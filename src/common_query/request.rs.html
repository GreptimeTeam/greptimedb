54/// Serialized predicate payload for remote dynamic filter updates.
55///
56/// The payload is tagged in JSON so receivers can reject unsupported encodings
57/// before decoding engine-specific bytes. For DataFusion expressions the
58/// `payload` bytes are serialized by `serde_json` as a base64 string, for example:
59///
60/// ```json
61/// { "kind": "datafusion", "payload": "CQgH" }
62/// ```
63#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
64#[non_exhaustive]
65#[serde(tag = "kind", content = "payload", rename_all = "snake_case")]
66pub enum DynFilterPayload {
67    /// A serialized DataFusion [`PhysicalExpr`] encoded as a protobuf
68    /// [`PhysicalExprNode`].
69    Datafusion(#[serde(with = "base64_serde::bytes")] Vec<u8>),
70}
71
72impl DynFilterPayload {
73    /// Encodes a DataFusion physical expression into a bounded dynamic filter payload.
74    ///
75    /// Runtime-only hash lookup predicates are degraded to `true` before encoding so
76    /// serializable min/max bounds around them can still be shipped to remote scans.
77    /// If the full serializable predicate is still larger than `max_payload_bytes`, large
78    /// membership predicates (`IN (...)`) are also degraded to `true` as a bounds-only fallback.
79    pub fn from_datafusion_expr(
80        expr: &Arc<dyn PhysicalExpr>,
81        max_payload_bytes: usize,
82    ) -> DataFusionResult<Self> {
83        match encode_remote_dyn_filter_expr(expr, max_payload_bytes, false) {
84            Ok(bytes) => Ok(Self::Datafusion(bytes)),
85            Err(CommonQueryError::DynFilterPayloadTooLarge { .. }) => {
86                encode_remote_dyn_filter_expr(expr, max_payload_bytes, true)
87                    .map(Self::Datafusion)
88                    .map_err(DataFusionError::from)
89            }
90            Err(error) => Err(DataFusionError::from(error)),
91        }
92    }
93
94    /// Decodes a DataFusion dynamic filter payload against the provided input schema.
95    ///
96    /// The decoded expression is validated to ensure column indexes stay within the receiver-side
97    /// schema, column names are consistent (defensive check), and the payload stays within
98    /// `max_payload_bytes`.
99    pub fn decode_datafusion_expr(
100        &self,
101        task_ctx: &TaskContext,
102        input_schema: &datafusion::arrow::datatypes::Schema,
103        max_payload_bytes: usize,
104    ) -> DataFusionResult<Arc<dyn PhysicalExpr>> {
105        let Self::Datafusion(bytes) = self;
106        validate_payload_size(bytes.len(), max_payload_bytes).map_err(DataFusionError::from)?;
107        let codec = DefaultPhysicalExtensionCodec {};
108        let proto = PhysicalExprNode::decode(bytes.as_slice()).map_err(|e| {
109            DataFusionError::Internal(format!("Failed to decode PhysicalExprNode: {e}"))
110        })?;
111
112        let expr = parse_physical_expr(&proto, task_ctx, input_schema, &codec)?;
113        validate_supported_payload_expr(&expr)?;
114        validate_decoded_payload_expr(&expr, input_schema)?;
115        Ok(expr)
116    }
117}

119fn encode_physical_expr_to_bytes(expr: &Arc<dyn PhysicalExpr>) -> DataFusionResult<Vec<u8>> {
120    let codec = DefaultPhysicalExtensionCodec {};
121    let proto = serialize_physical_expr(expr, &codec)?;
122    let mut bytes = Vec::new();
123    proto.encode(&mut bytes).map_err(|e| {
124        DataFusionError::Internal(format!("Failed to encode PhysicalExprNode: {e}"))
125    })?;
126    Ok(bytes)
127}

129struct BoundedPhysicalExprEncoder {
130    max_payload_bytes: usize,
131    encoded_bytes: Cell<usize>,
132}
133
134impl PhysicalProtoConverterExtension for BoundedPhysicalExprEncoder {
135    fn proto_to_execution_plan(
136        &self,
137        proto: &PhysicalPlanNode,
138        ctx: &PhysicalPlanDecodeContext<'_>,
139    ) -> DataFusionResult<Arc<dyn ExecutionPlan>> {
140        DefaultPhysicalProtoConverter {}.proto_to_execution_plan(proto, ctx)
141    }
142
143    fn execution_plan_to_proto(
144        &self,
145        plan: &Arc<dyn ExecutionPlan>,
146        codec: &dyn PhysicalExtensionCodec,
147    ) -> DataFusionResult<PhysicalPlanNode> {
148        DefaultPhysicalProtoConverter {}.execution_plan_to_proto(plan, codec)
149    }
150
151    fn proto_to_physical_expr(
152        &self,
153        proto: &PhysicalExprNode,
154        input_schema: &datafusion::arrow::datatypes::Schema,
155        ctx: &PhysicalPlanDecodeContext<'_>,
156    ) -> DataFusionResult<Arc<dyn PhysicalExpr>> {
157        DefaultPhysicalProtoConverter {}.proto_to_physical_expr(proto, input_schema, ctx)
158    }
159
160    fn physical_expr_to_proto(
161        &self,
162        expr: &Arc<dyn PhysicalExpr>,
163        codec: &dyn PhysicalExtensionCodec,
164    ) -> DataFusionResult<PhysicalExprNode> {
165        let before_children = self.encoded_bytes.get();
166        let proto = serialize_physical_expr_with_converter(expr, codec, self)?;
167        // Replace the children's contribution with the whole subtree, including protobuf framing.
168        // Checking each child stops large IN lists before materializing all their Arrow IPC literals.
169        let encoded_bytes = before_children.saturating_add(proto.encoded_len());
170        self.encoded_bytes.set(encoded_bytes);
171        validate_payload_size(encoded_bytes, self.max_payload_bytes)
172            .map_err(DataFusionError::from)?;
173        Ok(proto)
174    }
175}

177fn encode_bounded_physical_expr_to_bytes(
178    expr: &Arc<dyn PhysicalExpr>,
179    max_payload_bytes: usize,
180) -> Result<Vec<u8>, CommonQueryError> {
181    let codec = DefaultPhysicalExtensionCodec {};
182    let encoder = BoundedPhysicalExprEncoder {
183        max_payload_bytes,
184        encoded_bytes: Cell::new(0),
185    };
186    let proto = encoder.physical_expr_to_proto(expr, &codec);
187    // Preserve the size error so the caller can retry with bounds only.
188    validate_payload_size(encoder.encoded_bytes.get(), max_payload_bytes)?;
189    Ok(proto.map_err(CommonQueryError::from)?.encode_to_vec())
190}
191
192fn encode_remote_dyn_filter_expr(
193    expr: &Arc<dyn PhysicalExpr>,
194    max_payload_bytes: usize,
195    bounds_only: bool,
196) -> Result<Vec<u8>, CommonQueryError> {
197    let expr = portable_remote_dyn_filter_expr(Arc::clone(expr), bounds_only)
198        .map_err(CommonQueryError::from)?;
199    encode_bounded_physical_expr_to_bytes(&expr, max_payload_bytes)
200}
201
202fn portable_remote_dyn_filter_expr(
203    expr: Arc<dyn PhysicalExpr>,
204    bounds_only: bool,
205) -> DataFusionResult<Arc<dyn PhysicalExpr>> {
206    expr.transform_up(|node| {
207        if node.is::<HashTableLookupExpr>() || (bounds_only && node.is::<InListExpr>()) {
208            Ok(Transformed::yes(lit(true)))
209        } else {
210            Ok(Transformed::no(node))
211        }
212    })
213    .map(|transformed| transformed.data)
214}

216pub(crate) fn decode_physical_expr_from_bytes(
217    bytes: &[u8],
218    task_ctx: &TaskContext,
219    input_schema: &datafusion::arrow::datatypes::Schema,
220    max_payload_bytes: usize,
221) -> DataFusionResult<Arc<dyn PhysicalExpr>> {
222    validate_payload_size(bytes.len(), max_payload_bytes).map_err(DataFusionError::from)?;
223    let codec = DefaultPhysicalExtensionCodec {};
224    let proto = PhysicalExprNode::decode(bytes).map_err(|e| {
225        DataFusionError::Internal(format!("Failed to decode PhysicalExprNode: {e}"))
226    })?;
227
228    let expr = parse_physical_expr(&proto, task_ctx, input_schema, &codec)?;
229    validate_supported_payload_expr(&expr)?;
230    validate_decoded_payload_expr(&expr, input_schema)?;
231    Ok(expr)
232}
233
234fn validate_payload_size(
235    payload_size_bytes: usize,
236    max_payload_bytes: usize,
237) -> Result<(), CommonQueryError> {
238    ensure!(
239        payload_size_bytes <= max_payload_bytes,
240        DynFilterPayloadTooLargeSnafu {
241            payload_size_bytes,
242            max_payload_bytes,
243        }
244    );
245
246    Ok(())
247}
248
249fn validate_supported_payload_expr(expr: &Arc<dyn PhysicalExpr>) -> DataFusionResult<()> {
250    expr.apply(|node| {
251        if node.is::<HashTableLookupExpr>() {
252            return Err(DataFusionError::Plan(
253                "HashTableLookupExpr cannot be encoded into DynFilterPayload::Datafusion"
254                    .to_string(),
255            ));
256        }
257
258        Ok(TreeNodeRecursion::Continue)
259    })?;
260
261    Ok(())
262}

264/// Validates decoded dynamic filter physical expressions against the receiver-side schema.
265///
266/// Rejects out-of-bounds column indexes and, as a defensive check, rejects columns whose
267/// name disagrees with the corresponding schema field. DataFusion physical `Column` is
268/// index-authoritative, so a name mismatch usually indicates a coordinator/receiver
269/// schema inconsistency that should be surfaced loudly.
270fn validate_decoded_payload_expr(
271    expr: &Arc<dyn PhysicalExpr>,
272    input_schema: &datafusion::arrow::datatypes::Schema,
273) -> DataFusionResult<()> {
274    expr.apply(|node| {
275        if let Some(column) = node.downcast_ref::<Column>() {
276            let Some(field) = input_schema.fields().get(column.index()) else {
277                return Err(DataFusionError::Plan(format!(
278                    "Decoded Column '{}' references out-of-bounds index {} for input schema of size {}",
279                    column.name(),
280                    column.index(),
281                    input_schema.fields().len()
282                )));
283            };
284
285            if field.name() != column.name() {
286                return Err(DataFusionError::Plan(format!(
287                    "Decoded Column name/index mismatch: payload has '{}' at index {}, but schema field is '{}'",
288                    column.name(),
289                    column.index(),
290                    field.name()
291                )));
292            }
293        }
294
295        Ok(TreeNodeRecursion::Continue)
296    })?;
297
298    Ok(())
299}

301/// A remote dynamic filter update sent from a query coordinator to region servers.
302///
303/// `generation` is monotonic within a `query_id`/`filter_id` pair and matches the
304/// gRPC field name used by `RemoteDynFilterUpdate`. Receivers use it to ignore
305/// stale updates while `is_complete` marks the final payload for the filter.
306#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
307pub struct DynFilterUpdate {
308    /// Protocol version used by this update payload.
309    pub protocol_version: u32,
310    /// Internal query identifier that owns this dynamic filter lifecycle.
311    pub query_id: String,
312    /// Identifier of the dynamic filter within the query.
313    pub filter_id: String,
314    /// Monotonic update generation for this filter.
315    pub generation: u64,
316    /// Whether this update completes the dynamic filter stream.
317    pub is_complete: bool,
318    /// Serialized predicate payload carried by this update.
319    pub payload: DynFilterPayload,
320}
321
322impl DynFilterUpdate {
323    /// Creates a dynamic filter update with the current protocol version.
324    pub fn new(
325        query_id: String,
326        filter_id: String,
327        generation: u64,
328        is_complete: bool,
329        payload: DynFilterPayload,
330    ) -> Self {
331        Self {
332            protocol_version: DYN_FILTER_PROTOCOL_VERSION,
333            query_id,
334            filter_id,
335            generation,
336            is_complete,
337            payload,
338        }
339    }
340}

664        let mut found = false;
665        expr.apply(|node| {
666            if node.is::<T>() {
667                found = true;
668                Ok(TreeNodeRecursion::Stop)
669            } else {
670                Ok(TreeNodeRecursion::Continue)
671            }
672        })
673        .unwrap();
