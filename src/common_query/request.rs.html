61/// Smallest number of encoded bytes a single `IN` list literal can take.
62///
63/// Every list entry is a `PhysicalExprNode` whose `literal` arm holds the
64/// [`ScalarValue`] message itself, so the oneof field tag plus its length prefix are
65/// already 2 bytes even for an empty `Utf8` value (whose `ScalarValue` body may encode to
66/// nothing at all). Protobuf sizes add up across nesting, so a payload of `B` bytes can
67/// never hold more than `B / 2` literals. This is the floor the value bound is derived
68/// from; it is not an encoder policy constant.
69const MIN_ENCODED_LITERAL_BYTES: usize = 2;
70
71/// Upper bound on how many `IN` list literals `max_payload_bytes` can hold.
72///
73/// The count bound is derived from the transport budget with
74/// [`MIN_ENCODED_LITERAL_BYTES`] rather than being an arbitrary cap: a larger budget
75/// admits proportionally more build keys, and a small budget refuses the membership
76/// before a single literal is built.
77fn max_membership_values(max_payload_bytes: usize) -> usize {
78    max_payload_bytes / MIN_ENCODED_LITERAL_BYTES
79}
80
81/// Frame size reported for a payload refused because it does not fit the budget.
82///
83/// A refused payload is never materialized, so there is no measured frame length. The
84/// error reports the largest lower bound that is known instead: the price of the literals
85/// the refused membership would have written (`lower_bound`), or - when nothing about the
86/// shape is known - one byte over the budget. Either way the number is genuinely above
87/// `max_payload_bytes`; it is a bound, never a measurement, and it is only used for
88/// diagnostics.
89fn refused_payload_size(lower_bound: usize, max_payload_bytes: usize) -> usize {
90    lower_bound.max(max_payload_bytes.saturating_add(1))
91}

101/// ```

103#[non_exhaustive]

105pub enum DynFilterPayload {
106    /// A serialized DataFusion [`PhysicalExpr`] encoded as a protobuf
107    /// [`PhysicalExprNode`].
108    Datafusion(#[serde(with = "base64_serde::bytes")] Vec<u8>),
109}

111impl DynFilterPayload {
112    /// Encodes a DataFusion physical expression into a bounded dynamic filter payload.
113    ///
114    /// # Narrow contract: a key list used as a positive pruning hint
115    ///
116    /// A runtime [`HashTableLookupExpr`] is written as the existing
117    /// `PhysicalInListNode` — `key IN (build key, ...)` — built from the lookup's
118    /// published build-key list, so no new wire variant is needed. The expression tree
119    /// around it is preserved node for node: `IS NULL`, `NOT`, `CASE`, `AND`, `OR` and
120    /// every other parent are serialized as-is, and a `NOT` over a lookup therefore
121    /// becomes `NOT IN`.
122    ///
123    /// The membership is a **hint for pruning only**, not a re-statement of the join
124    /// semantics. For supported positive pruning, the list preserves actual build-key
125    /// matches; it is not universal Boolean equivalence. `IN`/`NOT IN` propagate
126    /// `NULL` while a hash lookup can be definite on a `NULL` probe key, and hash
127    /// presence can admit collisions, so a predicate that inspects the *negation* of a
128    /// lookup is not universally equivalent.
129    ///
130    /// The conversion is decided by the lookup alone (single-column `Utf8`, non-empty,
131    /// `NULL`-free build keys — see [`HashTableLookupExpr::build_key_membership`]).
132    /// Exactly one node changes: every [`HashTableLookupExpr`] is written as its key
133    /// list, wherever it sits in the tree (`NOT lookup` included, which is why it
134    /// becomes `NOT IN`). No node is ever replaced by a *weaker stand-in* — in
135    /// particular never by `true` — and no membership is dropped from inside a
136    /// predicate: when a lookup cannot be written at all, the whole top-level `AND`
137    /// conjunct that holds it is dropped (see below) or the payload is rejected.
138    ///
139    /// # Top-level conjunctive retention
140    ///
141    /// If a lookup has no such key list (accessor limitation) or the serialized predicate
142    /// does not fit `max_payload_bytes`, [`split_conjunction`] splits the original
143    /// expression and the whole conjuncts that cannot be shipped are dropped. Dropping an
144    /// `AND` term can only widen a predicate, so the result never drops a row the producer
145    /// would keep. Which conjuncts go depends on why the payload was refused:
146    ///
147    /// - a lookup has no wire form: only conjuncts holding a [`HashTableLookupExpr`] are
148    ///   dropped. A static [`InListExpr`] is representable, so `lookup AND id IN (...)`
149    ///   keeps the list and stays `id IN (...)`;
150    /// - the payload is over budget: conjuncts holding a [`HashTableLookupExpr`] *or* an
151    ///   [`InListExpr`] are dropped, because a static `IN` list is the node that usually
152    ///   blows the budget. `bounds AND oversized IN (...)` becomes `bounds`.
153    ///
154    /// Either way a conjunct is kept or dropped as a whole: a membership nested inside
155    /// `NOT`, `CASE` or `OR` drops that entire conjunct instead of being rewritten.
156    /// Retained conjuncts are encoded again under the same budget, and when the first two
157    /// rules still leave an over-budget payload the second one is applied as well, so a
158    /// static `IN` list that is too large for the budget is dropped last.
159    ///
160    /// # Encoder-level no-substitution rule
161    ///
162    /// This encoder never replaces a node in the producer expression with `true`: it
163    /// serializes the expression, retries after dropping whole top-level conjuncts under
164    /// the retention rules above, or returns an error. A caller decides what a refused
165    /// generation means for its receiver. Initial capture keeps the parent filter local;
166    /// the long-lived query fanout path
167    /// (`remote_dyn_filter_registry::fanout_snapshot_for_query`) publishes an explicit
168    /// `true` reset to invalidate a stale region-side predicate. That reset is not a
169    /// representation of the refused producer expression.
170    ///
171    /// The two refusals stay distinct so callers can tell a shape problem from a size
172    /// problem:
173    ///
174    /// - [`CommonQueryError::DynFilterUnsupported`]: no conjunct can be retained because
175    ///   every one of them holds a lookup with no UTF-8 key list (`NOT lookup`, `CASE` /
176    ///   `OR` over such a membership). The wire has no form for the predicate.
177    /// - [`CommonQueryError::DynFilterPayloadTooLarge`]: the predicate is representable but
178    ///   does not fit `max_payload_bytes` - either a key list itself exceeds the budget, or
179    ///   the retained conjuncts still do not fit. An over-budget key list is never reported
180    ///   as unsupported; it is a size refusal and uses
181    ///   [`ConjunctRetention::OverBudget`].
182    ///
183    /// These refusals do not fail the query in either caller: initial capture leaves the
184    /// parent filter local, while fanout sends an explicit `true` reset instead of leaving
185    /// a stale narrower predicate installed.
186    pub fn from_datafusion_expr(
187        expr: &Arc<dyn PhysicalExpr>,
188        max_payload_bytes: usize,
189    ) -> DataFusionResult<Self> {
190        match encode_bounded_physical_expr_to_bytes(expr, max_payload_bytes) {
191            Ok(bytes) => Ok(Self::Datafusion(bytes)),
192            // The predicate cannot be shipped as it stands: either a lookup has no key
193            // list, or the frame does not fit the budget. Keep the widest *safe* part of
194            // the predicate instead of publishing an always-true filter.
195            Err(CommonQueryError::DynFilterUnsupported { .. }) => retain_membership_free_conjuncts(
196                expr,
197                max_payload_bytes,
198                ConjunctRetention::UnrepresentableLookup,
199            ),
200            Err(CommonQueryError::DynFilterPayloadTooLarge { .. }) => {
201                retain_membership_free_conjuncts(
202                    expr,
203                    max_payload_bytes,
204                    ConjunctRetention::OverBudget,
205                )
206            }
207            Err(error) => Err(DataFusionError::from(error)),
208        }
209    }

211    /// Decodes a DataFusion dynamic filter payload against the provided input schema.
212    ///
213    /// The decoded expression is validated to ensure column indexes stay within the receiver-side
214    /// schema, column names are consistent (defensive check), and the payload stays within
215    /// `max_payload_bytes`.
216    ///
217    /// # Precondition: pruning only, never row filtering
218    ///
219    /// The decoded predicate is only a **pruning hint**. The sole production consumer —
220    /// `Predicate::dyn_filters` (in the `table` crate) and its pruning path, reached by
221    /// mito2's row-group pruner — uses it to skip containers whose statistics prove that
222    /// no row can match, so a predicate that is merely *wider* than the producer's costs
223    /// nothing but a wasted scan.
224    ///
225    /// The payload is not a re-statement of the producer predicate: the membership is the
226    /// join's build-key list, so `IN`/`NOT IN` `NULL` propagation and hash collisions can
227    /// make it disagree with a runtime lookup, and the encoder drops whole top-level
228    /// conjuncts it cannot ship (see [`DynFilterPayload::from_datafusion_expr`]). A
229    /// consumer that *filters rows* would turn every such widening into lost data, so row
230    /// filtering must not consume this payload unconditionally: it would need to prove the
231    /// predicate is equivalent for the rows it drops, which this format does not carry.
232    pub fn decode_datafusion_expr(
233        &self,
234        task_ctx: &TaskContext,
235        input_schema: &datafusion::arrow::datatypes::Schema,
236        max_payload_bytes: usize,
237    ) -> DataFusionResult<Arc<dyn PhysicalExpr>> {
238        let Self::Datafusion(bytes) = self;
239        validate_payload_size(bytes.len(), max_payload_bytes).map_err(DataFusionError::from)?;
240        let codec = DefaultPhysicalExtensionCodec {};
241        let proto = PhysicalExprNode::decode(bytes.as_slice()).map_err(|e| {
242            DataFusionError::Internal(format!("Failed to decode PhysicalExprNode: {e}"))
243        })?;
244
245        let expr = parse_physical_expr(&proto, task_ctx, input_schema, &codec)?;
246        validate_supported_payload_expr(&expr)?;
247        validate_decoded_payload_expr(&expr, input_schema)?;
248        Ok(expr)
249    }
250}

252fn encode_physical_expr_to_bytes(expr: &Arc<dyn PhysicalExpr>) -> DataFusionResult<Vec<u8>> {
253    let codec = DefaultPhysicalExtensionCodec {};
254    let proto = serialize_physical_expr(expr, &codec)?;
255    let mut bytes = Vec::new();
256    proto.encode(&mut bytes).map_err(|e| {
257        DataFusionError::Internal(format!("Failed to encode PhysicalExprNode: {e}"))
258    })?;
259    Ok(bytes)
260}

262struct BoundedPhysicalExprEncoder {
263    max_payload_bytes: usize,
264    encoded_bytes: Cell<usize>,
265    /// Set when the encoding aborted on a runtime lookup that has no serializable key
266    /// list.
267    ///
268    /// The abort travels out as an ordinary [`DataFusionError`], which would lose the
269    /// typed dynamic filter error on the way, so the flag is what lets
270    /// `encode_bounded_physical_expr_to_bytes` rebuild
271    /// [`CommonQueryError::DynFilterUnsupported`]. No placeholder is ever encoded in its
272    /// place.
273    unsupported: Cell<bool>,
274    /// Set to the reported frame size when the encoding aborted because a build-key
275    /// membership does not fit `max_payload_bytes`.
276    ///
277    /// This is a size refusal, not a shape one, so
278    /// `encode_bounded_physical_expr_to_bytes` rebuilds
279    /// [`CommonQueryError::DynFilterPayloadTooLarge`] from it instead of reporting the
280    /// predicate as having no wire form.
281    over_budget: Cell<Option<usize>>,
282}
283
284impl BoundedPhysicalExprEncoder {
285    fn new(max_payload_bytes: usize) -> Self {
286        Self {
287            max_payload_bytes,
288            encoded_bytes: Cell::new(0),
289            unsupported: Cell::new(false),
290            over_budget: Cell::new(None),
291        }
292    }
293
294    /// Charges `proto` (including its encoded children) against the payload budget.
295    ///
296    /// The accounting is additive over siblings and exact for the top-level node, so
297    /// no branch can push the payload over budget unnoticed.
298    fn finish(
299        &self,
300        before_children: usize,
301        proto: PhysicalExprNode,
302    ) -> DataFusionResult<PhysicalExprNode> {
303        let encoded_bytes = before_children.saturating_add(proto.encoded_len());
304        self.encoded_bytes.set(encoded_bytes);
305        validate_payload_size(encoded_bytes, self.max_payload_bytes)
306            .map_err(DataFusionError::from)?;
307        Ok(proto)
308    }
309
310    /// Encodes a lookup as its build-key membership, charging every list literal and
311    /// node against the payload budget.
312    ///
313    /// The decision only looks at the lookup itself: if DataFusion can hand out the
314    /// single-column `Utf8` build-key list it is written as `key IN (...)`; otherwise the
315    /// whole payload aborts. A lookup is a real predicate that DataFusion only degrades to
316    /// `true` in its own proto encoder, so this transport must not write any substitute for
317    /// it, and the abort keeps the two reasons apart: no key list at all (unsupported) is
318    /// not the same thing as a key list that exceeds the budget (too large).
319    fn membership_to_proto(
320        &self,
321        lookup: &HashTableLookupExpr,
322        codec: &dyn PhysicalExtensionCodec,
323    ) -> DataFusionResult<PhysicalExprNode> {
324        let membership = match RemoteMembership::from_lookup(lookup, self.max_payload_bytes) {
325            Ok(membership) => membership,
326            // A lookup without a key list has no wire form at all: that is a shape
327            // refusal, not a size one.
328            Err(MembershipRefusal::NoKeyList) => return Err(self.abort_unsupported_lookup()),
329            // The key list exists but does not fit the budget: report it as a size
330            // refusal so it is not counted as "no wire form".
331            Err(MembershipRefusal::OverBudget { build_keys }) => {
332                return Err(self.abort_over_budget_membership(build_keys));
333            }
334        };
335        // The count bound is enforced once more before any literal exists, so a build side
336        // larger than the budget can hold is never materialized. This cannot fail for any
337        // other reason here: the accessor already guaranteed a non-empty, `NULL`-free
338        // `Utf8` list, so a refusal is again a size refusal.
339        let Some(literals) = utf8_list_literals(
340            &membership.values,
341            max_membership_values(self.max_payload_bytes),
342        ) else {
343            return Err(self.abort_over_budget_membership(membership.values.len()));
344        };
345
346        let before_children = self.encoded_bytes.get();
347        let mut list = Vec::with_capacity(literals.len());
348        for literal in literals {
349            list.push(self.physical_expr_to_proto(&literal, codec)?);
350        }
351
352        // The `InList` proto is built by hand instead of going through the generic
353        // serializer: that path needs an `InListExpr`, whose only constructors
354        // (`InListExpr::try_new` / `try_new_from_array`) require an input `Schema` to
355        // resolve the key expression's data type, and this encoder converts a bare
356        // expression without one. The key is known to be `Utf8` (see
357        // `HashTableLookupExpr::build_key_membership`), which is exactly what the list
358        // below writes, so no type check is lost.
359        let proto = PhysicalExprNode {
360            expr_id: None,
361            expr_type: Some(PhysicalExprNodeType::InList(Box::new(PhysicalInListNode {
362                expr: Some(Box::new(
363                    self.physical_expr_to_proto(&membership.key, codec)?,
364                )),
365                list,
366                negated: false,
367            }))),
368        };
369
370        self.finish(before_children, proto)
371    }
372
373    /// Aborts the payload because the wire has no form for `lookup`.
374    ///
375    /// The returned error only carries the abort; [`Self::unsupported`] is what the caller
376    /// reads to rebuild the typed dynamic filter error.
377    fn abort_unsupported_lookup(&self) -> DataFusionError {
378        self.unsupported.set(true);
379        DataFusionError::NotImplemented(
380            "hash table lookup has no UTF-8 build-key membership to encode".to_string(),
381        )
382    }
383
384    /// Aborts the payload because a build-key membership does not fit `max_payload_bytes`.
385    ///
386    /// This is a *size* refusal, never a "no wire form" one, so the caller rebuilds
387    /// [`CommonQueryError::DynFilterPayloadTooLarge`] from [`Self::over_budget`] instead of
388    /// [`CommonQueryError::DynFilterUnsupported`]. `build_keys` is the number of build keys
389    /// the refused list would have written; every one of them costs at least
390    /// [`MIN_ENCODED_LITERAL_BYTES`], so that product is a real lower bound of the frame.
391    fn abort_over_budget_membership(&self, build_keys: usize) -> DataFusionError {
392        self.over_budget.set(Some(refused_payload_size(
393            build_keys.saturating_mul(MIN_ENCODED_LITERAL_BYTES),
394            self.max_payload_bytes,
395        )));
396        DataFusionError::NotImplemented(
397            "hash table lookup build-key membership exceeds the payload budget".to_string(),
398        )
399    }
400}

443        let proto = serialize_physical_expr_with_converter(expr, codec, self)?;
444        // Replace the children's contribution with the whole subtree, including protobuf framing.
445        // Checking each child stops large IN lists before materializing all their Arrow IPC literals.
446        self.finish(before_children, proto)
447    }

450/// Encodes `expr` for the remote wire under `max_payload_bytes`.
451///
452/// Returns the typed [`CommonQueryError::DynFilterUnsupported`] when the predicate holds a
453/// lookup whose key list cannot be serialized, and
454/// [`CommonQueryError::DynFilterPayloadTooLarge`] when the frame (or a build-key
455/// membership) does not fit the budget. The two refusals stay distinct: an over-budget
456/// membership is a size refusal and is never reported as a predicate without a wire form.
457/// The caller answers both with [`retain_membership_free_conjuncts`]; neither is ever
458/// replaced with `true`.
459fn encode_bounded_physical_expr_to_bytes(
460    expr: &Arc<dyn PhysicalExpr>,
461    max_payload_bytes: usize,
462) -> Result<Vec<u8>, CommonQueryError> {
463    let codec = DefaultPhysicalExtensionCodec {};
464    let encoder = BoundedPhysicalExprEncoder::new(max_payload_bytes);
465    let proto = encoder.physical_expr_to_proto(expr, &codec);
466    // A build-key membership that exceeds the budget is a size refusal even though the
467    // frame that would have held it is never materialized.
468    if let Some(payload_size_bytes) = encoder.over_budget.get() {
469        return Err(DynFilterPayloadTooLargeSnafu {
470            payload_size_bytes,
471            max_payload_bytes,
472        }
473        .build());
474    }
475    let unsupported = encoder.unsupported.get();
476    // The encoder stops at the node that went over budget, so this reports the size
477    // error unwrapped instead of the wrapped DataFusion error.
478    validate_payload_size(encoder.encoded_bytes.get(), max_payload_bytes)?;
479    if unsupported {
480        return Err(DynFilterUnsupportedSnafu {
481            reason: "the predicate holds a hash table lookup without a UTF-8 build-key \
482                     membership",
483        }
484        .build());
485    }
486    Ok(proto.map_err(CommonQueryError::from)?.encode_to_vec())
487}

489/// Which top-level conjuncts [`retain_membership_free_conjuncts`] has to drop.
490#[derive(Clone, Copy, Debug)]
491enum ConjunctRetention {
492    /// The payload holds a [`HashTableLookupExpr`] without a wire form, so such a lookup
493    /// is the only unshippable node: a static [`InListExpr`] is representable and stays.
494    UnrepresentableLookup,
495    /// The payload exceeds its budget. A static [`InListExpr`] is the node most likely to
496    /// be the oversized one, so it is dropped with the lookups.
497    OverBudget,
498}
499
500/// Keeps the top-level conjuncts of a predicate that cannot be shipped as it stands.
501///
502/// Dropping an `AND` term can only widen a predicate, so the retained payload still never
503/// drops a row the producer would keep. `retention` decides which conjuncts have to go. It
504/// only widens as far as it has to: when the narrow retention leaves conjuncts that still do
505/// not fit `max_payload_bytes`, the [`InListExpr`]-bearing ones are dropped as well, because
506/// a static `IN` list is the node most likely to be the oversized one.
507///
508/// The refusal keeps the two reasons apart instead of folding them into "no wire form":
509/// nothing retained under [`ConjunctRetention::UnrepresentableLookup`] means the wire
510/// really has no form for the predicate
511/// ([`CommonQueryError::DynFilterUnsupported`]), while nothing retained under
512/// [`ConjunctRetention::OverBudget`] (or a retained payload that still does not fit) is a
513/// size refusal ([`CommonQueryError::DynFilterPayloadTooLarge`]). Transport never
514/// substitutes `true` for either.
515fn retain_membership_free_conjuncts(
516    expr: &Arc<dyn PhysicalExpr>,
517    max_payload_bytes: usize,
518    retention: ConjunctRetention,
519) -> DataFusionResult<DynFilterPayload> {
520    match encode_retained_conjuncts_to_bytes(expr, max_payload_bytes, retention) {
521        Ok(Some(bytes)) => Ok(DynFilterPayload::Datafusion(bytes)),
522        // Nothing could be retained. Which refusal that is depends on *why* the conjuncts
523        // had to go, and the two must not be mixed up: an over-budget predicate is
524        // representable, it only does not fit.
525        Ok(None) => Err(match retention {
526            ConjunctRetention::UnrepresentableLookup => DynFilterUnsupportedSnafu {
527                reason: "every top-level conjunct of the predicate holds a hash table \
528                         lookup without a UTF-8 build-key membership, so the wire has no \
529                         form for it",
530            }
531            .build()
532            .into(),
533            ConjunctRetention::OverBudget => DynFilterPayloadTooLargeSnafu {
534                payload_size_bytes: refused_payload_size(0, max_payload_bytes),
535                max_payload_bytes,
536            }
537            .build()
538            .into(),
539        }),
540        Err(CommonQueryError::DynFilterPayloadTooLarge { .. })
541            if matches!(retention, ConjunctRetention::UnrepresentableLookup) =>
542        {
543            retain_membership_free_conjuncts(expr, max_payload_bytes, ConjunctRetention::OverBudget)
544        }
545        Err(error) => Err(error.into()),
546    }
547}

549/// Encodes the top-level conjuncts of `expr` that `retention` allows to keep.
550///
551/// [`split_conjunction`] hands out the top-level `AND` terms of the original
552/// expression; a term is dropped as a whole when its subtree holds a node the retention
553/// cannot ship (see [`ConjunctRetention`]). Membership nested inside `NOT`, `CASE` or `OR`
554/// therefore drops that whole conjunct instead of being rewritten in place, so the payload
555/// only ever loses top-level `AND` terms, which can only widen a predicate.
556///
557/// The retained terms are encoded again under the same `max_payload_bytes`, so a caller
558/// supplied budget still bounds the frame. Returns `Ok(None)` when no conjunct can be
559/// retained; the caller decides which typed refusal that is from `retention`, because an
560/// empty retention caused by an unrepresentable lookup is not the same thing as an empty
561/// retention caused by the budget.
562fn encode_retained_conjuncts_to_bytes(
563    expr: &Arc<dyn PhysicalExpr>,
564    max_payload_bytes: usize,
565    retention: ConjunctRetention,
566) -> Result<Option<Vec<u8>>, CommonQueryError> {
567    let mut retained = Vec::new();
568    for conjunct in split_conjunction(expr) {
569        if !retention.drops_conjunct(conjunct)? {
570            retained.push(Arc::clone(conjunct));
571        }
572    }
573    if retained.is_empty() {
574        return Ok(None);
575    }
576    encode_bounded_physical_expr_to_bytes(&conjunction(retained), max_payload_bytes).map(Some)
577}
578
579impl ConjunctRetention {
580    /// Whether the whole subtree of `expr` holds a node this retention cannot ship.
581    ///
582    /// This is a "whole conjunct" test on purpose: a conjunct that reaches such a node
583    /// anywhere is dropped by the retention instead of being rewritten around it.
584    fn drops_conjunct(self, expr: &Arc<dyn PhysicalExpr>) -> DataFusionResult<bool> {
585        let mut found = false;
586        expr.apply(|node| {
587            let unshippable = match self {
588                Self::UnrepresentableLookup => node.is::<HashTableLookupExpr>(),
589                Self::OverBudget => node.is::<HashTableLookupExpr>() || node.is::<InListExpr>(),
590            };
591            if unshippable {
592                found = true;
593                Ok(TreeNodeRecursion::Stop)
594            } else {
595                Ok(TreeNodeRecursion::Continue)
596            }
597        })?;
598        Ok(found)
599    }
600}

602/// Build-key list of one hash lookup, written as `probe_key IN (value, ...)`.
603///
604/// The membership reuses the existing `PhysicalInListNode`, so no new wire variant
605/// and no in-payload version tag are needed, and it is charged against the normal
606/// payload budget. Only the build-key list itself is consulted here; the wire form
607/// never depends on where the lookup sits in the predicate tree.
608struct RemoteMembership {
609    /// Probe-side key expression, reused as the `InList` operand.
610    key: Arc<dyn PhysicalExpr>,
611    /// Published build key values, tied to the lookup's build lease.
612    values: ArrayRef,
613}
614
615impl RemoteMembership {
616    /// Key-list membership of `lookup`, or the reason it cannot be written.
617    ///
618    /// The count bound handed to DataFusion is derived from the byte budget (see
619    /// [`max_membership_values`]) instead of being a policy constant, and the byte bound
620    /// is the raw payload budget itself, which every build-key buffer has to fit before it
621    /// can be written as literals. Both are conservative early guards, not the exact
622    /// serialized frame length: the encoder still charges every literal against the budget
623    /// node by node.
624    ///
625    /// The accessor keeps the value-level limitations — it returns `None` for lookups
626    /// without a build lease (e.g. a test-only lookup), composite keys, non-`Utf8` keys
627    /// (integer/float/decimal/dictionary/unknown), and `NULL`-containing or empty build
628    /// keys — and [`utf8_list_literals`] re-checks everything it materializes. Those are
629    /// accessor and materialization limitations, not encoder policy.
630    ///
631    /// The accessor folds its shape and size checks into one `None`, so a key list that
632    /// exists but exceeds the bounds is told apart by asking again with the bounds lifted:
633    /// only a list that exists at all can be refused for its size, and a size refusal must
634    /// not be reported as a predicate without a wire form (it is
635    /// [`CommonQueryError::DynFilterPayloadTooLarge`], not
636    /// [`CommonQueryError::DynFilterUnsupported`]).
637    fn from_lookup(
638        lookup: &HashTableLookupExpr,
639        max_payload_bytes: usize,
640    ) -> Result<Self, MembershipRefusal> {
641        match lookup
642            .build_key_membership(max_membership_values(max_payload_bytes), max_payload_bytes)
643        {
644            Some(membership) => Ok(Self {
645                key: Arc::clone(membership.key()),
646                values: Arc::clone(membership.values()),
647            }),
648            None => Err(match lookup.build_key_membership(usize::MAX, usize::MAX) {
649                Some(unbounded) => MembershipRefusal::OverBudget {
650                    build_keys: unbounded.values().len(),
651                },
652                None => MembershipRefusal::NoKeyList,
653            }),
654        }
655    }
656}
657
658/// Why a lookup's build-key membership cannot be written as `probe_key IN (...)`.
659///
660/// The two reasons are answered differently: [`Self::NoKeyList`] means the wire has no
661/// form for the node, while [`Self::OverBudget`] means the membership is real but too
662/// large for the payload budget, which the encoder reports as a size refusal.
663#[derive(Clone, Copy, Debug, PartialEq, Eq)]
664enum MembershipRefusal {
665    /// The lookup has no single-column `Utf8` build-key list at all: no build lease,
666    /// composite key, unsupported key type, or an empty / `NULL`-containing build side.
667    NoKeyList,
668    /// The build-key list exists but exceeds the caller's bounds. `build_keys` is how many
669    /// literals it would have written, measured from the unbounded accessor.
670    OverBudget { build_keys: usize },
671}
672
673/// Materializes the `IN` list literals for a build-key membership.
674///
675/// This is the authoritative value check: the list is only built for a non-empty,
676/// `NULL`-free, non-dictionary `Utf8` array of at most `max_values` entries. `max_values`
677/// is checked *before* a [`Literal`] is created for every build key, so an over-large
678/// build side is refused instead of being materialized and then measured. The caller has
679/// already been through [`RemoteMembership::from_lookup`], so a refusal here can only be
680/// the count bound and is reported as a size refusal, not as a missing wire form.
681fn utf8_list_literals(values: &ArrayRef, max_values: usize) -> Option<Vec<Arc<dyn PhysicalExpr>>> {
682    let values = values.as_any().downcast_ref::<StringArray>()?;
683    if values.is_empty() || values.null_count() != 0 || values.len() > max_values {
684        return None;
685    }
686    values
687        .iter()
688        .map(|value| {
689            value.map(|value| {
690                Arc::new(Literal::new(ScalarValue::Utf8(Some(value.to_owned()))))
691                    as Arc<dyn PhysicalExpr>
692            })
693        })
694        .collect()
695}

697pub(crate) fn decode_physical_expr_from_bytes(
698    bytes: &[u8],
699    task_ctx: &TaskContext,
700    input_schema: &datafusion::arrow::datatypes::Schema,
701    max_payload_bytes: usize,
702) -> DataFusionResult<Arc<dyn PhysicalExpr>> {
703    validate_payload_size(bytes.len(), max_payload_bytes).map_err(DataFusionError::from)?;
704    let codec = DefaultPhysicalExtensionCodec {};
705    let proto = PhysicalExprNode::decode(bytes).map_err(|e| {
706        DataFusionError::Internal(format!("Failed to decode PhysicalExprNode: {e}"))
707    })?;
708
709    let expr = parse_physical_expr(&proto, task_ctx, input_schema, &codec)?;
710    validate_supported_payload_expr(&expr)?;
711    validate_decoded_payload_expr(&expr, input_schema)?;
712    Ok(expr)
713}

761/// Whether `error` reports a dynamic filter predicate the wire cannot represent.
762///
763/// Transport never rewrites a producer predicate to `true`, so a producer answers this
764/// error by not shipping the producer predicate and keeping the query running. Callers
765/// that keep a long-lived region-side predicate treat it as an invalidation trigger: the
766/// stale predicate is replaced by an explicit `true` reset (see
767/// `remote_dyn_filter_registry`), so a producer that later widens its filter cannot be
768/// pruned away by it.
769pub fn is_dyn_filter_unsupported(error: &DataFusionError) -> bool {
770    matches!(
771        error,
772        DataFusionError::External(source)
773            if source
774                .downcast_ref::<CommonQueryError>()
775                .is_some_and(CommonQueryError::is_dyn_filter_unsupported)
776    )
777}

779/// Whether `error` reports a dynamic filter payload that exceeded `max_payload_bytes`.
780///
781/// This is the other refusal [`DynFilterPayload::from_datafusion_expr`] reports, next to
782/// [`is_dyn_filter_unsupported`]: the predicate exists and is representable, but even the

785/// shipping the producer predicate (the query keeps running) and never by narrowing or

790    matches!(
791        error,
792        DataFusionError::External(source)
793            if source
794                .downcast_ref::<CommonQueryError>()
795                .is_some_and(|error| matches!(
796                    error,
797                    CommonQueryError::DynFilterPayloadTooLarge { .. }
798                ))
799    )
800}

817/// Validates decoded dynamic filter physical expressions against the receiver-side schema.
818///
819/// Rejects out-of-bounds column indexes and, as a defensive check, rejects columns whose
820/// name disagrees with the corresponding schema field. DataFusion physical `Column` is
821/// index-authoritative, so a name mismatch usually indicates a coordinator/receiver
822/// schema inconsistency that should be surfaced loudly.
823fn validate_decoded_payload_expr(
824    expr: &Arc<dyn PhysicalExpr>,
825    input_schema: &datafusion::arrow::datatypes::Schema,
826) -> DataFusionResult<()> {
827    expr.apply(|node| {
828        if let Some(column) = node.downcast_ref::<Column>() {
829            let Some(field) = input_schema.fields().get(column.index()) else {
830                return Err(DataFusionError::Plan(format!(
831                    "Decoded Column '{}' references out-of-bounds index {} for input schema of size {}",
832                    column.name(),
833                    column.index(),
834                    input_schema.fields().len()
835                )));
836            };
837
838            if field.name() != column.name() {
839                return Err(DataFusionError::Plan(format!(
840                    "Decoded Column name/index mismatch: payload has '{}' at index {}, but schema field is '{}'",
841                    column.name(),
842                    column.index(),
843                    field.name()
844                )));
845            }
846        }
847
848        Ok(TreeNodeRecursion::Continue)
849    })?;
850
851    Ok(())
852}
