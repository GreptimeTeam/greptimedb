170    /// Regression test for <https://github.com/GreptimeTeam/greptimedb/issues/9231>:
171    /// prometheus remote write/read against a physical metric table
172    /// pre-created with a non-millisecond time index. Millisecond samples are
173    /// converted to the physical table's unit on write (lossless widening,
174    /// truncating narrowing) and narrowed back to milliseconds on remote read.
175    async fn test_prom_store_remote_rw_non_millisecond_physical_table(
176        instance: &Arc<Instance>,
177        sql_ts_type: &str,
178        expected_unit: TimeUnit,
179        expected_first: i64,
180        expected_second: i64,
181    ) {
182        let db = "prometheus_non_ms";
183        let ctx = Arc::new(QueryContext::with(DEFAULT_CATALOG_NAME, db));
184        assert!(
185            SqlQueryHandler::do_query(
186                instance.as_ref(),
187                &format!("CREATE DATABASE IF NOT EXISTS {db}"),
188                ctx.clone(),
189            )
190            .await
191            .first()
192            .unwrap()
193            .is_ok()
194        );
195
196        let mut output = instance
197            .do_query(
198                &format!(
199                    "CREATE TABLE greptime_physical_table (\
200                 greptime_timestamp {sql_ts_type} NOT NULL, \
201                 greptime_value DOUBLE NULL, \
202                 TIME INDEX (greptime_timestamp)) \
203                 ENGINE = metric WITH ('physical_metric_table' = 'true')"
204                ),
205                ctx.clone(),
206            )
207            .await;
208        assert!(output.remove(0).is_ok());
209
210        let write_request = WriteRequest {
211            timeseries: vec![prom_store::mock_timeseries()[0].clone()],
212            ..Default::default()
213        };
214        let (row_inserts, _) = to_grpc_row_insert_requests(&write_request).unwrap();
215        instance
216            .write(row_inserts, ctx.clone(), true)
217            .await
218            .unwrap();
219
220        let read_request = ReadRequest {
221            queries: vec![Query {
222                start_timestamp_ms: 1000,
223                end_timestamp_ms: 2000,
224                matchers: vec![LabelMatcher {
225                    name: prom_store::METRIC_NAME_LABEL.to_string(),
226                    value: "metric1".to_string(),
227                    r#type: 0,
228                }],
229                ..Default::default()
230            }],
231            ..Default::default()
232        };
233        let resp = instance.read(read_request, ctx.clone()).await.unwrap();
234        let body = prom_store::snappy_decompress(&resp.body).unwrap();
235        let read_response = ReadResponse::decode(&body[..]).unwrap();
236        assert_eq!(1, read_response.results.len());
237        assert_eq!(1, read_response.results[0].timeseries.len());
238        let timeseries = &read_response.results[0].timeseries[0];
239        assert_eq!(
240            timeseries.samples,
241            vec![
242                Sample {
243                    value: 1.0,
244                    timestamp: 1000,
245                },
246                Sample {
247                    value: 2.0,
248                    timestamp: 2000,
249                },
250            ]
251        );
252
253        // The stored timestamps keep the physical table's time index unit.
254        let mut output = instance
255            .do_query("SELECT greptime_timestamp FROM metric1", ctx.clone())
256            .await;
257        let OutputData::Stream(stream) = output.remove(0).unwrap().data else {
258            unreachable!()
259        };
260        let batches = common_recordbatch::RecordBatches::try_collect(stream)
261            .await
262            .unwrap()
263            .take();
264        assert_eq!(batches[0].num_rows(), 2);
265        let ts_column = batches[0].column(0);
266        assert_eq!(
267            ts_column.data_type(),
268            &DataType::Timestamp(expected_unit, None),
269            "unexpected time index type"
270        );
271        let stored = |row: usize| match expected_unit {
272            TimeUnit::Second => ts_column.as_primitive::<TimestampSecondType>().value(row),
273            TimeUnit::Millisecond => ts_column
274                .as_primitive::<TimestampMillisecondType>()
275                .value(row),
276            TimeUnit::Microsecond => ts_column
277                .as_primitive::<TimestampMicrosecondType>()
278                .value(row),
279            TimeUnit::Nanosecond => ts_column
280                .as_primitive::<TimestampNanosecondType>()
281                .value(row),
282        };
283        assert_eq!(stored(0), expected_first);
284        assert_eq!(stored(1), expected_second);
285
286        // Negative, non-aligned timestamps must floor towards negative
287        // infinity on remote read, matching `Timestamp::convert_to` on the
288        // ingestion path: -1001us/-1001000ns -> -2ms, not -1ms.
289        if matches!(expected_unit, TimeUnit::Microsecond | TimeUnit::Nanosecond) {
290            let (ts_datatype, negative_value) = if expected_unit == TimeUnit::Microsecond {
291                (
292                    ColumnDataType::TimestampMicrosecond,
293                    ValueData::TimestampMicrosecondValue(-1001),
294                )
295            } else {
296                (
297                    ColumnDataType::TimestampNanosecond,
298                    ValueData::TimestampNanosecondValue(-1_001_000),
299                )
300            };
301            let negative_request = RowInsertRequests {
302                inserts: vec![RowInsertRequest {
303                    table_name: "metric1".to_string(),
304                    rows: Some(Rows {
305                        schema: vec![
306                            api::v1::ColumnSchema {
307                                column_name: "greptime_timestamp".to_string(),
308                                datatype: ts_datatype as i32,
309                                semantic_type: SemanticType::Timestamp as i32,
310                                datatype_extension: None,
311                                options: None,
312                            },
313                            api::v1::ColumnSchema {
314                                column_name: "greptime_value".to_string(),
315                                datatype: ColumnDataType::Float64 as i32,
316                                semantic_type: SemanticType::Field as i32,
317                                datatype_extension: None,
318                                options: None,
319                            },
320                            api::v1::ColumnSchema {
321                                column_name: "job".to_string(),
322                                datatype: ColumnDataType::String as i32,
323                                semantic_type: SemanticType::Tag as i32,
324                                datatype_extension: None,
325                                options: None,
326                            },
327                        ],
328                        rows: vec![Row {
329                            values: vec![
330                                Value {
331                                    value_data: Some(negative_value),
332                                },
333                                Value {
334                                    value_data: Some(ValueData::F64Value(3.0)),
335                                },
336                                Value {
337                                    value_data: Some(ValueData::StringValue("spark".to_string())),
338                                },
339                            ],
340                        }],
341                    }),
342                }],
343            };
344            instance
345                .write(negative_request, ctx.clone(), true)
346                .await
347                .unwrap();
348
349            let read_request = ReadRequest {
350                queries: vec![Query {
351                    start_timestamp_ms: -2,
352                    end_timestamp_ms: 0,
353                    matchers: vec![LabelMatcher {
354                        name: prom_store::METRIC_NAME_LABEL.to_string(),
355                        value: "metric1".to_string(),
356                        r#type: 0,
357                    }],
358                    ..Default::default()
359                }],
360                ..Default::default()
361            };
362            let resp = instance.read(read_request, ctx.clone()).await.unwrap();
363            let body = prom_store::snappy_decompress(&resp.body).unwrap();
364            let read_response = ReadResponse::decode(&body[..]).unwrap();
365            assert_eq!(1, read_response.results.len());
366            assert_eq!(
367                read_response.results[0].timeseries[0].samples,
368                vec![Sample {
369                    value: 3.0,
370                    timestamp: -2,
371                }],
372                "remote read must floor negative non-aligned timestamps"
373            );
374        }
375    }

377    #[tokio::test(flavor = "multi_thread")]
378    async fn test_standalone_prom_store_write_existing_logical_table_of_other_physical_unit() {
379        common_telemetry::init_default_ut_logging();
380        let standalone = GreptimeDbStandaloneBuilder::new(
381            "test_prom_store_write_existing_logical_other_physical_unit",
382        )
383        .build()
384        .await;
385        let instance = standalone.fe_instance();
386
387        // Two physical metric tables with different time index units.
388        let db = "prometheus_mixed_units";
389        let ctx = Arc::new(QueryContext::with(DEFAULT_CATALOG_NAME, db));
390        assert!(
391            SqlQueryHandler::do_query(
392                instance.as_ref(),
393                &format!("CREATE DATABASE IF NOT EXISTS {db}"),
394                ctx.clone(),
395            )
396            .await
397            .first()
398            .unwrap()
399            .is_ok()
400        );
401        let mut output = instance
402            .do_query(
403                "CREATE TABLE phy_us (greptime_timestamp TIMESTAMP(6) NOT NULL, \
404                 greptime_value DOUBLE NULL, TIME INDEX (greptime_timestamp)) \
405                 ENGINE = metric WITH ('physical_metric_table' = 'true')",
406                ctx.clone(),
407            )
408            .await;
409        assert!(output.remove(0).is_ok());
410
411        // First create the logical table `shared_metric` on the default
412        // (millisecond) physical table.
413        let shared_series = TimeSeries {
414            labels: vec![
415                Label {
416                    name: prom_store::METRIC_NAME_LABEL.to_string(),
417                    value: "shared_metric".to_string(),
418                },
419                Label {
420                    name: "job".to_string(),
421                    value: "demo".to_string(),
422                },
423            ],
424            samples: vec![Sample {
425                value: 1.0,
426                timestamp: 1000,
427            }],
428            ..Default::default()
429        };
430        let (row_inserts, _) = to_grpc_row_insert_requests(&WriteRequest {
431            timeseries: vec![shared_series.clone()],
432            ..Default::default()
433        })
434        .unwrap();
435        instance
436            .write(row_inserts, ctx.clone(), true)
437            .await
438            .unwrap();
439
440        // Now select the microsecond physical table while writing to BOTH the
441        // existing millisecond `shared_metric` (bound to the default physical
442        // table) and a new table: the existing table's request must keep the
443        // millisecond unit, and only the new table uses microsecond.
444        let mut hint_ctx = QueryContext::with(DEFAULT_CATALOG_NAME, db);
445        hint_ctx.set_extension(PHYSICAL_TABLE_PARAM, "phy_us".to_string());
446        let hint_ctx = Arc::new(hint_ctx);
447        let (row_inserts, _) = to_grpc_row_insert_requests(&WriteRequest {
448            timeseries: vec![
449                TimeSeries {
450                    samples: vec![Sample {
451                        value: 1.5,
452                        timestamp: 1500,
453                    }],
454                    ..shared_series
455                },
456                TimeSeries {
457                    labels: vec![
458                        Label {
459                            name: prom_store::METRIC_NAME_LABEL.to_string(),
460                            value: "fresh_us_metric".to_string(),
461                        },
462                        Label {
463                            name: "job".to_string(),
464                            value: "demo".to_string(),
465                        },
466                    ],
467                    samples: vec![Sample {
468                        value: 2.5,
469                        timestamp: 1500,
470                    }],
471                    ..Default::default()
472                },
473            ],
474            ..Default::default()
475        })
476        .unwrap();
477        instance
478            .write(row_inserts, hint_ctx.clone(), true)
479            .await
480            .unwrap();
481
482        // The existing table keeps its millisecond unit and precision.
483        let mut output = instance
484            .do_query(
485                "SELECT greptime_timestamp FROM shared_metric ORDER BY greptime_timestamp",
486                ctx.clone(),
487            )
488            .await;
489        let OutputData::Stream(stream) = output.remove(0).unwrap().data else {
490            unreachable!()
491        };
492        let batches = common_recordbatch::RecordBatches::try_collect(stream)
493            .await
494            .unwrap()
495            .take();
496        let shared_ts = batches[0]
497            .column(0)
498            .as_primitive::<TimestampMillisecondType>();
499        assert_eq!((shared_ts.value(0), shared_ts.value(1)), (1000, 1500));
500
501        // The new table is created on the selected physical table with the
502        // microsecond unit.
503        let mut output = instance
504            .do_query(
505                "SELECT greptime_timestamp FROM fresh_us_metric",
506                ctx.clone(),
507            )
508            .await;
509        let OutputData::Stream(stream) = output.remove(0).unwrap().data else {
510            unreachable!()
511        };
512        let batches = common_recordbatch::RecordBatches::try_collect(stream)
513            .await
514            .unwrap()
515            .take();
516        let fresh_ts = batches[0]
517            .column(0)
518            .as_primitive::<TimestampMicrosecondType>();
519        assert_eq!(fresh_ts.value(0), 1_500_000);
520    }

522    async fn test_prom_store_remote_rw(instance: &Arc<Instance>, physical_table: Option<String>) {
523        let write_request = WriteRequest {
524            timeseries: prom_store::mock_timeseries(),
525            ..Default::default()
526        };
527
528        let db = "prometheus";
529        let mut ctx = Arc::into_inner(QueryContext::with(DEFAULT_CATALOG_NAME, db).into()).unwrap();
530
531        // set physical table if provided
532        if let Some(physical_table) = &physical_table {
533            ctx.set_extension(PHYSICAL_TABLE_PARAM.to_string(), physical_table.clone());
534        }
535        let ctx = Arc::new(ctx);
536
537        assert!(
538            SqlQueryHandler::do_query(
539                instance.as_ref(),
540                "CREATE DATABASE IF NOT EXISTS prometheus",
541                ctx.clone(),
542            )
543            .await
544            .first()
545            .unwrap()
546            .is_ok()
547        );
548
549        let (row_inserts, _) = to_grpc_row_insert_requests(&write_request).unwrap();
550        instance
551            .write(row_inserts, ctx.clone(), true)
552            .await
553            .unwrap();
554
555        let read_request = ReadRequest {
556            queries: vec![
557                Query {
558                    start_timestamp_ms: 1000,
559                    end_timestamp_ms: 2000,
560                    matchers: vec![LabelMatcher {
561                        name: prom_store::METRIC_NAME_LABEL.to_string(),
562                        value: "metric1".to_string(),
563                        r#type: 0,
564                    }],
565                    ..Default::default()
566                },
567                Query {
568                    start_timestamp_ms: 1000,
569                    end_timestamp_ms: 3000,
570                    matchers: vec![
571                        LabelMatcher {
572                            name: prom_store::METRIC_NAME_LABEL.to_string(),
573                            value: "metric3".to_string(),
574                            r#type: 0,
575                        },
576                        LabelMatcher {
577                            name: "app".to_string(),
578                            value: "biz".to_string(),
579                            r#type: MatcherType::Eq as i32,
580                        },
581                    ],
582                    ..Default::default()
583                },
584            ],
585            ..Default::default()
586        };
587
588        let resp = instance.read(read_request, ctx.clone()).await.unwrap();
589        assert_eq!(resp.content_type, "application/x-protobuf");
590        assert_eq!(resp.content_encoding, "snappy");
591        let body = prom_store::snappy_decompress(&resp.body).unwrap();
592        let read_response = ReadResponse::decode(&body[..]).unwrap();
593        let query_results = read_response.results;
594        assert_eq!(2, query_results.len());
595
596        assert_eq!(1, query_results[0].timeseries.len());
597        let timeseries = &query_results[0].timeseries[0];
598
599        assert_eq!(
600            vec![
601                Label {
602                    name: prom_store::METRIC_NAME_LABEL.to_string(),
603                    value: "metric1".to_string(),
604                },
605                Label {
606                    name: "job".to_string(),
607                    value: "spark".to_string(),
608                },
609            ],
610            timeseries.labels
611        );
612
613        assert_eq!(
614            timeseries.samples,
615            vec![
616                Sample {
617                    value: 1.0,
618                    timestamp: 1000,
619                },
620                Sample {
621                    value: 2.0,
622                    timestamp: 2000,
623                }
624            ]
625        );
626
627        assert_eq!(1, query_results[1].timeseries.len());
628        let timeseries = &query_results[1].timeseries[0];
629
630        assert_eq!(
631            vec![
632                Label {
633                    name: prom_store::METRIC_NAME_LABEL.to_string(),
634                    value: "metric3".to_string(),
635                },
636                Label {
637                    name: "app".to_string(),
638                    value: "biz".to_string(),
639                },
640                Label {
641                    name: "idc".to_string(),
642                    value: "z002".to_string(),
643                },
644            ],
645            timeseries.labels
646        );
647
648        assert_eq!(
649            timeseries.samples,
650            vec![
651                Sample {
652                    value: 5.0,
653                    timestamp: 1000,
654                },
655                Sample {
656                    value: 6.0,
657                    timestamp: 2000,
658                },
659                Sample {
660                    value: 7.0,
661                    timestamp: 3000,
662                }
663            ]
664        );
665
666        // check physical table if provided
667        if let Some(physical_table) = physical_table {
668            let sql = format!("DESC TABLE {physical_table};");
669            instance.do_query(&sql, ctx).await[0].as_ref().unwrap();
670        }
671    }
