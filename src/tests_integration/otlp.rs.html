59        let standalone = GreptimeDbStandaloneBuilder::new("test_otlp_fixed_schema")
60            .with_auto_create_table(false)
61            .build()
62            .await;
63        let instance = standalone.fe_instance();
64        let ctx = Arc::new(QueryContext::with(DEFAULT_CATALOG_NAME, "public"));
65        let mut output = instance
66            .do_query(
67                "CREATE TABLE fixed_delta_total (\
68                 \"stream\" STRING, greptime_timestamp TIMESTAMP(3) NOT NULL, \
69                 greptime_value DOUBLE, TIME INDEX (greptime_timestamp), \
70                 PRIMARY KEY (\"stream\")) ENGINE=mito",
71                ctx.clone(),
72            )
73            .await;
74        let result = output.remove(0);
75        assert!(result.is_ok(), "{result:?}");
76
77        let error = instance
78            .metrics(
79                build_sum_request("fixed.delta", AggregationTemporality::Delta, &[(60, 10)]),
80                ctx.clone(),
81            )
82            .await
83            .unwrap_err();
84        assert!(format!("{error:?}").contains("otlp_aggregation_temporality"));
85
86        let mut output = instance
87            .do_query("SELECT COUNT(*) FROM fixed_delta_total", ctx)
88            .await;
89        let OutputData::Stream(stream) = output.remove(0).unwrap().data else {
90            unreachable!()
91        };
92        assert!(
93            RecordBatches::try_collect(stream)

95                .unwrap()
96                .pretty_print()

476    fn build_request() -> ExportMetricsServiceRequest {
477        let data_points = vec![
478            NumberDataPoint {
479                attributes: vec![keyvalue("container.name", "testsevrer")],
480                time_unix_nano: 100,
481                value: Some(Value::AsInt(100)),
482                ..Default::default()
483            },
484            NumberDataPoint {
485                attributes: vec![keyvalue("container.name", "testserver")],
486                time_unix_nano: 105,
487                value: Some(Value::AsInt(105)),
488                ..Default::default()
489            },
490        ];
491        let gauge = Gauge { data_points };
492
493        let histo_data_points = vec![HistogramDataPoint {
494            attributes: vec![keyvalue("host", "testserver")],
495            time_unix_nano: 100,
496            count: 4,
497            bucket_counts: vec![1, 2, 1],
498            explicit_bounds: vec![1.0f64, 5.0f64],
499            sum: Some(51f64),
500            ..Default::default()
501        }];
502
503        let histo = Histogram {
504            data_points: histo_data_points,
505            aggregation_temporality: 0,
506        };
507
508        ExportMetricsServiceRequest {
509            resource_metrics: vec![ResourceMetrics {
510                scope_metrics: vec![ScopeMetrics {
511                    metrics: vec![
512                        Metric {
513                            name: "my.test.metric".into(),
514                            description: "my ignored desc".into(),
515                            unit: "my ignored unit".into(),
516                            metadata: vec![],
517                            data: Some(metric::Data::Gauge(gauge)),
518                        },
519                        Metric {
520                            name: "my.test.histo".into(),
521                            description: "my ignored desc".into(),
522                            unit: "my ignored unit".into(),
523                            metadata: vec![],
524                            data: Some(metric::Data::Histogram(histo)),
525                        },
526                    ],
527                    scope: Some(InstrumentationScope {
528                        attributes: vec![
529                            keyvalue("scope", "otel"),
530                            keyvalue("telemetry.sdk.name", "java"),
531                        ],
532                        ..Default::default()
533                    }),
534                    ..Default::default()
535                }],
536                resource: Some(Resource {
537                    attributes: vec![keyvalue("resource", "greptimedb")],
538                    dropped_attributes_count: 0,
539                    entity_refs: vec![],
540                }),
541                ..Default::default()
542            }],
543        }
544    }

546    fn build_sum_request(
547        name: &str,
548        temporality: AggregationTemporality,
549        points: &[(u64, i64)],
550    ) -> ExportMetricsServiceRequest {
551        let data_points = points
552            .iter()
553            .map(|(seconds, value)| NumberDataPoint {
554                attributes: vec![keyvalue("stream", "same")],
555                time_unix_nano: *seconds * 1_000_000_000,
556                value: Some(Value::AsInt(*value)),
557                ..Default::default()
558            })
559            .collect();
560        ExportMetricsServiceRequest {
561            resource_metrics: vec![ResourceMetrics {
562                scope_metrics: vec![ScopeMetrics {
563                    metrics: vec![Metric {
564                        name: name.to_string(),
565                        data: Some(metric::Data::Sum(Sum {
566                            data_points,
567                            aggregation_temporality: temporality as i32,
568                            is_monotonic: true,
569                        })),
570                        ..Default::default()
571                    }],
572                    ..Default::default()
573                }],
574                ..Default::default()
575            }],
576        }
577    }

579    fn build_histogram_request(
580        name: &str,
581        points: Vec<HistogramDataPoint>,
582    ) -> ExportMetricsServiceRequest {
583        ExportMetricsServiceRequest {
584            resource_metrics: vec![ResourceMetrics {
585                scope_metrics: vec![ScopeMetrics {
586                    metrics: vec![Metric {
587                        name: name.to_string(),
588                        data: Some(metric::Data::Histogram(Histogram {
589                            data_points: points,
590                            aggregation_temporality: AggregationTemporality::Delta as i32,
591                        })),
592                        ..Default::default()
593                    }],
594                    ..Default::default()
595                }],
596                ..Default::default()
597            }],
598        }
599    }
