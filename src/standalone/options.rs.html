37#[derive(Clone, Debug, Serialize, Deserialize, PartialEq)]
38#[serde(default)]
39pub struct StandaloneOptions {
40    pub enable_telemetry: bool,
41    pub default_timezone: Option<String>,
42    pub default_column_prefix: Option<String>,
43    /// Server-side global switch for auto table creation on write.
44    /// Upper bound: when `false`, missing tables are never auto-created even if a
45    /// request sets the `auto_create_table` hint to `true`. Default: `true`.
46    pub auto_create_table: bool,
47    /// Enables experimental Parquet database exports using shared Metric scans.
48    pub experimental_metric_export: bool,
49    /// Maximum total memory for all concurrent write request bodies and messages (HTTP, gRPC, Flight).
50    /// Set to 0 to disable the limit. Default: "0" (unlimited)
51    pub max_in_flight_write_bytes: ReadableSize,
52    /// Policy when write bytes quota is exhausted.
53    /// Options: "wait" (default, 10s), "wait(<duration>)", "fail"
54    pub write_bytes_exhausted_policy: OnExhaustedPolicy,
55    pub http: HttpOptions,
56    pub grpc: GrpcOptions,
57    pub mysql: MysqlOptions,
58    pub postgres: PostgresOptions,
59    pub opentsdb: OpentsdbOptions,
60    pub influxdb: InfluxdbOptions,
61    /// Ordinary-table batching with independent logical-table controls.
62    pub pending_rows_batcher: PendingRowsBatcherOptions,
63    pub jaeger: JaegerOptions,
64    pub otlp: OtlpOptions,
65    pub prom_store: PromStoreOptions,
66    pub wal: DatanodeWalConfig,
67    pub storage: StorageConfig,
68    pub metadata_store: KvBackendConfig,
69    pub procedure: ProcedureConfig,
70    pub flow: FlowConfig,
71    pub logging: LoggingOptions,
72    pub user_provider: Option<String>,
73    /// Options for different store engines.
74    pub region_engine: Vec<RegionEngineConfig>,
75    pub tracing: TracingOptions,
76    pub init_regions_in_background: bool,
77    pub init_regions_parallelism: usize,
78    pub slow_query: SlowQueryOptions,
79    pub query: QueryOptions,
80    pub memory: MemoryOptions,
81    /// The pipeline options.
82    pub pipeline: PipelineOptions,
83    /// The event recorder options.
84    pub event_recorder: EventRecorderOptions,
85    /// Environment variable keys to read and report in heartbeat messages.
86    pub heartbeat_env_vars: Vec<String>,
87}

145/// The [`StandaloneOptions`] is only defined in `standalone` crate,
146/// we don't want to make `frontend` depends on it, so impl [`Into`]
147/// rather than [`From`].
148#[allow(clippy::from_over_into)]
149impl Into<FrontendOptions> for StandaloneOptions {
150    fn into(self) -> FrontendOptions {
151        self.frontend_options()
152    }
153}

167            postgres: cloned_opts.postgres,
168            opentsdb: cloned_opts.opentsdb,

186    pub fn datanode_options(&self) -> DatanodeOptions {

191            wal: cloned_opts.wal,
192            storage: cloned_opts.storage,

194            grpc: cloned_opts.grpc,

196            init_regions_parallelism: cloned_opts.init_regions_parallelism,
197            query: cloned_opts.query,
198            heartbeat_env_vars: cloned_opts.heartbeat_env_vars,
199            ..Default::default()
200        }
201    }
202
203    /// Sanitize the `StandaloneOptions` to ensure the config is valid.
204    pub fn sanitize(&mut self) {
205        if self.storage.is_object_storage() {
206            self.storage
207                .store
208                .cache_config_mut()
209                .unwrap()
210                .sanitize(&self.storage.data_home);
211        }
212    }
213}

224    #[test]
225    fn test_logical_batcher_config_forwarding() {
226        let opts: StandaloneOptions = toml::from_str("[pending_rows_batcher]\nprotocols = ['influxdb']\n[pending_rows_batcher.logical_table]\nprotocols = ['otlp', 'prom']\npending_rows_flush_interval = '10ms'").unwrap();
227        let frontend = opts.frontend_options();
228        assert_eq!(frontend.pending_rows_batcher, opts.pending_rows_batcher);
229        assert_eq!(
230            frontend.pending_rows_batcher.logical_table,
231            opts.pending_rows_batcher.logical_table
232        );
233        let restored: StandaloneOptions = toml::from_str(&toml::to_string(&opts).unwrap()).unwrap();
234        assert_eq!(
235            restored.pending_rows_batcher.logical_table,
236            opts.pending_rows_batcher.logical_table
237        );
238        assert!(
239            toml::from_str::<StandaloneOptions>(
240                "[pending_rows_batcher.logical_table]\nprotocols = ['logs']"
241            )
242            .is_err()
243        );
244    }

295    #[test]
296    fn test_protocol_pending_rows_batcher_config() {
297        let defaults: StandaloneOptions = toml::from_str("").unwrap();
298        assert!(
299            !defaults
300                .pending_rows_batcher
301                .table
302                .pending_rows_batching_enabled()
303        );
304        let options: StandaloneOptions = toml::from_str(
305            r#"
306[pending_rows_batcher]
307protocols = ["influxdb", "http_sql"]
308pending_rows_flush_interval = "5ms"
309max_batch_rows = 25
310flow_notification_queue_capacity = 17
311"#,
312        )
313        .unwrap();
314        assert_eq!(options.pending_rows_batcher.table.max_batch_rows, 25);
315        assert_eq!(options.pending_rows_batcher.table.protocols.len(), 2);
316        assert_eq!(
317            options
318                .pending_rows_batcher
319                .table
320                .flow_notification_queue_capacity
321                .get(),
322            17
323        );
324        assert!(
325            options
326                .pending_rows_batcher
327                .table
328                .pending_rows_batching_enabled()
329        );
330        let serialized = toml::to_string(&options).unwrap();
331        let parsed: StandaloneOptions = toml::from_str(&serialized).unwrap();
332        assert_eq!(options.influxdb, parsed.influxdb);
333        assert_eq!(options.opentsdb, parsed.opentsdb);
334        assert_eq!(options.pending_rows_batcher, parsed.pending_rows_batcher);
335        let frontend = options.frontend_options();
336        assert_eq!(options.influxdb, frontend.influxdb);
337        assert_eq!(options.opentsdb, frontend.opentsdb);
338        assert_eq!(options.pending_rows_batcher, frontend.pending_rows_batcher);
339    }

341    #[test]
342    fn test_event_recorder_event_types_preserve_filter_semantics() {
343        let all: StandaloneOptions = toml::from_str("").unwrap();
344        let none: StandaloneOptions = toml::from_str("[event_recorder]\nevent_types = []").unwrap();
345        let selected: StandaloneOptions =
346            toml::from_str("[event_recorder]\nevent_types = ['create_database']").unwrap();
347
348        assert!(all.event_recorder.event_types.allows("future_event"));
349        assert_eq!(
350            none.event_recorder.event_types.as_ref(),
351            &EventTypeFilter::Only(Default::default())
352        );
353        assert!(
354            selected
355                .event_recorder
356                .event_types
357                .allows("create_database")
358        );
359        assert!(!selected.event_recorder.event_types.allows("drop_database"));
360
361        let frontend_options = selected.frontend_options();
362        assert!(Arc::ptr_eq(
363            &selected.event_recorder.event_types,
364            &frontend_options.event_recorder.event_types,
365        ));
366    }
