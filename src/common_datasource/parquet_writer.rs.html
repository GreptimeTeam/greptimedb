32/// Limits for one Parquet file. Flush thresholds are not hard memory caps.
33#[derive(Clone, Copy, Debug)]
34pub struct ParquetWriterLimits {
35    /// Maximum rows in a row group.
36    pub row_group_rows: usize,
37    /// Flush when encoder memory or encoded size reaches this threshold.
38    pub flush_threshold_bytes: usize,
39    /// Maximum row groups retained in the file footer.
40    pub max_row_groups: usize,
41}
42
43/// Encodes batches on a blocking runtime and writes one object-store file.
44/// Callers must await each operation before aborting; dropping an in-flight
45/// operation can leave a filesystem worker running after cleanup.
46pub struct ParquetFileWriter {
47    encoder: Option<ArrowWriter<Vec<u8>>>,
48    sink: Writer,
49    store: ObjectStore,
50    path: String,
51    limits: Option<ParquetWriterLimits>,
52}
53
54impl ParquetFileWriter {
55    /// Open a file using COPY's encoding settings. Destination ownership and
56    /// overwrite policy belong to the caller. None preserves Parquet's defaults.
57    pub async fn open(
58        schema: SchemaRef,
59        store: ObjectStore,
60        path: &str,
61        concurrency: usize,
62        limits: Option<ParquetWriterLimits>,
63    ) -> Result<Self> {
64        let mut props = WriterProperties::builder()
65            .set_compression(Compression::ZSTD(ZstdLevel::default()))
66            .set_statistics_truncate_length(None)
67            .set_column_index_truncate_length(None);
68        if let Some(limits) = limits {
69            ensure!(
70                limits.row_group_rows > 0
71                    && limits.flush_threshold_bytes > 0
72                    && limits.max_row_groups > 0,
73                error::InvalidParquetWriterLimitsSnafu
74            );
75            props = props
76                .set_max_row_group_row_count(Some(limits.row_group_rows))
77                .set_max_row_group_bytes(None);
78        }
79        for field in schema.fields() {
80            if matches!(field.data_type(), DataType::Timestamp(_, _)) {
81                let column = ColumnPath::new(vec![field.name().clone()]);
82                props = props
83                    .set_column_dictionary_enabled(column.clone(), false)
84                    .set_column_encoding(column, Encoding::DELTA_BINARY_PACKED);
85            }
86        }
87        let encoder = ArrowWriter::try_new(Vec::new(), schema, Some(props.build()))
88            .context(error::WriteParquetSnafu { path })?;
89        let sink = store
90            .writer_with(path)
91            .concurrent(concurrency)
92            .chunk(DEFAULT_WRITE_BUFFER_SIZE.as_bytes() as usize)
93            .await
94            .context(error::WriteObjectSnafu { path })?;
95        Ok(Self {
96            encoder: Some(encoder),
97            sink,
98            store,
99            path: path.to_owned(),
100            limits,
101        })
102    }
103
104    /// Write a batch, enforcing file limits across batch and row-group boundaries.
105    pub async fn write(
106        &mut self,
107        batch: RecordBatch,
108        cancellation: Option<&CancellationToken>,
109    ) -> Result<()> {
110        let mut offset = 0;
111        while offset < batch.num_rows() {
112            check_cancelled(cancellation)?;
113            let mut encoder = self.encoder.take().ok_or_else(|| {
114                error::WriteParquetSnafu { path: &self.path }
115                    .into_error(ParquetError::General("Parquet writer is closed".into()))
116            })?;
117            let len = self.limits.map_or(batch.num_rows() - offset, |limits| {
118                (limits.row_group_rows - encoder.in_progress_rows()).min(batch.num_rows() - offset)
119            });
120            let slice = batch.slice(offset, len);
121            let limits = self.limits;
122            let path = self.path.clone();
123            let (encoder, bytes) = common_runtime::spawn_blocking_global(move || {
124                if let Some(limits) = limits {
125                    ensure!(
126                        encoder.flushed_row_groups().len() < limits.max_row_groups,
127                        error::ParquetWriterResourceSnafu {
128                            reason: "Parquet row-group metadata budget exceeded"
129                        }
130                    );
131                }
132                encoder
133                    .write(&slice)
134                    .context(error::WriteParquetSnafu { path: &path })?;
135                if limits.is_some_and(|limits| {
136                    encoder.memory_size() >= limits.flush_threshold_bytes
137                        || encoder.in_progress_size() >= limits.flush_threshold_bytes
138                }) {
139                    encoder
140                        .flush()
141                        .context(error::WriteParquetSnafu { path: &path })?;
142                }
143                // Draining preserves ArrowWriter's cumulative file offsets.
144                let bytes = std::mem::take(encoder.inner_mut());
145                Ok::<_, error::Error>((encoder, bytes))
146            })
147            .await
148            .context(error::JoinHandleSnafu)??;
149            self.encoder = Some(encoder);
150            check_cancelled(cancellation)?;
151            self.write_bytes(bytes).await?;
152            check_cancelled(cancellation)?;
153            offset += len;
154        }
155        Ok(())
156    }
157
158    async fn write_bytes(&mut self, bytes: Vec<u8>) -> Result<()> {
159        if !bytes.is_empty() {
160            self.sink
161                .write(bytes)
162                .await
163                .context(error::WriteObjectSnafu { path: &self.path })?;
164        }
165        Ok(())
166    }
167
168    /// Write the footer and close the file. Retains the handle for cleanup on error.
169    pub async fn finish(&mut self, cancellation: Option<&CancellationToken>) -> Result<()> {
170        check_cancelled(cancellation)?;
171        let mut encoder = self.encoder.take().ok_or_else(|| {
172            error::WriteParquetSnafu { path: &self.path }
173                .into_error(ParquetError::General("Parquet writer is closed".into()))
174        })?;
175        let path = self.path.clone();
176        let bytes = common_runtime::spawn_blocking_global(move || {
177            encoder
178                .finish()
179                .context(error::WriteParquetSnafu { path: &path })?;
180            Ok::<_, error::Error>(std::mem::take(encoder.inner_mut()))
181        })
182        .await
183        .context(error::JoinHandleSnafu)??;
184        self.write_bytes(bytes).await?;
185        check_cancelled(cancellation)?;
186        self.sink
187            .close()
188            .await
189            .context(error::WriteObjectSnafu { path: &self.path })?;
190        check_cancelled(cancellation)?;
191        Ok(())
192    }
193
194    /// Abort an exclusively owned file after all in-flight operations have completed.
195    /// If the backend cannot abort, deletion assumes this attempt owns the path.
196    pub async fn abort(mut self) -> Result<()> {
197        let result = self.sink.abort().await;
198        if result
199            .as_ref()
200            .is_err_and(|e| e.kind() == object_store::ErrorKind::Unsupported)
201        {
202            let store = self.store.clone();
203            let path = self.path.clone();
204            // Secure filesystem writers require dropping the handle before deletion.
205            drop(self);
206            store
207                .delete(&path)
208                .await
209                .context(error::WriteObjectSnafu { path })?;
210        } else {
211            result.context(error::WriteObjectSnafu { path: &self.path })?;
212        }
213        Ok(())
214    }
215}
216
217fn check_cancelled(cancellation: Option<&CancellationToken>) -> Result<()> {
218    ensure!(
219        cancellation.is_none_or(|token| !token.is_cancelled()),
220        error::ParquetWriteCancelledSnafu
221    );
222    Ok(())
223}

225/// Bridges opendal [Writer] with parquet [AsyncFileWriter].
226pub struct AsyncWriter {
227    inner: Writer,
228}
229
230impl AsyncWriter {
231    /// Create a [`AsyncWriter`] by given [`Writer`].
232    pub fn new(writer: Writer) -> Self {
233        Self { inner: writer }
234    }
235}
236
237impl AsyncFileWriter for AsyncWriter {
238    fn write(&mut self, bs: Bytes) -> BoxFuture<'_, parquet::errors::Result<()>> {
239        Box::pin(async move {
240            self.inner
241                .write(bs)
242                .await
243                .map_err(|err| ParquetError::External(Box::new(err)))
244        })
245    }
246
247    fn complete(&mut self) -> BoxFuture<'_, parquet::errors::Result<()>> {
248        Box::pin(async move {
249            self.inner
250                .close()
251                .await
252                .map(|_| ())
253                .map_err(|err| ParquetError::External(Box::new(err)))
254        })
255    }
256}

293        let store = ObjectStore::new(object_store::services::Memory::default()).unwrap();
294        let batch = batch();
295        for (row_group_rows, flush_threshold_bytes, split) in [(2, usize::MAX, 3), (100, 1, 2)] {
296            let mut writer = ParquetFileWriter::open(
297                batch.schema(),
298                store.clone(),
299                "groups.parquet",

302                    row_group_rows,
303                    flush_threshold_bytes,
304                    max_row_groups: 2,
305                }),
306            )
307            .await
308            .unwrap();
309            writer.write(batch.slice(0, split), None).await.unwrap();

321                assert!(!encodings.contains(&Encoding::RLE_DICTIONARY));
322                assert_eq!(

326            }
327            let actual = reader
328                .build()
329                .unwrap()
330                .collect::<std::result::Result<Vec<_>, _>>()
331                .unwrap();
332            assert_eq!(
333                arrow::compute::concat_batches(&batch.schema(), &actual).unwrap(),

418    struct PausedFooterWriter {
419        inner: object_store::layers::mock::oio::Writer,
420        paused: bool,
421        started: Arc<tokio::sync::Notify>,
422        release: Arc<tokio::sync::Notify>,
423        closed: Arc<std::sync::atomic::AtomicBool>,
424    }
425
426    impl object_store::layers::mock::oio::Write for PausedFooterWriter {
427        async fn write(&mut self, bytes: object_store::Buffer) -> object_store::Result<()> {
428            if !self.paused {
429                self.paused = true;
430                self.started.notify_one();
431                self.release.notified().await;
432            }
433            self.inner.write(bytes).await
434        }
435        async fn close(&mut self) -> object_store::Result<object_store::layers::mock::Metadata> {
436            self.closed.store(true, std::sync::atomic::Ordering::SeqCst);
437            self.inner.close().await
438        }
439        async fn abort(&mut self) -> object_store::Result<()> {
440            self.inner.abort().await
441        }
442    }

444    #[tokio::test]
445    async fn cancellation_during_footer_write_waits_then_aborts_before_close() {
446        use object_store::layers::mock::{MockLayerBuilder, MockWriterFactory};
447        let started = Arc::new(tokio::sync::Notify::new());
448        let release = Arc::new(tokio::sync::Notify::new());
449        let closed = Arc::new(std::sync::atomic::AtomicBool::new(false));
450        let factory: MockWriterFactory = Arc::new({
451            let (started, release, closed) = (started.clone(), release.clone(), closed.clone());
452            move |_, _, inner| {
453                Box::new(PausedFooterWriter {
454                    inner,
455                    paused: false,
456                    started: started.clone(),
457                    release: release.clone(),
458                    closed: closed.clone(),
459                })
460            }
461        });
462        let store = ObjectStore::new(object_store::services::Memory::default())
463            .unwrap()
464            .layer(
465                MockLayerBuilder::default()
466                    .writer_factory(factory)
467                    .build()
468                    .unwrap(),
469            );
470        let mut writer =
471            ParquetFileWriter::open(batch().schema(), store.clone(), "cancel.parquet", 1, None)
472                .await
473                .unwrap();
474        // Force footer bytes through the sink before its close operation.
475        writer.sink = store.writer_with("cancel.parquet").chunk(1).await.unwrap();
476        let cancellation = CancellationToken::new();
477        let result = {
478            let finish = writer.finish(Some(&cancellation));
479            tokio::pin!(finish);
480            tokio::select! {
481                result = &mut finish => panic!("finished before footer write: {result:?}"),
482                _ = started.notified() => {},
483            }
484            cancellation.cancel();
485            assert!(futures::poll!(&mut finish).is_pending());
486            release.notify_one();
487            finish.await
488        };
489        assert!(matches!(
490            result,
491            Err(error::Error::ParquetWriteCancelled {})
492        ));
493        assert!(!closed.load(std::sync::atomic::Ordering::SeqCst));
494        writer.abort().await.unwrap();
495        assert!(!store.exists("cancel.parquet").await.unwrap());
496    }
