520/// Bound source for materializing labels from raw series chunks.
521#[derive(Clone, Debug)]
522pub enum RawLabelSource {
523    /// Scan-bound metadata for labels. A producer may expose encoded keys or
524    /// complete explicit labels; its stream decode specification distinguishes them.
525    PrimaryKey {
526        /// Metric name associated with the bound logical source.
527        metric_name: Arc<str>,
528        /// Physical metadata snapshot used to decode primary-key values.
529        physical_metadata: RegionMetadataRef,
530        /// Logical tag names mapped to physical column identifiers.
531        logical_tags: Arc<[BoundTag]>,
532    },
533    /// The stream supplies a complete metric name and label set in each chunk, not a partial
534    /// label projection; `Some(metric)` is a complete name/label set (the name itself may be
535    /// absent), or, for a stream that has no identity at all, see
536    /// [`NO_BOUND_LABEL_SOURCE`].
537    Explicit,
538}
539
540/// The identity a stream reports when no scan-bound label source was established for it.
541///
542/// [`RawLabelSource`] has no "absent" variant, and [`RawLabelSource::Explicit`] is the only one that
543/// carries no physical region key layout, so it stands in for "no identity" here. It is not a claim
544/// that the chunks carry labels: [`RawSeriesStream::label_decode_spec`] answers that, and a stream
545/// reporting this value can still return `Some` description and leave `metric` as `None` in every
546/// chunk. Decoding therefore asks the description, and only an identity check asks this.
547pub const NO_BOUND_LABEL_SOURCE: RawLabelSource = RawLabelSource::Explicit;
548
549/// A stream of raw series chunks with stable schema, label binding, and ordering.
550///
551/// Every series has strictly increasing timestamps within and across chunks; timestamps are
552/// unique per series. `Interleaved` permits a series to resume after another series; `Contiguous`
553/// and `KeyThenTime` guarantee each series' chunks are contiguous. `KeyThenTime` uses the complete
554/// stream identity compatible with its label binding, not just a TSID or hash. These are
555/// per-stream guarantees, not query-global order, partition ownership, or finality. `is_last`
556/// marks the last chunk of that series in this stream only; merging requires recomputing
557/// per-input-series finalization across all related inputs, not aggregate-group finality.
558/// Label interpretation is carried by the stream: [`RawSeriesStream::label_source`] states the
559/// scan-bound identity a producer read under, and [`RawSeriesStream::label_decode_spec`] states the
560/// label layout the stream's own chunks declare. Stream finality applies to all related input
561/// series, not aggregate groups or vector-match state.
562pub trait RawSeriesStream: Stream<Item = Result<RawSeriesChunk, BoxedError>> + Send {
563    /// Returns the schema shared by all chunks; raw schemas have no evaluation grid.
564    fn schema(&self) -> &Arc<SeriesSchema>;
565    /// Returns scan-bound label metadata. The binding does not dictate representation:
566    /// chunks with a decode specification leave `metric` as `None`; otherwise they
567    /// supply the complete explicit label set. The metric name itself may be absent.
568    ///
569    /// A stream that was decoded from a raw frame has no scan-bound identity and reports
570    /// [`NO_BOUND_LABEL_SOURCE`], which reads as `Explicit` while its chunks may still carry no
571    /// labels at all. That value therefore answers "is this the region I read under", never "are
572    /// these labels complete": [`Self::label_decode_spec`] is the stream's statement about its
573    /// encoding, and every consumer that decides whether to decode reads that instead.
574    fn label_source(&self) -> &RawLabelSource;
575    /// Returns the label decode description this stream declares for key-encoded labels.
576    ///
577    /// Every stream whose chunks carry key-encoded labels declares the description here, whether
578    /// it is a storage producer or a stream reassembled from a raw frame. A consumer holding only
579    /// that description can therefore decode labels without a scan-bound [`RawLabelSource`]: the
580    /// description, not the binding, describes the encoding, and a receiver reporting
581    /// [`NO_BOUND_LABEL_SOURCE`] still states through this description that its chunks are
582    /// undecoded key chunks. A stream that carries complete labels instead answers `None`. This is
583    /// the description the stream declares, not an authenticated statement about where the values
584    /// came from.
585    fn label_decode_spec(&self) -> Option<&LabelDecodeSpec> {
586        None
587    }
588    /// Returns this stream's ordering guarantee.
589    fn ordering(&self) -> SeriesOrdering;
590}

606/// An immutable, finite selected set of series prepared for independent range reads.
607///
608/// The count may include series with no samples after filtering. Ordinals are stable local
609/// indexes, not protocol identifiers; each has exactly one complete identity returned by
610/// [`PreparedSeriesSource::series_info`], and that identity matches chunks for that ordinal.
611/// Every reader exposes the same fixed schema, label binding, and ordering before polling,
612/// including for an empty range, and describes labels independently of this source's lifetime.
613/// Key-encoded output has `metric: None`; explicit-label output has a complete
614/// `Some(metric)` matching each chunk, independently of its scan binding.
615/// Ordering describes the stream, not ordinal enumeration.
616///
617/// Preparation, identity lookup, and reader creation do not load sample payloads or perform
618/// hidden blocking I/O. The selected set is bound to the scanner's immutable MVCC view,
619/// predicates, authorization, projection, labels, and existing scan resources. Readers own
620/// resources needed to continue after this source or scanner is dropped. Dropping one reader
621/// does not cancel sibling readers; normal ownership cleanup applies without detached work.
622///
623/// A range must satisfy `start <= end <= num_series()`. Invalid ranges and invalid ordinals
624/// return errors; an empty in-bounds range returns an ordinary empty stream. Repeated and
625/// overlapping reads are independent and do not consume shared iteration state.
626pub trait PreparedSeriesSource: Send + Sync {
627    /// Returns the exact, stable number of selected local series ordinals.
628    fn num_series(&self) -> usize;
629
630    /// Returns label binding shared by series metadata and every reader, without opening or
631    /// polling a reader; the binding is self-contained for remote consumers.
632    fn label_source(&self) -> &RawLabelSource;
633
634    /// Returns identity metadata for an ordinal without polling sample data.
635    fn series_info(&self, ordinal: usize) -> Result<SeriesInfo, BoxedError>;
636
637    /// Creates an independent reader for the half-open ordinal range.
638    fn read_series(
639        &self,
640        range: std::ops::Range<usize>,

644/// A scanner that provides a way to scan the region concurrently.
645///
646/// The scanner splits the region into partitions so that each partition can be scanned concurrently.
647/// You can use this trait to implement an [`ExecutionPlan`](datafusion_physical_plan::ExecutionPlan).
648pub trait RegionScanner: Debug + DisplayAs + Send {
649    fn name(&self) -> &str;
650
651    /// Returns the properties of the scanner.
652    fn properties(&self) -> &ScannerProperties;
653
654    /// Returns the schema of the record batches.
655    fn schema(&self) -> SchemaRef;
656
657    /// Returns the metadata of the region.
658    fn metadata(&self) -> RegionMetadataRef;
659
660    /// Prepares the scanner with the given partition ranges.
661    ///
662    /// This method is for the planner to adjust the scanner's behavior based on the partition ranges.
663    fn prepare(&mut self, request: PrepareRequest) -> Result<(), BoxedError>;
664
665    /// Scans the partition and returns a stream of record batches.
666    ///
667    /// # Panics
668    /// Panics if the `partition` is out of bound.
669    fn scan_partition(
670        &self,
671        ctx: &QueryScanContext,
672        metrics_set: &ExecutionPlanMetricsSet,
673        partition: usize,
674    ) -> Result<SendableRecordBatchStream, BoxedError>;
675
676    /// Checks whether this scanner can provide a raw-series scan for the given labels.
677    ///
678    /// This is a non-consuming preflight: it must not start a scan or otherwise consume
679    /// scanner state. It checks immutable label binding, schema, and typed residual-predicate
680    /// eligibility. Returning `Some` requires overriding `scan_series_partition` and
681    /// promises that it can execute with this label source. `None` means the raw path is
682    /// unsupported and an ordinary scan remains available. Execution must revalidate and
683    /// return errors rather than falling back if the binding or schema changed since preflight.
684    fn raw_series_schema(
685        &self,
686        _label_source: &RawLabelSource,
687    ) -> Result<Option<Arc<SeriesSchema>>, BoxedError> {
688        Ok(None)
689    }
690
691    /// Whether raw chunks carry complete explicit labels under this scan binding.
692    /// This non-consuming declaration must agree with the stream before its first poll.
693    /// Returning true does not replace the immutable source binding or authorization checks.
694    fn raw_series_has_explicit_labels(&self, _labels: &RawLabelSource) -> Result<bool, BoxedError> {
695        Ok(false)
696    }
697
698    /// Returns the guarantee for subsequent raw scans without consuming scanner state.
699    ///
700    /// Legacy raw sources have no ordering declaration, so `Contiguous` is the conservative
701    /// default; `Interleaved` is not representable on the raw frame wire.
702    fn raw_series_ordering(&self) -> Result<SeriesOrdering, BoxedError> {
703        Ok(SeriesOrdering::Contiguous)
704    }
705
706    /// Scans a partition as raw series when this source supports that path.
707    ///
708    /// `None` means raw capability is absent before execution and permits a legacy scan instead.
709    /// `Some` selects this entrypoint, including for an empty successful stream. Errors propagate;
710    /// callers must not replay a failed raw scan through the legacy path. Use only one entrypoint
711    /// for a given scanner partition per execution. The explicit label source is owned by the
712    /// returned stream and must be bound to this scanner's metadata snapshot.
713    ///
714    /// Output preserves the ordinary storage reader's pushdown and MVCC semantics. Inexact
715    /// pushed-down predicates can leave false-positive candidates, just as in `scan_partition`.
716    /// Implementations must preserve typed residual-predicate semantics before any lossy value
717    /// conversion, or return `None` before execution when they cannot. Callers must retain
718    /// residual predicate columns until those predicates are evaluated and apply the final
719    /// projection afterward; raw capability does not mean filtering is complete.
720    ///
721    /// # Panics
722    ///
723    /// The default implementation panics. Implementations must override this method to support
724    /// raw series scanning.
725    fn scan_series_partition(
726        &self,
727        _ctx: &QueryScanContext,
728        _metrics_set: &ExecutionPlanMetricsSet,
729        _partition: usize,
730        _label_source: &RawLabelSource,
731    ) -> Result<Option<SendableRawSeriesStream>, BoxedError> {
732        todo!()
733    }
734
735    /// Prepares scanner-wide candidates once for independent ordinal-range reads.
736    ///
737    /// Metadata I/O is asynchronous; sample payloads are loaded only by polling readers.
738    /// The owned future does not borrow the scanner. `None` means unsupported before
739    /// execution; errors must propagate without replay through a different scan path.
740    /// Implementations may share preparation with their raw partition adapter.
741    fn prepare_series(
742        &self,
743        _ctx: &QueryScanContext,
744        _metrics_set: &ExecutionPlanMetricsSet,
745        _label_source: &RawLabelSource,
746    ) -> futures::future::BoxFuture<
747        'static,
748        Result<Option<Arc<dyn PreparedSeriesSource>>, BoxedError>,
749    > {
750        Box::pin(async { Ok(None) })
751    }
752
753    /// Check if there is any predicate exclude region partition exprs that may be executed in this scanner.
754    fn has_predicate_without_region(&self) -> bool;
755
756    /// Add the given dynamic filter expressions to the predicate of the scanner.
757    /// Returns a vector of booleans indicating which filter expressions were applied.
758    /// true indicates the filter expression was applied(will be use by scanner to prune by stat for row group),
759    /// false otherwise.
760    fn add_dyn_filter_to_predicate(
761        &mut self,
762        filter_exprs: Vec<Arc<dyn PhysicalExpr>>,
763    ) -> Vec<bool>;
764
765    /// Resets execution state after streams from the previous execution have been dropped.
766    ///
767    /// Callers must not concurrently execute, reset, or install filters through scanner aliases.
768    /// Implementations retaining dynamic predicates must discard them and derived pruning state,
769    /// while preserving static predicates, the source snapshot, and prepared properties.
770    /// The default is a no-op and does not make inherently one-shot sources replayable.
771    fn reset_state(&mut self) {}
772
773    /// Sets whether the scanner reads one logical metric region, including a
774    /// physical metric region constrained by a table-ID filter.
775    fn set_logical_region(&mut self, logical_region: bool);
776
777    /// Sets the region id that should receive query-load metrics.
778    fn set_query_load_region_id(&mut self, region_id: RegionId);
779
780    fn snapshot_sequence(&self) -> Option<SequenceNumber> {
781        None
782    }

1346/// The source regions that publish their files into a target region another source publishes
1347/// into too.
1348///
1349/// This is the shape in which a repartition can publish one object twice: two distinct sources
1350/// hand their file sets to the same target, so a shared object and a materialized copy of it can
1351/// both become visible there. It is also the exact shape that the two ownership checks guard - the
1352/// pre-publication check of the repartition procedure, which asks [`Self`] which sources to
1353/// validate, and the remap-time guard
1354/// (`mito2::remap_manifest::RemapManifest::is_split`), which exempts the other shapes.
1355///
1356/// The predicate is deliberately stated per *target*, not per source count: whether one source
1357/// feeds one target or several says nothing about combining file sets. A one-to-many mapping
1358/// hands each child its own part of every object, and a one-to-one mapping hands the object to
1359/// the only target, so neither combines two sources' copies. Only a target fed by two or more
1360/// distinct sources does.
1361///
1362/// Returned in ascending region id order, so a message that names them is stable.
1363pub fn sources_sharing_a_target(mapping: &HashMap<RegionId, Vec<RegionId>>) -> Vec<RegionId> {
1364    let mut sources_per_target: HashMap<RegionId, BTreeSet<RegionId>> = HashMap::new();
1365    for (source, targets) in mapping {
1366        for target in targets {
1367            sources_per_target
1368                .entry(*target)
1369                .or_default()
1370                .insert(*source);
1371        }
1372    }
1373    let mut shared: BTreeSet<RegionId> = BTreeSet::new();
1374    for sources in sources_per_target
1375        .values()
1376        .filter(|sources| sources.len() > 1)
1377    {
1378        shared.extend(sources.iter().copied());
1379    }
1380    shared.into_iter().collect()
1381}
1382
1383/// Whether every target region receives files from at most one source region, i.e. whether
1384/// no repartition in this mapping can publish an object twice.
1385///
1386/// The negation of [`sources_sharing_a_target`] being empty.
1387pub fn every_target_has_a_single_source(mapping: &HashMap<RegionId, Vec<RegionId>>) -> bool {
1388    sources_sharing_a_target(mapping).is_empty()
1389}

1447    /// Handles batch open region requests.
1448    async fn handle_batch_open_requests(
1449        &self,
1450        parallelism: usize,
1451        requests: Vec<(RegionId, RegionOpenRequest)>,
1452    ) -> Result<BatchResponses, BoxedError> {
1453        let semaphore = Arc::new(Semaphore::new(parallelism));
1454        let mut tasks = Vec::with_capacity(requests.len());
1455
1456        for (region_id, request) in requests {
1457            let semaphore_moved = semaphore.clone();
1458
1459            tasks.push(async move {
1460                // Safety: semaphore must exist
1461                let _permit = semaphore_moved.acquire().await.unwrap();
1462                let result = self
1463                    .handle_request(region_id, RegionRequest::Open(request))
1464                    .await;
1465                (region_id, result)
1466            });
1467        }
1468
1469        Ok(join_all(tasks).await)
1470    }

1501    /// each put on its own. An engine that writes several puts together may answer them with
1502    /// one response whose affected rows count all of them.

1504        self: Arc<Self>,
1505        requests: Vec<(RegionId, RegionPutRequest)>,
1506    ) -> Vec<BatchPutResponse>
1507    where
1508        Self: 'static,
1509    {
1510        requests
1511            .into_iter()
1512            .map(|(region_id, request)| {
1513                let engine = self.clone();
1514                let response = async move {
1515                    engine
1516                        .handle_request(region_id, RegionRequest::Put(request))
1517                        .await
1518                }
1519                .boxed();

1526        &self,
1527        request: BatchRegionDdlRequest,
1528    ) -> Result<RegionResponse, BoxedError> {

1530
1531        let mut affected_rows = 0;
1532        let mut extensions = HashMap::new();
1533
1534        for (region_id, request) in requests {
1535            let result = self.handle_request(region_id, request).await?;

1537            extensions.extend(result.extensions);
1538        }
1539
1540        Ok(RegionResponse {
1541            affected_rows,
1542            extensions,
1543            metadata: Vec::new(),
1544        })
1545    }
