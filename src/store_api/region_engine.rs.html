517/// Bound source for materializing labels from raw series chunks.
518#[derive(Clone, Debug)]
519pub enum RawLabelSource {
520    /// Scan-bound metadata for labels. A producer may expose encoded keys or
521    /// complete explicit labels; its stream decode specification distinguishes them.
522    PrimaryKey {
523        /// Metric name associated with the bound logical source.
524        metric_name: Arc<str>,
525        /// Physical metadata snapshot used to decode primary-key values.
526        physical_metadata: RegionMetadataRef,
527        /// Logical tag names mapped to physical column identifiers.
528        logical_tags: Arc<[BoundTag]>,
529    },
530    /// The stream supplies a complete metric name and label set in each chunk, not a partial
531    /// label projection; `Some(metric)` is a complete name/label set (the name itself may be
532    /// absent), or, for a stream that has no identity at all, see
533    /// [`NO_BOUND_LABEL_SOURCE`].
534    Explicit,
535}
536
537/// The identity a stream reports when no scan-bound label source was established for it.
538///
539/// [`RawLabelSource`] has no "absent" variant, and [`RawLabelSource::Explicit`] is the only one that
540/// carries no physical region key layout, so it stands in for "no identity" here. It is not a claim
541/// that the chunks carry labels: [`RawSeriesStream::label_decode_spec`] answers that, and a stream
542/// reporting this value can still return `Some` description and leave `metric` as `None` in every
543/// chunk. Decoding therefore asks the description, and only an identity check asks this.
544pub const NO_BOUND_LABEL_SOURCE: RawLabelSource = RawLabelSource::Explicit;
545
546/// A stream of raw series chunks with stable schema, label binding, and ordering.
547///
548/// Every series has strictly increasing timestamps within and across chunks; timestamps are
549/// unique per series. `Interleaved` permits a series to resume after another series; `Contiguous`
550/// and `KeyThenTime` guarantee each series' chunks are contiguous. `KeyThenTime` uses the complete
551/// stream identity compatible with its label binding, not just a TSID or hash. These are
552/// per-stream guarantees, not query-global order, partition ownership, or finality. `is_last`
553/// marks the last chunk of that series in this stream only; merging requires recomputing
554/// per-input-series finalization across all related inputs, not aggregate-group finality.
555/// Label interpretation is carried by the stream: [`RawSeriesStream::label_source`] states the
556/// scan-bound identity a producer read under, and [`RawSeriesStream::label_decode_spec`] states the
557/// label layout the stream's own chunks declare. Stream finality applies to all related input
558/// series, not aggregate groups or vector-match state.
559pub trait RawSeriesStream: Stream<Item = Result<RawSeriesChunk, BoxedError>> + Send {
560    /// Returns the schema shared by all chunks; raw schemas have no evaluation grid.
561    fn schema(&self) -> &Arc<SeriesSchema>;
562    /// Returns scan-bound label metadata. The binding does not dictate representation:
563    /// chunks with a decode specification leave `metric` as `None`; otherwise they
564    /// supply the complete explicit label set. The metric name itself may be absent.
565    ///
566    /// A stream that was decoded from a raw frame has no scan-bound identity and reports
567    /// [`NO_BOUND_LABEL_SOURCE`], which reads as `Explicit` while its chunks may still carry no
568    /// labels at all. That value therefore answers "is this the region I read under", never "are
569    /// these labels complete": [`Self::label_decode_spec`] is the stream's statement about its
570    /// encoding, and every consumer that decides whether to decode reads that instead.
571    fn label_source(&self) -> &RawLabelSource;
572    /// Returns the label decode description this stream declares for key-encoded labels.
573    ///
574    /// Every stream whose chunks carry key-encoded labels declares the description here, whether
575    /// it is a storage producer or a stream reassembled from a raw frame. A consumer holding only
576    /// that description can therefore decode labels without a scan-bound [`RawLabelSource`]: the
577    /// description, not the binding, describes the encoding, and a receiver reporting
578    /// [`NO_BOUND_LABEL_SOURCE`] still states through this description that its chunks are
579    /// undecoded key chunks. A stream that carries complete labels instead answers `None`. This is
580    /// the description the stream declares, not an authenticated statement about where the values
581    /// came from.
582    fn label_decode_spec(&self) -> Option<&LabelDecodeSpec> {
583        None
584    }
585    /// Returns this stream's ordering guarantee.
586    fn ordering(&self) -> SeriesOrdering;
587}

603/// An immutable, finite selected set of series prepared for independent range reads.
604///
605/// The count may include series with no samples after filtering. Ordinals are stable local
606/// indexes, not protocol identifiers; each has exactly one complete identity returned by
607/// [`PreparedSeriesSource::series_info`], and that identity matches chunks for that ordinal.
608/// Every reader exposes the same fixed schema, label binding, and ordering before polling,
609/// including for an empty range, and describes labels independently of this source's lifetime.
610/// Key-encoded output has `metric: None`; explicit-label output has a complete
611/// `Some(metric)` matching each chunk, independently of its scan binding.
612/// Ordering describes the stream, not ordinal enumeration.
613///
614/// Preparation, identity lookup, and reader creation do not load sample payloads or perform
615/// hidden blocking I/O. The selected set is bound to the scanner's immutable MVCC view,
616/// predicates, authorization, projection, labels, and existing scan resources. Readers own
617/// resources needed to continue after this source or scanner is dropped. Dropping one reader
618/// does not cancel sibling readers; normal ownership cleanup applies without detached work.
619///
620/// A range must satisfy `start <= end <= num_series()`. Invalid ranges and invalid ordinals
621/// return errors; an empty in-bounds range returns an ordinary empty stream. Repeated and
622/// overlapping reads are independent and do not consume shared iteration state.
623pub trait PreparedSeriesSource: Send + Sync {
624    /// Returns the exact, stable number of selected local series ordinals.
625    fn num_series(&self) -> usize;
626
627    /// Returns label binding shared by series metadata and every reader, without opening or
628    /// polling a reader; the binding is self-contained for remote consumers.
629    fn label_source(&self) -> &RawLabelSource;
630
631    /// Returns identity metadata for an ordinal without polling sample data.
632    fn series_info(&self, ordinal: usize) -> Result<SeriesInfo, BoxedError>;
633
634    /// Creates an independent reader for the half-open ordinal range.
635    fn read_series(
636        &self,
637        range: std::ops::Range<usize>,
638    ) -> Result<SendableRawSeriesStream, BoxedError>;
639}

643/// The scanner splits the region into partitions so that each partition can be scanned concurrently.
644/// You can use this trait to implement an [`ExecutionPlan`](datafusion_physical_plan::ExecutionPlan).
645pub trait RegionScanner: Debug + DisplayAs + Send {
646    fn name(&self) -> &str;
647
648    /// Returns the properties of the scanner.
649    fn properties(&self) -> &ScannerProperties;
650
651    /// Returns the schema of the record batches.
652    fn schema(&self) -> SchemaRef;
653
654    /// Returns the metadata of the region.
655    fn metadata(&self) -> RegionMetadataRef;
656
657    /// Prepares the scanner with the given partition ranges.
658    ///
659    /// This method is for the planner to adjust the scanner's behavior based on the partition ranges.
660    fn prepare(&mut self, request: PrepareRequest) -> Result<(), BoxedError>;
661
662    /// Scans the partition and returns a stream of record batches.
663    ///
664    /// # Panics
665    /// Panics if the `partition` is out of bound.
666    fn scan_partition(
667        &self,
668        ctx: &QueryScanContext,
669        metrics_set: &ExecutionPlanMetricsSet,
670        partition: usize,
671    ) -> Result<SendableRecordBatchStream, BoxedError>;
672
673    /// Checks whether this scanner can provide a raw-series scan for the given labels.
674    ///
675    /// This is a non-consuming preflight: it must not start a scan or otherwise consume
676    /// scanner state. It checks immutable label binding, schema, and typed residual-predicate
677    /// eligibility. Returning `Some` requires overriding `scan_series_partition` and
678    /// promises that it can execute with this label source. `None` means the raw path is
679    /// unsupported and an ordinary scan remains available. Execution must revalidate and
680    /// return errors rather than falling back if the binding or schema changed since preflight.
681    fn raw_series_schema(
682        &self,
683        _label_source: &RawLabelSource,
684    ) -> Result<Option<Arc<SeriesSchema>>, BoxedError> {
685        Ok(None)
686    }
687
688    /// Whether raw chunks carry complete explicit labels under this scan binding.
689    /// This non-consuming declaration must agree with the stream before its first poll.
690    /// Returning true does not replace the immutable source binding or authorization checks.
691    fn raw_series_has_explicit_labels(&self, _labels: &RawLabelSource) -> Result<bool, BoxedError> {
692        Ok(false)
693    }
694
695    /// Returns the guarantee for subsequent raw scans without consuming scanner state.
696    ///
697    /// Legacy raw sources have no ordering declaration, so `Contiguous` is the conservative
698    /// default; `Interleaved` is not representable on the raw frame wire.
699    fn raw_series_ordering(&self) -> Result<SeriesOrdering, BoxedError> {
700        Ok(SeriesOrdering::Contiguous)
701    }
702
703    /// Scans a partition as raw series when this source supports that path.
704    ///
705    /// `None` means raw capability is absent before execution and permits a legacy scan instead.
706    /// `Some` selects this entrypoint, including for an empty successful stream. Errors propagate;
707    /// callers must not replay a failed raw scan through the legacy path. Use only one entrypoint
708    /// for a given scanner partition per execution. The explicit label source is owned by the
709    /// returned stream and must be bound to this scanner's metadata snapshot.
710    ///
711    /// Output preserves the ordinary storage reader's pushdown and MVCC semantics. Inexact
712    /// pushed-down predicates can leave false-positive candidates, just as in `scan_partition`.
713    /// Implementations must preserve typed residual-predicate semantics before any lossy value
714    /// conversion, or return `None` before execution when they cannot. Callers must retain
715    /// residual predicate columns until those predicates are evaluated and apply the final
716    /// projection afterward; raw capability does not mean filtering is complete.
717    ///
718    /// # Panics
719    ///
720    /// The default implementation panics. Implementations must override this method to support
721    /// raw series scanning.
722    fn scan_series_partition(
723        &self,
724        _ctx: &QueryScanContext,
725        _metrics_set: &ExecutionPlanMetricsSet,
726        _partition: usize,
727        _label_source: &RawLabelSource,
728    ) -> Result<Option<SendableRawSeriesStream>, BoxedError> {
729        todo!()
730    }
731
732    /// Prepares scanner-wide candidates once for independent ordinal-range reads.
733    ///
734    /// Metadata I/O is asynchronous; sample payloads are loaded only by polling readers.
735    /// The owned future does not borrow the scanner. `None` means unsupported before
736    /// execution; errors must propagate without replay through a different scan path.
737    /// Implementations may share preparation with their raw partition adapter.
738    fn prepare_series(
739        &self,
740        _ctx: &QueryScanContext,
741        _metrics_set: &ExecutionPlanMetricsSet,
742        _label_source: &RawLabelSource,
743    ) -> futures::future::BoxFuture<
744        'static,
745        Result<Option<Arc<dyn PreparedSeriesSource>>, BoxedError>,
746    > {
747        Box::pin(async { Ok(None) })
748    }
749
750    /// Check if there is any predicate exclude region partition exprs that may be executed in this scanner.
751    fn has_predicate_without_region(&self) -> bool;
752
753    /// Add the given dynamic filter expressions to the predicate of the scanner.
754    /// Returns a vector of booleans indicating which filter expressions were applied.
755    /// true indicates the filter expression was applied(will be use by scanner to prune by stat for row group),
756    /// false otherwise.
757    fn add_dyn_filter_to_predicate(
758        &mut self,
759        filter_exprs: Vec<Arc<dyn PhysicalExpr>>,
760    ) -> Vec<bool>;
761
762    /// Resets execution state after streams from the previous execution have been dropped.
763    ///
764    /// Callers must not concurrently execute, reset, or install filters through scanner aliases.
765    /// Implementations retaining dynamic predicates must discard them and derived pruning state,
766    /// while preserving static predicates, the source snapshot, and prepared properties.
767    /// The default is a no-op and does not make inherently one-shot sources replayable.
768    fn reset_state(&mut self) {}
769
770    /// Sets whether the scanner reads one logical metric region, including a
771    /// physical metric region constrained by a table-ID filter.
772    fn set_logical_region(&mut self, logical_region: bool);
773
774    /// Sets the region id that should receive query-load metrics.
775    fn set_query_load_region_id(&mut self, region_id: RegionId);
776
777    fn snapshot_sequence(&self) -> Option<SequenceNumber> {
778        None
779    }

1389    /// Handles batch open region requests.
1390    async fn handle_batch_open_requests(
1391        &self,
1392        parallelism: usize,
1393        requests: Vec<(RegionId, RegionOpenRequest)>,
1394    ) -> Result<BatchResponses, BoxedError> {
1395        let semaphore = Arc::new(Semaphore::new(parallelism));
1396        let mut tasks = Vec::with_capacity(requests.len());
1397
1398        for (region_id, request) in requests {
1399            let semaphore_moved = semaphore.clone();
1400
1401            tasks.push(async move {
1402                // Safety: semaphore must exist
1403                let _permit = semaphore_moved.acquire().await.unwrap();
1404                let result = self
1405                    .handle_request(region_id, RegionRequest::Open(request))
1406                    .await;
1407                (region_id, result)
1408            });
1409        }
1410
1411        Ok(join_all(tasks).await)
1412    }
1413
1414    async fn handle_batch_catchup_requests(
1415        &self,
1416        parallelism: usize,
1417        requests: Vec<(RegionId, RegionCatchupRequest)>,
1418    ) -> Result<BatchResponses, BoxedError> {
1419        let semaphore = Arc::new(Semaphore::new(parallelism));
1420        let mut tasks = Vec::with_capacity(requests.len());
1421
1422        for (region_id, request) in requests {
1423            let semaphore_moved = semaphore.clone();
1424
1425            tasks.push(async move {
1426                // Safety: semaphore must exist
1427                let _permit = semaphore_moved.acquire().await.unwrap();
1428                let result = self

1430                    .await;
1431                (region_id, result)
1432            });
1433        }
1434

1436    }

1438    /// Handles the puts of one request into regions of this engine and returns the responses
1439    /// to await on their own, each with the region of every put it answers for: two puts to one
1440    /// region are two entries.
1441    ///
1442    /// The default handles each put with [`handle_request`](Self::handle_request) and answers
1443    /// each put on its own. An engine that writes several puts together may answer them with
1444    /// one response whose affected rows count all of them.
1445    fn handle_batch_put_requests(
1446        self: Arc<Self>,
1447        requests: Vec<(RegionId, RegionPutRequest)>,
1448    ) -> Vec<BatchPutResponse>
1449    where
1450        Self: 'static,
1451    {
1452        requests
1453            .into_iter()
1454            .map(|(region_id, request)| {
1455                let engine = self.clone();
1456                let response = async move {
1457                    engine
1458                        .handle_request(region_id, RegionRequest::Put(request))
1459                        .await
1460                }
1461                .boxed();
1462                (vec![region_id], response)
1463            })
1464            .collect()
1465    }
1466
1467    async fn handle_batch_ddl_requests(
1468        &self,
1469        request: BatchRegionDdlRequest,
1470    ) -> Result<RegionResponse, BoxedError> {
1471        let requests = request.into_region_requests();
1472
1473        let mut affected_rows = 0;
1474        let mut extensions = HashMap::new();
1475
1476        for (region_id, request) in requests {
1477            let result = self.handle_request(region_id, request).await?;
1478            affected_rows += result.affected_rows;
1479            extensions.extend(result.extensions);
1480        }
1481
1482        Ok(RegionResponse {
1483            affected_rows,
1484            extensions,
1485            metadata: Vec::new(),
1486        })
1487    }
