76/// A live region-level compaction cycle. Completed cycles are not retained.
77///
78/// String fields are shared so cloning a snapshot does not copy parameter payloads.
79/// Time ranges are envelopes and may contain holes. SQL endpoints are NULL when
80/// unknown, unbounded, or outside the representable nanosecond timestamp range.
81#[derive(Debug, Clone)]
82pub struct CompactionTaskEntry {
83    /// UUID retained from acceptance until this cycle leaves the scheduler.
84    pub task_id: Arc<str>,
85    /// Physical region whose scheduler owns this cycle.
86    pub region_id: RegionId,
87    /// Physical table ID derived from the region ID.
88    pub table_id: TableId,
89    /// Reporting datanode ID (0 in standalone), not the remote executor ID.
90    pub node_id: Option<u64>,
91    /// Whether this cycle was triggered automatically or manually.
92    pub trigger_type: Arc<str>,
93    /// Queued, running, or cancelling, aggregated across the cycle.
94    pub state: Arc<str>,
95    /// Phase label: `waiting`, `planning`, `executing`, `committing`, or `remote`.
96    pub phase: Arc<str>,
97    /// Acceptance time in milliseconds since the Unix epoch.
98    pub submitted_at: i64,
99    /// Monotonic elapsed time since acceptance, including waiting.
100    pub elapsed_millis: u64,
101    /// Distinct plan input files, including expired SSTs; unknown before plan acceptance.
102    pub input_files: Option<u64>,
103    /// Total size of plan inputs, including expired SSTs; not bytes processed.
104    pub input_bytes: Option<u64>,
105    /// Number of local atomic units; unavailable for remote execution.
106    pub total_units: Option<u64>,
107    /// Local units with a terminal result, including failures.
108    pub completed_units: Option<u64>,
109    /// JSON snapshot of request arguments, including requested parallelism.
110    pub request_options: Arc<str>,
111    /// JSON snapshot of resolved settings and picker output, if available.
112    pub effective_options: Option<Arc<str>>,
113    /// Inclusive earliest timestamp in plan input SSTs, including expired SSTs.
114    pub input_time_start: Option<Timestamp>,
115    /// Inclusive latest timestamp in plan input SSTs, including expired SSTs.
116    pub input_time_end: Option<Timestamp>,
117    /// Inclusive start of the planned output envelope, not actual output SST bounds.
118    pub output_time_start: Option<Timestamp>,
119    /// Exclusive end of the planned output envelope, not actual output SST bounds.
120    pub output_time_end: Option<Timestamp>,
121}

123impl CompactionTaskEntry {
124    /// Returns the public inspection schema.
125    pub fn schema() -> SchemaRef {
126        use datatypes::prelude::ConcreteDataType as Ty;
127        Arc::new(Schema::new(vec![
128            ColumnSchema::new(TASK_ID, Ty::string_datatype(), false),
129            ColumnSchema::new(REGION_ID, Ty::uint64_datatype(), false),
130            ColumnSchema::new(TABLE_ID, Ty::uint32_datatype(), false),
131            ColumnSchema::new(NODE_ID, Ty::uint64_datatype(), true),
132            ColumnSchema::new(TRIGGER_TYPE, Ty::string_datatype(), false),
133            ColumnSchema::new(STATE, Ty::string_datatype(), false),
134            ColumnSchema::new(PHASE, Ty::string_datatype(), false),
135            ColumnSchema::new(SUBMITTED_AT, Ty::timestamp_millisecond_datatype(), false),
136            ColumnSchema::new(ELAPSED_MILLIS, Ty::uint64_datatype(), false),
137            ColumnSchema::new(INPUT_FILES, Ty::uint64_datatype(), true),
138            ColumnSchema::new(INPUT_BYTES, Ty::uint64_datatype(), true),
139            ColumnSchema::new(TOTAL_UNITS, Ty::uint64_datatype(), true),
140            ColumnSchema::new(COMPLETED_UNITS, Ty::uint64_datatype(), true),
141            ColumnSchema::new(REQUEST_OPTIONS, Ty::string_datatype(), false),
142            ColumnSchema::new(EFFECTIVE_OPTIONS, Ty::string_datatype(), true),
143            ColumnSchema::new(INPUT_TIME_START, Ty::timestamp_nanosecond_datatype(), true),
144            ColumnSchema::new(INPUT_TIME_END, Ty::timestamp_nanosecond_datatype(), true),
145            ColumnSchema::new(OUTPUT_TIME_START, Ty::timestamp_nanosecond_datatype(), true),
146            ColumnSchema::new(OUTPUT_TIME_END, Ty::timestamp_nanosecond_datatype(), true),
147        ]))
148    }
149
150    /// Prunes using known values, leaving unsupported predicates to DataFusion.
151    pub fn matches(&self, filters: &MetadataFilter) -> bool {
152        filters.matches(|name| {
153            Some(match name {
154                TASK_ID => ScalarValue::Utf8(Some(self.task_id.to_string())),
155                REGION_ID => ScalarValue::UInt64(Some(self.region_id.as_u64())),
156                TABLE_ID => ScalarValue::UInt32(Some(self.table_id)),
157                NODE_ID => ScalarValue::UInt64(self.node_id),
158                TRIGGER_TYPE => ScalarValue::Utf8(Some(self.trigger_type.to_string())),
159                STATE => ScalarValue::Utf8(Some(self.state.to_string())),
160                PHASE => ScalarValue::Utf8(Some(self.phase.to_string())),
161                SUBMITTED_AT => ScalarValue::TimestampMillisecond(Some(self.submitted_at), None),
162                ELAPSED_MILLIS => ScalarValue::UInt64(Some(self.elapsed_millis)),
163                INPUT_FILES => ScalarValue::UInt64(self.input_files),
164                INPUT_BYTES => ScalarValue::UInt64(self.input_bytes),
165                TOTAL_UNITS => ScalarValue::UInt64(self.total_units),
166                COMPLETED_UNITS => ScalarValue::UInt64(self.completed_units),
167                REQUEST_OPTIONS => ScalarValue::Utf8(Some(self.request_options.to_string())),
168                EFFECTIVE_OPTIONS => {
169                    ScalarValue::Utf8(self.effective_options.as_deref().map(str::to_string))
170                }
171                INPUT_TIME_START => {
172                    ScalarValue::TimestampNanosecond(timestamp_nanos(self.input_time_start), None)
173                }
174                INPUT_TIME_END => {
175                    ScalarValue::TimestampNanosecond(timestamp_nanos(self.input_time_end), None)
176                }
177                OUTPUT_TIME_START => {
178                    ScalarValue::TimestampNanosecond(timestamp_nanos(self.output_time_start), None)
179                }
180                OUTPUT_TIME_END => {
181                    ScalarValue::TimestampNanosecond(timestamp_nanos(self.output_time_end), None)
182                }
183                _ => return None,
184            })
185        })
186    }
187
188    /// Converts compaction task entries to a record batch.
189    pub fn to_record_batch(entries: &[Self]) -> Result<DfRecordBatch, ArrowError> {
190        Self::to_projected_record_batch(entries, None)
191    }
192
193    /// Builds only the requested columns, including empty and repeated projections.
194    fn to_projected_record_batch(
195        entries: &[Self],
196        projection: Option<&[usize]>,
197    ) -> Result<DfRecordBatch, ArrowError> {
198        let schema = Self::schema().arrow_schema().clone();
199        let schema = match projection {
200            Some(projection) => Arc::new(schema.project(projection)?),
201            None => schema,
202        };
203        CompactionTaskBatchBuilder::new(schema).build(entries)
204    }
205
206    /// Internal name, inaccessible through user table resolution.
207    pub fn reserved_table_name_for_inspection() -> &'static str {
208        "__inspect/__mito/__compaction_tasks"
209    }
210
211    /// Builds a remote scan with filtering before projection and limit.
212    pub fn build_plan(request: ScanRequest) -> Result<LogicalPlan, DataFusionError> {
213        inspect::build_plan(
214            request,
215            Self::reserved_table_name_for_inspection(),
216            Self::schema(),
217        )
218    }

221/// Converts compaction task batches using a schema prepared once per scan.
222/// Repeated projected columns share the same Arrow array.
223pub struct CompactionTaskBatchBuilder {
224    projected: ProjectedBatchBuilder,
225}
226
227impl CompactionTaskBatchBuilder {
228    /// Creates a builder for a projected compaction task schema.
229    pub fn new(schema: ArrowSchemaRef) -> Self {
230        Self {
231            projected: ProjectedBatchBuilder::new(schema),
232        }
233    }
234
235    /// Builds the projected columns, copying string contents only into Arrow buffers.
236    pub fn build(&self, entries: &[CompactionTaskEntry]) -> Result<DfRecordBatch, ArrowError> {
237        self.projected.build(entries.len(), |name| {
238            let array: ArrayRef = match name {
239                TASK_ID => Arc::new(StringArray::from_iter_values(
240                    entries.iter().map(|e| e.task_id.as_ref()),
241                )),
242                REGION_ID => Arc::new(UInt64Array::from_iter_values(
243                    entries.iter().map(|e| e.region_id.as_u64()),
244                )),
245                TABLE_ID => Arc::new(UInt32Array::from_iter_values(
246                    entries.iter().map(|e| e.table_id),
247                )),
248                NODE_ID => Arc::new(UInt64Array::from_iter(entries.iter().map(|e| e.node_id))),
249                TRIGGER_TYPE => Arc::new(StringArray::from_iter_values(
250                    entries.iter().map(|e| e.trigger_type.as_ref()),
251                )),
252                STATE => Arc::new(StringArray::from_iter_values(
253                    entries.iter().map(|e| e.state.as_ref()),
254                )),
255                PHASE => Arc::new(StringArray::from_iter_values(
256                    entries.iter().map(|e| e.phase.as_ref()),
257                )),
258                SUBMITTED_AT => Arc::new(TimestampMillisecondArray::from_iter_values(
259                    entries.iter().map(|e| e.submitted_at),
260                )),
261                ELAPSED_MILLIS => Arc::new(UInt64Array::from_iter_values(
262                    entries.iter().map(|e| e.elapsed_millis),
263                )),
264                INPUT_FILES => Arc::new(UInt64Array::from_iter(
265                    entries.iter().map(|e| e.input_files),
266                )),
267                INPUT_BYTES => Arc::new(UInt64Array::from_iter(
268                    entries.iter().map(|e| e.input_bytes),
269                )),
270                TOTAL_UNITS => Arc::new(UInt64Array::from_iter(
271                    entries.iter().map(|e| e.total_units),
272                )),
273                COMPLETED_UNITS => Arc::new(UInt64Array::from_iter(
274                    entries.iter().map(|e| e.completed_units),
275                )),
276                REQUEST_OPTIONS => Arc::new(StringArray::from_iter_values(
277                    entries.iter().map(|e| e.request_options.as_ref()),
278                )),
279                EFFECTIVE_OPTIONS => Arc::new(StringArray::from_iter(
280                    entries.iter().map(|e| e.effective_options.as_deref()),
281                )),
282                INPUT_TIME_START => Arc::new(TimestampNanosecondArray::from_iter(
283                    entries.iter().map(|e| timestamp_nanos(e.input_time_start)),
284                )),
285                INPUT_TIME_END => Arc::new(TimestampNanosecondArray::from_iter(
286                    entries.iter().map(|e| timestamp_nanos(e.input_time_end)),
287                )),
288                OUTPUT_TIME_START => Arc::new(TimestampNanosecondArray::from_iter(
289                    entries.iter().map(|e| timestamp_nanos(e.output_time_start)),
290                )),
291                OUTPUT_TIME_END => Arc::new(TimestampNanosecondArray::from_iter(
292                    entries.iter().map(|e| timestamp_nanos(e.output_time_end)),
293                )),
294                _ => {
295                    return Err(ArrowError::SchemaError(format!(
296                        "Unknown compaction task column: {name}"
297                    )));
298                }
299            };
300            Ok(array)
301        })
302    }
303}

305/// Converts endpoints identically for predicate pruning and Arrow output.
306fn timestamp_nanos(timestamp: Option<Timestamp>) -> Option<i64> {
307    timestamp
308        .and_then(|ts| ts.convert_to(TimeUnit::Nanosecond))
309        .map(|ts| ts.value())
310}

564        .into_iter()
565        .enumerate()
566        {
567            let values = batch
568                .column(index)
569                .as_any()
570                .downcast_ref::<TimestampNanosecondArray>()

573                values,
574                &TimestampNanosecondArray::from(vec![Some(nanos), None, None])
575            );
576            let equal = col(name).eq(lit(ScalarValue::TimestampNanosecond(Some(nanos), None)));
577            let different =
578                col(name).eq(lit(ScalarValue::TimestampNanosecond(Some(nanos + 1), None)));
579            assert!(entries[0].matches(&MetadataFilter::new(std::slice::from_ref(&equal))));
580            assert!(!entries[0].matches(&MetadataFilter::new(&[different])));
581            assert!(!entries[0].matches(&MetadataFilter::new(&[col(name).is_null()])));
582            for entry in &entries[1..] {
583                assert!(entry.matches(&MetadataFilter::new(&[col(name).is_null()])));
584                assert!(!entry.matches(&MetadataFilter::new(&[col(name).is_not_null()])));
585                assert!(!entry.matches(&MetadataFilter::new(std::slice::from_ref(&equal))));
586            }
587        }
588        let full = CompactionTaskEntry::to_record_batch(&entries).unwrap();
589        let projection = [18, 16, 15, 18, 17];
590        assert_eq!(
591            CompactionTaskEntry::to_projected_record_batch(&entries, Some(&projection)).unwrap(),
592            full.project(&projection).unwrap()
593        );
594    }
