175) -> bool {
176    region_write_count > 1
177}

179pub(in crate::batcher::logical_table) async fn flush_region_writes_concurrently(
180    node_manager: &(impl PhysicalFlushNodeRequester + ?Sized),
181    writes: Vec<FlushRegionWrite>,
182) -> Result<usize> {
183    let mut affected_rows = 0;
184    if !should_dispatch_concurrently(writes.len()) {
185        for write in writes {
186            let _timer = PENDING_ROWS_BATCH_FLUSH_STAGE_ELAPSED
187                .with_label_values(&["flush_write_region"])
188                .start_timer();
189            affected_rows += node_manager
190                .handle(&write.datanode, write.request)
191                .await?
192                .affected_rows;
193        }
194        return Ok(affected_rows);
195    }
196
197    let write_futures = writes.into_iter().map(|write| async move {
198        let _timer = PENDING_ROWS_BATCH_FLUSH_STAGE_ELAPSED
199            .with_label_values(&["flush_write_region"])
200            .start_timer();
201
202        let response = node_manager.handle(&write.datanode, write.request).await?;
203        Ok::<_, Error>(response.affected_rows)
204    });
205
206    // todo(hl): should be bounded.
207    let affected_rows = futures::future::try_join_all(write_futures)
208        .await?
209        .into_iter()
210        .sum();
211    Ok(affected_rows)
212}

226    combined_batch: RecordBatch,

235pub(in crate::batcher::logical_table) fn plan_region_batch(
236    stripped_batch: &RecordBatch,
237    physical_table_id: TableId,
238    region_number: u32,
239    mask: &partition::partition::RegionMask,
240) -> Result<Option<PlannedRegionBatch>> {
241    if mask.select_none() {
242        return Ok(None);
243    }
244
245    let region_batch = if mask.select_all() {
246        stripped_batch.clone()
247    } else {
248        let _timer = PENDING_ROWS_BATCH_FLUSH_STAGE_ELAPSED
249            .with_label_values(&["flush_physical_filter_record_batch"])
250            .start_timer();
251        filter_record_batch(stripped_batch, mask.array()).context(error::ArrowSnafu)?
252    };
253
254    let row_count = region_batch.num_rows();
255    if row_count == 0 {
256        return Ok(None);
257    }
258
259    Ok(Some(PlannedRegionBatch {
260        region_id: RegionId::new(physical_table_id, region_number),
261        batch: region_batch,
262    }))
263}

265pub(in crate::batcher::logical_table) fn plan_region_batches(
266    combined_batch: RecordBatch,
267    physical_table_id: TableId,
268    partition_rule: &dyn partition::partition::PartitionRule,
269    partition_columns: &[String],
270) -> Result<Vec<PlannedRegionBatch>> {
271    let region_masks = split_combined_batch_by_region(&combined_batch, partition_rule)?;
272    let stripped_batch = prepare_physical_region_routing_batch(combined_batch, partition_columns)?;
273
274    let mut planned_batches = Vec::new();
275    for (region_number, mask) in region_masks {
276        if let Some(planned_batch) =
277            plan_region_batch(&stripped_batch, physical_table_id, region_number, &mask)?
278        {
279            planned_batches.push(planned_batch);
280        }
281    }
282
283    Ok(planned_batches)
284}

286pub(in crate::batcher::logical_table) async fn resolve_region_targets(
287    planned_batches: Vec<PlannedRegionBatch>,
288    partition_manager: &(impl PhysicalFlushPartitionProvider + ?Sized),
289) -> Result<Vec<ResolvedRegionBatch>> {
290    let mut resolved_batches = Vec::with_capacity(planned_batches.len());
291    for planned in planned_batches {
292        let datanode = {
293            let _timer = PENDING_ROWS_BATCH_FLUSH_STAGE_ELAPSED
294                .with_label_values(&["flush_physical_resolve_region_leader"])
295                .start_timer();
296            partition_manager
297                .find_region_leader(planned.region_id)
298                .await?
299        };
300
301        resolved_batches.push(ResolvedRegionBatch { planned, datanode });
302    }
303
304    Ok(resolved_batches)
305}

307pub(in crate::batcher::logical_table) fn encode_region_write_requests(
308    resolved_batches: Vec<ResolvedRegionBatch>,
309    skip_wal: bool,
310) -> Result<Vec<FlushRegionWrite>> {
311    let mut region_writes = Vec::with_capacity(resolved_batches.len());
312    for resolved in resolved_batches {
313        let region_id = resolved.planned.region_id;
314        let (schema_bytes, data_header, payload) = {
315            let _timer = PENDING_ROWS_BATCH_FLUSH_STAGE_ELAPSED
316                .with_label_values(&["flush_physical_encode_ipc"])
317                .start_timer();
318            record_batch_to_ipc(resolved.planned.batch).map_err(map_ipc_error)?
319        };
320
321        let request = RegionRequest {
322            header: Some(RegionRequestHeader {
323                tracing_context: TracingContext::from_current_span().to_w3c(),
324                ..Default::default()
325            }),
326            body: Some(region_request::Body::BulkInsert(BulkInsertRequest {
327                skip_wal,
328                region_id: region_id.as_u64(),
329                partition_expr_version: None,
330                // Set aligned_schema_version to None so that datanode will check the batch schema again to see if any
331                // column is missing.
332                aligned_schema_version: None,
333                body: Some(bulk_insert_request::Body::ArrowIpc(ArrowIpc {
334                    schema: schema_bytes,
335                    data_header,
336                    payload,
337                })),
338            })),
339        };
340
341        region_writes.push(FlushRegionWrite {
342            datanode: resolved.datanode,
343            request,
344        });
345    }
346
347    Ok(region_writes)
348}

350pub(in crate::batcher::logical_table) fn map_ipc_error(error: GrpcError) -> Error {
351    match error {
352        GrpcError::NotSupported { feat } => Error::NotSupported { feat },

354        error => Error::Internal {
355            err_msg: error.to_string(),
356        },

531    impl Datanode for ConcurrentMockDatanode {
532        async fn handle(&self, _request: RegionRequest) -> MetaResult<RegionResponse> {
533            let now = self.inflight.fetch_add(1, Ordering::SeqCst) + 1;
534            loop {
535                let max = self.max_inflight.load(Ordering::SeqCst);
536                if now <= max {
537                    break;
538                }
539                if self
540                    .max_inflight
541                    .compare_exchange(max, now, Ordering::SeqCst, Ordering::SeqCst)
542                    .is_ok()
543                {
544                    break;
545                }
546            }
547
548            sleep(self.delay).await;
549            self.inflight.fetch_sub(1, Ordering::SeqCst);
550            Ok(RegionResponse::new(0))
551        }

554            &self,
555            _request: QueryRequest,
556        ) -> MetaResult<SendableRecordBatchStream> {
557            unimplemented!()
558        }
