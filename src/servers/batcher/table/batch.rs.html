32/// A detached batch owns every submission until its completion is reported.
33pub(in crate::batcher::table) struct Batch {
34    pub submissions: Vec<PendingBatch>,
35    pub total_rows: usize,
36}
37
38pub(in crate::batcher::table) async fn flush_batch(
39    batch: Batch,
40    inserter: Arc<Inserter>,
41    notifier: FlowNotifier,
42) {
43    let started = Instant::now();
44    let result = send_batch(&batch, &inserter).await;
45    FLUSH_ELAPSED.observe(started.elapsed().as_secs_f64());
46    match result {
47        Ok((table, combined)) => {
48            FLUSH_TOTAL.inc();
49            FLUSH_ROWS.observe(batch.total_rows as f64);
50            DIST_INGEST_ROW_COUNT
51                .with_label_values(&[batch.submissions[0].ctx.get_db_string().as_str()])
52                .inc_by(batch.total_rows as u64);
53            notify_batches(batch.submissions, Ok(()));
54            // Admission is best-effort and only occurs after affected-row validation.
55            notifier.notify(table, &combined);
56        }
57        Err(error) => {
58            FLUSH_FAILURES.inc();
59            FLUSH_DROPPED_ROWS.inc_by(batch.total_rows as u64);
60            notify_batches(batch.submissions, Err(Arc::new(error)));
61        }
62    }
63}
64
65async fn send_batch(batch: &Batch, inserter: &Inserter) -> Result<(TableInfoRef, RecordBatch)> {
66    let first = batch.submissions.first().ok_or_else(|| {
67        UnexpectedSnafu {
68            violated: "cannot execute an empty flush".to_string(),
69        }
70        .build()
71    })?;
72    let combined = combine_batches(batch.submissions.iter().map(|pending| &pending.batch))?;
73    let affected_rows = inserter
74        .flush_bulk_batch(
75            first.table_info.clone(),
76            combined.clone(),
77            first.ctx.clone(),
78        )
79        .await?;
80    validate_affected_rows(batch.total_rows, affected_rows)?;
81    Ok((first.table_info.clone(), combined))
82}
83
84fn combine_batches<'a>(batches: impl IntoIterator<Item = &'a RecordBatch>) -> Result<RecordBatch> {
85    let batches = batches.into_iter().collect::<Vec<_>>();
86    let Some(first) = batches.first() else {
87        return UnexpectedSnafu {
88            violated: "cannot combine an empty flush".to_string(),
89        }
90        .fail();
91    };
92    let schema = first.schema();
93    if batches.iter().any(|batch| batch.schema() != schema) {
94        return UnexpectedSnafu {
95            violated: "cannot combine different Arrow schemas".to_string(),
96        }
97        .fail();
98    }
99    concat_batches(&schema, batches).context(ComputeArrowSnafu)
100}
101
102fn validate_affected_rows(expected: usize, actual: usize) -> Result<()> {
103    if expected != actual {
104        return UnexpectedSnafu {
105            violated: format!("batched write affected {actual} rows, expected {expected}; individual results cannot be attributed"),
106        }.fail();
107    }
108    Ok(())
109}

111#[cfg(test)]
112mod tests {
113    use arrow::array::Int32Array;
114    use arrow::datatypes::{DataType, Field, Schema};
115
116    use crate::batcher::table::batch::*;
117
118    fn batch(name: &str, values: Vec<i32>) -> RecordBatch {
119        RecordBatch::try_new(
120            Arc::new(Schema::new(vec![Field::new(name, DataType::Int32, false)])),
121            vec![Arc::new(Int32Array::from(values))],
122        )
123        .unwrap()
124    }
125
126    #[test]
127    fn test_combine_preserves_rows_and_rejects_schema_changes() {
128        let first = batch("a", vec![1, 2]);
129        let second = batch("a", vec![3]);
130        assert_eq!(
131            batch("a", vec![1, 2, 3]),
132            combine_batches([&first, &second]).unwrap()
133        );
134        let different = batch("b", vec![4]);
135        assert!(combine_batches([&first, &different]).is_err());
136        let schema = first.schema();
137        let metadata = Schema::new_with_metadata(
138            schema.fields().clone(),
139            [("version".to_string(), "2".to_string())]
140                .into_iter()
141                .collect(),
142        );
143        let changed = RecordBatch::try_new(Arc::new(metadata), first.columns().to_vec()).unwrap();
144        assert!(combine_batches([&first, &changed]).is_err());
145    }
146
147    #[test]
148    fn test_affected_rows_must_match_before_attribution() {
149        for (expected, actual, valid) in [(0, 0, true), (3, 3, true), (3, 2, false), (3, 4, false)]
150        {
151            assert_eq!(valid, validate_affected_rows(expected, actual).is_ok());
152        }
153    }
