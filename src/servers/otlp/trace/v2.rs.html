33// Preallocate for the fixed v2 schema: 3 timing columns, 3 span/trace IDs,
34// 2 kind/name columns, 2 status columns, 1 trace state, 2 scope name/version
35// columns, 1 service name, 3 JSON2 attribute columns, and 2 JSON event/link columns.
36const APPROXIMATE_COLUMN_COUNT: usize = 19;
37
38/// Converts trace spans into row insert requests for the main v2 trace table.
39pub(super) fn v2_to_grpc_main_insert_requests(
40    spans: &[TraceSpan],
41    table_name: &str,
42) -> Result<(RowInsertRequests, usize)> {
43    let mut tables = MultiTableData::default();
44    tables.add_table_data(table_name, build_trace_table_data(spans)?);
45    Ok(tables.into_row_insert_requests())
46}
47
48/// Builds the fixed row-oriented payload for the main v2 trace table.
49fn build_trace_table_data(spans: &[TraceSpan]) -> Result<TableData> {
50    let mut writer = TableData::new(APPROXIMATE_COLUMN_COUNT, spans.len());
51    for span in spans.iter().cloned() {
52        write_span_to_row(&mut writer, span)?;
53    }
54    Ok(writer)
55}
56
57fn write_span_to_row(writer: &mut TableData, span: TraceSpan) -> Result<()> {
58    ensure!(
59        span.start_in_nanosecond <= i64::MAX as u64,
60        TimestampOverflowSnafu {
61            error: "`span.start_in_nanosecond`",
62        }
63    );
64    ensure!(
65        span.end_in_nanosecond <= i64::MAX as u64,
66        TimestampOverflowSnafu {
67            error: "`span.end_in_nanosecond`",
68        }
69    );
70    let mut row = writer.alloc_one_row();
71
72    row_writer::write_ts_to_nanos(
73        writer,
74        TIMESTAMP_COLUMN,
75        Some(span.start_in_nanosecond as i64),
76        Precision::Nanosecond,
77        &mut row,
78    )?;
79    row_writer::write_fields(
80        writer,
81        vec![
82            make_column_data(
83                TIMESTAMP_END_COLUMN,
84                ColumnDataType::TimestampNanosecond,
85                Some(ValueData::TimestampNanosecondValue(
86                    span.end_in_nanosecond as i64,
87                )),
88            ),
89            make_column_data(
90                DURATION_NANO_COLUMN,
91                ColumnDataType::Int64,
92                Some(ValueData::I64Value(span_duration_nano(&span))),
93            ),
94            make_string_column_data(PARENT_SPAN_ID_COLUMN, span.parent_span_id),
95            make_string_column_data(TRACE_ID_COLUMN, Some(span.trace_id)),
96            make_string_column_data(SPAN_ID_COLUMN, Some(span.span_id)),
97            make_string_column_data(SPAN_KIND_COLUMN, Some(span.span_kind)),
98            make_string_column_data(SPAN_NAME_COLUMN, Some(span.span_name)),
99            make_string_column_data(SPAN_STATUS_CODE, Some(span.span_status_code)),
100            make_string_column_data(SPAN_STATUS_MESSAGE_COLUMN, Some(span.span_status_message)),
101            make_string_column_data(TRACE_STATE_COLUMN, Some(span.trace_state)),
102            make_string_column_data(SCOPE_NAME_COLUMN, Some(span.scope_name)),
103            make_string_column_data(SCOPE_VERSION_COLUMN, Some(span.scope_version)),
104        ]
105        .into_iter(),
106        &mut row,
107    )?;
108    row_writer::write_by_schema(
109        writer,
110        std::iter::once((
111            ColumnSchema {
112                column_name: SERVICE_NAME_COLUMN.to_string(),
113                datatype: ColumnDataType::String as i32,
114                semantic_type: SemanticType::Tag as i32,
115                ..Default::default()
116            },
117            span.service_name.map(ValueData::StringValue),
118        )),
119        &mut row,
120    )?;
121
122    row_writer::write_json2(
123        writer,
124        SPAN_ATTRIBUTES_COLUMN,
125        span.span_attributes,
126        &mut row,
127    )?;
128    row_writer::write_json2(
129        writer,
130        SCOPE_ATTRIBUTES_COLUMN,
131        span.scope_attributes,
132        &mut row,
133    )?;
134    row_writer::write_json2(
135        writer,
136        RESOURCE_ATTRIBUTES_COLUMN,
137        span.resource_attributes,
138        &mut row,
139    )?;
140
141    row_writer::write_json(
142        writer,
143        SPAN_EVENTS_COLUMN,
144        span.span_events.into(),
145        &mut row,
146    )?;
147    row_writer::write_json(writer, "span_links", span.span_links.into(), &mut row)?;
148
149    writer.add_row(row);
150    Ok(())
151}
