70struct BorrowedRequest<'a> {
71    symbols: Vec<&'a str>,
72    timeseries: Vec<&'a [u8]>,
73}
74
75impl<'a> BorrowedRequest<'a> {
76    fn decode(mut buf: &'a [u8]) -> std::result::Result<Self, DecodeError> {
77        let mut symbols = Vec::new();
78        let mut timeseries = Vec::new();
79
80        while buf.has_remaining() {
81            let (tag, wire_type) = decode_key(&mut buf)?;
82            match tag {
83                4 => {
84                    let value =
85                        take_length_delimited(wire_type, &mut buf).map_err(|mut error| {
86                            error.push("Request", "symbols");
87                            error
88                        })?;
89                    let symbol = std::str::from_utf8(value).map_err(|_| {
90                        let mut error =
91                            DecodeError::new("invalid string value: data is not UTF-8 encoded");
92                        error.push("Request", "symbols");
93                        error
94                    })?;
95                    symbols.push(symbol);
96                }
97                5 => {
98                    let series =
99                        take_length_delimited(wire_type, &mut buf).map_err(|mut error| {
100                            error.push("Request", "timeseries");
101                            error
102                        })?;
103                    timeseries.push(series);
104                }
105                _ => skip_field(wire_type, tag, &mut buf, DecodeContext::default())?,
106            }
107        }
108
109        Ok(Self {
110            symbols,
111            timeseries,
112        })
113    }
114}

126pub(crate) fn decode_remote_write_v2(
127    is_zstd: bool,
128    body: Bytes,
129    native_histograms_enabled: bool,
130) -> Result<RemoteWriteV2WriteRequests> {
131    let decode_timer = crate::metrics::METRIC_HTTP_PROM_STORE_CODEC_ELAPSED
132        .with_label_values(&["decode", REMOTE_WRITE_V2_VERSION])
133        .start_timer();
134
135    // Match the v1 decoder's VictoriaMetrics fallback: some clients may send a
136    // mismatched content-encoding header, so try the other compression on failure.
137    let buf = if let Ok(buf) = try_decompress(is_zstd, &body[..]) {
138        buf
139    } else {
140        try_decompress(!is_zstd, &body[..])?
141    };
142    let request = BorrowedRequest::decode(&buf).context(error::DecodePromRemoteRequestSnafu)?;
143    drop(decode_timer);
144
145    let _convert_timer = crate::metrics::METRIC_HTTP_PROM_STORE_CODEC_ELAPSED
146        .with_label_values(&["convert", REMOTE_WRITE_V2_VERSION])
147        .start_timer();
148    convert_remote_write_v2(request, native_histograms_enabled)
149}

151fn convert_remote_write_v2(
152    request: BorrowedRequest<'_>,
153    native_histograms_enabled: bool,
154) -> Result<RemoteWriteV2WriteRequests> {
155    ensure!(
156        request.symbols.first().copied() == Some(""),
157        error::InvalidPromRemoteRequestSnafu {
158            msg: "remote write v2 symbols must start with an empty string".to_string(),
159        }
160    );
161
162    let mut sample_tables = HashMap::<PromCtx, HashMap<String, TableData>>::new();
163    let mut histogram_tables = HashMap::<PromCtx, HashMap<String, TableData>>::new();
164    let mut label_names = HashSet::new();
165    let mut sample_count_total = 0;
166    let mut histogram_count_total = 0;
167    let mut labels_refs = Vec::new();
168    let mut metadata = Metadata::default();
169    let mut scratch = LeafScratch::default();
170    let mut semantic_index = SemanticIndexes::default();
171
172    for series in request.timeseries {
173        let counts = scan_series(series, &mut labels_refs, &mut metadata)
174            .context(error::DecodePromRemoteRequestSnafu)?;
175
176        ensure!(
177            native_histograms_enabled || counts.histograms == 0,
178            error::InvalidPromRemoteRequestSnafu {
179                msg: "prometheus remote write v2 native histogram ingestion is experimental; set prom_store.experimental_enable_prometheus_native_histogram = true to enable it"
180                    .to_string(),
181            }
182        );
183
184        if counts.samples == 0 && counts.histograms == 0 {
185            decode_series_leaves(series, None, Vec::new(), 0, &mut scratch)?;
186            continue;
187        }
188
189        let (prom_ctx, table_name, tags) =
190            resolve_series_labels(&request.symbols, &labels_refs, &mut label_names)?;
191        ensure_no_internal_histogram_labels(&tags)?;
192        record_series_metadata(
193            &mut semantic_index,
194            &request.symbols,
195            &metadata,
196            &prom_ctx,
197            &table_name,
198        )?;
199        let has_other_value_type = if counts.samples > 0 {
200            counts.histograms > 0
201                || histogram_tables
202                    .get(&prom_ctx)
203                    .is_some_and(|tables| tables.contains_key(&table_name))
204        } else {
205            sample_tables
206                .get(&prom_ctx)
207                .is_some_and(|tables| tables.contains_key(&table_name))
208        };
209        ensure!(
210            !has_other_value_type,
211            error::InvalidPromRemoteRequestSnafu {
212                msg: format!(
213                    "remote write v2 metric `{table_name}` contains both samples and native histograms"
214                ),
215            }
216        );
217
218        let column_count =
219            tags.len()
220                .checked_add(2)
221                .with_context(|| error::InvalidPromRemoteRequestSnafu {
222                    msg: "remote write v2 series has too many labels".to_string(),
223                })?;
224        let (writer, row_count) = if counts.samples > 0 {
225            (
226                SeriesWriter::Samples(get_or_create_table_data(
227                    &mut sample_tables,
228                    prom_ctx,
229                    table_name,
230                    column_count,
231                    counts.samples,
232                )),
233                counts.samples,
234            )
235        } else {
236            (
237                SeriesWriter::Histograms(get_or_create_table_data(
238                    &mut histogram_tables,
239                    prom_ctx,
240                    table_name,
241                    column_count,
242                    counts.histograms,
243                )),
244                counts.histograms,
245            )
246        };
247
248        decode_series_leaves(series, Some(writer), tags, row_count, &mut scratch)?;
249        sample_count_total = checked_total(sample_count_total, counts.samples, "sample")?;
250        histogram_count_total =
251            checked_total(histogram_count_total, counts.histograms, "histogram")?;
252    }
253
254    Ok(RemoteWriteV2WriteRequests {
255        samples: into_context_req(sample_tables),
256        histograms: into_context_req(histogram_tables),
257        sample_count: sample_count_total,
258        histogram_count: histogram_count_total,
259        semantic_index,
260    })
261}

263/// Stamps the series' inline metadata for the written table's auto-create: an
264/// explicit metric type upgrades the table's metadata quality to `declared`,
265/// `UNSPECIFIED` series keep the request-level `inferred` stamp, and units are
266/// canonicalised from OpenMetrics words to UCUM. Type and unit stamp
267/// independently, as OpenMetrics defines them. Help text is not persisted.
268///
269/// Every non-zero symbol reference is validated up front, independent of what
270/// ends up persisted: the spec requires all references to point into the
271/// symbol table.
272fn record_series_metadata(
273    index: &mut SemanticIndexes,
274    symbols: &[&str],
275    series_metadata: &Metadata,
276    prom_ctx: &PromCtx,
277    table_name: &str,
278) -> Result<()> {
279    // Symbol 0 is the mandatory empty string: no help / no unit.
280    if series_metadata.help_ref != 0 {
281        symbol_ref(symbols, series_metadata.help_ref, "metadata help")?;
282    }
283    let unit = if series_metadata.unit_ref != 0 {
284        Some(symbol_ref(
285            symbols,
286            series_metadata.unit_ref,
287            "metadata unit",
288        )?)
289    } else {
290        None
291    };
292    let metric_type = metric_type_value(series_metadata.r#type);
293    let ucum = unit.and_then(|unit| openmetrics_unit_to_ucum(unit.trim()));
294    if metric_type.is_none() && ucum.is_none() {
295        return Ok(());
296    }
297
298    let index = index.index_for(prom_ctx.schema.as_deref());
299    if let Some(metric_type) = metric_type {
300        index.record_scalar(table_name, SEMANTIC_METRIC_TYPE, metric_type);
301        index.record_scalar(
302            table_name,
303            SEMANTIC_METRIC_METADATA_QUALITY,
304            METADATA_QUALITY_DECLARED,
305        );
306    }
307    if let Some(ucum) = ucum {
308        index.record_scalar(table_name, SEMANTIC_METRIC_UNIT, ucum);
309    }
310    Ok(())
311}

313/// The `greptime.semantic.metric.type` value for a wire metric type; `None`
314/// for `UNSPECIFIED` (nothing was declared) and out-of-range values.
315fn metric_type_value(wire_type: i32) -> Option<&'static str> {
316    match metadata::MetricType::try_from(wire_type).ok()? {
317        metadata::MetricType::Unspecified => None,
318        metadata::MetricType::Counter => Some(METRIC_TYPE_COUNTER),
319        metadata::MetricType::Gauge => Some(METRIC_TYPE_GAUGE),
320        metadata::MetricType::Histogram => Some(METRIC_TYPE_HISTOGRAM),
321        metadata::MetricType::Gaugehistogram => Some(METRIC_TYPE_GAUGE_HISTOGRAM),
322        metadata::MetricType::Summary => Some(METRIC_TYPE_SUMMARY),
323        metadata::MetricType::Info => Some(METRIC_TYPE_INFO),
324        metadata::MetricType::Stateset => Some(METRIC_TYPE_STATESET),
325    }
326}

334fn scan_series(
335    mut buf: &[u8],
336    labels_refs: &mut Vec<u32>,
337    metadata: &mut Metadata,
338) -> std::result::Result<SeriesCounts, DecodeError> {
339    labels_refs.clear();
340    metadata.clear();
341    let mut counts = SeriesCounts::default();
342
343    while buf.has_remaining() {
344        let (tag, wire_type) = decode_key(&mut buf)?;
345        match tag {
346            TIME_SERIES_LABELS_REFS_TAG => {
347                uint32::merge_repeated(wire_type, labels_refs, &mut buf, DecodeContext::default())
348                    .map_err(|mut error| {
349                    error.push("TimeSeries", "labels_refs");
350                    error
351                })?
352            }
353            TIME_SERIES_SAMPLES_TAG => {
354                take_length_delimited(wire_type, &mut buf).map_err(|mut error| {
355                    error.push("TimeSeries", "samples");
356                    error
357                })?;
358                counts.samples = counts.samples.checked_add(1).ok_or_else(|| {
359                    DecodeError::new("remote write v2 sample count overflows usize")
360                })?;
361            }
362            TIME_SERIES_HISTOGRAMS_TAG => {
363                take_length_delimited(wire_type, &mut buf).map_err(|mut error| {
364                    error.push("TimeSeries", "histograms");
365                    error
366                })?;
367                counts.histograms = counts.histograms.checked_add(1).ok_or_else(|| {
368                    DecodeError::new("remote write v2 histogram count overflows usize")
369                })?;
370            }
371            TIME_SERIES_EXEMPLARS_TAG => {
372                take_length_delimited(wire_type, &mut buf)?;
373            }
374            TIME_SERIES_METADATA_TAG => {
375                message::merge(wire_type, metadata, &mut buf, DecodeContext::default()).map_err(
376                    |mut error| {
377                        error.push("TimeSeries", "metadata");
378                        error
379                    },
380                )?
381            }
382            _ => skip_field(wire_type, tag, &mut buf, DecodeContext::default())?,
383        }
384    }
385
386    Ok(counts)
387}

401fn decode_series_leaves(
402    mut buf: &[u8],
403    mut writer: Option<SeriesWriter<'_>>,
404    mut tags: PromTags<'_>,
405    mut rows_remaining: usize,
406    scratch: &mut LeafScratch,
407) -> Result<()> {
408    let mut sample_row_template = None;
409
410    while buf.has_remaining() {
411        let (tag, wire_type) = decode_key(&mut buf).context(error::DecodePromRemoteRequestSnafu)?;
412        match tag {
413            TIME_SERIES_LABELS_REFS_TAG => {
414                skip_field(wire_type, tag, &mut buf, DecodeContext::default())
415                    .context(error::DecodePromRemoteRequestSnafu)?
416            }
417            TIME_SERIES_SAMPLES_TAG => {
418                scratch.sample.clear();
419                message::merge(
420                    wire_type,
421                    &mut scratch.sample,
422                    &mut buf,
423                    DecodeContext::default(),
424                )
425                .map_err(|mut error| {
426                    error.push("TimeSeries", "samples");
427                    error
428                })
429                .context(error::DecodePromRemoteRequestSnafu)?;
430                rows_remaining = rows_remaining.checked_sub(1).with_context(|| {
431                    error::InvalidPromRemoteRequestSnafu {
432                        msg: "remote write v2 sample count changed between scans".to_string(),
433                    }
434                })?;
435                if let Some(SeriesWriter::Samples(table_data)) = &mut writer {
436                    if sample_row_template.is_none() {
437                        let timestamp_index = table_data.ensure_column_by_name(
438                            greptime_timestamp(),
439                            ColumnDataType::TimestampMillisecond,
440                            SemanticType::Timestamp,
441                        )?;
442                        let value_index = table_data.ensure_column_by_name(
443                            greptime_value(),
444                            ColumnDataType::Float64,
445                            SemanticType::Field,
446                        )?;
447                        let mut row = table_data.alloc_one_row();
448                        row_writer::write_tags(
449                            table_data,
450                            std::mem::take(&mut tags).into_iter(),
451                            &mut row,
452                        )?;
453                        sample_row_template = Some((row, timestamp_index, value_index));
454                    }
455
456                    if let Some((row, timestamp_index, value_index)) = sample_row_template.as_mut()
457                    {
458                        row[*timestamp_index].value_data = Some(
459                            ValueData::TimestampMillisecondValue(scratch.sample.timestamp),
460                        );
461                        row[*value_index].value_data =
462                            Some(ValueData::F64Value(scratch.sample.value));
463                        let row = if rows_remaining == 0 {
464                            std::mem::take(row)
465                        } else {
466                            row.clone()
467                        };
468                        table_data.add_row(row);
469                    }
470                }
471            }
472            TIME_SERIES_HISTOGRAMS_TAG => {
473                scratch.histogram.clear();
474                message::merge(
475                    wire_type,
476                    &mut scratch.histogram,
477                    &mut buf,
478                    DecodeContext::default(),
479                )
480                .map_err(|mut error| {
481                    error.push("TimeSeries", "histograms");
482                    error
483                })
484                .context(error::DecodePromRemoteRequestSnafu)?;
485                rows_remaining = rows_remaining.checked_sub(1).with_context(|| {
486                    error::InvalidPromRemoteRequestSnafu {
487                        msg: "remote write v2 histogram count changed between scans".to_string(),
488                    }
489                })?;
490                if let Some(SeriesWriter::Histograms(table_data)) = &mut writer {
491                    if rows_remaining == 0 {
492                        write_native_histogram(
493                            table_data,
494                            &scratch.histogram,
495                            std::mem::take(&mut tags).into_iter(),
496                        )?;
497                    } else {
498                        write_native_histogram(
499                            table_data,
500                            &scratch.histogram,
501                            tags.iter().cloned(),
502                        )?;
503                    }
504                }
505            }
506            TIME_SERIES_EXEMPLARS_TAG => {
507                scratch.exemplar.clear();
508                message::merge(
509                    wire_type,
510                    &mut scratch.exemplar,
511                    &mut buf,
512                    DecodeContext::default(),
513                )
514                .map_err(|mut error| {
515                    error.push("TimeSeries", "exemplars");
516                    error
517                })
518                .context(error::DecodePromRemoteRequestSnafu)?;
519            }
520            TIME_SERIES_METADATA_TAG => {
521                take_length_delimited(wire_type, &mut buf)
522                    .map_err(|mut error| {
523                        error.push("TimeSeries", "metadata");
524                        error
525                    })
526                    .context(error::DecodePromRemoteRequestSnafu)?;
527            }
528            _ => skip_field(wire_type, tag, &mut buf, DecodeContext::default())
529                .context(error::DecodePromRemoteRequestSnafu)?,
530        }
531    }
532
533    ensure!(
534        rows_remaining == 0,
535        error::InvalidPromRemoteRequestSnafu {
536            msg: "remote write v2 row count changed between scans".to_string(),
537        }
538    );
539
540    Ok(())
541}

543fn take_length_delimited<'a>(
544    wire_type: WireType,
545    buf: &mut &'a [u8],
546) -> std::result::Result<&'a [u8], DecodeError> {
547    if wire_type != WireType::LengthDelimited {
548        return Err(DecodeError::new(format!(
549            "invalid wire type: {wire_type:?} (expected LengthDelimited)"
550        )));
551    }
552
553    let len = decode_varint(buf)?;
554    let len =
555        usize::try_from(len).map_err(|_| DecodeError::new("length delimiter exceeds usize"))?;
556    if len > buf.len() {
557        return Err(DecodeError::new("buffer underflow"));
558    }
559    let (value, remaining) = buf.split_at(len);
560    *buf = remaining;
561    Ok(value)
562}

564fn checked_total(total: u64, count: usize, name: &str) -> Result<u64> {
565    let count =
566        u64::try_from(count)
567            .ok()
568            .with_context(|| error::InvalidPromRemoteRequestSnafu {
569                msg: format!("remote write v2 {name} count exceeds u64"),
570            })?;
571    total
572        .checked_add(count)
573        .with_context(|| error::InvalidPromRemoteRequestSnafu {
574            msg: format!("remote write v2 {name} count overflows u64"),
575        })
576}

578fn get_or_create_table_data(
579    tables: &mut HashMap<PromCtx, HashMap<String, TableData>>,
580    prom_ctx: PromCtx,
581    table_name: String,
582    column_count: usize,
583    row_count: usize,
584) -> &mut TableData {
585    match tables.entry(prom_ctx).or_default().entry(table_name) {
586        Entry::Occupied(entry) => {
587            let table_data = entry.into_mut();
588            table_data.reserve_rows(row_count);
589            table_data
590        }
591        Entry::Vacant(entry) => entry.insert(TableData::new(column_count, row_count)),
592    }
593}

595fn write_native_histogram<'a>(
596    table_data: &mut TableData,
597    histogram: &Histogram,
598    tags: impl Iterator<Item = (&'a str, String)>,
599) -> Result<()> {
600    // Persist both int and float families into the logical table schema. Only one
601    // family is populated per row; the other is written as NULL so PromQL can
602    // infer the original histogram flavor without a separate type column.
603    let mut row = table_data.alloc_one_row();
604    row_writer::write_ts_to_millis(
605        table_data,
606        greptime_timestamp(),
607        Some(histogram.timestamp),
608        Precision::Millisecond,
609        &mut row,
610    )?;
611
612    write_native_histogram_value(table_data, histogram, &mut row)?;
613
614    row_writer::write_tags(table_data, tags, &mut row)?;
615    table_data.add_row(row);
616
617    Ok(())
618}

650fn native_histogram_struct_value(histogram: &Histogram) -> Result<ValueData> {
651    let uses_float_counts = native_histogram_uses_float_counts(histogram)?;
652    validate_native_histogram(histogram, uses_float_counts)?;
653
654    let mut items = Vec::with_capacity(NATIVE_HISTOGRAM_FIELD_NAMES.len());
655    let positive_span_lengths = i32_span_lengths("positive", &histogram.positive_spans)?;
656    let negative_span_lengths = i32_span_lengths("negative", &histogram.negative_spans)?;
657    items.extend([
658        pb_value(ValueData::I32Value(histogram.schema)),
659        pb_value(ValueData::F64Value(histogram.zero_threshold)),
660        pb_value(ValueData::F64Value(histogram.sum)),
661        pb_value(ValueData::I32Value(histogram.reset_hint)),
662        optional_pb_value((histogram.start_timestamp != 0).then_some(
663            ValueData::TimestampMillisecondValue(histogram.start_timestamp),
664        )),
665        f64_list_value(histogram.custom_values.iter().copied()),
666        i32_list_value(histogram.positive_spans.iter().map(|span| span.offset)),
667        i32_list_value(positive_span_lengths.iter().copied()),
668        i32_list_value(histogram.negative_spans.iter().map(|span| span.offset)),
669        i32_list_value(negative_span_lengths.iter().copied()),
670    ]);
671
672    if uses_float_counts {
673        validate_float_native_histogram_counts(histogram)?;
674        let count = match histogram.count.as_ref() {
675            Some(Count::CountFloat(count)) => *count,
676            _ => 0.0,
677        };
678        let zero_count = match histogram.zero_count.as_ref() {
679            Some(ZeroCount::ZeroCountFloat(zero_count)) => *zero_count,
680            _ => 0.0,
681        };
682        items.extend([
683            null_pb_value(),
684            null_pb_value(),
685            i64_list_value(std::iter::empty()),
686            i64_list_value(std::iter::empty()),
687            pb_value(ValueData::F64Value(count)),
688            pb_value(ValueData::F64Value(zero_count)),
689            f64_list_value(histogram.positive_counts.iter().copied()),
690            f64_list_value(histogram.negative_counts.iter().copied()),
691        ]);
692    } else {
693        let count = match histogram.count.as_ref() {
694            Some(Count::CountInt(count)) => *count,
695            _ => 0,
696        };
697        let zero_count = match histogram.zero_count.as_ref() {
698            Some(ZeroCount::ZeroCountInt(zero_count)) => *zero_count,
699            _ => 0,
700        };
701        let positive_buckets = bucket_counts_from_deltas(&histogram.positive_deltas)?;
702        let negative_buckets = bucket_counts_from_deltas(&histogram.negative_deltas)?;
703        validate_integer_native_histogram_counts(histogram, &positive_buckets, &negative_buckets)?;
704        let count = i64::try_from(count)
705            .ok()
706            .context(error::InvalidPromRemoteRequestSnafu {
707                msg: format!(
708                    "remote write v2 native histogram integer count {count} overflows i64"
709                ),
710            })?;
711        let zero_count = i64::try_from(zero_count).ok().context(
712            error::InvalidPromRemoteRequestSnafu {
713                msg: format!(
714                    "remote write v2 native histogram integer zero_count {zero_count} overflows i64"
715                ),
716            },
717        )?;
718        items.extend([
719            pb_value(ValueData::I64Value(count)),
720            pb_value(ValueData::I64Value(zero_count)),
721            i64_list_value(positive_buckets.iter().copied()),
722            i64_list_value(negative_buckets.iter().copied()),
723            null_pb_value(),
724            null_pb_value(),
725            f64_list_value(std::iter::empty()),
726            f64_list_value(std::iter::empty()),
727        ]);
728    }
729
730    Ok(ValueData::StructValue(api::v1::StructValue { items }))
731}

733fn validate_native_histogram(histogram: &Histogram, uses_float_counts: bool) -> Result<()> {
734    let exponential_overflow_index = validate_native_histogram_schema(histogram.schema)?;
735    validate_native_histogram_custom_values(histogram)?;
736
737    if histogram.schema == CUSTOM_BUCKETS_SCHEMA {
738        ensure!(
739            histogram.zero_threshold == 0.0 && native_histogram_zero_count_is_zero(histogram),
740            error::InvalidPromRemoteRequestSnafu {
741                msg: "remote write v2 custom native histogram must not use a zero bucket"
742                    .to_string(),
743            }
744        );
745        ensure!(
746            histogram.negative_spans.is_empty()
747                && histogram.negative_deltas.is_empty()
748                && histogram.negative_counts.is_empty(),
749            error::InvalidPromRemoteRequestSnafu {
750                msg: "remote write v2 custom native histogram must not use negative buckets"
751                    .to_string(),
752            }
753        );
754    }
755
756    let (positive_buckets, negative_buckets) = if uses_float_counts {
757        (
758            histogram.positive_counts.len(),
759            histogram.negative_counts.len(),
760        )
761    } else {
762        (
763            histogram.positive_deltas.len(),
764            histogram.negative_deltas.len(),
765        )
766    };
767    let bucket_index_range = if let Some(overflow_index) = exponential_overflow_index {
768        (i32::MIN, overflow_index)
769    } else {
770        (
771            0,
772            i32::try_from(histogram.custom_values.len()).ok().context(
773                error::InvalidPromRemoteRequestSnafu {
774                    msg: "remote write v2 custom native histogram has too many custom_values"
775                        .to_string(),
776                },
777            )?,
778        )
779    };
780    validate_native_histogram_spans(
781        "positive",
782        &histogram.positive_spans,
783        positive_buckets,
784        bucket_index_range,
785    )?;
786    validate_native_histogram_spans(
787        "negative",
788        &histogram.negative_spans,
789        negative_buckets,
790        bucket_index_range,
791    )?;
792
793    Ok(())
794}

796fn validate_native_histogram_schema(schema: i32) -> Result<Option<i32>> {
797    if schema == CUSTOM_BUCKETS_SCHEMA {
798        return Ok(None);
799    }
800
801    if let Some(overflow_index) = exponential_overflow_bucket_index(schema) {
802        return Ok(Some(overflow_index));
803    }
804
805    if (MAX_REMOTE_WRITE_V2_SCHEMA + 1..=MAX_REDUCIBLE_REMOTE_WRITE_V2_SCHEMA).contains(&schema) {
806        error::InvalidPromRemoteRequestSnafu {
807            msg: format!(
808                "remote write v2 native histogram schema {schema} must be reduced before ingestion"
809            ),
810        }
811        .fail()
812    } else {
813        error::InvalidPromRemoteRequestSnafu {
814            msg: format!("remote write v2 native histogram schema {schema} is unsupported"),
815        }
816        .fail()
817    }
818}

820fn validate_native_histogram_custom_values(histogram: &Histogram) -> Result<()> {
821    if histogram.schema != CUSTOM_BUCKETS_SCHEMA {
822        ensure!(
823            histogram.custom_values.is_empty(),
824            error::InvalidPromRemoteRequestSnafu {
825                msg: "remote write v2 standard native histogram must not use custom_values"
826                    .to_string(),
827            }
828        );
829        return Ok(());
830    }
831
832    for value in &histogram.custom_values {
833        ensure!(
834            !value.is_nan() && *value != f64::INFINITY,
835            error::InvalidPromRemoteRequestSnafu {
836                msg: "remote write v2 custom native histogram custom_values must not contain +Inf or NaN"
837                    .to_string(),
838            }
839        );
840    }
841    for values in histogram.custom_values.windows(2) {
842        ensure!(
843            values[0] < values[1],
844            error::InvalidPromRemoteRequestSnafu {
845                msg: "remote write v2 custom native histogram custom_values must be sorted"
846                    .to_string(),
847            }
848        );
849    }
850
851    Ok(())

854fn validate_native_histogram_spans(
855    name: &str,
856    spans: &[BucketSpan],
857    bucket_count: usize,
858    bucket_index_range: (i32, i32),
859) -> Result<()> {
860    let span_len = spans
861        .iter()
862        .try_fold(0usize, |sum, span| sum.checked_add(span.length as usize))
863        .with_context(|| error::InvalidPromRemoteRequestSnafu {
864            msg: format!("remote write v2 native histogram {name} spans overflow"),
865        })?;
866    ensure!(
867        span_len == bucket_count,
868        error::InvalidPromRemoteRequestSnafu {
869            msg: format!(
870                "remote write v2 native histogram {name} spans describe {span_len} buckets, found {bucket_count}"
871            ),
872        }
873    );
874
875    let mut current_index = 0i32;
876    for (span_index, span) in spans.iter().enumerate() {
877        ensure!(
878            span.offset >= 0 || (span_index == 0 && bucket_index_range.0 == i32::MIN),
879            error::InvalidPromRemoteRequestSnafu {
880                msg: format!(
881                    "remote write v2 native histogram {name} span {} has negative offset {}",
882                    span_index + 1,
883                    span.offset
884                ),
885            }
886        );
887        current_index = if span_index == 0 {
888            span.offset
889        } else {
890            current_index.checked_add(span.offset).with_context(|| {
891                error::InvalidPromRemoteRequestSnafu {
892                    msg: format!(
893                        "remote write v2 native histogram {name} span index overflows i32"
894                    ),
895                }
896            })?
897        };
898
899        for _ in 0..span.length {
900            ensure!(
901                (bucket_index_range.0..=bucket_index_range.1).contains(&current_index),
902                error::InvalidPromRemoteRequestSnafu {
903                    msg: format!(
904                        "remote write v2 native histogram {name} bucket index {current_index} is out of range"
905                    ),
906                }
907            );
908            current_index =
909                current_index
910                    .checked_add(1)
911                    .context(error::InvalidPromRemoteRequestSnafu {
912                        msg: format!(
913                            "remote write v2 native histogram {name} span index overflows i32"
914                        ),
915                    })?;
916        }
917    }
918
919    Ok(())
920}

922fn validate_float_native_histogram_counts(histogram: &Histogram) -> Result<()> {
923    let count = match histogram.count.as_ref() {
924        Some(Count::CountFloat(count)) => *count,
925        _ => 0.0,
926    };
927    ensure!(
928        count >= 0.0 || count.is_nan(),
929        error::InvalidPromRemoteRequestSnafu {
930            msg: "remote write v2 native histogram float count must not be negative".to_string(),
931        }
932    );
933
934    let zero_count = match histogram.zero_count.as_ref() {
935        Some(ZeroCount::ZeroCountFloat(zero_count)) => *zero_count,
936        _ => 0.0,
937    };
938    ensure!(
939        zero_count >= 0.0 || zero_count.is_nan(),
940        error::InvalidPromRemoteRequestSnafu {
941            msg: "remote write v2 native histogram float zero_count must not be negative"
942                .to_string(),
943        }
944    );
945
946    for (name, counts) in [
947        ("positive", &histogram.positive_counts),
948        ("negative", &histogram.negative_counts),
949    ] {
950        for (index, count) in counts.iter().enumerate() {
951            ensure!(
952                *count >= 0.0 || count.is_nan(),
953                error::InvalidPromRemoteRequestSnafu {
954                    msg: format!(
955                        "remote write v2 native histogram {name} bucket {} count must not be negative",
956                        index + 1
957                    ),
958                }
959            );
960        }
961    }
962
963    Ok(())
964}

966fn validate_integer_native_histogram_counts(
967    histogram: &Histogram,
968    positive_buckets: &[i64],
969    negative_buckets: &[i64],
970) -> Result<()> {
971    let count = match histogram.count.as_ref() {
972        Some(Count::CountInt(count)) => *count,
973        _ => 0,
974    };
975    let zero_count = match histogram.zero_count.as_ref() {
976        Some(ZeroCount::ZeroCountInt(zero_count)) => *zero_count,
977        _ => 0,
978    };
979    let bucket_count = positive_buckets
980        .iter()
981        .chain(negative_buckets)
982        .try_fold(zero_count, |total, bucket| {
983            total.checked_add(*bucket as u64)
984        })
985        .with_context(|| error::InvalidPromRemoteRequestSnafu {
986            msg: "remote write v2 native histogram bucket total overflows u64".to_string(),
987        })?;
988    ensure!(
989        if histogram.sum.is_nan() {
990            bucket_count <= count
991        } else {
992            bucket_count == count
993        },
994        error::InvalidPromRemoteRequestSnafu {
995            msg: format!(
996                "remote write v2 native histogram has {bucket_count} observations in buckets, count is {count}"
997            ),
998        }
999    );
1000
1001    Ok(())
1002}

1012fn native_histogram_uses_float_counts(histogram: &Histogram) -> Result<bool> {
1013    let uses_float_count = matches!(histogram.count, Some(Count::CountFloat(_)))
1014        || matches!(histogram.zero_count, Some(ZeroCount::ZeroCountFloat(_)));
1015    let uses_int_count = matches!(histogram.count, Some(Count::CountInt(_)))
1016        || matches!(histogram.zero_count, Some(ZeroCount::ZeroCountInt(_)));
1017    let uses_float_buckets =
1018        !histogram.positive_counts.is_empty() || !histogram.negative_counts.is_empty();
1019    let uses_int_buckets =
1020        !histogram.positive_deltas.is_empty() || !histogram.negative_deltas.is_empty();
1021
1022    if matches!(
1023        (&histogram.count, &histogram.zero_count),
1024        (Some(Count::CountInt(_)), Some(ZeroCount::ZeroCountFloat(_)))
1025            | (Some(Count::CountFloat(_)), Some(ZeroCount::ZeroCountInt(_)))
1026    ) {
1027        return error::InvalidPromRemoteRequestSnafu {
1028            msg: "remote write v2 native histogram count and zero_count must use the same integer or float family".to_string(),
1029        }
1030        .fail();
1031    }
1032
1033    ensure!(
1034        !(uses_float_buckets && uses_int_buckets),
1035        error::InvalidPromRemoteRequestSnafu {
1036            msg: "remote write v2 native histogram bucket counts must use either integer deltas or float counts".to_string(),
1037        }
1038    );
1039    ensure!(
1040        !(uses_float_count && uses_int_buckets),
1041        error::InvalidPromRemoteRequestSnafu {
1042            msg: "remote write v2 float native histogram must not use integer bucket deltas"
1043                .to_string(),
1044        }
1045    );
1046    ensure!(
1047        !(uses_int_count && uses_float_buckets),
1048        error::InvalidPromRemoteRequestSnafu {
1049            msg: "remote write v2 integer native histogram must not use float bucket counts"
1050                .to_string(),
1051        }
1052    );
1053
1054    Ok(uses_float_count || uses_float_buckets)
1055}

1142fn resolve_series_labels<'a>(
1143    symbols: &'a [&str],
1144    labels_refs: &[u32],
1145    label_names: &mut HashSet<&'a str>,
1146) -> Result<ResolvedSeriesLabels<'a>> {
1147    ensure!(
1148        labels_refs.len().is_multiple_of(2),
1149        error::InvalidPromRemoteRequestSnafu {
1150            msg: "remote write v2 labels_refs must contain name/value pairs".to_string(),
1151        }
1152    );
1153
1154    let mut prom_ctx = PromCtx::default();
1155    let mut table_name = None;
1156    let mut tags = Vec::with_capacity(labels_refs.len() / 2);
1157    label_names.clear();
1158
1159    for pair in labels_refs.chunks_exact(2) {
1160        let name = symbol_ref(symbols, pair[0], "label name")?;
1161        let value = symbol_ref(symbols, pair[1], "label value")?;
1162        validate_label(name)?;
1163        ensure!(
1164            label_names.insert(name),
1165            error::InvalidPromRemoteRequestSnafu {
1166                msg: format!("remote write v2 label name `{name}` is repeated"),
1167            }
1168        );
1169
1170        if name == METRIC_NAME_LABEL {
1171            table_name = Some(value.to_string());
1172            continue;
1173        }
1174        if apply_remote_write_special_label(name, value, &mut prom_ctx) {
1175            continue;
1176        }
1177
1178        tags.push((name, value.to_string()));
1179    }
1180
1181    let table_name = table_name.with_context(|| error::InvalidPromRemoteRequestSnafu {
1182        msg: "missing '__name__' label in time-series".to_string(),
1183    })?;
1184    ensure!(
1185        !table_name.is_empty(),
1186        error::InvalidPromRemoteRequestSnafu {
1187            msg: "remote write v2 label `__name__` value must not be empty".to_string(),
1188        }
1189    );
1190
1191    Ok((prom_ctx, table_name, tags))
1192}

1271    }

1276    }

1361    ) -> Request {

1366            labels_refs.push(push_symbol(&mut symbols, value));
1367        }

1369        Request {
1370            symbols,
1371            timeseries: vec![TimeSeries {
1372                labels_refs,

1374                histograms,
1375                exemplars: Vec::new(),
1376                metadata: None,
1377            }],
1378        }
1379    }

1381    fn push_symbol(symbols: &mut Vec<String>, symbol: &str) -> u32 {
1382        if let Some(idx) = symbols.iter().position(|s| s == symbol) {
1383            return idx as u32;
1384        }
1385
1386        let idx = symbols.len();
1387        symbols.push(symbol.to_string());
1388        idx as u32
1389    }

2530    fn encoded_message_field(tag: u32, value: &[u8]) -> Vec<u8> {
2531        let mut field = Vec::new();
2532        prost::encoding::encode_key(tag, WireType::LengthDelimited, &mut field);
2533        prost::encoding::encode_varint(u64::try_from(value.len()).unwrap(), &mut field);
2534        field.extend_from_slice(value);
2535        field
2536    }

2636    fn metadata_request(series: Vec<MetadataSeries<'_>>) -> Request {
2637        let mut symbols = vec!["".to_string()];
2638        let timeseries = series
2639            .into_iter()
2640            .map(|(labels, metric_type, unit)| {
2641                let mut labels_refs = Vec::with_capacity(labels.len() * 2);
2642                for (name, value) in labels {
2643                    labels_refs.push(push_test_symbol(&mut symbols, name));
2644                    labels_refs.push(push_test_symbol(&mut symbols, value));
2645                }
2646                let unit_ref = unit.map_or(0, |unit| push_test_symbol(&mut symbols, unit));
2647                TimeSeries {
2648                    labels_refs,
2649                    samples: vec![Sample {
2650                        value: 1.0,
2651                        timestamp: 1000,
2652                        start_timestamp: 0,
2653                    }],
2654                    histograms: Vec::new(),
2655                    exemplars: Vec::new(),
2656                    metadata: Some(Metadata {
2657                        r#type: metric_type,
2658                        help_ref: 0,
2659                        unit_ref,
2660                    }),
2661                }
2662            })
2663            .collect();
2664        Request {
2665            symbols,
2666            timeseries,
2667        }
2668    }
