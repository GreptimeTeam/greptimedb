37/// Create a query context from the grpc metadata.
38pub fn create_query_context_from_grpc_metadata(
39    headers: &MetadataMap,
40) -> TonicResult<QueryContextRef> {
41    let (catalog, schema) = if let Some(db) = extract_header(headers, &[GREPTIME_DB_HEADER_NAME])? {
42        parse_catalog_and_schema_from_db_string(db)
43    } else {
44        (
45            DEFAULT_CATALOG_NAME.to_string(),
46            DEFAULT_SCHEMA_NAME.to_string(),
47        )
48    };
49
50    let ctx = QueryContextBuilder::default()
51        .current_catalog(catalog)
52        .current_schema(schema)
53        .channel(Channel::Grpc)
54        .build();
55    // OTEL Arrow uses ordinary inserts. Accept only its request-level WAL hint,
56    // leaving unrelated hints and reserved internal extensions unchanged.
57    if let Some((key, value)) = hint_headers::extract_hints(headers)
58        .into_iter()
59        .find(|(key, _)| key == INSERT_SKIP_WAL_HINT)
60    {
61        let skip_wal = value.parse::<bool>().map_err(|_| {
62            InvalidParameterSnafu {
63                reason: format!("Invalid {key} hint: expected true or false, got {value:?}"),
64            }
65            .build()
66        })?;
67        ctx.set_skip_wal(skip_wal);
68    }
69    Ok(Arc::new(ctx))
70}

72/// Helper function to extract a header from the metadata map.
73/// Can be multiple keys, and the first one found will be returned.
74pub fn extract_header<'a>(headers: &'a MetadataMap, keys: &[&str]) -> TonicResult<Option<&'a str>> {
75    let mut value = None;
76    for key in keys {
77        if let Some(v) = headers.get(*key) {
78            value = Some(v);
79            break;
80        }
81    }
82
83    let Some(v) = value else {
84        return Ok(None);
85    };
86    let Ok(v) = std::str::from_utf8(v.as_bytes()) else {
87        return Err(InvalidParameterSnafu {
88            reason: "expect valid UTF-8 value",
89        }
90        .build()
91        .into());
92    };
93    Ok(Some(v))
94}

96/// Helper function to extract the header from the metadata and authenticate the user.
97pub async fn check_auth(
98    user_provider: Option<UserProviderRef>,
99    headers: &MetadataMap,
100    query_ctx: QueryContextRef,
101) -> TonicResult<bool> {
102    if user_provider.is_none() {
103        return Ok(true);
104    }
105
106    let auth_schema = extract_header(
107        headers,
108        &[AUTHORIZATION_HEADER, http::header::AUTHORIZATION.as_str()],
109    )?
110    .map(|x| {
111        if x.len() > 5 && x[0..5].eq_ignore_ascii_case("Basic") {
112            x.try_into()
113        } else {
114            // compatible with old version
115            format!("Basic {}", x).as_str().try_into()
116        }
117    })
118    .transpose()?
119    .map(|x: crate::http::authorize::AuthScheme| x.into());
120
121    let auth_schema = auth_schema.context(NotFoundAuthHeaderSnafu)?;
122    let header = RequestHeader {
123        authorization: Some(AuthHeader {
124            auth_scheme: Some(auth_schema),
125        }),
126        catalog: query_ctx.current_catalog().to_string(),
127        schema: query_ctx.current_schema(),
128        ..Default::default()
129    };
130
131    match auth(user_provider, Some(&header), &query_ctx).await {
132        Ok(user_info) => {
133            query_ctx.set_current_user(user_info);
134            Ok(true)
135        }
136        Err(_) => Err(Status::unauthenticated("auth failed")),
137    }
138}

140/// Authenticate the user based on the header and query context.
141pub async fn auth(
142    user_provider: Option<UserProviderRef>,
143    header: Option<&RequestHeader>,
144    query_ctx: &QueryContextRef,
145) -> Result<UserInfoRef> {
146    let Some(user_provider) = user_provider else {
147        return Ok(auth::userinfo_by_name(None));
148    };
149
150    let auth_scheme = header
151        .and_then(|header| {
152            header
153                .authorization
154                .as_ref()
155                .and_then(|x| x.auth_scheme.clone())
156        })
157        .context(NotFoundAuthHeaderSnafu)?;
158
159    match auth_scheme {
160        AuthScheme::Basic(api::v1::Basic { username, password }) => user_provider
161            .auth(
162                Identity::UserId(&username, None),
163                Password::PlainText(password.into()),
164                query_ctx.current_catalog(),
165                &query_ctx.current_schema(),
166            )
167            .await
168            .context(AuthSnafu),
169        AuthScheme::Token(_) => Err(UnsupportedAuthScheme {
170            name: "Token AuthScheme".to_string(),
171        }),
172    }
173    .inspect_err(|e| {
174        METRIC_AUTH_FAILURE
175            .with_label_values(&[e.status_code().as_ref()])
176            .inc();
177    })
178}

180#[cfg(test)]
181mod tests {
182    use session::hints::{HINTS_KEY, REMOTE_QUERY_ID_EXTENSION_KEY, RESERVED_EXTENSION_KEYS};
183
184    use super::*;
185
186    #[test]
187    fn test_arrow_insert_hint_does_not_accept_reserved_extensions() {
188        let mut headers = MetadataMap::new();
189        assert_eq!(
190            create_query_context_from_grpc_metadata(&headers)
191                .unwrap()
192                .extension(INSERT_SKIP_WAL_HINT),
193            None
194        );
195        for (value, expected) in [("true", true), ("false", false)] {
196            let mut hints = format!("insert_skip_wal={value},ttl=7d");
197            for key in RESERVED_EXTENSION_KEYS {
198                hints.push_str(&format!(",{key}=external"));
199            }
200            headers.insert(HINTS_KEY, hints.parse().unwrap());
201            let ctx = create_query_context_from_grpc_metadata(&headers).unwrap();
202            assert_eq!(ctx.skip_wal(), expected);
203            assert_eq!(ctx.extension(INSERT_SKIP_WAL_HINT), None);
204            assert_eq!(ctx.extension("ttl"), None);
205            for key in RESERVED_EXTENSION_KEYS {
206                if key == REMOTE_QUERY_ID_EXTENSION_KEY {
207                    // The builder generates this ID; external hints must not replace it.
208                    assert!(ctx.remote_query_id().is_some());
209                    assert_ne!(ctx.extension(key), Some("external"));
210                } else {
211                    assert_eq!(ctx.extension(key), None);
212                }
213            }
214        }
215        // Only the first matching hint is parsed and applied.
216        for (hints, expected) in [
217            ("insert_skip_wal=true,insert_skip_wal=false", true),
218            ("insert_skip_wal=false,insert_skip_wal=true", false),
219            ("insert_skip_wal=true,insert_skip_wal=invalid", true),
220        ] {
221            headers.insert(HINTS_KEY, hints.parse().unwrap());
222            let ctx = create_query_context_from_grpc_metadata(&headers).unwrap();
223            assert_eq!(ctx.skip_wal(), expected);
224        }
225        for value in ["", "TRUE", "1", "invalid"] {
226            headers.insert(
227                HINTS_KEY,
228                format!("insert_skip_wal={value}").parse().unwrap(),
229            );
230            assert!(create_query_context_from_grpc_metadata(&headers).is_err());
231        }
232    }
