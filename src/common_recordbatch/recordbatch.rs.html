39/// A two-dimensional batch of column-oriented data with a defined schema.
40#[derive(Clone, Debug, PartialEq)]
41pub struct RecordBatch {
42    pub schema: SchemaRef,
43    df_record_batch: DfRecordBatch,
44}

46impl RecordBatch {
47    /// Create a new [`RecordBatch`] from `schema` and `columns`.
48    pub fn new<I: IntoIterator<Item = VectorRef>>(
49        schema: SchemaRef,
50        columns: I,
51    ) -> Result<RecordBatch> {
52        let columns: Vec<_> = columns.into_iter().collect();
53        let arrow_arrays = columns.iter().map(|v| v.to_arrow_array()).collect();
54
55        // Casting the arrays here to match the schema, is a temporary solution to support Arrow's
56        // view array types (`StringViewArray` and `BinaryViewArray`).
57        // As to "support": the arrays here are created from vectors, which do not have types
58        // corresponding to view arrays. What we can do is to only cast them.
59        // As to "temporary": we are planing to use Arrow's RecordBatch directly in the read path.
60        // the casting here will be removed in the end.
61        // TODO(LFC): Remove the casting here once `Batch` is no longer used.
62        let arrow_arrays = Self::cast_view_arrays(schema.arrow_schema(), arrow_arrays)?;
63
64        let df_record_batch = DfRecordBatch::try_new(schema.arrow_schema().clone(), arrow_arrays)
65            .context(error::NewDfRecordBatchSnafu)?;
66
67        Ok(RecordBatch {
68            schema,
69            df_record_batch,
70        })
71    }

73    pub fn to_df_record_batch<I: IntoIterator<Item = VectorRef>>(
74        arrow_schema: ArrowSchemaRef,
75        columns: I,
76    ) -> Result<DfRecordBatch> {
77        let columns: Vec<_> = columns.into_iter().collect();
78        let arrow_arrays = columns.iter().map(|v| v.to_arrow_array()).collect();
79
80        // Casting the arrays here to match the schema, is a temporary solution to support Arrow's
81        // view array types (`StringViewArray` and `BinaryViewArray`).
82        // As to "support": the arrays here are created from vectors, which do not have types
83        // corresponding to view arrays. What we can do is to only cast them.
84        // As to "temporary": we are planing to use Arrow's RecordBatch directly in the read path.
85        // the casting here will be removed in the end.
86        // TODO(LFC): Remove the casting here once `Batch` is no longer used.
87        let arrow_arrays = Self::cast_view_arrays(&arrow_schema, arrow_arrays)?;
88
89        let df_record_batch = DfRecordBatch::try_new(arrow_schema, arrow_arrays)
90            .context(error::NewDfRecordBatchSnafu)?;
91
92        Ok(df_record_batch)
93    }

95    fn cast_view_arrays(
96        schema: &ArrowSchemaRef,
97        mut arrays: Vec<ArrayRef>,
98    ) -> Result<Vec<ArrayRef>> {
99        for (f, a) in schema.fields().iter().zip(arrays.iter_mut()) {
100            let expected = f.data_type();
101            let actual = a.data_type();
102            if matches!(
103                (expected, actual),
104                (ArrowDataType::Utf8View, ArrowDataType::Utf8)
105                    | (ArrowDataType::BinaryView, ArrowDataType::Binary)
106            ) {
107                *a = compute::cast(a, expected).context(ArrowComputeSnafu)?;
108            }
109        }
110        Ok(arrays)
111    }

195    pub fn column_vectors(
196        &self,
197        table_name: &str,
198        table_schema: SchemaRef,
199    ) -> Result<HashMap<String, VectorRef>> {
200        let mut vectors = HashMap::with_capacity(self.num_columns());
201
202        // column schemas in recordbatch must match its vectors, otherwise it's corrupted
203        for (field, array) in self
204            .df_record_batch
205            .schema()
206            .fields()
207            .iter()
208            .zip(self.df_record_batch.columns().iter())
209        {
210            let column_name = field.name();
211            let column_schema =
212                table_schema
213                    .column_schema_by_name(column_name)
214                    .context(ColumnNotExistsSnafu {
215                        table_name,
216                        column_name,
217                    })?;
218            let vector = if field.data_type() != &column_schema.data_type.as_arrow_type() {
219                let array = compute::cast(array, &column_schema.data_type.as_arrow_type())
220                    .context(ArrowComputeSnafu)?;
221                Helper::try_into_vector(array).context(DataTypesSnafu)?
222            } else {
223                Helper::try_into_vector(array).context(DataTypesSnafu)?
224            };
225
226            let _ = vectors.insert(column_name.clone(), vector);
227        }
228
229        Ok(vectors)
230    }

241        ensure!(
242            offset + len <= self.num_rows(),
243            error::RecordBatchSliceIndexOverflowSnafu {
244                size: self.num_rows(),
245                visit_index: offset + len

249        Ok(RecordBatch::from_df_record_batch(
250            self.schema.clone(),
251            sliced,
252        ))
253    }

255    /// Returns the total number of bytes of memory pointed to by the arrays in this `RecordBatch`.
256    ///
257    /// The buffers store bytes in the Arrow memory format, and include the data as well as the validity map.
258    /// Note that this does not always correspond to the exact memory usage of an array,
259    /// since multiple arrays can share the same buffers or slices thereof.
260    pub fn buffer_memory_size(&self) -> usize {
261        self.df_record_batch
262            .columns()
263            .iter()
264            .map(|array| array.get_buffer_memory_size())
265            .sum()
266    }
267
268    /// Returns the logical memory size of this batch's array slices.
269    ///
270    /// This sums Arrow's logical visible slice buffers rather than the capacity of their shared
271    /// backing buffers. View out-of-line payloads and nested custom payloads are not separately
272    /// traversed or accounted. It is not an exact measure of live physical memory. If Arrow cannot
273    /// calculate a slice's size, the full buffer size is used conservatively.
274    ///
275    /// Flat columns (fixed-width primitives, timestamps, booleans, and `Utf8`/`LargeUtf8`/
276    /// `Binary`/`LargeBinary`) are measured directly from their typed arrays using Arrow's own
277    /// formula (see [`flat_slice_memory_size`]), so the scan hot path does not materialize an
278    /// `ArrayData` per column per batch. All other types use Arrow's calculation.
279    ///
280    /// Mito's current scan paths do not produce top-level View arrays. If they do in the future,
281    /// their out-of-line payload accounting must be reassessed here.
282    pub fn logical_slice_memory_size(&self) -> usize {
283        self.df_record_batch
284            .columns()
285            .iter()
286            .fold(0, |total, array| {
287                total.saturating_add(logical_column_memory_size(array))
288            })
289    }

291    /// Iterate the values as strings in the column at index `i`.
292    ///
293    /// Note that if the underlying array is not a valid GreptimeDB vector, an empty iterator is
294    /// returned.
295    ///
296    /// # Panics
297    /// if index `i` is out of bound.
298    pub fn iter_column_as_string(&self, i: usize) -> Box<dyn Iterator<Item = Option<String>> + '_> {
299        macro_rules! iter {
300            ($column: ident) => {
301                Box::new(
302                    (0..$column.len())
303                        .map(|i| $column.is_valid(i).then(|| $column.value(i).to_string())),
304                )
305            };
306        }
307
308        let column = self.df_record_batch.column(i);
309        match column.data_type() {
310            ArrowDataType::Utf8 => {
311                let column = column.as_string::<i32>();
312                let iter = iter!(column);
313                iter as _
314            }
315            ArrowDataType::LargeUtf8 => {
316                let column = column.as_string::<i64>();
317                iter!(column)
318            }
319            ArrowDataType::Utf8View => {
320                let column = column.as_string_view();
321                iter!(column)
322            }
323            _ => {
324                if let Ok(column) = Helper::try_into_vector(column) {
325                    Box::new(
326                        (0..column.len())
327                            .map(move |i| (!column.is_null(i)).then(|| column.get(i).to_string())),
328                    )
329                } else {
330                    Box::new(std::iter::empty())
331                }
332            }
333        }
334    }

337/// Returns Arrow's logical slice memory size of a single column.
338///
339/// This is `array.to_data().get_slice_memory_size()`, with the full buffer size as a conservative
340/// fallback when Arrow cannot calculate the slice size. Flat columns are measured from their typed
341/// array (see [`flat_slice_memory_size`]) so that no `ArrayData` is materialized for them.
342fn logical_column_memory_size(array: &dyn Array) -> usize {
343    flat_slice_memory_size(array).unwrap_or_else(|| {
344        array
345            .to_data()
346            .get_slice_memory_size()
347            .unwrap_or_else(|_| array.get_buffer_memory_size())
348    })
349}
350
351/// Computes `array.to_data().get_slice_memory_size()` for flat arrays without building the
352/// `ArrayData`.
353///
354/// The formulas mirror Arrow's `ArrayData::get_slice_memory_size` (arrow-data 59.2.0,
355/// `get_slice_memory_size` in `arrow-data/src/data.rs`):
356///
357/// - a fixed-width value buffer costs `len * byte_width`, where the byte width is the one of the
358///   type's buffer layout, as reported by `DataType::primitive_width`,
359/// - the value bitmap of a boolean array costs `ceil(len / 8)`,
360/// - the offsets buffer of a variable-width array holds `len + 1` boundaries and costs
361///   `(len + 1) * offset_width`,
362/// - the visible value bytes of a variable-width array cost `offset[len] - offset[0]`,
363/// - the validity bitmap costs `ceil(len / 8)`, but only if the slice contains a null: converting
364///   an array to `ArrayData` drops a validity bitmap without any null (`ArrayDataBuilder::build`),
365///   so Arrow only sees one when `null_count` is non-zero.
366///
367/// Returns `None` for data types that are left to Arrow's general computation (nested, view,
368/// dictionary, union, `FixedSizeBinary`, `Null`, ...) and when the size computation overflows
369/// `usize`, which Arrow reports as an error. Callers must then fall back to Arrow.
370fn flat_slice_memory_size(array: &dyn Array) -> Option<usize> {
371    let len = array.len();
372    let validity_size = if array.null_count() != 0 {
373        bit_util::ceil(len, 8)
374    } else {
375        0
376    };
377
378    match array.data_type() {
379        ArrowDataType::Boolean => bit_util::ceil(len, 8).checked_add(validity_size),
380        ArrowDataType::Utf8 => {
381            offset_slice_memory_size(len, array.as_string::<i32>().offsets(), validity_size)
382        }
383        ArrowDataType::LargeUtf8 => {
384            offset_slice_memory_size(len, array.as_string::<i64>().offsets(), validity_size)
385        }
386        ArrowDataType::Binary => {
387            offset_slice_memory_size(len, array.as_binary::<i32>().offsets(), validity_size)
388        }
389        ArrowDataType::LargeBinary => {
390            offset_slice_memory_size(len, array.as_binary::<i64>().offsets(), validity_size)
391        }
392        data_type => data_type
393            .primitive_width()
394            .and_then(|byte_width| fixed_width_slice_memory_size(len, byte_width, validity_size)),
395    }
396}

398/// Returns `len * byte_width + validity_size`, or `None` on overflow.
399#[inline]
400fn fixed_width_slice_memory_size(
401    len: usize,
402    byte_width: usize,
403    validity_size: usize,
404) -> Option<usize> {
405    len.checked_mul(byte_width)?.checked_add(validity_size)
406}
407
408/// Returns `(len + 1) * offset_width + (offsets[len] - offsets[0]) + validity_size`, or `None` on
409/// overflow.
410///
411/// `offsets` are the offsets of the visible slice, so `offsets[0]` is the first visible byte and
412/// `offsets[len]` is the end of the last visible value. The offsets buffer size is checked before
413/// `offsets[len]` is read, so a `len` that cannot come from a valid offsets buffer is reported as
414/// an overflow instead of panicking.
415#[inline]
416fn offset_slice_memory_size<O: OffsetSizeTrait>(
417    len: usize,
418    offsets: &[O],
419    validity_size: usize,
420) -> Option<usize> {
421    let offsets_size = len.checked_add(1)?.checked_mul(size_of::<O>())?;
422    let values_size = (offsets[len] - offsets[0]).as_usize();
423    offsets_size
424        .checked_add(values_size)?
425        .checked_add(validity_size)
426}

428impl Serialize for RecordBatch {
429    fn serialize<S>(&self, serializer: S) -> std::result::Result<S::Ok, S::Error>
430    where
431        S: Serializer,
432    {
433        // TODO(yingwen): arrow and arrow2's schemas have different fields, so
434        // it might be better to use our `RawSchema` as serialized field.
435        let mut s = serializer.serialize_struct("record", 2)?;
436        s.serialize_field("schema", &**self.schema.arrow_schema())?;
437
438        let columns = self.df_record_batch.columns();
439        let columns = Helper::try_into_vectors(columns).map_err(Error::custom)?;
440        let vec = columns
441            .iter()
442            .map(|c| c.serialize_to_json())
443            .collect::<std::result::Result<Vec<_>, _>>()
444            .map_err(S::Error::custom)?;
445
446        s.serialize_field("columns", &vec)?;
447        s.end()
448    }
449}

451/// merge multiple recordbatch into a single
452pub fn merge_record_batches(schema: SchemaRef, batches: &[RecordBatch]) -> Result<RecordBatch> {
453    let batches_len = batches.len();
454    if batches_len == 0 {
455        return Ok(RecordBatch::new_empty(schema));
456    }
457
458    let record_batch = compute::concat_batches(
459        schema.arrow_schema(),
460        batches.iter().map(|x| x.df_record_batch()),
461    )
462    .context(ArrowComputeSnafu)?;
463
464    // Create a new RecordBatch with merged columns
465    Ok(RecordBatch::from_df_record_batch(schema, record_batch))
466}

670    fn arrow_slice_memory_size(array: &dyn Array) -> usize {
671        array
672            .to_data()
673            .get_slice_memory_size()
674            .unwrap_or_else(|_| array.get_buffer_memory_size())

677    /// Slices probed for every array: the whole array, empty slices at both ends, single element
678    /// slices, a non byte aligned range, and slices that skip a prefix or a suffix.
679    fn probe_ranges(len: usize) -> Vec<(usize, usize)> {
680        if len == 0 {
681            return vec![(0, 0)];
682        }
683
684        let mut ranges = vec![
685            (0, 0),
686            (len, 0),
687            (0, 1),
688            (len - 1, 1),
689            (1, len.saturating_sub(1)),
690            (len / 3, len / 3),
691        ];
692        if len >= 8 {
693            ranges.push((3, 5));
694        }
695        ranges
696    }

698    /// Asserts the fast path covers `array` and agrees with Arrow's own formula.
699    #[track_caller]
700    fn assert_fast_path_matches_arrow(name: &str, array: &dyn Array) {
701        let expected = arrow_slice_memory_size(array);
702        let Some(actual) = flat_slice_memory_size(array) else {
703            panic!("{name}: fast path must cover {:?}", array.data_type());
704        };
705
706        assert_eq!(
707            expected,
708            actual,
709            "{name}: fast path disagrees with Arrow for {:?} (len {}, null count {})",
710            array.data_type(),
711            array.len(),
712            array.null_count()
713        );
714        assert_eq!(
715            expected,
716            logical_column_memory_size(array),
717            "{name}: column size disagrees with Arrow for {:?}",
718            array.data_type()
719        );
720    }

722    /// Asserts the fast path agrees with Arrow for `array` and all probe slices of it.
723    fn assert_fast_path_matches_arrow_on_slices(name: &str, array: &ArrayRef) {
724        assert_fast_path_matches_arrow(name, array.as_ref());
725
726        for (offset, len) in probe_ranges(array.len()) {
727            let slice = array.slice(offset, len);

735    /// Returns a full array, an array with a few nulls and an all-null array of `len` values.
736    fn primitive_cases<T: ArrowPrimitiveType>(len: usize) -> Vec<ArrayRef> {
737        let values = (0..len)
738            .map(|index| Some(T::Native::usize_as(index)))
739            .collect::<PrimitiveArray<T>>();
740        let with_nulls = (0..len)
741            .map(|index| (index % 3 != 0).then(|| T::Native::usize_as(index)))
742            .collect::<PrimitiveArray<T>>();
743        let all_null = (0..len)
744            .map(|_| None::<T::Native>)
745            .collect::<PrimitiveArray<T>>();
746
747        vec![
748            Arc::new(values) as ArrayRef,
749            Arc::new(with_nulls),
750            Arc::new(all_null),
751        ]
752    }

819    fn test_flat_slice_memory_size_matches_arrow_for_byte_arrays() {
820        // Empty, short (inline) and long (out of line) values, plus nulls.
821        let strings: Vec<Option<&str>> = vec![
822            Some(""),
823            Some("a"),
824            Some("greptime"),
825            Some("a value that is much longer than twelve bytes"),
826            None,
827            Some(""),
828            Some("z"),
829            Some("another long value that exceeds the inline limit"),
830            None,
831        ];
832        let bytes: Vec<Option<&[u8]>> = strings
833            .iter()
834            .map(|value| value.map(str::as_bytes))
835            .collect();
836
837        let cases: Vec<(&str, ArrayRef)> = vec![
838            ("utf8", Arc::new(StringArray::from(strings.clone()))),
839            (
840                "large-utf8",
841                Arc::new(LargeStringArray::from(strings.clone())),
842            ),
843            ("binary", Arc::new(BinaryArray::from(bytes.clone()))),
844            ("large-binary", Arc::new(LargeBinaryArray::from(bytes))),
845        ];
846
847        for (name, array) in cases {
848            assert_fast_path_matches_arrow_on_slices(name, &array);
849
850            // `(len + 1) * offset_width + visible value bytes`, plus the validity bitmap.
851            let len = array.len();
852            let offset_width = match array.data_type() {
853                DataType::Utf8 | DataType::Binary => 4,
854                DataType::LargeUtf8 | DataType::LargeBinary => 8,
855                other => panic!("unexpected type {other}"),
856            };
857            let visible: usize = strings.iter().map(|value| value.map_or(0, str::len)).sum();
858            let mut expected = (len + 1) * offset_width + visible;
859            if array.null_count() != 0 {
860                expected += bit_util::ceil(len, 8);
861            }
862            assert_eq!(
863                Some(expected),
864                flat_slice_memory_size(array.as_ref()),
865                "{name}"
866            );
867        }
868
869        // Zero rows keep one offset boundary per array, and nothing else.
870        let empty = Arc::new(StringArray::from(Vec::<Option<&str>>::new())) as ArrayRef;
871        assert_eq!(Some(4), flat_slice_memory_size(empty.as_ref()));
872        assert_eq!(4, arrow_slice_memory_size(empty.as_ref()));
873        assert_fast_path_matches_arrow("utf8-empty", empty.as_ref());
874
875        let empty_large = Arc::new(LargeStringArray::from(Vec::<Option<&str>>::new())) as ArrayRef;
876        assert_eq!(Some(8), flat_slice_memory_size(empty_large.as_ref()));
877        assert_fast_path_matches_arrow("large-utf8-empty", empty_large.as_ref());
878    }
