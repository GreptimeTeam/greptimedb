15//! Preflight validation of persisted partition expressions against an ALTER TABLE schema change.
16//!
17//! Persisted partition bounds keep the types they were written with. A column whose
18//! type changes while a partition expression references it breaks the region on both
19//! paths that consume the bounds:
20//!
21//! * On scan, the region coerces the bound literal and the new column type to a
22//!   common comparison type while building the partition predicate. If no common type
23//!   exists, or the bound cannot be cast to it (a classic example is a millisecond
24//!   timestamp bound that overflows when the time index column is widened to
25//!   nanoseconds), every scan of that region fails at that point.
26//! * On write, the region routes rows through `MultiDimPartitionRule::find_region`,
27//!   which compares the persisted bound against row values directly (`Value::cmp`): a
28//!   bound whose variant no longer matches the column's panics on ordered comparisons
29//!   and misroutes silently on equality. Only a timestamp unit widening keeps the two
30//!   comparable — `Timestamp` comparisons convert units before comparing — so every
31//!   other type change of a referenced column is rejected outright, and the widening
32//!   keeps the scan check below.
33//!
34//! Such an ALTER must be rejected before it is submitted to any region, otherwise
35//! already-altered regions keep failing while the rest of the table is untouched.
36//!
37//! The `partition` crate depends on `common-meta`, so this crate cannot depend on
38//! `partition` back and cannot reuse [`partition::expr::PartitionExpr`]. The `Persisted*`
39//! types below mirror the JSON representation of `partition::partition::PartitionBound` and
40//! `partition::expr::PartitionExpr` and must be kept in sync with `partition/src/partition.rs`
41//! and `partition/src/expr.rs`; round-trip tests in `partition` run real serialized
42//! expressions through this validator so drift fails in CI. The compatibility check itself
43//! deliberately repeats the same conversion (value -> [`ScalarValue`]) and the same
44//! comparison coercion the region performs when it builds the partition predicate
45//! (`ExprSimplifier::coerce` / [`comparison_coercion`]): both sides are cast to the common
46//! comparison type, and the persisted bound must be castable to that type with
47//! [`ScalarValue::cast_to`]. A bound that would fail on scan is rejected here as well.

65/// Mirrors `partition::partition::PartitionBound`.
66#[derive(Debug, Deserialize)]
67enum PersistedPartitionBound {
68    /// Deprecated since 0.9.0. Only deserialized to keep unknown old formats parseable.
69    #[allow(dead_code)]
70    Value(Value),
71    /// Deprecated since 0.15.0.
72    MaxValue,
73    Expr(PersistedPartitionExpr),
74}

76/// Mirrors `partition::expr::PartitionExpr`.
77#[derive(Debug, Deserialize)]
78struct PersistedPartitionExpr {
79    lhs: Box<PersistedOperand>,
80    op: PersistedOp,
81    rhs: Box<PersistedOperand>,
82}
83
84/// Mirrors `partition::expr::Operand`.
85#[derive(Debug, Deserialize)]
86enum PersistedOperand {
87    Column(String),
88    Value(Value),
89    Expr(PersistedPartitionExpr),
90}

92/// Mirrors `partition::expr::RestrictedOp`.
93#[derive(Debug, Deserialize)]
94enum PersistedOp {
95    Eq,
96    NotEq,
97    Lt,
98    LtEq,
99    Gt,
100    GtEq,
101    And,
102    Or,
103}

105/// Returns the columns whose data type differs between `old_schema` and `new_schema`,
106/// mapped to (old type, new type). Columns only present in one of the schemas (added or
107/// dropped) are not included; dropping a column referenced by a partition expression is
108/// already rejected by the table metadata validation.
109pub(crate) fn changed_column_types(
110    old_schema: &Schema,
111    new_schema: &Schema,
112) -> HashMap<String, (ConcreteDataType, ConcreteDataType)> {
113    let mut changed = HashMap::new();
114    for new_column in new_schema.column_schemas() {
115        let Some(old_column) = old_schema.column_schema_by_name(&new_column.name) else {
116            continue;
117        };
118        if old_column.data_type != new_column.data_type {
119            changed.insert(
120                new_column.name.clone(),
121                (old_column.data_type.clone(), new_column.data_type.clone()),
122            );
123        }
124    }
125
126    changed
127}

129/// Validates that the persisted partition bounds of every region are still representable
130/// with the types of `changed_columns`. Returns an error naming the first offending
131/// region without modifying anything.
132///
133/// Public so that round-trip tests in the `partition` crate can run real serialized
134/// partition expressions through the same mirror used here.
135pub fn validate_region_partition_bounds(
136    changed_columns: &HashMap<String, (ConcreteDataType, ConcreteDataType)>,
137    region_routes: &[RegionRoute],
138) -> Result<()> {
139    if changed_columns.is_empty() {
140        return Ok(());
141    }
142
143    for route in region_routes {
144        let expr_json = route.region.partition_expr();
145        if expr_json.is_empty() {
146            continue;
147        }
148
149        let bound: PersistedPartitionBound = match serde_json::from_str(&expr_json) {
150            Ok(bound) => bound,
151            Err(err) => {
152                // A malformed persisted expression already fails every scan of this
153                // region. It is unrelated to the schema change, so don't block the ALTER.
154                warn!(
155                    "Skipping partition expression validation for region {}, \
156                     the persisted expression is not parseable: {}, expr: {}",
157                    route.region.id, err, expr_json
158                );
159                continue;
160            }
161        };
162        let PersistedPartitionBound::Expr(expr) = bound else {
163            // Deprecated bound formats don't carry expressions and are not used
164            // when the region builds its partition predicate.
165            continue;
166        };
167
168        validate_expr(&expr, changed_columns, route.region.id)?;
169    }
170
171    Ok(())
172}

174fn validate_expr(
175    expr: &PersistedPartitionExpr,
176    changed_columns: &HashMap<String, (ConcreteDataType, ConcreteDataType)>,
177    region_id: RegionId,
178) -> Result<()> {
179    match expr.op {
180        PersistedOp::And | PersistedOp::Or => {
181            validate_operand(&expr.lhs, changed_columns, region_id)?;
182            validate_operand(&expr.rhs, changed_columns, region_id)?;
183        }
184        PersistedOp::Eq
185        | PersistedOp::NotEq
186        | PersistedOp::Lt
187        | PersistedOp::LtEq
188        | PersistedOp::Gt
189        | PersistedOp::GtEq => {
190            // Comparison between a column and a bound literal in either order.
191            validate_comparison(&expr.lhs, &expr.rhs, changed_columns, region_id)?;
192            validate_comparison(&expr.rhs, &expr.lhs, changed_columns, region_id)?;
193        }
194    }
195
196    Ok(())
197}
198
199fn validate_operand(
200    operand: &PersistedOperand,
201    changed_columns: &HashMap<String, (ConcreteDataType, ConcreteDataType)>,
202    region_id: RegionId,
203) -> Result<()> {
204    match operand {
205        PersistedOperand::Expr(expr) => validate_expr(expr, changed_columns, region_id),
206        PersistedOperand::Column(_) | PersistedOperand::Value(_) => Ok(()),
207    }
208}
209
210fn validate_comparison(
211    operand: &PersistedOperand,
212    other: &PersistedOperand,
213    changed_columns: &HashMap<String, (ConcreteDataType, ConcreteDataType)>,
214    region_id: RegionId,
215) -> Result<()> {
216    let (PersistedOperand::Column(column), PersistedOperand::Value(value)) = (operand, other)
217    else {
218        return Ok(());
219    };
220    let Some((old_type, target_type)) = changed_columns.get(column) else {
221        return Ok(());
222    };
223    // Values not representable as a partition bound (lists, structs, json) are
224    // rejected by the partition parser and cannot appear in a persisted expression.
225    let Some(scalar) = value_to_scalar_value(value) else {
226        return Ok(());
227    };
228
229    // The write path (`MultiDimPartitionRule::find_region`) compares the persisted
230    // bound against row values directly: `Value::cmp` panics on mismatched variants
231    // and equality misroutes silently. Only a timestamp unit widening keeps the
232    // bound and the rows comparable (`Timestamp` comparisons convert units first),
233    // so every other type change of a referenced column is rejected here.
234    if !old_type.is_timestamp_unit_widening_to(target_type) {
235        warn!(
236            "Persisted partition bound would be compared against values of a different \
237             type after the ALTER, region_id: {}, column: '{}', target_type: {}",
238            region_id, column, target_type
239        );
240        return PartitionExprIncompatibleSnafu {
241            region_id,
242            column: column.clone(),
243            target_type: target_type.to_string(),
244        }
245        .fail();
246    }
247
248    // For the widening, mirror `ExprSimplifier::coerce` in `PredicateGroup::new`:
249    // the region casts both the column and the persisted bound to the common
250    // comparison type. The bound is not cast to the column type itself, so the
251    // widening is accepted as long as the bound survives the common type; a bound
252    // that cannot be cast to it — or a pair of types without a common comparison
253    // type — fails every scan.
254    let target_arrow_type = target_type.as_arrow_type();
255    let Some(common_type) = comparison_coercion(&target_arrow_type, &scalar.data_type()) else {
256        warn!(
257            "Persisted partition bound has no common comparison type with the altered \
258             column type, region_id: {}, column: '{}', target_type: {}",
259            region_id, column, target_type
260        );
261        return PartitionExprIncompatibleSnafu {
262            region_id,
263            column: column.clone(),
264            target_type: target_type.to_string(),
265        }
266        .fail();
267    };
268    if scalar.data_type() != common_type
269        && let Err(error) = scalar.cast_to(&common_type)
270    {
271        warn!(
272            error;
273            "Persisted partition bound is not representable with the altered column type, \
274             region_id: {}, column: '{}', target_type: {}",
275            region_id,
276            column,
277            target_type
278        );
279        return PartitionExprIncompatibleSnafu {
280            region_id,
281            column: column.clone(),
282            target_type: target_type.to_string(),
283        }
284        .fail();
285    }
286
287    Ok(())
288}

290/// Converts a persisted bound value to a [`ScalarValue`], mirroring
291/// `partition::expr::Operand::try_as_logical_expr`. Returns `None` for values that
292/// cannot appear in a persisted partition expression.
293fn value_to_scalar_value(value: &Value) -> Option<ScalarValue> {
294    Some(match value {
295        Value::Boolean(v) => ScalarValue::Boolean(Some(*v)),
296        Value::UInt8(v) => ScalarValue::UInt8(Some(*v)),
297        Value::UInt16(v) => ScalarValue::UInt16(Some(*v)),
298        Value::UInt32(v) => ScalarValue::UInt32(Some(*v)),
299        Value::UInt64(v) => ScalarValue::UInt64(Some(*v)),
300        Value::Int8(v) => ScalarValue::Int8(Some(*v)),
301        Value::Int16(v) => ScalarValue::Int16(Some(*v)),
302        Value::Int32(v) => ScalarValue::Int32(Some(*v)),
303        Value::Int64(v) => ScalarValue::Int64(Some(*v)),
304        Value::Float32(v) => ScalarValue::Float32(Some(v.0)),
305        Value::Float64(v) => ScalarValue::Float64(Some(v.0)),
306        Value::String(v) => ScalarValue::Utf8(Some(v.as_utf8().to_string())),
307        Value::Binary(v) => ScalarValue::Binary(Some(v.to_vec())),
308        Value::Date(v) => ScalarValue::Date32(Some(v.val())),
309        Value::Null => ScalarValue::Null,
310        Value::Timestamp(t) => timestamp_to_scalar_value(t.unit(), Some(t.value())),
311        Value::Time(t) => time_to_scalar_value(*t.unit(), Some(t.value())).ok()?,
312        Value::IntervalYearMonth(v) => ScalarValue::IntervalYearMonth(Some(v.to_i32())),
313        Value::IntervalDayTime(v) => ScalarValue::IntervalDayTime(Some((*v).into())),
314        Value::IntervalMonthDayNano(v) => ScalarValue::IntervalMonthDayNano(Some((*v).into())),
315        Value::Duration(d) => duration_to_scalar_value(d.unit(), Some(d.value())),
316        Value::Decimal128(d) => {
317            let (v, p, s) = d.to_scalar_value();
318            ScalarValue::Decimal128(v, p, s)
319        }
320        Value::List(_) | Value::Struct(_) | Value::Json(_) => return None,
321    })
322}

332    fn schema(ts_type: ConcreteDataType) -> Schema {
333        Schema::new(vec![
334            ColumnSchema::new("ts", ts_type, false),
335            ColumnSchema::new("host", ConcreteDataType::string_datatype(), true),
336        ])
337    }
338
339    fn ts_bound(millis: i64) -> String {
340        format!(
341            r#"{{"Expr":{{"lhs":{{"Column":"ts"}},"op":"Lt","rhs":{{"Value":{{"Timestamp":{{"value":{millis},"unit":"Millisecond"}}}}}}}}}}"#
342        )
343    }
344
345    fn routes(exprs: &[&str]) -> Vec<RegionRoute> {
346        exprs
347            .iter()
348            .enumerate()
349            .map(|(index, expr)| RegionRoute {
350                region: Region {
351                    id: RegionId::new(1024, index as u32 + 1),
352                    partition_expr: expr.to_string(),
353                    ..Default::default()
354                },
355                ..Default::default()
356            })
357            .collect()
358    }
359
360    fn changed(
361        column: &str,
362        old_type: ConcreteDataType,
363        new_type: ConcreteDataType,
364    ) -> HashMap<String, (ConcreteDataType, ConcreteDataType)> {
365        HashMap::from([(column.to_string(), (old_type, new_type))])
366    }

369    fn test_changed_column_types() {
370        let old = schema(ConcreteDataType::timestamp_millisecond_datatype());
371        let new = schema(ConcreteDataType::timestamp_nanosecond_datatype());
372        let changed = changed_column_types(&old, &new);
373        assert_eq!(changed.len(), 1);
374        assert_eq!(
375            changed["ts"],
376            (
377                ConcreteDataType::timestamp_millisecond_datatype(),
378                ConcreteDataType::timestamp_nanosecond_datatype()
379            )
380        );
381
382        // Added and dropped columns are not type changes.
383        let mut new_columns = old.column_schemas().to_vec();
384        new_columns.push(ColumnSchema::new(
385            "extra",
386            ConcreteDataType::int64_datatype(),
387            true,
388        ));
389        let new = Schema::new(new_columns);
390        assert!(changed_column_types(&old, &new).is_empty());
391    }
