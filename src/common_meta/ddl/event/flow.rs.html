30pub(crate) const CREATE_FLOW_EVENT_TYPE: &str = "create_flow";
31pub(crate) const DROP_FLOW_EVENT_TYPE: &str = "drop_flow";
32
33const PAYLOAD_VERSION: u8 = 1;
34
35/// The bounded Create Flow intent allowed in a submitted event payload.
36#[derive(Debug)]
37pub(crate) struct CreateFlowEventIntent {
38    pub(crate) or_replace: bool,
39    pub(crate) create_if_not_exists: bool,
40    pub(crate) expire_after: Option<i64>,
41    pub(crate) eval_interval_secs: Option<i64>,
42}
43
44#[derive(Debug, Serialize)]
45struct CreateFlowPayload {
46    version: u8,
47    or_replace: bool,
48    create_if_not_exists: bool,
49    expire_after: Option<i64>,
50    eval_interval_secs: Option<i64>,
51}
52
53#[derive(Debug, Serialize)]
54struct DropFlowPayload {
55    version: u8,
56    drop_if_exists: bool,
57}
58
59#[derive(Debug, Serialize)]
60#[serde(untagged)]
61enum FlowDdlPayload {
62    Create(CreateFlowPayload),
63    Drop(DropFlowPayload),
64}
65
66/// A Flow DDL event with a schema shared by every lifecycle trigger.
67#[derive(Debug)]
68pub(crate) struct FlowDdlEvent {
69    event_type: &'static str,
70    catalog_name: Option<String>,
71    flow_name: Option<String>,
72    flow_id: Option<u32>,
73    payload: Option<FlowDdlPayload>,
74    event_context: Option<EventContext>,
75}
76
77impl FlowDdlEvent {
78    pub(crate) fn create_submitted(
79        catalog_name: &str,
80        flow_name: &str,
81        intent: CreateFlowEventIntent,
82        event_context: EventContext,
83    ) -> Self {
84        Self {
85            event_type: CREATE_FLOW_EVENT_TYPE,
86            catalog_name: Some(catalog_name.to_string()),
87            flow_name: Some(flow_name.to_string()),
88            flow_id: None,
89            payload: Some(FlowDdlPayload::Create(CreateFlowPayload {
90                version: PAYLOAD_VERSION,
91                or_replace: intent.or_replace,
92                create_if_not_exists: intent.create_if_not_exists,
93                expire_after: intent.expire_after,
94                eval_interval_secs: intent.eval_interval_secs,
95            })),
96            event_context: Some(event_context),
97        }
98    }
99
100    pub(crate) fn drop_submitted(
101        catalog_name: &str,
102        flow_name: &str,
103        flow_id: u32,
104        drop_if_exists: bool,
105        event_context: EventContext,
106    ) -> Self {
107        Self {
108            event_type: DROP_FLOW_EVENT_TYPE,
109            catalog_name: Some(catalog_name.to_string()),
110            flow_name: Some(flow_name.to_string()),
111            flow_id: Some(flow_id),
112            payload: Some(FlowDdlPayload::Drop(DropFlowPayload {
113                version: PAYLOAD_VERSION,
114                drop_if_exists,
115            })),
116            event_context: Some(event_context),
117        }
118    }
119
120    /// Builds a lightweight Create Flow lifecycle event.
121    pub(crate) fn create_lifecycle() -> Self {
122        Self::lifecycle(CREATE_FLOW_EVENT_TYPE)
123    }
124
125    /// Builds a successful Create Flow event containing only a resolved ID.
126    pub(crate) fn create_succeeded(flow_id: Option<u32>) -> Self {
127        Self {
128            flow_id,
129            ..Self::lifecycle(CREATE_FLOW_EVENT_TYPE)
130        }
131    }
132
133    /// Builds a lightweight Drop Flow lifecycle event.
134    pub(crate) fn drop_lifecycle() -> Self {
135        Self::lifecycle(DROP_FLOW_EVENT_TYPE)
136    }
137
138    fn lifecycle(event_type: &'static str) -> Self {
139        Self {
140            event_type,
141            catalog_name: None,
142            flow_name: None,
143            flow_id: None,
144            payload: None,
145            event_context: None,
146        }
147    }
148}
149
150impl Event for FlowDdlEvent {
151    fn event_type(&self) -> &str {
152        self.event_type
153    }
154
155    fn json_payload(&self) -> Result<serde_json::Value> {
156        match &self.payload {
157            Some(payload) => serde_json::to_value(payload).context(SerializeEventSnafu),
158            None => Ok(serde_json::Value::Null),
159        }
160    }
161
162    fn extra_schema(&self) -> Vec<ColumnSchema> {
163        column_schemas([
164            &CATALOG_NAME_COLUMN,
165            &FLOW_NAME_COLUMN,
166            &FLOW_ID_COLUMN,
167            &EVENT_CONTEXT_COLUMN,
168        ])
169    }
170
171    fn extra_rows(&self) -> Result<Vec<Row>> {
172        let event_context = self
173            .event_context
174            .as_ref()
175            .map(serde_json::to_value)
176            .transpose()
177            .context(SerializeEventSnafu)?;
178        Ok(vec![Row {
179            values: vec![
180                nullable_string(self.catalog_name.as_deref()),
181                nullable_string(self.flow_name.as_deref()),
182                nullable_value(self.flow_id.map(ValueData::U32Value)),
183                nullable_json(event_context.as_ref()),
184            ],
185        }])
186    }
187
188    fn as_any(&self) -> &dyn Any {
189        self
190    }
191}
