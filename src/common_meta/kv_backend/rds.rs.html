59#[cfg(any(feature = "pg_kvbackend", feature = "mysql_kvbackend"))]
60pub(crate) fn ensure_rustls_crypto_provider_installed() -> Result<()> {
61    RUSTLS_CRYPTO_PROVIDER_INIT
62        .get_or_init(|| {
63            if rustls::crypto::CryptoProvider::get_default().is_some() {
64                return Ok(());
65            }
66
67            match rustls::crypto::CryptoProvider::install_default(
68                rustls::crypto::aws_lc_rs::default_provider(),
69            ) {
70                Ok(()) => Ok(()),
71                Err(_provider) if rustls::crypto::CryptoProvider::get_default().is_some() => {
72                    Ok(())
73                }
74                Err(provider) => Err(format!(
75                    "Failed to install rustls CryptoProvider, existing default: {:?}, attempted provider: {:?}",
76                    rustls::crypto::CryptoProvider::get_default(),
77                    provider
78                )),
79            }
80        })
81        .clone()
82        .map_err(|err_msg| {
83            info!("Failed to install rustls crypto provider: {err_msg}");
84            UnexpectedSnafu { err_msg }.build()
85        })
86}

88/// Query executor for rds. It can execute queries or generate a transaction executor.
89#[async_trait::async_trait]
90pub trait Executor: Send + Sync {
91    type Transaction<'a>: 'a + Transaction<'a>
92    where
93        Self: 'a;
94
95    fn name() -> &'static str;
96
97    async fn query(&mut self, query: &str, params: &[&Vec<u8>]) -> Result<Vec<KeyValue>>;
98
99    /// Some queries don't need to return any result, such as `DELETE`.
100    async fn execute(&mut self, query: &str, params: &[&Vec<u8>]) -> Result<()> {
101        self.query(query, params).await?;
102        Ok(())
103    }
104
105    async fn txn_executor<'a>(&'a mut self) -> Result<Self::Transaction<'a>>;
106}
107
108/// Transaction query executor for rds. It can execute queries in transaction or commit the transaction.
109#[async_trait::async_trait]
110pub trait Transaction<'a>: Send + Sync {
111    async fn query(&mut self, query: &str, params: &[&Vec<u8>]) -> Result<Vec<KeyValue>>;
112
113    async fn execute(&mut self, query: &str, params: &[&Vec<u8>]) -> Result<()> {
114        self.query(query, params).await?;
115        Ok(())
116    }
117
118    async fn commit(self) -> Result<()>;
119}
120
121/// Factory for creating default and transaction query executors.
122#[async_trait::async_trait]
123pub trait ExecutorFactory<T: Executor>: Send + Sync {
124    /// Closes the connection pool to new acquisitions. This affects all shared pool handles.
125    /// MySQL waits for checked-out connections to be returned; PostgreSQL allows them to
126    /// finish and releases them when returned. Stop consumers before closing the backend.
127    async fn close(&self);
128
129    async fn default_executor(&self) -> Result<T>;
130
131    async fn txn_executor<'a>(&self, default_executor: &'a mut T) -> Result<T::Transaction<'a>>;
132}

147pub enum ExecutorImpl<'a, T: Executor + 'a> {
148    Default(T),
149    Txn(T::Transaction<'a>),
150}
151
152impl<T: Executor> ExecutorImpl<'_, T> {
153    async fn query(&mut self, query: &str, params: &Vec<&Vec<u8>>) -> Result<Vec<KeyValue>> {
154        match self {
155            Self::Default(executor) => executor.query(query, params).await,
156            Self::Txn(executor) => executor.query(query, params).await,
157        }
158    }
159
160    #[allow(dead_code)] // Used in #[cfg(feature = "mysql_kvbackend")]
161    async fn execute(&mut self, query: &str, params: &Vec<&Vec<u8>>) -> Result<()> {
162        match self {
163            Self::Default(executor) => executor.execute(query, params).await,
164            Self::Txn(executor) => executor.execute(query, params).await,
165        }
166    }
167
168    async fn commit(self) -> Result<()> {
169        match self {
170            Self::Txn(executor) => executor.commit().await,
171            _ => Ok(()),
172        }
173    }
174}

176#[async_trait::async_trait]
177pub trait KvQueryExecutor<T: Executor> {
178    async fn range_with_query_executor(
179        &self,
180        query_executor: &mut ExecutorImpl<'_, T>,
181        req: RangeRequest,
182    ) -> Result<RangeResponse>;
183
184    async fn put_with_query_executor(
185        &self,
186        query_executor: &mut ExecutorImpl<'_, T>,
187        req: PutRequest,
188    ) -> Result<PutResponse> {
189        let kv = KeyValue {
190            key: req.key,
191            value: req.value,
192        };
193        let mut res = self
194            .batch_put_with_query_executor(
195                query_executor,
196                BatchPutRequest {
197                    kvs: vec![kv],
198                    prev_kv: req.prev_kv,
199                },
200            )
201            .await?;
202
203        if !res.prev_kvs.is_empty() {
204            debug_assert!(req.prev_kv);
205            return Ok(PutResponse {
206                prev_kv: Some(res.prev_kvs.remove(0)),
207            });
208        }
209        Ok(PutResponse::default())
210    }
211
212    async fn batch_put_with_query_executor(
213        &self,
214        query_executor: &mut ExecutorImpl<'_, T>,
215        req: BatchPutRequest,
216    ) -> Result<BatchPutResponse>;
217
218    /// Batch get with certain client. It's needed for a client with transaction.
219    async fn batch_get_with_query_executor(
220        &self,
221        query_executor: &mut ExecutorImpl<'_, T>,
222        req: BatchGetRequest,
223    ) -> Result<BatchGetResponse>;
224
225    async fn delete_range_with_query_executor(
226        &self,
227        query_executor: &mut ExecutorImpl<'_, T>,
228        req: DeleteRangeRequest,
229    ) -> Result<DeleteRangeResponse>;
230
231    async fn batch_delete_with_query_executor(
232        &self,
233        query_executor: &mut ExecutorImpl<'_, T>,
234        req: BatchDeleteRequest,
235    ) -> Result<BatchDeleteResponse>;
236}

238impl<T, S, R> RdsStore<T, S, R>
239where
240    Self: KvQueryExecutor<T> + Send + Sync,
241    T: Executor + Send + Sync,
242    S: ExecutorFactory<T> + Send + Sync,
243{
244    async fn execute_txn_cmp(
245        &self,
246        query_executor: &mut ExecutorImpl<'_, T>,
247        cmp: &[Compare],
248    ) -> Result<bool> {
249        let batch_get_req = BatchGetRequest {
250            keys: cmp.iter().map(|c| c.key.clone()).collect(),
251        };
252        let res = self
253            .batch_get_with_query_executor(query_executor, batch_get_req)
254            .await?;
255        debug!("batch get res: {:?}", res);
256        let res_map = res
257            .kvs
258            .into_iter()
259            .map(|kv| (kv.key, kv.value))
260            .collect::<HashMap<Vec<u8>, Vec<u8>>>();
261        for c in cmp {
262            let value = res_map.get(&c.key);
263            if !c.compare_value(value) {
264                return Ok(false);
265            }
266        }
267        Ok(true)
268    }
269
270    /// Execute a batch of transaction operations. This function is only used for transactions with the same operation type.
271    async fn try_batch_txn(
272        &self,
273        query_executor: &mut ExecutorImpl<'_, T>,
274        txn_ops: &[TxnOp],
275    ) -> Result<Option<Vec<TxnOpResponse>>> {
276        if !check_txn_ops(txn_ops)? {
277            return Ok(None);
278        }
279        // Safety: txn_ops is not empty
280        match txn_ops.first().unwrap() {
281            TxnOp::Delete(_) => self.handle_batch_delete(query_executor, txn_ops).await,
282            TxnOp::Put(_, _) => self.handle_batch_put(query_executor, txn_ops).await,
283            TxnOp::Get(_) => self.handle_batch_get(query_executor, txn_ops).await,
284        }
285    }
286
287    async fn handle_batch_delete(
288        &self,
289        query_executor: &mut ExecutorImpl<'_, T>,
290        txn_ops: &[TxnOp],
291    ) -> Result<Option<Vec<TxnOpResponse>>> {
292        let mut batch_del_req = BatchDeleteRequest {
293            keys: vec![],
294            prev_kv: true,
295        };
296        for op in txn_ops {
297            if let TxnOp::Delete(key) = op {
298                batch_del_req.keys.push(key.clone());
299            }
300        }
301        let res = self
302            .batch_delete_with_query_executor(query_executor, batch_del_req)
303            .await?;
304        let res_map = res
305            .prev_kvs
306            .into_iter()
307            .map(|kv| (kv.key, kv.value))
308            .collect::<HashMap<Vec<u8>, Vec<u8>>>();
309        let mut resps = Vec::with_capacity(txn_ops.len());
310        for op in txn_ops {
311            if let TxnOp::Delete(key) = op {
312                let value = res_map.get(key);
313                resps.push(TxnOpResponse::ResponseDelete(DeleteRangeResponse {
314                    deleted: if value.is_some() { 1 } else { 0 },
315                    prev_kvs: vec![],
316                }));
317            }
318        }
319        Ok(Some(resps))
320    }
321
322    async fn handle_batch_put(
323        &self,
324        query_executor: &mut ExecutorImpl<'_, T>,
325        txn_ops: &[TxnOp],
326    ) -> Result<Option<Vec<TxnOpResponse>>> {
327        let mut batch_put_req = BatchPutRequest {
328            kvs: vec![],
329            prev_kv: false,
330        };
331        for op in txn_ops {
332            if let TxnOp::Put(key, value) = op {
333                batch_put_req.kvs.push(KeyValue {
334                    key: key.clone(),
335                    value: value.clone(),
336                });
337            }
338        }
339        let _ = self
340            .batch_put_with_query_executor(query_executor, batch_put_req)
341            .await?;
342        let mut resps = Vec::with_capacity(txn_ops.len());
343        for op in txn_ops {
344            if let TxnOp::Put(_, _) = op {
345                resps.push(TxnOpResponse::ResponsePut(PutResponse { prev_kv: None }));
346            }
347        }
348        Ok(Some(resps))
349    }
350
351    async fn handle_batch_get(
352        &self,
353        query_executor: &mut ExecutorImpl<'_, T>,
354        txn_ops: &[TxnOp],
355    ) -> Result<Option<Vec<TxnOpResponse>>> {
356        let mut batch_get_req = BatchGetRequest { keys: vec![] };
357        for op in txn_ops {
358            if let TxnOp::Get(key) = op {
359                batch_get_req.keys.push(key.clone());
360            }
361        }
362        let res = self
363            .batch_get_with_query_executor(query_executor, batch_get_req)
364            .await?;
365        let res_map = res
366            .kvs
367            .into_iter()
368            .map(|kv| (kv.key, kv.value))
369            .collect::<HashMap<Vec<u8>, Vec<u8>>>();
370        let mut resps = Vec::with_capacity(txn_ops.len());
371        for op in txn_ops {
372            if let TxnOp::Get(key) = op {
373                let value = res_map.get(key);
374                resps.push(TxnOpResponse::ResponseGet(RangeResponse {
375                    kvs: value
376                        .map(|v| {
377                            vec![KeyValue {
378                                key: key.clone(),
379                                value: v.clone(),
380                            }]
381                        })
382                        .unwrap_or_default(),
383                    more: false,
384                }));
385            }
386        }
387        Ok(Some(resps))
388    }
389
390    async fn execute_txn_op(
391        &self,
392        query_executor: &mut ExecutorImpl<'_, T>,
393        op: &TxnOp,
394    ) -> Result<TxnOpResponse> {
395        match op {
396            TxnOp::Put(key, value) => {
397                let res = self
398                    .put_with_query_executor(
399                        query_executor,
400                        PutRequest {
401                            key: key.clone(),
402                            value: value.clone(),
403                            prev_kv: false,
404                        },
405                    )
406                    .await?;
407                Ok(TxnOpResponse::ResponsePut(res))
408            }
409            TxnOp::Get(key) => {
410                let res = self
411                    .range_with_query_executor(
412                        query_executor,
413                        RangeRequest {
414                            key: key.clone(),
415                            range_end: vec![],
416                            limit: 1,
417                            keys_only: false,
418                        },
419                    )
420                    .await?;
421                Ok(TxnOpResponse::ResponseGet(res))
422            }
423            TxnOp::Delete(key) => {
424                let res = self
425                    .delete_range_with_query_executor(
426                        query_executor,
427                        DeleteRangeRequest {
428                            key: key.clone(),
429                            range_end: vec![],
430                            prev_kv: false,
431                        },
432                    )
433                    .await?;
434                Ok(TxnOpResponse::ResponseDelete(res))
435            }
436        }
437    }
438
439    async fn txn_inner(&self, txn: &KvTxn) -> Result<KvTxnResponse> {
440        let mut default_executor = self.executor_factory.default_executor().await?;
441        let mut txn_executor = ExecutorImpl::Txn(
442            self.executor_factory
443                .txn_executor(&mut default_executor)
444                .await?,
445        );
446        let mut success = true;
447        if txn.c_when {
448            success = self
449                .execute_txn_cmp(&mut txn_executor, &txn.req.compare)
450                .await?;
451        }
452        let mut responses = vec![];
453        if success && txn.c_then {
454            match self
455                .try_batch_txn(&mut txn_executor, &txn.req.success)
456                .await?
457            {
458                Some(res) => responses.extend(res),
459                None => {
460                    for txnop in &txn.req.success {
461                        let res = self.execute_txn_op(&mut txn_executor, txnop).await?;
462                        responses.push(res);
463                    }
464                }
465            }
466        } else if !success && txn.c_else {
467            match self
468                .try_batch_txn(&mut txn_executor, &txn.req.failure)
469                .await?
470            {
471                Some(res) => responses.extend(res),
472                None => {
473                    for txnop in &txn.req.failure {
474                        let res = self.execute_txn_op(&mut txn_executor, txnop).await?;
475                        responses.push(res);
476                    }
477                }
478            }
479        }
480
481        txn_executor.commit().await?;
482        Ok(KvTxnResponse {
483            responses,
484            succeeded: success,
485        })
486    }
487}

547#[async_trait::async_trait]
548impl<T, S, R> TxnService for RdsStore<T, S, R>
549where
550    Self: KvQueryExecutor<T> + Send + Sync,
551    T: Executor + 'static,
552    S: ExecutorFactory<T> + 'static,
553{
554    type Error = Error;
555
556    async fn txn(&self, txn: KvTxn) -> Result<KvTxnResponse> {
557        let _timer = METRIC_META_TXN_REQUEST
558            .with_label_values(&[T::name(), "txn"])
559            .start_timer();
560
561        let mut backoff = ExponentialBuilder::default()
562            .with_min_delay(Duration::from_millis(10))
563            .with_max_delay(Duration::from_millis(200))
564            .with_max_times(self.txn_retry_count)
565            .build();
566
567        loop {
568            match self.txn_inner(&txn).await {
569                Ok(res) => return Ok(res),
570                Err(e) => {
571                    if e.is_serialization_error() {
572                        let d = backoff.next();
573                        if let Some(d) = d {
574                            tokio::time::sleep(d).await;
575                            continue;
576                        }
577                        break;
578                    } else {
579                        return Err(e);
580                    }
581                }
582            }
583        }
584
585        RdsTransactionRetryFailedSnafu {}.fail()
586    }
587
588    fn max_txn_ops(&self) -> usize {
589        self.max_txn_ops
590    }
591}

607/// Checks if the transaction operations are the same type.
608fn check_txn_ops(txn_ops: &[TxnOp]) -> Result<bool> {
609    if txn_ops.is_empty() {
610        return Ok(false);
611    }
612    let same = txn_ops.windows(2).all(|a| {
613        matches!(
614            (&a[0], &a[1]),
615            (TxnOp::Put(_, _), TxnOp::Put(_, _))
616                | (TxnOp::Get(_), TxnOp::Get(_))
617                | (TxnOp::Delete(_), TxnOp::Delete(_))
618        )
619    });
620    Ok(same)
621}

623#[macro_export]
624macro_rules! record_rds_sql_execute_elapsed {
625    ($result:expr, $label_store:expr,$label_op:expr,$label_type:expr) => {{
626        let timer = std::time::Instant::now();
627        $result
628            .inspect(|_| {
629                $crate::metrics::RDS_SQL_EXECUTE_ELAPSED
630                    .with_label_values(&[$label_store, "success", $label_op, $label_type])
631                    .observe(timer.elapsed().as_millis() as f64)
632            })
633            .inspect_err(|_| {
634                $crate::metrics::RDS_SQL_EXECUTE_ELAPSED
635                    .with_label_values(&[$label_store, "error", $label_op, $label_type])
636                    .observe(timer.elapsed().as_millis() as f64);
637            })
638    }};
