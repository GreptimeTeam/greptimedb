74/// Type of range template.
75#[derive(Debug, Clone, Copy, AsRefStr)]
76enum RangeTemplateType {
77    Point,
78    Range,
79    Full,
80    LeftBounded,
81    Prefix,
82}
83
84/// Builds params for the given range template type.
85impl RangeTemplateType {
86    /// Builds the parameters for the given range template type.
87    /// You can check out the conventions at [RangeRequest]
88    fn build_params(&self, mut key: Vec<u8>, range_end: Vec<u8>) -> Vec<Vec<u8>> {
89        match self {
90            RangeTemplateType::Point => vec![key],
91            RangeTemplateType::Range => vec![key, range_end],
92            RangeTemplateType::Full => vec![],
93            RangeTemplateType::LeftBounded => vec![key],
94            RangeTemplateType::Prefix => {
95                key.push(b'%');
96                vec![key]
97            }
98        }
99    }
100}

102/// Templates for range request.
103#[derive(Debug, Clone)]
104struct RangeTemplate {
105    point: String,
106    range: String,
107    full: String,
108    left_bounded: String,
109    prefix: String,
110}
111
112impl RangeTemplate {
113    /// Gets the template for the given type.
114    fn get(&self, typ: RangeTemplateType) -> &str {
115        match typ {
116            RangeTemplateType::Point => &self.point,
117            RangeTemplateType::Range => &self.range,
118            RangeTemplateType::Full => &self.full,
119            RangeTemplateType::LeftBounded => &self.left_bounded,
120            RangeTemplateType::Prefix => &self.prefix,
121        }
122    }
123
124    /// Adds limit to the template.
125    fn with_limit(template: &str, limit: i64) -> String {
126        if limit == 0 {
127            return format!("{};", template);
128        }
129        format!("{} LIMIT {};", template, limit)
130    }
131}

133fn is_prefix_range(start: &[u8], end: &[u8]) -> bool {
134    if start.len() != end.len() {
135        return false;
136    }
137    let l = start.len();
138    let same_prefix = start[0..l - 1] == end[0..l - 1];
139    if let (Some(rhs), Some(lhs)) = (start.last(), end.last()) {
140        return same_prefix && (*rhs + 1) == *lhs;
141    }
142    false
143}
144
145/// Determine the template type for range request.
146fn range_template(key: &[u8], range_end: &[u8]) -> RangeTemplateType {
147    match (key, range_end) {
148        (_, &[]) => RangeTemplateType::Point,
149        (EMPTY, EMPTY) => RangeTemplateType::Full,
150        (_, EMPTY) => RangeTemplateType::LeftBounded,
151        (start, end) => {
152            if is_prefix_range(start, end) {
153                RangeTemplateType::Prefix
154            } else {
155                RangeTemplateType::Range
156            }
157        }
158    }
159}

161/// Generate in placeholders for MySQL.
162fn mysql_generate_in_placeholders(from: usize, to: usize) -> Vec<String> {
163    (from..=to).map(|_| "?".to_string()).collect()

168    table_name: &'a str,
169}

171impl<'a> MySqlTemplateFactory<'a> {
172    /// Creates a new [`SqlTemplateFactory`] with the given table name.
173    fn new(table_name: &'a str) -> Self {
174        Self { table_name }
175    }
176
177    /// Builds the template set for the given table name.
178    fn build(&self) -> MySqlTemplateSet {
179        let table_name = self.table_name;
180        // Some of queries don't end with `;`, because we need to add `LIMIT` clause.
181        MySqlTemplateSet {
182            table_name: table_name.to_string(),
183            create_table_statement: format!(
184                // Cannot be more than 3072 bytes in PRIMARY KEY
185                "CREATE TABLE IF NOT EXISTS `{table_name}`(k VARBINARY(3072) PRIMARY KEY, v MEDIUMBLOB);",
186            ),
187            show_create_table_statement: format!("SHOW CREATE TABLE `{table_name}`"),
188            alter_value_column_statement: format!(
189                "ALTER TABLE `{table_name}` MODIFY COLUMN v MEDIUMBLOB;"
190            ),
191            range_template: RangeTemplate {
192                point: format!("SELECT k, v FROM `{table_name}` WHERE k = ?"),
193                range: format!("SELECT k, v FROM `{table_name}` WHERE k >= ? AND k < ? ORDER BY k"),
194                full: format!("SELECT k, v FROM `{table_name}` ORDER BY k"),
195                left_bounded: format!("SELECT k, v FROM `{table_name}` WHERE k >= ? ORDER BY k"),
196                prefix: format!("SELECT k, v FROM `{table_name}` WHERE k LIKE ? ORDER BY k"),
197            },
198            delete_template: RangeTemplate {
199                point: format!("DELETE FROM `{table_name}` WHERE k = ?;"),
200                range: format!("DELETE FROM `{table_name}` WHERE k >= ? AND k < ?;"),
201                full: format!("DELETE FROM `{table_name}`"),
202                left_bounded: format!("DELETE FROM `{table_name}` WHERE k >= ?;"),
203                prefix: format!("DELETE FROM `{table_name}` WHERE k LIKE ?;"),
204            },
205        }
206    }
207}

220impl MySqlTemplateSet {
221    /// Generates the sql for batch get.
222    fn generate_batch_get_query(&self, key_len: usize) -> String {
223        let table_name = &self.table_name;
224        let in_clause = mysql_generate_in_placeholders(1, key_len).join(", ");
225        format!(
226            "SELECT k, v FROM `{table_name}` WHERE k in ({});",
227            in_clause
228        )
229    }
230
231    /// Generates the sql for batch delete.
232    fn generate_batch_delete_query(&self, key_len: usize) -> String {
233        let table_name = &self.table_name;
234        let in_clause = mysql_generate_in_placeholders(1, key_len).join(", ");
235        format!("DELETE FROM `{table_name}` WHERE k in ({});", in_clause)
236    }
237
238    /// Generates the sql for batch upsert.
239    /// For MySQL, it also generates a select query to get the previous values.
240    fn generate_batch_upsert_query(&self, kv_len: usize) -> (String, String) {
241        let table_name = &self.table_name;
242        let in_placeholders: Vec<String> = (1..=kv_len).map(|_| "?".to_string()).collect();
243        let in_clause = in_placeholders.join(", ");
244        let mut values_placeholders = Vec::new();
245        for _ in 0..kv_len {
246            values_placeholders.push("(?, ?)".to_string());
247        }
248        let values_clause = values_placeholders.join(", ");
249
250        (
251            format!(r#"SELECT k, v FROM `{table_name}` WHERE k IN ({in_clause})"#,),
252            format!(
253                r#"INSERT INTO `{table_name}` (k, v) VALUES {values_clause} ON DUPLICATE KEY UPDATE v = VALUES(v);"#,
254            ),
255        )
256    }
257}

259#[async_trait::async_trait]
260impl Executor for MySqlClient {
261    type Transaction<'a>
262        = MySqlTxnClient
263    where
264        Self: 'a;
265
266    fn name() -> &'static str {
267        "MySql"
268    }
269
270    async fn query(&mut self, raw_query: &str, params: &[&Vec<u8>]) -> Result<Vec<KeyValue>> {
271        // Safety: `raw_query` is generated by `MySqlTemplateSet`, and all dynamic data
272        // is passed via bind parameters.
273        let query = sqlx::query(AssertSqlSafe(raw_query));
274        let query = params.iter().fold(query, |query, param| query.bind(param));
275        let rows = query
276            .fetch_all(&**self)
277            .await
278            .context(MySqlExecutionSnafu { sql: raw_query })?;
279        Ok(rows.into_iter().map(key_value_from_row).collect())
280    }
281
282    async fn execute(&mut self, raw_query: &str, params: &[&Vec<u8>]) -> Result<()> {
283        // Safety: `raw_query` is generated by `MySqlTemplateSet`, and all dynamic data
284        // is passed via bind parameters.
285        let query = sqlx::query(AssertSqlSafe(raw_query));
286        let query = params.iter().fold(query, |query, param| query.bind(param));
287        query
288            .execute(&**self)
289            .await
290            .context(MySqlExecutionSnafu { sql: raw_query })?;
291        Ok(())
292    }
293
294    async fn txn_executor<'a>(&'a mut self) -> Result<Self::Transaction<'a>> {
295        // MySQL requires SET TRANSACTION separately from BEGIN. Execute both on
296        // the connection acquired by begin_with so the isolation level applies.
297        let txn = self
298            .begin_with("SET TRANSACTION ISOLATION LEVEL SERIALIZABLE; BEGIN")
299            .await
300            .context(MySqlTransactionSnafu { operation: "begin" })?;
301        Ok(MySqlTxnClient(txn))
302    }
303
304    async fn read_only_txn_executor<'a>(&'a mut self) -> Result<Self::Transaction<'a>> {
305        let txn = self
306            .begin_with("SET TRANSACTION ISOLATION LEVEL REPEATABLE READ, READ ONLY; BEGIN")
307            .await
308            .context(MySqlTransactionSnafu {
309                operation: "begin read-only",
310            })?;
311        Ok(MySqlTxnClient(txn))
312    }
313}

315#[async_trait::async_trait]
316impl Transaction<'_> for MySqlTxnClient {
317    async fn query(&mut self, raw_query: &str, params: &[&Vec<u8>]) -> Result<Vec<KeyValue>> {
318        // Safety: `raw_query` is generated by `MySqlTemplateSet`, and all dynamic data
319        // is passed via bind parameters.
320        let query = sqlx::query(AssertSqlSafe(raw_query));
321        let query = params.iter().fold(query, |query, param| query.bind(param));
322        // As said in https://docs.rs/sqlx/latest/sqlx/trait.Executor.html, we need a `&mut *transaction`. Weird.
323        let rows = query
324            .fetch_all(&mut *(self.0))
325            .await
326            .context(MySqlExecutionSnafu { sql: raw_query })?;
327        Ok(rows.into_iter().map(key_value_from_row).collect())
328    }
329
330    async fn execute(&mut self, raw_query: &str, params: &[&Vec<u8>]) -> Result<()> {
331        // Safety: `raw_query` is generated by `MySqlTemplateSet`, and all dynamic data
332        // is passed via bind parameters.
333        let query = sqlx::query(AssertSqlSafe(raw_query));
334        let query = params.iter().fold(query, |query, param| query.bind(param));
335        // As said in https://docs.rs/sqlx/latest/sqlx/trait.Executor.html, we need a `&mut *transaction`. Weird.
336        query
337            .execute(&mut *(self.0))
338            .await
339            .context(MySqlExecutionSnafu { sql: raw_query })?;
340        Ok(())
341    }
342
343    /// Caution: sqlx will stuck on the query if two transactions conflict with each other.
344    /// Don't know if it's a feature or it depends on the database. Be careful.
345    async fn commit(self) -> Result<()> {
346        self.0.commit().await.context(MySqlTransactionSnafu {
347            operation: "commit",
348        })?;
349        Ok(())
350    }

379#[async_trait::async_trait]
380impl KvQueryExecutor<MySqlClient> for MySqlStore {
381    async fn range_with_query_executor(
382        &self,
383        query_executor: &mut ExecutorImpl<'_, MySqlClient>,
384        req: RangeRequest,
385    ) -> Result<RangeResponse> {
386        let template_type = range_template(&req.key, &req.range_end);
387        let template = self.sql_template_set.range_template.get(template_type);
388        let params = template_type.build_params(req.key, req.range_end);
389        let params_ref = params.iter().collect::<Vec<_>>();
390        // Always add 1 to limit to check if there is more data
391        let query =
392            RangeTemplate::with_limit(template, if req.limit == 0 { 0 } else { req.limit + 1 });
393        let limit = req.limit as usize;
394        debug!("query: {:?}, params: {:?}", query, params);
395        let mut kvs = crate::record_rds_sql_execute_elapsed!(
396            query_executor.query(&query, &params_ref).await,
397            MYSQL_STORE_NAME,
398            RDS_STORE_OP_RANGE_QUERY,
399            template_type.as_ref()
400        )?;
401        if req.keys_only {
402            kvs.iter_mut().for_each(|kv| kv.value = vec![]);
403        }
404        // If limit is 0, we always return all data
405        if limit == 0 || kvs.len() <= limit {
406            return Ok(RangeResponse { kvs, more: false });
407        }
408        // If limit is greater than the number of rows, we remove the last row and set more to true
409        let removed = kvs.pop();
410        debug_assert!(removed.is_some());
411        Ok(RangeResponse { kvs, more: true })
412    }

414    async fn batch_put_with_query_executor(
415        &self,
416        query_executor: &mut ExecutorImpl<'_, MySqlClient>,
417        req: BatchPutRequest,
418    ) -> Result<BatchPutResponse> {
419        let kvs = &req.kvs;
420        if kvs.is_empty() {
421            return Ok(BatchPutResponse::default());
422        }
423        if (req.prev_kv || kvs.len() > (*RDS_BATCH_SIZE).min(MAX_BATCH_PUT_KVS))
424            && let ExecutorImpl::Default(executor) = query_executor
425        {
426            let mut txn = ExecutorImpl::Txn(executor.txn_executor().await?);
427            let response = self.put_chunks(&mut txn, kvs, req.prev_kv).await?;
428            txn.commit().await?;
429            return Ok(response);
430        }
431        self.put_chunks(query_executor, kvs, req.prev_kv).await
432    }

434    async fn batch_get_with_query_executor(
435        &self,
436        query_executor: &mut ExecutorImpl<'_, MySqlClient>,
437        req: BatchGetRequest,
438    ) -> Result<BatchGetResponse> {
439        let mut keys = req.keys;
440        keys.sort_unstable();
441        keys.dedup();
442        let chunk_size = *RDS_BATCH_SIZE;
443        if keys.len() > chunk_size
444            && let ExecutorImpl::Default(client) = query_executor
445        {
446            // Share one snapshot without serializable read conflicts or row locks.
447            let mut transaction = ExecutorImpl::Txn(client.read_only_txn_executor().await?);
448            let response = self.get_chunks(&mut transaction, &keys, chunk_size).await?;
449            transaction.commit().await?;
450            return Ok(response);
451        }
452        self.get_chunks(query_executor, &keys, chunk_size).await
453    }

455    async fn delete_range_with_query_executor(
456        &self,
457        query_executor: &mut ExecutorImpl<'_, MySqlClient>,
458        req: DeleteRangeRequest,
459    ) -> Result<DeleteRangeResponse> {
460        // Since we need to know the number of deleted keys, we have no fast path here.
461        // Should use transaction to ensure atomicity.
462        if let ExecutorImpl::Default(query_executor) = query_executor {
463            let txn = query_executor.txn_executor().await?;
464            let mut txn = ExecutorImpl::Txn(txn);
465            let res = self.delete_range_with_query_executor(&mut txn, req).await;
466            txn.commit().await?;
467            return res;
468        }
469        let range_get_req = RangeRequest {
470            key: req.key.clone(),
471            range_end: req.range_end.clone(),
472            limit: 0,
473            keys_only: false,
474        };
475        let prev_kvs = self
476            .range_with_query_executor(query_executor, range_get_req)
477            .await?
478            .kvs;
479        let template_type = range_template(&req.key, &req.range_end);
480        let template = self.sql_template_set.delete_template.get(template_type);
481        let params = template_type.build_params(req.key, req.range_end);
482        let params_ref = params.iter().map(|x| x as _).collect::<Vec<_>>();
483        crate::record_rds_sql_execute_elapsed!(
484            query_executor.execute(template, &params_ref).await,
485            MYSQL_STORE_NAME,
486            RDS_STORE_OP_RANGE_DELETE,
487            template_type.as_ref()
488        )?;
489        let mut resp = DeleteRangeResponse::new(prev_kvs.len() as i64);
490        if req.prev_kv {
491            resp.with_prev_kvs(prev_kvs);
492        }
493        Ok(resp)
494    }
495
496    async fn batch_delete_with_query_executor(
497        &self,
498        query_executor: &mut ExecutorImpl<'_, MySqlClient>,
499        req: BatchDeleteRequest,
500    ) -> Result<BatchDeleteResponse> {
501        let keys = req.keys;
502        if keys.is_empty() {
503            return Ok(BatchDeleteResponse::default());
504        }
505        if (req.prev_kv || keys.len() > *RDS_BATCH_SIZE)
506            && let ExecutorImpl::Default(executor) = query_executor
507        {
508            let mut txn = ExecutorImpl::Txn(executor.txn_executor().await?);
509            let response = self.delete_chunks(&mut txn, &keys, req.prev_kv).await?;
510            txn.commit().await?;
511            return Ok(response);
512        }
513        self.delete_chunks(query_executor, &keys, req.prev_kv).await
514    }
515}

517impl MySqlStore {
518    async fn get_chunks(
519        &self,
520        query_executor: &mut ExecutorImpl<'_, MySqlClient>,
521        keys: &[Vec<u8>],
522        chunk_size: usize,
523    ) -> Result<BatchGetResponse> {
524        let mut kvs = Vec::new();
525        let mut params = Vec::with_capacity(keys.len().min(chunk_size));
526        for keys in keys.chunks(chunk_size) {
527            let query = self.sql_template_set.generate_batch_get_query(keys.len());
528            params.clear();
529            params.extend(keys);
530            kvs.extend(crate::record_rds_sql_execute_elapsed!(
531                query_executor.query(&query, &params).await,
532                MYSQL_STORE_NAME,
533                RDS_STORE_OP_BATCH_GET,
534                ""
535            )?);
536        }
537        Ok(BatchGetResponse { kvs })
538    }
539
540    async fn put_chunks(
541        &self,
542        executor: &mut ExecutorImpl<'_, MySqlClient>,
543        kvs: &[KeyValue],
544        prev_kv: bool,
545    ) -> Result<BatchPutResponse> {
546        let mut prev_kvs = Vec::new();
547        let chunk_size = (*RDS_BATCH_SIZE).min(MAX_BATCH_PUT_KVS);
548        let mut params = Vec::with_capacity(kvs.len().min(chunk_size) * 2);
549        for chunk in kvs.chunks(chunk_size) {
550            let (select, update) = self
551                .sql_template_set
552                .generate_batch_upsert_query(chunk.len());
553            if prev_kv {
554                params.clear();
555                params.extend(chunk.iter().map(|kv| &kv.key));
556                prev_kvs.extend(crate::record_rds_sql_execute_elapsed!(
557                    executor.query(&select, &params).await,
558                    MYSQL_STORE_NAME,
559                    RDS_STORE_OP_BATCH_PUT,
560                    ""
561                )?);
562            }
563            params.clear();
564            params.extend(chunk.iter().flat_map(|kv| [&kv.key, &kv.value]));
565            crate::record_rds_sql_execute_elapsed!(
566                executor.execute(&update, &params).await,
567                MYSQL_STORE_NAME,
568                RDS_STORE_OP_BATCH_PUT,
569                ""
570            )?;
571        }
572        Ok(BatchPutResponse { prev_kvs })
573    }
574
575    async fn delete_chunks(
576        &self,
577        executor: &mut ExecutorImpl<'_, MySqlClient>,
578        keys: &[Vec<u8>],
579        prev_kv: bool,
580    ) -> Result<BatchDeleteResponse> {
581        let mut prev_kvs = Vec::new();
582        let mut params = Vec::with_capacity(keys.len().min(*RDS_BATCH_SIZE));
583        for chunk in keys.chunks(*RDS_BATCH_SIZE) {
584            params.clear();
585            params.extend(chunk);
586            if prev_kv {
587                let select = self.sql_template_set.generate_batch_get_query(chunk.len());
588                prev_kvs.extend(crate::record_rds_sql_execute_elapsed!(
589                    executor.query(&select, &params).await,
590                    MYSQL_STORE_NAME,
591                    RDS_STORE_OP_BATCH_DELETE,
592                    ""
593                )?);
594            }
595            let delete = self
596                .sql_template_set
597                .generate_batch_delete_query(chunk.len());

639        pool: &Pool<MySql>,

641    ) -> Result<()> {
642        let table_name = &sql_template_set.table_name;
643        let create_table_sql = Self::fetch_create_table_sql(pool, sql_template_set)
644            .await?

660            }
661            Some(ValueBlobType::MediumBlob | ValueBlobType::LongBlob) => {
662                debug!("MySQL metadata value column for `{table_name}` is already compatible");
663            }
664            None => {
665                warn!(
666                    "Failed to determine MySQL metadata value column type from table definition for `{table_name}`, skip automatic MEDIUMBLOB upgrade"

668            }
669        }
