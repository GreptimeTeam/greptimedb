31/// [NodeExpiryListener] periodically checks all node info in memory and removes
32/// expired node info to prevent memory leak.
33pub struct NodeExpiryListener {
34    handle: Mutex<Option<JoinHandle<()>>>,
35    max_idle_time: Duration,
36    role_idle_limits: HashMap<Role, Duration>,
37    in_memory: ResettableKvBackendRef,
38}
39
40impl Drop for NodeExpiryListener {
41    fn drop(&mut self) {
42        self.stop();
43    }
44}
45
46impl NodeExpiryListener {
47    pub fn new(max_idle_time: Duration, in_memory: ResettableKvBackendRef) -> Self {
48        Self {
49            handle: Mutex::new(None),
50            max_idle_time,
51            role_idle_limits: HashMap::new(),
52            in_memory,
53        }
54    }
55
56    /// Caps the idle time for registrations of the given role.
57    pub fn with_role_idle_limit(mut self, role: Role, limit: Duration) -> Self {
58        self.role_idle_limits
59            .insert(role, limit.min(self.max_idle_time));
60        self
61    }
62
63    /// Starts periodic cleanup on the active metasrv.
64    pub fn start(&self) {
65        let mut handle = self.handle.lock().unwrap();
66        if handle.is_none() {
67            let in_memory = self.in_memory.clone();
68
69            let max_idle_time = self.max_idle_time;
70            let role_idle_limits = self.role_idle_limits.clone();
71            let ticker_loop = tokio::spawn(async move {
72                // Run clean task every minute.
73                let mut interval = interval(Duration::from_secs(60));
74                interval.set_missed_tick_behavior(MissedTickBehavior::Skip);
75                loop {
76                    interval.tick().await;
77                    if let Err(e) = Self::clean_expired_nodes(
78                        &in_memory,
79                        max_idle_time,
80                        &role_idle_limits,
81                        common_time::util::current_time_millis(),
82                    )
83                    .await
84                    {
85                        error!(e; "Failed to clean expired node");
86                    }
87                }
88            });
89            *handle = Some(ticker_loop);
90        }
91    }
92
93    /// Stops periodic cleanup.
94    pub fn stop(&self) {
95        if let Some(handle) = self.handle.lock().unwrap().take() {
96            handle.abort();
97            info!("Node expiry listener stopped")
98        }
99    }
100
101    /// Cleans expired nodes from memory.
102    async fn clean_expired_nodes(
103        in_memory: &ResettableKvBackendRef,
104        max_idle_time: Duration,
105        role_idle_limits: &HashMap<Role, Duration>,
106        now: i64,
107    ) -> error::Result<()> {
108        let req = RangeRequest::new().with_prefix(NodeInfoKey::key_prefix());
109        for KeyValue { key, value } in in_memory.range(req).await?.kvs {
110            let Ok(node_key) = NodeInfoKey::try_from(key.clone()).inspect_err(|e| {
111                warn!(e; "Unrecognized node info key");
112            }) else {
113                continue;
114            };
115            let Ok(info) = serde_json::from_slice::<NodeInfo>(&value).inspect_err(|e| {
116                warn!(e; "Unrecognized node info value");
117            }) else {
118                continue;
119            };
120            let limit = role_idle_limits
121                .get(&node_key.role)
122                .unwrap_or(&max_idle_time);
123            let idle_ms = now.saturating_sub(info.last_activity_ts).max(0) as u64;
124            if Duration::from_millis(idle_ms) > *limit {
125                match in_memory.txn(delete_if_unchanged(key, value)).await {
126                    Ok(resp) if resp.succeeded => {
127                        debug!("Deleted expired node key: {:?}", node_key)
128                    }
129                    Err(e) => warn!(e; "Failed to delete expired node: {:?}", node_key),
130                    _ => {}
131                }
132            }
133        }
134        Ok(())
135    }
136}
137
138// A heartbeat may have refreshed the registration since the scan.
139fn delete_if_unchanged(key: Vec<u8>, value: Vec<u8>) -> Txn {
140    Txn::new()
141        .when(vec![Compare::with_value(
142            key.clone(),
143            CompareOp::Equal,
144            value,
145        )])
146        .and_then(vec![TxnOp::Delete(key)])
147}

149#[async_trait::async_trait]
150impl LeadershipChangeListener for NodeExpiryListener {
151    fn name(&self) -> &str {
152        "NodeExpiryListener"
153    }
154
155    async fn on_leader_start(&self) -> error::Result<()> {
156        self.start();
157        info!(
158            "On leader start, node expiry listener started with max idle time: {:?}",
159            self.max_idle_time
160        );
161        Ok(())
162    }
163
164    async fn on_leader_stop(&self) -> error::Result<()> {
165        self.stop();
166        info!("On leader stop, node expiry listener stopped");
167        Ok(())
168    }
169}

171#[cfg(test)]
172mod tests {
173    use std::sync::Arc;
174
175    use super::*;
176    use crate::kv_backend::memory::MemoryKvBackend;
177    use crate::rpc::store::PutRequest;
178
179    #[tokio::test]
180    async fn test_cleanup_preserves_refreshed_registration() {
181        let backend: ResettableKvBackendRef = Arc::new(MemoryKvBackend::new());
182        let key = Vec::from(&NodeInfoKey {
183            role: Role::Datanode,
184            node_id: 1,
185        });
186        let scanned = b"first heartbeat".to_vec();
187        let refreshed = b"next heartbeat".to_vec();
188        for value in [&scanned, &refreshed] {
189            backend
190                .put(
191                    PutRequest::new()
192                        .with_key(key.clone())
193                        .with_value(value.clone()),
194                )
195                .await
196                .unwrap();
197        }
198        let response = backend
199            .txn(delete_if_unchanged(key.clone(), scanned))
200            .await
201            .unwrap();
202        assert!(!response.succeeded);
203        assert_eq!(backend.get(&key).await.unwrap().unwrap().value, refreshed);
204    }
205
206    #[tokio::test]
207    async fn test_role_idle_limit() {
208        let now = 100_000_000;
209        let cases = [
210            (86_400, None, vec![1, 2, 3, 4]),
211            (86_400, Some(60), vec![1, 3, 4]),
212            (30, Some(60), vec![4]),
213        ];
214        for (global_secs, datanode_secs, expected) in cases {
215            let backend: ResettableKvBackendRef = Arc::new(MemoryKvBackend::new());
216            let mut listener =
217                NodeExpiryListener::new(Duration::from_secs(global_secs), backend.clone());
218            if let Some(limit) = datanode_secs {
219                listener =
220                    listener.with_role_idle_limit(Role::Datanode, Duration::from_secs(limit));
221            }
222            let mut keys = Vec::new();
223            for (node_id, role, idle_ms) in [
224                (1, Role::Datanode, 60_000),
225                (2, Role::Datanode, 60_001),
226                (3, Role::Frontend, 86_400_000),
227                (4, Role::Flownode, 30_000),
228                (5, Role::Flownode, 86_400_001),
229            ] {
230                let key = Vec::from(&NodeInfoKey { role, node_id });
231                let value = serde_json::json!({
232                    "peer": {"id": node_id, "addr": "localhost"},
233                    "last_activity_ts": now - idle_ms,
234                    "status": "Standalone",
235                    "version": "", "git_commit": "", "start_time_ms": 0
236                });
237                backend
238                    .put(
239                        PutRequest::new()
240                            .with_key(key.clone())
241                            .with_value(value.to_string()),
242                    )
243                    .await
244                    .unwrap();
245                keys.push((node_id, key));
246            }
247
248            NodeExpiryListener::clean_expired_nodes(
249                &backend,
250                listener.max_idle_time,
251                &listener.role_idle_limits,
252                now,
253            )
254            .await
255            .unwrap();
256            let mut remaining = Vec::new();
257            for (id, key) in keys {
258                if backend.get(&key).await.unwrap().is_some() {
259                    remaining.push(id);
260                }
261            }
262            assert_eq!(remaining, expected);
263        }
264    }
