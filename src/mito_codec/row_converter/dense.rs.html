74    fn estimated_size_by_type(data_type: &ConcreteDataType) -> usize {
75        match data_type {
76            ConcreteDataType::Boolean(_) => 2,
77            ConcreteDataType::Int8(_) | ConcreteDataType::UInt8(_) => 2,
78            ConcreteDataType::Int16(_) | ConcreteDataType::UInt16(_) => 3,
79            ConcreteDataType::Int32(_) | ConcreteDataType::UInt32(_) => 5,
80            ConcreteDataType::Int64(_) | ConcreteDataType::UInt64(_) => 9,
81            ConcreteDataType::Float32(_) => 5,
82            ConcreteDataType::Float64(_) => 9,
83            ConcreteDataType::Binary(_)
84            | ConcreteDataType::Json(_)
85            | ConcreteDataType::Vector(_) => 11,
86            ConcreteDataType::String(_) => 11, // a non-empty string takes at least 11 bytes.
87            ConcreteDataType::Date(_) => 5,
88            ConcreteDataType::Timestamp(_) => 10,
89            ConcreteDataType::Time(_) => 10,
90            ConcreteDataType::Duration(_) => 10,
91            ConcreteDataType::Interval(_) => 18,
92            ConcreteDataType::Decimal128(_) => 19,
93            ConcreteDataType::Null(_)
94            | ConcreteDataType::List(_)
95            | ConcreteDataType::Struct(_)
96            | ConcreteDataType::Dictionary(_) => 0,
97        }
98    }

109    fn serialize_by_type(
110        data_type: &ConcreteDataType,
111        serializer: &mut Serializer<&mut Vec<u8>>,
112        value: &ValueRef,
113    ) -> Result<()> {
114        macro_rules! cast_value_and_serialize {
115            (
116                $data_type: ident;
117                $serializer: ident;
118                $(
119                    $ty: ident, $f: ident
120                ),*
121            ) => {
122                match $data_type {
123                $(
124                    ConcreteDataType::$ty(_) => {
125                        paste!{
126                            value
127                            .[<try_into_ $f>]()
128                            .context(FieldTypeMismatchSnafu)?
129                            .serialize($serializer)
130                            .context(SerializeFieldSnafu)?;
131                        }
132                    }
133                )*
134                    ConcreteDataType::Timestamp(_) => {
135                        let timestamp = value.try_into_timestamp().context(FieldTypeMismatchSnafu)?;
136                        timestamp
137                            .map(|t|t.value())
138                            .serialize($serializer)
139                            .context(SerializeFieldSnafu)?;
140                    }
141                    ConcreteDataType::Interval(IntervalType::YearMonth(_)) => {
142                        let interval = value.try_into_interval_year_month().context(FieldTypeMismatchSnafu)?;
143                        interval.map(|i| i.to_i32())
144                            .serialize($serializer)
145                            .context(SerializeFieldSnafu)?;
146                    }
147                    ConcreteDataType::Interval(IntervalType::DayTime(_)) => {
148                        let interval = value.try_into_interval_day_time().context(FieldTypeMismatchSnafu)?;
149                        interval.map(|i| i.to_i64())
150                            .serialize($serializer)
151                            .context(SerializeFieldSnafu)?;
152                    }
153                    ConcreteDataType::Interval(IntervalType::MonthDayNano(_)) => {
154                        let interval = value.try_into_interval_month_day_nano().context(FieldTypeMismatchSnafu)?;
155                        interval.map(|i| i.to_i128())
156                            .serialize($serializer)
157                            .context(SerializeFieldSnafu)?;
158                    }
159                    ConcreteDataType::List(_) |
160                    ConcreteDataType::Struct(_) |
161                    ConcreteDataType::Dictionary(_) |
162                    ConcreteDataType::Null(_) => {
163                        return error::NotSupportedFieldSnafu {
164                            data_type: $data_type.clone()
165                        }.fail()
166                    }
167                }
168            };
169        }
170        cast_value_and_serialize!(data_type; serializer;
171            Boolean, boolean,
172            Binary, binary,
173            Int8, i8,
174            UInt8, u8,
175            Int16, i16,
176            UInt16, u16,
177            Int32, i32,
178            UInt32, u32,
179            Int64, i64,
180            UInt64, u64,
181            Float32, f32,
182            Float64, f64,
183            String, string,
184            Date, date,
185            Time, time,
186            Duration, duration,
187            Decimal128, decimal128,
188            Json, binary,
189            Vector, binary
190        );
191
192        Ok(())
193    }

200    fn deserialize_by_type<B: Buf>(
201        data_type: &ConcreteDataType,
202        deserializer: &mut Deserializer<B>,
203    ) -> Result<Value> {
204        macro_rules! deserialize_and_build_value {
205            (
206                $data_type: ident;
207                $serializer: ident;
208                $(
209                    $ty: ident, $f: ident
210                ),*
211            ) => {
212
213                match $data_type {
214                    $(
215                        ConcreteDataType::$ty(_) => {
216                            Ok(Value::from(Option::<$f>::deserialize(deserializer).context(error::DeserializeFieldSnafu)?))
217                        }
218                    )*
219                    ConcreteDataType::Binary(_) | ConcreteDataType::Json(_) | ConcreteDataType::Vector(_) => Ok(Value::from(
220                        Option::<Vec<u8>>::deserialize(deserializer)
221                            .context(error::DeserializeFieldSnafu)?
222                            .map(Bytes::from),
223                    )),
224                    ConcreteDataType::Timestamp(ty) => {
225                        let timestamp = Option::<i64>::deserialize(deserializer)
226                            .context(error::DeserializeFieldSnafu)?
227                            .map(|t|ty.create_timestamp(t));
228                        Ok(Value::from(timestamp))
229                    }
230                    ConcreteDataType::Interval(IntervalType::YearMonth(_)) => {
231                        let interval = Option::<i32>::deserialize(deserializer)
232                            .context(error::DeserializeFieldSnafu)?
233                            .map(IntervalYearMonth::from_i32);
234                        Ok(Value::from(interval))
235                    }
236                    ConcreteDataType::Interval(IntervalType::DayTime(_)) => {
237                        let interval = Option::<i64>::deserialize(deserializer)
238                            .context(error::DeserializeFieldSnafu)?
239                            .map(IntervalDayTime::from_i64);
240                        Ok(Value::from(interval))
241                    }
242                    ConcreteDataType::Interval(IntervalType::MonthDayNano(_)) => {
243                        let interval = Option::<i128>::deserialize(deserializer)
244                            .context(error::DeserializeFieldSnafu)?
245                            .map(IntervalMonthDayNano::from_i128);
246                        Ok(Value::from(interval))
247                    }
248                    ConcreteDataType::List(l) => NotSupportedFieldSnafu {
249                        data_type: ConcreteDataType::List(l.clone()),
250                    }
251                    .fail(),
252                    ConcreteDataType::Struct(f) => NotSupportedFieldSnafu {
253                        data_type: ConcreteDataType::Struct(f.clone()),
254                    }
255                    .fail(),
256                    ConcreteDataType::Dictionary(d) => NotSupportedFieldSnafu {
257                        data_type: ConcreteDataType::Dictionary(d.clone()),
258                    }
259                    .fail(),
260                    ConcreteDataType::Null(n) => NotSupportedFieldSnafu {
261                        data_type: ConcreteDataType::Null(n.clone()),
262                    }
263                    .fail(),
264                }
265            };
266        }
267        deserialize_and_build_value!(data_type; deserializer;
268            Boolean, bool,
269            Int8, i8,
270            Int16, i16,
271            Int32, i32,
272            Int64, i64,
273            UInt8, u8,
274            UInt16, u16,
275            UInt32, u32,
276            UInt64, u64,
277            Float32, f32,
278            Float64, f64,
279            String, String,
280            Date, Date,
281            Time, Time,
282            Duration, Duration,
283            Decimal128, Decimal128
284        )
285    }

287    /// Skip deserializing this field, returns the length of it.
288    pub(crate) fn skip_deserialize(
289        &self,
290        bytes: &[u8],
291        deserializer: &mut Deserializer<&[u8]>,
292    ) -> Result<usize> {
293        let pos = deserializer.position();
294        let remaining = bytes
295            .get(pos..)
296            .ok_or(memcomparable::Error::Eof)
297            .context(error::DeserializeFieldSnafu)?;
298        let len = Self::encoded_len(self.encode_data_type(), remaining)?;
299        if len > 1 && self.encode_data_type().is_boolean() && remaining[1] > 1 {
300            return Err(memcomparable::Error::InvalidBoolEncoding(remaining[1]))
301                .context(error::DeserializeFieldSnafu);
302        }
303        deserializer.advance(len);
304        Ok(len)
305    }

307    /// Checks field boundaries before any unchecked reads in memcomparable.
308    fn encoded_len(data_type: &ConcreteDataType, bytes: &[u8]) -> Result<usize> {
309        let marker = bytes
310            .first()
311            .copied()
312            .ok_or(memcomparable::Error::Eof)
313            .context(error::DeserializeFieldSnafu)?;
314        match marker {
315            0 => return Ok(1),
316            1 => {}
317            value => {
318                return Err(memcomparable::Error::InvalidTagEncoding(value as usize))
319                    .context(error::DeserializeFieldSnafu);
320            }
321        }
322        let to_skip = match data_type {
323            ConcreteDataType::Boolean(_) => 2,
324            ConcreteDataType::Int8(_) | ConcreteDataType::UInt8(_) => 2,
325            ConcreteDataType::Int16(_) | ConcreteDataType::UInt16(_) => 3,
326            ConcreteDataType::Int32(_) | ConcreteDataType::UInt32(_) => 5,
327            ConcreteDataType::Int64(_) | ConcreteDataType::UInt64(_) => 9,
328            ConcreteDataType::Float32(_) => 5,
329            ConcreteDataType::Float64(_) => 9,
330            ConcreteDataType::Binary(_)
331            | ConcreteDataType::Json(_)
332            | ConcreteDataType::Vector(_) => {
333                // Binary is encoded as a sequence of bytes, not chunked strings.
334                return encoded_binary_len(bytes).context(error::DeserializeFieldSnafu);
335            }
336            ConcreteDataType::String(_) => {
337                return encoded_string_len(bytes).context(error::DeserializeFieldSnafu);
338            }
339            ConcreteDataType::Date(_) => 5,
340            ConcreteDataType::Timestamp(_) => 9, // We treat timestamp as Option<i64>
341            ConcreteDataType::Time(_) => 10,     // i64 and 1 byte time unit
342            ConcreteDataType::Duration(_) => 10,
343            ConcreteDataType::Interval(IntervalType::YearMonth(_)) => 5,
344            ConcreteDataType::Interval(IntervalType::DayTime(_)) => 9,
345            ConcreteDataType::Interval(IntervalType::MonthDayNano(_)) => 17,
346            ConcreteDataType::Decimal128(_) => 19,
347            ConcreteDataType::Null(_)
348            | ConcreteDataType::List(_)
349            | ConcreteDataType::Struct(_)
350            | ConcreteDataType::Dictionary(_) => {
351                return NotSupportedFieldSnafu {
352                    data_type: data_type.clone(),
353                }
354                .fail();
355            }
356        };
357        if bytes.len() < to_skip {
358            return Err(memcomparable::Error::Eof).context(error::DeserializeFieldSnafu);
359        }
360        Ok(to_skip)
361    }

363    /// Decodes primitive fields straight into a reference value for column
364    /// builders, avoiding owned Value construction and conversion. The caller has
365    /// already checked the encoded boundary.
366    fn deserialize_primitive_ref<B: Buf>(
367        data_type: &ConcreteDataType,
368        deserializer: &mut Deserializer<B>,
369    ) -> Option<Result<ValueRef<'static>>> {
370        macro_rules! decode_primitive {
371            ($($variant:ident, $native:ty),* $(,)?) => {
372                match data_type {
373                    $(ConcreteDataType::$variant(_) => Some(
374                        Option::<$native>::deserialize(deserializer)
375                            .map(ValueRef::from)
376                            .context(error::DeserializeFieldSnafu)
377                    ),)*
378                    _ => None,
379                }
380            };
381        }
382        decode_primitive!(
383            Boolean, bool, Int8, i8, Int16, i16, Int32, i32, Int64, i64, UInt8, u8, UInt16, u16,
384            UInt32, u32, UInt64, u64, Float32, f32, Float64, f64,
385        )
386    }
387}

389/// The Option marker has already been checked by SortField.
390fn encoded_binary_len(bytes: &[u8]) -> memcomparable::Result<usize> {
391    let mut current = 1;
392    loop {
393        match bytes.get(current).copied() {
394            Some(0) => return Ok(current + 1),
395            Some(1) if bytes.get(current + 1).is_some() => current += 2,
396            Some(1) | None => return Err(memcomparable::Error::Eof),
397            Some(marker) => return Err(memcomparable::Error::InvalidSeqEncoding(marker)),
398        }
399    }
400}

418impl DensePrimaryKeyCodec {
419    pub fn new(metadata: &RegionMetadata) -> Self {
420        let ordered_primary_key_columns = metadata
421            .primary_key_columns()
422            .map(|c| {
423                (
424                    c.column_id,
425                    SortField::new(c.column_schema.data_type.clone()),

427            })
428            .collect::<Vec<_>>();

430        Self::with_fields(ordered_primary_key_columns)
431    }

436        }

439    fn encode_dense<'a, I>(&self, row: I, buffer: &mut Vec<u8>) -> Result<()>
440    where
441        I: Iterator<Item = ValueRef<'a>>,
442    {
443        let mut serializer = Serializer::new(buffer);
444        for (idx, value) in row.enumerate() {
445            self.field_at(idx).serialize(&mut serializer, &value)?;
446        }
447        Ok(())
448    }

450    /// Decode primary key values from bytes.
451    pub fn decode_dense(&self, bytes: &[u8]) -> Result<Vec<(ColumnId, Value)>> {
452        let mut deserializer = Deserializer::new(bytes);
453        let mut values = Vec::with_capacity(self.ordered_primary_key_columns.len());
454        for (column_id, field) in self.ordered_primary_key_columns.iter() {
455            let value = field.deserialize(&mut deserializer)?;
456            values.push((*column_id, value));
457        }
458        Ok(values)
459    }

461    /// Counts complete fields in a key encoded with a prefix of this codec's schema.
462    ///
463    /// Dense keys contain neither column ids nor types: callers must ensure that
464    /// existing fields have the same order and types. Only EOF between fields is
465    /// accepted; truncated fields and bytes beyond the schema return an error.
466    /// Field boundaries are checked in all builds; full value validation is only
467    /// enabled in debug builds and this crate's unit tests.
468    pub fn decode_prefix_len(&self, bytes: &[u8]) -> Result<usize> {
469        let mut deserializer = Deserializer::new(bytes);
470        for (index, (_, field)) in self.ordered_primary_key_columns.iter().enumerate() {
471            if !deserializer.has_remaining() {
472                return Ok(index);
473            }
474            let start = deserializer.position();
475            // Preserve the prefix error contract used by primary-key range mapping.
476            let len = SortField::encoded_len(field.encode_data_type(), &bytes[start..]).map_err(
477                |source| match source {
478                    error::Error::DeserializeField { .. } => error::InvalidDensePrimaryKeySnafu {
479                        reason: "truncated field or invalid encoding",
480                    }
481                    .build(),
482                    source => source,
483                },
484            )?;
485            // Production range mapping only needs boundaries, not allocated field values.
486            #[cfg(any(debug_assertions, test))]
487            field.deserialize(&mut Deserializer::new(&bytes[start..start + len]))?;
488            deserializer.advance(len);
489        }
490        snafu::ensure!(
491            !deserializer.has_remaining(),
492            error::InvalidDensePrimaryKeySnafu {
493                reason: "key contains bytes beyond the primary key schema",
494            }
495        );
496        Ok(self.num_fields())
497    }

499    /// Decode primary key values from bytes without column id.
500    pub fn decode_dense_without_column_id(&self, bytes: &[u8]) -> Result<Vec<Value>> {
501        let mut deserializer = Deserializer::new(bytes);
502        let mut values = Vec::with_capacity(self.ordered_primary_key_columns.len());
503        for (_, field) in self.ordered_primary_key_columns.iter() {
504            let value = field.deserialize(&mut deserializer)?;
505            values.push(value);
506        }
507        Ok(values)
508    }

510    /// Iterates over all source PK fields in schema order, checking each field's
511    /// encoded boundary before deserializing it. Stops after the first error.
512    ///
513    /// Unlike positional access, this requires no offsets or value cache. Callers
514    /// can consume each value immediately instead of retaining the entire key.
515    pub fn decode_dense_iter<'a>(
516        &'a self,
517        bytes: &'a [u8],
518    ) -> impl Iterator<Item = Result<(ColumnId, Value)>> + 'a {
519        self.ordered_primary_key_columns.iter().scan(
520            Some(Deserializer::new(bytes)),
521            move |state, (id, field)| {
522                let deserializer = state.as_mut()?;
523                let remaining = &bytes[deserializer.position()..];
524                let decoded = SortField::encoded_len(field.encode_data_type(), remaining)
525                    .and_then(|_| field.deserialize(deserializer).map(|value| (*id, value)));
526                if decoded.is_err() {
527                    *state = None;
528                }
529                Some(decoded)
530            },
531        )
532    }

539    /// Decodes all fields in source order into a consumer, checking boundaries.
540    /// Strings borrow the reusable buffer for the duration of each callback, so
541    /// materializing them into column builders needs no per-value owned string.
542    pub fn decode_dense_with(
543        &self,
544        bytes: &[u8],
545        value_buf: &mut Vec<u8>,
546        mut consume: impl FnMut(usize, ValueRef<'_>),
547    ) -> Result<()> {
548        let mut deserializer = Deserializer::new(bytes);
549        for (pos, (_, field)) in self.ordered_primary_key_columns.iter().enumerate() {
550            let data_type = field.encode_data_type();
551            let remaining = &bytes[deserializer.position()..];
552            SortField::encoded_len(data_type, remaining)?;
553            if data_type.is_string() && remaining[0] != 0 {
554                deserializer.advance(1);
555                deserializer
556                    .read_bytes_into(value_buf)
557                    .context(error::DeserializeFieldSnafu)?;
558                let value = std::str::from_utf8(value_buf).map_err(|err| {
559                    error::InvalidDensePrimaryKeySnafu {
560                        reason: format!("string is not valid UTF-8: {err}"),
561                    }
562                    .build()
563                })?;
564                consume(pos, ValueRef::String(value));
565            } else if let Some(value) =
566                SortField::deserialize_primitive_ref(data_type, &mut deserializer)
567            {
568                consume(pos, value?);
569            } else {
570                let value = field.deserialize(&mut deserializer)?;
571                consume(pos, value.as_value_ref());
572            }
573        }
574        Ok(())
575    }

585    /// Advances `deserializer` to the start of value at `pos`.
586    ///
587    /// Returns the offset of the value at `pos` in `bytes`.
588    fn advance_to_value_at(
589        &self,
590        bytes: &[u8],
591        pos: usize,
592        offsets_buf: &mut Vec<usize>,
593        deserializer: &mut Deserializer<&[u8]>,
594    ) -> Result<usize> {
595        ensure!(
596            pos < self.num_fields(),
597            error::InvalidDensePrimaryKeySnafu {
598                reason: format!(
599                    "field position {pos} exceeds field count {}",
600                    self.num_fields()
601                ),
602            }
603        );
604        if offsets_buf.is_empty() {
605            // The schema bounds this cache. Reserve once even when the first
606            // requested field is near the end of a wide key.
607            offsets_buf.reserve_exact(self.num_fields());
608            offsets_buf.push(0);
609        }
610        // Start at the requested field if cached, otherwise resume discovery
611        // from the furthest known boundary.
612        let value_start = pos.min(offsets_buf.len() - 1);
613        let mut offset = offsets_buf[value_start];
614        ensure!(
615            offset <= bytes.len(),
616            error::InvalidDensePrimaryKeySnafu {
617                reason: "cached offset exceeds key length",
618            }
619        );
620        deserializer.advance(offset);
621        for i in value_start..pos {
622            offset += self.field_at(i).skip_deserialize(bytes, deserializer)?;
623            offsets_buf.push(offset);
624        }
625        Ok(offset)
626    }

628    /// Decode value at `pos` in `bytes`.
629    ///
630    /// The i-th element in the offsets buffer is the start of field i. The buffer
631    /// must only be reused for the same encoded key and codec.
632    #[inline]
633    pub fn decode_value_at(
634        &self,
635        bytes: &[u8],
636        pos: usize,
637        offsets_buf: &mut Vec<usize>,
638    ) -> Result<Value> {
639        let encoded = self.encoded_value_at(bytes, pos, offsets_buf)?;
640        self.field_at(pos)
641            .deserialize(&mut Deserializer::new(encoded))
642    }
643
644    /// Returns the encoded bytes at `pos` in `bytes`.
645    ///
646    /// The i-th element in the offsets buffer is the start of field i. The buffer
647    /// must only be reused for the same encoded key and codec. Validates the
648    /// framing up to this field, without validating or decoding later fields.
649    #[inline]
650    pub fn encoded_value_at<'a>(
651        &self,
652        bytes: &'a [u8],
653        pos: usize,
654        offsets_buf: &mut Vec<usize>,
655    ) -> Result<&'a [u8]> {
656        let mut deserializer = Deserializer::new(bytes);
657        let offset = self.advance_to_value_at(bytes, pos, offsets_buf, &mut deserializer)?;
658
659        let len = self
660            .field_at(pos)
661            .skip_deserialize(bytes, &mut deserializer)?;
662        // We already found the next field's start while validating this one.
663        // Reuse it instead of scanning this field again on the next lookup.
664        if offsets_buf.len() == pos + 1 && pos + 1 < self.num_fields() {
665            offsets_buf.push(offset + len);
666        }
667        Ok(&bytes[offset..offset + len])
668    }
669
670    pub fn estimated_size(&self) -> usize {
671        self.ordered_primary_key_columns
672            .iter()
673            .map(|(_, f)| f.estimated_size())
674            .sum()
675    }
676
677    pub fn num_fields(&self) -> usize {
678        self.ordered_primary_key_columns.len()
679    }
680}

682impl PrimaryKeyCodec for DensePrimaryKeyCodec {
683    fn encode_key_value(&self, key_value: &KeyValue, buffer: &mut Vec<u8>) -> Result<()> {
684        self.encode_dense(key_value.primary_keys(), buffer)
685    }
686
687    fn encode_values(&self, values: &[(ColumnId, Value)], buffer: &mut Vec<u8>) -> Result<()> {
688        self.encode_dense(values.iter().map(|(_, v)| v.as_value_ref()), buffer)
689    }
690
691    fn encode_value_refs(
692        &self,
693        values: &[(ColumnId, ValueRef)],
694        buffer: &mut Vec<u8>,
695    ) -> Result<()> {
696        let iter = values.iter().map(|(_, v)| v.clone());
697        self.encode_dense(iter, buffer)
698    }
699
700    fn estimated_size(&self) -> Option<usize> {
701        Some(self.estimated_size())
702    }
703
704    fn num_fields(&self) -> Option<usize> {
705        Some(self.num_fields())
706    }
707
708    fn encoding(&self) -> PrimaryKeyEncoding {
709        PrimaryKeyEncoding::Dense
710    }
711
712    fn as_dense(&self) -> Option<&DensePrimaryKeyCodec> {
713        Some(self)
714    }
715
716    fn primary_key_filter(
717        &self,
718        metadata: &RegionMetadataRef,
719        filters: Arc<Vec<SimpleFilterEvaluator>>,
720    ) -> Box<dyn PrimaryKeyFilter> {
721        Box::new(DensePrimaryKeyFilter::new(
722            metadata.clone(),
723            filters,
724            self.clone(),
725        ))
726    }
727
728    fn decode(&self, bytes: &[u8]) -> Result<CompositeValues> {
729        Ok(CompositeValues::Dense(self.decode_dense(bytes)?))
730    }

747    fn check_encode_and_decode(data_types: &[ConcreteDataType], row: Vec<Value>) {
748        let encoder = DensePrimaryKeyCodec::with_fields(
749            data_types
750                .iter()
751                .map(|t| (0, SortField::new(t.clone())))
752                .collect::<Vec<_>>(),
753        );
754
755        let value_ref = row.iter().map(|v| v.as_value_ref()).collect::<Vec<_>>();
756
757        let result = encoder.encode(value_ref.iter().cloned()).unwrap();
758        let boundaries: Vec<_> = (0..=row.len())
759            .map(|count| {
760                encoder
761                    .encode(value_ref[..count].iter().cloned())
762                    .unwrap()
763                    .len()
764            })
765            .collect();
766        for end in 0..=result.len() {
767            match boundaries.iter().position(|boundary| *boundary == end) {
768                Some(count) => {
769                    assert_eq!(count, encoder.decode_prefix_len(&result[..end]).unwrap())
770                }
771                None => assert!(
772                    encoder.decode_prefix_len(&result[..end]).is_err(),
773                    "truncated {data_types:?} at {end}"
774                ),
775            }
776        }
777        let decoded = encoder.decode(&result).unwrap().into_dense();
778        assert_eq!(decoded, row);
779        let mut value_buf = Vec::new();
780        let mut borrowed = Vec::new();
781        encoder
782            .decode_dense_with(&result, &mut value_buf, |_, value| {
783                borrowed.push(Value::from(value))
784            })
785            .unwrap();
786        assert_eq!(borrowed, row);
787        let sequential = encoder
788            .decode_dense_iter(&result)
789            .collect::<Result<Vec<_>>>()
790            .unwrap();
791        assert_eq!(
792            sequential
793                .iter()
794                .map(|(_, value)| value)
795                .collect::<Vec<_>>(),
796            row.iter().collect::<Vec<_>>()
797        );
798        for end in 0..result.len() {
799            assert!(
800                encoder
801                    .decode_dense_with(&result[..end], &mut value_buf, |_, _| {})
802                    .is_err()
803            );
804            assert!(
805                encoder
806                    .decode_dense_iter(&result[..end])
807                    .collect::<Result<Vec<_>>>()
808                    .is_err(),
809                "truncated at {end}"
810            );
811        }
812        // Every supported type must have the same boundaries in positional and
813        // full decoding. Exercise every truncation both at the requested field
814        // and while skipping it to reach a later field.
815        let mut offsets = Vec::new();
816        let mut end = 0;
817        for (pos, value) in row.iter().enumerate() {
818            let encoded = encoder
819                .encoded_value_at(&result, pos, &mut offsets)
820                .unwrap();
821            assert_eq!(
822                encoder
823                    .field_at(pos)
824                    .deserialize(&mut Deserializer::new(encoded))
825                    .unwrap(),
826                *value
827            );
828            let start = end;
829            end += encoded.len();
830            for len in start..end {
831                assert!(
832                    encoder
833                        .encoded_value_at(&result[..len], pos, &mut Vec::new())
834                        .is_err()
835                );
836                assert!(
837                    encoder
838                        .decode_value_at(&result[..len], pos, &mut Vec::new())
839                        .is_err()
840                );
841                if pos + 1 < row.len() {
842                    assert!(
843                        encoder
844                            .encoded_value_at(&result[..len], pos + 1, &mut Vec::new())
845                            .is_err()
846                    );
847                }
848            }
849        }
850        assert_eq!(end, result.len());
851        let mut decoded = Vec::new();
852        let mut offsets = Vec::new();
853        // Iter two times to test offsets buffer.
854        for _ in 0..2 {
855            decoded.clear();
856            for i in 0..data_types.len() {
857                let value = encoder.decode_value_at(&result, i, &mut offsets).unwrap();
858                decoded.push(value);
859            }
860            assert_eq!(data_types.len(), offsets.len(), "offsets: {offsets:?}");
861            assert_eq!(decoded, row);
862        }
863    }
