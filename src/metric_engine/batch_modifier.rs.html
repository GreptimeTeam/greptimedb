30/// Info about a tag column for TSID computation and sparse primary key encoding.
31#[allow(dead_code)]
32pub struct TagColumnInfo {
33    /// Column name (used for label-name hash).
34    pub name: String,
35    /// Column index in the RecordBatch.
36    pub index: usize,
37    /// Column ID in the physical region.
38    pub column_id: ColumnId,
39}
40
41/// Computes the TSID for each row in a [RecordBatch].
42///
43/// The TSID is a stable hash of the set of labels (tags) present in each row.
44/// It accounts for both the names and values of all non-null tag columns.
45///
46/// # Logic
47/// - If a row has no nulls across all `sorted_tag_columns`, it uses a precomputed hash of all label names.
48/// - If a row has nulls, it dynamically computes a hash of the names of labels that are present (non-null).
49/// - In both cases, it then hashes the values of those present labels in the order specified by `sorted_tag_columns`.
50pub fn compute_tsid_array(
51    batch: &RecordBatch,
52    sorted_tag_columns: &[TagColumnInfo],
53    tag_arrays: &[&ArrayRef],
54) -> UInt64Array {
55    let num_rows = batch.num_rows();
56
57    let label_name_hash = {
58        let mut hasher = FxHasher::default();
59        for tag_col in sorted_tag_columns {
60            hasher.write(tag_col.name.as_bytes());
61            hasher.write_u8(0xff);
62        }
63        hasher.finish()
64    };
65
66    let mut tsid_values = Vec::with_capacity(num_rows);
67    for row in 0..num_rows {
68        let has_null = tag_arrays.iter().any(|arr| is_string_null_at(arr, row));
69
70        let tsid = if !has_null {
71            let mut hasher = FxHasher::default();
72            hasher.write_u64(label_name_hash);
73            for arr in tag_arrays {
74                if let Some(value) = string_array_value_at_index(arr, row) {
75                    hasher.write(value.as_bytes());
76                    hasher.write_u8(0xff);
77                }
78            }
79            hasher.finish()
80        } else {
81            let mut name_hasher = FxHasher::default();
82            for (tc, arr) in sorted_tag_columns.iter().zip(tag_arrays.iter()) {
83                if !is_string_null_at(arr, row) {
84                    name_hasher.write(tc.name.as_bytes());
85                    name_hasher.write_u8(0xff);
86                }
87            }
88            let row_label_hash = name_hasher.finish();
89
90            let mut val_hasher = FxHasher::default();
91            val_hasher.write_u64(row_label_hash);
92            for arr in tag_arrays {
93                if let Some(value) = string_array_value_at_index(arr, row) {
94                    val_hasher.write(value.as_bytes());
95                    val_hasher.write_u8(0xff);
96                }
97            }
98            val_hasher.finish()
99        };
100
101        tsid_values.push(tsid);
102    }
103
104    UInt64Array::from(tsid_values)
105}

107fn build_tag_arrays<'a>(
108    batch: &'a RecordBatch,
109    sorted_tag_columns: &[TagColumnInfo],
110) -> Result<Vec<&'a ArrayRef>> {
111    sorted_tag_columns
112        .iter()
113        .map(|tc| {
114            let array = batch.column(tc.index);
115            match array.data_type() {
116                DataType::Utf8 | DataType::LargeUtf8 | DataType::Utf8View => Ok(array),
117                DataType::Dictionary(key_type, value_type)
118                    if key_type.is_integer() && value_type.is_string() =>
119                {
120                    Ok(array)
121                }
122                data_type => UnexpectedRequestSnafu {
123                    reason: format!(
124                        "Tag column '{}' must be a string, given: {data_type}",
125                        tc.name
126                    ),
127                }
128                .fail(),
129            }
130        })
131        .collect()
132}

134/// Modifies a [RecordBatch] to include a sparse primary key column.
135///
136/// This function transforms the input `batch` into a new `RecordBatch` where the first column
137/// is the generated primary key (named [PRIMARY_KEY_COLUMN_NAME]), followed by columns
138/// indicated by `extra_column_indices`.
139///
140/// The primary key uses a "sparse" encoding, which compactly represents the row's identity
141/// by only including non-null tag values. The encoding, handled by [SparsePrimaryKeyCodec],
142/// consists of:
143/// 1. The `table_id`.
144/// 2. A `tsid` (Time Series ID), which is a hash of the present tags.
145/// 3. The actual non-null tag values paired with their `column_id`.
146///
147/// # Parameters
148/// - `batch`: The source [RecordBatch].
149/// - `table_id`: The ID of the table.
150/// - `sorted_tag_columns`: Metadata for tag columns, used for both TSID computation and PK encoding.
151/// - `extra_column_indices`: Indices of columns from the original batch to keep in the output
152///   (typically the timestamp and value fields).
153pub fn modify_batch_sparse(
154    batch: RecordBatch,
155    table_id: u32,
156    sorted_tag_columns: &[TagColumnInfo],
157    extra_column_indices: &[usize],
158) -> Result<RecordBatch> {
159    let num_rows = batch.num_rows();
160    let codec = SparsePrimaryKeyCodec::schemaless();
161    let tag_arrays = build_tag_arrays(&batch, sorted_tag_columns)?;
162    let tsid_array = compute_tsid_array(&batch, sorted_tag_columns, &tag_arrays);
163
164    let mut pk_builder = BinaryBuilder::with_capacity(num_rows, 0);
165    let mut buffer = Vec::new();
166    for row in 0..num_rows {
167        buffer.clear();
168        codec
169            .encode_internal(table_id, tsid_array.value(row), &mut buffer)
170            .context(EncodePrimaryKeySnafu)?;
171
172        let tags = sorted_tag_columns
173            .iter()
174            .zip(tag_arrays.iter())
175            .filter_map(|(tc, arr)| {
176                string_array_value_at_index(arr, row).map(|value| (tc.column_id, value.as_bytes()))
177            });
178        codec
179            .encode_raw_tag_value(tags, &mut buffer)
180            .context(EncodePrimaryKeySnafu)?;
181
182        pk_builder.append_value(&buffer);
183    }
184
185    let pk_array = pk_builder.finish();
186
187    let mut fields = vec![Arc::new(Field::new(
188        PRIMARY_KEY_COLUMN_NAME,
189        DataType::Binary,
190        false,
191    ))];
192    let mut columns: Vec<Arc<dyn Array>> = vec![Arc::new(pk_array)];
193
194    for &idx in extra_column_indices {
195        fields.push(batch.schema().fields()[idx].clone());
196        columns.push(batch.column(idx).clone());
197    }
198
199    let new_schema = Arc::new(ArrowSchema::new(fields));
200    RecordBatch::try_new(new_schema, columns).map_err(|e| {
201        UnexpectedRequestSnafu {
202            reason: format!("Failed to build modified sparse RecordBatch: {e}"),
203        }
204        .build()
205    })
206}

473        // Tag (label) columns may arrive as any string representation (`Utf8`,

482                "greptimedb",
483            ])),
484            Arc::new(datatypes::arrow::array::DictionaryArray::<
485                datatypes::arrow::datatypes::UInt32Type,
486            >::new(
487                datatypes::arrow::array::UInt32Array::from(vec![0]),
488                Arc::new(StringArray::from(vec!["greptimedb"])),
489            )),
490        ];
491        let tag_columns = vec![TagColumnInfo {
492            name: "namespace".to_string(),
493            index: 2,
494            column_id: 2,
495        }];
496        let non_tag_indices = vec![0, 1];
497
498        let primary_keys = tag_arrays
499            .into_iter()
500            .map(|tag_array| {
501                let schema = Arc::new(ArrowSchema::new(vec![
502                    Field::new("greptime_timestamp", DataType::Int64, false),
503                    Field::new("greptime_value", DataType::Float64, true),
504                    Field::new("namespace", tag_array.data_type().clone(), true),
505                ]));
506                let batch = RecordBatch::try_new(

509                        Arc::new(Int64Array::from(vec![1000])),

528    }

530    #[test]
531    fn test_compute_tsid_utf8view_with_nulls_matches_utf8() {
532        // Regression test for Utf8View tag/label columns: the whole
533        // `build_tag_arrays` -> `compute_tsid_array` -> sparse primary key path must handle
534        // `Utf8View` columns exactly like `Utf8`, including rows with null tags, instead of
535        // panicking or producing different TSIDs.
536        let namespace_values: Vec<Option<&str>> =
537            vec![Some("ns-a"), Some("ns-b"), None, Some("ns-c")];
538        let host_values: Vec<Option<&str>> = vec![Some("host-1"), None, Some("host-2"), None];
539
540        let build_batch = |data_type: DataType| {
541            let (namespace, host): (ArrayRef, ArrayRef) = match &data_type {
542                DataType::Utf8 => (
543                    Arc::new(StringArray::from(namespace_values.clone())),
544                    Arc::new(StringArray::from(host_values.clone())),
545                ),
546                DataType::Utf8View => (
547                    Arc::new(datatypes::arrow::array::StringViewArray::from(
548                        namespace_values.clone(),
549                    )),
550                    Arc::new(datatypes::arrow::array::StringViewArray::from(
551                        host_values.clone(),
552                    )),
553                ),
554                _ => unreachable!(),
555            };
556            let schema = Arc::new(ArrowSchema::new(vec![
557                Field::new("greptime_timestamp", DataType::Int64, false),
558                Field::new("greptime_value", DataType::Float64, true),
559                Field::new("namespace", data_type.clone(), true),
560                Field::new("host", data_type, true),
561            ]));
562            RecordBatch::try_new(
563                schema,
564                vec![
565                    Arc::new(Int64Array::from(vec![1000, 1001, 1002, 1003])),
566                    Arc::new(datatypes::arrow::array::Float64Array::from(vec![
567                        1.0, 2.0, 3.0, 4.0,
568                    ])),
569                    namespace,
570                    host,
571                ],
572            )
573            .unwrap()
574        };
575
576        let tag_columns = vec![
577            TagColumnInfo {
578                name: "host".to_string(),
579                index: 3,
580                column_id: 3,
581            },
582            TagColumnInfo {
583                name: "namespace".to_string(),
584                index: 2,
585                column_id: 2,
586            },
587        ];
588
589        let utf8_batch = build_batch(DataType::Utf8);
590        let utf8view_batch = build_batch(DataType::Utf8View);
591
592        // Both representations must be accepted by `build_tag_arrays`.
593        let utf8_tag_arrays = build_tag_arrays(&utf8_batch, &tag_columns).unwrap();
594        let utf8view_tag_arrays = build_tag_arrays(&utf8view_batch, &tag_columns).unwrap();
595
596        // TSIDs must be identical across representations, including rows with null tags.
597        let utf8_tsids = compute_tsid_array(&utf8_batch, &tag_columns, &utf8_tag_arrays);
598        let utf8view_tsids =
599            compute_tsid_array(&utf8view_batch, &tag_columns, &utf8view_tag_arrays);
600        assert_eq!(utf8_tsids, utf8view_tsids);
601
602        // The full sparse primary key (TSID + tag values) must also be identical.
603        let non_tag_indices = vec![0, 1];
604        let modified_utf8 =
605            modify_batch_sparse(utf8_batch, 1025, &tag_columns, &non_tag_indices).unwrap();
606        let modified_utf8view =
607            modify_batch_sparse(utf8view_batch, 1025, &tag_columns, &non_tag_indices).unwrap();
608        let utf8_pks = modified_utf8
609            .column(0)
610            .as_any()
611            .downcast_ref::<BinaryArray>()
612            .unwrap();
613        let utf8view_pks = modified_utf8view
614            .column(0)
615            .as_any()
616            .downcast_ref::<BinaryArray>()
617            .unwrap();
618        for row in 0..utf8_pks.len() {
619            assert_eq!(utf8_pks.value(row), utf8view_pks.value(row));
620        }
621    }
