39/// This is a generic handler like [MetricEngine](crate::engine::MetricEngine). It
40/// will handle all the data related operations across physical tables. Thus
41/// every operation should be associated to a [RegionId], which is the physical
42/// table id + region sequence. This handler will transform the region group by
43/// itself.
44pub struct DataRegion {
45    mito: MitoEngine,
46}
47
48impl DataRegion {
49    pub(crate) fn new(mito: MitoEngine) -> Self {
50        Self { mito }
51    }
52
53    /// Submit an alter request to underlying physical region.
54    ///
55    /// This method will change the nullability of those given columns.
56    /// [SemanticType::Tag] will become nullable column as it's shared between
57    /// logical regions.
58    ///
59    /// Invoker don't need to set up or verify the column id. This method will adjust
60    /// it using underlying schema, and reuse the persisted id of a column that the
61    /// physical region already has.
62    ///
63    /// This method will also set the nullable marker to true. All of those change are applies
64    /// to `columns` in-place.
65    ///
66    /// Callers must hold the column-add lock of the physical region, otherwise two
67    /// concurrent requests could read the same metadata and assign one column id to
68    /// different columns. Use
69    /// [MetricEngineInner::add_physical_columns](crate::engine::MetricEngineInner::add_physical_columns).
70    pub async fn add_columns(
71        &self,
72        region_id: RegionId,
73        columns: Vec<ColumnMetadata>,
74        index_options: IndexOptions,
75    ) -> Result<()> {
76        // Return early if no new columns are added.
77        if columns.is_empty() {
78            return Ok(());
79        }
80
81        let region_id = utils::to_data_region_id(region_id);
82
83        let Some((new_column_count, request)) = self
84            .assemble_alter_request(region_id, columns, index_options)
85            .await?
86        else {
87            // Every requested column already exists.
88            return Ok(());
89        };
90
91        let _timer = MITO_DDL_DURATION.start_timer();
92
93        let _ = self
94            .mito
95            .handle_request(region_id, request)
96            .await
97            .context(MitoWriteOperationSnafu)?;
98
99        PHYSICAL_COLUMN_COUNT.add(new_column_count as _);
100
101        Ok(())
102    }
103
104    /// Generate wrapped [RegionAlterRequest] with given [ColumnMetadata].
105    ///
106    /// Columns that the physical region already has are not added again: their
107    /// definitions are validated and their persisted column ids are reused.
108    /// Returns `None` if every given column already exists.
109    async fn assemble_alter_request(
110        &self,
111        region_id: RegionId,
112        columns: Vec<ColumnMetadata>,
113        index_options: IndexOptions,
114    ) -> Result<Option<(usize, RegionRequest)>> {
115        let region_metadata = self
116            .mito
117            .get_metadata(region_id)
118            .await
119            .context(MitoReadOperationSnafu)?;
120
121        // The caller derives the columns to add from a cached schema that a cancelled or
122        // concurrent alter may have left behind, so the names are checked against the
123        // real metadata instead of being trusted.
124        let mut seen: HashMap<String, usize> = HashMap::with_capacity(columns.len());
125        let mut columns_to_add = Vec::with_capacity(columns.len());
126        for column in columns {
127            let name = column.column_schema.name.clone();
128            if let Some(existing) = region_metadata.column_by_name(&name) {
129                validate_shared_column(region_id, existing, &column)?;
130                continue;
131            }
132            // The batch itself may repeat a column that is not installed yet: the duplicate
133            // must agree with the first occurrence, and only one of them is added.
134            if let Some(&index) = seen.get(&name) {
135                validate_shared_column(region_id, &columns_to_add[index], &column)?;
136                continue;
137            }
138            seen.insert(name, columns_to_add.len());
139            columns_to_add.push(column);
140        }
141
142        if columns_to_add.is_empty() {
143            debug!("No new columns to add to physical region {region_id:?}");
144            return Ok(None);
145        }
146        let new_column_count = columns_to_add.len();
147
148        // find the max column id
149        let new_column_id_start = 1 + region_metadata
150            .column_metadatas
151            .iter()
152            .filter_map(|c| {
153                if ReservedColumnId::is_reserved(c.column_id) {
154                    None
155                } else {
156                    Some(c.column_id)
157                }
158            })
159            .max()
160            .unwrap_or(0);
161
162        // overwrite semantic type
163        let new_columns = columns_to_add
164            .into_iter()
165            .enumerate()
166            .map(|(delta, mut c)| {
167                match c.semantic_type {
168                    SemanticType::Tag => {
169                        if !c.column_schema.data_type.is_string() {
170                            return ColumnTypeMismatchSnafu {
171                                expect: ConcreteDataType::string_datatype(),
172                                actual: c.column_schema.data_type.clone(),
173                            }
174                            .fail();
175                        }
176                    }
177                    // Field columns can only be added to the shared physical
178                    // table for native histograms; ordinary metric fields are
179                    // created with the logical table.
180                    SemanticType::Field
181                        if is_native_histogram_value_type(&c.column_schema.data_type) => {}
182                    _ => {
183                        return AddingFieldColumnSnafu {
184                            name: &c.column_schema.name,
185                        }
186                        .fail();
187                    }
188                }
189
190                c.column_id = new_column_id_start + delta as u32;
191                c.column_schema.set_nullable();
192                if c.semantic_type == SemanticType::Tag {
193                    match index_options {
194                        IndexOptions::None => {}
195                        IndexOptions::Inverted => {
196                            c.column_schema.set_inverted_index(true);
197                        }
198                        IndexOptions::Skipping {
199                            granularity,
200                            false_positive_rate,
201                        } => {
202                            c.column_schema
203                                .set_skipping_options(
204                                    &SkippingIndexOptions::new(
205                                        granularity,
206                                        false_positive_rate,
207                                        SkippingIndexType::BloomFilter,
208                                    )
209                                    .context(SetSkippingIndexOptionSnafu)?,
210                                )
211                                .context(SetSkippingIndexOptionSnafu)?;
212                        }
213                    }
214                }
215
216                Ok(AddColumn {
217                    column_metadata: c.clone(),
218                    location: None,
219                })
220            })
221            .collect::<Result<_>>()?;
222
223        debug!("Adding (Column id assigned) columns {new_columns:?} to region {region_id:?}");
224        // assemble alter request
225        let alter_request = RegionRequest::Alter(RegionAlterRequest {
226            kind: AlterKind::AddColumns {
227                columns: new_columns,
228            },
229        });
230
231        Ok(Some((new_column_count, alter_request)))
232    }

234    pub async fn write_data(
235        &self,
236        region_id: RegionId,
237        request: RegionRequest,
238    ) -> Result<AffectedRows> {
239        let region_id = utils::to_data_region_id(region_id);
240        self.mito
241            .handle_request(region_id, request)
242            .await
243            .context(MitoWriteOperationSnafu)
244            .map(|result| result.affected_rows)
245    }
246
247    pub async fn physical_columns(
248        &self,
249        physical_region_id: RegionId,
250    ) -> Result<Vec<ColumnMetadata>> {
251        let data_region_id = utils::to_data_region_id(physical_region_id);
252        let metadata = self
253            .mito
254            .get_metadata(data_region_id)
255            .await
256            .context(MitoReadOperationSnafu)?;
257        Ok(metadata.column_metadatas.clone())
258    }
259
260    pub async fn alter_region_options(
261        &self,
262        region_id: RegionId,
263        request: RegionAlterRequest,
264    ) -> Result<AffectedRows> {
265        match request.kind {
266            AlterKind::SetRegionOptions { options: _ }
267            | AlterKind::UnsetRegionOptions { keys: _ }
268            | AlterKind::SetIndexes { options: _ }
269            | AlterKind::UnsetIndexes { options: _ }
270            | AlterKind::SetJsonSettings { .. }
271            | AlterKind::SyncColumns {
272                column_metadatas: _,
273            } => {
274                let region_id = utils::to_data_region_id(region_id);
275                self.mito
276                    .handle_request(region_id, RegionRequest::Alter(request))
277                    .await
278                    .context(MitoWriteOperationSnafu)
279                    .map(|result| result.affected_rows)
280            }
281            _ => {
282                info!(
283                    "Metric region received alter request {request:?} on physical region {region_id:?}"
284                );
285                FORBIDDEN_OPERATION_COUNT.inc();
286
287                ForbiddenPhysicalAlterSnafu.fail()
288            }
289        }
290    }

302    fn tag_column(name: &str) -> ColumnMetadata {
303        ColumnMetadata {
304            column_id: 0,

310    #[tokio::test]
311    async fn test_add_columns() {
312        let env = TestEnv::new().await;
313        env.init_metric_region().await;
314
315        let current_version = env
316            .mito()
317            .get_metadata(utils::to_data_region_id(env.default_physical_region_id()))
318            .await
319            .unwrap()
320            .schema_version;
321        // TestEnv will create a logical region which changes the version to 1.
322        assert_eq!(current_version, 1);
323
324        let new_columns = vec![
325            ColumnMetadata {
326                column_id: 0,
327                semantic_type: SemanticType::Tag,
328                column_schema: ColumnSchema::new(
329                    "tag2",
330                    ConcreteDataType::string_datatype(),
331                    false,
332                ),
333            },
334            ColumnMetadata {
335                column_id: 0,
336                semantic_type: SemanticType::Tag,
337                column_schema: ColumnSchema::new(
338                    "tag3",
339                    ConcreteDataType::string_datatype(),
340                    false,
341                ),
342            },
343        ];
344        env.data_region()
345            .add_columns(
346                env.default_physical_region_id(),
347                new_columns,
348                IndexOptions::Inverted,
349            )
350            .await
351            .unwrap();
352
353        let new_metadata = env
354            .mito()
355            .get_metadata(utils::to_data_region_id(env.default_physical_region_id()))
356            .await
357            .unwrap();
358        let column_names = new_metadata
359            .column_metadatas
360            .iter()
361            .map(|c| &c.column_schema.name)
362            .collect::<Vec<_>>();
363        let expected = vec![
364            greptime_timestamp(),
365            greptime_value(),
366            "__table_id",
367            "__tsid",
368            "job",
369            "tag2",
370            "tag3",
371        ];
372        assert_eq!(column_names, expected);
373    }

431    // A retry passes the column it added before, because the caller derives the columns to
432    // add from a cache that a cancelled request may not have refreshed. The column keeps the
433    // id it was installed with.
434    #[tokio::test]
435    async fn test_add_columns_reuses_the_existing_column_id() {
436        let env = TestEnv::new().await;
437        env.init_metric_region().await;
438        let physical_region_id = env.default_physical_region_id();
439        let data_region_id = utils::to_data_region_id(physical_region_id);
440
441        env.data_region()
442            .add_columns(
443                physical_region_id,
444                vec![tag_column("tag2")],
445                IndexOptions::Inverted,
446            )
447            .await
448            .unwrap();
449        let tag2_id = env
450            .mito()
451            .get_metadata(data_region_id)
452            .await
453            .unwrap()
454            .column_by_name("tag2")
455            .unwrap()
456            .column_id;
457
458        env.data_region()
459            .add_columns(
460                physical_region_id,
461                vec![tag_column("tag2"), tag_column("tag3")],
462                IndexOptions::Inverted,
463            )
464            .await
465            .unwrap();
466
467        let metadata = env.mito().get_metadata(data_region_id).await.unwrap();
468        assert_eq!(metadata.column_by_name("tag2").unwrap().column_id, tag2_id);
469        assert_eq!(
470            metadata.column_by_name("tag3").unwrap().column_id,
471            tag2_id + 1
472        );
473    }
