39    pub async fn alter_regions(
40        &self,
41        mut requests: Vec<(RegionId, RegionAlterRequest)>,
42        extension_return_value: &mut HashMap<String, Vec<u8>>,
43    ) -> Result<AffectedRows> {
44        if requests.is_empty() {
45            return Ok(0);
46        }
47
48        let first_region_id = &requests.first().unwrap().0;
49        if self.is_physical_region(*first_region_id) {
50            ensure!(
51                requests.len() == 1,
52                UnexpectedRequestSnafu {
53                    reason: "Physical table must be altered with single request".to_string(),
54                }
55            );
56            let (region_id, request) = requests.pop().unwrap();
57            self.alter_physical_region(region_id, request).await?;
58        } else {
59            // Fast path for single logical region alter request
60            if requests.len() == 1 {
61                // Safety: requests is not empty
62                let region_id = requests.first().unwrap().0;
63                let physical_region_id = self
64                    .state
65                    .read()
66                    .unwrap()
67                    .get_physical_region_id(region_id)
68                    .with_context(|| LogicalRegionNotFoundSnafu { region_id })?;
69                let mut manifest_infos = Vec::with_capacity(1);
70                self.alter_logical_regions(physical_region_id, requests, extension_return_value)
71                    .await?;
72                append_manifest_info(&self.mito, physical_region_id, &mut manifest_infos);
73                encode_manifest_info_to_extensions(&manifest_infos, extension_return_value)?;
74            } else {
75                let grouped_requests =
76                    self.group_logical_region_requests_by_physical_region_id(requests)?;
77                let mut manifest_infos = Vec::with_capacity(grouped_requests.len());
78                for (physical_region_id, requests) in grouped_requests {
79                    self.alter_logical_regions(
80                        physical_region_id,
81                        requests,
82                        extension_return_value,
83                    )
84                    .await?;
85                    append_manifest_info(&self.mito, physical_region_id, &mut manifest_infos);
86                }
87                encode_manifest_info_to_extensions(&manifest_infos, extension_return_value)?;
88            }
89        }
90        Ok(0)
91    }

93    /// Groups the alter logical region requests by physical region id.
94    fn group_logical_region_requests_by_physical_region_id(
95        &self,
96        requests: Vec<(RegionId, RegionAlterRequest)>,
97    ) -> Result<HashMap<RegionId, Vec<(RegionId, RegionAlterRequest)>>> {
98        let mut result = HashMap::with_capacity(requests.len());
99        let state = self.state.read().unwrap();
100
101        for (region_id, request) in requests {
102            let physical_region_id = state
103                .get_physical_region_id(region_id)
104                .with_context(|| LogicalRegionNotFoundSnafu { region_id })?;
105            result
106                .entry(physical_region_id)
107                .or_insert_with(Vec::new)
108                .push((region_id, request));
109        }
110
111        Ok(result)
112    }

114    /// Alter multiple logical regions on the same physical region.
115    pub async fn alter_logical_regions(
116        &self,
117        physical_region_id: RegionId,
118        requests: Vec<(RegionId, RegionAlterRequest)>,
119        extension_return_value: &mut HashMap<String, Vec<u8>>,
120    ) -> Result<AffectedRows> {
121        // Checks all alter requests are add columns.
122        validate_alter_region_requests(&requests)?;
123        self.validate_logical_field_alters(physical_region_id, &requests)
124            .await?;
125
126        // Finds new columns to add
127        let mut new_column_indices = HashMap::new();
128        let mut new_columns_to_add = vec![];
129
130        let index_options = {
131            let state = &self.state.read().unwrap();
132            let region_state = state
133                .physical_region_states()
134                .get(&physical_region_id)
135                .with_context(|| PhysicalRegionNotFoundSnafu {
136                    region_id: physical_region_id,
137                })?;
138            let physical_columns = region_state.physical_columns();
139
140            extract_new_columns(
141                &requests,
142                physical_columns,
143                &mut new_column_indices,
144                &mut new_columns_to_add,
145            )?;
146
147            region_state.options().index
148        };
149
150        // Acquire logical region locks in a deterministic order to avoid deadlocks when multiple
151        // alter operations target overlapping regions concurrently.
152        let region_ids = requests
153            .iter()
154            .map(|(region_id, _)| *region_id)
155            .collect::<BTreeSet<_>>();
156
157        let mut write_guards = Vec::with_capacity(region_ids.len());
158        for region_id in region_ids {
159            write_guards.push(
160                self.metadata_region
161                    .write_lock_logical_region(region_id)
162                    .await?,
163            );
164        }
165
166        self.add_physical_columns(physical_region_id, new_columns_to_add, index_options)
167            .await?;
168
169        let physical_metadata = self
170            .data_region
171            .physical_metadata(physical_region_id)
172            .await?;
173
174        // The physical region now has every requested column, either because this call
175        // added it or because it was already there with the same definition.
176        let mut logical_region_columns = Vec::with_capacity(requests.len());
177        for (region_id, request) in &requests {
178            let AlterKind::AddColumns { columns } = &request.kind else {
179                unreachable!()
180            };
181            let mut column_metadata = HashMap::with_capacity(columns.len());
182            for col in columns {
183                let column_name = col.column_metadata.column_schema.name.as_str();
184                let metadata = region_metadata::physical_column(&physical_metadata, column_name)?;
185                column_metadata.insert(column_name, metadata);
186            }
187            logical_region_columns.push((*region_id, column_metadata));
188        }
189
190        self.cache_new_physical_columns(&physical_metadata, new_column_indices.keys().copied())?;
191
192        // Writes logical regions metadata to metadata region
193        let result = self
194            .metadata_region
195            .add_logical_regions(
196                physical_region_id,
197                false,
198                logical_region_columns.into_iter(),
199            )
200            .await;
201        // A failed write may still be applied, e.g. when the region lease expires after the WAL
202        // append, so the cached logical columns are dropped either way.
203        self.state
204            .write()
205            .unwrap()
206            .invalid_logical_regions_cache(requests.iter().map(|(region_id, _)| *region_id));
207        result?;
208
209        extension_return_value.insert(
210            ALTER_PHYSICAL_EXTENSION_KEY.to_string(),
211            ColumnMetadata::encode_list(&physical_metadata.column_metadatas)
212                .context(SerializeColumnMetadataSnafu)?,
213        );
214
215        Ok(0)
216    }

218    async fn validate_logical_field_alters(
219        &self,
220        physical_region_id: RegionId,
221        requests: &[(RegionId, RegionAlterRequest)],
222    ) -> Result<()> {
223        // Logical metric tables have one field column. Native histograms are a
224        // special struct field, so field alters must leave exactly that field.
225        for (region_id, request) in requests {
226            let AlterKind::AddColumns { columns } = &request.kind else {
227                unreachable!()
228            };
229            let added_fields = columns
230                .iter()
231                .filter(|col| col.column_metadata.semantic_type == SemanticType::Field)
232                .collect::<Vec<_>>();
233            let Some(&first_added_field) = added_fields.first() else {
234                continue;
235            };
236
237            let existing_columns = self
238                .load_logical_columns(physical_region_id, *region_id)
239                .await?;
240            ensure!(
241                added_fields.len() == 1
242                    && is_native_histogram_value_type(
243                        &first_added_field.column_metadata.column_schema.data_type
244                    )
245                    && existing_columns
246                        .iter()
247                        .all(|col| col.semantic_type != SemanticType::Field),
248                AddingFieldColumnSnafu {
249                    name: first_added_field.column_metadata.column_schema.name.clone(),
250                }
251            );
252        }
253
254        Ok(())
255    }

488    /// Holds the result of an armed physical alter in Mito, after its manifest is
489    /// updated and before the new metadata becomes visible, and reports the requests
490    /// the region worker receives.
491    struct HoldAlterResultListener {
492        armed: AtomicBool,
493        altering: Notify,
494        release: Notify,
495        received: mpsc::UnboundedSender<usize>,
496    }
497
498    #[async_trait::async_trait]
499    impl EventListener for HoldAlterResultListener {
500        fn on_recv_requests(&self, request_num: usize) {
501            let _ = self.received.send(request_num);
502        }
503
504        async fn on_notify_region_change_result_begin(&self, _region_id: RegionId) {
505            if self.armed.swap(false, Ordering::SeqCst) {
506                self.altering.notify_one();
507                self.release.notified().await;
508            }
509        }
510    }

512    // Pins the residual race documented by #10372: the column-add lock is released
513    // with a dropped caller while Mito still applies its alter, so a request for
514    // another column reads the same metadata, allocates the same column id, and is
515    // rejected once the first alter lands. The rejection is not retryable, while a
516    // manual retry succeeds with the next id. Tracked by #10377.
517    #[tokio::test]
518    async fn test_add_physical_columns_cancelled_while_alter_pending() {
519        common_telemetry::init_default_ut_logging();
520        let mut mito_env = MitoTestEnv::with_prefix("cancel-pending-add-column").await;
521        let (received_tx, mut received_rx) = mpsc::unbounded_channel();
522        let listener = Arc::new(HoldAlterResultListener {
523            armed: AtomicBool::new(false),
524            altering: Notify::new(),
525            release: Notify::new(),
526            received: received_tx,
527        });
528        let mito = mito_env
529            .create_engine_with(MitoConfig::default(), None, Some(listener.clone()), None)
530            .await;
531        let env = TestEnv::with_mito_engine(mito_env, mito);
532        env.init_metric_region().await;
533        let physical_region_id = env.default_physical_region_id();
534        let data_region_id = to_data_region_id(physical_region_id);
535        let tag = |name: &str| ColumnMetadata {
536            column_id: 0,
537            semantic_type: SemanticType::Tag,
538            column_schema: ColumnSchema::new(name, ConcreteDataType::string_datatype(), false),
539        };
540        let inner = env.metric().inner.clone();
541
542        // Request A allocates an id for `x` and its alter is held in Mito.
543        listener.armed.store(true, Ordering::SeqCst);
544        let altering = listener.altering.notified();
545        let request_a = {
546            let inner = inner.clone();
547            let column = tag("x");
548            tokio::spawn(async move {
549                inner
550                    .add_physical_columns(physical_region_id, vec![column], IndexOptions::Inverted)
551                    .await
552            })
553        };
554        tokio::time::timeout(Duration::from_secs(10), altering)
555            .await
556            .expect("alter of request A should reach Mito");
557
558        // Drop the caller of A: its guard is released while Mito still applies A.
559        request_a.abort();
560        assert!(request_a.await.unwrap_err().is_cancelled());
561        let pending_id = {
562            let metadata = env.mito().get_metadata(data_region_id).await.unwrap();
563            assert!(metadata.column_by_name("x").is_none());
564            1 + metadata
565                .column_metadatas
566                .iter()
567                .map(|c| c.column_id)
568                .filter(|id| !ReservedColumnId::is_reserved(*id))
569                .max()
570                .unwrap()
571        };
572
573        // Request B, for another column, reads the same metadata and allocates the same
574        // id. Its alter waits in the worker behind the pending alter of A.
575        while received_rx.try_recv().is_ok() {}
576        let request_b = {
577            let inner = inner.clone();
578            let column = tag("y");
579            tokio::spawn(async move {
580                inner
581                    .add_physical_columns(physical_region_id, vec![column], IndexOptions::Inverted)
582                    .await
583            })
584        };
585        tokio::time::timeout(Duration::from_secs(10), received_rx.recv())
586            .await
587            .expect("alter of request B should reach Mito");
588        assert!(!request_b.is_finished());
589
590        // A lands first and takes the id, so Mito rejects B.
591        listener.release.notify_one();
592        let err = tokio::time::timeout(Duration::from_secs(10), request_b)
593            .await
594            .expect("request B should finish once A is applied")
595            .unwrap()
596            .unwrap_err();
597        assert_eq!(err.status_code(), StatusCode::InvalidArguments, "{err:?}");
598        assert_eq!(err.retry_hint(), RetryHint::NonRetryable, "{err:?}");
599        assert!(
600            format!("{err:?}").contains(&format!("column id {pending_id} already exists")),
601            "{err:?}"
602        );
603
604        let metadata = env.mito().get_metadata(data_region_id).await.unwrap();
605        assert_eq!(metadata.column_by_name("x").unwrap().column_id, pending_id);
606        assert!(metadata.column_by_name("y").is_none());
607
608        // A manual retry sees `y` absent and allocates the next id.
609        inner
610            .add_physical_columns(physical_region_id, vec![tag("y")], IndexOptions::Inverted)
611            .await
612            .unwrap();
613        let metadata = env.mito().get_metadata(data_region_id).await.unwrap();
614        assert_eq!(metadata.column_by_name("x").unwrap().column_id, pending_id);
615        assert_eq!(
616            metadata.column_by_name("y").unwrap().column_id,
617            pending_id + 1
618        );
619    }

622    // physical column installed while the cache still misses it. The retry must reuse the

624    #[tokio::test]
625    async fn test_alter_logical_region_after_cancelled_physical_alter() {
626        let env = TestEnv::new().await;
627        env.init_metric_region().await;
628        let physical_region_id = env.default_physical_region_id();
629        let logical_region_id = env.default_logical_region_id();
630        let data_region_id = to_data_region_id(physical_region_id);
631
632        env.data_region()
633            .add_columns(
634                physical_region_id,
635                vec![ColumnMetadata {
636                    column_id: 0,
637                    semantic_type: SemanticType::Tag,
638                    column_schema: ColumnSchema::new(
639                        "tag1",
640                        ConcreteDataType::string_datatype(),
641                        false,
642                    ),
643                }],
644                IndexOptions::Inverted,
645            )
646            .await
647            .unwrap();
648        let installed_id = env
649            .mito()
650            .get_metadata(data_region_id)
651            .await
652            .unwrap()
653            .column_by_name("tag1")
654            .unwrap()
655            .column_id;
656
657        env.metric()
658            .handle_batch_ddl_requests(BatchRegionDdlRequest::Alter(vec![(
659                logical_region_id,
660                alter_logical_region_request(&["tag1"]),
661            )]))
662            .await
663            .unwrap();
664
665        let metadata = env.mito().get_metadata(data_region_id).await.unwrap();
666        assert_eq!(
667            metadata.column_by_name("tag1").unwrap().column_id,
668            installed_id
669        );
670        assert_eq!(
671            env.metadata_region()
672                .column_semantic_type(physical_region_id, logical_region_id, "tag1")
673                .await
674                .unwrap(),
675            Some(SemanticType::Tag)
676        );
677
678        let engine = env.metric();
679        let state = engine.inner.state.read().unwrap();
680        let physical_columns = state
681            .physical_region_states()
682            .get(&data_region_id)
683            .unwrap()
684            .physical_columns();
685        assert!(physical_columns.contains_key("tag1"));
686    }
