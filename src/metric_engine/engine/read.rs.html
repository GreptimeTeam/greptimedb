37    pub async fn read_region(
38        &self,
39        region_id: RegionId,
40        request: ScanRequest,
41    ) -> Result<RegionScannerRef> {
42        let is_reading_physical_region = self.is_physical_region(region_id);
43
44        if is_reading_physical_region {
45            debug!(
46                "Metric region received read request {request:?} on physical region {region_id:?}"
47            );
48            self.read_physical_region(region_id, request).await
49        } else {
50            self.read_logical_region(region_id, request).await
51        }
52    }

54    /// Proxy the read request to underlying physical region (mito engine).
55    async fn read_physical_region(
56        &self,
57        region_id: RegionId,
58        request: ScanRequest,
59    ) -> Result<RegionScannerRef> {
60        let _timer = MITO_OPERATION_ELAPSED
61            .with_label_values(&["read_physical"])
62            .start_timer();
63
64        // PromQL scans physical regions with a table-ID filter to access `__tsid`.
65        let logical_region = request
66            .filters
67            .iter()
68            .flat_map(logical_expr::utils::split_conjunction)
69            .any(|expr| {
70                let Expr::BinaryExpr(expr) = expr else {
71                    return false;
72                };
73                expr.op == Operator::Eq
74                    && matches!(
75                        (expr.left.as_ref(), expr.right.as_ref()),
76                        (Expr::Column(column), Expr::Literal(ScalarValue::UInt32(Some(_)), _))
77                            | (Expr::Literal(ScalarValue::UInt32(Some(_)), _), Expr::Column(column))
78                            if column.name == DATA_SCHEMA_TABLE_ID_COLUMN_NAME
79                    )
80            });
81        let mut scanner = self
82            .mito
83            .handle_query(region_id, request)
84            .await
85            .context(MitoReadOperationSnafu)?;
86        scanner.set_logical_region(logical_region);
87
88        Ok(scanner)
89    }
90
91    async fn read_logical_region(
92        &self,
93        logical_region_id: RegionId,
94        request: ScanRequest,
95    ) -> Result<RegionScannerRef> {
96        let _timer = MITO_OPERATION_ELAPSED
97            .with_label_values(&["read"])
98            .start_timer();
99
100        let physical_region_id = self.get_physical_region_id(logical_region_id).await?;
101        let data_region_id = utils::to_data_region_id(physical_region_id);
102        let request = self
103            .transform_request(physical_region_id, logical_region_id, request)
104            .await?;
105        let mut scanner = self
106            .mito
107            .handle_query(data_region_id, request)
108            .await
109            .context(MitoReadOperationSnafu)?;
110        scanner.set_logical_region(true);
111        scanner.set_query_load_region_id(data_region_id);
112
113        Ok(scanner)
114    }

162    /// Transform the [ScanRequest] from logical region to physical data region.
163    async fn transform_request(
164        &self,
165        physical_region_id: RegionId,
166        logical_region_id: RegionId,
167        mut request: ScanRequest,
168    ) -> Result<ScanRequest> {
169        // transform projection
170        let physical_projection = match request.projection.as_ref() {
171            Some(projection) => {
172                self.transform_projection(physical_region_id, logical_region_id, projection)
173                    .await?
174            }
175            None => {
176                self.default_projection(physical_region_id, logical_region_id)
177                    .await?
178            }
179        };
180
181        // Rewrite the projection from logical-region schema indices to
182        // physical-region schema indices.
183        request.projection = Some(physical_projection);
184
185        request
186            .filters
187            .push(self.table_id_filter(logical_region_id));
188
189        Ok(request)
190    }
191
192    /// Generate a filter on the table id column.
193    fn table_id_filter(&self, logical_region_id: RegionId) -> Expr {
194        logical_expr::col(DATA_SCHEMA_TABLE_ID_COLUMN_NAME)
195            .eq(logical_expr::lit(logical_region_id.table_id()))
196    }

198    /// Transform the projection from logical region to physical region.
199    ///
200    /// This method will not preserve internal columns.
201    pub async fn transform_projection(
202        &self,
203        physical_region_id: RegionId,
204        logical_region_id: RegionId,
205        origin_projection: &[usize],
206    ) -> Result<Vec<usize>> {
207        // project on logical columns
208        let all_logical_columns = self
209            .load_logical_column_names(physical_region_id, logical_region_id)
210            .await?;
211        let projected_logical_names = origin_projection
212            .iter()
213            .map(|&index| {
214                all_logical_columns
215                    .get(index)
216                    .map(String::as_str)
217                    .with_context(|| InvalidRequestSnafu {
218                        region_id: logical_region_id,
219                        reason: format!("projection index {index} is out of bounds"),
220                    })
221            })
222            .collect::<Result<Vec<_>>>()?;
223
224        // generate physical projection
225        let mut physical_projection = Vec::with_capacity(origin_projection.len());
226        let data_region_id = utils::to_data_region_id(physical_region_id);
227        let physical_metadata = self
228            .mito
229            .get_metadata(data_region_id)
230            .await
231            .context(MitoReadOperationSnafu)?;
232
233        for name in projected_logical_names {
234            // Safety: logical columns is a strict subset of physical columns
235            physical_projection.push(physical_metadata.column_index_by_name(name).unwrap());
236        }
237
238        Ok(physical_projection)
239    }
240
241    /// Default projection for a logical region. Includes non-internal columns
242    pub async fn default_projection(
243        &self,
244        physical_region_id: RegionId,
245        logical_region_id: RegionId,
246    ) -> Result<Vec<usize>> {
247        let logical_columns = self
248            .load_logical_column_names(physical_region_id, logical_region_id)
249            .await?;
250        let mut projection = Vec::with_capacity(logical_columns.len());
251        let data_region_id = utils::to_data_region_id(physical_region_id);
252        let physical_metadata = self
253            .mito
254            .get_metadata(data_region_id)
255            .await
256            .context(MitoReadOperationSnafu)?;
257        for name in logical_columns {
258            // Safety: logical columns is a strict subset of physical columns
259            projection.push(physical_metadata.column_index_by_name(&name).unwrap());
260        }
261
262        Ok(projection)
263    }
264
265    pub async fn logical_region_metadata(
266        &self,
267        physical_region_id: RegionId,
268        logical_region_id: RegionId,
269    ) -> Result<RegionMetadataRef> {
270        let logical_columns = self
271            .load_logical_columns(physical_region_id, logical_region_id)
272            .await?;
273
274        let primary_keys = logical_columns
275            .iter()
276            .filter_map(|col| {
277                if col.semantic_type == SemanticType::Tag {
278                    Some(col.column_id)
279                } else {
280                    None
281                }
282            })
283            .collect::<Vec<_>>();
284
285        let mut logical_metadata_builder = RegionMetadataBuilder::new(logical_region_id);
286        for col in logical_columns {
287            logical_metadata_builder.push_column_metadata(col);
288        }
289        logical_metadata_builder.primary_key(primary_keys);
290        let logical_metadata = logical_metadata_builder
291            .build()
292            .context(InvalidMetadataSnafu)?;
293
294        Ok(Arc::new(logical_metadata))
295    }

319                .await
320                .map_err(common_error::ext::BoxedError::new)?;
321            self.mito
322                .scan_to_stream(physical_region_id, request)
323                .await
324                .map_err(common_error::ext::BoxedError::new)
325        }
326    }

359    async fn count_scanner_rows(scanner: &mut RegionScannerRef, partitions: usize) -> usize {
360        let ranges = scanner
361            .properties()
362            .partitions
363            .iter()
364            .flatten()
365            .copied()
366            .collect::<Vec<_>>();
367        let mut prepared = vec![Vec::new(); partitions];
368        prepared[0] = ranges;
369        scanner
370            .prepare(
371                PrepareRequest::default()
372                    .with_ranges(prepared)
373                    .with_target_partitions(partitions),
374            )
375            .unwrap();
376
377        let metrics = ExecutionPlanMetricsSet::default();
378        let context = QueryScanContext::default();
379        let streams = (0..partitions)
380            .map(|partition| {
381                scanner
382                    .scan_partition(&context, &metrics, partition)
383                    .unwrap()
384            })
385            .collect::<Vec<_>>();
386        try_join_all(streams.into_iter().map(|stream| async move {
387            stream
388                .try_fold(0, |rows, batch| async move { Ok(rows + batch.num_rows()) })
389                .await
390        }))
391        .await
392        .unwrap()
393        .into_iter()
394        .sum()
395    }
