32impl MetricEngineInner {
33    /// Syncs the logical regions from the source region to the target region in the metric engine.
34    ///
35    /// This operation:
36    /// 1. Copies SST files from source metadata region to target metadata region
37    /// 2. Transforms logical region metadata (updates region numbers to match target)
38    /// 3. Edits target manifest to remove old file entries (copied files)
39    /// 4. Recovers states and returns newly opened logical region IDs
40    ///
41    /// **Note**: Only the metadata region is synced. The data region is not affected.
42    pub(crate) async fn sync_region_from_region(
43        &self,
44        region_id: RegionId,
45        source_region_id: RegionId,
46        parallelism: usize,
47    ) -> Result<SyncRegionFromResponse> {
48        let source_metadata_region_id = utils::to_metadata_region_id(source_region_id);
49        let target_metadata_region_id = utils::to_metadata_region_id(region_id);
50        let target_data_region_id = utils::to_data_region_id(region_id);
51        let source_data_region_id = utils::to_data_region_id(source_region_id);
52        info!(
53            "Syncing region from region {} to region {}, parallelism: {}",
54            source_region_id, region_id, parallelism
55        );
56
57        // Cleanup removes the original copied file IDs, so compaction must not replace them.
58        let guard = self
59            .mito
60            .pause_compaction(target_metadata_region_id)
61            .await
62            .map_err(BoxedError::new)
63            .context(MitoPauseCompactionOperationSnafu {
64                region_id: target_metadata_region_id,
65            })?;
66        let target_region = self.mito.find_region(target_metadata_region_id).context(
67            PhysicalRegionNotFoundSnafu {
68                region_id: target_metadata_region_id,
69            },
70        )?;
71        // Under the compaction guard, only SSTs present before import can retain
72        // source keys after we remove this attempt's copied files.
73        let had_ssts = target_region.has_ssts();
74        let res = self
75            .mito
76            .copy_region_from(
77                target_metadata_region_id,
78                MitoCopyRegionFromRequest {
79                    source_region_id: source_metadata_region_id,
80                    parallelism,
81                },
82            )
83            .await
84            .map_err(BoxedError::new)
85            .context(MitoCopyRegionFromOperationSnafu {
86                source_region_id: source_metadata_region_id,
87                target_region_id: target_metadata_region_id,
88            })?;
89
90        if !res.copied_file_ids.is_empty() {
91            self.transform_imported_files(
92                &target_region,
93                source_data_region_id,
94                &res.copied_file_ids,
95            )
96            .await?;
97        }
98        if had_ssts && target_region.has_ssts() {
99            self.metadata_region
100                .remove_imported_keys(target_metadata_region_id, source_data_region_id)
101                .await?;
102        }
103        drop(guard);
104        if res.copied_file_ids.is_empty() {
105            info!(
106                "No files were copied from source region {} to target region {}, copied file ids are empty",
107                source_metadata_region_id, target_metadata_region_id
108            );
109            return Ok(SyncRegionFromResponse::Metric {
110                metadata_synced: false,
111                data_synced: false,
112                new_opened_logical_region_ids: vec![],
113            });
114        }
115
116        let now = Instant::now();
117        // Always recover states from the target metadata region after syncing
118        // from the source metadata region.
119        let physical_region_options = *self
120            .state
121            .read()
122            .unwrap()
123            .physical_region_states()
124            .get(&target_data_region_id)
125            .context(PhysicalRegionNotFoundSnafu {
126                region_id: target_data_region_id,
127            })?
128            .options();
129        let new_opened_logical_region_ids = self
130            .recover_states(target_data_region_id, physical_region_options)
131            .await?;
132        info!(
133            "Sync metadata region from source region {} to target region {}, recover states cost: {:?}, new opened logical region ids: {:?}",
134            source_metadata_region_id,
135            target_metadata_region_id,
136            now.elapsed(),
137            new_opened_logical_region_ids
138        );
139
140        Ok(SyncRegionFromResponse::Metric {
141            metadata_synced: true,
142            data_synced: false,
143            new_opened_logical_region_ids,
144        })
145    }

147    async fn transform_imported_files(
148        &self,
149        target_region: &MitoRegion,
150        source_data_region_id: RegionId,
151        file_ids: &[FileId],
152    ) -> Result<()> {
153        let target_metadata_region_id = target_region.region_id();
154        let target_data_region_id = utils::to_data_region_id(target_metadata_region_id);
155        let files_to_remove = target_region.file_metas(file_ids).await;
156        let missing_file_ids = file_ids
157            .iter()
158            .zip(&files_to_remove)
159            .filter_map(|(file_id, maybe_meta)| {
160                if maybe_meta.is_none() {
161                    Some(*file_id)
162                } else {
163                    None
164                }
165            })
166            .collect::<Vec<_>>();
167        let transform = if missing_file_ids.is_empty() {
168            self.metadata_region
169                .transform_logical_region_metadata(target_data_region_id, source_data_region_id)
170                .await
171        } else {
172            MissingFilesSnafu {
173                region_id: target_metadata_region_id,
174                file_ids: missing_file_ids,
175            }
176            .fail()
177        };
178        let files_to_remove: Vec<_> = files_to_remove.into_iter().flatten().collect();
179        if !files_to_remove.is_empty() {
180            let edit = RegionEdit {
181                files_to_remove,
182                timestamp_ms: Some(chrono::Utc::now().timestamp_millis()),
183                ..Default::default()
184            };
185            let cleanup = self
186                .mito
187                .edit_region(target_metadata_region_id, edit)
188                .await
189                .map_err(BoxedError::new)
190                .context(MitoEditRegionSnafu {
191                    region_id: target_metadata_region_id,
192                });
193            if let Err(err) = cleanup {
194                if transform.is_ok() {
195                    return Err(err);
196                }
197                warn!(err; "Failed to remove copied SSTs from {target_metadata_region_id} after sync failed");
198            }
199        }
200        transform
201    }

265    async fn test_metadata_sync_blocks_compaction_until_cleanup() {
266        let env = TestEnv::new().await;
267        env.init_metric_region().await;
268        let source = env.default_physical_region_id();
269        let logical = env.default_logical_region_id();
270        let target = RegionId::new(source.table_id(), 7);
271        env.create_physical_region(target, &TestEnv::default_table_dir(), vec![])
272            .await;
273        // Leave target rows in the memtable so the intervening flush creates an SST.
274        let existing = RegionId::new(4, target.region_number());
275        env.create_logical_region(target, existing).await;
276        let metric = env.metric();
277        metric
278            .handle_request(source, RegionRequest::Flush(Default::default()))
279            .await
280            .unwrap();
281        let target_metadata = crate::utils::to_metadata_region_id(target);
282        let metadata = &metric.inner.metadata_region;
283        let mito = env.mito();
284
285        // A source-numbered orphan column in the target memtable makes an unnecessary
286        // fallback scan observable: the transform ignores it, but the fallback deletes it.
287        let probe =
288            MetadataRegion::concat_column_key(RegionId::new(99, source.region_number()), "probe");
289        let put = MetadataRegion::build_put_request_from_iter(std::iter::once((
290            probe.clone(),
291            String::new(),
292        )));
293        mito.handle_request(target_metadata, RegionRequest::Put(put))
294            .await
295            .unwrap();
296        assert!(!mito.find_region(target_metadata).unwrap().has_ssts());
297
298        // Hold the transform at its metadata write. The copied source marker is
299        // observable through a key-only scan while the sync waits for the lock.
300        let observe = metadata.with_writes_paused(target_metadata, async {
301            tokio::time::timeout(std::time::Duration::from_secs(10), async {
302                while !metadata
303                    .logical_regions(target)
304                    .await
305                    .unwrap()
306                    .contains(&logical)
307                {
308                    tokio::task::yield_now().await;
309                }
310            })
311            .await
312            .unwrap();
313            mito.handle_request(target_metadata, RegionRequest::Flush(Default::default()))
314                .await
315                .unwrap();
316            let err = mito
317                .handle_request(target_metadata, RegionRequest::Compact(Default::default()))
318                .await
319                .unwrap_err();
320            assert!(matches!(
321                err.into_inner()
322                    .as_any()
323                    .downcast_ref::<mito2::error::Error>(),
324                Some(mito2::error::Error::CompactionCancelled { .. })
325            ));
326        });
327        let sync = metric.sync_region(
328            target,
329            SyncRegionFromRequest::FromRegion {
330                source_region_id: source,
331                parallelism: 1,
332            },
333        );
334        let ((), result) = tokio::join!(observe, sync);
335        result.unwrap();
336        let remapped = RegionId::new(logical.table_id(), target.region_number());
337        let mut recovered = metadata.logical_regions(target).await.unwrap();
338        recovered.sort_unstable();
339        assert_eq!(recovered, [remapped, existing]);
340        assert!(mito.find_region(target_metadata).unwrap().has_ssts());
341        assert!(
342            metadata
343                .get(target_metadata, &probe)
344                .await
345                .unwrap()
346                .is_some()
347        );
348        mito.handle_request(target_metadata, RegionRequest::Compact(Default::default()))
349            .await
350            .unwrap();
351    }

358    async fn test_sync_cleans_imported_files_on_error(
359        #[case] invalid_column: bool,
360        #[case] fail_cleanup: bool,
361    ) {
362        use mito2::test_util::{DeltaFaults, TestEnv as MitoTestEnv};
363        use store_api::region_engine::MitoCopyRegionFromRequest;
364
365        let faults = DeltaFaults::default();
366        let mito_env = MitoTestEnv::with_prefix("sync-cleanup")
367            .await
368            .with_mock_layer(faults.layer());
369        let env = TestEnv::with_mito_env(mito_env).await;
370        env.init_metric_region().await;
371        let source = env.default_physical_region_id();
372        let logical = env.default_logical_region_id();
373        let target = RegionId::new(source.table_id(), 7);
374        env.create_physical_region(target, &TestEnv::default_table_dir(), vec![])
375            .await;
376        let source_meta = crate::utils::to_metadata_region_id(source);
377        let target_meta = crate::utils::to_metadata_region_id(target);
378        let mito = env.mito();
379        let source_guard = mito.pause_compaction(source_meta).await.unwrap();
380        if invalid_column {
381            let entries = [(format!("__column_{}_!", logical.as_u64()), String::new())];
382            let put = MetadataRegion::build_put_request_from_iter(entries.into_iter());
383            mito.handle_request(source_meta, RegionRequest::Put(put))
384                .await
385                .unwrap();
386        }
387        mito.handle_request(source_meta, RegionRequest::Flush(Default::default()))
388            .await
389            .unwrap();
390        if !invalid_column {
391            // Give compaction distinct flushed SSTs without changing the source schema.
392            for _ in 0..4 {
393                let marker = (MetadataRegion::concat_region_key(logical), String::new());
394                let put = MetadataRegion::build_put_request_from_iter(std::iter::once(marker));
395                mito.handle_request(source_meta, RegionRequest::Put(put))
396                    .await
397                    .unwrap();
398                mito2::test_util::flush_region(&mito, source_meta, None).await;
399            }
400        }
401        // Model a retry after import. The next manifest write will be the cleanup edit.
402        let imported = mito
403            .copy_region_from(
404                target_meta,
405                MitoCopyRegionFromRequest {
406                    source_region_id: source_meta,
407                    parallelism: 1,
408                },
409            )
410            .await
411            .unwrap();
412        assert!(!imported.copied_file_ids.is_empty());
413        if fail_cleanup {
414            faults.fail_next_write();
415        }
416        let err = env
417            .metric()
418            .inner
419            .sync_region_from_region(target, source, 1)
420            .await
421            .unwrap_err();
422        if invalid_column {
423            assert!(
424                matches!(err, crate::error::Error::DecodeColumnValue { .. }),
425                "{err}"
426            );
427        } else {
428            assert!(
429                matches!(err, crate::error::Error::MitoEditRegion { .. }),
430                "{err}"
431            );
432        }
433        if !invalid_column {
434            drop(source_guard);
435            mito.handle_request(source_meta, RegionRequest::Compact(Default::default()))
436                .await
437                .unwrap();
438            let source_region = mito.find_region(source_meta).unwrap();
439            assert!(
440                source_region
441                    .file_metas(&imported.copied_file_ids)
442                    .await
443                    .iter()
444                    .any(Option::is_none)
445            );
446            // An uncertain manifest edit closes the target. Reopen it before retrying.
447            mito.handle_request(
448                target_meta,
449                RegionRequest::Open(RegionOpenRequest {
450                    engine: mito2::engine::MITO_ENGINE_NAME.into(),
451                    table_dir: TestEnv::default_table_dir(),
452                    path_type: PathType::Metadata,
453                    options: Default::default(),
454                    skip_wal_replay: false,
455                    checkpoint: None,
456                    requirements: Default::default(),
457                }),
458            )
459            .await
460            .unwrap();
461            mito.set_region_role(target_meta, store_api::region_engine::RegionRole::Leader)
462                .unwrap();
463            env.metric()
464                .inner
465                .sync_region_from_region(target, source, 1)
466                .await
467                .unwrap();
468            let metadata = env.metadata_region();
469            let remapped = RegionId::new(logical.table_id(), target.region_number());
470            assert_eq!(metadata.logical_regions(target).await.unwrap(), [remapped]);
471            assert_logical_table_columns(&metadata, target, logical, &[]).await;
472            assert_logical_table_columns(
473                &metadata,
474                target,
475                remapped,
476                &["greptime_timestamp", "greptime_value", "job"],
477            )
478            .await;
479        }
480        if !fail_cleanup {
481            let region = mito.find_region(target_meta).unwrap();
482            assert!(
483                region
484                    .file_metas(&imported.copied_file_ids)
485                    .await
486                    .iter()
487                    .all(Option::is_none)
488            );
489        }
490    }

631        let dir = TestEnv::default_table_dir();
632        env.init_metric_region().await;
633        env.create_physical_region(target, &dir, vec![]).await;
634        let engine = env.metric();
635        engine.inner.flush_task.stop().await.unwrap();
636        engine
637            .handle_request(source, RegionRequest::Flush(Default::default()))
638            .await
639            .unwrap();
640
641        // Model a crash after the input SSTs are installed and one column put
642        // has reached the WAL, before the new region marker has been written.
643        let target_metadata = crate::utils::to_metadata_region_id(target);
644        env.mito()
645            .copy_region_from(
646                target_metadata,
647                MitoCopyRegionFromRequest {
648                    source_region_id: crate::utils::to_metadata_region_id(source),
649                    parallelism: 1,
650                },
651            )
652            .await
653            .unwrap();
654        let (name, column) = env
655            .metadata_region()
656            .logical_columns(source, logical)
657            .await
658            .unwrap()
659            .pop()
660            .unwrap();
661        let put = MetadataRegion::build_put_request_from_iter(std::iter::once((
662            MetadataRegion::concat_column_key(target_logical, &name),
663            MetadataRegion::serialize_column_metadata(&column),
664        )));
665        env.mito()
666            .handle_request(target_metadata, RegionRequest::Put(put))
667            .await
668            .unwrap();
669        env.reopen_engine(Default::default()).await;
670        let engine = env.metric();
671        engine
672            .handle_request(
673                target,
674                RegionRequest::Open(RegionOpenRequest {
675                    engine: METRIC_ENGINE_NAME.to_string(),
676                    table_dir: dir,
677                    path_type: PathType::Bare,
678                    options: [(PHYSICAL_TABLE_METADATA_KEY.to_string(), String::new())]
679                        .into_iter()
680                        .collect(),
681                    skip_wal_replay: false,
682                    checkpoint: None,
683                    requirements: Default::default(),
684                }),
685            )
686            .await
687            .unwrap();
688        // The heartbeat sync handler only sends FromRegion to a writable leader.
689        engine.set_region_role(target, RegionRole::Leader).unwrap();
690        let result = engine
691            .sync_region(
692                target,
693                SyncRegionFromRequest::FromRegion {
694                    source_region_id: source,
695                    parallelism: 1,
696                },
697            )
698            .await
699            .unwrap();
700        assert_eq!(
701            result.new_opened_logical_region_ids().unwrap(),
702            vec![target_logical]
703        );
704        assert_eq!(
705            engine.logical_regions(target).await.unwrap(),
706            vec![target_logical]
707        );
708        assert_logical_table_columns(
709            &env.metadata_region(),
710            target,
711            target_logical,
712            &["greptime_timestamp", "greptime_value", "job"],
713        )
714        .await;
715    }
