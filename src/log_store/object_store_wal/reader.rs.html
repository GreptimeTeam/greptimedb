46/// Node-wide limit on concurrent remote object index reads. Startup recovery
47/// keeps its existing, separate concurrency setting.
48const FOOTER_CONCURRENCY: usize = 16;
49
50/// A normalized WAL node prefix: `<root>/datanodes/<node>/epochs/<generation>`.
51/// Construct it from a persisted source prefix or
52/// [`ObjectStoreWalConfig::node_prefix`](common_wal::config::object_store::ObjectStoreWalConfig::node_prefix).
53#[derive(Debug, Clone, PartialEq, Eq, Hash)]
54pub struct NodeWalPrefix(String);
55
56impl NodeWalPrefix {
57    /// Validates the node layout and removes a trailing slash.
58    pub fn try_new(prefix: &str) -> Result<Self> {
59        let prefix = io::normalize_prefix(prefix)?;
60        let mut parts = prefix.rsplitn(5, '/');
61        let generation = parts.next();
62        let epochs = parts.next();
63        let node = parts.next();
64        let datanodes = parts.next();
65        let root = parts.next();
66        let is_id = |part: Option<&str>| {
67            part.is_some_and(|part| {
68                (part == "0" || !part.starts_with('0'))
69                    && part.bytes().all(|byte| byte.is_ascii_digit())
70                    && part.parse::<u64>().is_ok()
71            })
72        };
73        ensure!(
74            is_id(generation)
75                && epochs == Some("epochs")
76                && is_id(node)
77                && datanodes == Some("datanodes")
78                && root.is_some(),
79            InvalidWalObjectStoreSnafu {
80                reason: format!("expected a WAL node prefix, got {prefix:?}"),
81            }
82        );
83        Ok(Self(prefix))
84    }
85
86    /// Returns the normalized prefix used in object keys and providers.
87    pub fn as_str(&self) -> &str {
88        &self.0
89    }
90}

92/// Readers shared by source prefix on one destination node and one WAL backend.
93/// Keep this registry for the node's lifetime, and hold each returned reader
94/// while any recovery range still references that source. Dropping the last
95/// reader releases its cached indexes; no background task or writer is started.
96pub struct ObjectStoreWalReaders {
97    store: ObjectStore,
98    readers: Mutex<HashMap<NodeWalPrefix, Weak<ObjectStoreWalReader>>>,
99    footer_budget: Arc<Semaphore>,
100}
101
102impl ObjectStoreWalReaders {
103    /// Creates an empty registry. Construction performs no storage requests.
104    pub fn new(store: ObjectStore) -> Self {
105        Self {
106            store,
107            readers: Mutex::new(HashMap::new()),
108            footer_budget: Arc::new(Semaphore::new(FOOTER_CONCURRENCY)),
109        }
110    }
111
112    /// Returns the shared reader for a full node prefix, without probing writes
113    /// or listing objects. Only read/list permissions are needed on the backend.
114    pub fn reader(&self, prefix: &NodeWalPrefix) -> Result<Arc<ObjectStoreWalReader>> {
115        let mut readers = self.readers.lock().unwrap_or_else(PoisonError::into_inner);
116        if let Some(reader) = readers.get(prefix).and_then(Weak::upgrade) {
117            return Ok(reader);
118        }
119        readers.retain(|_, reader| reader.strong_count() != 0);
120        let reader = Arc::new(ObjectStoreWalReader {
121            io: Arc::new(ObjectStoreIo::new_readonly(
122                self.store.clone(),
123                prefix.as_str(),
124            )?),
125            indexes: CachedObject::cache(),
126            footer_budget: self.footer_budget.clone(),
127            prefetch_budget: Arc::new(Semaphore::new(prefetch::READ_PREFETCH_BYTES)),
128        });
129        readers.insert(prefix.clone(), Arc::downgrade(&reader));
130        Ok(reader)
131    }
132}

134/// Reads remote WAL ranges without claiming an epoch or modifying the source.
135/// The reader never honors missing-history or corrupted-segment skip options.
136/// It does not own region roles, manifests, release positions or replay tasks.
137pub struct ObjectStoreWalReader {
138    io: Arc<dyn WalObjectRead>,
139    /// Immutable indexes keyed by listed metadata, sharing loads within this prefix.
140    indexes: Cache<ListedObject, Arc<CachedObject>>,
141    /// Footer-load concurrency shared by all readers in the node's registry.
142    footer_budget: Arc<Semaphore>,
143    /// Prefetched bytes shared by reads of this source prefix only.
144    prefetch_budget: Arc<Semaphore>,
145}
146
147/// Cached metadata for an immutable WAL object, without recovery release records.
148struct CachedObject {
149    object: ListedObject,
150    header: Header,
151    footer: Vec<FooterEntry>,
152}
153
154impl CachedObject {
155    /// Bounds retained index weight per source prefix. Eviction is asynchronous;
156    /// active reads can retain their own snapshots beyond this capacity.
157    fn cache() -> Cache<ListedObject, Arc<Self>> {
158        Cache::new(
159            "object_store_wal_indexes",
160            64 * 1024 * 1024,
161            |key: &ListedObject, value: &Arc<Self>| {
162                let bytes = std::mem::size_of::<ListedObject>()
163                    + key.path.capacity()
164                    + std::mem::size_of::<Self>()
165                    + value.object.path.capacity()
166                    + value.footer.capacity() * std::mem::size_of::<FooterEntry>();
167                u32::try_from(bytes).unwrap_or(u32::MAX)
168            },
169            |_, _, _| {},
170        )
171    }
172}

174impl ObjectStoreWalReader {
175    /// The normalized node prefix used by every entry's provider.
176    pub fn prefix(&self) -> &str {
177        self.io.prefix()
178    }
179
180    /// Lists keys only, returning a bound at or above every entry in the listed
181    /// objects. An empty prefix is an error, not an empty tail. No object index
182    /// or data is read on this readiness path. The bound covers the whole prefix;
183    /// `region_id` identifies the recovery request in missing-history errors.
184    pub async fn list_upper_bound(&self, region_id: RegionId) -> Result<EntryId> {
185        let objects = self.io.list().await?;
186        let seq = objects
187            .iter()
188            .map(|object| object.object_seq)
189            .max()
190            .with_context(|| MissingSourceWalHistorySnafu {
191                region_id,
192                prefix: self.prefix().to_string(),
193                storage: self.io.storage(),
194                reason: format!(
195                    "empty WAL prefix {} while determining takeover bound",
196                    self.prefix()
197                ),
198            })?;
199        ensure!(
200            seq < OBJECT_SEQ_LIMIT,
201            CorruptedWalObjectSnafu {
202                reason: format!(
203                    "listed key {} has sequence {seq} outside the entry-id range",
204                    self.io.object_path(seq)
205                ),
206            }
207        );
208        Ok(entry_id(seq, POSITION_LIMIT - 1))
209    }
210
211    /// Prepares a fresh view and streams only this region's `(after, upper]`.
212    /// Call after the recovery range has been registered in the manifest. Each
213    /// call (including a retry) starts a new LIST; cached indexes never replace
214    /// that LIST. Previously indexed, still-listed immutable objects are reused.
215    ///
216    /// The store writing the prefix may collect listed objects before their
217    /// indexes are read. An object gone below the sequence where the read stops
218    /// following links to predecessors (a reclaim boundary, or the object of
219    /// `after` when it is higher in the `LastEntry` mode) is skipped, as
220    /// startup recovery skips it, unless it holds the reported entry. One gone
221    /// at or above it, on the chain or not, fails the read; a retry lists the
222    /// prefix again.
223    ///
224    /// Errors can occur after entries have been yielded. The caller must finish
225    /// the stream successfully before publishing recovery progress. In the
226    /// `LastEntry` mode success also proves that the reported entry was read.
227    pub async fn read(
228        &self,
229        region_id: RegionId,
230        after: EntryId,
231        upper: WalReadUpperBound,
232    ) -> Result<SendableEntryStream<'static, Entry, Error>> {
233        let upto = upper.entry_id();
234        ensure!(
235            after < upto,
236            InvalidWalEntryRangeSnafu {
237                region_id,
238                start_entry_id: after,
239                end_entry_id: upto,
240            }
241        );
242        let catalog = self.prepare(region_id, after, upper).await?;
243        let objects = catalog
244            .objects_for_entry_range(region_id, after + 1, upto)?
245            .into_iter()
246            .map(|(seq, entry)| (seq, entry.clone()))
247            .collect();
248        let mut segments =
249            prefetch::segments(self.io.clone(), objects, self.prefetch_budget.clone());
250        let provider = Provider::object_store_provider(region_id, self.prefix().to_string());
251        Ok(Box::pin(try_stream! {
252            let mut reached_upto = false;
253            while let Some((_, _, fetched)) = segments.next().await {
254                let entries = fetched?.into_iter()
255                    .filter(|record| record.entry_id > after && record.entry_id <= upto)
256                    .map(|record| {
257                        reached_upto |= record.entry_id == upto;
258                        Entry::Naive(NaiveEntry {
259                            provider: provider.clone(), region_id,
260                            entry_id: record.entry_id, data: record.payload.into(),
261                        })
262                    }).collect::<Vec<_>>();
263                if !entries.is_empty() { yield entries; }
264            }
265            if matches!(upper, WalReadUpperBound::LastEntry(_)) && !reached_upto {
266                Err(CorruptedWalObjectSnafu {
267                    reason: format!("WAL recovery of region {region_id} did not reach reported entry {upto}"),
268                }.build())?;
269            }
270        }))
271    }
272
273    async fn prepare(
274        &self,
275        region_id: RegionId,
276        after: EntryId,
277        upper: WalReadUpperBound,
278    ) -> Result<ObjectCatalog> {
279        let start = match upper {
280            WalReadUpperBound::LastEntry(_) => after >> POSITION_BITS,
281            WalReadUpperBound::Listed(_) => 0,
282        };
283        let listed = self.io.list_from(start).await?;
284        let present = listed.iter().cloned().collect::<HashSet<_>>();
285        // Each read owns its LIST snapshot. Drop cached indexes outside it,
286        // accepting a refetch for later reads with a lower watermark. In-flight
287        // loads may still insert old indexes; capacity also bounds retention.
288        let _ = self
289            .indexes
290            .invalidate_if(move |object, _| !present.contains(object));
291        ensure!(
292            !listed.is_empty(),
293            MissingSourceWalHistorySnafu {
294                region_id,
295                prefix: self.prefix().to_string(),
296                storage: self.io.storage(),
297                reason: format!("no WAL objects from sequence {start} in {}", self.prefix()),
298            }
299        );
300        if let WalReadUpperBound::Listed(upto) = upper {
301            ensure!(
302                listed
303                    .iter()
304                    .any(|object| object.object_seq >= upto >> POSITION_BITS),
305                CorruptedWalObjectSnafu {
306                    reason: format!(
307                        "WAL listing of {} does not reach recorded bound {upto}",
308                        self.prefix()
309                    ),
310                }
311            );
312        }
313        let (objects, vanished) = index::fetch_listed(
314            futures::stream::iter(listed.into_iter().map(Ok)),
315            FOOTER_CONCURRENCY,
316            |object| async move {
317                let seq = object.object_seq;
318                // Include listing metadata in the key so a load for a changed
319                // object cannot join an older in-flight load of the same sequence.
320                let cached = self
321                    .indexes
322                    .try_get_with(&object, async {
323                        // Waiters share the loader without consuming footer permits.
324                        let _permit = self.footer_budget.acquire().await.map_err(|_| {
325                            CorruptedWalObjectSnafu {
326                                reason: "remote WAL footer budget closed".to_string(),
327                            }
328                            .build()
329                        })?;
330                        let fetched = index::fetch_footer(self.io.as_ref(), &object).await?;
331                        Ok::<_, Error>(Arc::new(CachedObject {
332                            object: object.clone(),
333                            header: fetched.header,
334                            footer: fetched.footer,
335                        }))
336                    })
337                    .await
338                    .context(ObjectStoreWalSnafu)?;
339                Ok::<_, Error>((seq, cached))
340            },
341        )
342        .await?;
343        let objects = objects.into_iter().collect::<BTreeMap<_, _>>();
344        let headers = objects
345            .iter()
346            .map(|(seq, object)| (*seq, &object.header))
347            .collect::<BTreeMap<_, _>>();
348        let chain = match upper {
349            WalReadUpperBound::LastEntry(last) => {
350                // A later writer may already have reclaimed the oldest part.
351                // The reported L itself must still exist and be replayed.
352                let boundary = headers
353                    .values()
354                    .map(|header| header.reclaim_boundary)
355                    .max()
356                    .unwrap_or(0)
357                    .max(start);
358                index::ensure_vanished_below(vanished, boundary)?;
359                index::chain_from(&headers, last >> POSITION_BITS, boundary).with_context(|| {
360                    CorruptedWalObjectSnafu {
361                        reason: format!(
362                            "missing WAL chain ending at reported entry {last} in {}",
363                            self.prefix()
364                        ),
365                    }
366                })?
367            }
368            WalReadUpperBound::Listed(_) => {
369                let chain = index::select_chain(&headers);
370                let boundary = chain.last().map_or(0, |tip| headers[tip].reclaim_boundary);
371                index::ensure_vanished_below(vanished, boundary)?;
372                let tip = chain.last().with_context(|| CorruptedWalObjectSnafu {
373                    reason: format!("no complete WAL chain in {}", self.prefix()),
374                })?;
375                let max_epoch = headers
376                    .values()
377                    .map(|header| header.epoch)
378                    .max()
379                    .unwrap_or(0);
380                ensure!(
381                    headers[tip].epoch == max_epoch,
382                    CorruptedWalObjectSnafu {
383                        reason: format!(
384                            "WAL chain in {} would fall back from epoch {max_epoch} to {}",
385                            self.prefix(),
386                            headers[tip].epoch
387                        ),
388                    }
389                );
390                chain
391            }
392        };
393        let mut catalog = ObjectCatalog::default();
394        let mut history_end = 0;
395        for seq in chain {
396            let object = &objects[&seq];
397            history_end = history_end.max(
398                object
399                    .footer
400                    .iter()
401                    .map(|entry| sequence_floor(entry.max_entry_id))
402                    .max()
403                    .unwrap_or(0),
404            );
405            // Keep only this region in the per-read catalog. All regions share
406            // the source's cached footers, rather than copying the entire node
407            // index into every concurrent region recovery.
408            let footer = object
409                .footer
410                .iter()
411                .filter(|entry| entry.region_id == region_id)
412                .cloned()
413                .collect();
414            catalog
415                .insert_object(seq, footer)
416                .with_context(|_| InvalidWalObjectSnafu {
417                    path: object.object.path.clone(),
418                })?;
419        }
420        if matches!(upper, WalReadUpperBound::Listed(_))
421            && let Some(reason) =
422                history::missing_history(&catalog, region_id, after, false, history_end)
423        {
424            return MissingSourceWalHistorySnafu {
425                region_id,
426                prefix: self.prefix().to_string(),
427                storage: self.io.storage(),
428                reason,
429            }
430            .fail();
431        }
432        Ok(catalog)
433    }
