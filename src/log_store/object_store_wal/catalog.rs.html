30/// Indexes objects by sequence and, per region, the objects that hold entries
31/// of that region.
32#[derive(Debug, Default)]
33pub(crate) struct ObjectCatalog {
34    objects: BTreeMap<u64, Vec<FooterEntry>>,
35    regions: BTreeMap<RegionId, BTreeMap<u64, FooterEntry>>,
36    /// Largest entry id ever indexed per region, kept after the object that
37    /// held it was removed.
38    max_entry_ids: HashMap<RegionId, EntryId>,
39}
40
41impl ObjectCatalog {
42    /// Indexes the footer of the object `object_seq`. Objects may be inserted
43    /// in any order, which lets recovery index them as it discovers them.
44    /// Inserting a sequence that is already indexed is rejected, whether or not
45    /// the footer matches the indexed one. An object without segments holds no
46    /// entries but still takes its sequence.
47    pub(crate) fn insert_object(
48        &mut self,
49        object_seq: u64,
50        mut footer: Vec<FooterEntry>,
51    ) -> Result<()> {
52        footer.sort_unstable_by_key(|entry| entry.region_id);
53        for entries in footer.windows(2) {
54            ensure!(
55                entries[0].region_id != entries[1].region_id,
56                CorruptedWalObjectSnafu {
57                    reason: format!(
58                        "object {object_seq} has duplicate footer entries for region {}",
59                        entries[0].region_id
60                    ),
61                }
62            );
63        }
64        for entry in &footer {
65            ensure!(
66                entry.entry_count > 0 && entry.min_entry_id <= entry.max_entry_id,
67                CorruptedWalObjectSnafu {
68                    reason: format!(
69                        "object {object_seq} has invalid entry range {}..={} with {} entries for region {}",
70                        entry.min_entry_id, entry.max_entry_id, entry.entry_count, entry.region_id
71                    ),
72                }
73            );
74        }
75
76        // Object sequences are unique: recovery indexes every listed key once, and
77        // accepting a repeated insertion would hide a caller that lost track of it.
78        // Retrying an identical write stays an object store concern.
79        ensure!(
80            !self.objects.contains_key(&object_seq),
81            CorruptedWalObjectSnafu {
82                reason: format!("object {object_seq} is already indexed"),
83            }
84        );
85
86        // Validate every region before mutating either index so insertion is atomic.
87        for entry in &footer {
88            let Some(region_objects) = self.regions.get(&entry.region_id) else {
89                continue;
90            };
91            if let Some((&previous_seq, previous)) = region_objects.range(..object_seq).next_back()
92            {
93                ensure!(
94                    previous.max_entry_id < entry.min_entry_id,
95                    CorruptedWalObjectSnafu {
96                        reason: out_of_order_reason(
97                            entry.region_id,
98                            previous_seq,
99                            previous.max_entry_id,
100                            object_seq,
101                            entry.min_entry_id
102                        ),
103                    }
104                );
105            }
106            if let Some((&next_seq, next)) = region_objects
107                .range((Excluded(object_seq), Unbounded))
108                .next()
109            {
110                ensure!(
111                    entry.max_entry_id < next.min_entry_id,
112                    CorruptedWalObjectSnafu {
113                        reason: out_of_order_reason(
114                            entry.region_id,
115                            object_seq,
116                            entry.max_entry_id,
117                            next_seq,
118                            next.min_entry_id
119                        ),
120                    }
121                );
122            }
123        }
124
125        for entry in &footer {
126            self.regions
127                .entry(entry.region_id)
128                .or_default()
129                .insert(object_seq, entry.clone());
130            self.max_entry_ids
131                .entry(entry.region_id)
132                .and_modify(|max| *max = (*max).max(entry.max_entry_id))
133                .or_insert(entry.max_entry_id);
134        }
135        self.objects.insert(object_seq, footer);
136        Ok(())
137    }
138
139    /// Removes the object `object_seq` from the index once it was deleted and
140    /// returns the footer it held, or `None` if it was not indexed. The
141    /// largest entry id of every region it held is kept.
142    pub(crate) fn remove_object(&mut self, object_seq: u64) -> Option<Vec<FooterEntry>> {
143        let footer = self.objects.remove(&object_seq)?;
144        for entry in &footer {
145            if let Some(objects) = self.regions.get_mut(&entry.region_id) {
146                objects.remove(&object_seq);
147                if objects.is_empty() {
148                    self.regions.remove(&entry.region_id);
149                }
150            }
151        }
152        Some(footer)
153    }
154
155    /// Returns true while the object `object_seq` is indexed.
156    pub(crate) fn contains_object(&self, object_seq: u64) -> bool {
157        self.objects.contains_key(&object_seq)
158    }
159
160    /// Returns the reclaim boundary the indexed objects and the obsolete
161    /// watermarks allow, given that every object below `from` holds no live
162    /// segment: the first indexed object at or above `from` that holds a
163    /// live segment, or the sequence after the last indexed object when none
164    /// does, and never less than `from`.
165    ///
166    /// A segment is live when its maximum entry id is above the watermark of
167    /// its region, or its region has no watermark; a region whose watermark
168    /// is [`EntryId::MAX`] holds no live segment. Watermarks never move down
169    /// and a region first appears in an object above the boundary, so the
170    /// scan starts at the previous boundary and never reads an object twice
171    /// once the boundary has passed it.
172    pub(crate) fn reclaim_boundary(
173        &self,
174        from: u64,
175        obsolete_entry_ids: &HashMap<RegionId, EntryId>,
176    ) -> u64 {
177        let live = self.objects.range(from..).find(|(_, footer)| {
178            footer.iter().any(|entry| {
179                obsolete_entry_ids
180                    .get(&entry.region_id)
181                    .is_none_or(|obsolete| entry.max_entry_id > *obsolete)
182            })
183        });
184        match live {
185            Some((&object_seq, _)) => object_seq,
186            None => self
187                .objects
188                .last_key_value()
189                .map_or(from, |(&object_seq, _)| {
190                    object_seq.saturating_add(1).max(from)
191                }),
192        }
193    }
194
195    /// Returns the objects that hold entries of `region_id` overlapping
196    /// `start_entry_id..=end_entry_id`, ordered by object sequence.
197    pub(crate) fn objects_for_entry_range(
198        &self,
199        region_id: RegionId,
200        start_entry_id: u64,
201        end_entry_id: u64,
202    ) -> Result<Vec<(u64, &FooterEntry)>> {
203        ensure!(
204            start_entry_id <= end_entry_id,
205            InvalidWalEntryRangeSnafu {
206                region_id,
207                start_entry_id,
208                end_entry_id,
209            }
210        );
211
212        let Some(objects) = self.regions.get(&region_id) else {
213            return Ok(Vec::new());
214        };
215        Ok(objects
216            .iter()
217            .filter(|(_, entry)| {
218                entry.max_entry_id >= start_entry_id && entry.min_entry_id <= end_entry_id
219            })
220            .map(|(&object_seq, entry)| (object_seq, entry))
221            .collect())
222    }
223
224    /// Returns the largest entry id ever indexed for `region_id`, which is
225    /// kept after the object holding it was removed.
226    pub(crate) fn region_max_entry_id(&self, region_id: RegionId) -> Option<u64> {
227        self.max_entry_ids.get(&region_id).copied()
228    }
229
230    /// Returns the sequence to assign to the next object written after recovery.
231    ///
232    /// An empty catalog starts at zero, so the first object of a prefix always
233    /// takes sequence zero. Otherwise the sequence continues after the largest
234    /// indexed one, which recovery discovers regardless of insertion order, and
235    /// is raised further when the largest entry id of a region lies at or above
236    /// the ids that sequence would assign: ids assigned under the earlier
237    /// contiguous scheme carry no object information, and every new id of a
238    /// region must be greater than every id it already has. A sequence at or
239    /// above [`OBJECT_SEQ_LIMIT`] does not fit an entry id and is rejected.
240    pub(crate) fn next_object_seq(&self) -> Result<u64> {
241        let after_last = match self.objects.last_key_value() {
242            None => 0,
243            Some((&last_object_seq, _)) => last_object_seq
244                .checked_add(1)
245                .context(WalObjectSequenceExhaustedSnafu { last_object_seq })?,
246        };
247        let floor = self
248            .max_entry_ids
249            .values()
250            .copied()
251            .map(sequence_floor)
252            .max()
253            .unwrap_or(0);
254        let next_object_seq = after_last.max(floor);
255        ensure!(
256            next_object_seq < OBJECT_SEQ_LIMIT,
257            WalObjectSequenceExhaustedSnafu {
258                last_object_seq: next_object_seq - 1,
259            }
260        );
261        Ok(next_object_seq)
262    }
263
264    /// Iterates over the indexed objects ordered by object sequence.
265    pub(crate) fn objects_in_order(&self) -> impl Iterator<Item = (u64, &[FooterEntry])> + '_ {
266        self.objects
267            .iter()
268            .map(|(&object_seq, footer)| (object_seq, footer.as_slice()))
269    }
270}
271
272fn out_of_order_reason(
273    region_id: RegionId,
274    lower_object_seq: u64,
275    lower_max_entry_id: u64,
276    upper_object_seq: u64,
277    upper_min_entry_id: u64,
278) -> String {
279    format!(
280        "entry ranges of region {region_id} are not strictly increasing, object {lower_object_seq} ends at {lower_max_entry_id}, object {upper_object_seq} starts at {upper_min_entry_id}"
281    )
282}

290    #[test]
291    fn test_catalog_indexes_objects_and_queries_ranges() {
292        let region_one = RegionId::new(1, 1);
293        let region_two = RegionId::new(2, 1);
294        let mut catalog = ObjectCatalog::default();
295
296        // Recovery may discover objects out of order.
297        catalog
298            .insert_object(
299                2,
300                vec![
301                    footer_entry(region_one, 4, 6),
302                    footer_entry(region_two, 8, 9),
303                ],
304            )
305            .unwrap();
306        catalog
307            .insert_object(1, vec![footer_entry(region_one, 1, 3)])
308            .unwrap();
309        catalog
310            .insert_object(4, vec![footer_entry(region_one, 10, 12)])
311            .unwrap();
312
313        assert_eq!(Some(12), catalog.region_max_entry_id(region_one));
314        assert_eq!(Some(9), catalog.region_max_entry_id(region_two));
315        assert_eq!(None, catalog.region_max_entry_id(RegionId::new(3, 1)));
316
317        let objects = catalog.objects_for_entry_range(region_one, 3, 10).unwrap();
318        assert_eq!(vec![1, 2, 4], object_seqs(&objects));
319        assert_eq!(
320            vec![1, 2, 4],
321            catalog
322                .objects_in_order()
323                .map(|(object_seq, _)| object_seq)
324                .collect::<Vec<_>>()
325        );
326    }

329    fn test_catalog_rejects_duplicate_object_sequences() {
330        let region_id = RegionId::new(1, 1);
331        let mut catalog = ObjectCatalog::default();
332        let first = footer_entry(region_id, 1, 2);
333        let second = footer_entry(RegionId::new(2, 1), 4, 5);

335        catalog
336            .insert_object(1, vec![first.clone(), second.clone()])
337            .unwrap();
338
339        // An identical footer is rejected just like a conflicting one.
340        assert_corrupted(
341            catalog.insert_object(1, vec![second, first.clone()]),
342            "object 1 is already indexed",

344
345        let mut conflicting = first;
346        conflicting.segment_offset += 1;
347        assert_corrupted(
348            catalog.insert_object(1, vec![conflicting]),
349            "object 1 is already indexed",
350        );
351        assert_eq!(1, catalog.objects_in_order().count());
352    }

378        let region_one = RegionId::new(1, 1);
379        let region_two = RegionId::new(1, 2);
380        let mut catalog = ObjectCatalog::default();
381
382        // Ids that fit below the ids of the next sequence leave it alone.
383        catalog

402                3,
403                vec![footer_entry(region_two, entry_id(7, 4), entry_id(7, 4))],

406        assert_eq!(8, catalog.next_object_seq().unwrap());

496    #[test]
497    fn test_catalog_reclaim_boundary_follows_the_watermarks() {
498        let region_one = RegionId::new(1, 1);
499        let region_two = RegionId::new(2, 1);
500        let mut catalog = ObjectCatalog::default();
501        catalog.insert_object(0, Vec::new()).unwrap();
502        catalog
503            .insert_object(
504                1,
505                vec![
506                    footer_entry(region_one, entry_id(1, 1), entry_id(1, 2)),
507                    footer_entry(region_two, entry_id(1, 1), entry_id(1, 1)),
508                ],
509            )
510            .unwrap();
511        catalog
512            .insert_object(
513                2,
514                vec![footer_entry(region_one, entry_id(2, 1), entry_id(2, 1))],
515            )
516            .unwrap();
517        catalog
518            .insert_object(
519                3,
520                vec![footer_entry(region_two, entry_id(3, 1), entry_id(3, 1))],
521            )
522            .unwrap();
523        let boundary = |catalog: &ObjectCatalog, obsolete: &[(RegionId, EntryId)]| {
524            catalog.reclaim_boundary(0, &obsolete.iter().copied().collect())
525        };
526
527        // A region without a watermark holds the boundary at its first object.
528        assert_eq!(1, boundary(&catalog, &[]));
529        assert_eq!(1, boundary(&catalog, &[(region_one, entry_id(2, 1))]));
530        // A watermark inside an object keeps that object.
531        assert_eq!(
532            1,
533            boundary(
534                &catalog,
535                &[(region_one, entry_id(1, 1)), (region_two, EntryId::MAX)]
536            )
537        );
538        assert_eq!(
539            2,
540            boundary(
541                &catalog,
542                &[(region_one, entry_id(1, 2)), (region_two, entry_id(1, 1))]
543            )
544        );
545        assert_eq!(
546            3,
547            boundary(
548                &catalog,
549                &[(region_one, entry_id(2, 1)), (region_two, entry_id(1, 1))]
550            )
551        );
552        // A region of `obsolete_all` contributes nothing, and when no region
553        // does, the boundary is past the last object.
554        assert_eq!(
555            4,
556            boundary(
557                &catalog,
558                &[(region_one, entry_id(2, 1)), (region_two, EntryId::MAX)]
559            )
560        );
561
562        // Objects below the previous boundary are not scanned again.
563        assert_eq!(3, catalog.reclaim_boundary(3, &HashMap::new()));
564        assert_eq!(5, catalog.reclaim_boundary(5, &HashMap::new()));
565
566        // Removing objects keeps the largest entry id of every region, and a
567        // region without indexed objects contributes nothing.
568        for object_seq in [0, 1, 2] {
569            assert!(catalog.remove_object(object_seq).is_some());
570        }
571        assert!(catalog.remove_object(2).is_none());
572        assert!(!catalog.contains_object(2));
573        assert!(catalog.contains_object(3));
574        assert_eq!(
575            Some(entry_id(2, 1)),
576            catalog.region_max_entry_id(region_one)
577        );
578        assert!(
579            catalog
580                .objects_for_entry_range(region_one, 0, EntryId::MAX)
581                .unwrap()
582                .is_empty()
583        );
584        assert_eq!(3, boundary(&catalog, &[]));
585        assert_eq!(4, catalog.next_object_seq().unwrap());
586    }

588    fn footer_entry(region_id: RegionId, min_entry_id: u64, max_entry_id: u64) -> FooterEntry {
589        FooterEntry {
590            region_id,
591            min_entry_id,
592            max_entry_id,
593            entry_count: max_entry_id
594                .checked_sub(min_entry_id)
595                .and_then(|count| count.checked_add(1))
596                .unwrap_or(0) as u32,
597            segment_offset: min_entry_id.wrapping_mul(100),
598            segment_len: 100,
599            segment_crc32: min_entry_id as u32,
600        }
601    }
602
603    fn object_seqs(objects: &[(u64, &FooterEntry)]) -> Vec<u64> {
604        objects.iter().map(|(object_seq, _)| *object_seq).collect()
605    }
606
607    fn assert_corrupted(result: Result<()>, expected_reason: &str) {
608        match result {
609            Err(Error::CorruptedWalObject { reason, .. }) => assert!(
610                reason.contains(expected_reason),
611                "expected reason to contain {expected_reason:?}, actual {reason:?}"
612            ),
613            other => panic!("expected a corrupted object error, actual {other:?}"),
614        }
615    }
