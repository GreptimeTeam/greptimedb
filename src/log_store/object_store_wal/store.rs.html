47const COMMAND_BUFFER: usize = 1024;
48const MIN_FLUSH_INTERVAL: Duration = Duration::from_millis(10);
49/// Number of objects whose footers recovery fetches at a time.
50const RECOVERY_CONCURRENCY: usize = 8;
51/// Bytes recovery reads from the end of an object in one request. The window
52/// holds the trailer and the footer of an object with up to 1364 regions of
53/// 48 bytes each, so a second request for the footer is rare.
54const RECOVERY_TAIL_WINDOW: usize = 64 * 1024;

76impl ObjectStoreLogStore {
77    /// Builds the store under the node and generation prefix derived from `config`,
78    /// recovering the catalog from the objects that already exist. Recovery
79    /// fails on the first corrupted or conflicting object.
80    pub(crate) async fn try_new(
81        object_store: ObjectStore,
82        config: &ObjectStoreWalConfig,
83        node_id: u64,
84        generation: u64,
85    ) -> Result<Arc<Self>> {
86        let prefix = config.node_prefix(node_id, generation);
87        let io = ObjectStoreIo::new(object_store, &prefix)?;
88        Self::open(Arc::new(io), config, prefix).await
89    }
90
91    async fn open(
92        io: Arc<dyn WalObjectIo>,
93        config: &ObjectStoreWalConfig,
94        prefix: String,
95    ) -> Result<Arc<Self>> {
96        ensure!(
97            config.flush_interval >= MIN_FLUSH_INTERVAL,
98            InvalidWalObjectStoreSnafu {
99                reason: format!(
100                    "flush interval {:?} is shorter than {MIN_FLUSH_INTERVAL:?}",
101                    config.flush_interval
102                ),
103            }
104        );
105
106        positive_bytes(config.max_batch_bytes.as_bytes(), "max batch bytes")?;
107        let (catalog, next_object_seq, durable_entry_ids) = recover(io.as_ref()).await?;
108        let catalog = Arc::new(RwLock::new(catalog));
109        let terminal_error = TerminalError::default();
110        let stopped = Arc::new(AtomicBool::new(false));
111        let (command_tx, command_rx) = mpsc::channel(COMMAND_BUFFER);
112        let actor = Actor {
113            terminal_error: terminal_error.clone(),
114            stopped: stopped.clone(),
115            command_rx,
116            issued_entry_ids: durable_entry_ids,
117            next_object_seq: Some(next_object_seq),
118            stop: Vec::new(),
119        };
120        common_runtime::spawn_global(actor.run());
121        Ok(Arc::new(Self {
122            prefix,
123            io,
124            catalog,
125            terminal_error,
126            stopped,
127            command_tx,
128        }))
129    }
130
131    fn durable_entry_id_of(&self, region_id: RegionId) -> EntryId {
132        let catalog = self.catalog.read().unwrap_or_else(PoisonError::into_inner);
133        catalog.region_max_entry_id(region_id).unwrap_or(0)
134    }
135
136    /// Returns the region of `provider`, which must select this store's prefix.
137    fn region_of(&self, provider: &Provider) -> Result<RegionId> {
138        let provider =
139            provider
140                .as_object_store_provider()
141                .with_context(|| InvalidProviderSnafu {
142                    expected: ObjectStoreProvider::type_name(),
143                    actual: provider.type_name(),
144                })?;
145        ensure!(
146            provider.prefix == self.prefix,
147            MismatchedWalPrefixSnafu {
148                expected: self.prefix.clone(),
149                actual: provider.prefix.clone(),
150            }
151        );
152        Ok(provider.region_id)
153    }
154
155    fn check_terminal(&self) -> Result<()> {
156        match terminal(&self.terminal_error) {
157            Some(error) => Err(shared(&error)),
158            None => Ok(()),
159        }
160    }
161}

163fn positive_bytes(bytes: u64, name: &str) -> Result<usize> {
164    usize::try_from(bytes)
165        .ok()
166        .filter(|bytes| *bytes > 0)
167        .with_context(|| InvalidWalObjectStoreSnafu {
168            reason: format!("{name} {bytes} is zero or too large"),
169        })

247enum Command {
248    Stop {
249        response: oneshot::Sender<Result<()>>,
250    },
251}
252
253struct Actor {
254    terminal_error: TerminalError,
255    stopped: Arc<AtomicBool>,
256    command_rx: mpsc::Receiver<Command>,
257    issued_entry_ids: HashMap<RegionId, EntryId>,
258    next_object_seq: Option<u64>,
259    stop: Vec<oneshot::Sender<Result<()>>>,
260}
261
262impl Actor {
263    async fn run(mut self) {
264        while let Some(Command::Stop { response }) = self.command_rx.recv().await {
265            self.handle_stop(response);
266            if self.finish_stop() {
267                return;
268            }
269        }
270    }
271
272    fn handle_stop(&mut self, response: oneshot::Sender<Result<()>>) {
273        self.stop.push(response);
274    }
275
276    fn finish_stop(&mut self) -> bool {
277        if !self.is_stopped() || self.stop.is_empty() {
278            return false;
279        }
280        for response in self.stop.drain(..) {
281            let _ = response.send(Ok(()));
282        }
283        true
284    }
285
286    fn is_stopped(&self) -> bool {
287        self.stopped.load(Ordering::Acquire)
288    }
289}

313/// Rebuilds the catalog from object footers,
314/// so recovery costs a few small reads per object however large the objects
315/// are. Segments are not read; a segment checksum is verified by the read that
316/// decodes it. Footers are fetched for up to [`RECOVERY_CONCURRENCY`] objects
317/// at a time and indexed in sequence order, so the catalog checks the entry
318/// ranges of every object against its predecessors like a sequential replay.
319async fn recover(io: &dyn WalObjectIo) -> Result<(ObjectCatalog, u64, HashMap<RegionId, EntryId>)> {
320    let objects = io.list().await?;
321    let mut catalog = ObjectCatalog::default();
322    for (object, footer) in fetch_footers(io, objects, RECOVERY_CONCURRENCY).await? {
323        catalog
324            .insert_object(object.object_seq, footer)
325            .with_context(|_| InvalidWalObjectSnafu { path: object.path })?;
326    }
327    finish_recovery(catalog)
328}
329
330fn finish_recovery(
331    catalog: ObjectCatalog,
332) -> Result<(ObjectCatalog, u64, HashMap<RegionId, EntryId>)> {
333    let next_object_seq = catalog.next_object_seq()?;
334    let durable_entry_ids = durable_entry_ids(&catalog);
335    Ok((catalog, next_object_seq, durable_entry_ids))
336}

338/// Fetches and verifies the footers of `objects`, up to `concurrency` objects
339/// at a time, and returns them ordered by object sequence whatever the order
340/// the fetches complete in. The first failure abandons the remaining fetches.
341async fn fetch_footers(
342    io: &dyn WalObjectIo,
343    objects: Vec<ListedObject>,
344    concurrency: usize,
345) -> Result<Vec<(ListedObject, Vec<FooterEntry>)>> {
346    let mut footers = futures::stream::iter(objects)
347        .map(|object| async move {
348            let footer = fetch_footer(io, &object).await?;
349            Ok((object, footer))
350        })
351        .buffer_unordered(concurrency)
352        .try_collect::<Vec<_>>()
353        .await?;
354    footers.sort_unstable_by_key(|(object, _)| object.object_seq);
355    Ok(footers)
356}

358/// Reads the header, trailer and footer of `object` and verifies them: the
359/// header must carry the sequence of the key, the trailer must be well formed,
360/// the footer must match the checksum the trailer holds and its segments must
361/// tile the object body.
362///
363/// A short object is read whole. Otherwise the header and a window at the end
364/// of the object are read concurrently, and the footer is read separately only
365/// when it starts before the window.
366async fn fetch_footer(io: &dyn WalObjectIo, object: &ListedObject) -> Result<Vec<FooterEntry>> {
367    let ListedObject {
368        object_seq, size, ..
369    } = *object;
370    let invalid = |source: Error| {
371        InvalidWalObjectSnafu {
372            path: object.path.clone(),
373        }
374        .into_error(source)
375    };
376    let object_len = usize::try_from(size)
377        .ok()
378        .filter(|len| *len >= MIN_OBJECT_LEN)
379        .with_context(|| CorruptedWalObjectSnafu {
380            reason: format!(
381                "truncated object, expected at least {MIN_OBJECT_LEN} bytes, actual {size}"
382            ),
383        })
384        .map_err(invalid)?;
385
386    let window = object_len.min(RECOVERY_TAIL_WINDOW);
387    let tail_start = object_len - window;
388    let (head, tail) = if tail_start == 0 {
389        let bytes = io.get(object_seq).await?;
390        (bytes.clone(), bytes)
391    } else {
392        futures::try_join!(
393            io.get_range(object_seq, 0, HEADER_LEN as u64),
394            io.get_range(object_seq, tail_start as u64, window as u64)
395        )?
396    };
397    if head.len() < HEADER_LEN || tail.len() != window {
398        return Err(invalid(
399            CorruptedWalObjectSnafu {
400                reason: format!(
401                    "object holds fewer bytes than the listed {size}, head {} bytes, tail {} bytes",
402                    head.len(),
403                    tail.len()
404                ),
405            }
406            .build(),
407        ));
408    }
409
410    let (trailer, footer_range) =
411        locate_footer(object_seq, object_len, &head, &tail).map_err(invalid)?;
412    let footer = if footer_range.start >= tail_start {
413        tail.slice(footer_range.start - tail_start..footer_range.end - tail_start)
414    } else {
415        io.get_range(
416            object_seq,
417            footer_range.start as u64,
418            footer_range.len() as u64,
419        )
420        .await?
421    };
422    let footer = decode_footer(&footer, trailer).map_err(invalid)?;
423    verify_segment_ranges(&footer, footer_range.start).map_err(invalid)?;
424    Ok(footer)
425}

427/// Verifies the header and trailer of the object `object_seq` of `object_len`
428/// bytes from its first bytes `head` and its last bytes `tail`, and returns
429/// the trailer with the range the footer occupies in the object.
430fn locate_footer(
431    object_seq: u64,
432    object_len: usize,
433    head: &[u8],
434    tail: &[u8],
435) -> Result<(FixedTrailer, Range<usize>)> {
436    let header = decode_header(head)?;
437    ensure!(
438        header.object_seq == object_seq,
439        CorruptedWalObjectSnafu {
440            reason: format!(
441                "header sequence {} does not match key sequence {object_seq}",
442                header.object_seq
443            ),
444        }
445    );
446    let trailer = decode_trailer(&tail[tail.len() - TRAILER_LEN..])?;
447    let footer_range = footer_range(trailer, object_len)?;
448    Ok((trailer, footer_range))
449}

451fn durable_entry_ids(catalog: &ObjectCatalog) -> HashMap<RegionId, EntryId> {
452    let mut entry_ids = HashMap::new();
453    for (_, footer) in catalog.objects_in_order() {
454        for entry in footer {
455            entry_ids
456                .entry(entry.region_id)
457                .and_modify(|current: &mut EntryId| *current = (*current).max(entry.max_entry_id))
458                .or_insert(entry.max_entry_id);
459        }
460    }
461    entry_ids
462}

644    /// Rebuilds the catalog by decoding whole objects as a recovery oracle.
645    async fn recover_by_decoding(
646        io: &dyn WalObjectIo,
647    ) -> Result<(ObjectCatalog, u64, HashMap<RegionId, EntryId>)> {
648        let mut catalog = ObjectCatalog::default();
649        for ListedObject {
650            object_seq, path, ..
651        } in io.list().await?
652        {
653            let bytes = io.get(object_seq).await?;
654            decode_object(&bytes)
655                .and_then(|decoded| {
656                    ensure!(
657                        decoded.header.object_seq == object_seq,
658                        CorruptedWalObjectSnafu {
659                            reason: format!(
660                                "header sequence {} does not match key sequence {object_seq}",
661                                decoded.header.object_seq
662                            ),
663                        }
664                    );

726        let trailer = decode_trailer(&bytes[bytes.len() - TRAILER_LEN..]).unwrap();

853    /// Overwrites the byte range of footer entry `index` and refreshes the
854    /// footer checksum, so the footer is intact but describes the wrong bytes.
855    fn rewrite_segment_range(bytes: &mut [u8], index: usize, offset: u64, len: u64) {
856        let trailer_start = bytes.len() - TRAILER_LEN;
857        let (trailer, _) = footer_of(bytes);
858        let footer = footer_range(trailer, bytes.len()).unwrap();
859        let entry = footer.start + 4 + index * FOOTER_ENTRY_LEN;
860        bytes[entry + 28..entry + 36].copy_from_slice(&offset.to_be_bytes());
861        bytes[entry + 36..entry + 44].copy_from_slice(&len.to_be_bytes());
862        let checksum = crc32fast::hash(&bytes[footer]);
863        bytes[trailer_start + 16..trailer_start + 20].copy_from_slice(&checksum.to_be_bytes());
864    }

1324    #[async_trait::async_trait]
1325    impl WalObjectIo for ParkedIo {
1326        async fn put_if_absent(&self, object_seq: u64, content: Bytes) -> Result<PutResult> {
1327            self.inner.put_if_absent(object_seq, content).await
1328        }
1329
1330        async fn get(&self, object_seq: u64) -> Result<Bytes> {
1331            let in_flight = self.in_flight.fetch_add(1, Ordering::SeqCst) + 1;
1332            self.max_in_flight.fetch_max(in_flight, Ordering::SeqCst);
1333            let (release, released) = oneshot::channel();
1334            self.parked.send((object_seq, release)).unwrap();
1335            let success = released.await.unwrap();
1336            self.in_flight.fetch_sub(1, Ordering::SeqCst);
1337            if !success {
1338                return Err(object_store::Error::new(
1339                    object_store::ErrorKind::Unexpected,
1340                    "injected failure",
1341                )
1342                .set_temporary())
1343                .context(WalObjectStoreSnafu {
1344                    operation: "read",
1345                    path: self.object_path(object_seq),
1346                });
1347            }
1348            self.inner.get(object_seq).await
1349        }
1350
1351        async fn get_range(&self, object_seq: u64, offset: u64, len: u64) -> Result<Bytes> {
1352            self.inner.get_range(object_seq, offset, len).await
1353        }
1354
1355        async fn list(&self) -> Result<Vec<ListedObject>> {
1356            self.inner.list().await
1357        }
1358
1359        fn object_path(&self, object_seq: u64) -> String {
1360            self.inner.object_path(object_seq)
1361        }
1362    }
