115    /// How long an object of the object store logstore takes to become durable
116    /// once its batch is sealed, sampled for every conditional create that
117    /// succeeds, an identical retry included. The time runs from the seal, so
118    /// it includes any wait for a create slot and, in the `enqueued` mode,
119    /// every repeated attempt and the delay before it.
120    ///
121    /// A create is a few milliseconds on a local object store and tens of
122    /// milliseconds on a cloud one, so the buckets start at 1ms and reach 8s.
123    pub static ref METRIC_OBJECT_STORE_WAL_SEAL_TO_DURABLE_SECONDS: Histogram = register_histogram!(
124        "greptime_logstore_object_store_wal_seal_to_durable_seconds",
125        "object store logstore seconds from sealing a batch to the success of its conditional create, slot waits and retries included",
126        exponential_buckets(0.001, 2.0, 14).unwrap(),
127    )
128    .unwrap();
129
130    /// How long an append of the object store logstore waits for its
131    /// acknowledgement in the `durable` mode, which is the write latency a
132    /// caller sees.
133    ///
134    /// The wait is one seal plus every earlier object plus one create, so with
135    /// the default 100ms flush interval it sits in the tens of milliseconds;
136    /// the buckets straddle that and leave room for a stalled uploader.
137    pub static ref METRIC_OBJECT_STORE_WAL_APPEND_ACK_SECONDS: Histogram = register_histogram!(
138        "greptime_logstore_object_store_wal_append_ack_seconds",
139        "object store logstore seconds an append waits for its acknowledgement in the durable mode",
140        exponential_buckets(0.002, 2.0, 14).unwrap(),
141    )
142    .unwrap();
143
144    /// How large the objects of the object store logstore are, which says
145    /// whether batches seal on size or on the flush interval.
146    ///
147    /// The buckets are powers of two from 1KiB, the size of a timer-sealed
148    /// batch of one small entry, to 16MiB, twice the 8MiB default batch size,
149    /// which is a threshold rather than an upper bound.
150    pub static ref METRIC_OBJECT_STORE_WAL_OBJECT_BYTES: Histogram = register_histogram!(
151        "greptime_logstore_object_store_wal_object_bytes",
152        "object store logstore bytes of each object sealed",
153        exponential_buckets(1024.0, 2.0, 15).unwrap(),
154    )
155    .unwrap();
156
157    /// How many entries the objects of the object store logstore hold, which
158    /// says how many writes one request amortises.
159    ///
160    /// The largest finite bucket is 2^20, the position limit of one region in
161    /// one object. An object holds the entries of many regions, so its total
162    /// can pass that; such an object falls in the overflow bucket.
163    pub static ref METRIC_OBJECT_STORE_WAL_OBJECT_ENTRIES: Histogram = register_histogram!(
164        "greptime_logstore_object_store_wal_object_entries",
165        "object store logstore entries of each object sealed",
166        exponential_buckets(1.0, 4.0, 11).unwrap(),
167    )
168    .unwrap();

268    /// How long such an append stays held back, until it is admitted or until
269    /// it is refused, a stop, a terminal error or the teardown of the store
270    /// releasing it included, which is the latency the `enqueued` mode pays

276        "greptime_logstore_object_store_wal_stalled_append_seconds",
277        "object store logstore seconds a stalled append is held back before it is admitted or refused",
278        exponential_buckets(0.005, 3.0, 9).unwrap(),
279    )
