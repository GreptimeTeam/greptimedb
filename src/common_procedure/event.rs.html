35/// `ProcedureEvent` represents an event emitted by a procedure during its execution lifecycle.
36#[derive(Debug)]
37pub struct ProcedureEvent {
38    /// Unique identifier associated with the originating procedure instance.
39    pub procedure_id: ProcedureId,
40    /// The timestamp of the event.
41    pub timestamp: Timestamp,
42    /// The state of the procedure.
43    pub state: ProcedureState,
44    /// The lifecycle trigger that caused the event to be emitted.
45    pub trigger: EventTrigger,
46    /// The event emitted by the procedure. It's generated by [Procedure::event].
47    pub internal_event: Box<dyn Event>,
48}
49
50impl ProcedureEvent {
51    pub fn new(
52        procedure_id: ProcedureId,
53        internal_event: Box<dyn Event>,
54        state: ProcedureState,
55        trigger: EventTrigger,
56    ) -> Self {
57        Self {
58            procedure_id,
59            internal_event,
60            timestamp: Timestamp::current_time(TimeUnit::Nanosecond),
61            state,
62            trigger,
63        }
64    }
65}
66
67impl Event for ProcedureEvent {
68    fn event_type(&self) -> &str {
69        self.internal_event.event_type()
70    }
71
72    fn timestamp(&self) -> Timestamp {
73        self.timestamp
74    }
75
76    fn json_payload(&self) -> Result<serde_json::Value> {
77        self.internal_event.json_payload()
78    }
79
80    fn extra_schema(&self) -> Vec<ColumnSchema> {
81        let mut schema = procedure_event_column_schemas();
82        schema.append(&mut self.internal_event.extra_schema());
83        schema
84    }
85
86    fn extra_rows(&self) -> Result<Vec<Row>> {
87        let mut internal_event_extra_rows = self.internal_event.extra_rows()?;
88        let mut rows = Vec::with_capacity(internal_event_extra_rows.len());
89        let procedure_id = self.procedure_id.to_string();
90        let state = self.state.as_str_name().to_string();
91        let error = match &self.state {
92            ProcedureState::Failed { error }
93            | ProcedureState::PrepareRollback { error }
94            | ProcedureState::RollingBack { error }
95            | ProcedureState::Retrying { error }
96            | ProcedureState::Poisoned { error, .. } => format!("{error:?}"),
97            _ => String::new(),
98        };
99        let trigger = serde_json::to_value(&self.trigger).context(SerializeEventSnafu)?;
100
101        for internal_event_extra_row in internal_event_extra_rows.iter_mut() {
102            let mut values = Vec::with_capacity(4 + internal_event_extra_row.values.len());
103            values.extend([
104                ValueData::StringValue(procedure_id.clone()).into(),
105                ValueData::StringValue(state.clone()).into(),
106                ValueData::StringValue(error.clone()).into(),
107                jsonb_value(&trigger),
108            ]);
109            values.append(&mut internal_event_extra_row.values);
110            rows.push(Row { values });
111        }
112
113        Ok(rows)
114    }
115
116    fn as_any(&self) -> &dyn Any {
117        self
118    }
119}

287    #[test]
288    fn test_event_trigger_serialization() {
289        let procedure_id = ProcedureId::parse_str("00000000-0000-0000-0000-000000000001").unwrap();
290
291        for (trigger, name) in [
292            (EventTrigger::Submitted, "Submitted"),
293            (EventTrigger::Recovered, "Recovered"),
294            (EventTrigger::RollingBack, "RollingBack"),
295            (EventTrigger::Succeeded, "Succeeded"),
296            (EventTrigger::Failed, "Failed"),
297            (EventTrigger::Poisoned, "Poisoned"),
298        ] {
299            assert_eq!(
300                serde_json::to_value(trigger).unwrap(),
301                json!({"type": name})
302            );
303        }
304        assert_eq!(
305            serde_json::to_value(EventTrigger::ChildSubmitted {
306                procedure_id,
307                outcome: ChildSubmissionOutcome::Accepted,
308            })
309            .unwrap(),
310            json!({
311                "type": "ChildSubmitted",
312                "procedure_id": "00000000-0000-0000-0000-000000000001",
313                "outcome": "Accepted",
314            })
315        );
316        assert_eq!(
317            serde_json::to_value(EventTrigger::Retrying {
318                phase: RetryPhase::Execute,
319                attempt: 2,
320            })
321            .unwrap(),
322            json!({"type": "Retrying", "phase": "Execute", "attempt": 2})
323        );
324    }
