15//! Segmenting a native tree, so each per-series prefix runs where its input is read.
16//!
17//! A tree that reads across series answers one partition and gathers whatever it reads. Its
18//! per-series prefixes do not need that gather, so each is cut out as a [`FusedSeries`] node of its
19//! own and keeps the partitions it was scanned with: `cut_as_segment` decides what can be cut,
20//! [`cut_segments`] rewrites the tree into one segment per cut, and [`segment_tree`] builds the
21//! nodes, recursing into a cut that wraps a whole-vector consumer.
22//!
23//! Everything here reads the bound tree alone. The columns a node publishes are the planner's
24//! answer, so the builders ask for them through [`OutputTags`], [`FieldNames`] and [`RootSchema`]
25//! instead of deriving them a second time.

51/// Derives the output tag columns a node publishes, from the tag columns each source declares.
52///
53/// The planner that owns the result layout names those columns with its own walk over the tree. A
54/// segment only reads that answer - it never renames a column - so it asks through this leaf
55/// rather than deriving the columns a second time.
56pub type OutputTags = fn(&NativeNode, &[Vec<String>]) -> Option<Vec<String>>;

58/// Derives the value column names a statement's expression gives the compiled `fields`.
59///
60/// Like [`OutputTags`], this is the planner's answer: a segment publishes the values its tree
61/// computes under the names the statement gave them, so the builders ask through this leaf instead
62/// of publishing the compiler's own field names. The tree alone does not carry the statement's
63/// expression, which is why the expression text is an input rather than something derived here.
64pub type FieldNames = fn(&str, &[String]) -> Vec<String>;

66/// The row schema a statement root publishes for `promql`, or `None` when its columns cannot be
67/// named once.
68///
69/// Like [`OutputTags`], this is the caller's answer: a segment that publishes rows publishes
70/// exactly the columns the planner would have published for the same statement.
71pub type RootSchema = fn(&str, &NativeTree, &[Vec<String>], &str, FieldKind) -> Option<DFSchemaRef>;

73/// Whether `node` reads samples point-wise, so that one partitioned execution is one answer.
74///
75/// A node's output type is not checked here: [`NativeNode::validate`] already refuses a selection,
76/// a map or a binary operation whose operands are not evaluated vectors, so every branch below is
77/// one that produces values rather than raw samples.
78pub fn pointwise_operand(node: &NativeNode) -> bool {
79    match &node.expr {
80        // A selection reads the raw samples below it; anything else below it is a value computed
81        // from a whole series, not a scan.
82        NativeExpr::Instant { input, .. } | NativeExpr::Window { input, .. } => {
83            matches!(input.expr, NativeExpr::Input(_))
84        }
85        NativeExpr::Map { input, .. } | NativeExpr::Timestamp { input } => pointwise_operand(input),
86        // A parameter that reads no data has the same value in every execution.
87        NativeExpr::ParameterizedMap {
88            input, parameters, ..
89        } => pointwise_operand(input) && parameters.iter().all(NativeNode::is_data_free),
90        // A scalar that reads no data is broadcast over the vector side's positions. Two vectors
91        // are a match between two series sets, which is the shape this route cannot split.
92        NativeExpr::Binary {
93            left,
94            right,
95            matching,
96        } => {
97            let scalar = |side: &NativeNode| matches!(side.output, NativeType::Scalar(_));
98            matching.cardinality == Cardinality::OneToOne
99                && match (scalar(left), scalar(right)) {
100                    (true, false) => left.is_data_free() && pointwise_operand(right),
101                    (false, true) => right.is_data_free() && pointwise_operand(left),
102                    _ => false,
103                }
104        }
105        // An aggregate folds a group, a vector operation reads the whole vector, and a bare input or
106        // scalar is not a per-series evaluation. A parameterized topk or histogram reads the whole
107        // vector, and `scalar()` answers from it.
108        NativeExpr::Input(_)
109        | NativeExpr::Scalar(_)
110        | NativeExpr::EvalTime { .. }
111        | NativeExpr::VectorOf { .. }
112        | NativeExpr::ScalarOf { .. }
113        | NativeExpr::ParameterizedTopK { .. }
114        | NativeExpr::ParameterizedHistogram { .. }
115        | NativeExpr::Aggregate { .. }
116        | NativeExpr::Vector { .. }
117        | NativeExpr::Set { .. }
118        | NativeExpr::Merge { .. }
119        | NativeExpr::Info { .. } => false,
120    }
121}

123pub fn independent_scan_pair(root: &NativeNode) -> bool {
124    let (left, right) = match &root.expr {
125        NativeExpr::Binary { left, right, .. } => match (&left.expr, &right.expr) {
126            (NativeExpr::Scalar(_), NativeExpr::Scalar(_)) => return false,
127            (NativeExpr::Scalar(_), _) => return independent_scan_pair(right),
128            (_, NativeExpr::Scalar(_)) => return independent_scan_pair(left),
129            _ => (left, right),
130        },
131        NativeExpr::Map { input, .. } => return independent_scan_pair(input),
132        NativeExpr::Set { left, right, .. } => {
133            return scan_operand(left) && scan_operand(right);
134        }
135        NativeExpr::Info { base, info, .. } => {
136            return pointwise_operand(base) && pointwise_operand(info);
137        }
138        _ => return false,
139    };
140
141    scan_operand(left) && scan_operand(right)
142}
143
144fn scan_operand(node: &NativeNode) -> bool {
145    pointwise_operand(node) || independent_scan_pair(node)
146}

148/// Whether a grouping in `node` keeps a label the existing planner's column list cannot name.
149///
150/// An aggregate's `by(...)` keeps the labels it lists that its input carries, and `__name__` is one
151/// of them: the aggregate folds its groups by the metric name as by any other label. It is not a
152/// column of any scan, though - it is the name the raw frame carries - so the existing planner's
153/// `by(__name__)` finds no such column and reports no label at all. A schema taken from that list
154/// would drop the metric name the native aggregate kept, and two groups differing only in their
155/// metric would come out as two rows under one label set, which is a series identity lost rather
156/// than an answer given.
157///
158/// `without(...)` needs no such check: both routes drop `__name__` there. Nor does a tree with one
159/// source: a discovered selector's source declares `__name__` per metric, and a named one folds one
160/// metric whose name cannot merge two groups. What the check is for is the tree reading more than one
161/// source, where only a discovered source can report a name at all.
162pub fn groups_by_metric_name(node: &NativeNode) -> bool {
163    let grouping_names_metric = |op: &BoundAggregate| matches!(&op.group, Grouping::By(names) if names.iter().any(|name| name.as_ref() == METRIC_NAME));
164    match &node.expr {
165        NativeExpr::Input(_)
166        | NativeExpr::Scalar(_)
167        | NativeExpr::EvalTime { .. }
168        | NativeExpr::VectorOf { .. } => false,
169        // `scalar()` answers one value with no labels at all, so a grouping by the metric name
170        // below it cannot leave two groups under one label set: the walk stops there.
171        NativeExpr::ScalarOf { .. } => false,
172        NativeExpr::Instant { input, .. }
173        | NativeExpr::Merge { input }
174        | NativeExpr::Window { input, .. }
175        | NativeExpr::Map { input, .. }
176        | NativeExpr::Timestamp { input } => groups_by_metric_name(input),
177        NativeExpr::Aggregate { input, op } => {
178            grouping_names_metric(op) || groups_by_metric_name(input)
179        }
180        NativeExpr::ParameterizedMap {
181            input, parameters, ..
182        } => groups_by_metric_name(input) || parameters.iter().any(groups_by_metric_name),
183        NativeExpr::ParameterizedTopK {
184            input, parameter, ..
185        } => groups_by_metric_name(input) || groups_by_metric_name(parameter),
186        NativeExpr::ParameterizedHistogram {
187            input, parameters, ..
188        } => groups_by_metric_name(input) || parameters.iter().any(groups_by_metric_name),
189        NativeExpr::Vector { input, op } => {
190            matches!(op, BoundVectorOp::CountValues { group: Grouping::By(names), .. }
191                if names.iter().any(|name| name.as_ref() == METRIC_NAME))
192                || groups_by_metric_name(input)
193        }
194        NativeExpr::Binary { left, right, .. } | NativeExpr::Set { left, right, .. } => {
195            groups_by_metric_name(left) || groups_by_metric_name(right)
196        }
197        NativeExpr::Info { base, info, .. } => {
198            groups_by_metric_name(base) || groups_by_metric_name(info)
199        }
200    }
201}

203/// Whether a segment can hand `node`'s values to another segment.
204///
205/// A boundary carries an evaluated series frame: values addressed on a grid, under the labels of
206/// the series they belong to. A scalar is broadcast rather than carried, aggregate state is
207/// internal, and raw samples are what a scan produces - none of them is a frame, so a subtree
208/// producing one stays inside the segment that reads it.
209fn exportable(node: &NativeNode) -> bool {
210    node.output
211        .samples()
212        .is_some_and(|schema| schema.eval_grid.is_some())
213}

215/// Whether `node` is a window over an evaluated input - the shape a subquery over anything but a
216/// bare selection compiles to.
217fn evaluated_input_window(node: &NativeNode) -> bool {
218    matches!(&node.expr, NativeExpr::Window { op, .. } if op.input_grid.is_some())
219}

221/// Whether the dist analyzer places a segment over `node` region-locally.
222///
223/// A segment is a `FusedSeries` node the analyzer has to move below the `MergeScan` for the physical
224/// planner to read the scans it names. The `FusedSeries` arm of `Categorizer::check_extension_plan`
225/// (`src/query/src/dist_plan/commutativity.rs`) is what decides that, from the bound tree alone: a
226/// root that [`reads_across_series`] is `NonCommutative` unless it is a per-series evaluation or the
227/// `Partial` half of a split fold. A segment it refuses stays above the merge, where the physical
228/// planner finds neither a region scan nor an explicit-label frame below it, so the statement fails
229/// while being planned instead of keeping the plan it had. Segmentation therefore asks that question
230/// before it cuts, and `cut_as_segment` declines the nodes this refuses.
231pub fn region_local_segment(node: &NativeNode) -> bool {
232    !reads_across_series(node) || pointwise_operand(node) || independent_scan_pair(node)
233}

235/// Whether `node` becomes a segment of its own under a parent that must gather.
236///
237/// Cut local prefixes, global consumers and canonical merges separately so the prefix keeps its
238/// scan width, and every global consumer sees the canonical boundary below it. A pointwise wrapper
239/// of a global consumer is cut as well, so it cannot hide that consumer from recursive lowering.
240fn cut_as_segment(node: &NativeNode, canonical_segmentation: bool) -> bool {
241    // A window over an evaluated input answers per series on its own grid, so its own shape says
242    // nothing about where a segment over it can go: it drops the metric name, so two of the series
243    // it answers can end under one label set, and the dist analyzer reads it as cross-series
244    // ([`region_local_segment`]). Cutting one where the analyzer will not place it leaves a
245    // `FusedSeries` above the `MergeScan` that the physical planner cannot build, so the segment is
246    // only taken where the analyzer places it region-locally; otherwise the per-series prefix
247    // inside the window is cut instead, and where there is none the statement keeps the plan it had.
248    if evaluated_input_window(node) && !region_local_segment(node) {
249        return false;
250    }
251    // A window over an evaluated input is answered per series on its own grid, so it is not the
252    // global gather a raw window's locality reports.
253    let evaluated_window = evaluated_input_window(node);
254    if !exportable(node)
255        || matches!(node.expr, NativeExpr::Input(_))
256        || (!canonical_segmentation
257            && !evaluated_window
258            && matches!(input_locality(node), InputLocality::SinglePartition))
259    {
260        return false;
261    }
262    if canonical_segmentation && (contains_global_consumer(node) || is_canonical_cut(node)) {
263        return true;
264    }
265    // A segment builder needs an input, so data-free subtrees stay with their parent.
266    let mut sources = Vec::new();
267    collect_segment_inputs(node, &mut sources);
268    !sources.is_empty()
269}

271/// Whether any canonical merge in `node` answers with histogram samples, given the kind its input
272/// carries.
273///
274/// A merge collects the pieces of one label set and hands on the samples it was given, so its answer
275/// kind is its child's. A histogram answer standing on a merge would travel through the canonical
276/// boundary carrying histogram frames, a pairing no execution exercises yet: the planner keeps such
277/// a statement on the existing plan until that pairing is covered, and it has to look at every
278/// merge, not only a root one.
279pub fn merge_answers_histogram(node: &NativeNode, input_kind: FieldKind) -> bool {
280    node.any(&|node| {
281        matches!(node.expr, NativeExpr::Merge { .. })
282            && node.output_field_kind(Some(input_kind)) == Some(FieldKind::Histogram)
283    })
284}

286/// Whether the merge producer's entire prefix can run on each scan partition independently.
287pub fn canonical_local_prefix(node: &NativeNode) -> bool {
288    match &node.expr {
289        NativeExpr::Input(_) => true,
290        NativeExpr::Instant { input, .. } | NativeExpr::Window { input, .. } => {
291            matches!(input.expr, NativeExpr::Input(_))
292        }
293        NativeExpr::Map { input, .. } | NativeExpr::Timestamp { input } => {
294            canonical_local_prefix(input)
295        }
296        NativeExpr::ParameterizedMap {
297            input, parameters, ..
298        } => canonical_local_prefix(input) && parameters.iter().all(NativeNode::is_data_free),
299        NativeExpr::Binary { left, right, .. } => {
300            (left.is_data_free() && canonical_local_prefix(right))
301                || (right.is_data_free() && canonical_local_prefix(left))
302        }
303        _ => false,
304    }
305}

315/// Whole-vector consumers remain global, but can be cut out as evaluated segments so a pointwise
316/// parent does not inherit their single-partition placement.
317fn is_global_consumer(node: &NativeNode) -> bool {
318    matches!(
319        node.expr,
320        NativeExpr::Aggregate { .. } | NativeExpr::Vector { .. }
321    ) && matches!(input_locality(node), InputLocality::SinglePartition)
322}

324/// Whether a subtree can be cut without inheriting its parent's global gather.
325///
326/// A local prefix stays on its source partitions. A canonical merge becomes its own routed
327/// boundary, and a parent reads the assembled frames rather than gathering the scan.
328fn is_canonical_cut(node: &NativeNode) -> bool {
329    matches!(node.expr, NativeExpr::Merge { .. })
330        || !matches!(input_locality(node), InputLocality::SinglePartition)
331        || canonical_local_prefix(node)
332}

334/// One segment: its tree rewritten to read only its own inputs, the scans it reads, and the
335/// subtrees cut out of it.
336///
337/// A segment has one index space of its own - the scans it reads first, in the order it first names
338/// them, then the subtrees cut out of it - because `FusedSeries` requires its source indices to be
339/// dense over its children.
340pub struct Segment {
341    pub node: NativeNode,
342    /// Original scan indices, in the order this segment's indices name them.
343    pub sources: Vec<usize>,
344    /// The subtrees cut out of this segment, each still in the original index space.
345    pub cut: Vec<NativeNode>,
346}
347
348/// Collects a segment's scans and cut subtrees, without descending into a cut.
349fn collect_segment(
350    node: &NativeNode,
351    sources: &mut Vec<usize>,
352    cut: &mut Vec<NativeNode>,
353    canonical_segmentation: bool,
354) {
355    match &node.expr {
356        NativeExpr::Input(input) => {
357            if !sources.contains(&input.source_index) {
358                sources.push(input.source_index);
359            }
360        }
361        expr => {
362            for child in expr.children() {
363                if cut_as_segment(child, canonical_segmentation) {
364                    cut.push(child.clone());
365                } else {
366                    collect_segment(child, sources, cut, canonical_segmentation);
367                }
368            }
369        }
370    }
371}

373/// Collects every scan input in a selected segment without looking for nested cuts.
374fn collect_segment_inputs(node: &NativeNode, sources: &mut Vec<usize>) {
375    for input in node.inputs() {
376        if !sources.contains(&input.source_index) {
377            sources.push(input.source_index);
378        }
379    }
380}

382/// Rewrites every input leaf in a selected segment to its index in the segment's own space.
383fn rewrite_inner_segment(node: &NativeNode, sources: &[usize]) -> NativeNode {
384    let expr = match &node.expr {
385        NativeExpr::Input(input) => NativeExpr::Input(BoundInput {
386            source_index: sources
387                .iter()
388                .position(|&source| source == input.source_index)
389                .expect("the segment collected every scan it reads"),
390            output: input.output.clone(),
391            ordering: input.ordering,
392        }),
393        expr => expr.with_children(
394            expr.children()
395                .into_iter()
396                .map(|child| rewrite_inner_segment(child, sources))
397                .collect(),
398        ),
399    };
400    NativeNode {
401        expr,
402        output: node.output.clone(),
403    }
404}

406/// Rewrites a segment's tree so every leaf names its index in the segment's own space.
407fn rewrite_segment(
408    node: &NativeNode,
409    sources: &[usize],
410    cut_base: usize,
411    next_cut: &mut usize,
412    canonical_segmentation: bool,
413) -> NativeNode {
414    let expr = match &node.expr {
415        NativeExpr::Input(input) => NativeExpr::Input(BoundInput {
416            source_index: sources
417                .iter()
418                .position(|&source| source == input.source_index)
419                .expect("the segment collected every scan it reads"),
420            output: input.output.clone(),
421            ordering: input.ordering,
422        }),
423        expr => {
424            let children = expr
425                .children()
426                .into_iter()
427                .map(|child| {
428                    if cut_as_segment(child, canonical_segmentation) {
429                        let index = cut_base + *next_cut;
430                        *next_cut += 1;
431                        NativeNode {
432                            expr: NativeExpr::Input(BoundInput {
433                                source_index: index,
434                                output: child.output.clone(),
435                                ordering: NATIVE_ORDERING,
436                            }),
437                            output: child.output.clone(),
438                        }
439                    } else {
440                        rewrite_segment(child, sources, cut_base, next_cut, canonical_segmentation)
441                    }
442                })
443                .collect::<Vec<_>>();
444            expr.with_children(children)
445        }
446    };
447    NativeNode {
448        expr,
449        output: node.output.clone(),
450    }
451}

453/// Splits `node` into one segment and the subtrees cut out of it.
454pub fn cut_segments(node: &NativeNode, canonical_segmentation: bool) -> Segment {
455    let mut sources = Vec::new();
456    let mut cut = Vec::new();
457    collect_segment(node, &mut sources, &mut cut, canonical_segmentation);
458    let cut_base = sources.len();
459    let mut next_cut = 0;
460    let rewritten = rewrite_segment(
461        node,
462        &sources,
463        cut_base,
464        &mut next_cut,
465        canonical_segmentation,
466    );
467    Segment {
468        node: rewritten,
469        sources,
470        cut,
471    }
472}

474/// Appends the per-row name column to a rows layout's tag list.
475///
476/// A scalar root declares no name column: it publishes one value and no series, so the row it
477/// answers has no name to report. The column is appended, so the layout keeps its order - the time
478/// index, then the value fields, then the tags - and a label set that already reports the name keeps
479/// the one column it has.
480pub fn with_row_name_column(node: &NativeNode, mut tags: Vec<String>) -> Vec<String> {
481    if !matches!(&node.output, NativeType::Samples(_)) {
482        return tags;
483    }
484    // A layout can already report the name: a set operation takes it from its sides, and a
485    // discovered source sorts its tags for the union. Neither leaves that column last - `or` keeps
486    // the left operand's order and only appends the right side's unseen tags, and sorting puts the
487    // name wherever its bytes fall - so the existing one is removed and appended instead: every rows
488    // layout then answers one order, the time index, the value fields, the labels, and the name
489    // last.
490    if let Some(index) = tags.iter().position(|tag| tag == METRIC_NAME) {
491        tags.remove(index);
492    }
493    tags.push(METRIC_NAME.to_string());
494    tags
495}

497/// The ordering this route compiles its trees against: the weakest both scanners provide.
498///
499/// A native scan declares `KeyThenTime` and a two-phase scan `Contiguous`, and the logical planner
500/// cannot read which one a statement will get. `Contiguous` is the weaker of the two - every
501/// series' samples contiguous, the series themselves unordered - and a native evaluation only
502/// refuses `Interleaved`, so a tree compiled against this ordering runs on either scanner.
503pub const NATIVE_ORDERING: SeriesOrdering = SeriesOrdering::Contiguous;

505/// The columns a [`FusedSeries`] node produces.
506///
507/// With `experimental_native_promql` enabled, a value column is named by the caller's [`FieldNames`]
508/// answer for the statement's expression, not legacy planner expression text. Legacy headers in
509/// `tests/cases/**` are unaffected because the native route is opt-in and off by default.
510///
511/// The shape is the one the existing unary pipeline consumes — the time index, then the series
512/// fields, then the tags — and each column is declared the way the node's own conversion reads it
513/// back:
514/// the time column is a non-nullable millisecond timestamp with no timezone, because every
515/// evaluated chunk is addressed on the statement's grid and carries one; a field is a nullable
516/// `Float64`, because a grid point inside the statement's interval can have no sample; and a tag is
517/// a nullable `Utf8`, because a range function drops the metric name and a series can be missing a
518/// label entirely.
519/// The output schema of a fused node: its time index, one column per field, then its tags.
520///
521/// `field_kind` says what a field column carries. The physical conversion reads the same kind off
522/// this schema and dispatches on it, so the schema and the batch it describes cannot disagree.
523pub fn fused_output_schema(
524    fields: &[String],
525    tags: &[String],
526    time_index: &str,
527    field_kind: FieldKind,
528) -> datafusion::common::Result<DFSchemaRef> {
529    let field_type = match field_kind {
530        FieldKind::Float => ArrowDataType::Float64,
531        FieldKind::Histogram => native_histogram_value_type().as_arrow_type(),
532    };
533    let mut columns = Vec::with_capacity(fields.len() + tags.len() + 1);
534    // The time index is marked as one, the way the existing planner's plans mark the table's own
535    // time index: what a plan publishes is a result either way, and a reader of the marker - a
536    // `RANGE` query over the plan, a schema converted back to GreptimeDB's - has to see the same
537    // column on both routes.
538    columns.push((
539        None,
540        Arc::new(
541            Field::new(
542                time_index,
543                ArrowDataType::Timestamp(ArrowTimeUnit::Millisecond, None),
544                false,
545            )
546            .with_metadata(std::collections::HashMap::from([(
547                datatypes::schema::TIME_INDEX_KEY.to_string(),
548                "true".to_string(),
549            )])),
550        ),
551    ));
552    for name in fields {
553        columns.push((None, Arc::new(Field::new(name, field_type.clone(), true))));
554    }
555    for name in tags {
556        let mut field = Field::new(name, ArrowDataType::Utf8, true);
557        // The runtime answers `__name__` from the metric name of the series it reports, so a tag
558        // column of that name is the name of each row rather than one name for the whole result.
559        // Declaring it keeps the response layer from naming every row after the statement.
560        if name == METRIC_NAME {
561            field = field.with_metadata(datatypes::schema::metric_name_field_metadata());
562        }
563        columns.push((None, Arc::new(field)));
564    }
565
566    // GreptimeDB's schemas carry their version, and a result schema is no different: a schema
567    // read back from this plan is one, exactly as the existing planner's answer for the same
568    // statement is.
569    let metadata = std::collections::HashMap::from([(
570        datatypes::schema::VERSION_KEY.to_string(),
571        datatypes::schema::Schema::INITIAL_VERSION.to_string(),
572    )]);
573    Ok(Arc::new(DFSchema::new_with_metadata(columns, metadata)?))
574}

576/// The grid one `FusedSeries` publishes on: the grid its root declares it produces, or the
577/// statement's grid when the root declares none.
578///
579/// A root that produces one value per grid point says which grid that is, and that declared grid is
580/// the one its consumer reads it on: a window over an evaluated input - a subquery - produces on the
581/// window's own output grid, and a non-window root inside such a window, like the `Map` an
582/// expression of a subquery compiles to, produces on the subquery's grid rather than on the
583/// statement's. A statement root declares the statement's grid, so the two agree there. A root with
584/// no declared grid - raw samples - is read where it was scanned and keeps the caller's grid.
585/// Every lowering path binds the effective grid (the subquery's child grid for a matrix root, the
586/// statement's grid otherwise), so a window root answers that same grid here.
587fn fused_publish_grid(root: &NativeNode, statement_grid: &Arc<EvalGrid>) -> Arc<EvalGrid> {
588    root.output
589        .grid()
590        .map_or_else(|| Arc::clone(statement_grid), Arc::clone)
591}

593/// Lowers a tree whose root is a canonical merge into a producer and a canonical router.
594///
595/// The tree says the statement answers one series per canonical label set; what it cannot say is
596/// where the pieces of one set are, because a rewrite created them after the scan split the rows.
597/// So the pieces have to meet before the answer is written, and there are only two honest ways to
598/// make them meet: read every partition in one execution, or route each label set to one owner. The
599/// first takes the caller's parallelism away for the whole query; this is the second.
600///
601/// Two stages come out of it, and each has one job:
602///
603/// - the **producer** runs the tree's root merge on every scan partition. Its merge collects the
604///   pieces that partition holds - which is what makes a same-partition collision an error here,
605///   before anything is routed - and it exports one series per label set as a series frame. It is
606///   explicitly partition-local: the same node without that distinction would gather its inputs and
607///   answer everything in one execution.
608/// - the **router** is `SeriesRepartitionExec`'s canonical routing, which collects the pieces of one
609///   label set from *every* input partition. It is the conservative router - a label set is
610///   published only once every producer contributed it or ended - because a producer promises one
611///   series per set *per partition*, not that each set lives in one partition. A second piece of one
612///   set that holds a position another piece already holds is the same error, reported while that
613///   label set is assembled, and each output of the router then holds whole label sets: one series
614///   per set, complementary points joined.
615///
616/// `None` when the boundary cannot be built, so the caller keeps the plan it would have chosen.
617/// The stages of a canonical boundary over `tree`, and the columns its frames carry.
618///
619/// `format` says what the caller above reads: a statement root reads the boundary's rows, and
620/// anything else reads the assembled frames as an ordinary evaluated input.
621#[allow(clippy::too_many_arguments)]
622pub fn canonical_boundary_stages(
623    promql: Arc<str>,
624    tree: &NativeTree,
625    children: &[LogicalPlan],
626    source_tags: &[Vec<String>],
627    grid: &Arc<EvalGrid>,
628    time_index: &str,
629    tags: &[String],
630    metric: &Option<Arc<str>>,
631    format: FusedOutputFormat,
632    root_kind: FieldKind,
633    output_tags_of: OutputTags,
634    field_names: FieldNames,
635) -> Option<(LogicalPlan, Vec<String>)> {
636    let NativeExpr::Merge { input } = &tree.root().expr else {
637        return None;
638    };
639    if !canonical_local_prefix(input) {
640        return None;
641    }
642    let NativeType::Samples(schema) = &tree.root().output else {
643        return None;
644    };
645    schema.eval_grid.as_ref()?;
646    let fields = schema
647        .fields
648        .iter()
649        .map(|field| field.name.to_string())
650        .collect::<Vec<_>>();
651    let output_tags = output_tags_of(tree.root(), source_tags)?;
652    // The terminal rows of this boundary report the name of each row they publish, so the schema
653    // declares it exactly as the other rows layouts do. The tags stay the walk's own: they name the
654    // labels the frames carry, and the binding below is built from them.
655    let row_tags = with_row_name_column(tree.root(), output_tags.clone());
656    // The values are named after the expression, as in every other rows layout of this statement.
657    // A schema this boundary cannot declare declines here: the caller's error type is not one this
658    // crate can build, so a name collision is reported by declining rather than as an internal
659    // error, and the caller keeps the plan it would have chosen.
660    let value_names = field_names(&promql, &fields);
661    let output_schema = fused_output_schema(&value_names, &row_tags, time_index, root_kind).ok()?;
662
663    // The producer: the statement's own tree, run per partition, exporting a frame. It runs above
664    // the regions, where its router collects each label set from every region's contribution.
665    let producer = FusedSeries::try_new_canonical_producer(
666        Arc::clone(&promql),
667        tree.clone(),
668        FusedSeriesBinding {
669            grid: Arc::clone(grid),
670            time_index: time_index.to_string(),
671            tags: tags.to_vec(),
672            metric_name: metric.clone(),
673        },
674        children.to_vec(),
675    )
676    .ok()?;
677    let producer = LogicalPlan::Extension(Extension {
678        node: Arc::new(producer),
679    });
680
681    // What reads the router depends on where this boundary sits. A consumer still above it reads
682    // whole label sets, and the router already merged the pieces of one label set into one series,
683    // so that consumer is an ordinary evaluated input of its own and needs no merge. The statement's
684    // root instead publishes rows, which is this boundary's own frame converted as usual.
685    if format == FusedOutputFormat::SeriesFrame {
686        return Some((producer, output_tags));
687    }
688    let evaluated = NativeType::Samples(Arc::clone(schema));
689    let consumer_tree = NativeTree::try_new(NativeNode {
690        expr: NativeExpr::Input(BoundInput {
691            source_index: 0,
692            output: evaluated.clone(),
693            ordering: NATIVE_ORDERING,
694        }),
695        output: evaluated,
696    })
697    .ok()?;
698    // A frame names each series' own metric, so the binding names one only when a tag column does.
699    let consumer_metric = if output_tags.iter().any(|tag| tag == METRIC_NAME) {
700        metric.clone()
701    } else {
702        None
703    };
704    let consumer = FusedSeries::try_new_with_format(
705        Arc::clone(&promql),
706        consumer_tree,
707        FusedSeriesBinding {
708            grid: Arc::clone(grid),
709            time_index: time_index.to_string(),
710            tags: output_tags.clone(),
711            metric_name: consumer_metric,
712        },
713        vec![producer],
714        output_schema,
715        FusedOutputFormat::PlainRows,
716    )
717    .ok()?;
718    Some((
719        LogicalPlan::Extension(Extension {
720            node: Arc::new(consumer),
721        }),
722        output_tags,
723    ))
724}

739/// Whether `plan` is the typed empty input a discovered selector that matched no table gets.
740///
741/// It reads no region and holds no scan, so it is not one of the tables a node's binding is derived
742/// from: two selectors are "of one table" only if the tables they scan are one. The remote
743/// admission gate reads the same shape back, to tell a node with nothing to ship from one that does
744/// read regions, so the declared frame is part of the check: an empty relation that does not
745/// declare the raw explicit-label frame is some other plan, not this input.
746pub fn is_empty_discovered_input(plan: &LogicalPlan) -> bool {
747    let LogicalPlan::EmptyRelation(relation) = plan else {
748        return false;
749    };
750    if relation.produce_one_row {
751        return false;
752    }
753    let Ok(declaration) = SeriesFrameDeclaration::parse(relation.schema.as_arrow()) else {
754        return false;
755    };
756    if declaration.content() != FrameContent::Raw
757        || declaration.labels() != FrameLabelMode::Explicit
758    {
759        return false;
760    }
761    let Ok(frame) = raw_series_frame_schema_with_ordering(
762        declaration.series(),
763        &RawLabelSource::Explicit,
764        NATIVE_ORDERING,
765    ) else {
766        return false;
767    };
768    relation.schema.as_arrow() == frame.as_ref()
769}

771#[allow(clippy::too_many_arguments, clippy::type_complexity)]
772fn build_evaluated_segment(
773    promql: Arc<str>,
774    node: &NativeNode,
775    children: &[LogicalPlan],
776    source_tags: &[Vec<String>],
777    source_ids: &[Option<usize>],
778    per_source: &[(Option<Arc<str>>, String, Vec<String>)],
779    grid: &Arc<EvalGrid>,
780    time_index: &str,
781    root_kind: FieldKind,
782    output_tags_of: OutputTags,
783    field_names: FieldNames,
784) -> Option<(LogicalPlan, Vec<String>)> {
785    let mut sources = Vec::new();
786    collect_segment_inputs(node, &mut sources);
787    if sources.is_empty() {
788        return None;
789    }
790    let mut inputs = Vec::with_capacity(sources.len());
791    let mut local_tags = Vec::with_capacity(sources.len());
792    let tree = NativeTree::try_new(rewrite_inner_segment(node, &sources)).ok()?;
793    let source_fields = tree_source_fields(&tree).ok()?;
794    for (local_index, &index) in sources.iter().enumerate() {
795        let child = children.get(index)?.clone();
796        let input = find_input_node(node, index)?;
797        if input
798            .output
799            .samples()
800            .is_some_and(|schema| schema.eval_grid.is_some())
801            || is_raw_series_source(&child)
802            // The typed empty frame of a selector that matched no table is an input in its own
803            // right: it reads no region and carries no metric, so there is nothing to wrap it in.
804            // A segment over it is still an explicit-label frame with no series.
805            || is_empty_discovered_input(&child)
806        {
807            inputs.push(child);
808            local_tags.push(source_tags.get(index)?.clone());
809        } else {
810            let original_index = *source_ids.get(index)?.as_ref()?;
811            let (Some((Some(metric), _, tags)), Some(fields)) = (
812                per_source.get(original_index),
813                source_fields.get(local_index),
814            ) else {
815                return None;
816            };
817            inputs.push(LogicalPlan::Extension(Extension {
818                node: Arc::new(
819                    RawSeriesSource::try_new(
820                        Arc::clone(metric),
821                        tags.clone(),
822                        fields.clone(),
823                        child,
824                    )
825                    .ok()?,
826                ),
827            }));
828            local_tags.push(tags.clone());
829        }
830    }
831    // A canonical merge is not a segment: it is the middle of a boundary whose two frame nodes and
832    // router are what answer one series per label set across partitions, so it is built as that
833    // boundary and read as a frame.
834    if matches!(tree.root().expr, NativeExpr::Merge { .. }) {
835        let (plan, tags) = canonical_boundary_stages(
836            Arc::clone(&promql),
837            &tree,
838            &inputs,
839            &local_tags,
840            grid,
841            time_index,
842            &[],
843            &None,
844            FusedOutputFormat::SeriesFrame,
845            root_kind,
846            output_tags_of,
847            field_names,
848        )?;
849        return Some((plan, tags));
850    }
851    let output_tags = output_tags_of(tree.root(), &local_tags)?;
852    if !native_tree_supported(&tree) {
853        return None;
854    }
855    let frame = FusedSeries::series_frame_schema(&tree).ok()?;
856    let frame = DFSchema::try_from(frame.as_ref().clone()).ok()?;
857    let publish_grid = fused_publish_grid(tree.root(), grid);
858    let plan = FusedSeries::try_new_with_format(
859        Arc::from("segment"),
860        tree,
861        FusedSeriesBinding {
862            grid: publish_grid,
863            time_index: time_index.to_string(),
864            tags: output_tags.clone(),
865            metric_name: None,
866        },
867        inputs,
868        Arc::new(frame),
869        FusedOutputFormat::SeriesFrame,
870    )
871    .ok()?;
872    Some((
873        LogicalPlan::Extension(Extension {
874            node: Arc::new(plan),
875        }),
876        output_tags,
877    ))
878}

880/// Builds one segment's plan from a subtree, binding its scan inputs without cutting again.
881///
882/// Returns the plan and the output columns the segment reports, which its parent names its own
883/// columns by. `None` when any part cannot be built, so the statement keeps the plan it had.
884#[allow(clippy::type_complexity)]
885fn build_inner_segment(
886    node: &NativeNode,
887    bare_scans: &[LogicalPlan],
888    per_source: &[(Option<Arc<str>>, String, Vec<String>)],
889    grid: &Arc<EvalGrid>,
890    output_tags_of: OutputTags,
891) -> Option<(LogicalPlan, Vec<String>)> {
892    let mut sources = Vec::new();
893    collect_segment_inputs(node, &mut sources);
894    if sources.is_empty() {
895        return None;
896    }
897    let scans = sources
898        .iter()
899        .map(|&index| bare_scans.get(index).cloned())
900        .collect::<Option<Vec<_>>>()?;
901    let source_tags = sources
902        .iter()
903        .map(|&index| per_source.get(index).map(|(_, _, tags)| tags.clone()))
904        .collect::<Option<Vec<_>>>()?;
905    let tree = NativeTree::try_new(rewrite_inner_segment(node, &sources)).ok()?;
906    let output_tags = output_tags_of(tree.root(), &source_tags)?;
907    // A segment publishes on the grid its own root declares, which is the grid its parent reads it
908    // on: the segments of a subquery's inner expression are evaluated on the subquery's grid, not on
909    // the statement's, whatever their root computes. The statement's grid is only the fallback for a
910    // root that declares none.
911    let segment_grid = fused_publish_grid(tree.root(), grid);
912    let (inputs, metric, time_index, tags) = if let [index] = sources.as_slice() {
913        let (metric, time_index, tags) = per_source.get(*index)?;
914        (scans, metric.clone(), time_index.clone(), tags.clone())
915    } else {
916        // A binding cannot describe different source identities. Give each child the existing
917        // explicit-label source boundary used for separate metric tables.
918        let source_fields = tree_source_fields(&tree).ok()?;
919        let mut children = Vec::with_capacity(sources.len());
920        for (local_index, &source_index) in sources.iter().enumerate() {
921            let (Some((Some(metric), _, source_tags)), Some(fields), Some(scan)) = (
922                per_source.get(source_index),
923                source_fields.get(local_index),
924                scans.get(local_index),
925            ) else {
926                return None;
927            };
928            children.push(LogicalPlan::Extension(Extension {
929                node: Arc::new(
930                    RawSeriesSource::try_new(
931                        Arc::clone(metric),
932                        source_tags.clone(),
933                        fields.clone(),
934                        scan.clone(),
935                    )
936                    .ok()?,
937                ),
938            }));
939        }
940        let time_index = per_source.get(*sources.first()?)?.1.clone();
941        (children, None, time_index, output_tags.clone())
942    };
943    if !native_tree_supported(&tree) {
944        return None;
945    }
946    let frame = FusedSeries::series_frame_schema(&tree).ok()?;
947    let frame = DFSchema::try_from(frame.as_ref().clone()).ok()?;
948    let plan = FusedSeries::try_new_with_format(
949        Arc::from("segment"),
950        tree,
951        FusedSeriesBinding {
952            grid: segment_grid,
953            time_index: time_index.clone(),
954            tags: tags.clone(),
955            metric_name: metric.clone(),
956        },
957        inputs,
958        Arc::new(frame),
959        FusedOutputFormat::SeriesFrame,
960    )
961    .ok()?;
962    Some((
963        LogicalPlan::Extension(Extension {
964            node: Arc::new(plan),
965        }),
966        output_tags,
967    ))
968}

970/// Plans `tree` as several segments, or `None` when it is one node already.
971///
972/// A tree that reads across series answers one partition and gathers whatever it reads. Its
973/// per-series prefixes do not need that gather, so each becomes a segment of its own and keeps the
974/// partitions it was scanned with. Global consumers and canonical merges are cut in their original
975/// order, and recursive segments publish frames for their parents to consume. Only the statement
976/// root publishes rows, and it does so under the statement's own value names: [`FieldNames`]
977/// answers those, and the rows schema asks [`RootSchema`] for the rest of the layout.
978#[allow(clippy::too_many_arguments, clippy::type_complexity)]
979pub fn segment_tree(
980    promql: Arc<str>,
981    tree: &NativeTree,
982    children: &[LogicalPlan],
983    source_tags: &[Vec<String>],
984    bare_scans: &[LogicalPlan],
985    per_source: &[(Option<Arc<str>>, String, Vec<String>)],
986    grid: &Arc<EvalGrid>,
987    time_index: &str,
988    tags: &[String],
989    metric: &Option<Arc<str>>,
990    format: FusedOutputFormat,
991    canonical_segmentation: bool,
992    root_kind: FieldKind,
993    remote_raw_boundary: bool,
994    output_tags_of: OutputTags,
995    field_names: FieldNames,
996    root_schema: RootSchema,
997) -> Option<LogicalPlan> {
998    // Only a tree the analyzer gathers is worth splitting: its per-series prefixes then run where
999    // their input is read. A window over an evaluated input can need that split even where
1000    // [`input_locality`] reports a local tree - it drops the metric name, so [`reads_across_series`]
1001    // reads it as cross-series - and the per-series prefix inside it is what keeps the partitions it
1002    // scanned. Proceeding only admits the attempt: every cut is admitted by `cut_as_segment`
1003    // individually, which is where the analyzer's placement is asked.
1004    if !matches!(input_locality(tree.root()), InputLocality::SinglePartition)
1005        && !evaluated_input_window(tree.root())
1006    {
1007        return None;
1008    }
1009    // A segment boundary has to carry something, and the only thing it can carry is a per-series
1010    // prefix: a subtree that answers each series on its own and can be recomputed. A tree with no
1011    // such subtree inside it has nothing to cut, and cutting anything else - an aggregate's own
1012    // state, a constant - is either impossible or a plain recomputation of the same work in one
1013    // more node. Cutting on locality alone would accept one here.
1014    //
1015    // The window this tree may be rooted at is not one of those consumers, but it is no boundary
1016    // either: `cut_as_segment` never cuts one, and what keeps its input's partitions is the
1017    // per-series prefix below it, which the cuts below are admitted for.
1018    // A grouping by `__name__` keeps that label while the fused node's declared columns cannot
1019    // name it. With multiple sources, groups differing only by metric name would therefore be
1020    // reported under one label set. A single-source tree cannot merge distinct metric-name groups
1021    // this way, so this restriction is specific to multi-source segmentation.
1022    if children.len() > 1 && groups_by_metric_name(tree.root()) {
1023        return None;
1024    }
1025    let mut current_tree = tree.clone();
1026    let mut current_children = children.to_vec();
1027    let mut current_tags = source_tags.to_vec();
1028    let mut current_source_ids = (0..children.len()).map(Some).collect::<Vec<_>>();
1029    let mut first_iteration = true;
1030    let mut changed = false;
1031    loop {
1032        // A tree the analyzer keeps partition-local needs no segment; a window over an evaluated
1033        // input is still worth attempting one for, so its per-series prefix keeps its partitions.
1034        if !matches!(
1035            input_locality(current_tree.root()),
1036            InputLocality::SinglePartition
1037        ) && !evaluated_input_window(current_tree.root())
1038        {
1039            break;
1040        }
1041        let segment = cut_segments(current_tree.root(), canonical_segmentation);
1042        if segment.cut.is_empty() {
1043            break;
1044        }
1045        let mut inputs = Vec::with_capacity(segment.sources.len() + segment.cut.len());
1046        let mut child_tags = Vec::with_capacity(inputs.capacity());
1047        for &index in &segment.sources {
1048            let child = current_children.get(index)?;
1049            inputs.push(child.clone());
1050            child_tags.push(current_tags.get(index).cloned().unwrap_or_default());
1051        }
1052        let mut child_source_ids = segment
1053            .sources
1054            .iter()
1055            .map(|&index| current_source_ids.get(index).copied().flatten())
1056            .collect::<Vec<_>>();
1057        for cut in &segment.cut {
1058            // A cut may wrap a global consumer in a pointwise rewrite or a later merge. Segment
1059            // that subtree first so its inner canonical boundary stays below the consumer. A local
1060            // canonical merge instead goes directly to the boundary builder.
1061            let canonical = matches!(cut.expr, NativeExpr::Merge { .. });
1062            let built = if first_iteration && contains_global_consumer(cut) {
1063                let cut_tree = NativeTree::try_new(cut.clone()).ok()?;
1064                let plan = segment_tree(
1065                    Arc::clone(&promql),
1066                    &cut_tree,
1067                    &current_children,
1068                    &current_tags,
1069                    bare_scans,
1070                    per_source,
1071                    grid,
1072                    time_index,
1073                    tags,
1074                    metric,
1075                    FusedOutputFormat::SeriesFrame,
1076                    canonical_segmentation,
1077                    root_kind,
1078                    remote_raw_boundary,
1079                    output_tags_of,
1080                    field_names,
1081                    root_schema,
1082                );
1083                match plan {
1084                    Some(plan) => {
1085                        let cut_tags = output_tags_of(cut, &current_tags)?;
1086                        Some((plan, cut_tags))
1087                    }
1088                    None => None,
1089                }
1090            } else if first_iteration && !canonical && !remote_raw_boundary {
1091                build_inner_segment(cut, bare_scans, per_source, grid, output_tags_of)
1092            } else {
1093                None
1094            };
1095            if built.is_none() && contains_global_consumer(cut) {
1096                return None;
1097            }
1098            // A cut the first iteration cannot build from the scans is built from the frames this
1099            // loop has already published: the canonical pass inserts a merge under a label rewrite,
1100            // and that boundary - rebuilt here as a producer, a frame and a router - is what answers
1101            // one series per label set across partitions. Which cuts are taken is
1102            // `cut_as_segment`'s decision, and it declines the evaluated-input window the analyzer
1103            // will not place region-locally, so this fallback only ever sees the cuts it admitted.
1104            let built = built.or_else(|| {
1105                build_evaluated_segment(
1106                    Arc::clone(&promql),
1107                    cut,
1108                    &current_children,
1109                    &current_tags,
1110                    &current_source_ids,
1111                    per_source,
1112                    grid,
1113                    time_index,
1114                    root_kind,
1115                    output_tags_of,
1116                    field_names,
1117                )
1118            });
1119            let (plan, cut_tags) = built?;
1120            inputs.push(plan);
1121            child_tags.push(cut_tags);
1122            child_source_ids.push(None);
1123        }
1124        let parent = NativeTree::try_new(segment.node).ok()?;
1125        first_iteration = false;
1126        current_tree = parent;
1127        current_children = inputs;
1128        current_tags = child_tags;
1129        current_source_ids = child_source_ids;
1130        changed = true;
1131    }
1132    if !changed {
1133        return None;
1134    }
1135    let parent = current_tree;
1136    // Segmentation is useful even when the resulting parent is still single-partition: each
1137    // exported per-series subtree remains a separate input and is bound to its own source.
1138    let mut inputs = current_children;
1139    let mut child_tags = current_tags;
1140    // A merge at this segment's root still needs its own boundary. An outer rewrite above a
1141    // recursively cut ranking runs after the ranking, then is canonicalized before its caller.
1142    if let NativeExpr::Merge { input } = &parent.root().expr {
1143        let source_fields = tree_source_fields(&parent).ok();
1144        for index in 0..inputs.len() {
1145            let bound = find_input_node(input, index)?;
1146            if bound
1147                .output
1148                .samples()
1149                .is_some_and(|schema| schema.eval_grid.is_none())
1150                && !is_raw_series_source(&inputs[index])
1151            {
1152                let source_index = current_source_ids.get(index).copied().flatten()?;
1153                let (Some((Some(source_metric), _, source_tags)), Some(fields)) = (
1154                    per_source.get(source_index),
1155                    source_fields.as_ref().and_then(|fields| fields.get(index)),
1156                ) else {
1157                    return None;
1158                };
1159                let Ok(source) = RawSeriesSource::try_new(
1160                    Arc::clone(source_metric),
1161                    source_tags.clone(),
1162                    fields.clone(),
1163                    inputs[index].clone(),
1164                ) else {
1165                    return None;
1166                };
1167                inputs[index] = LogicalPlan::Extension(Extension {
1168                    node: Arc::new(source),
1169                });
1170                child_tags[index] = source_tags.clone();
1171            }
1172        }
1173        return canonical_boundary_stages(
1174            Arc::clone(&promql),
1175            &parent,
1176            &inputs,
1177            &child_tags,
1178            grid,
1179            time_index,
1180            tags,
1181            metric,
1182            format,
1183            root_kind,
1184            output_tags_of,
1185            field_names,
1186        )
1187        .map(|(plan, _)| plan);
1188    }
1189    // Nothing below can fall back once physical planning starts, so every tree this route commits
1190    // to has to be one the runtime executes. A shape the runtime refuses - a grouping by a value
1191    // field, an operation with no kernel - therefore keeps the plan it had.
1192    if !native_tree_supported(&parent) {
1193        return None;
1194    }
1195    let output_schema = match format {
1196        FusedOutputFormat::PlainRows => {
1197            root_schema(&promql, &parent, &child_tags, time_index, root_kind)?
1198        }
1199        FusedOutputFormat::SeriesFrame => {
1200            let Ok(frame) = FusedSeries::series_frame_schema(&parent) else {
1201                return None;
1202            };
1203            let Ok(schema) = DFSchema::try_from(frame.as_ref().clone()) else {
1204                return None;
1205            };
1206            Arc::new(schema)
1207        }
1208    };
1209    // A recursively constructed segment still has its original source binding. Its parent may
1210    // combine another metric; giving the segment that parent's metric would relabel both inputs.
1211    // Direct scan segments keep their established builder and binding unchanged.
1212    let segment_binding = |source_ids: &[Option<usize>]| -> (Option<Arc<str>>, String) {
1213        let mut sources = source_ids.iter().flatten().copied();
1214        let Some(source) = sources.next() else {
1215            return (metric.clone(), time_index.to_string());
1216        };
1217        if sources.next().is_some() {
1218            return (None, time_index.to_string());
1219        }
1220        per_source.get(source).map_or_else(
1221            || (metric.clone(), time_index.to_string()),
1222            |(metric, time, _)| (metric.clone(), time.clone()),
1223        )
1224    };
1225    let (segment_metric, segment_time_index) = segment_binding(&current_source_ids);
1226    // A segment publishes on the grid its root declares, which is the grid its consumer reads it
1227    // on. Segmentation recurses into a cut subtree with the caller's grid, and that subtree can sit
1228    // inside a subquery window: the canonical segmentation of a collapsing rewrite cuts the window's
1229    // input while its caller still holds the statement's grid. Binding the caller's grid here
1230    // rejects such a segment, and the rejection aborts the whole route.
1231    let publish_grid = fused_publish_grid(parent.root(), grid);
1232    let Ok(plan) = FusedSeries::try_new_with_format(
1233        Arc::clone(&promql),
1234        parent,
1235        FusedSeriesBinding {
1236            grid: publish_grid,
1237            time_index: segment_time_index,
1238            tags: tags.to_vec(),
1239            metric_name: segment_metric,
1240        },
1241        inputs,
1242        output_schema,
1243        format,
1244    ) else {
1245        return None;
1246    };
1247    Some(LogicalPlan::Extension(Extension {
1248        node: Arc::new(plan),
1249    }))
1250}
