15//! The bound tree a native PromQL evaluation compiles to.
16//!
17//! A query is compiled into [`NativeTree`]: [`NativeNode`]s whose operations are already bound to
18//! kernels, fields and grids. Nothing in this module interprets a function name, resolves a label
19//! or picks a column at run time; those decisions are made before the tree exists, so the hot
20//! path only feeds batches to kernels it was handed.
21//!
22//! The tree is also the unit of legality. [`NativeNode::validate`] checks the contract every
23//! operation states about its input, which is what lets a new combination of supported operations
24//! be an ordinary tree instead of another hand-written shape.
25//!
26//! What the tree cannot express yet is deliberate for this stage: [`NativeExpr::Scalar`] holds a
27//! constant, so `time()` and `scalar(v)` have no node. The legality below is therefore about the
28//! operations that exist, not about PromQL's whole surface.

65/// A conservative set of reportable label names for `node`.
66///
67/// `source_tags` is indexed by each leaf's `source_index`. The canonical pass supplies runtime
68/// label names, including `__name__` for named scans; the planner supplies declared tag columns
69/// to derive output columns. The result need not describe the labels of any individual series:
70/// `or` includes a nullable `__name__` column even when both operands drop their names. Math
71/// operations remove only `__name__` here, while their runtime also removes `__type__` and
72/// `__unit__`.
73///
74/// They follow Prometheus' result labels: a one-to-one match keeps the left side's labels, narrowed
75/// by `on` or `ignoring`; `and` and `unless` keep the left side's series and `or` adds the right
76/// side's; `info` adds the data labels of the info table; an operation with a scalar or a function
77/// of each value keeps its operand's; a scalar and `vector(s)` have none; `count_values` keeps its
78/// grouping and the label it writes; a fill that answers a missing many-side sample files the one
79/// side's matching labels, so they join the set, while a grouped `include` label joins them only when
80/// the one side carries it or the matching keeps it in the key a fill reads. A label rewrite writes
81/// its destination, and a destination that names the metric name writes that name.
82///
83/// A node that reports a value of its own drops the name where its kernel does - `timestamp`, a
84/// range function whose result is computed (`rate`, `increase`, ...), a math or histogram map,
85/// `histogram_quantile`, every non-filtering binary - so a name rewrite below one of them cannot
86/// make the name reappear above it.
87///
88/// `None` for a node whose labels this walk does not derive - a `count_values` whose destination is
89/// empty, or an input whose source has no entry. `None` proves nothing: the canonical pass treats
90/// it as "may alias" rather than as "distinct", and the planner keeps the statement off the native
91/// route.
92pub fn fused_output_tags(node: &NativeNode, source_tags: &[Vec<String>]) -> Option<Vec<String>> {
93    match &node.expr {
94        NativeExpr::Input(input) => source_tags.get(input.source_index).cloned(),
95        NativeExpr::Scalar(_) | NativeExpr::EvalTime { .. } | NativeExpr::VectorOf { .. } => {
96            Some(Vec::new())
97        }
98        // These report a value of their own rather than the sample they read, and the runtime drops
99        // the name for each of them. `source_tags` is the authority on what an input carries, so a
100        // discovery input arrives carrying `__name__` and these arms are what decides it does not
101        // survive.
102        NativeExpr::Instant { input, op } => {
103            let tags = fused_output_tags(input, source_tags)?;
104            Some(match op.value {
105                InstantValue::Timestamp => without_metric_name(tags),
106                InstantValue::Sample | InstantValue::InfoTimestamp => tags,
107            })
108        }
109        NativeExpr::Window { input, op } => {
110            let tags = fused_output_tags(input, source_tags)?;
111            Some(if op.function.drops_metric_name() {
112                without_metric_name(tags)
113            } else {
114                tags
115            })
116        }
117        NativeExpr::Timestamp { input } => {
118            Some(without_metric_name(fused_output_tags(input, source_tags)?))
119        }
120        // A label rewrite changes the label set itself: `label_replace` adds its destination label
121        // and an empty replacement removes it, so the input's labels are not the output's. A math or
122        // histogram kernel drops the metric name where it runs, and the ops apply in written order,
123        // so what follows a rewrite sees the set the rewrite left.
124        NativeExpr::Map { input, op } => {
125            let mut tags = fused_output_tags(input, source_tags)?;
126            for operation in op.ops.iter() {
127                match operation {
128                    MapOp::Label(rewrite) => {
129                        let destination = rewrite.destination();
130                        if tags.iter().any(|tag| tag == destination) {
131                            // The input carries the label, so the rewrite changes its value - and an
132                            // empty replacement may remove it, which the column still represents.
133                            tags.retain(|tag| tag != destination);
134                            tags.push(destination.to_string());
135                        } else if rewrite.writes_value() {
136                            // A fresh destination is a label the output only carries when the
137                            // rewrite can write a value; a delete-only rewrite of a label no input
138                            // carried publishes nothing.
139                            tags.push(destination.to_string());
140                        }
141                    }
142                    MapOp::Math(_) | MapOp::Histogram(_) => tags = without_metric_name(tags),
143                }
144            }
145            Some(tags)
146        }
147        // Its parameters are scalars, which carry no labels; the map computes a value of its own
148        // with the input, and its kernel drops the metric name.
149        NativeExpr::ParameterizedMap { input, .. } => {
150            Some(without_metric_name(fused_output_tags(input, source_tags)?))
151        }
152        // `scalar()` reports one value and no labels, like the scalar it produces.
153        NativeExpr::ScalarOf { .. } => Some(Vec::new()),
154        // Its parameter is a scalar, which carries no labels, and ranking keeps each selected
155        // series' own labels.
156        NativeExpr::ParameterizedTopK { input, .. } => fused_output_tags(input, source_tags),
157        // A parameterized histogram function folds the bucket series that share the other labels
158        // into one series and drops `le` along with the metadata labels, exactly as its bound
159        // counterpart does. Its parameters are scalars and carry no labels.
160        NativeExpr::ParameterizedHistogram { input, .. } => {
161            let mut tags = fused_output_tags(input, source_tags)?;
162            tags.retain(|tag| tag != LE_LABEL && !PROMQL_METADATA_LABELS.contains(&tag.as_str()));
163            Some(tags)
164        }
165        // An aggregate reports the grouping labels its input carries: `by(...)` keeps exactly the
166        // ones it names, and `without(...)` keeps the rest and drops the metric name, as Prometheus
167        // does. A grouping by `__name__` keeps the name where the input reports one; a named scan
168        // only reports it among the runtime labels, so the aggregate keeps it there and publishes no
169        // column for it, which is what `sum by (__name__)` over one metric reports.
170        NativeExpr::Aggregate { input, op } => {
171            let input_tags = fused_output_tags(input, source_tags)?;
172            Some(match &op.group {
173                Grouping::By(kept) => input_tags
174                    .into_iter()
175                    .filter(|tag| kept.iter().any(|name| name.as_ref() == tag.as_str()))
176                    .collect(),
177                Grouping::Without(dropped) => input_tags
178                    .into_iter()
179                    .filter(|tag| {
180                        tag != METRIC_NAME
181                            && !dropped.iter().any(|name| name.as_ref() == tag.as_str())
182                    })
183                    .collect(),
184            })
185        }
186        // Selecting some of the input's series keeps each one's own labels, and `count_values`
187        // keeps its grouping's; `histogram` folds bucket series into one histogram, so it does not
188        // report the input's set.
189        NativeExpr::Vector { input, op } => match op {
190            BoundVectorOp::TopK { .. }
191            | BoundVectorOp::LimitK { .. }
192            | BoundVectorOp::LimitRatio { .. }
193            | BoundVectorOp::Sort(_) => fused_output_tags(input, source_tags),
194            // `absent` reports the labels its own matchers name, not the input's set and not the
195            // table's columns: it answers one series carrying exactly those, so a label the input
196            // does not carry is still a label of the result.
197            BoundVectorOp::Absent { labels } => Some(
198                labels
199                    .tags()
200                    .iter()
201                    .map(|label| label.name.to_string())
202                    .collect(),
203            ),
204            // `count_values` writes the value it counts into its own label and then keeps its
205            // grouping: `by(...)` keeps the labels named plus that destination, `without(...)`
206            // keeps the rest minus the labels named and the metric name. What the node reports is
207            // therefore the grouping applied to the labels its input carries, with the destination
208            // among them - not the input's set, and not the scan's, since the grouping narrows both.
209            //
210            // The destination is a label the node creates, so a rewrite of it above this node is a
211            // rewrite of an existing label. A destination naming the metric name writes the name
212            // the node groups by, which is the one label its result carries. This walk derives the
213            // labels; it does not decide which groupings the kernel accepts, and those refusals stay
214            // where they are.
215            BoundVectorOp::CountValues { label, group } if !label.is_empty() => {
216                let mut tags = fused_output_tags(input, source_tags)?;
217                if !tags.contains(&label.to_string()) {
218                    tags.push(label.to_string());
219                }
220                Some(match group {
221                    Grouping::By(names) => {
222                        let mut output = tags
223                            .into_iter()
224                            .filter(|tag| names.iter().any(|name| name.as_ref() == tag))
225                            .collect::<Vec<_>>();
226                        // `by` keeps the label the node writes whether the input carried it or not.
227                        if !output.contains(&label.to_string()) {
228                            output.push(label.to_string());
229                        }
230                        output
231                    }
232                    Grouping::Without(names) => tags
233                        .into_iter()
234                        .filter(|tag| {
235                            tag != METRIC_NAME && !names.iter().any(|name| name.as_ref() == tag)
236                        })
237                        .collect(),
238                })
239            }
240            BoundVectorOp::CountValues { .. } => None,
241            // A classic histogram function folds the bucket series that share the other labels into
242            // one series and drops `le`, along with the metadata labels the runtime does not put on
243            // its result. What it keeps is everything else its input carried.
244            BoundVectorOp::Histogram(_) => {
245                let mut tags = fused_output_tags(input, source_tags)?;
246                tags.retain(|tag| {
247                    tag != LE_LABEL && !PROMQL_METADATA_LABELS.contains(&tag.as_str())
248                });
249                Some(tags)
250            }
251        },
252        NativeExpr::Merge { input } => fused_output_tags(input, source_tags),
253        NativeExpr::Binary {
254            left,
255            right,
256            matching,
257        } => {
258            let keeps_name = keeps_metric_name(matching);
259            let scalar_side = matches!(&left.output, NativeType::Scalar(_))
260                || matches!(&right.output, NativeType::Scalar(_));
261            let tags = match (&left.output, &right.output) {
262                (NativeType::Scalar(_), _) => fused_output_tags(right, source_tags)?,
263                (_, NativeType::Scalar(_)) => fused_output_tags(left, source_tags)?,
264                _ => {
265                    let kept = |tags: Vec<String>| -> Vec<String> {
266                        match &matching.labels {
267                            MatchingLabels::On(names) => tags
268                                .into_iter()
269                                .filter(|tag| {
270                                    names.iter().any(|name| name.as_ref() == tag.as_str())
271                                })
272                                .collect(),
273                            MatchingLabels::Ignoring(names) => tags
274                                .into_iter()
275                                .filter(|tag| {
276                                    !names.iter().any(|name| name.as_ref() == tag.as_str())
277                                })
278                                .collect(),
279                        }
280                    };
281                    let left_tags = fused_output_tags(left, source_tags)?;
282                    let right_tags = fused_output_tags(right, source_tags)?;
283                    // One-to-one matching narrows the left side's labels with `on`/`ignoring`.
284                    // A group retains the many side's labels and copies the named labels from the one
285                    // side: `group_left(container)` keeps the left labels and adds `container` from
286                    // the right.
287                    let mut tags = match &matching.cardinality {
288                        Cardinality::OneToOne => kept(left_tags.clone()),
289                        Cardinality::ManyToOne(_) => left_tags.clone(),
290                        Cardinality::OneToMany(_) => right_tags.clone(),
291                    };
292                    // `result_metric` drops the metadata labels before it matches or copies an
293                    // included label, so an arithmetic result starts without the name. An include can
294                    // still put it back from the one side, which is why the final rule below does not
295                    // run on this branch.
296                    if !keeps_name {
297                        tags.retain(|tag| tag != METRIC_NAME);
298                    }
299                    let (includes, include_tags): (&[Arc<str>], &Vec<String>) =
300                        match &matching.cardinality {
301                            Cardinality::OneToOne => (&[], &left_tags),
302                            Cardinality::ManyToOne(labels) => (labels, &right_tags),
303                            Cardinality::OneToMany(labels) => (labels, &left_tags),
304                        };
305                    for tag in include_tags {
306                        if includes.iter().any(|label| label.as_ref() == tag.as_str())
307                            && !tags.contains(tag)
308                        {
309                            tags.push(tag.clone());
310                        }
311                    }
312                    // A grouped result carries an include label only when a row can give it one.
313                    // `result_metric` removes the label from the many side's labels unless the one side
314                    // has it, so a paired row and a row filed from a one-side series take it from the
315                    // one series, which is what the loop above declares. A row filed from a many-side
316                    // series takes it from that series' match key instead, so it carries the label only
317                    // when the matching keeps the label in the key. Everything else the many side
318                    // carries is the base above, so a label neither of those can provide has to leave
319                    // the set: nothing would ever fill its column.
320                    if !includes.is_empty() {
321                        let many_rows_filled = match &matching.cardinality {
322                            // `group_right` puts the one side on the left, so the fills trade places:
323                            // `fill_left` is the one that answers a missing one-side sample there.
324                            Cardinality::OneToMany(_) => matching.fill.lhs,
325                            Cardinality::ManyToOne(_) | Cardinality::OneToOne => matching.fill.rhs,
326                        }
327                        .is_some();
328                        let in_match_key = |label: &str| match &matching.labels {
329                            MatchingLabels::On(names) => {
330                                names.iter().any(|name| name.as_ref() == label)
331                            }
332                            // A match key is built from a series' tags, which never hold the metric name.
333                            MatchingLabels::Ignoring(names) => {
334                                label != "__name__"
335                                    && !names.iter().any(|name| name.as_ref() == label)
336                            }
337                        };
338                        tags.retain(|tag| {
339                            !includes.iter().any(|label| label.as_ref() == tag.as_str())
340                                || include_tags.contains(tag)
341                                || (many_rows_filled && in_match_key(tag))
342                        });
343                    }
344                    // A fill answers a missing sample from the series of the other side, and the kernel
345                    // files that row at the label set the series it has gives it
346                    // (`vector_match::binary_field`): a many-side series keeps all of its labels, which
347                    // `tags` already holds, while a one-side series keeps only its matching labels plus
348                    // the group labels the modifier names, which the loop above adds. So what a fill can
349                    // add here is the one side's matching labels, and only a fill that can file a
350                    // one-side row has any to add: one where the many side is the missing side. Which
351                    // fill that is follows the kernel, which trades the two for `group_right`, whose one
352                    // side is the left - there it is `fill_right` that leaves the many side missing.
353                    let (one_tags, fills_many_side) = match &matching.cardinality {
354                        Cardinality::OneToMany(_) => (&left_tags, matching.fill.rhs),
355                        Cardinality::ManyToOne(_) | Cardinality::OneToOne => {
356                            (&right_tags, matching.fill.lhs)
357                        }
358                    };
359                    // A filled row keeps only the one side's matching labels, so the name can reach
360                    // it only through an `on` key that names the metric name, and only while the
361                    // operation reports its samples' names.
362                    let fill_carries_name = keeps_name
363                        && matches!(&matching.labels, MatchingLabels::On(names)
364                            if names.iter().any(|name| name.as_ref() == METRIC_NAME));
365                    if fills_many_side.is_some() {
366                        for tag in kept(one_tags.clone()) {
367                            if tag == METRIC_NAME && !fill_carries_name {
368                                continue;
369                            }
370                            if !tags.contains(&tag) {
371                                tags.push(tag);
372                            }
373                        }
374                    }
375                    tags
376                }
377            };
378            Some(if scalar_side && !keeps_name {
379                without_metric_name(tags)
380            } else {
381                tags
382            })
383        }
384        NativeExpr::Set { left, right, op } => {
385            let mut tags = fused_output_tags(left, source_tags)?;
386            if op.op == SetOp::Or {
387                for tag in fused_output_tags(right, source_tags)? {
388                    if !tags.contains(&tag) {
389                        tags.push(tag);
390                    }
391                }
392                // `or` reports the series of both sides, and each row keeps the name of the side it
393                // came from, so the name is a label of the result even when neither source declares
394                // it as a tag column. A side whose kernel dropped the name leaves the column empty.
395                if !tags.iter().any(|tag| tag == METRIC_NAME) {
396                    tags.push(METRIC_NAME.to_string());
397                }
398            }
399            Some(tags)
400        }
401        // A base series keeps its labels and takes the data labels of its info series: the ones the
402        // selector names, or all of them when it names none.
403        NativeExpr::Info {
404            base,
405            info,
406            selector,
407        } => {
408            let mut tags = fused_output_tags(base, source_tags)?;
409            for tag in fused_output_tags(info, source_tags)? {
410                let copied = selector.data_labels.is_empty()
411                    || selector
412                        .data_labels
413                        .iter()
414                        .any(|label| label.as_ref() == tag.as_str());
415                if copied && !tags.contains(&tag) {
416                    tags.push(tag);
417                }
418            }
419            Some(tags)
420        }
421    }
422}
423/// Whether a binary operation reports the samples of the side it keeps, name included.
424///
425/// A filtering comparison reports those samples, so the name survives it. Every other binary
426/// operation computes new values, and `bool` reports the comparison's own outcome rather than
427/// either operand's samples, so both drop the name.
428fn keeps_metric_name(matching: &BinaryMatching) -> bool {
429    matching.op.is_comparison() && !matching.return_bool
430}

432/// The labels a node reports after the metric name is dropped, which is what a kernel that computes
433/// a value of its own does to the samples it reports.
434fn without_metric_name(tags: Vec<String>) -> Vec<String> {
435    tags.into_iter().filter(|tag| tag != METRIC_NAME).collect()
436}

438/// What one node produces.
439///
440/// A sample type with an evaluation grid is an evaluated series: one value per grid point. Without
441/// a grid it is raw samples carrying their own timestamps. A scalar is one value per grid point
442/// and no labels. Intermediate aggregate state is a different type altogether, so an operation
443/// that cannot consume it is rejected by type rather than by convention.
444#[derive(Clone, Debug, serde::Serialize, serde::Deserialize, PartialEq, Hash, Eq)]
445pub enum NativeType {
446    /// Samples, evaluated when the schema carries a grid and raw otherwise.
447    Samples(Arc<SeriesSchema>),
448    /// One value per grid point and no labels, which binary operations broadcast.
449    ///
450    /// PromQL scalars vary with the evaluation time (`time()`, `scalar(v)`), so this carries the
451    /// grid it is evaluated on. Its value carrier is a series without labels, but its type is
452    /// what keeps a broadcast from becoming label matching.
453    Scalar(Arc<EvalGrid>),
454    /// Intermediate aggregate state, not a query value.
455    AggregateState(Arc<AggregateStateSchema>),
456}
457
458impl NativeType {
459    /// The schema of a sample type, or `None` for a scalar or aggregate state.
460    pub fn samples(&self) -> Option<&Arc<SeriesSchema>> {
461        match self {
462            Self::Samples(schema) => Some(schema),
463            Self::Scalar(_) | Self::AggregateState(_) => None,
464        }
465    }
466
467    /// The grid this type is evaluated on, or `None` when it holds raw samples or state.
468    ///
469    /// A scalar is evaluated on its grid like a vector, which is what lets a binary operation
470    /// line it up with a vector point by point; only raw samples and state have no positions.
471    pub fn grid(&self) -> Option<&Arc<EvalGrid>> {
472        match self {
473            Self::Samples(schema) => schema.eval_grid.as_ref(),
474            Self::Scalar(grid) => Some(grid),
475            Self::AggregateState(_) => None,
476        }
477    }
478
479    /// What this type is, for a rejection that has to say what a consumer was handed.
480    fn describe(&self) -> &'static str {
481        match self {
482            Self::Samples(schema) if schema.eval_grid.is_some() => "an evaluated vector",
483            Self::Samples(_) => "raw samples",
484            Self::Scalar(_) => "a scalar",
485            Self::AggregateState(_) => "aggregate state",
486        }
487    }
488
489    /// Whether `self` binds the same thing as `other`: the same samples or the same state.
490    ///
491    /// Bindings compare their semantic content, not process-local allocation identity. Scalars are
492    /// produced by a node and never handed to one by a source, so they never match.
493    fn same_binding(&self, other: &NativeType) -> bool {
494        match (self, other) {
495            (Self::Samples(mine), Self::Samples(theirs)) => **mine == **theirs,
496            (Self::AggregateState(mine), Self::AggregateState(theirs)) => **mine == **theirs,
497            _ => false,
498        }
499    }
500}

502/// The [`FieldKind`] every referenced field of a scan carries, or `None` when they disagree or one
503/// is a type this route does not answer.
504///
505/// The route declares one output column per field, so a statement whose selectors carry different
506/// kinds has no column to declare them in and keeps the existing plan. `Float64` and the canonical
507/// native histogram struct are the two kinds this route answers.
508pub fn uniform_field_kind<'a>(
509    kinds: impl IntoIterator<Item = Option<&'a DataType>>,
510) -> Option<FieldKind> {
511    let mut answer = None;
512    for kind in kinds {
513        let this = match kind? {
514            DataType::Float64 => FieldKind::Float,
515            data_type if data_type == &native_histogram_arrow_type() => FieldKind::Histogram,
516            _ => return None,
517        };
518        match answer {
519            None => answer = Some(this),
520            Some(seen) if seen == this => {}
521            Some(_) => return None,
522        }
523    }
524    answer
525}

527/// The output kind of a binary operation over two evaluated vectors, including fill pairs.
528pub fn binary_output_field_kind(
529    left: FieldKind,
530    right: FieldKind,
531    op: BinaryOp,
532    return_bool: bool,
533    fill: FillValues,
534) -> Option<FieldKind> {
535    if return_bool {
536        return Some(FieldKind::Float);
537    }
538    match op {
539        BinaryOp::Add
540        | BinaryOp::Sub
541        | BinaryOp::Mul
542        | BinaryOp::Div
543        | BinaryOp::Mod
544        | BinaryOp::Pow
545        | BinaryOp::Atan2 => {
546            // An unsupported pair emits no sample, not a float. Fill values are floats, so one
547            // field can be produced by the real pair and by either enabled synthetic pair.
548            let payload_kind = |left, right| match (left, op, right) {
549                (FieldKind::Float, _, FieldKind::Float) => Some(FieldKind::Float),
550                (FieldKind::Histogram, BinaryOp::Add | BinaryOp::Sub, FieldKind::Histogram)
551                | (FieldKind::Histogram, BinaryOp::Mul | BinaryOp::Div, FieldKind::Float)
552                | (FieldKind::Float, BinaryOp::Mul, FieldKind::Histogram) => {
553                    Some(FieldKind::Histogram)
554                }
555                _ => None,
556            };
557            let candidates = [
558                payload_kind(left, right),
559                fill.rhs.and_then(|_| payload_kind(left, FieldKind::Float)),
560                fill.lhs.and_then(|_| payload_kind(FieldKind::Float, right)),
561            ];
562            let mut kind = None;
563            for candidate in candidates.into_iter().flatten() {
564                if kind.is_some_and(|existing| existing != candidate) {
565                    return None;
566                }
567                kind = Some(candidate);
568            }
569            // With no possible samples, retain the legacy float layout for an empty result.
570            Some(kind.unwrap_or(FieldKind::Float))
571        }
572        // Comparison and bool behavior is separate from arithmetic's empty-field layout.
573        _ if left == right => Some(left),
574        _ => None,
575    }
576}

578impl NativeNode {
579    /// The kind this node's output field carries, given the kind its sources carry.
580    ///
581    /// `None` means this walk cannot name it: either the caller has no kind for the statement's
582    /// sources and this tree reads one, or the shape's samples are ones the model does not describe.
583    /// Both keep the existing plan rather than declaring a column the first batch might not fill.
584    pub fn output_field_kind(&self, input: Option<FieldKind>) -> Option<FieldKind> {
585        let kind_of = |node: &NativeNode| node.output_field_kind(input);
586        match &self.expr {
587            NativeExpr::Input(_) => input,
588            // A tree that reads no source answers floats: a constant, the evaluation time, `vector()`
589            // over one of them, and a date function of the evaluation time. None of them carries a
590            // sample kind of its own, so none of them needs the source kind either.
591            NativeExpr::Scalar(_)
592            | NativeExpr::EvalTime { .. }
593            | NativeExpr::VectorOf { .. }
594            | NativeExpr::Timestamp { .. } => Some(FieldKind::Float),
595            // The window functions that answer with a histogram are the ones the range kernel says
596            // yield one; every other range function reduces its window to a float.
597            NativeExpr::Window { input, op } => {
598                let inner = kind_of(input)?;
599                (!op.function.yields_histograms())
600                    .then_some(FieldKind::Float)
601                    .or(Some(inner))
602            }
603            // An instant selection reads one sample of the field, so it keeps the kind; the two
604            // timestamp forms replace the sample with a float of its own time.
605            NativeExpr::Instant { input, op } => match op.value {
606                InstantValue::Sample => kind_of(input),
607                InstantValue::Timestamp | InstantValue::InfoTimestamp => Some(FieldKind::Float),
608            },
609            // `histogram_count` and the other native-histogram readers answer with one float per
610            // sample; the map kernels that keep the samples keep their kind as well.
611            NativeExpr::Map { input, op } => {
612                let mut kind = kind_of(input)?;
613                for op in op.ops.iter() {
614                    kind = match op {
615                        // The native-histogram readers answer one float per sample.
616                        MapOp::Histogram(_) => FieldKind::Float,
617                        // A label rewrite writes labels only, so the samples it passes on are the
618                        // ones it was given, kind included.
619                        MapOp::Label(_) => kind,
620                        // No math function keeps a histogram, so a histogram reaching one is a
621                        // shape this walk does not name rather than a silent float.
622                        MapOp::Math(_) if kind == FieldKind::Histogram => return None,
623                        MapOp::Math(_) => FieldKind::Float,
624                    };
625                }
626                Some(kind)
627            }
628            // A constant beside a vector is broadcast over it, so the operation answers with the
629            // vector's own samples: a histogram times a float is still a histogram. Two vectors
630            // use the arithmetic result's kind, including any samples emitted by fill, and an
631            // empty float field only when no possible pairing can emit a sample.
632            NativeExpr::Binary {
633                left,
634                right,
635                matching,
636            } => {
637                // `bool` makes a comparison answer 1 or 0 whatever its operands hold, which is
638                // what the kernel does before it looks at the kinds.
639                if matching.return_bool {
640                    return Some(FieldKind::Float);
641                }
642                match (&left.expr, &right.expr) {
643                    (NativeExpr::Scalar(_), NativeExpr::Scalar(_)) => Some(FieldKind::Float),
644                    (NativeExpr::Scalar(_), _) => kind_of(right),
645                    (_, NativeExpr::Scalar(_)) => kind_of(left),
646                    (_, _) => {
647                        let left_kind = kind_of(left)?;
648                        let right_kind = kind_of(right)?;
649                        binary_output_field_kind(
650                            left_kind,
651                            right_kind,
652                            matching.op,
653                            false,
654                            matching.fill,
655                        )
656                    }
657                }
658            }
659            // A canonical merge collects pieces of one label set: it assembles the samples it was
660            // given and never converts them, so the field keeps the kind it had.
661            NativeExpr::Merge { input } => kind_of(input),
662            // A set operation keeps the payloads it selects - `and` and `unless` keep the left
663            // samples, `or` keeps the left's and the right's unmatched ones - so a pair of
664            // different kinds would answer a field carrying both. This route declares one type per
665            // field, so it names the answer only when both sides agree. That is this planner's
666            // condition for a uniformly typed column, not a rule Prometheus states.
667            NativeExpr::Set { left, right, .. } => {
668                let left_kind = kind_of(left)?;
669                (left_kind == kind_of(right)?).then_some(left_kind)
670            }
671            // `scalar()` answers one float per grid point, whatever its input holds.
672            NativeExpr::ScalarOf { .. } => Some(FieldKind::Float),
673            // A ranking hands on the series it selected, so the field keeps the kind it read; its
674            // parameter is a scalar, which carries no samples of its own.
675            NativeExpr::ParameterizedTopK { input, .. } => kind_of(input),
676            // `histogram_quantile` and `histogram_fraction` reduce a histogram to one float per
677            // sample whatever their arguments are, constant or per point.
678            NativeExpr::ParameterizedHistogram { .. } => Some(FieldKind::Float),
679            // An aggregate answers the kind its own state carries: `count`/`group` answer a count
680            // or a `1` whatever the samples they counted, `min`/`max`/`stddev`/`stdvar`/`quantile`
681            // fold the float samples and skip the histogram ones, and `sum`/`avg` answer the
682            // payload they folded - a native histogram when their binding is a histogram state, a
683            // float otherwise. Naming it here is what lets `sum(count_over_time(h[5m]))` - a float
684            // built from histograms - declare a float column, and `sum(h)` declare a histogram one.
685            NativeExpr::Aggregate { op, .. } => Some(aggregate_output_kind(op)),
686            _ => None,
687        }
688    }
689}

694/// differ in both directions: `count` over a histogram answers a float, and so does every aggregate

721/// Returns the state type implied by an aggregate over an input of `field_kind`, if that state can
722/// be merged across a partition boundary.
723///
724/// Two families share this table. A numeric input keeps the components the accumulator kernel
725/// declares, and the component list comes from that kernel rather than a second mapping, so changing
726/// a kernel's state fields changes the native binding in lockstep. A native-histogram input binds
727/// the histogram state of the same aggregate: it carries the payload and the compensation that
728/// belongs to it, plus the evidence a merge of two partitions has to report. A quantile has no
729/// partial kernel at all, whatever it reads.
730pub fn aggregate_state_binding(
731    kind: AggregateKind,
732    field_kind: FieldKind,
733) -> Option<(AggregateStateKind, Vec<StateComponent>)> {
734    if field_kind == FieldKind::Histogram {
735        let state_kind = match kind {
736            AggregateKind::Sum => AggregateStateKind::HistogramSum,
737            AggregateKind::Avg => AggregateStateKind::HistogramAvg,
738            // Every other aggregate folds a float answer out of the samples it reads - a count is a
739            // count whatever it counted, and the extreme and deviation kinds skip a histogram
740            // sample - so a histogram input keeps their numeric state.
741            _ => return float_state_binding(kind),
742        };
743        // The layout is the kind's own canonical list, which is the list its frame accepts.
744        return Some((state_kind, state_kind.components().to_vec()));
745    }
746    float_state_binding(kind)
747}

749/// The numeric state one aggregate keeps, from the components its kernel declares.
750fn float_state_binding(kind: AggregateKind) -> Option<(AggregateStateKind, Vec<StateComponent>)> {
751    use crate::aggregate::{AggregateOp, StateField};
752
753    let (state_kind, op) = match kind {
754        AggregateKind::Sum => (AggregateStateKind::Sum, AggregateOp::Sum),
755        AggregateKind::Count => (AggregateStateKind::Count, AggregateOp::Count),
756        AggregateKind::Avg => (AggregateStateKind::Avg, AggregateOp::Avg),
757        AggregateKind::Min => (AggregateStateKind::Min, AggregateOp::Min),
758        AggregateKind::Max => (AggregateStateKind::Max, AggregateOp::Max),
759        AggregateKind::Group => (AggregateStateKind::Group, AggregateOp::Group),
760        AggregateKind::Stddev => (AggregateStateKind::Stddev, AggregateOp::Stddev),
761        AggregateKind::Stdvar => (AggregateStateKind::Stdvar, AggregateOp::Stdvar),
762        AggregateKind::Quantile { .. } => return None,
763    };
764    let components = op
765        .state_fields()
766        .iter()
767        .map(|field| match field {
768            StateField::Value => StateComponent::Value,
769            StateField::Mean => StateComponent::Mean,
770            StateField::Compensation => StateComponent::Compensation,
771            StateField::Count => StateComponent::Count,
772            StateField::IncrementalMean => StateComponent::IncrementalMean,
773        })
774        .collect();
775    Some((state_kind, components))
776}

1305/// Whether this tree can hand over two series that share one canonical label set.
1306///
1307/// A canonical label set is the identity a [`NativeExpr::Merge`] groups by: the metric name plus the
1308/// ordinary labels, which is exactly what [`SeriesLabels`] holds. One scan partition holds one
1309/// series per such set - a stored series is one identity, and the scan splits rows by storage key
1310/// rather than by label set - so only a computation that *writes* an existing label can turn two of
1311/// those identities into one. A `label_replace` or `label_join` does exactly that, and it may name
1312/// the very label that told its input's series apart.
1313///
1314/// The rewrite is judged by the labels it writes and the ones its input carries, which is why the
1315/// caller supplies what it knows about each source's tags, indexed by the `source_index` the leaves
1316/// were bound to: a destination that names no ordinary label of the input writes a fresh one, and a
1317/// fresh ordinary label cannot make two series equal. That is the difference between
1318/// `label_replace(x, "dst", "$1", "host", "(.*)")` - which no merge has to collect - and
1319/// `label_replace(x, "host", "same", "host", ".*")`, which folds every series of `x` into one label
1320/// set.
1321///
1322/// A source's entry is its complete runtime label set, so `__name__` is one of its labels whenever
1323/// its series carry a name - a scan bound to one metric names every series from its binding. That is
1324/// what makes `label_replace(x, "__name__", "same", "__name__", ".*")` a rewrite of an existing
1325/// label: two metrics whose ordinary labels are equal can be told apart by name alone, and rewriting
1326/// that name folds them into one identity. A name write is collected even when the derived set no
1327/// longer carries `__name__` - after `abs`, a `timestamp` or a range function dropped it - because
1328/// the scan below those kernels can still hold several metrics whose ordinary labels are equal, and
1329/// writing one name puts them under one identity; the walk has no proof that the input reads a
1330/// single metric, so the conservative answer is the one that collects.
1331///
1332/// Everything else either keeps the set it read or answers a set of its own:
1333///
1334/// - A map of math drops the metric name, and an arithmetic broadcast does too, but within one
1335///   metric the remaining tags are the primary key: two stored series of one metric that agree on
1336///   every other label are the same series, so dropping the name cannot make them one. Each of
1337///   those nodes hands its input's answer on rather than claiming one of its own.
1338/// - An `Aggregate` answers its groups, a `Vector` that folds series answers its result sets, and a
1339///   `Binary` that reads whole vectors files its rows by the label set a sample *ends* under and
1340///   refuses two values of a field at one point. A `Set` does the same, and `info` merges the
1341///   results that share a label set. Each answers one series per set it names.
1342/// - A node that hands its input's series over unchanged - a selection, a window, a timestamp, a
1343///   parameterized map, `topk` and its ranking siblings - passes the answer on: an alias below it is
1344///   still an alias above it.
1345///
1346/// This is the fact [`merge_label_rewrites`](crate::native::canonicalize::merge_label_rewrites)
1347/// inserts a `Merge` for: a consumer that reads a whole label set cannot answer from pieces, and this
1348/// says where pieces of one set can exist at all. A `Merge` is the collection itself, so it answers
1349/// `false` - which is what makes the pass idempotent.
1350pub fn duplicates_canonical_labels(node: &NativeNode, source_tags: &[Vec<String>]) -> bool {
1351    match &node.expr {
1352        NativeExpr::Input(_)
1353        | NativeExpr::Scalar(_)
1354        | NativeExpr::EvalTime { .. }
1355        | NativeExpr::VectorOf { .. } => false,
1356        // The collection itself: one series per label set comes out of it.
1357        NativeExpr::Merge { .. } => false,
1358        // These hand each series over as they read it, so what aliases below them still aliases.
1359        NativeExpr::Instant { input, .. }
1360        | NativeExpr::Window { input, .. }
1361        | NativeExpr::Timestamp { input }
1362        | NativeExpr::ParameterizedMap { input, .. }
1363        | NativeExpr::ParameterizedTopK { input, .. } => {
1364            duplicates_canonical_labels(input, source_tags)
1365        }
1366        // `scalar()` answers one value, which names no label set at all.
1367        NativeExpr::ScalarOf { .. } => false,
1368        // The histogram functions fold the bucket series of one histogram into one series, so two
1369        // pieces of one label set cannot survive them.
1370        NativeExpr::ParameterizedHistogram { .. } => false,
1371        // A rewrite writes a caller-chosen destination label, so two series can end under one set -
1372        // but only if that label was one of the input's, because writing a label no series carries
1373        // leaves every series' identity distinct. `map_writes_existing_label` is that question, and
1374        // an input whose labels this walk cannot derive is collected rather than trusted. Even a
1375        // whole-vector consumer can emit distinct series whose labels a later rewrite makes equal.
1376        NativeExpr::Map { input, op } => {
1377            map_writes_existing_label(input, op, source_tags).unwrap_or(true)
1378                || duplicates_canonical_labels(input, source_tags)
1379        }
1380        // A fold answers one series per group, which is one label set per group.
1381        NativeExpr::Aggregate { .. } => false,
1382        NativeExpr::Vector { input, op } => match op {
1383            // These keep and order the series they read, so they alias exactly as their input does.
1384            BoundVectorOp::TopK { .. }
1385            | BoundVectorOp::LimitK { .. }
1386            | BoundVectorOp::LimitRatio { .. }
1387            | BoundVectorOp::Sort(_) => duplicates_canonical_labels(input, source_tags),
1388            // `absent` reports one series, `count_values` one per distinct value, and the histogram
1389            // functions fold the bucket series of one histogram into one series.
1390            BoundVectorOp::Absent { .. }
1391            | BoundVectorOp::Histogram(_)
1392            | BoundVectorOp::CountValues { .. } => false,
1393        },
1394        NativeExpr::Binary { left, right, .. } => {
1395            let vector = |side: &NativeNode| side.output.samples().is_some();
1396            if vector(left) && vector(right) {
1397                // Two vectors are matched by label, and the kernel files its rows by the label set a
1398                // sample ends under, refusing two values of one field at one point.
1399                false
1400            } else {
1401                // A broadcast reaches each series of its vector side as it is.
1402                let carried = if vector(left) { left } else { right };
1403                duplicates_canonical_labels(carried, source_tags)
1404            }
1405        }
1406        // A set operation keeps each side's samples and files them by result label set, refusing two
1407        // values of one field at one point.
1408        NativeExpr::Set { .. } => false,
1409        // `info` indexes the info side by identifying labels and merges the results that share a
1410        // label set.
1411        NativeExpr::Info { .. } => false,
1412    }
1413}

1415/// Whether `op`, applied to `input`'s series, writes a label those series already carry.
1416///
1417/// The walk is over the ops in written order, starting from the labels the input reports. A
1418/// destination the input already carries can fold two series into one; a fresh destination cannot.
1419/// A `__name__` destination is always treated as carried, even when the derived set has lost it: a
1420/// kernel such as `abs`, `timestamp` or a range function drops the name, but the scan below it can
1421/// still hold several metrics whose ordinary labels are equal, and writing one name puts those
1422/// identities under one label set. Nothing here proves the input reads a single metric, so the
1423/// conservative answer is the one that collects.
1424///
1425/// `None` when the input's label set is not derivable, which is treated as "it may carry the
1426/// destination" by the caller: an unknown set proves nothing, so the conservative answer is the one
1427/// that collects.
1428fn map_writes_existing_label(
1429    input: &NativeNode,
1430    op: &crate::native::plan::BoundMap,
1431    source_tags: &[Vec<String>],
1432) -> Option<bool> {
1433    let mut tags = fused_output_tags(input, source_tags)?;
1434    let mut existing = false;
1435    for operation in op.ops.iter() {
1436        if let MapOp::Label(rewrite) = operation {
1437            let destination = rewrite.destination();
1438            if destination == METRIC_NAME || tags.iter().any(|tag| tag == destination) {
1439                existing = true;
1440            }
1441            // A label the input did not carry is one the rewrite publishes only when it can write a
1442            // value; a delete-only rewrite adds nothing a later op could read as carried.
1443            if rewrite.writes_value() && !tags.iter().any(|tag| tag == destination) {
1444                tags.push(destination.to_string());
1445            }
1446        }
1447    }
1448    Some(existing)
1449}
1450/// Whether `node` reads the whole input vector at a point rather than one series at a time.
1451///
1452/// An `Aggregate` folds every series of its group, a `Binary` and a `Set` pair two vectors' series
1453/// by label, an `Info` looks the info series of each series up by its identifying labels, and a
1454/// `Vector` operation (`sort`, `absent`, `topk`, `limitk`, `histogram_quantile`, ...) reorders or
1455/// reads the whole vector at a grid point.
1456///
1457/// `Merge` answers one series per label set, and the label sets it answers span every input series:
1458/// two of them can collapse into the one it emits. It is therefore whole-vector too, which is why it
1459/// also cannot be moved below a boundary that holds only part of a metric. Everything else - an
1460/// input, a scalar, a selection, a window, a map, and a binary operation with a constant on one
1461/// side, whose broadcast reaches each series the other side answers - answers one series from one
1462/// series' rows.
1463///
1464/// Both a region-side placement and the native route's admission ask this, and they must agree: a
1465/// tree admitted here but refused there becomes a node the physical layer cannot build, and one
1466/// refused here but admitted there would be pushed to a region that holds part of the metric.
1467pub fn reads_across_series(node: &NativeNode) -> bool {
1468    match &node.expr {
1469        NativeExpr::Input(_) | NativeExpr::Scalar(_) | NativeExpr::EvalTime { .. } => false,
1470        NativeExpr::Window { input, .. } => {
1471            matches!(unifies_its_own_contributions(input), Some(false))
1472                || reads_across_series(input)
1473        }
1474        NativeExpr::Instant { input, .. }
1475        | NativeExpr::VectorOf { input }
1476        | NativeExpr::Map { input, .. }
1477        | NativeExpr::Timestamp { input } => reads_across_series(input),
1478        // A parameterized operation reads its parameters too, and one of them can read across series
1479        // as well: `clamp_min(b, scalar(a))` bounds each series of `b` by a value that only exists
1480        // once the whole of `a` is gathered, so the node reads across series wherever its parameter
1481        // does.
1482        NativeExpr::ParameterizedMap {
1483            input, parameters, ..
1484        } => reads_across_series(input) || parameters.iter().any(reads_across_series),
1485        // `scalar()` needs the whole vector: which series carries the one value at a point decides
1486        // the answer, so the node reads across series.
1487        NativeExpr::ScalarOf { .. } => true,
1488        NativeExpr::Binary { left, right, .. } => {
1489            let scalar = |side: &NativeNode| matches!(side.output, NativeType::Scalar(_));
1490            if scalar(left) || scalar(right) {
1491                reads_across_series(left) || reads_across_series(right)
1492            } else {
1493                // Two vectors are matched by label, which pairs series of the whole vector.
1494                true
1495            }
1496        }
1497        // A parameterized operation reads its parameter too, and a `k` that changes per point
1498        // decides which series survive.
1499        NativeExpr::ParameterizedTopK { .. } | NativeExpr::ParameterizedHistogram { .. } => true,
1500        NativeExpr::Aggregate { .. }
1501        | NativeExpr::Vector { .. }
1502        | NativeExpr::Set { .. }
1503        | NativeExpr::Info { .. }
1504        | NativeExpr::Merge { .. } => true,
1505    }
1506}

1508/// Whether `node` is a supported child of a [`NativeExpr::Merge`], and the evaluated vector it hands
1509/// over.
1510///
1511/// A merge receives fragments and groups them by canonical label set, combining the ones a label set
1512/// shares by grid position; it names neither the tree nor the operator that produced a fragment. What
1513/// it reads is therefore the whole of its requirement: samples evaluated on one grid in one field
1514/// layout, which is an evaluated vector and nothing else. This is that type check and no more, so no
1515/// node kind is refused for what it computes: a scan's samples, a window's rate, an aggregate's fold
1516/// and a binary operation's match all hand over the same thing, and which one of them did says
1517/// nothing about whether the fragments merge.
1518///
1519/// The collection does rely on the shape of the stream - a series arrives whole, its chunks
1520/// contiguous and its last one marked as closing it - and every node that answers an evaluated vector
1521/// keeps it. A window, a map, an instant selection, a timestamp and a scalar broadcast hand their
1522/// input's closing on, and an aggregate, a vector operation, a binary or set operation and a merge
1523/// itself close each series they answer, because each decides a whole series at once. There is hence
1524/// no operator above a merge that could break the collection, and none has to be listed here; the
1525/// collector checks the shape chunk by chunk before it keeps a fragment, which is where a stream that
1526/// broke it is refused.
1527fn merge_shape_supported(node: &NativeNode) -> Option<(&Arc<SeriesSchema>, &Arc<EvalGrid>)> {
1528    evaluated_vector(&node.output)
1529}

1586/// Whether the native runtime executes `tree`, answered from the tree itself.
1587///
1588/// This is the question the native route asks before it commits a statement: a tree the runtime
1589/// refuses has to keep the plan it had, because a refusal raised during physical planning cannot
1590/// fall back. It used to be answered by building a
1591/// [`NativeExecutionState`](crate::native::execute::NativeExecutionState) over the tree and
1592/// dropping it, which allocates a runtime node per node per partition only to throw it away. Every
1593/// term of the answer is bound on the tree, so the answer is read from the tree instead.
1594///
1595/// # What is refused
1596///
1597/// This lists what `RuntimeNode::build` (and the root check in
1598/// `NativeExecutionState::try_new`) refuses, less what [`NativeNode::validate`] has already
1599/// refused: a `NativeTree` exists only through [`NativeTree::try_new`] or `Deserialize`, and both
1600/// validate, so a compiled tree's per-node type contracts hold by the time this runs. What is left
1601/// is the shape of the nodes, and the arithmetic each of them leaves to the runtime:
1602///
1603/// - a root that answers raw samples, which has no positions to answer on;
1604/// - an input leaf whose series chunks may interleave, which no consumer of a leaf can order;
1605/// - `scalar()` over a vector carrying more than one field;
1606/// - a binary node that is neither a vector–vector operation nor a broadcast the runtime performs:
1607///   two constant operands (no series to carry the result), a scalar broadcast carrying
1608///   vector-matching modifiers, or `bool` on an arithmetic operator, which has no `bool` form;
1609/// - a `quantile` aggregate at any stage but `Complete`, which would need a partial state it does
1610///   not keep (`validate` refuses one for the same reason; this arm keeps the predicate and the
1611///   runtime agreeing on it);
1612/// - an aggregate below `Final` whose `by` names a value field, which would fold series the
1613///   existing plan keeps apart;
1614/// - `topk`, `bottomk`, `limitk`, `sort` and `sort_desc` over a vector carrying more than one
1615///   field, whose series have no one value to rank;
1616/// - a `topk`, `bottomk`, `limitk` or `count_values` grouping that names a value field, and a
1617///   `count_values` label that is empty;
1618/// - a window whose `anchored` or `smoothed` modifier reads evaluated input, names a function its
1619///   kernel does not correct, or carries a negative lookback;
1620/// - a window whose timestamps do not fit: the fold offset an `@` anchor turns into, or the one
1621///   `predict_linear` adds to a point of the grid, where the sum leaves `i64`. Not a shape, but it
1622///   is decided by the grid and the window alone, so this is where it is decided (see
1623///   `window_instants_fit`).
1624///
1625/// # What `NativeExecutionState::try_new` still decides
1626///
1627/// Not every refusal is decided by the tree, and the ones below stay where the runtime builds:
1628///
1629/// - a reservation the memory pool refuses, which no tree carries. Every site that admits a tree
1630///   ran the trial build over an
1631///   [`UnboundedMemoryPool`](datafusion::execution::memory_pool::UnboundedMemoryPool), which
1632///   refuses nothing, and none of them builds a runtime any more, so this is not a refusal
1633///   admission can meet at any site;
1634/// - a tree a test assembled directly, whose nodes disagree about their types, which
1635///   [`NativeNode::validate`] rejects for every compiled one.
1636pub fn native_tree_supported(tree: &NativeTree) -> bool {
1637    // A constant root and a state root are answers the state produces itself; a root of raw
1638    // samples is not one, because raw samples carry their own timestamps instead of positions.
1639    let root_supported = match &tree.root().output {
1640        NativeType::Scalar(_) | NativeType::AggregateState(_) => true,
1641        NativeType::Samples(schema) => schema.eval_grid.is_some(),
1642    };
1643    root_supported && native_node_supported(tree.root())
1644}
1645
1646/// Whether `node` and every input below it is a shape the runtime builds.
1647fn native_node_supported(node: &NativeNode) -> bool {
1648    stacker::maybe_grow(VALIDATE_STACK_RED_ZONE, VALIDATE_STACK_GROWTH, || {
1649        native_node_supported_within_stack(node)
1650    })
1651}
1652
1653/// [`native_node_supported`] one tree level down.
1654///
1655/// Every arm reads this node's own shape and recurses into exactly the inputs its variant holds,
1656/// so a new [`NativeExpr`] variant does not compile until it is classified here.
1657fn native_node_supported_within_stack(node: &NativeNode) -> bool {
1658    match &node.expr {
1659        NativeExpr::Input(bound) => bound.ordering != SeriesOrdering::Interleaved,
1660        NativeExpr::Scalar(_) | NativeExpr::EvalTime { .. } => true,
1661        NativeExpr::VectorOf { input } | NativeExpr::Instant { input, .. } => {
1662            native_node_supported(input)
1663        }
1664        NativeExpr::Merge { input } | NativeExpr::Map { input, .. } => native_node_supported(input),
1665        NativeExpr::Timestamp { input } => native_node_supported(input),
1666        NativeExpr::Window { input, op } => {
1667            native_node_supported(input)
1668                && window_modifier_supported(input, op)
1669                && window_instants_fit(op)
1670        }
1671        NativeExpr::ParameterizedMap {
1672            input, parameters, ..
1673        } => native_node_supported(input) && parameters.iter().all(native_node_supported),
1674        NativeExpr::ParameterizedHistogram {
1675            input, parameters, ..
1676        } => native_node_supported(input) && parameters.iter().all(native_node_supported),
1677        NativeExpr::Aggregate { input, op } => {
1678            native_node_supported(input) && aggregate_supported(node, input, op)
1679        }
1680        NativeExpr::Vector { input, op } => {
1681            native_node_supported(input)
1682                && node
1683                    .output
1684                    .samples()
1685                    .is_some_and(|schema| vector_op_supported(schema, op))
1686        }
1687        NativeExpr::ParameterizedTopK {
1688            input,
1689            op,
1690            parameter,
1691        } => {
1692            native_node_supported(input)
1693                && native_node_supported(parameter)
1694                // A per-point `k` ranks the same series as the constant one, one value per series.
1695                && node.output.samples().is_some_and(|schema| {
1696                    schema.fields.len() == 1 && vector_op_supported(schema, op)
1697                })
1698        }
1699        NativeExpr::ScalarOf { input } => {
1700            native_node_supported(input)
1701                // `scalar()` answers the sole value of a vector, so a vector of several fields has
1702                // more than one value to answer with.
1703                && input
1704                    .output
1705                    .samples()
1706                    .is_some_and(|schema| schema.fields.len() == 1)
1707        }
1708        NativeExpr::Binary {
1709            left,
1710            right,
1711            matching,
1712        } => {
1713            native_node_supported(left)
1714                && native_node_supported(right)
1715                && binary_supported(left, right, matching)
1716        }
1717        NativeExpr::Set { left, right, .. } => {
1718            native_node_supported(left) && native_node_supported(right)
1719        }
1720        NativeExpr::Info { base, info, .. } => {
1721            native_node_supported(base) && native_node_supported(info)
1722        }
1723    }
1724}

1726/// Whether a binary node over `left` and `right` is a shape the runtime builds.
1727///
1728/// Two evaluated vectors are matched by `matching`, which the kernel takes as it is bound. Every
1729/// other pair is a broadcast over one scalar side, and a broadcast has no labels to match and no
1730/// series to pair: two constants leave no series to carry the result, a modifier a vector match
1731/// would use has no meaning over a scalar, and arithmetic has no `bool` form.
1732fn binary_supported(left: &NativeNode, right: &NativeNode, matching: &BinaryMatching) -> bool {
1733    let evaluated = |node: &NativeNode| {
1734        node.output
1735            .samples()
1736            .is_some_and(|schema| schema.eval_grid.is_some())
1737    };
1738    if evaluated(left) && evaluated(right) {
1739        return true;
1740    }
1741    let literal = |node: &NativeNode| matches!(&node.expr, NativeExpr::Scalar(_));
1742    let broadcastable = match (&left.output, &right.output) {
1743        (NativeType::Samples(vector), NativeType::Scalar(_))
1744        | (NativeType::Scalar(_), NativeType::Samples(vector)) => vector.eval_grid.is_some(),
1745        (NativeType::Scalar(_), NativeType::Scalar(_)) => !(literal(left) && literal(right)),
1746        _ => false,
1747    };
1748    broadcastable
1749        && matching.labels == MatchingLabels::Ignoring(Vec::new())
1750        && matching.cardinality == Cardinality::OneToOne
1751        && matching.fill == FillValues::default()
1752        && !(!matching.op.is_comparison() && matching.return_bool)
1753}

1755/// Whether the runtime folds `op` over `input` with the layout `node` declares.
1756///
1757/// `quantile` has no kernel on this route, so it is refused at every stage. A `Complete` or
1758/// `Partial` node folds values into groups it keys by series labels only, while `Final` merges
1759/// state that was already grouped, which is why only the folding stages are checked against the
1760/// schema.
1761fn aggregate_supported(node: &NativeNode, input: &NativeNode, op: &BoundAggregate) -> bool {
1762    match op.kind {
1763        AggregateKind::Sum
1764        | AggregateKind::Count
1765        | AggregateKind::Avg
1766        | AggregateKind::Min
1767        | AggregateKind::Max
1768        | AggregateKind::Group
1769        | AggregateKind::Stddev
1770        | AggregateKind::Stdvar => {}
1771        AggregateKind::Quantile { .. } => {
1772            // A quantile is one `Complete` fold over its group's whole input: it keeps no state,
1773            // so `validate` refuses a `Partial` or `Final` one and there is nothing to merge. The
1774            // runtime answers the `Complete` stage, which is the only shape compiled.
1775            if op.stage != AggregateStage::Complete {
1776                return false;
1777            }
1778        }
1779    }
1780    if op.stage == AggregateStage::Final {
1781        return true;
1782    }
1783    node.output
1784        .samples()
1785        .or_else(|| input.output.samples())
1786        .is_some_and(|schema| grouping_supported(schema, &op.group))
1787}
1788
1789/// Whether `op` is an operation the vector kernel runs over a vector of `schema`.
1790fn vector_op_supported(schema: &SeriesSchema, op: &BoundVectorOp) -> bool {
1791    // These five rank or order one value per series, so a vector whose samples carry several
1792    // fields has no one value to rank.
1793    let one_field = matches!(
1794        op,
1795        BoundVectorOp::TopK { .. }
1796            | BoundVectorOp::LimitK { .. }
1797            | BoundVectorOp::Sort(SortOrder::Asc | SortOrder::Desc)
1798    );
1799    if one_field && schema.fields.len() != 1 {
1800        return false;
1801    }
1802    match op {
1803        BoundVectorOp::TopK { group, .. } | BoundVectorOp::LimitK { group, .. } => {
1804            grouping_supported(schema, group)
1805        }
1806        BoundVectorOp::CountValues { label, group } => {
1807            let names = group_names(group);
1808            crate::aggregate::validate_count_values(
1809                schema,
1810                label.as_ref(),
1811                kernel_grouping(group, &names),
1812            )
1813            .is_ok()
1814        }
1815        BoundVectorOp::LimitRatio { .. }
1816        | BoundVectorOp::Sort(_)
1817        | BoundVectorOp::Absent { .. }
1818        | BoundVectorOp::Histogram(_) => true,
1819    }
1820}

1822/// Whether the kernel folds `schema`'s series the way `group` asks.
1823///
1824/// The kernel validates its grouping by the schema alone: the group is a set of series labels, and
1825/// a `by` naming one of the schema's fields would group by a value that varies between samples.
1826/// This is the check `FloatAggregator::try_new_budgeted` and `VectorRuntime::try_new` run, spelled
1827/// once so a reader of either name finds the same rule.
1828fn grouping_supported(schema: &SeriesSchema, group: &Grouping) -> bool {
1829    let names = group_names(group);
1830    crate::aggregate::validate_vector_grouping(schema, kernel_grouping(group, &names)).is_ok()
1831}
1832
1833/// The labels `group` names, which is what the kernel's grouping is built over.
1834fn group_names(group: &Grouping) -> Vec<&str> {
1835    match group {
1836        Grouping::By(names) | Grouping::Without(names) => names.iter().map(AsRef::as_ref).collect(),
1837    }
1838}
1839
1840/// `group` spelled with the kernel's own [`Grouping`](crate::aggregate::Grouping), over the labels
1841/// [`group_names`] reports for it: `by` and `without` are not interchangeable to the kernel.
1842fn kernel_grouping<'a>(group: &Grouping, names: &'a [&'a str]) -> crate::aggregate::Grouping<'a> {
1843    match group {
1844        Grouping::By(_) => crate::aggregate::Grouping::By(names),
1845        Grouping::Without(_) => crate::aggregate::Grouping::Without(names),
1846    }
1847}

1849/// Whether the runtime evaluates `op` over `input`, with or without a range modifier.
1850///
1851/// An `anchored` or `smoothed` window reads samples around the window to pin its boundary values,
1852/// which only a raw sample stream can offer and only a kernel that corrects its boundaries reads.
1853/// That is one rule, not two: [`check_modifier`](crate::range_eval::check_modifier) states it for
1854/// [`RangeEvaluator::try_new`](crate::range_eval::RangeEvaluator::try_new), and it is asked here
1855/// over the schema the tree binds, so a modifier a runtime refuses is one admission declines.
1856fn window_modifier_supported(input: &NativeNode, op: &BoundWindow) -> bool {
1857    // Without a modifier there is nothing for the rule to decide.
1858    let Some(modifier) = op.window.modifier else {
1859        return true;
1860    };
1861    input.output.samples().is_some_and(|schema| {
1862        crate::range_eval::check_modifier(schema, modifier, op.function).is_ok()
1863    })
1864}
1865
1866/// Whether the runtime can compute the timestamps `op` reads.
1867///
1868/// Unlike the window itself, which a point of the grid opens alone, these timestamps are sums of
1869/// the window's own parts: an `@` anchor becomes an offset from the grid's first point, and
1870/// `predict_linear` adds that offset to the first and last point of the grid. Every term of either
1871/// sum is bound on the node, but a sum of two that fit can still leave `i64`, and the runtime
1872/// refuses the window when one does. Asking [`fold_offset`] and [`predict_linear_instants`] here
1873/// rather than repeating their arithmetic keeps the two verdicts the same one: a query whose only
1874/// refusal is this one keeps the plan it had instead of failing while the runtime is built.
1875fn window_instants_fit(op: &BoundWindow) -> bool {
1876    let grid = op.output_grid.spec();
1877    let Ok(fold_offset) = fold_offset(grid, op.window) else {
1878        return false;
1879    };
1880    match op.function {
1881        // Only `predict_linear` reads the instant it reports; every other function reads the
1882        // window it opens, which the grid alone decides.
1883        RangeFunction::PredictLinear(_) => predict_linear_instants(grid, fold_offset).is_ok(),
1884        _ => true,
1885    }
1886}

1888/// Stack one [`NativeNode::validate`] level needs below it before the check grows its stack.
1889///
1890/// The check recurses once per tree level, and a debug build's frame for one level is large:
1891/// measured on an 80-term chain, the whole check needed about 1.9 MiB, which overflows the 2 MiB
1892/// stack a test thread has. There is no depth limit on the trees this route compiles, so the check
1893/// grows its stack on demand rather than relying on the caller's.
1894const VALIDATE_STACK_RED_ZONE: usize = 256 * 1024;
1895
1896/// Stack a growth adds when the reserve above is spent. It holds several levels of the same check.
1897const VALIDATE_STACK_GROWTH: usize = 4 * 1024 * 1024;
1898
1899impl NativeNode {
1900    /// Checks this node and its inputs against the contract of its operation.
1901    pub fn validate(&self) -> Result<()> {
1902        stacker::maybe_grow(VALIDATE_STACK_RED_ZONE, VALIDATE_STACK_GROWTH, || {
1903            self.validate_within_stack()
1904        })
1905    }
1906
1907    fn validate_within_stack(&self) -> Result<()> {
1908        match &self.expr {
1909            NativeExpr::Input(input) => {
1910                // An input node's output is exactly the binding it was bound to: a scan hands over
1911                // the samples it was created for, and an exchange boundary the state.
1912                if !self.output.same_binding(&input.output) {
1913                    return Err(invalid(
1914                        "an input node's output is not the binding it was bound to",
1915                    ));
1916                }
1917            }
1918            NativeExpr::Scalar(_) | NativeExpr::EvalTime { .. } => {
1919                if !matches!(self.output, NativeType::Scalar(_)) {
1920                    return Err(invalid("a scalar node does not produce a scalar"));
1921                }
1922            }
1923            NativeExpr::ScalarOf { input } => {
1924                input.validate()?;
1925                let Some((_, input_grid)) = evaluated_vector(&input.output) else {
1926                    return Err(invalid("scalar() consumes an evaluated vector"));
1927                };
1928                let NativeType::Scalar(output_grid) = &self.output else {
1929                    return Err(invalid("scalar() does not produce a scalar"));
1930                };
1931                if output_grid.spec() != input_grid.spec() {
1932                    return Err(invalid("scalar() output is not on its input's grid"));
1933                }
1934            }
1935            NativeExpr::VectorOf { input } => {
1936                input.validate()?;
1937                let NativeType::Scalar(input_grid) = &input.output else {
1938                    return Err(invalid("vector() consumes a scalar"));
1939                };
1940                ensure_samples(&self.output, "vector()")?;
1941                if !self.output.samples().is_some_and(|s| s.fields.len() == 1) {
1942                    return Err(invalid("vector() produces one field"));
1943                }
1944                // The scalar is materialized on the grid it is evaluated on, so an output on another
1945                // grid would address positions the scalar does not have.
1946                ensure_grid(&self.output, input_grid, "vector()")?;
1947            }
1948            NativeExpr::Instant { input, op } => {
1949                input.validate()?;
1950                // An instant selection reads raw samples and stamps them onto its own grid: an
1951                // evaluated vector is already stamped, and a scalar has no timestamps at all.
1952                match &input.output {
1953                    NativeType::Samples(schema) if schema.eval_grid.is_none() => {}
1954                    NativeType::Samples(_) => {
1955                        return Err(invalid("an instant selection reads evaluated values"));
1956                    }
1957                    other => {
1958                        return Err(invalid(&format!(
1959                            "an instant selection consumes raw samples, not {}",
1960                            other.describe()
1961                        )));
1962                    }
1963                }
1964                ensure_not_interleaved(input)?;
1965                ensure_samples(&self.output, "an instant selection")?;
1966                // A selection is one source row per grid point: `select_instants` picks row
1967                // indices and every field is taken at them, so it carries one value per input
1968                // field.
1969                ensure_same_fields(&input.output, &self.output, "an instant selection")?;
1970                ensure_grid(&self.output, &op.output_grid, "an instant selection")?;
1971            }
1972            NativeExpr::Merge { input } => {
1973                input.validate()?;
1974                // A merge canonicalizes evaluated fragments, so an evaluated vector is the whole of
1975                // what it needs of its input: [`merge_shape_supported`] is that check, and its
1976                // documentation says why no shape of tree above those fragments matters.
1977                let Some((_input_schema, input_grid)) = merge_shape_supported(input) else {
1978                    return Err(invalid(&format!(
1979                        "a merge consumes an evaluated vector, not {}",
1980                        input.output.describe()
1981                    )));
1982                };
1983                ensure_samples(&self.output, "a merge")?;
1984                ensure_same_fields(&input.output, &self.output, "a merge")?;
1985                ensure_grid(&self.output, input_grid, "a merge")?;
1986            }
1987            NativeExpr::Window { input, op } => {
1988                input.validate()?;
1989                // A window reads a range of samples per grid point, so it consumes samples: state
1990                // is not a sample stream and a scalar has no timestamps.
1991                let NativeType::Samples(schema) = &input.output else {
1992                    return Err(invalid(&format!(
1993                        "a window consumes samples, not {}",
1994                        input.output.describe()
1995                    )));
1996                };
1997                // A range that is not positive reads nothing at every grid point.
1998                if op.window.range_ms <= 0 {
1999                    return Err(invalid(
2000                        "a window reads an empty range: its range must be positive",
2001                    ));
2002                }
2003                ensure_not_interleaved(input)?;
2004                // What is checked here and just below is that the declared input grid equals the
2005                // input's grid, and that the output grid matches the declared one. That the grid
2006                // covers the window's read range
2007                // `(out.start - offset - range, out.end - offset]` is not checked here: only a
2008                // subquery feeds an evaluated input to a window, and deriving that coverage is the
2009                // compiler's, which builds the subquery's child grid to span it.
2010                match (op.input_grid.as_ref(), schema.eval_grid.as_ref()) {
2011                    (None, None) => {}
2012                    (Some(bound), Some(actual)) if bound.spec() == actual.spec() => {}
2013                    _ => {
2014                        return Err(invalid(
2015                            "a window's declared input grid is not the input's grid",
2016                        ));
2017                    }
2018                }
2019                ensure_samples(&self.output, "a window")?;
2020                ensure_same_fields(&input.output, &self.output, "a window")?;
2021                ensure_grid(&self.output, &op.output_grid, "a window")?;
2022            }
2023            NativeExpr::Map { input, op } => {
2024                input.validate()?;
2025                // A map is point-wise over an evaluated vector: raw samples have no shared
2026                // positions to be point-wise over, and a scalar is not a vector.
2027                let Some((_, input_grid)) = evaluated_vector(&input.output) else {
2028                    return Err(invalid(&format!(
2029                        "a map consumes an evaluated vector, not {}",
2030                        input.output.describe()
2031                    )));
2032                };
2033                if op.ops.is_empty() {
2034                    return Err(invalid("a map holds no operation"));
2035                }
2036                ensure_samples(&self.output, "a map")?;
2037                ensure_same_fields(&input.output, &self.output, "a map")?;
2038                ensure_grid(&self.output, input_grid, "a map")?;
2039            }
2040            NativeExpr::ParameterizedMap {
2041                input,
2042                op,
2043                parameters,
2044            } => {
2045                input.validate()?;
2046                // `max_of` and `min_of` are functions of two scalars, so their input is the first
2047                // of the two rather than a vector's fields; every other function here maps an
2048                // evaluated vector and answers one of the same shape.
2049                let input_grid = if op.takes_a_scalar_input() {
2050                    let NativeType::Scalar(input_grid) = &input.output else {
2051                        return Err(invalid(&format!(
2052                            "{op:?} consumes a scalar, not {}",
2053                            input.output.describe()
2054                        )));
2055                    };
2056                    Arc::clone(input_grid)
2057                } else {
2058                    let Some((_, input_grid)) = evaluated_vector(&input.output) else {
2059                        return Err(invalid(&format!(
2060                            "a map consumes an evaluated vector, not {}",
2061                            input.output.describe()
2062                        )));
2063                    };
2064                    Arc::clone(input_grid)
2065                };
2066                if parameters.len() != op.arity() {
2067                    return Err(invalid(&format!(
2068                        "{op:?} reads {} parameter(s), but {} are bound",
2069                        op.arity(),
2070                        parameters.len()
2071                    )));
2072                }
2073                for parameter in parameters {
2074                    parameter.validate()?;
2075                    if !matches!(&parameter.output, NativeType::Scalar(grid) if grid.spec() == input_grid.spec())
2076                    {
2077                        return Err(invalid(
2078                            "a map parameter is not a scalar on its input's grid",
2079                        ));
2080                    }
2081                }
2082                if op.takes_a_scalar_input() {
2083                    if !matches!(&self.output, NativeType::Scalar(grid) if grid.spec() == input_grid.spec())
2084                    {
2085                        return Err(invalid(
2086                            "a map over a scalar does not answer a scalar on its input's grid",
2087                        ));
2088                    }
2089                } else {
2090                    ensure_samples(&self.output, "a parameterized map")?;
2091                    ensure_same_fields(&input.output, &self.output, "a parameterized map")?;
2092                    ensure_grid(&self.output, &input_grid, "a parameterized map")?;
2093                }
2094            }
2095            NativeExpr::ParameterizedTopK {
2096                input,
2097                op,
2098                parameter,
2099            } => {
2100                input.validate()?;
2101                parameter.validate()?;
2102                if !matches!(op, BoundVectorOp::TopK { .. }) {
2103                    return Err(invalid("a parameterized topk operation is not TopK"));
2104                }
2105                let Some((_, output_grid)) = evaluated_vector(&self.output) else {
2106                    return Err(invalid(
2107                        "a parameterized topk does not produce an evaluated vector",
2108                    ));
2109                };
2110                if !matches!(&parameter.output, NativeType::Scalar(grid) if grid.spec() == output_grid.spec())
2111                {
2112                    return Err(invalid(
2113                        "a topk parameter is not a scalar on its output grid",
2114                    ));
2115                }
2116                let Some((_, input_grid)) = evaluated_vector(&input.output) else {
2117                    return Err(invalid("a parameterized topk consumes a non-vector"));
2118                };
2119                ensure_same_fields(&input.output, &self.output, "a vector operation")?;
2120                ensure_grid(&input.output, input_grid, "a vector operation")?;
2121                ensure_grid(&self.output, input_grid, "a vector operation")?;
2122            }
2123            NativeExpr::ParameterizedHistogram {
2124                input,
2125                op,
2126                parameters,
2127            } => {
2128                input.validate()?;
2129                if !matches!(op, BoundVectorOp::Histogram(_)) {
2130                    return Err(invalid(
2131                        "a parameterized histogram operation is not Histogram",
2132                    ));
2133                }
2134                let expected = match op {
2135                    BoundVectorOp::Histogram(HistogramFunction::Quantile(_)) => 1,
2136                    BoundVectorOp::Histogram(HistogramFunction::Fraction { .. }) => 2,
2137                    _ => {
2138                        return Err(invalid(
2139                            "parameterized histogram operation is not Histogram",
2140                        ));
2141                    }
2142                };
2143                if parameters.len() != expected {
2144                    return Err(invalid("parameterized histogram has the wrong arity"));
2145                }
2146                let Some((_, input_grid)) = evaluated_vector(&input.output) else {
2147                    return Err(invalid("a parameterized histogram consumes a non-vector"));
2148                };
2149                for parameter in parameters {
2150                    parameter.validate()?;
2151                    if !matches!(&parameter.output, NativeType::Scalar(grid) if grid.spec() == input_grid.spec())
2152                    {
2153                        return Err(invalid(
2154                            "a histogram parameter is not a scalar on its input grid",
2155                        ));
2156                    }
2157                }
2158                ensure_same_fields(&input.output, &self.output, "a vector operation")?;
2159                ensure_grid(&self.output, input_grid, "a vector operation")?;
2160            }
2161            NativeExpr::Vector { input, .. } => {
2162                input.validate()?;
2163                // A vector operation keeps the input's fields and grid, so it consumes and produces
2164                // the same evaluated vector.
2165                let Some((_, input_grid)) = evaluated_vector(&input.output) else {
2166                    return Err(invalid(&format!(
2167                        "a vector operation consumes an evaluated vector, not {}",
2168                        input.output.describe()
2169                    )));
2170                };
2171                ensure_samples(&self.output, "a vector operation")?;
2172                ensure_same_fields(&input.output, &self.output, "a vector operation")?;
2173                ensure_grid(&self.output, input_grid, "a vector operation")?;
2174            }
2175            NativeExpr::Timestamp { input } => {
2176                input.validate()?;
2177                let Some((_, input_grid)) = evaluated_vector(&input.output) else {
2178                    return Err(invalid("timestamp consumes an evaluated vector"));
2179                };
2180                ensure_samples(&self.output, "timestamp")?;
2181                ensure_same_fields(&input.output, &self.output, "timestamp")?;
2182                ensure_grid(&self.output, input_grid, "timestamp")?;
2183            }
2184            NativeExpr::Aggregate { input, op } => {
2185                input.validate()?;
2186                // A stage's contract is a pair of types, not only what it produces: `Partial`
2187                // turns a vector into state and `Final` turns state back into a vector, so state
2188                // never sits where a value is expected and a value never sits where state is.
2189                match op.stage {
2190                    AggregateStage::Complete => {
2191                        // Every contribution lies within this execution: vector in, vector out,
2192                        // both on the input's grid, one value per input field.
2193                        let Some((_, input_grid)) = evaluated_vector(&input.output) else {
2194                            return Err(invalid(&format!(
2195                                "a complete aggregate consumes an evaluated vector, not {}",
2196                                input.output.describe()
2197                            )));
2198                        };
2199                        ensure_samples(&self.output, "a complete aggregate")?;
2200                        ensure_same_fields(&input.output, &self.output, "a complete aggregate")?;
2201                        ensure_grid(&self.output, input_grid, "a complete aggregate")?;
2202                    }
2203                    AggregateStage::Partial => {
2204                        // Only this execution's contributions were seen, so the output is state
2205                        // for a later merge: on the input's grid, grouped the same way, one
2206                        // accumulator per input field.
2207                        let Some((input_schema, input_grid)) = evaluated_vector(&input.output)
2208                        else {
2209                            return Err(invalid(&format!(
2210                                "a partial aggregate consumes an evaluated vector, not {}",
2211                                input.output.describe()
2212                            )));
2213                        };
2214                        let NativeType::AggregateState(state) = &self.output else {
2215                            return Err(invalid(&format!(
2216                                "a partial aggregate produces aggregate state, not {}",
2217                                self.output.describe()
2218                            )));
2219                        };
2220                        ensure_field_count(
2221                            state.accumulators.len(),
2222                            input_schema.fields.len(),
2223                            "a partial aggregate",
2224                        )?;
2225                        let Some((expected_kind, expected_components)) =
2226                            aggregate_state_binding(op.kind, op.field_kind)
2227                        else {
2228                            return Err(DataFusionError::NotImplemented(format!(
2229                                "aggregate state is not defined for {:?}",
2230                                op.kind
2231                            )));
2232                        };
2233                        if state.kind != expected_kind
2234                            || state.components.as_ref() != expected_components
2235                        {
2236                            return Err(DataFusionError::NotImplemented(format!(
2237                                "partial {:?} aggregate requires state kind {:?} with components {:?}, found {:?} with {:?}",
2238                                op.kind,
2239                                expected_kind,
2240                                expected_components,
2241                                state.kind,
2242                                state.components
2243                            )));
2244                        }
2245                        if state.grid.spec() != input_grid.spec() {
2246                            return Err(invalid(
2247                                "aggregate state is indexed on a grid the input does not produce",
2248                            ));
2249                        }
2250                        if state.group != op.group {
2251                            return Err(invalid(
2252                                "aggregate state groups differ from the binding: state built for \
2253                                 one `by` or `without` cannot be merged into another",
2254                            ));
2255                        }
2256                    }
2257                    AggregateStage::Final => {
2258                        // The state was collected from every contributing execution, so the
2259                        // output is the complete value on the grid that state is indexed on.
2260                        let NativeType::AggregateState(state) = &input.output else {
2261                            return Err(invalid(&format!(
2262                                "a final aggregate consumes aggregate state, not {}",
2263                                input.output.describe()
2264                            )));
2265                        };
2266                        let Some((expected_kind, expected_components)) =
2267                            aggregate_state_binding(op.kind, op.field_kind)
2268                        else {
2269                            return Err(DataFusionError::NotImplemented(format!(
2270                                "aggregate state is not defined for {:?}",
2271                                op.kind
2272                            )));
2273                        };
2274                        if state.kind != expected_kind
2275                            || state.components.as_ref() != expected_components
2276                        {
2277                            return Err(DataFusionError::NotImplemented(format!(
2278                                "final {:?} aggregate requires state kind {:?} with components {:?}, found {:?} with {:?}",
2279                                op.kind,
2280                                expected_kind,
2281                                expected_components,
2282                                state.kind,
2283                                state.components
2284                            )));
2285                        }
2286                        if state.group != op.group {
2287                            return Err(invalid(
2288                                "aggregate state groups differ from the binding: state built for \
2289                                 one `by` or `without` cannot be merged into another",
2290                            ));
2291                        }
2292                        let output = ensure_samples(&self.output, "a final aggregate")?;
2293                        // The merged value carries one value per accumulator the state holds.
2294                        ensure_field_count(
2295                            output.fields.len(),
2296                            state.accumulators.len(),
2297                            "a final aggregate",
2298                        )?;
2299                        ensure_grid(&self.output, &state.grid, "a final aggregate")?;
2300                    }
2301                }
2302            }
2303            NativeExpr::Binary { left, right, .. } => {
2304                left.validate()?;
2305                right.validate()?;
2306                // A binary operation is point-wise, so both sides are evaluated on the statement's
2307                // grid: state has no positions at all, and raw samples carry their own timestamps
2308                // instead of positions on a shared grid.
2309                let Some(left_grid) = left.output.grid() else {
2310                    return Err(invalid(&format!(
2311                        "a binary operation consumes evaluated values, not {} on its left",
2312                        left.output.describe()
2313                    )));
2314                };
2315                let Some(right_grid) = right.output.grid() else {
2316                    return Err(invalid(&format!(
2317                        "a binary operation consumes evaluated values, not {} on its right",
2318                        right.output.describe()
2319                    )));
2320                };
2321                if left_grid.spec() != right_grid.spec() {
2322                    return Err(invalid("a binary operation aligns two different grids"));
2323                }
2324                match (&left.output, &right.output) {
2325                    // Two scalars make a scalar: there is nothing to match and nothing to keep.
2326                    (NativeType::Scalar(_), NativeType::Scalar(_)) => {
2327                        if !matches!(&self.output, NativeType::Scalar(grid) if grid.spec() == left_grid.spec())
2328                        {
2329                            return Err(invalid(
2330                                "two scalars do not produce a scalar on the statement's grid",
2331                            ));
2332                        }
2333                    }
2334                    // A scalar side carries no labels and no fields, so the operation is a
2335                    // broadcast on the statement's grid and `matching` is not consulted. A scalar
2336                    // has no fields of its own, so the broadcast keeps the vector side's fields.
2337                    (NativeType::Scalar(_), NativeType::Samples(vector))
2338                    | (NativeType::Samples(vector), NativeType::Scalar(_)) => {
2339                        let Some(output) = self.output.samples() else {
2340                            return Err(invalid(&format!(
2341                                "a scalar broadcast produces the vector side's samples, not {}",
2342                                self.output.describe()
2343                            )));
2344                        };
2345                        ensure_grid(&self.output, left_grid, "a scalar broadcast")?;
2346                        if output.fields.len() != vector.fields.len() {
2347                            return Err(invalid(
2348                                "a scalar broadcast does not produce the vector side's fields",
2349                            ));
2350                        }
2351                    }
2352                    // Two vectors are matched by `matching` on the grid they share, and the match
2353                    // pairs fields by position, so it keeps the shorter side's fields.
2354                    (NativeType::Samples(left_vector), NativeType::Samples(right_vector)) => {
2355                        let Some(output) = self.output.samples() else {
2356                            return Err(invalid(&format!(
2357                                "a binary operation over two vectors produces samples, not {}",
2358                                self.output.describe()
2359                            )));
2360                        };
2361                        ensure_field_count(
2362                            output.fields.len(),
2363                            left_vector.fields.len().min(right_vector.fields.len()),
2364                            "a binary operation over two vectors",
2365                        )?;
2366                        ensure_grid(&self.output, left_grid, "a binary operation")?;
2367                    }
2368                    // State has no grid, so the grid checks above return for either side before
2369                    // this match and these two arms cannot be reached at run time. They are
2370                    // spelled out because `NativeType` names `AggregateState` and the match has to
2371                    // cover the pair; the message says so rather than pretending a value arrived.
2372                    (NativeType::AggregateState(_), _) | (_, NativeType::AggregateState(_)) => {
2373                        return Err(invalid(
2374                            "a binary operation reached aggregate state, which has no grid: the \
2375                             grid check above should have rejected it",
2376                        ));
2377                    }
2378                }
2379            }
2380            NativeExpr::Set { left, right, .. } => {
2381                left.validate()?;
2382                right.validate()?;
2383                // A set operation keeps samples of whole vectors on the grid both share: a scalar
2384                // has no labels to match, and PromQL refuses it on either side.
2385                let Some((left_schema, left_grid)) = evaluated_vector(&left.output) else {
2386                    return Err(invalid(&format!(
2387                        "a set operation consumes evaluated vectors, not {} on its left",
2388                        left.output.describe()
2389                    )));
2390                };
2391                let Some((right_schema, right_grid)) = evaluated_vector(&right.output) else {
2392                    return Err(invalid(&format!(
2393                        "a set operation consumes evaluated vectors, not {} on its right",
2394                        right.output.describe()
2395                    )));
2396                };
2397                if left_grid.spec() != right_grid.spec() {
2398                    return Err(invalid("a set operation aligns two different grids"));
2399                }
2400                // `or` hands the right side's samples on under the left side's fields, so the
2401                // kernel pairs every field by position and refuses sides of different counts.
2402                if left_schema.fields.len() != right_schema.fields.len() {
2403                    return Err(invalid(&format!(
2404                        "a set operation pairs fields by position, and its sides carry {} and {}",
2405                        left_schema.fields.len(),
2406                        right_schema.fields.len()
2407                    )));
2408                }
2409                ensure_samples(&self.output, "a set operation")?;
2410                ensure_same_fields(&left.output, &self.output, "a set operation")?;
2411                ensure_grid(&self.output, left_grid, "a set operation")?;
2412            }
2413            NativeExpr::Info { base, info, .. } => {
2414                base.validate()?;
2415                info.validate()?;
2416                let Some((_, base_grid)) = evaluated_vector(&base.output) else {
2417                    return Err(invalid(&format!(
2418                        "info consumes an evaluated vector, not {}",
2419                        base.output.describe()
2420                    )));
2421                };
2422                // The kernel reads the info side's values as sample times, which only an instant
2423                // selection that reports them produces. A canonical merge above that selection only
2424                // collects the pieces of one label set into one series, so it hands the times on
2425                // unchanged and the check looks through it. Nothing else is looked through: an
2426                // evaluated value vector is not a set of sample times, and reading one as times
2427                // would compare a sample's value against another's.
2428                let mut source = info.as_ref();
2429                while let NativeExpr::Merge { input } = &source.expr {
2430                    source = input;
2431                }
2432                let NativeExpr::Instant { op, .. } = &source.expr else {
2433                    return Err(invalid(
2434                        "info reads its info series from an instant selection",
2435                    ));
2436                };
2437                if op.value != InstantValue::InfoTimestamp {
2438                    return Err(invalid(
2439                        "info reads the sample times of its info series, not their values",
2440                    ));
2441                }
2442                let Some((info_schema, info_grid)) = evaluated_vector(&info.output) else {
2443                    return Err(invalid("info's info series are not an evaluated vector"));
2444                };
2445                if info_schema.fields.len() != 1 {
2446                    return Err(invalid(&format!(
2447                        "info reads one sample time per info series, and its info series carry {} \
2448                         fields",
2449                        info_schema.fields.len()
2450                    )));
2451                }
2452                if base_grid.spec() != info_grid.spec() {
2453                    return Err(invalid("info aligns two different grids"));
2454                }
2455                ensure_samples(&self.output, "info")?;
2456                ensure_same_fields(&base.output, &self.output, "info")?;
2457                ensure_grid(&self.output, base_grid, "info")?;
2458            }
2459        }
2460        Ok(())
2461    }

2465///
2466/// `BroadcastRuntime::drops_labels` in [`execute`](crate::native::execute) is the rule the runtime
2467/// applies: arithmetic and a `bool` comparison answer a series under the metric name dropped, and a
2468/// filtering comparison keeps the label set it read.
2469fn broadcast_drops_labels(matching: &BinaryMatching) -> bool {
2470    !matching.op.is_comparison() || matching.return_bool
2471}

2473/// Whether `node` hands over one logical series per canonical label set, or several.
2474///
2475/// This is the rule the runtime collectors use to decide whether a consumer above `node` has to
2476/// collect what `node` emits. `RuntimeNode::build` in [`execute`](crate::native::execute) and the
2477/// Window and `limitk` collectors share this predicate so their decisions cannot disagree.
2478///
2479/// Three answers, because what a consumer may do about a contribution carries two facts and a
2480/// boundary answers neither:
2481///
2482/// - `Some(true)`: the output is one series per canonical label set, established here, so a
2483///   consumer that needs one unified series already has it and collects nothing.
2484/// - `Some(false)`: the output may hold several series of one label set, and this node *says so*,
2485///   so a consumer that needs one unified series collects them through `MergeInput`. A
2486///   [`NativeExpr::Merge`] above such a node would make it stream an already-merged series and
2487///   turn that collection off.
2488/// - `None`: the output may hold several, and this node says nothing: it has neither a proof to
2489///   offer nor a report to make. Only a boundary ([`NativeExpr::Input`]) answers this, and a node
2490///   that hands its input's series back unchanged passes the answer on. Runtime collectors leave
2491///   this case alone because they collect only when the answer is `Some(false)`.
2492///
2493/// The collection condition is `matches!(answer, Some(false))`: the Window `MergeInput` collector
2494/// at runtime and the `limitk` collector both ask this predicate of their input.
2495#[allow(clippy::match_same_arms)] // Keep each expression's contribution guarantee explicit.
2496pub fn unifies_its_own_contributions(node: &NativeNode) -> Option<bool> {
2497    match &node.expr {
2498        // A boundary hands over the pieces some upstream execution produced: what it holds is one
2499        // storage series at a time for a scan and one partition's fragments of a label set for an
2500        // exchange, and nothing in the tree can tell which series those fragments belong to. A
2501        // merge binds samples evaluated on a grid rather than proving the pieces are one series
2502        // per label set, so a proof is not this function's to give: it answers `None`. Runtime
2503        // collectors gather only when this node reports fragments with `Some(false)`.
2504        NativeExpr::Input(_) => None,
2505        // A constant holds no label set at all, so it has nothing that could arrive twice.
2506        NativeExpr::Scalar(_) | NativeExpr::EvalTime { .. } | NativeExpr::VectorOf { .. } => {
2507            Some(true)
2508        }
2509        // A selection reads raw samples - `validate` refuses an evaluated input - and raw samples
2510        // arrive one whole storage series at a time, so it answers one series per series it read,
2511        // under the label set it read. `timestamp(x)` is the exception: it reports the sample's own
2512        // time under the metric name dropped, which can give two series one label set.
2513        NativeExpr::Instant { op, .. } => Some(op.value != InstantValue::Timestamp),
2514        // The merge is the collection: whatever its input handed over, it groups the fragments by
2515        // canonical label set and emits one series per set, so what it emits shares none.
2516        NativeExpr::Merge { .. } => Some(true),
2517        // A window answers three ways, and what the function does to labels decides the first:
2518        //
2519        // - the function drops the metric name, so what it emits can coincide with another series'
2520        //   label set whatever its input held: `Some(false)` (this is a real report, and the case
2521        //   `Map`, `Timestamp` and a label-dropping broadcast are in);
2522        // - the function keeps the metric name, and the child answers one series per label set -
2523        //   `Some(true)` - or reports fragments the window's `MergeInput` collects - `Some(false)`.
2524        //   Either way the window emits one series per label set and answers `Some(true)`: an
2525        //   already-unified child keeps that property through the labels the window retains, and
2526        //   fragments are collected before the window folds them;
2527        // - the function keeps the metric name, but the child is a boundary - `None`. The window
2528        //   folds each contribution on its own, so what it emits does alias its input's fragments,
2529        //   and `Some(false)` is exact.
2530        NativeExpr::Window { input, op } => {
2531            Some(!op.function.drops_metric_name() && unifies_its_own_contributions(input).is_some())
2532        }
2533        // A map can rewrite labels, so two series can end under one label set and the runtime
2534        // records what it wrote in an `Emitted` table.
2535        NativeExpr::Map { .. } | NativeExpr::ParameterizedMap { .. } => Some(false),
2536        // `topk` hands back the series it read with their labels, so the input's answer is its own -
2537        // exactly as for the bound operation.
2538        NativeExpr::ParameterizedTopK { input, .. } => unifies_its_own_contributions(input),
2539        // A histogram function groups its rows by label set itself, so it emits one series per set.
2540        NativeExpr::ParameterizedHistogram { .. } => Some(true),
2541        // `scalar()` answers one value per point, which is not several series.
2542        NativeExpr::ScalarOf { .. } => Some(true),
2543        // A timestamp reports a position under the metric name dropped.
2544        NativeExpr::Timestamp { .. } => Some(false),
2545        NativeExpr::Binary {
2546            left,
2547            right,
2548            matching,
2549        } => {
2550            if evaluated_vector(&left.output).is_some() && evaluated_vector(&right.output).is_some()
2551            {
2552                // Two vectors are matched by the kernel, which files its rows by the label set a
2553                // sample ends under and emits one series per set.
2554                Some(true)
2555            } else if broadcast_drops_labels(matching) {
2556                // A broadcast whose operation drops the metric name can give two series of its
2557                // vector side one label set.
2558                Some(false)
2559            } else {
2560                // A filtering comparison keeps the label set it read, so the side that carries
2561                // series answers for this node. Which side that is follows the rule the runtime's
2562                // broadcast matches on: an evaluated vector carries them, and with two scalars the
2563                // carrier is the side that is not a bound constant (`broadcast_runtime`'s
2564                // `literal(left).is_none()`), which is what the second test below reads.
2565                //
2566                // The two selection rules agree wherever it is observable: every scalar node kind
2567                // answers `Some(true)` (`Scalar`, `EvalTime`, `VectorOf`), so a scalar-scalar pair
2568                // answers `Some(true)` whichever side is named and a consumer collects nothing
2569                // either way. The constant check is still what keeps the two readers describing
2570                // the same node rather than coinciding by accident.
2571                let carries_series = |node: &NativeNode| {
2572                    if node.output.samples().is_some() {
2573                        return true;
2574                    }
2575                    !matches!(node.expr, NativeExpr::Scalar(BoundScalar { .. }))
2576                };
2577                let vector = if carries_series(left) { left } else { right };
2578                unifies_its_own_contributions(vector)
2579            }
2580        }
2581        // A set operation keeps each side's samples and files them by label set, one series per
2582        // set.
2583        NativeExpr::Set { .. } => Some(true),
2584        // `info` indexes the info side by identifying labels and merges the results that share a
2585        // label set.
2586        NativeExpr::Info { .. } => Some(true),
2587        // A finished aggregate emits one series per group, and two groups are two label sets.
2588        // `Partial` state is not a series at all, which is not several either.
2589        NativeExpr::Aggregate { .. } => Some(true),
2590        // `topk`, `limit_ratio` and `sort` hand back the series they read with their labels, so the
2591        // input's answer is theirs. The four that read the whole vector - `limitk`, `absent`,
2592        // `histogram_*` and `count_values` - emit one series per label set: `limitk` counts series
2593        // in input order, so its input carries the merge the runtime performs when that input reports fragments, and the other three group their rows
2594        // by label set themselves.
2595        NativeExpr::Vector { input, op } => {
2596            let whole = matches!(
2597                op,
2598                BoundVectorOp::LimitK { .. }
2599                    | BoundVectorOp::Absent { .. }
2600                    | BoundVectorOp::Histogram(_)
2601                    | BoundVectorOp::CountValues { .. }
2602            );
2603            if whole {
2604                Some(true)
2605            } else {
2606                unifies_its_own_contributions(input)
2607            }
2608        }
2609    }
2610}

2612/// The schema and grid of an evaluated vector, or `None` when `ty` is raw samples, a scalar or
2613/// state.
2614///
2615/// An operation over values needs both halves: that its input is samples at all, which rules out
2616/// a scalar, and that those samples are stamped onto a grid, which rules out raw samples. The
2617/// schema comes back with the grid because a consumer that needs the grid also declares a field
2618/// layout for the fields the schema carries.
2619fn evaluated_vector(ty: &NativeType) -> Option<(&Arc<SeriesSchema>, &Arc<EvalGrid>)> {
2620    let schema = ty.samples().filter(|schema| schema.eval_grid.is_some())?;
2621    Some((schema, schema.eval_grid.as_ref()?))
2622}

2624/// Checks that a node declares the field layout its input produces.
2625///
2626/// A window, a map, an instant selection and a complete aggregate all report one value per input
2627/// field, so they keep the layout of their input. A node that declares a different number of
2628/// fields than it consumes describes a computation this model has no meaning for, and letting it
2629/// through would only surface at run time as a field count mismatch against a schema the tree was
2630/// supposed to have validated.
2631fn ensure_same_fields(input: &NativeType, output: &NativeType, what: &str) -> Result<()> {
2632    let (NativeType::Samples(input), NativeType::Samples(output)) = (input, output) else {
2633        return Ok(());
2634    };
2635    ensure_field_count(output.fields.len(), input.fields.len(), what)
2636}

2638/// Checks that a node declares `declared` fields where its input provides `expected`.
2639///

2641/// value per accumulator the state holds, and a vector-vector binary operation produces one field

2643/// kernel produces, so the message names the node and both numbers.

2645    if declared != expected {
2646        return Err(invalid(&format!(
2647            "{what} declares {declared} field(s) but its input provides {expected}"
2648        )));
2649    }
2650    Ok(())
2651}

2659/// Checks that `ty` is evaluated on the grid `expected` describes.
2660///
2661/// The comparison is by [`EvalGrid::spec`], not by allocation: that is the rule the kernels apply
2662/// (`vector_match::same_grid`), so a compiler that allocates one grid per window but binds equal
2663/// specs produces a tree the kernel runs correctly.
2664fn ensure_grid(ty: &NativeType, expected: &Arc<EvalGrid>, what: &str) -> Result<()> {
2665    match ty.grid() {
2666        Some(actual) if actual.spec() == expected.spec() => Ok(()),
2667        Some(_) => Err(invalid(&format!(
2668            "{what} declares a grid its output does not carry"
2669        ))),
2670        None => Err(invalid(&format!("{what} produces values without a grid"))),
2671    }
2672}

2674/// Checks that `input`, when it is a [`NativeExpr::Input`], does not declare interleaved series.
2675///
2676/// A native evaluation reads one series to its last chunk before the next: a range evaluator folds
2677/// a series completely before it starts another, and an instant selection's consumer collects one
2678/// series' chunks into one buffer. Neither can be handed chunks of different series that
2679/// interleave, which is the one rule both consumers state. Only an input node carries an ordering,
2680/// so only there is there one to check.
2681fn ensure_not_interleaved(input: &NativeNode) -> Result<()> {
2682    if let NativeExpr::Input(binding) = &input.expr
2683        && binding.ordering == SeriesOrdering::Interleaved
2684    {
2685        return Err(invalid(
2686            "a native evaluation reads one series to its last chunk before the next, so \
2687             interleaved chunks cannot be fed to it",
2688        ));
2689    }
2690    Ok(())
2691}

2693fn invalid(reason: &str) -> DataFusionError {
2694    // A rejected tree is a compiler defect, not a query error: the compiler only builds a node
2695    // after deciding that this combination is supported. `Internal` keeps that meaning on the wire
2696    // instead of reporting a mistake in the client's statement, which `Plan` would.
2697    DataFusionError::Internal(reason.to_string())
2698}
