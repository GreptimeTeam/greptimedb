15//! Per-node metric handles for native PromQL execution and their DataFusion registration.
16//!
17//! The whole native tree registers into the one [`ExecutionPlanMetricsSet`] of its enclosing
18//! operator (`FusedSeriesExec`), and readers aggregate by name, so each name carries the node's
19//! path in the runtime tree: `native.<path>.<metric>`, with a metric from [`NodeMetric::ALL`]. The
20//! path is valid only within the topology that produced it; it is not a persisted or public
21//! identifier. `peak_reserved_bytes` alone is named `native.p<partition>.<path>.<metric>`, because
22//! a sum of peaks is not a peak.
23
24use std::collections::HashMap;
25
26use datafusion::physical_plan::metrics::MetricValue;
27use datafusion::physical_plan::metrics::{
28    Count, ExecutionPlanMetricsSet, Gauge, MetricBuilder, Time,
29};

31/// The reserved first segment of every native metric name.
32const NATIVE_PREFIX: &str = "native";
33
34/// The metric names one node reports. [`Self::ALL`] is complete:
35/// `fresh_exports_register_exactly_the_declared_metrics_without_growth` checks it against
36/// [`NodeMetrics::register`].
37pub struct NodeMetric;
38
39impl NodeMetric {
40    /// How many batches this node consumed.
41    pub const INPUT_BATCHES: &'static str = "input_batches";
42    /// How many batches this node produced.
43    pub const OUTPUT_BATCHES: &'static str = "output_batches";
44    /// How many sample positions this node ingested.
45    pub const INPUT_SAMPLES: &'static str = "input_samples";
46    /// How many grid positions this node evaluated, including those with no valid result.
47    pub const EVALUATED_GRID_POINTS: &'static str = "evaluated_grid_points";
48    /// How many valid values this node produced.
49    pub const OUTPUT_VALUES: &'static str = "output_values";
50    /// How many series this node closed.
51    pub const COMPLETED_SERIES: &'static str = "completed_series";
52    /// Time this node spent in its own work, excluding its inputs.
53    pub const SELF_COMPUTE_TIME_NS: &'static str = "self_compute_time_ns";
54    /// Bytes this node currently reserves.
55    pub const CURRENT_RESERVED_BYTES: &'static str = "current_reserved_bytes";
56    /// The highest byte count this node reserved during the execution.
57    pub const PEAK_RESERVED_BYTES: &'static str = "peak_reserved_bytes";
58    /// How many groups this node tracks, counted per partition: a group reached by several
59    /// partitions is one entry in each.
60    pub const GROUPS: &'static str = "groups";
61    /// How many state entries this node tracks.
62    pub const STATE_ENTRIES: &'static str = "state_entries";
63
64    /// Every metric a node may report.
65    pub const ALL: &'static [&'static str] = &[
66        Self::INPUT_BATCHES,
67        Self::OUTPUT_BATCHES,
68        Self::INPUT_SAMPLES,
69        Self::EVALUATED_GRID_POINTS,
70        Self::OUTPUT_VALUES,
71        Self::COMPLETED_SERIES,
72        Self::SELF_COMPUTE_TIME_NS,
73        Self::CURRENT_RESERVED_BYTES,
74        Self::PEAK_RESERVED_BYTES,
75        Self::GROUPS,
76        Self::STATE_ENTRIES,
77    ];
78}

80/// The figures every node of one execution recorded, keyed by the path passed to
81/// [`NodeMetrics::register`] (`root`, `root.left`, ...). A node's partitions are combined: counts
82/// and times add up, and `peak_reserved_bytes` takes the highest.
83#[derive(Clone, Debug, Default, PartialEq, Eq)]
84pub struct NodeMetricsSnapshot {
85    nodes: HashMap<String, Vec<(String, usize)>>,
86}
87
88impl NodeMetricsSnapshot {
89    /// Reads every native node figure in `set`, ignoring other metrics. A plan that never ran
90    /// yields an empty snapshot.
91    ///
92    /// Partitions are combined here rather than by
93    /// [`MetricsSet::aggregate_by_name`](datafusion::physical_plan::metrics::MetricsSet::aggregate_by_name),
94    /// which adds at least one nanosecond to every time it touches: a node whose timer never ran
95    /// would then report time it did not spend.
96    pub fn read(set: &ExecutionPlanMetricsSet) -> Self {
97        let mut nodes: HashMap<String, Vec<(String, usize)>> = HashMap::new();
98        for metric in set.clone_inner().iter() {
99            let Some((path, name)) = split_name(metric.value().name()) else {
100                continue;
101            };
102            let value = metric.value().as_usize();
103            let figures = nodes.entry(path.to_string()).or_default();
104            match figures.iter_mut().find(|(figure, _)| figure == name) {
105                // Partitions peak at different moments, so the node's peak is the max, not the sum.
106                Some((_, accumulated)) if name == NodeMetric::PEAK_RESERVED_BYTES => {
107                    *accumulated = (*accumulated).max(value)
108                }
109                Some((_, accumulated)) => *accumulated = accumulated.saturating_add(value),
110                None => figures.push((name.to_string(), value)),
111            }
112        }
113        for figures in nodes.values_mut() {
114            figures.sort_by_key(|(name, _)| node_metric_order(name));
115        }
116        Self { nodes }
117    }
118
119    /// What this node recorded, dropping zero figures: a node registers every metric, so most are
120    /// zero on any one node.
121    pub fn described(&self, path: &str) -> Vec<(String, usize)> {
122        self.of(path)
123            .iter()
124            .filter(|(_, value)| *value != 0)
125            .cloned()
126            .collect()
127    }
128
129    /// What this node recorded, in [`NodeMetric::ALL`] order, or an empty list when it recorded
130    /// nothing.
131    pub fn of(&self, path: &str) -> &[(String, usize)] {
132        self.nodes.get(path).map(Vec::as_slice).unwrap_or_default()
133    }
134}

136/// Splits `native.<path>.<metric>` or `native.p<partition>.<path>.<metric>` into path and metric,
137/// dropping the partition. Returns `None` for any other name.
138fn split_name(name: &str) -> Option<(&str, &str)> {
139    let rest = name.strip_prefix(NATIVE_PREFIX)?.strip_prefix('.')?;
140    let rest = match rest.strip_prefix('p').and_then(|rest| rest.split_once('.')) {
141        Some((partition, rest))
142            if !partition.is_empty() && partition.bytes().all(|byte| byte.is_ascii_digit()) =>
143        {
144            rest
145        }
146        _ => rest,
147    };
148    let (path, metric) = rest.rsplit_once('.')?;
149    NodeMetric::ALL.contains(&metric).then_some((path, metric))
150}
151
152/// Where `metric` sits in [`NodeMetric::ALL`], so figures print in a stable order.
153fn node_metric_order(metric: &str) -> usize {
154    NodeMetric::ALL
155        .iter()
156        .position(|declared| *declared == metric)
157        .unwrap_or(usize::MAX)
158}

160/// The metric handles one node in one partition updates as it runs.
161#[derive(Clone, Debug)]
162pub struct NodeMetrics {
163    input_batches: Count,
164    output_batches: Count,
165    input_samples: Count,
166    evaluated_grid_points: Count,
167    output_values: Count,
168    completed_series: Count,
169    self_compute_time: Time,
170    current_reserved_bytes: Gauge,
171    peak_reserved_bytes: Gauge,
172    groups: Gauge,
173    state_entries: Gauge,
174}
175
176impl Default for NodeMetrics {
177    fn default() -> Self {
178        Self {
179            input_batches: Count::new(),
180            output_batches: Count::new(),
181            input_samples: Count::new(),
182            evaluated_grid_points: Count::new(),
183            output_values: Count::new(),
184            completed_series: Count::new(),
185            self_compute_time: Time::new(),
186            current_reserved_bytes: Gauge::new(),
187            peak_reserved_bytes: Gauge::new(),
188            groups: Gauge::new(),
189            state_entries: Gauge::new(),
190        }
191    }
192}

194/// Whether `name` is a figure a node of a native tree reports, rather than one of an operator.
195pub fn is_node_metric(name: &str) -> bool {
196    name.strip_prefix(NATIVE_PREFIX)
197        .is_some_and(|rest| rest.starts_with('.'))
198}
199
200/// The full metric name for `metric` reported by the node called `name`.
201fn named(name: &str, metric: &str) -> String {
202    format!("{NATIVE_PREFIX}.{name}.{metric}")
203}
204
205/// The full metric name for `metric` in `partition` of the node called `name`.
206fn partitioned(name: &str, metric: &str, partition: usize) -> String {
207    format!("{NATIVE_PREFIX}.p{partition}.{name}.{metric}")
208}
209
210impl NodeMetrics {
211    /// Registers this node's handles under `name` into `set`. Call once per execution, before it
212    /// starts; `set` then follows what the node records.
213    pub fn register(&self, name: &str, set: &ExecutionPlanMetricsSet, partition: usize) {
214        fn count(set: &ExecutionPlanMetricsSet, partition: usize, name: String, count: &Count) {
215            MetricBuilder::new(set)
216                .with_partition(partition)
217                .build(MetricValue::Count {
218                    name: name.into(),
219                    count: count.clone(),
220                });
221        }
222        fn gauge(set: &ExecutionPlanMetricsSet, partition: usize, name: String, gauge: &Gauge) {
223            MetricBuilder::new(set)
224                .with_partition(partition)
225                .build(MetricValue::Gauge {
226                    name: name.into(),
227                    gauge: gauge.clone(),
228                });
229        }
230
231        count(
232            set,
233            partition,
234            named(name, NodeMetric::INPUT_BATCHES),
235            &self.input_batches,
236        );
237        count(
238            set,
239            partition,
240            named(name, NodeMetric::OUTPUT_BATCHES),
241            &self.output_batches,
242        );
243        count(
244            set,
245            partition,
246            named(name, NodeMetric::INPUT_SAMPLES),
247            &self.input_samples,
248        );
249        count(
250            set,
251            partition,
252            named(name, NodeMetric::EVALUATED_GRID_POINTS),
253            &self.evaluated_grid_points,
254        );
255        count(
256            set,
257            partition,
258            named(name, NodeMetric::OUTPUT_VALUES),
259            &self.output_values,
260        );
261        count(
262            set,
263            partition,
264            named(name, NodeMetric::COMPLETED_SERIES),
265            &self.completed_series,
266        );
267        gauge(
268            set,
269            partition,
270            named(name, NodeMetric::CURRENT_RESERVED_BYTES),
271            &self.current_reserved_bytes,
272        );
273        // The collector sums gauges by name; the peak alone must not be summed.
274        gauge(
275            set,
276            partition,
277            partitioned(name, NodeMetric::PEAK_RESERVED_BYTES, partition),
278            &self.peak_reserved_bytes,
279        );
280        gauge(
281            set,
282            partition,
283            named(name, NodeMetric::GROUPS),
284            &self.groups,
285        );
286        gauge(
287            set,
288            partition,
289            named(name, NodeMetric::STATE_ENTRIES),
290            &self.state_entries,
291        );
292        MetricBuilder::new(set)
293            .with_partition(partition)
294            .build(MetricValue::Time {
295                name: named(name, NodeMetric::SELF_COMPUTE_TIME_NS).into(),
296                time: self.self_compute_time.clone(),
297            });
298    }
299
300    /// Records that this node ingested `batches` batches carrying `samples` sample positions.
301    pub fn record_input(&self, batches: usize, samples: usize) {
302        self.input_batches.add(batches);
303        self.input_samples.add(samples);
304    }
305
306    /// Records that this node produced `batches` batches carrying `values` valid values.
307    pub fn record_output(&self, batches: usize, values: usize) {
308        self.output_batches.add(batches);
309        self.output_values.add(values);
310    }
311
312    /// Records that this node evaluated `points` grid positions.
313    pub fn record_grid_points(&self, points: usize) {
314        self.evaluated_grid_points.add(points);
315    }
316
317    /// Records that this node closed `series` series.
318    pub fn record_completed_series(&self, series: usize) {
319        self.completed_series.add(series);
320    }
321
322    /// Records that this node now reserves `bytes`, raising its peak if higher.
323    pub fn record_reserved_bytes(&self, bytes: usize) {
324        self.current_reserved_bytes.set(bytes);
325        self.peak_reserved_bytes.set_max(bytes);
326    }
327
328    /// Records the size of the grouping and state this node holds. No production node calls this
329    /// yet, so `groups` and `state_entries` report zero.
330    pub fn record_state(&self, groups: usize, entries: usize) {
331        self.groups.set(groups);
332        self.state_entries.set(entries);
333    }
334
335    /// Starts timing this node's own work. Drop the guard before polling an input: self time
336    /// excludes the inputs' work.
337    pub fn timer(&self) -> datafusion::physical_plan::metrics::ScopedTimerGuard<'_> {
338        self.self_compute_time.timer()
339    }
340}

350    fn metrics_named(set: &ExecutionPlanMetricsSet) -> Vec<String> {
351        set.clone_inner()
352            .iter()

357    #[test]
358    fn fresh_exports_register_exactly_the_declared_metrics_without_growth() {
359        assert_eq!(NodeMetric::ALL.len(), 11);
360        let metrics = NodeMetrics::default();
361        for _ in 0..2 {
362            let set = ExecutionPlanMetricsSet::new();
363            metrics.register("root", &set, 0);
364            let names = metrics_named(&set);
365            assert_eq!(names.len(), NodeMetric::ALL.len());
366            let mut names: Vec<_> = names
367                .iter()
368                .map(|name| name.rsplit('.').next().unwrap().to_string())
369                .collect();
370            names.sort();
371            let mut declared: Vec<_> = NodeMetric::ALL
372                .iter()
373                .map(|name| name.to_string())
374                .collect();
375            declared.sort();
376            assert_eq!(names, declared);
377        }
378    }

380    #[test]
381    fn exported_counters_have_recorded_values() {
382        let metrics = NodeMetrics::default();
383        metrics.record_input(3, 17);
384        metrics.record_output(2, 11);
385        let set = ExecutionPlanMetricsSet::new();
386        metrics.register("root", &set, 0);
387        let registered = set.clone_inner();
388        let counter = |name: &str| {
389            registered.iter().find_map(|metric| match metric.value() {
390                MetricValue::Count {
391                    name: metric_name,
392                    count,
393                } if metric_name.as_ref() == name => Some(count.value()),
394                _ => None,
395            })
396        };
397        assert_eq!(counter("native.root.input_batches"), Some(3));
398        assert_eq!(counter("native.root.input_samples"), Some(17));
399        assert_eq!(counter("native.root.output_batches"), Some(2));
400        assert_eq!(counter("native.root.output_values"), Some(11));
401    }

403    #[test]
404    fn peak_exports_are_distinct_per_partition() {
405        let first_metrics = NodeMetrics::default();
406        first_metrics.record_reserved_bytes(1024);
407        let first = ExecutionPlanMetricsSet::new();
408        first_metrics.register("root", &first, 0);
409        let second_metrics = NodeMetrics::default();
410        second_metrics.record_reserved_bytes(2048);
411        let second = ExecutionPlanMetricsSet::new();
412        second_metrics.register("root", &second, 1);
413        let peak = |set: &ExecutionPlanMetricsSet, expected: &str| {
414            set.clone_inner()
415                .iter()
416                .find_map(|metric| match metric.value() {
417                    MetricValue::Gauge { name, gauge } if name.as_ref() == expected => {
418                        Some(gauge.value())
419                    }
420                    _ => None,
421                })
422        };
423        assert_eq!(
424            peak(&first, "native.p0.root.peak_reserved_bytes"),
425            Some(1024)
426        );
427        assert_eq!(
428            peak(&second, "native.p1.root.peak_reserved_bytes"),
429            Some(2048)
430        );
431    }

433    #[test]
434    fn a_node_reports_one_set_of_figures_across_its_partitions() {
435        let set = ExecutionPlanMetricsSet::new();
436        for (partition, batches, samples, peak) in [(0, 2, 10, 1024), (1, 3, 20, 4096)] {
437            let metrics = NodeMetrics::default();
438            metrics.record_input(batches, samples);
439            metrics.record_reserved_bytes(peak);
440            metrics.register("root", &set, partition);
441        }
442
443        let snapshot = NodeMetricsSnapshot::read(&set);
444        let figure = |name: &str| {
445            snapshot
446                .of("root")
447                .iter()
448                .find_map(|(figure, value)| (figure == name).then_some(*value))
449                .unwrap_or_else(|| panic!("{name} is reported"))
450        };
451        assert_eq!(figure(NodeMetric::INPUT_BATCHES), 5);
452        assert_eq!(figure(NodeMetric::INPUT_SAMPLES), 30);
453        assert_eq!(figure(NodeMetric::PEAK_RESERVED_BYTES), 4096);
454    }
