15//! One line per node of a [`NativeTree`].
16//!
17//! The tree is displayed by recursively visiting its bound expressions. The display is an
18//! observation of a compiled tree: it runs nothing, and every line reads as the node it is,
19//! making no claim about whether that node could run.
20//!
21//! A node that recorded figures shows them at the end of its own line, and only its own: the walk
22//! carries the path the runtime registered that node under, so the figures of one node are never
23//! reported as those of another. A tree whose nodes recorded nothing - one that was never run,
24//! which is what `EXPLAIN` shows - prints structure only.
25

45/// The path the runtime registers a tree's root node under.
46///
47/// Every other path is this one and the segments of [`child_segments`], so the walk below asks for
48/// a node's figures by the same key its registration used.
49pub const ROOT_PATH: &str = "root";
50
51/// One displayed node: its line, the figures it recorded, and the nodes that consume its output.
52///
53/// The same report feeds the tree's text and, through the operator that owns the tree, the
54/// structure a machine-readable `ANALYZE` publishes, so the two cannot describe different trees.
55#[derive(Clone, Debug, PartialEq, Eq)]
56pub struct NativeNodeReport {
57    /// The node's line without figures, as `<kind>: <what it computes>`.
58    pub line: String,
59    /// The figures this node recorded, in [`crate::native::metrics::NodeMetric::ALL`] order.
60    pub metrics: Vec<(String, usize)>,
61    /// The nodes that consume this one's output, in the order the runtime feeds them.
62    pub children: Vec<NativeNodeReport>,
63}
64
65impl From<NativeNodeReport> for NativeTreeMetrics {
66    /// The same nodes, as the plain data a machine-readable `ANALYZE` publishes.
67    ///
68    /// The text and that record are built from one report, so a reader of either sees the same
69    /// nodes with the same figures.
70    fn from(report: NativeNodeReport) -> Self {
71        Self {
72            line: report.line,
73            metrics: report.metrics,
74            children: report.children.into_iter().map(Self::from).collect(),
75        }
76    }
77}

79/// A grid as `<start>/<step>x<len>`.
80fn grid_name(grid: &EvalGrid) -> String {
81    let spec = grid.spec();
82    format!(
83        "{}/{step}x{len}",
84        spec.start_ms,
85        step = spec.step_ms,
86        len = spec.len
87    )
88}

90/// A label list in the order it was bound.
91fn label_list(labels: &[Arc<str>]) -> String {
92    labels
93        .iter()
94        .map(|label| label.to_string())
95        .collect::<Vec<_>>()
96        .join(", ")
97}

99/// The ordering an input guarantees, when it is weaker than the scan's default.
100fn ordering(ordering: SeriesOrdering) -> &'static str {
101    // `KeyThenTime` is what a scan that orders anything at all provides, so it is the
102    // uninteresting case; a weaker guarantee is what a consumer has to know about.
103    match ordering {
104        SeriesOrdering::KeyThenTime => "",
105        SeriesOrdering::Contiguous => ", contiguous",
106        SeriesOrdering::Interleaved => ", interleaved",
107    }
108}

110/// The grid a node's output is evaluated on, or `-` for one that has none.
111fn grid_of(node: &NativeNode) -> String {
112    node.output
113        .grid()
114        .map_or_else(|| "-".to_string(), |grid| grid_name(grid))
115}

117/// The operation a node computes, in the words of the query that produced it.
118fn describe(node: &NativeNode) -> String {
119    match &node.expr {
120        NativeExpr::Input(input) => match &input.output {
121            NativeType::Samples(schema) if schema.eval_grid.is_some() => {
122                format!(
123                    "evaluated samples source={}{}",
124                    input.source_index,
125                    ordering(input.ordering)
126                )
127            }
128            NativeType::Samples(_) => format!(
129                "raw samples source={}{}",
130                input.source_index,
131                ordering(input.ordering)
132            ),
133            NativeType::Scalar(grid) => format!("scalar on grid {}", grid_name(grid)),
134            NativeType::AggregateState(_) => "aggregate state".to_string(),
135        },
136        NativeExpr::Scalar(scalar) => scalar.value.to_string(),
137        NativeExpr::EvalTime { date: None } => format!("time() on grid {}", grid_of(node)),
138        NativeExpr::EvalTime { date: Some(date) } => {
139            format!("{}() on grid {}", date.name(), grid_of(node))
140        }
141        NativeExpr::VectorOf { .. } => "vector()".to_string(),
142        NativeExpr::ScalarOf { .. } => "scalar()".to_string(),
143        NativeExpr::ParameterizedTopK { op, .. } => match op {
144            BoundVectorOp::TopK { top: true, .. } => "parameterized topk".to_string(),
145            BoundVectorOp::TopK { top: false, .. } => "parameterized bottomk".to_string(),
146            _ => "parameterized topk".to_string(),
147        },
148        NativeExpr::ParameterizedHistogram { op, parameters, .. } => {
149            format!(
150                "{} ({} parameters)",
151                match op {
152                    BoundVectorOp::Histogram(HistogramFunction::Quantile(_)) =>
153                        "histogram_quantile",
154                    BoundVectorOp::Histogram(HistogramFunction::Fraction { .. }) =>
155                        "histogram_fraction",
156                    _ => "histogram",
157                },
158                parameters.len()
159            )
160        }
161        NativeExpr::ParameterizedMap { op, .. } => match op {
162            ParameterizedOp::Round => "round".to_string(),
163            ParameterizedOp::Clamp => "clamp".to_string(),
164            ParameterizedOp::ClampMin => "clamp_min".to_string(),
165            ParameterizedOp::ClampMax => "clamp_max".to_string(),
166            ParameterizedOp::MaxOf => "max_of".to_string(),
167            ParameterizedOp::MinOf => "min_of".to_string(),
168        },
169        NativeExpr::Instant { op, .. } => {
170            let mut text = format!(
171                "lookback {}ms offset {} on grid {}",
172                op.lookback_ms,
173                op.offset_ms,
174                grid_name(&op.output_grid)
175            );
176            if let Some(at) = op.at_ms {
177                text.push_str(&format!(" @ {at}ms"));
178            }
179            match op.value {
180                InstantValue::Sample => {}
181                InstantValue::Timestamp => text.push_str(" timestamp"),
182                InstantValue::InfoTimestamp => text.push_str(" info timestamp"),
183            }
184            text
185        }
186        NativeExpr::Window { op, .. } => window(op),
187        NativeExpr::Merge { .. } => "canonical label-set merge".to_string(),
188        NativeExpr::Timestamp { .. } => "evaluation time".to_string(),
189        NativeExpr::Map { op, .. } => {
190            let ops: Vec<String> = op.ops.iter().map(map_op).collect();
191            ops.join(", ")
192        }
193        NativeExpr::Vector { op, .. } => match op {
194            BoundVectorOp::TopK { k, top, group } => format!(
195                "{} {k} {}",
196                if *top { "topk" } else { "bottomk" },
197                grouping(group)
198            ),
199            BoundVectorOp::LimitK { k, group } => format!("limitk {k} {}", grouping(group)),
200            BoundVectorOp::LimitRatio { ratio } => format!("limit_ratio {ratio}"),
201            BoundVectorOp::Sort(order) => match order {
202                SortOrder::Asc => "sort".to_string(),
203                SortOrder::Desc => "sort_desc".to_string(),
204                SortOrder::ByLabel(labels) => format!("sort_by_label [{}]", label_list(labels)),
205                SortOrder::ByLabelDesc(labels) => {
206                    format!("sort_by_label_desc [{}]", label_list(labels))
207                }
208            },
209            BoundVectorOp::Absent { labels } => format!("absent {{{}}}", label_pairs(labels)),
210            BoundVectorOp::Histogram(HistogramFunction::Quantile(quantile)) => {
211                format!("histogram_quantile {quantile}")
212            }
213            BoundVectorOp::Histogram(HistogramFunction::Fraction { lower, upper }) => {
214                format!("histogram_fraction {lower} {upper}")
215            }
216            BoundVectorOp::CountValues { label, group } => {
217                format!("count_values {label:?} {}", grouping(group))
218            }
219        },
220        NativeExpr::Aggregate { op, .. } => format!(
221            "{} {} [{}] {}",
222            aggregate_kind(op.kind),
223            match &op.group {
224                Grouping::By(_) => "by",
225                Grouping::Without(_) => "without",
226            },
227            label_list(match &op.group {
228                Grouping::By(labels) | Grouping::Without(labels) => labels,
229            }),
230            match op.stage {
231                AggregateStage::Complete => "complete",
232                AggregateStage::Partial => "partial",
233                AggregateStage::Final => "final",
234            }
235        ),
236        NativeExpr::Binary { matching, .. } => format!(
237            "{}{} {} {}{}",
238            binary_op(matching.op),
239            if matching.return_bool { " bool" } else { "" },
240            matching_labels(&matching.labels),
241            match &matching.cardinality {
242                Cardinality::OneToOne => "one-to-one".to_string(),
243                Cardinality::ManyToOne(labels) => format!("many-to-one [{}]", label_list(labels)),
244                Cardinality::OneToMany(labels) => format!("one-to-many [{}]", label_list(labels)),
245            },
246            fill(matching.fill)
247        ),
248        NativeExpr::Set { op, .. } => format!(
249            "{} {}",
250            match op.op {
251                SetOp::And => "and",
252                SetOp::Or => "or",
253                SetOp::Unless => "unless",
254            },
255            matching_labels(&op.labels)
256        ),
257        NativeExpr::Info { selector, .. } => format!(
258            "names [{}] data [{}]{}",
259            selector
260                .name_matchers
261                .iter()
262                .map(|(op, value)| format!(
263                    "__name__{}\"{value}\"",
264                    match op {
265                        MatchOp::Equal => "=",
266                        MatchOp::NotEqual => "!=",
267                        MatchOp::Regex => "=~",
268                        MatchOp::NotRegex => "!~",
269                    }
270                ))
271                .collect::<Vec<_>>()
272                .join(", "),
273            label_list(&selector.data_labels),
274            if selector.keep_unmatched {
275                " keep unmatched"
276            } else {
277                ""
278            }
279        ),
280    }
281}

283fn matching_labels(labels: &MatchingLabels) -> String {
284    match labels {
285        MatchingLabels::On(labels) => format!("on [{}]", label_list(labels)),
286        MatchingLabels::Ignoring(labels) => format!("ignoring [{}]", label_list(labels)),
287    }

290fn fill(values: FillValues) -> String {
291    let side = |name: &str, value: Option<f64>| {
292        value
293            .map(|value| format!(" {name} {value}"))
294            .unwrap_or_default()
295    };
296    format!(
297        "{}{}",
298        side("fill_left", values.lhs),
299        side("fill_right", values.rhs)
300    )
301}

318/// A window: what it runs, over which range, and where it lands.
319fn window(op: &BoundWindow) -> String {
320    format!(
321        "{} [{}ms]{} offset {} anchor {} on grid {}",
322        function(op.function),
323        op.window.range_ms,
324        match op.window.modifier {
325            Some(modifier) => format!(" {} lookback {}ms", modifier.name(), modifier.before_ms()),
326            None => String::new(),
327        },
328        op.window.offset_ms,
329        match op.window.anchor_ms {
330            Some(anchor_ms) => anchor_ms.to_string(),
331            None => "-".to_string(),
332        },
333        grid_name(&op.output_grid)
334    )
335}

481/// The child nodes of `node` and the path segment each is registered under, in child order.
482///
483/// The segments are the ones the runtime registers, not the names the bound tree happens to give
484/// the operands: `info` binds its operands as `base` and `info`, while the runtime registers them
485/// as the kernel's `left` and `right`, so the display asks for the figures where they were put.
486fn child_segments(node: &NativeNode) -> Vec<(&NativeNode, Cow<'static, str>)> {
487    let mut children = Vec::new();
488    fn child<'a>(
489        children: &mut Vec<(&'a NativeNode, Cow<'static, str>)>,
490        segment: &'static str,
491        node: &'a NativeNode,
492    ) {
493        children.push((node, Cow::Borrowed(segment)));
494    }
495    match &node.expr {
496        NativeExpr::Input(_) | NativeExpr::Scalar(_) | NativeExpr::EvalTime { .. } => {}
497        NativeExpr::Instant { input, .. }
498        | NativeExpr::Window { input, .. }
499        | NativeExpr::Merge { input }
500        | NativeExpr::Map { input, .. }
501        | NativeExpr::Timestamp { input }
502        | NativeExpr::Aggregate { input, .. }
503        | NativeExpr::Vector { input, .. }
504        | NativeExpr::VectorOf { input }
505        | NativeExpr::ScalarOf { input } => child(&mut children, "input", input),
506        NativeExpr::ParameterizedTopK {
507            input, parameter, ..
508        } => {
509            child(&mut children, "input", input);
510            child(&mut children, "parameter", parameter);
511        }
512        NativeExpr::ParameterizedHistogram {
513            input, parameters, ..
514        } => {
515            child(&mut children, "input", input);
516            for (index, parameter) in parameters.iter().enumerate() {
517                children.push((parameter, Cow::Owned(format!("parameter{index}"))));
518            }
519        }
520        NativeExpr::ParameterizedMap {
521            input, parameters, ..
522        } => {
523            child(&mut children, "input", input);
524            for (index, parameter) in parameters.iter().enumerate() {
525                children.push((parameter, Cow::Owned(format!("parameter{index}"))));
526            }
527        }
528        NativeExpr::Binary { left, right, .. } | NativeExpr::Set { left, right, .. } => {
529            child(&mut children, "left", left);
530            child(&mut children, "right", right);
531        }
532        NativeExpr::Info { base, info, .. } => {
533            child(&mut children, "left", base);
534            child(&mut children, "right", info);
535        }
536    }
537    children
538}

540/// The node at `path` and everything below it, with the figures each node recorded.
541fn report_node(node: &NativeNode, path: &str, metrics: &NodeMetricsSnapshot) -> NativeNodeReport {
542    NativeNodeReport {
543        line: format!("{}: {}", kind(&node.expr), describe(node)),
544        metrics: metrics.described(path),
545        children: child_segments(node)
546            .into_iter()
547            .map(|(child, segment)| report_node(child, &format!("{path}.{segment}"), metrics))
548            .collect(),
549    }
550}

552impl NativeTree {
553    /// This tree, one report per node, carrying the figures each node recorded.
554    ///
555    /// A snapshot with no entries reports the tree's structure only, which is what a plan that was
556    /// never executed has to show.
557    pub fn report(&self, metrics: &NodeMetricsSnapshot) -> NativeNodeReport {
558        report_node(self.root(), ROOT_PATH, metrics)
559    }
560
561    /// This tree rendered from column zero, with each node's own figures when `metrics` holds them.
562    pub fn display_with<'a>(&'a self, metrics: &'a NodeMetricsSnapshot) -> NativeTreeDisplay<'a> {
563        NativeTreeDisplay {
564            tree: self,
565            metrics,
566        }
567    }
568}

570/// A tree as it appears below the operator line: one node per line, two spaces per depth, and each
571/// node's figures on its own line.
572pub struct NativeTreeDisplay<'a> {
573    tree: &'a NativeTree,
574    metrics: &'a NodeMetricsSnapshot,
575}
576
577impl fmt::Display for NativeTreeDisplay<'_> {
578    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
579        // The root starts at column zero: the block the caller brackets the tree in carries the
580        // indentation, so each line here is depth-relative to that block rather than to the
581        // caller's own place in the plan.
582        write_report(f, &self.tree.report(self.metrics), 0)
583    }
584}
585
586fn write_report(
587    f: &mut fmt::Formatter<'_>,
588    report: &NativeNodeReport,
589    depth: usize,
590) -> fmt::Result {
591    write!(f, "{:indent$}{}", "", report.line, indent = depth * 2)?;
592    if !report.metrics.is_empty() {
593        let figures = report
594            .metrics
595            .iter()
596            .map(|(name, value)| format_metric(name, *value))
597            .collect::<Vec<_>>()
598            .join(", ");
599        write!(f, ", metrics=[{figures}]")?;
600    }
601    for child in &report.children {
602        f.write_char('\n')?;
603        write_report(f, child, depth + 1)?;
604    }
605    Ok(())
606}
607
608impl fmt::Display for NativeTree {
609    /// One line per node, outermost first, from column zero with two spaces per depth and without
610    /// figures: this is the tree's own shape, which is what a display that holds no metrics
611    /// snapshot can show.
612    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
613        write_report(f, &self.report(&NodeMetricsSnapshot::default()), 0)
614    }
615}
