15//! The grid positions a field occupies, as sorted, disjoint runs.
16//!
17//! A node that rewrites labels — a `Map` dropping the metric name, a `Window` dropping the name and
18//! the empty-valued labels — can turn two input series into one label set, and every layer above
19//! then reads the two chunks as one series. What decides whether that pair is legal is which
20//! positions of each field already hold a value: two values of one field at one position cannot be
21//! merged away, while a chunk that fills positions the set left empty merges cleanly, as
22//! [`crate::grid_series::merge_same_labelset`] does.
23//!
24//! The native runtime's partition-local [`crate::native::execute::Emitted`] table keeps this state
25//! per label set and field as runs, so a chunk of many positions costs one run rather than one
26//! entry per position.
27
28/// One run of grid positions, half-open, that hold a value in one field.
29#[derive(Clone, Copy, Debug)]
30pub(crate) struct Run {
31    pub(crate) start: usize,
32    pub(crate) end: usize,
33}
34
35impl Run {
36    /// Whether two runs share a position.
37    pub(crate) fn overlaps(self, other: Self) -> bool {
38        self.start < other.end && other.start < self.end
39    }
40}

42// How many times `absorb` rebuilt the stored runs instead of appending to them.
43//
44// Test-only, and per thread: the runs stored afterwards are the same either way, so a rebuild shows
45// only in what it costs, and a test that reads its own thread's figure is not disturbed by another
46// test's chunks.
47#[cfg(test)]
48thread_local! {
49    static ABSORB_REBUILDS: std::cell::Cell<usize> = const { std::cell::Cell::new(0) };
50}
51
52/// How many times this thread's [`absorb`] rebuilt the stored runs.
53#[cfg(test)]
54pub(crate) fn absorb_rebuilds() -> usize {
55    ABSORB_REBUILDS.with(std::cell::Cell::get)
56}

58/// Merges `incoming` into `stored`, keeping the result in position order and joining the runs that
59/// touch.
60///
61/// Within a chunk one pass over both sides is what keeps this linear, since each side is sorted and
62/// disjoint. Adding the runs one at a time and rebuilding the array after each would cost the square
63/// of the runs a chunk yields: a chunk of one hundred thousand positions alternating between values
64/// and nulls yields fifty thousand runs, and re-merging after each is over a billion element moves
65/// for a single series.
66///
67/// One pass per chunk is still a pass over everything the series has stored so far, so it is not
68/// free once the *history* grows across chunks: a series split into `K` chunks of one position each,
69/// a hole apart, stores one more run per chunk and moves `K(K+1)/2` of them, with no label collision
70/// and no illegal input anywhere. A chunk that starts no earlier than the end of the last stored run
71/// only ever adds to the end — the stored runs are sorted and disjoint, so every one of them ends
72/// before anything this chunk holds — and it is appended instead, which is amortised constant per
73/// run. That is the shape a scan hands over as its positions move forward chunk after chunk. A chunk
74/// that starts before the last stored run can split a run or reach back past it, so it still costs
75/// the full merge.
76///
77/// Either way the result is sorted, disjoint, and has the runs that touch joined, which is what a
78/// look-up by [`Run::overlaps`] on the runs stored here relies on.
79pub(crate) fn absorb(stored: &mut Vec<Run>, incoming: &[Run]) {
80    if incoming.is_empty() {
81        return;
82    }
83    if stored
84        .last()
85        .is_none_or(|last| incoming[0].start >= last.end)
86    {
87        append_runs(stored, incoming);
88    } else {
89        merge_runs(stored, incoming);
90    }
91}
92
93/// Appends `incoming` to `stored`, all of which lies past the last stored run.
94///
95/// Nothing already stored has to be placed again: the two sides are each in position order and
96/// nothing `incoming` holds can reach back into what is stored, so its runs are pushed on the end and
97/// the vector's own growth is the only move — amortised over the runs appended. Where its first run
98/// touches the end of the last stored run the two are one run, as they are after a merge, so it is
99/// joined rather than left beside it.
100fn append_runs(stored: &mut Vec<Run>, incoming: &[Run]) {
101    let mut incoming = incoming;
102    if let Some(last) = stored.last_mut()
103        && let Some(first) = incoming.first()
104        && first.start <= last.end
105    {
106        last.end = last.end.max(first.end);
107        incoming = &incoming[1..];
108    }
109    stored.extend_from_slice(incoming);
110}

112/// Merges `incoming` into `stored`, where the two can interleave.
113///
114/// One pass is enough because each side is sorted and disjoint: whichever run starts first is
115/// placed, and it joins the run before it where the two touch.
116fn merge_runs(stored: &mut Vec<Run>, incoming: &[Run]) {
117    #[cfg(test)]
118    ABSORB_REBUILDS.with(|rebuilds| rebuilds.set(rebuilds.get() + 1));
119    let mut merged: Vec<Run> = Vec::with_capacity(stored.len() + incoming.len());
120    let (mut a, mut b) = (0, 0);
121    while a < stored.len() || b < incoming.len() {
122        let larger = match (stored.get(a), incoming.get(b)) {
123            (Some(left), Some(right)) if right.start < left.start => {
124                b += 1;
125                *right
126            }
127            (Some(left), _) => {
128                a += 1;
129                *left
130            }
131            (None, Some(right)) => {
132                b += 1;
133                *right
134            }
135            (None, None) => break,
136        };
137        match merged.last_mut() {
138            Some(last) if larger.start <= last.end => last.end = last.end.max(larger.end),
139            _ => merged.push(larger),
140        }
141    }
142    *stored = merged;
143}
