81/// A chunk on an internal edge.
82///
83/// A source hands over raw samples; a node above a window hands over evaluated chunks or group
84/// state, as the bound tree says.
85enum NativeChunk {
86    /// Raw source samples, with their own timestamps.
87    Raw(RawSeriesChunk),
88    /// Evaluated values, addressed on their absolute grid positions.
89    Evaluated(EvaluatedChunk),
90    /// Partial aggregate state, addressed on its group's absolute grid positions.
91    AggregateState(StateChunk),
92}
93
94impl NativeChunk {
95    /// The positions this chunk carries: decided grid positions, or raw timestamps.
96    fn positions(&self) -> usize {
97        match self {
98            Self::Raw(chunk) => chunk.time.len(),
99            Self::Evaluated(chunk) => chunk.series.time.len(),
100            Self::AggregateState(StateChunk::Series { series, .. }) => {
101                series.iter().map(|s| s.time.len()).sum()
102            }
103            Self::AggregateState(StateChunk::Row(row)) => row.time.len(),
104        }
105    }
106
107    /// What this chunk is, for a rejection that has to say which kind arrived.
108    fn describe(&self) -> &'static str {
109        match self {
110            Self::Raw(_) => "raw samples",
111            Self::Evaluated(_) => "an evaluated chunk",
112            Self::AggregateState(_) => "aggregate state",
113        }
114    }
115
116    /// Whether this chunk closes the source series whose data it carries.
117    ///
118    /// Not the label set it holds: two source series can share one output label set, and each ends
119    /// on its own chunk.
120    fn is_last(&self) -> bool {
121        match self {
122            Self::Raw(chunk) => chunk.is_last,
123            Self::Evaluated(chunk) => chunk.is_last,
124            // A state chunk is what a partition's fold produces once its input ended, so it only
125            // says the producer is done with these groups -- not that a group is complete; a
126            // `Final` closes a group on the end of its whole input, not on one chunk.
127            Self::AggregateState(_) => true,
128        }
129    }
130
131    /// An estimate of the bytes this chunk holds.
132    ///
133    /// Only what the chunk alone stands for, so a shared buffer or table is counted once per chunk.
134    fn bytes(&self) -> usize {
135        match self {
136            Self::Raw(chunk) => raw_bytes(chunk),
137            Self::Evaluated(chunk) => evaluated_bytes(chunk),
138            Self::AggregateState(chunk) => state_bytes(chunk),
139        }
140    }
141}

143/// The bytes a raw chunk holds: its identity, its timestamps and the Arrow buffers behind its
144/// fields.
145///
146/// A histogram field costs its column of indices, not one float per row.
147pub(crate) fn raw_bytes(chunk: &RawSeriesChunk) -> usize {
148    std::mem::size_of::<RawSeriesChunk>()
149        .saturating_add(buffer_bytes(&chunk.primary_key))
150        .saturating_add(chunk.metric.as_ref().map_or(0, metric_bytes))
151        .saturating_add(buffer_bytes(chunk.time.as_buffer().inner()))
152        .saturating_add(fields_bytes(&chunk.fields))
153}

162/// The bytes one chunk holds beside its samples: its axis, the runs that axis is made of, one
163/// descriptor per segment and its labels.
164///
165/// It is what is left of [`series_bytes`] once the allocations its samples are slices of are taken
166/// out and the `Series` struct itself is left to whatever holds the chunk: the struct is one slot of
167/// the container a retainer keeps its chunks in, so a retainer that counts that container -- a
168/// `quantile` counts its vector by capacity -- has already counted it, and one that does not count
169/// it anywhere would have to add it back.
170///
171/// This is the figure of a retainer that bills the allocations its samples live in itself
172/// ([`unbilled_sample_buffers`]): what the chunk holds on its own, plus that residency, once each.
173pub(crate) fn chunk_bytes_beside_samples(series: &Series) -> usize {
174    series_bytes(series)
175        .saturating_sub(fields_bytes(&series.fields))
176        .saturating_sub(std::mem::size_of::<Series>())
177}

179/// The bytes one chunk a binary operand retains holds on its own.
180///
181/// It is [`chunk_bytes_beside_samples`] with the samples a chunk's rows occupy in place of the
182/// allocations they are slices of, and with the struct the chunk's `Series` carries: a series
183/// delivered in a thousand chunks holds a thousand slices of one buffer, so counting that buffer
184/// once per chunk would make the operand's figure grow with the number of chunks rather than with
185/// the data in them. The slices of one series add up to the allocation they share, which is the
186/// figure a deduplication across the chunks would arrive at.
187///
188/// The struct is part of this figure because an operand keeps one entry per retained chunk and
189/// counts no container of its own, so the entry and the chunk in it are one figure here. A retainer
190/// that does count its container takes [`chunk_bytes_beside_samples`] instead.
191///
192/// The histogram tables a chunk's segments reference are not part of the figure: a table is one
193/// allocation however many chunks index into it, so it is not a per-chunk cost at all. Dropping it
194/// here is therefore only half the story -- it must not become a figure the operand never charges.
195/// [`unbilled_tables`] is the other half, and the operand bills each table once through it.
196pub(crate) fn operand_chunk_bytes(series: &Series) -> usize {
197    chunk_bytes_beside_samples(series)
198        .saturating_add(std::mem::size_of::<Series>())
199        .saturating_add(sample_slice_bytes(&series.fields))
200}

202/// The bytes the samples of one series' fields take as the slices they are.
203///
204/// A dense segment is counted by its rows and a native-histogram field by its index column; a nested
205/// segment is left to the chunks it is made of.
206fn sample_slice_bytes(fields: &SeriesFields) -> usize {
207    fields
208        .iter()
209        .flat_map(|field| field.segments())
210        .map(|segment| match segment.dense() {
211            Some(DenseSamplesRef::Float(values)) => {
212                values.len().saturating_mul(std::mem::size_of::<f64>())
213            }
214            Some(DenseSamplesRef::Histogram(histograms)) => histograms
215                .indices()
216                .len()
217                .saturating_mul(std::mem::size_of::<u32>()),
218            Some(DenseSamplesRef::Mixed(mixed)) => mixed
219                .floats()
220                .len()
221                .saturating_mul(std::mem::size_of::<f64>())
222                .saturating_add(
223                    mixed
224                        .histograms()
225                        .indices()
226                        .len()
227                        .saturating_mul(std::mem::size_of::<u32>()),
228                ),
229            // `segments` hands out dense segments only, so a nested chunk never reaches here.
230            None => 0,
231        })
232        .fold(0usize, usize::saturating_add)
233}

235/// The Arrow allocations `series`' samples live in that `billed` does not hold yet, and the bytes
236/// they hold.
237///
238/// [`sample_slice_bytes`] counts the rows a chunk holds, which is the figure of a caller that bills
239/// the samples without the allocations they are slices of. That is exact only when the chunks of a
240/// series are non-overlapping slices that add up to the buffers they point into. It is wrong in both
241/// directions otherwise: a chunk holding one row of a large batch keeps the whole buffer resident
242/// while its rows are eight bytes, and two chunks whose slices overlap hold the bytes between them
243/// once while each is charged for them. What is resident is the allocation, and an allocation is one
244/// however many chunks slice it, so it is counted once, by the first chunk to reference it --
245/// exactly as [`unbilled_tables`] counts a histogram table.
246///
247/// Counting every chunk for the whole buffer it slices would need no identity at all, but it
248/// over-estimates by the number of chunks: a series handed over as a thousand slices of one buffer
249/// would be charged that buffer a thousand times, which refuses an input whose buffers do fit the
250/// pool. The identity is therefore read, and read from Arrow's own handle on the allocation
251/// ([`Buffer::data_ptr`]): every slice of one buffer reports the address of the allocation it points
252/// into, whatever its offset inside it, and the address names that allocation for as long as it is
253/// remembered -- a chunk referencing a buffer keeps it alive, so no later allocation can be given
254/// the same address. The bytes are the buffer's capacity, which is the allocation and not the slice,
255/// plus the one slot the identity takes in the set that remembers it. The caller counts the figure
256/// as it stands and adds the identities to `billed` once the chunk is retained, which is what makes
257/// the two halves agree.
258///
259/// Every array a chunk holds samples in contributes: the values of a float segment, the indices of a
260/// histogram one, both of a mixed one, and the validity bitmap beside each of them -- a bitmap is an
261/// allocation of its own, and no figure that counts rows can see it. A buffer two segments or two
262/// fields of the one chunk reference is listed once.
263pub(crate) fn unbilled_sample_buffers(
264    series: &Series,
265    billed: &HashSet<usize>,
266) -> (Vec<usize>, usize) {
267    let mut fresh: Vec<usize> = Vec::new();
268    let mut bytes = 0usize;
269    for field in series.fields.iter() {
270        for segment in field.segments() {
271            match segment.dense() {
272                Some(DenseSamplesRef::Float(values)) => {
273                    unbilled_array_buffers(values, &mut fresh, billed, &mut bytes)
274                }
275                Some(DenseSamplesRef::Histogram(histograms)) => {
276                    unbilled_array_buffers(histograms.indices(), &mut fresh, billed, &mut bytes)
277                }
278                Some(DenseSamplesRef::Mixed(mixed)) => {
279                    unbilled_array_buffers(mixed.floats(), &mut fresh, billed, &mut bytes);
280                    unbilled_array_buffers(
281                        mixed.histograms().indices(),
282                        &mut fresh,
283                        billed,
284                        &mut bytes,
285                    );
286                }
287                // `segments` hands out dense segments only, so a nested chunk never reaches here.
288                None => {}
289            }
290        }
291    }
292    (fresh, bytes)
293}

295/// Adds the allocation a sample array's values live in, and the one its validity bitmap lives in,
296/// to `fresh` and `bytes`, skipping the ones `billed` holds and the ones already reported.
297fn unbilled_array_buffers<T: ArrowPrimitiveType>(
298    values: &PrimitiveArray<T>,
299    fresh: &mut Vec<usize>,
300    billed: &HashSet<usize>,
301    bytes: &mut usize,
302) {
303    unbilled_buffer(values.values().inner(), fresh, billed, bytes);
304    if let Some(nulls) = values.nulls() {
305        unbilled_buffer(nulls.buffer(), fresh, billed, bytes);
306    }
307}

309/// Adds the allocation `buffer` is a slice of to `fresh` and `bytes`, once.
310fn unbilled_buffer(
311    buffer: &Buffer,
312    fresh: &mut Vec<usize>,
313    billed: &HashSet<usize>,
314    bytes: &mut usize,
315) {
316    let held = buffer_bytes(buffer);
317    if held == 0 {
318        // An array whose values are all valid carries an empty bitmap, which owns nothing.
319        return;
320    }
321    let identity = buffer.data_ptr().as_ptr() as usize;
322    if billed.contains(&identity) || fresh.contains(&identity) {
323        return;
324    }
325    fresh.push(identity);
326    *bytes = bytes
327        .saturating_add(held)
328        .saturating_add(std::mem::size_of::<usize>());
329}

331/// The histogram tables `series` is the first to reference, and the bytes they hold.
332///
333/// A table is one allocation the chunks of a series share, so it is neither a per-chunk cost --
334/// counting it per chunk would make a series delivered in a thousand chunks cost a thousand tables
335/// -- nor no cost at all, which is what leaving it out of every chunk leaves the figure at. What it
336/// is, is a cost of the operand that first retains a chunk referencing it: the identities it returns
337/// are the tables `billed` does not hold yet, and the figure is what those tables hold
338/// ([`table_bytes`]). The caller bills the figure as it stands and adds the identities to `billed`
339/// once the chunk is retained, which is what makes the two halves of the figure agree.
340///
341/// The identity is the address of the allocation behind the `Arc`. It names one table for as long as
342/// it is remembered, because a chunk referencing a table keeps it alive and no later allocation can
343/// be given the same address. A table two segments or two fields of the one chunk reference is
344/// listed once. Each identity is also one slot in the set that remembers it, which is charged as it
345/// is written -- one entry at a time, like the boundary list's, and not from the capacity the set
346/// grows to -- so remembering a table is paid for beside the table itself.
347///
348/// Both nodes that retain chunks instead of folding them read this: a binary operand and a
349/// `quantile` input. Neither counts a table per chunk, and neither lets it go unbilled.
350pub(crate) fn unbilled_tables(series: &Series, billed: &HashSet<usize>) -> (Vec<usize>, usize) {
351    let mut fresh: Vec<usize> = Vec::new();
352    let mut bytes = 0usize;
353    for field in series.fields.iter() {
354        for segment in field.segments() {
355            let table = match segment.dense() {
356                Some(DenseSamplesRef::Histogram(histograms)) => histograms.batch(),
357                Some(DenseSamplesRef::Mixed(mixed)) => mixed.histograms().batch(),
358                _ => continue,
359            };
360            let identity = Arc::as_ptr(table) as usize;
361            if billed.contains(&identity) || fresh.contains(&identity) {
362                continue;
363            }
364            fresh.push(identity);
365            bytes = bytes
366                .saturating_add(table_bytes(table))
367                .saturating_add(std::mem::size_of::<usize>());
368        }
369    }
370    (fresh, bytes)
371}

373/// A rough workspace estimate for merging fragments that hold `rows` rows between them.
374///
375/// An estimate, not an allocation bound; the payload of values and histogram tables stays shared
376/// with the fragments.
377pub(crate) fn merge_workspace_bytes(rows: usize, fragments: usize, fields: usize) -> usize {
378    let sorted_rows = rows.saturating_mul(
379        std::mem::size_of::<(usize, usize, usize)>() + std::mem::size_of::<usize>(),
380    );
381    let readers = fragments.saturating_mul(
382        fields.saturating_mul(std::mem::size_of::<Samples>() + 4 * std::mem::size_of::<usize>()),
383    );
384    // Allow for run and flattened-field descriptors when rows alternate between fragments or
385    // missing values; this is an estimate rather than a count of the allocations made.
386    let merged_field_per_row =
387        2 * std::mem::size_of::<Samples>() + 3 * std::mem::size_of::<usize>();
388    let merged_fields = rows
389        .saturating_mul(fields)
390        .saturating_mul(merged_field_per_row);
391    let output = fragments.saturating_mul(
392        std::mem::size_of::<Series>() + fields.saturating_mul(std::mem::size_of::<Samples>()),
393    );
394    sorted_rows
395        .saturating_add(readers)
396        .saturating_add(merged_fields)
397        .saturating_add(output)
398}

429struct OpenSeries {
430    /// The identity of the open series, as the chunk that opened it named it.
431    metric: SeriesLabels,

433    last_position: Option<usize>,
434}

436/// The input a [`RuntimeOp::Window`] or a [`RuntimeOp::Merge`] collects until its input ends.
437///
438/// Neither node can answer one label set at a time: the points a window across the boundary needs, and
439/// the fragments a canonical merge combines, are those of closed series. The fragments are kept whole,
440/// merged into one series per label set ([`crate::grid_series::merge_same_labelset`]), and only then
441/// evaluated or emitted.
442///
443/// Collecting is also where the per-series contract is checked: the merge that follows would sort a
444/// stream that broke it into a plausible answer.
445struct MergeInput {
446    /// The schema the fragments are addressed on: their grid and their field count.
447    schema: Arc<SeriesSchema>,
448    /// The series received so far, in the order they arrived.
449    fragments: Vec<Series>,
450    /// An estimate of the bytes the fragments and the merge that follows them stand for.
451    bytes: usize,
452    /// The series the next chunk has to continue, if the last chunk this buffer took did not close
453    /// one. Nothing is kept once a series closed: the next chunk, if any, starts another series.
454    open: Option<OpenSeries>,
455}
456
457impl MergeInput {
458    /// A buffer for the fragments of one window's input, which are addressed on `schema`.
459    fn new(schema: Arc<SeriesSchema>) -> Self {
460        Self {
461            schema,
462            fragments: Vec::new(),
463            bytes: 0,
464            open: None,
465        }
466    }
467
468    /// Checks one received chunk against the contract a series keeps, and follows the open series.
469    ///
470    /// A chunk must match the input grid and schema, hold positions as its axis places rows,
471    /// continue the open series' identity and only move forward inside it. A chunk carrying `is_last`
472    /// closes the open series whether or not it holds rows.
473    fn check(&mut self, series: &Series, is_last: bool) -> Result<()> {
474        let on_grid = match (&self.schema.eval_grid, series.time.grid()) {
475            (Some(input), Some(grid)) => input.spec() == grid.spec(),
476            _ => false,
477        };
478        if !on_grid
479            || series.fields.len() != self.schema.fields.len()
480            || series
481                .fields
482                .iter()
483                .any(|field| field.len() != series.time.len())
484        {
485            return Err(DataFusionError::Execution(
486                "merge input does not match its input schema".to_string(),
487            ));
488        }
489        let (first, last) = (series.time.first_position(), series.time.last_position());
490        if !is_last && first.is_none() {
491            return Err(DataFusionError::Execution(
492                "merge input has no position before the last run of its series".to_string(),
493            ));
494        }
495        if let Some(open) = &self.open
496            && open.metric != series.metric
497        {
498            return Err(DataFusionError::Execution(
499                "merge input changed identity before its last run".to_string(),
500            ));
501        }
502        if let Some(previous) = self.open.as_ref().and_then(|open| open.last_position)
503            && let Some(first) = first
504            && first <= previous
505        {
506            return Err(DataFusionError::Execution(format!(
507                "merge input runs are not increasing: {first} after {previous}"
508            )));
509        }
510        self.open = if is_last {
511            None
512        } else {
513            Some(OpenSeries {
514                metric: series.metric.clone(),
515                last_position: last,
516            })
517        };
518        Ok(())
519    }
520
521    /// The bytes of the fragments and of the workspace merging them needs.
522    ///
523    /// The sample payload is shared with the fragments, so it is not counted twice here.
524    fn workspace_bytes(&self) -> usize {
525        let rows: usize = self
526            .fragments
527            .iter()
528            .map(|fragment| fragment.time.len())
529            .sum();
530        merge_workspace_bytes(rows, self.fragments.len(), self.schema.fields.len())
531    }
532}

534/// What every node holds: its metric handles and the bytes it reserves.
535///
536/// The reservation is the pool's, so a node that buffers input competes with every other operator in the
537/// query, and it is the figure itself: a node holds nothing beside what it reserved.
538//
539// Visible to the aggregate node so that its fold can reserve for what it is about to allocate.
540pub(super) struct NodeBuffer {
541    metrics: Arc<NodeMetrics>,
542    reservation: MemoryReservation,
543}
544
545impl NodeBuffer {
546    /// Registers a reservation for the node called `name` with `pool`.
547    ///
548    /// The name identifies the operator and its tree location, such as `native.window at
549    /// root.left`, for errors and the pool's report. It labels the reservation, it does not key it.
550    pub(super) fn new(
551        name: impl Into<String>,
552        metrics: Arc<NodeMetrics>,
553        pool: &Arc<dyn MemoryPool>,
554    ) -> Self {
555        let consumer = MemoryConsumer::new(name.into());
556        Self {
557            metrics,
558            reservation: consumer.register(pool),
559        }
560    }
561
562    /// Reserves `bytes` beside what this node already holds.
563    pub(super) fn grow(&mut self, bytes: usize) -> Result<()> {
564        self.reservation.try_grow(bytes)?;
565        self.record();
566        Ok(())
567    }
568
569    /// Reserves exactly `bytes`, which is what this node holds from now on: a node that consumes its
570    /// input chunk by chunk is set from the state that is left, not grown per chunk.
571    pub(super) fn resize(&mut self, bytes: usize) -> Result<()> {
572        self.reservation.try_resize(bytes)?;
573        self.record();
574        Ok(())
575    }
576
577    /// Gives `bytes` of what this node holds back to the pool.
578    fn shrink(&mut self, bytes: usize) -> Result<()> {
579        self.reservation.try_shrink(bytes)?;
580        self.record();
581        Ok(())
582    }
583
584    /// Moves `bytes` of what this node holds into a reservation of its own.
585    ///
586    /// The bytes stay charged to the pool; only who holds them changes.
587    fn transfer(&mut self, bytes: usize) -> Result<MemoryReservation> {
588        // A chunk is queued with the figure this node grew by, so the two agree unless the
589        // accounting drifted. Refusing is what keeps a drift from either panicking inside the
590        // memory path or handing out a chunk nobody is charged for.
591        if bytes > self.reservation.size() {
592            return Err(DataFusionError::Internal(format!(
593                "a queued chunk holds {bytes} bytes, more than the {} its node reserves",
594                self.reservation.size()
595            )));
596        }
597        let reservation = self.reservation.split(bytes);
598        self.record();
599        Ok(reservation)
600    }
601
602    /// Reports what this node now reserves.
603    fn record(&self) {
604        self.metrics.record_reserved_bytes(self.reservation.size());
605    }
606}

608impl Drop for NodeBuffer {
609    fn drop(&mut self) {
610        self.reservation.free();
611        self.record();
612    }

615/// The positions one node has handed out, per output label set and field.
616///
617/// A node that rewrites labels -- a `Map` dropping the metric name, a `Window` dropping the name and the
618/// empty-valued labels -- can turn two input series into one label set, and the node above would read the
619/// two chunks as one series. Recording which positions of each field were already handed out lets a second
620/// series onto a set be refused before it is evaluated.
621///
622/// Occupancy is per field and per position, and a chunk holding no value occupies nothing. A set is never
623/// retired: `is_last` closes a source series, not a label set. The table is one node's own.
624#[derive(Default)]
625struct Emitted {
626    sets: HashMap<LabelSet, History>,
627    /// How many run vectors, runs and label payload bytes every entry holds, folded in as the entries
628    /// change rather than by walking them, so [`Emitted::bytes`] is read in constant time.
629    counts: Counts,
630}
631
632/// What the entries of an [`Emitted`] table hold, as running totals.
633#[derive(Default)]
634struct Counts {
635    /// The run vectors, one per field of one entry.
636    vectors: usize,
637    /// The occupied runs in every vector.
638    runs: usize,
639    /// The payload of every label set: the bytes of its names and values with one row per label.
640    payload: usize,
641}
642
643/// What one node has handed out under one output label set.
644#[derive(Default)]
645struct History {
646    /// The occupied runs of each field, in position order and disjoint.
647    fields: Vec<Vec<Run>>,
648}
649
650/// The bytes a label set holds: each label's name and value with the pair that refers to them.
651fn label_bytes(labels: &LabelSet) -> usize {
652    labels
653        .iter()
654        .map(|(name, value)| std::mem::size_of::<(Arc<str>, Arc<str>)>() + name.len() + value.len())
655        .sum()
656}
657
658/// The bytes a metric name holds: its name and each tag's name and value, one row per label, which is
659/// the convention [`label_bytes`] keeps.
660fn metric_bytes(metric: &SeriesLabels) -> usize {
661    metric.name().map_or(0, |name| name.len())
662        + std::mem::size_of_val(metric.tags())
663        + metric
664            .tags()
665            .iter()
666            .map(|tag| tag.name.len() + tag.value.len())
667            .sum::<usize>()
668}
669
670impl Emitted {
671    /// Records what `chunk` occupies, rejecting it against what this node already handed out.
672    ///
673    /// Two values of one field at one position is the only thing refused: a second series that fills
674    /// only the positions the first one left empty merges into it.
675    fn occupy(&mut self, chunk: &EvaluatedChunk) -> Result<()> {
676        let labels = label_set(&chunk.series.metric);
677        let occupied: Vec<Vec<Run>> = chunk
678            .series
679            .fields
680            .iter()
681            .enumerate()
682            .map(|(index, field)| occupied_runs(index, field, &chunk.series.time))
683            .collect::<Result<_>>()?;
684        let empty = occupied.iter().all(Vec::is_empty);
685        if let Some(history) = self.sets.get(&labels) {
686            for (index, runs) in occupied.iter().enumerate() {
687                let stored = history.fields.get(index).map_or(&[][..], Vec::as_slice);
688                for run in runs {
689                    // The stored runs are sorted and disjoint, so only the one holding `run.start`
690                    // or the one just before it can share a position.
691                    let at = stored.partition_point(|stored| stored.end <= run.start);
692                    let previous = stored.get(at).filter(|stored| stored.overlaps(*run));
693                    if let Some(previous) = previous {
694                        return Err(occupied_position(&labels, index, run, previous));
695                    }
696                }
697            }
698        }
699        if empty {
700            // A chunk holding no value records nothing and closes nothing: a series that answered
701            // empty leaves every earlier series on this label set exactly where it was.
702            return Ok(());
703        }
704        let fields = chunk.series.fields.len();
705        let history = match self.sets.entry(labels) {
706            Entry::Occupied(entry) => entry.into_mut(),
707            Entry::Vacant(entry) => {
708                self.counts.payload += label_bytes(entry.key());
709                entry.insert(History::default())
710            }
711        };
712        if history.fields.len() < fields {
713            self.counts.vectors += fields - history.fields.len();
714            history.fields.resize_with(fields, Vec::new);
715        }
716        for (index, runs) in occupied.into_iter().enumerate() {
717            let stored = &mut history.fields[index];
718            let before = stored.len();
719            absorb(stored, &runs);
720            self.counts.runs = self.counts.runs - before + stored.len();
721        }
722        Ok(())
723    }
724
725    /// An estimate of the bytes this table holds, which the node reserves for.
726    ///
727    /// Read from running totals, so it is constant time; the map's buckets and a vector's slack are
728    /// left out.
729    fn bytes(&self) -> usize {
730        let slots =
731            self.sets.len() * (std::mem::size_of::<LabelSet>() + std::mem::size_of::<History>());
732        slots
733            + self.counts.vectors * std::mem::size_of::<Vec<Run>>()
734            + self.counts.runs * std::mem::size_of::<Run>()
735            + self.counts.payload
736    }
737}

739/// The runs of positions holding a value in field `index`, as the axis `time` places its rows.
740///
741/// A missing sample holds no value and a hole between rows is not a position, so a run ends where a hole
742/// begins. Every field must hold one sample per row the axis places.
743fn occupied_runs(index: usize, field: &Samples, time: &TimeAxis) -> Result<Vec<Run>> {
744    let mut runs: Vec<Run> = Vec::new();
745    let mut open: Option<usize> = None;
746    let mut previous: Option<usize> = None;
747    let mut cursor = field.cursor();
748    for (position, row) in time.positions() {
749        // The axis names the absolute position of every row, so the hole between two rows is
750        // visible as a jump in `position` while no row of this chunk sits in it.
751        if let Some(previous) = previous
752            && position != previous + 1
753            && let Some(start) = open.take()
754        {
755            runs.push(Run {
756                start,
757                end: previous + 1,
758            });
759        }
760        let sample = cursor
761            .at(row)
762            .ok_or_else(|| misaligned_field(index, time, field))?;
763        let present = !matches!(sample, SampleRef::Missing);
764        match (present, open) {
765            (true, None) => open = Some(position),
766            (false, Some(start)) => {
767                runs.push(Run {
768                    start,
769                    end: position,
770                });
771                open = None;
772            }
773            _ => {}
774        }
775        previous = Some(position);
776    }
777    if let Some(start) = open {
778        // A run still open at the last row ends just past it, not where the field's row count says:
779        // the rows need not be adjacent to the positions above them.
780        runs.push(Run {
781            start,
782            end: previous.map_or(start, |last| last + 1),
783        });
784    }
785    Ok(runs)
786}

788/// The rejection of a series whose axis and fields disagree on how many rows there are.
789///
790/// [`Series`] keeps its fields public, so a struct-literal chunk can carry fewer samples than the
791/// axis places rows.
792fn misaligned_field(index: usize, time: &TimeAxis, field: &Samples) -> DataFusionError {
793    DataFusionError::Execution(format!(
794        "a series was handed to the node whose axis and fields are misaligned: the axis places {} \
795         rows, but field {index} holds {} samples, so the axis names a row the field does not have",
796        time.len(),
797        field.len(),
798    ))
799}

808/// What an instant node holds: the chunks of the one series it is selecting, until that series ends.
809///
810/// [`select_instants`] takes a complete source axis, so this node buffers a whole series and hands one
811/// chunk over for it.
812struct InstantRuntime {
813    /// The bound selection: how far back it reads, the offset it reads at, and the grid its values
814    /// are addressed on.
815    selection: BoundInstant,
816    /// The layout the chunk it hands out is addressed on: its input's fields on that grid.
817    schema: Arc<SeriesSchema>,
818    /// Where the rows the selection picked are gathered, one series at a time.
819    output: GridSeriesBuilder,
820    /// The chunks of the series being selected, in arrival order; empty while no series is open.
821    chunks: Vec<RawSeriesChunk>,
822    /// What `chunks` hold, added as each chunk arrives and given back when the series is gathered.
823    held: usize,
824}

826impl InstantRuntime {
827    /// The run-time state of the instant selection `op`, reading the samples of `input`.
828    fn try_new(input: &SeriesSchema, op: &BoundInstant) -> Result<Self> {
829        // The chunk this node hands out is addressed on the selection's own grid, so the grid the
830        // rows are gathered on is the binding's rather than a copy of it.
831        let schema = Arc::new(SeriesSchema {
832            fields: Arc::clone(&input.fields),
833            eval_grid: Some(Arc::clone(&op.output_grid)),
834        });
835        Ok(Self {
836            selection: op.clone(),
837            output: GridSeriesBuilder::try_new(Arc::clone(&schema))?,
838            schema,
839            chunks: Vec::new(),
840            held: 0,
841        })
842    }

844    /// The bytes the buffered chunks of the open series hold.
845    fn buffered_bytes(&self) -> usize {
846        self.held
847    }
848
849    /// The bytes gathering the series `closing` ends is about to allocate beside its chunks.
850    ///
851    /// An upper bound derived from what the selection has in hand rather than a measurement: the rows the
852    /// series holds once `closing` has joined it, the fields it carries, the points the selection can decide
853    /// for it, and, per field, the largest histogram a row of it can carry. Each term is one thing the gather
854    /// allocates:
855    ///
856    /// - the concatenated axis ([`buffered_time`]) and the descriptors that read it
857    ///   ([`buffered_fields`]), including the three per segment [`Samples::try_from_chunks`] flattens them
858    ///   into;
859    /// - the two arrays [`select_instants`] returns, one `u64` source row and one `i64` timestamp per point
860    ///   it can reach;
861    /// - the rows [`GridSeriesBuilder`] keeps, including one validity byte per (row, field); a field of native
862    ///   histograms adds its index column and the payload of the histogram it names, charged per *point*,
863    ///   because one source row can be selected by every point;
864    /// - the row the selection fills and one [`RowCursor`] per field;
865    /// - doubling, since every column the builder keeps is a `Vec` that grows by doubling.
866    ///
867    /// The buffered chunks' own headers are not counted: the node's reservation already holds them.
868    fn gather_bytes(&self, closing: &RawSeriesChunk) -> usize {
869        // The points the selection decides, not the points the grid plans: the gather clips its
870        // scan to the series it holds, so a figure over the whole grid would charge the pool for
871        // rows this series can never produce.
872        let points = self.selected_points(closing);
873        let chunks = self.chunks.len().saturating_add(1);
874        let rows = self
875            .chunks
876            .iter()
877            .chain(std::iter::once(closing))
878            .map(|chunk| chunk.time.len())
879            .fold(0usize, usize::saturating_add);
880        let fields = self.schema.fields.len();
881        let descriptor = std::mem::size_of::<Samples>();
882        let segments = self
883            .chunks
884            .iter()
885            .chain(std::iter::once(closing))
886            .flat_map(|chunk| chunk.fields.as_slice())
887            .map(|field| field.segments().len())
888            .fold(0usize, usize::saturating_add);
889        let axis = rows.saturating_mul(std::mem::size_of::<i64>());
890        let layout = chunks
891            .saturating_mul(fields)
892            .saturating_mul(descriptor)
893            .saturating_add(segments.saturating_mul(3).saturating_mul(descriptor))
894            .saturating_add(fields.saturating_mul(std::mem::size_of::<Vec<Samples>>()));
895        let selection =
896            points.saturating_mul(std::mem::size_of::<u64>() + std::mem::size_of::<i64>());
897        let mut rows_of_points = points.saturating_mul(
898            std::mem::size_of::<usize>() + fields.saturating_mul(std::mem::size_of::<f64>() + 1),
899        );
900        for largest in self.largest_histograms(closing).iter().flatten() {
901            rows_of_points = rows_of_points.saturating_add(
902                points.saturating_mul(std::mem::size_of::<Option<u32>>().saturating_add(*largest)),
903            );
904        }
905        let row = fields.saturating_mul(
906            std::mem::size_of::<Option<GridSample>>() + std::mem::size_of::<RowCursor<'_>>(),
907        );
908        axis.saturating_add(layout)
909            .saturating_add(selection)
910            .saturating_add(row)
911            .saturating_add(rows_of_points.saturating_mul(2))
912    }

914    /// The number of candidate grid points [`select_instants`] reserves for the buffered series.
915    ///
916    /// Candidate points can select nothing, but charging the whole planned grid would refuse short
917    /// series that fit the pool.
918    fn selected_points(&self, closing: &RawSeriesChunk) -> usize {
919        let spec = self.selection.output_grid.spec();
920        if self.selection.at_ms.is_some() {
921            return usize::try_from(spec.len).unwrap_or(usize::MAX);
922        }
923        // The selection refuses these itself, before it can decide anything.
924        if spec.step_ms <= 0 || self.selection.lookback_ms < 0 {
925            return 0;
926        }
927        let query_end = grid_end_ms(spec);
928        if query_end < spec.start_ms {
929            return 0;
930        }
931        let series = || self.chunks.iter().chain(std::iter::once(closing));
932        let endpoints = series()
933            .find_map(|chunk| chunk.time.as_buffer().first().copied())
934            .zip(
935                series()
936                    .rev()
937                    .find_map(|chunk| chunk.time.as_buffer().last().copied()),
938            );
939        let span = instant_candidate_span(
940            endpoints,
941            spec.start_ms..=query_end,
942            spec.step_ms,
943            self.selection.lookback_ms,
944            self.selection.offset_ms,
945        );
946        usize::try_from(span.count).unwrap_or(usize::MAX)
947    }

949    /// The largest histogram a row of each field can carry, `None` for a field of floats.
950    ///
951    /// A histogram row is copied into the builder once per grid point that selects it, so the largest
952    /// row is what such a field costs per point.
953    fn largest_histograms(&self, closing: &RawSeriesChunk) -> Vec<Option<usize>> {
954        let mut largest = vec![None; self.schema.fields.len()];
955        for chunk in self.chunks.iter().chain(std::iter::once(closing)) {
956            for (slot, field) in largest.iter_mut().zip(chunk.fields.as_slice()) {
957                if let Some(bytes) = histogram_row_bytes(field) {
958                    *slot = Some(slot.unwrap_or(0).max(bytes));
959                }
960            }
961        }
962        largest
963    }

965    /// Buffers one chunk of the series being selected, and gathers that series when it ends.
966    ///
967    /// `None` means the series is still open: the selection reads one whole axis, so nothing is
968    /// decided for a series until the chunk carrying `is_last` arrives.
969    fn push(&mut self, chunk: RawSeriesChunk) -> Result<Option<EvaluatedChunk>> {
970        if let Some(open) = self.chunks.first()
971            && (open.primary_key.as_slice() != chunk.primary_key.as_slice()
972                || open.metric != chunk.metric)
973        {
974            // One axis is gathered before the kernel reads any of it, so a chunk of another series
975            // has nowhere to go: it would be selected as though its samples were the open series'.
976            return Err(series_changed());
977        }
978        let closing = chunk.is_last;
979        self.held = self.held.saturating_add(raw_bytes(&chunk));
980        self.chunks.push(chunk);
981        if !closing {
982            return Ok(None);
983        }
984        self.gather().map(Some)
985    }

987    /// Selects the instants of the buffered series, hands it over closed, and drops the buffer.
988    fn gather(&mut self) -> Result<EvaluatedChunk> {
989        let chunk = self.select()?;
990        // The selection copied what it needed out of the buffered chunks, so a series that has been
991        // answered must not keep charging the pool for them.
992        self.chunks.clear();
993        self.chunks.shrink_to_fit();
994        self.held = 0;
995        Ok(chunk)
996    }

998    /// The last sample at or before each grid point of the buffered series, as one chunk.
999    fn select(&mut self) -> Result<EvaluatedChunk> {
1000        let Some(first) = self.chunks.first() else {
1001            return Err(DataFusionError::Execution(
1002                "the instant selection holds no series to select".to_string(),
1003            ));
1004        };
1005        let metric = first.metric.clone().ok_or_else(no_labels)?;
1006        // The buffered chunks are one series, so they are read as the one axis the selection takes.
1007        let time = buffered_time(&self.chunks)?;
1008        let fields = buffered_fields(&self.chunks, self.schema.fields.len())?;
1009        let grid = Arc::clone(&self.selection.output_grid);
1010        let spec = grid.spec();
1011        // The grid's own points are what the selection decides, so the range and the step it is
1012        // asked for are the grid's: `start..=start + (len - 1) * step`, step by step.
1013        if let Some(at_ms) = self.selection.at_ms {
1014            let timestamp_window = if self.selection.value == InstantValue::Timestamp
1015                && self.selection.lookback_ms >= 0
1016            {
1017                let anchor = at_ms as i128;
1018                let shifted = anchor - self.selection.offset_ms as i128;
1019                let lower = anchor.max(shifted) - self.selection.lookback_ms as i128;
1020                let upper = anchor.min(shifted);
1021                if lower > upper || (lower == upper && self.selection.lookback_ms != 0) {
1022                    None
1023                } else {
1024                    let effective_lookback = i64::try_from(upper - lower).ok();
1025                    effective_lookback.and_then(|lookback| {
1026                        i64::try_from(upper).ok().map(|upper| (upper, lookback))
1027                    })
1028                }
1029            } else {
1030                Some((at_ms, self.selection.lookback_ms))
1031            };
1032            let Some((selection_at, selection_lookback)) = timestamp_window else {
1033                let metric = drop_metric_name(&metric)?;
1034                return no_row_chunk(&self.schema, metric);
1035            };
1036            let Some(index) = select_instant_at(
1037                &time,
1038                &fields,
1039                selection_at,
1040                selection_lookback,
1041                if self.selection.value == InstantValue::Timestamp {
1042                    0
1043                } else {
1044                    self.selection.offset_ms
1045                },
1046            )?
1047            else {
1048                return no_row_chunk(&self.schema, self.output_metric(&metric)?);
1049            };
1050            let grid_len = usize::try_from(spec.len).unwrap_or(usize::MAX);
1051            if self.selection.value != InstantValue::Sample {
1052                let indices = vec![index; grid_len];
1053                let timestamps = grid
1054                    .timestamps()
1055                    .map_err(crate::grid_series::external)?
1056                    .to_vec();
1057                return self.project(&metric, &time, &fields, &indices, &timestamps);
1058            }
1059            self.output.begin_series();
1060            let mut cursors: Vec<RowCursor<'_>> = fields.iter().map(Samples::cursor).collect();
1061            let mut row = Vec::with_capacity(fields.len());
1062            for cursor in &mut cursors {
1063                row.push(owned(
1064                    cursor
1065                        .at(index as usize)
1066                        .ok_or_else(|| missing_row(index as usize))?,
1067                )?);
1068            }
1069            for position in 0..grid_len {
1070                self.output.push_samples(position, &row)?;
1071            }
1072            return match self.output.finish(metric.clone())? {
1073                Some(series) => EvaluatedChunk::try_new(series, true),
1074                None => no_row_chunk(&self.schema, metric),
1075            };
1076        }
1077        let (indices, timestamps) = select_instants(
1078            &time,
1079            &fields,
1080            spec.start_ms..=grid_end_ms(spec),
1081            spec.step_ms,
1082            self.selection.lookback_ms,
1083            self.selection.offset_ms,
1084        )?;
1085        if self.selection.value != InstantValue::Sample {
1086            return self.project(&metric, &time, &fields, &indices, &timestamps);
1087        }
1088        self.output.begin_series();
1089        // One cursor per field, so a row is read as the field holds it rather than by walking the
1090        // chunks of a field from its start once per row.
1091        let mut cursors: Vec<RowCursor<'_>> = fields.iter().map(Samples::cursor).collect();
1092        // `GridSample` is not `Clone`, so the row is filled slot by slot rather than from one value.
1093        let mut row: Vec<Option<GridSample>> = (0..fields.len()).map(|_| None).collect();
1094        let mut read: Option<usize> = None;
1095        for (index, timestamp) in indices.iter().zip(&timestamps) {
1096            let index = *index as usize;
1097            // The selection steps the grid's range by its step, so every timestamp it reports is a
1098            // point of the grid and the position is the point's index in it.
1099            let position = grid_position(&grid, *timestamp)?;
1100            // Two grid points can select one source row, and the row is read once for both.
1101            if read != Some(index) {
1102                for (slot, cursor) in row.iter_mut().zip(&mut cursors) {
1103                    let sample = cursor.at(index).ok_or_else(|| missing_row(index))?;
1104                    *slot = owned(sample)?;
1105                }
1106                read = Some(index);
1107            }
1108            // A position no field holds a value at is not a row: the builder drops it, which is
1109            // what keeps a hole in the grid out of the result instead of padding it.
1110            self.output.push_samples(position, &row)?;
1111        }
1112        // The rows the selection picked are the rows this series holds, and the whole series is
1113        // decided at once, so this one chunk closes it.
1114        match self.output.finish(metric.clone())? {
1115            Some(series) => EvaluatedChunk::try_new(series, true),
1116            None => no_row_chunk(&self.schema, metric),
1117        }
1118    }

1128    /// The sample times of a selection, for a selection that reports them rather than the samples.
1129    fn project(
1130        &self,
1131        metric: &SeriesLabels,
1132        time: &RawTimestamps,
1133        fields: &SeriesFields,
1134        indices: &[u64],
1135        timestamps: &[i64],
1136    ) -> Result<EvaluatedChunk> {
1137        let projected = match self.selection.value {
1138            InstantValue::Timestamp => {
1139                timestamp_of_selection(&self.schema, metric, time, fields, indices, timestamps)?
1140            }
1141            InstantValue::InfoTimestamp => {
1142                info_sample_times(&self.schema, metric, time, fields, indices, timestamps)?
1143            }
1144            InstantValue::Sample => {
1145                return Err(DataFusionError::Internal(
1146                    "an instant selection of samples has no sample times to project".to_string(),
1147                ));
1148            }
1149        };
1150        match projected {
1151            Some(series) => EvaluatedChunk::try_new(series, true),
1152            None => no_row_chunk(&self.schema, self.output_metric(metric)?),
1153        }
1154    }
1155}

1157/// An input a node reads whole before it computes -- one operand of a binary node, or the input of a vector
1158/// operation -- chunk by chunk, until it ends.
1159///
1160/// The kernels read whole series, so the chunks are assembled into one series per logical series once the
1161/// input has ended.
1162#[derive(Default)]
1163struct RetainedInput {
1164    /// The series whose last chunk did not close it, with the position after its last row.
1165    ///
1166    /// `None` when no series is open. A label set may be closed more than once, so a chunk that starts a
1167    /// series again is not an error; one that starts it while another series is open, or goes back inside
1168    /// the open one, is.
1169    open: Option<(SeriesLabels, usize)>,
1170    /// The chunks this operand handed over, in arrival order.
1171    series: Vec<Series>,
1172    /// The chunk count at which each logical series closed, exclusive, in arrival order.
1173    ///
1174    /// A label set that closes and starts again is two logical series, which a list of ends keeps
1175    /// apart.
1176    ends: Vec<usize>,
1177    /// The histogram tables the chunks this operand holds index into, by the address of the allocation
1178    /// behind their `Arc`.
1179    ///
1180    /// A table is one allocation however many chunks index into it, so it is billed once, by the first chunk
1181    /// that references it. A chunk's own figure leaves its tables out ([`operand_chunk_bytes`]), so a table
1182    /// neither this operand nor this set bills is a table no reservation covers.
1183    tables: HashSet<usize>,
1184    /// The bytes `series` and `ends` hold, folded in as chunks arrive.
1185    ///
1186    /// Each chunk is measured once by [`operand_chunk_bytes`], and shared histogram tables are billed once
1187    /// through [`unbilled_tables`].
1188    held: usize,
1189}
1190
1191impl RetainedInput {
1192    /// The bytes `chunk` would add to what this operand holds, read without retaining it.
1193    ///
1194    /// The same figure [`Self::enqueue`] adds, so a caller reserves it before retaining the chunk and the pool
1195    /// admits the chunk before the operand holds it.
1196    fn pending(&self, chunk: &EvaluatedChunk) -> usize {
1197        let end = if chunk.is_last {
1198            std::mem::size_of::<usize>()
1199        } else {
1200            0
1201        };
1202        let (_, tables) = unbilled_tables(&chunk.series, &self.tables);
1203        tables
1204            .saturating_add(operand_chunk_bytes(&chunk.series))
1205            .saturating_add(end)
1206    }
1207
1208    /// Retains one chunk after checking that it continues what this input handed over.
1209    ///
1210    /// A chunk that starts another series while one is open, or goes back inside the open one, is
1211    /// refused before it is retained.
1212    fn retain(&mut self, chunk: EvaluatedChunk, reader: Reader) -> Result<()> {
1213        // An empty chunk closes a series without holding a row, so it says nothing about where that
1214        // series ended: only a chunk that carries rows can be read as going backwards.
1215        let (first, last) = (
1216            chunk.series.time.first_position(),
1217            chunk.series.time.last_position(),
1218        );
1219        let metric = chunk.series.metric.clone();
1220        // The position a chunk of the open series may start at: the row after the last one that
1221        // series handed over. Nothing is open after a series closed, so the next one starts over.
1222        let end = self.open.as_ref().map_or(0, |(_, end)| *end);
1223        if let Some((open, _)) = &self.open
1224            && *open != metric
1225        {
1226            let (owner, input) = (reader.owner(), reader.input());
1227            return Err(DataFusionError::Execution(format!(
1228                "{owner} reads whole series, and {input} handed over {metric:?} before {open:?} \
1229                 ended"
1230            )));
1231        }
1232        if let Some(first) = first
1233            && first < end
1234        {
1235            let (owner, input) = (reader.owner(), reader.input());
1236            return Err(DataFusionError::Execution(format!(
1237                "{owner} was handed chunks of {metric:?} by {input} that do not continue each \
1238                 other: one starts at position {first}, after the series already reached {end}"
1239            )));
1240        }
1241        self.open = (!chunk.is_last).then(|| (metric, last.map_or(end, |last| last + 1)));
1242        self.enqueue(chunk);
1243        Ok(())
1244    }
1245
1246    /// Refuses an input that ended inside a series: a chunk that did not say `is_last` left the last
1247    /// series it handed over open, and nothing came after it to close it.
1248    fn ensure_closed(&self, reader: Reader) -> Result<()> {
1249        match &self.open {
1250            Some((metric, _)) => Err(DataFusionError::Execution(format!(
1251                "the input of {} ended before the last chunk of a series: {} handed over \
1252                 {metric:?} without the chunk that closes it",
1253                reader.owner(),
1254                reader.input()
1255            ))),
1256            None => Ok(()),
1257        }
1258    }
1259
1260    /// Retains one chunk, folding the estimate it is charged with into what the operand holds.
1261    fn enqueue(&mut self, chunk: EvaluatedChunk) {
1262        let closing = chunk.is_last;
1263        // The tables this chunk is the first to reference, read before they are recorded: the same
1264        // figure `pending` billed, so what is reserved for the chunk and what it grows the operand
1265        // by are one.
1266        let (tables, bytes) = unbilled_tables(&chunk.series, &self.tables);
1267        // The boundary list is storage the operand keeps, and it grows before the chunk is recorded
1268        // in it: an end is charged for before it exists, the same figure `pending` reported.
1269        self.held = self
1270            .held
1271            .saturating_add(if closing {
1272                std::mem::size_of::<usize>()
1273            } else {
1274                0
1275            })
1276            .saturating_add(operand_chunk_bytes(&chunk.series))
1277            .saturating_add(bytes);
1278        // From here the operand keeps those tables alive itself, so a chunk arriving later that
1279        // references one is charged its own slices and not the table again.
1280        self.tables.extend(tables);
1281        self.series.push(chunk.series);
1282        if closing {
1283            self.ends.push(self.series.len());
1284        }
1285    }
1286
1287    /// The chunk ranges of the logical series this operand closed, in arrival order.
1288    fn boundaries(&self) -> impl Iterator<Item = std::ops::Range<usize>> + '_ {
1289        self.ends.iter().scan(0, |start, &end| {
1290            let range = *start..end;
1291            *start = end;
1292            Some(range)
1293        })
1294    }
1295
1296    /// The bytes assembling this operand's chunks into logical series allocate, before they do.
1297    ///
1298    /// Nothing copies a sample: the descriptors the chunks' fields are made of are collected and shared, so
1299    /// what is counted is metadata -- per series, per chunk and field, per segment and per run the boundary
1300    /// covers:
1301    ///
1302    /// - per series, the assembled entry and the field list ([`SeriesFields`] takes a one-field list out of
1303    ///   its vector and boxes a wider one, so neither copies);
1304    /// - per chunk and field, the descriptor collected out of it;
1305    /// - per segment, three descriptors: the one the flatten copies out, the slot the `Arc<[Samples]>` it is
1306    ///   copied into holds, and the doubling of the vector it is copied out of;
1307    /// - per run the boundary covers, the span [`Self::assemble`] collects and the span and row-offset arrays
1308    ///   [`TimeAxis::from_grid_spans`] copies them into.
1309    fn assembly_bytes(&self, schema: &SeriesSchema) -> Option<usize> {
1310        let samples = std::mem::size_of::<Samples>();
1311        let grids = std::mem::size_of::<GridSpan>();
1312        let width = schema.fields.len();
1313        let mut bytes = 0usize;
1314        for boundary in self.boundaries() {
1315            let chunks = &self.series[boundary];
1316            // The runs the boundary covers, and the segments its fields hold: a chunk's field can
1317            // itself be chunked, and flattening it copies one descriptor per segment.
1318            let mut runs = 0usize;
1319            let mut segments = 0usize;
1320            for chunk in chunks {
1321                runs = runs.checked_add(chunk.time.runs().count())?;
1322                for field in chunk.fields.iter() {
1323                    segments = segments.checked_add(field.segments().len())?;
1324                }
1325            }
1326            // The descriptors of a field's chunks, the flatten's descriptors and the `Arc` they are
1327            // copied into; a field with no segment still builds an empty `Samples`, which is one
1328            // descriptor.
1329            let descriptors = chunks
1330                .len()
1331                .checked_mul(width)?
1332                .checked_add(segments.checked_mul(3)?.checked_add(width)?)?;
1333            // The axis: the spans collected, and the span and offset arrays a run layout copies
1334            // them into.
1335            let axis = runs.checked_mul(
1336                grids
1337                    .checked_mul(2)?
1338                    .checked_add(std::mem::size_of::<usize>())?,
1339            )?;
1340            bytes = bytes
1341                .checked_add(std::mem::size_of::<Series>())?
1342                .checked_add(width.checked_mul(samples)?)?
1343                .checked_add(descriptors.checked_mul(samples)?)?
1344                .checked_add(axis)?;
1345        }
1346        Some(bytes)
1347    }
1348
1349    /// The logical series this operand closed, each as one series on `schema`'s grid.
1350    ///
1351    /// The chunks of a logical series are collected, not concatenated: the fields keep one descriptor
1352    /// per segment of every chunk and the axis keeps the runs the chunks cover, sharing the buffers
1353    /// with the chunks this operand still holds. [`Series::try_new`] checks that the series holds as
1354    /// many rows as its axis places, and [`Self::assembly_bytes`] is what a caller reserves.
1355    fn assemble(&self, schema: &SeriesSchema, reader: Reader) -> Result<Vec<Series>> {
1356        let grid = schema.eval_grid.clone().ok_or_else(|| {
1357            let (owner, input) = (reader.owner(), reader.input());
1358            DataFusionError::Execution(format!(
1359                "{owner} assembles {input} on an evaluation grid, and its schema names none"
1360            ))
1361        })?;
1362        let mut assembled = Vec::with_capacity(self.ends.len());
1363        for boundary in self.boundaries() {
1364            let chunks = &self.series[boundary];
1365            let metric = chunks
1366                .first()
1367                .map(|chunk| chunk.metric.clone())
1368                .ok_or_else(|| {
1369                    let (owner, input) = (reader.owner(), reader.input());
1370                    DataFusionError::Execution(format!(
1371                        "{owner} closed a logical series of {input} without a chunk to assemble"
1372                    ))
1373                })?;
1374            let runs = chunks
1375                .iter()
1376                .map(|chunk| chunk.time.runs().count())
1377                .fold(0usize, usize::saturating_add);
1378            let mut spans = Vec::with_capacity(runs);
1379            for chunk in chunks {
1380                for run in chunk.time.runs() {
1381                    spans.push(GridSpan::new(run.position, run.len));
1382                }
1383            }
1384            let mut fields = Vec::with_capacity(schema.fields.len());
1385            for index in 0..schema.fields.len() {
1386                let mut segments = Vec::with_capacity(chunks.len());
1387                for chunk in chunks {
1388                    let field = chunk.fields.get(index).ok_or_else(|| {
1389                        let (owner, input) = (reader.owner(), reader.input());
1390                        DataFusionError::Execution(format!(
1391                            "{owner} cannot assemble {input}: a chunk of it carries {} fields, not \
1392                             {}",
1393                            chunk.fields.len(),
1394                            schema.fields.len()
1395                        ))
1396                    })?;
1397                    segments.push(field.clone());
1398                }
1399                fields.push(Samples::try_from_chunks(segments).map_err(series_error)?);
1400            }
1401            let fields = SeriesFields::try_from_vec(fields).map_err(series_error)?;
1402            let time = TimeAxis::from_grid_spans(grid.clone(), spans).map_err(series_error)?;
1403            assembled.push(Series::try_new(schema, metric, time, fields).map_err(series_error)?);
1404        }
1405        Ok(assembled)
1406    }
1407
1408    /// Drops the retained chunks, giving back what they hold.
1409    fn release(&mut self) {
1410        self.open = None;
1411        self.series.clear();
1412        self.series.shrink_to_fit();
1413        self.ends.clear();
1414        self.ends.shrink_to_fit();
1415        // The tables the cleared chunks were keeping alive are gone with them, so their addresses
1416        // name nothing this operand holds any more.
1417        self.tables.clear();
1418        self.tables.shrink_to_fit();
1419        self.held = 0;
1420    }
1421}

1423/// What a binary or set node holds: both operands until they end.
1424///
1425/// The operation pairs two whole vectors point by point, so the node retains every chunk each operand
1426/// hands over, assembles each logical series into the one series the kernel matches, and runs the
1427/// kernel once both operands closed.

1429    node_index: usize,
1430    /// Evidence shared with the other output partitions of this fused-series execution.
1431    evidence: Option<Arc<DuplicateEvidence>>,
1432    /// The query's collector for dropped histogram sample pairs, when requested.
1433    annotations: Option<PromqlAnnotationCollector>,
1434    /// The kernel and the modifiers the compiler bound it with.
1435    kernel: PairKernel,
1436    /// The layout each operand is bound to, what a chunk of it is checked against: `0` is the left

1439    /// The retained chunks of each operand, in the same order.

1445/// The kernel a [`BinaryRuntime`] runs over its two assembled operands.

1499impl BinaryRuntime {
1500    /// Identifies this binary node's evidence slot for a field.
1501    fn slot(&self, field: usize) -> (usize, usize) {
1502        (self.node_index, field)
1503    }
1504
1505    /// The run-time state of `kernel` over its two operands.
1506    fn new(
1507        kernel: PairKernel,
1508        schemas: [Arc<SeriesSchema>; 2],
1509        node_index: usize,
1510        evidence: Option<Arc<DuplicateEvidence>>,
1511    ) -> Self {
1512        Self {
1513            node_index,
1514            evidence,
1515            annotations: None,
1516            kernel,
1517            schemas,
1518            operands: [RetainedInput::default(), RetainedInput::default()],
1519            computed: false,
1520        }
1521    }
1522
1523    /// The reader of the operand at `side`, which names this node in its messages.
1524    fn reader(&self, side: usize) -> Reader {
1525        Reader::Operand {
1526            side,
1527            node: self.kernel.node(),
1528        }
1529    }
1530
1531    /// The bytes the retained operands hold.
1532    fn retained_bytes(&self) -> usize {
1533        self.operands
1534            .iter()
1535            .map(|operand| operand.held)
1536            .fold(0usize, usize::saturating_add)
1537    }
1538
1539    /// The bytes assembling both operands into logical series allocates, before it does.
1540    fn assembly_bytes(&self) -> Option<usize> {
1541        let [left, right] = &self.operands;
1542        let [left_schema, right_schema] = &self.schemas;
1543        left.assembly_bytes(left_schema)?
1544            .checked_add(right.assembly_bytes(right_schema)?)
1545    }
1546
1547    /// The logical series both operands closed, ready for the kernel to match.
1548    fn assemble(&self) -> Result<[Vec<Series>; 2]> {
1549        let [left, right] = &self.operands;
1550        let [left_schema, right_schema] = &self.schemas;
1551        Ok([
1552            left.assemble(left_schema, self.reader(0))?,
1553            right.assemble(right_schema, self.reader(1))?,
1554        ])
1555    }
1556
1557    /// The bytes the chunk of `side` would add to that operand's retention.
1558    ///
1559    /// It is read before the chunk is retained, so it is the figure the operand grows by.
1560    fn pending(&self, side: usize, chunk: &EvaluatedChunk) -> Result<usize> {
1561        self.operands
1562            .get(side)
1563            .map(|operand| operand.pending(chunk))
1564            .ok_or_else(invalid_binary_operand)
1565    }
1566
1567    /// Reports an operand that ended inside a series.
1568    ///
1569    /// The kernel is not run over such an operand: it would answer the pairs of a series whose
1570    /// remaining rows were never read.
1571    fn closed(&self) -> Result<()> {
1572        for (side, operand) in self.operands.iter().enumerate() {
1573            operand.ensure_closed(self.reader(side))?;
1574        }
1575        Ok(())
1576    }
1577
1578    /// Retains one chunk of the operand at `side`, checking it against that operand's binding.
1579    fn retain(&mut self, side: usize, chunk: EvaluatedChunk) -> Result<()> {
1580        // What a bound tree says an operand produces is not proof of what a chunk carries: a `Series`
1581        // keeps its fields public, so a struct-literal chunk can carry another field count, or fewer
1582        // samples than its axis places rows. The kernel's own check runs here, before the chunk enters
1583        // the buffer.
1584        let Some(schema) = self.schemas.get(side) else {
1585            return Err(invalid_binary_operand());
1586        };
1587        validate_binary_operand(schema.as_ref(), std::slice::from_ref(&chunk.series)).map_err(
1588            |error| {
1589                DataFusionError::Execution(format!(
1590                    "{} rejected its {} operand: {error}",
1591                    self.kernel.node(),
1592                    side_name(side)
1593                ))
1594            },
1595        )?;
1596        let reader = self.reader(side);
1597        let Some(operand) = self.operands.get_mut(side) else {
1598            return Err(invalid_binary_operand());
1599        };
1600        operand.retain(chunk, reader)
1601    }
1602
1603    /// Runs the kernel over both operands and returns the series it produced.
1604    ///
1605    /// Each operand's chunks are assembled into their logical series first, because the kernel matches
1606    /// whole series by label set: a series in many chunks would otherwise be a candidate per chunk on
1607    /// its own side, which is quadratic in the chunk count of one series.
1608    ///
1609    /// The reservation covers the retained chunks, the assembled series and the kernel's own live
1610    /// workspace, and it keeps covering the returned results until the caller queues them.
1611    fn compute(&mut self, buffer: &mut NodeBuffer) -> Result<Option<Vec<Series>>> {
1612        // The kernel reads the operands once and drops them, so a second call has nothing to read
1613        // and nothing left to produce: what it produced the first time is in the node's queue, and
1614        // the reservation is already the queue's.
1615        if self.computed {
1616            return Ok(None);
1617        }
1618        self.closed()?;
1619        let [left_schema, right_schema] = &self.schemas;
1620        let retained = self.retained_bytes();
1621        // The assembled series are live while the kernel reads them and share the buffers the retained
1622        // chunks hold, so what they add is metadata, and it is reserved before it exists: a figure read
1623        // from it afterwards would be a figure the pool never admitted.
1624        let node = self.kernel.node();
1625        let assembly = self
1626            .assembly_bytes()
1627            .ok_or_else(|| reservation_overflow(node))?;
1628        buffer.grow(assembly)?;
1629        let assembled = match self.assemble() {
1630            Ok(assembled) => assembled,
1631            Err(error) => {
1632                // Nothing the assembly allocated outlives this call, so the node falls back to what
1633                // it still holds. A failure of that resize is the pool's to report; the driver stops
1634                // the execution either way and releases every node.
1635                buffer.resize(retained)?;
1636                return Err(error);
1637            }
1638        };
1639        let [left, right] = &assembled;
1640        // The operands are still held while the kernel runs, so what this node holds is the operands,
1641        // the assembled series and the workspace the kernel reports; the callback resizes the
1642        // reservation to that sum as the live workspace changes.
1643        let metrics = Arc::clone(&buffer.metrics);
1644        let output = {
1645            // The timer covers this node's own work only: it is dropped before the driver polls
1646            // the input again.
1647            let _timer = metrics.timer();
1648            let mut reserve = |workspace: usize| {
1649                let bytes = retained
1650                    .checked_add(assembly)
1651                    .and_then(|held| held.checked_add(workspace))
1652                    .ok_or_else(|| reservation_overflow(node))?;
1653                buffer.resize(bytes)
1654            };
1655            match &self.kernel {
1656                PairKernel::Binary(matching) => {
1657                    let evidence = self.evidence.as_deref();
1658                    binary_vectors_with_reservation(
1659                        matching,
1660                        left_schema,
1661                        left,
1662                        right_schema,
1663                        right,
1664                        evidence.map(|evidence| crate::vector_match::NodeEvidence {
1665                            node: self.slot(0).0,
1666                            evidence,
1667                        }),
1668                        self.annotations.as_ref(),
1669                        &mut reserve,
1670                    )
1671                }
1672                PairKernel::Set(set) => set_vectors_with_reservation(
1673                    set.op,
1674                    &set.labels,
1675                    left_schema,
1676                    left,
1677                    right_schema,
1678                    right,
1679                    &mut reserve,
1680                ),
1681                // The result keeps the base's fields on the base's grid, so the base's layout is
1682                // the one the kernel builds it on.
1683                PairKernel::Info(selector) => {
1684                    info_with_reservation(left_schema, left, right, selector, &mut reserve)
1685                }
1686            }
1687        };
1688        // The operands and the assembled series have been handed to the kernel and the reservation
1689        // covers the workspace it allocated past them, so a failure is the driver's to release: it
1690        // stops the execution and every node with it.
1691        let output = output?;
1692        // The assembled series were the kernel's input and are read by nothing after it, so they go
1693        // before the chunks they share their buffers with: what this node holds from here on is what
1694        // the kernel returned, and the pool is not charged for a vector nothing reads again.
1695        drop(assembled);
1696        for operand in &mut self.operands {
1697            operand.release();
1698        }
1699        self.computed = true;
1700        Ok(Some(output))
1701    }
1702}

1704/// What a vector node holds: its whole input until it ends.
1705///
1706/// Which samples `topk` or `limitk` keeps at a point depends on every series present there, so the node
1707/// retains every chunk and runs its kernel once the input ended, like a binary node.
1708struct VectorRuntime {
1709    op: BoundVectorOp,
1710    /// The layout of the input, which is also what the node produces.
1711    schema: Arc<SeriesSchema>,
1712    input: RetainedInput,
1713    /// Whether the input's pieces of one label set are merged before the kernel reads them, which
1714    /// `limitk` needs when its input can hold them.
1715    merge_input: bool,
1716    /// Whether the kernel has run. It runs at most once: the input it read is gone afterwards.
1717    computed: bool,
1718    annotations: Option<PromqlAnnotationCollector>,
1719    /// The per-point k of a parameterized topk/bottomk, in grid position order.
1720    parameter: Option<Vec<Option<f64>>>,
1721    /// The bytes charged for the parameter values.
1722    parameter_bytes: usize,
1723}
1724
1725impl VectorRuntime {
1726    /// Validates schema-dependent requirements before the native route is selected.
1727    fn try_new(op: BoundVectorOp, schema: Arc<SeriesSchema>) -> Result<Self> {
1728        // Every grouping these operations address their series by is checked here: `topk`,
1729        // `bottomk` and `limitk` key groups by series labels only, so a `by` naming a value field
1730        // would fold series the existing plan keeps apart into one group.
1731        let bound_grouping = match &op {
1732            BoundVectorOp::TopK { group, .. }
1733            | BoundVectorOp::LimitK { group, .. }
1734            | BoundVectorOp::CountValues { group, .. } => Some(group),
1735            _ => None,
1736        };
1737        if let Some(group) = bound_grouping {
1738            let names: Vec<&str> = match group {
1739                Grouping::By(names) | Grouping::Without(names) => {
1740                    names.iter().map(AsRef::as_ref).collect()
1741                }
1742            };
1743            let grouping = match group {
1744                Grouping::By(_) => aggregate::Grouping::By(&names),
1745                Grouping::Without(_) => aggregate::Grouping::Without(&names),
1746            };
1747            match &op {
1748                BoundVectorOp::CountValues { label, .. } => {
1749                    aggregate::validate_count_values(&schema, label, grouping)?
1750                }
1751                _ => aggregate::validate_vector_grouping(&schema, grouping)?,
1752            }
1753        }
1754        Ok(Self {
1755            op,
1756            schema,
1757            input: RetainedInput::default(),
1758            merge_input: false,
1759            computed: false,
1760            annotations: None,
1761            parameter: None,
1762            parameter_bytes: 0,
1763        })
1764    }
1765
1766    /// Runs the kernel over the whole input and returns the series it kept, or `None` when it already
1767    /// ran.
1768    ///
1769    /// The reservation covers the retained chunks, the assembled series and a rough estimate of the
1770    /// kernel's workspace and of the series it rebuilds ([`vector_workspace_bytes`]), which the kernels
1771    /// do not report as they allocate.
1772    fn compute(&mut self, buffer: &mut NodeBuffer) -> Result<Option<Vec<Series>>> {
1773        self.compute_with_histogram_parameters(buffer, &[])
1774    }
1775
1776    fn compute_with_histogram_parameters(
1777        &mut self,
1778        buffer: &mut NodeBuffer,
1779        parameters: &[&[Option<f64>]],
1780    ) -> Result<Option<Vec<Series>>> {
1781        if self.computed {
1782            return Ok(None);
1783        }
1784        self.input.ensure_closed(Reader::Vector)?;
1785        // What a failed assembly leaves the node holding: the input it retains beside the per-point
1786        // values it read its parameters into, which outlive the assembly.
1787        let retained = self.input.held.saturating_add(self.parameter_bytes);
1788        let assembly = self
1789            .input
1790            .assembly_bytes(&self.schema)
1791            .ok_or_else(vector_overflow)?;
1792        buffer.grow(assembly)?;
1793        let assembled = match self.input.assemble(&self.schema, Reader::Vector) {
1794            Ok(assembled) => assembled,
1795            Err(error) => {
1796                buffer.resize(retained)?;
1797                return Err(error);
1798            }
1799        };
1800        // `limitk` keeps the first series in the order its input hands them over, as Prometheus takes
1801        // the order of the vector it is given. Pieces of one label set are one series there, so they are
1802        // merged before they are ranked.
1803        let assembled = if self.merge_input {
1804            let rows = assembled.iter().map(|series| series.time.len()).sum();
1805            buffer.grow(merge_workspace_bytes(
1806                rows,
1807                assembled.len(),
1808                self.schema.fields.len(),
1809            ))?;
1810            merge_same_labelset(&self.schema, assembled)?
1811        } else {
1812            assembled
1813        };
1814        if !parameters.is_empty() && !matches!(self.op, BoundVectorOp::Histogram(_)) {
1815            return Err(DataFusionError::Internal(
1816                "histogram parameters supplied to non-histogram op".into(),
1817            ));
1818        }
1819        let assembled = if !parameters.is_empty() {
1820            let rows = assembled.iter().map(|series| series.time.len()).sum();
1821            buffer.grow(merge_workspace_bytes(
1822                rows,
1823                assembled.len(),
1824                self.schema.fields.len(),
1825            ))?;
1826            merge_same_labelset(&self.schema, assembled)?
1827        } else {
1828            assembled
1829        };
1830        if !parameters.is_empty() {
1831            let grid = self
1832                .schema
1833                .eval_grid
1834                .as_ref()
1835                .ok_or_else(|| DataFusionError::Internal("histogram grid missing".into()))?;
1836            if parameters
1837                .iter()
1838                .any(|values| values.len() != grid.spec().len as usize)
1839            {
1840                return Err(DataFusionError::Internal(
1841                    "histogram parameter length differs from its grid".into(),
1842                ));
1843            }
1844        }
1845        if self.parameter.is_some() && !matches!(self.op, BoundVectorOp::TopK { .. }) {
1846            return Err(DataFusionError::Internal(
1847                "only topk accepts a per-point k".into(),
1848            ));
1849        }
1850        if let Some(values) = self.parameter.as_ref() {
1851            let grid = self
1852                .schema
1853                .eval_grid
1854                .as_ref()
1855                .ok_or_else(|| DataFusionError::Internal("topk grid missing".into()))?;
1856            if values.len() != grid.spec().len as usize {
1857                return Err(DataFusionError::Internal(
1858                    "topk parameter length differs from its grid".into(),
1859                ));
1860            }
1861        }
1862        buffer.grow(vector_workspace_bytes(&self.op, &self.schema, &assembled))?;
1863        let metrics = Arc::clone(&buffer.metrics);
1864        let output = {
1865            let _timer = metrics.timer();
1866            if parameters.is_empty() {
1867                vector_series_with_k(
1868                    &self.op,
1869                    &self.schema,
1870                    &assembled,
1871                    self.parameter.as_deref(),
1872                    self.annotations.as_ref(),
1873                )?
1874            } else {
1875                let function = match self.op {
1876                    BoundVectorOp::Histogram(function) => function,
1877                    _ => {
1878                        return Err(DataFusionError::Internal(
1879                            "parameterized histogram op mismatch".into(),
1880                        ));
1881                    }
1882                };
1883                let histogram_parameters = match function {
1884                    HistogramFunction::Quantile(_) => HistogramParameters {
1885                        quantile: Some(parameters[0]),
1886                        lower: None,
1887                        upper: None,
1888                    },
1889                    HistogramFunction::Fraction { .. } => HistogramParameters {
1890                        quantile: None,
1891                        lower: Some(parameters[0]),
1892                        upper: Some(parameters[1]),
1893                    },
1894                };
1895                evaluate_histograms_with_annotations(
1896                    function,
1897                    &self.schema,
1898                    &assembled,
1899                    Some(histogram_parameters),
1900                    self.annotations.as_ref(),
1901                )?
1902            }
1903        };
1904        self.input.release();
1905        self.computed = true;
1906        Ok(Some(output))
1907    }
1908}

1919fn vector_series_with_k(
1920    op: &BoundVectorOp,
1921    schema: &Arc<SeriesSchema>,
1922    series: &[Series],
1923    values: Option<&[Option<f64>]>,
1924    collector: Option<&PromqlAnnotationCollector>,
1925) -> Result<Vec<Series>> {
1926    let grouping = |group: &Grouping| match group {
1927        Grouping::By(labels) => select::Grouping::By(labels.to_vec()),
1928        Grouping::Without(labels) => select::Grouping::Without(labels.to_vec()),
1929    };
1930    match op {
1931        BoundVectorOp::TopK { k, top, group } => {
1932            if let Some(values) = values {
1933                let grid = schema.eval_grid.as_ref().ok_or_else(|| {
1934                    DataFusionError::Internal("topk output has no evaluation grid".into())
1935                })?;
1936                if values.len() != grid.spec().len as usize {
1937                    return Err(DataFusionError::Internal(
1938                        "topk parameter length differs from its grid".into(),
1939                    ));
1940                }
1941                select::topk_at(
1942                    // A point the parameter series holds no sample at selects nothing, which is
1943                    // Prometheus' empty parameter there; a sample that is present and NaN is still
1944                    // an error, as [`select::validate_topk_parameters`] rules.
1945                    |point| match values[point] {
1946                        Some(k) => select::k_param(k, series.len()),
1947                        None => Ok(None),
1948                    },
1949                    *top,
1950                    &grouping(group),
1951                    schema,
1952                    series,
1953                )
1954            } else {
1955                select::topk(*k, *top, &grouping(group), schema, series)
1956            }
1957        }
1958        BoundVectorOp::LimitK { k, group } => select::limitk(*k, &grouping(group), schema, series),
1959        BoundVectorOp::LimitRatio { ratio } => select::limit_ratio(*ratio, schema, series),
1960        // A sort hands back the series it was given, rebuilding only those `sort` and `sort_desc`
1961        // drop histogram samples from.
1962        BoundVectorOp::Sort(order) => select::sort_instant(order, schema, series.to_vec()),
1963        // The kernel groups bucket series by label set and takes one native histogram series per
1964        // set, so pieces of one label set -- an empty tag and an absent one, or what a map folded
1965        // -- are merged first, and two of them holding one point fail as they do elsewhere.
1966        BoundVectorOp::Histogram(function) => evaluate_histograms_with_annotations(
1967            *function,
1968            schema,
1969            &merge_same_labelset(schema, series.to_vec())?,
1970            None,
1971            collector,
1972        ),
1973        BoundVectorOp::CountValues { label, group } => {
1974            let (by, labels) = match group {
1975                Grouping::By(labels) => (true, labels),
1976                Grouping::Without(labels) => (false, labels),
1977            };
1978            let labels: Vec<&str> = labels.iter().map(|label| &**label).collect();
1979            let grouping = if by {
1980                aggregate::Grouping::By(&labels)
1981            } else {
1982                aggregate::Grouping::Without(&labels)
1983            };
1984            aggregate::count_values(schema, series, label, grouping)
1985        }
1986        BoundVectorOp::Absent { labels } => Ok(select::absent(schema, series, labels.clone())?
1987            .into_iter()
1988            .collect()),
1989    }
1990}

1992/// A rough estimate of what the vector operation `op` allocates over `series`, beside the input it
1993/// reads.
1994///
1995/// - a ranking: per row the recorded `(point, row)` and its candidate, per series its place in the
1996///   label order, its group and result descriptors, and the histograms `limitk` and `limit_ratio`
1997///   copy into their results;
1998/// - a sort: the value each series is keyed by, and the floats `sort` and `sort_desc` rebuild a series
1999///   into when it holds histograms;
2000/// - `absent`: a presence flag and a result value per grid point and field;
2001/// - `count_values`: per field sample, the group's label set with the value written into it, its count
2002///   per field and the result series it becomes;
2003/// - `histogram_quantile` and `histogram_fraction`: the merge of label-set pieces, per series its group
2004///   entry and a cursor per field, and per row the bucket it contributes.
2005fn vector_workspace_bytes(op: &BoundVectorOp, schema: &SeriesSchema, series: &[Series]) -> usize {
2006    let fields = schema.fields.len();
2007    match op {
2008        BoundVectorOp::Sort(order) => {
2009            let rebuilds = matches!(order, SortOrder::Asc | SortOrder::Desc);
2010            return series.iter().fold(0usize, |total, item| {
2011                let floats = if rebuilds
2012                    && item
2013                        .fields
2014                        .iter()
2015                        .any(|field| histogram_row_bytes(field).is_some())
2016                {
2017                    item.time.len().saturating_mul(std::mem::size_of::<f64>())
2018                } else {
2019                    0
2020                };
2021                total
2022                    .saturating_add(std::mem::size_of::<(f64, Series)>())
2023                    .saturating_add(floats)
2024            });
2025        }
2026        BoundVectorOp::Absent { .. } => {
2027            let points = schema.eval_grid.as_ref().map_or(0, |grid| {
2028                usize::try_from(grid.spec().len).unwrap_or(usize::MAX)
2029            });
2030            return points.saturating_mul(fields.saturating_mul(1 + std::mem::size_of::<f64>()));
2031        }
2032        BoundVectorOp::Histogram(_) => {
2033            let rows: usize = series.iter().map(|item| item.time.len()).sum();
2034            let label = std::mem::size_of::<(Arc<str>, Arc<str>)>();
2035            let row = std::mem::size_of::<(f64, f64)>() + fields * std::mem::size_of::<f64>();
2036            let per_series = std::mem::size_of::<(f64, &Series)>()
2037                + fields * std::mem::size_of::<RowCursor<'_>>();
2038            return series.iter().fold(
2039                merge_workspace_bytes(rows, series.len(), fields),
2040                |total, item| {
2041                    total
2042                        .saturating_add(per_series)
2043                        .saturating_add((item.metric.tags().len() + 1).saturating_mul(label))
2044                        .saturating_add(item.time.len().saturating_mul(row))
2045                },
2046            );
2047        }
2048        BoundVectorOp::CountValues { .. } => {
2049            let label = std::mem::size_of::<(Arc<str>, Arc<str>)>();
2050            // The value label at the length of a 17-digit value with sign, point and exponent;
2051            // the `'f'` form writes a large magnitude out in full, which this does not follow.
2052            let value = 24;
2053            let group = std::mem::size_of::<Vec<f64>>()
2054                + fields * std::mem::size_of::<f64>()
2055                + std::mem::size_of::<Series>()
2056                + fields * std::mem::size_of::<Samples>();
2057            return series.iter().fold(0usize, |total, item| {
2058                let labels = 2 * (item.metric.tags().len() + 2) * label + value;
2059                let samples = item.time.len().saturating_mul(fields);
2060                total.saturating_add(samples.saturating_mul(group + labels))
2061            });
2062        }
2063        BoundVectorOp::TopK { .. }
2064        | BoundVectorOp::LimitK { .. }
2065        | BoundVectorOp::LimitRatio { .. } => {}
2066    }
2067    let row = std::mem::size_of::<(usize, usize)>()
2068        + std::mem::size_of::<(usize, usize, f64)>()
2069        + fields * std::mem::size_of::<f64>();
2070    let per_series = 2 * std::mem::size_of::<usize>()
2071        + std::mem::size_of::<Vec<(usize, usize)>>()
2072        + std::mem::size_of::<RowCursor<'_>>()
2073        + std::mem::size_of::<Series>()
2074        + fields * std::mem::size_of::<Samples>();
2075    let label = std::mem::size_of::<(Arc<str>, Arc<str>)>();
2076    series.iter().fold(0usize, |total, item| {
2077        let rows = item.time.len();
2078        let labels = item.metric.tags().len() + 1;
2079        let histograms = item
2080            .fields
2081            .iter()
2082            .filter_map(histogram_row_bytes)
2083            .fold(0usize, |bytes, widest| {
2084                bytes.saturating_add(widest.saturating_mul(rows))
2085            });
2086        total
2087            .saturating_add(rows.saturating_mul(row))
2088            .saturating_add(per_series)
2089            .saturating_add(labels.saturating_mul(label))
2090            .saturating_add(histograms)
2091    })
2092}

2094/// The operation a scalar broadcast applies, with the side the constant sits on already resolved.
2095#[derive(Clone, Copy, Debug)]

2099    /// A comparison: `Filter` keeps the positions that passed, `Bool` reports `1`/`0` for them.

2189impl BroadcastRuntime {
2190    /// Whether the scalar varies per point and has not arrived yet.
2191    fn awaits_scalar(&self) -> bool {
2192        matches!(self.scalar, BroadcastScalar::Dynamic { value: None, .. })
2193    }
2194
2195    /// Whether the operation rewrites the label set of what it hands out.
2196    ///
2197    /// Arithmetic and a `bool` comparison drop the metric name, so two input series can become one label
2198    /// set under this node; a filtering comparison keeps the label set it read. The answer is read off
2199    /// [`Self::op`], so an operation and its label policy cannot disagree.
2200    fn drops_labels(&self) -> bool {
2201        match self.op {
2202            BroadcastOp::Arithmetic(_) | BroadcastOp::Compare(Comparison::Bool(_)) => true,
2203            BroadcastOp::Compare(Comparison::Filter(_)) => false,
2204        }
2205    }
2206
2207    /// Reports the samples a binary operation had to drop because it has no histogram form.
2208    ///
2209    /// Prometheus answers no value for a histogram under an operator it has no meaning for, and failing
2210    /// the query instead would turn one unsupported sample into no answer at all.
2211    fn record_dropped_histograms(&self, dropped_histogram: bool, op: impl std::fmt::Display) {
2212        if dropped_histogram && let Some(collector) = &self.annotations {
2213            collector.record_info(format!(
2214                "{op}: dropped native histogram samples because this binary operation is not supported for native histograms"
2215            ));
2216        }
2217    }
2218
2219    /// The label set a chunk of the vector side's `metric` becomes under this operation.
2220    ///
2221    /// It is what a series closes under when the operation emptied every position.
2222    fn output_metric(&self, metric: &SeriesLabels) -> Result<SeriesLabels> {
2223        if self.drops_labels() {
2224            drop_metric_name(metric)
2225        } else {
2226            Ok(metric.clone())
2227        }
2228    }
2229
2230    /// Applies the operation to one chunk of the vector side.
2231    ///
2232    /// `None` means the operation emptied every position and the chunk does not close its series. A
2233    /// chunk that does close its series is answered even then -- with a series carrying no row --
2234    /// because a consumer tells "nothing more here" from "no more input" by that chunk alone.
2235    fn apply(&self, chunk: &EvaluatedChunk) -> Result<Option<Series>> {
2236        let output = match &self.scalar {
2237            BroadcastScalar::Literal(operand) => match self.op {
2238                BroadcastOp::Arithmetic(op) => {
2239                    let (output, dropped_histogram) =
2240                        map_arithmetic_literal(&self.sample_schema, &chunk.series, op, *operand)?;
2241                    self.record_dropped_histograms(dropped_histogram, op);
2242                    Some(output)
2243                }
2244                BroadcastOp::Compare(comparison) => {
2245                    let (output, dropped_histogram) = map_compare_literal(
2246                        &self.sample_schema,
2247                        &chunk.series,
2248                        comparison,
2249                        *operand,
2250                    )?;
2251                    self.record_dropped_histograms(dropped_histogram, comparison);
2252                    output
2253                }
2254            },
2255            BroadcastScalar::Dynamic { left, value } => {
2256                let value = value.as_ref().ok_or_else(|| {
2257                    DataFusionError::Execution(
2258                        "the binary node broadcast a scalar that has not arrived".to_string(),
2259                    )
2260                })?;
2261                let operand = if *left {
2262                    ScalarSeriesOperand::Left(value)
2263                } else {
2264                    ScalarSeriesOperand::Right(value)
2265                };
2266                match self.op {
2267                    BroadcastOp::Arithmetic(op) => {
2268                        if !is_float_series(&chunk.series) {
2269                            return Err(DataFusionError::NotImplemented(
2270                                "dynamic scalar arithmetic does not support histogram or mixed samples"
2271                                    .into(),
2272                            ));
2273                        }
2274                        map_arithmetic_series(&self.sample_schema, &chunk.series, op, operand)?
2275                    }
2276                    BroadcastOp::Compare(comparison) => {
2277                        let (output, dropped_histogram) = map_compare_series(
2278                            &self.sample_schema,
2279                            &chunk.series,
2280                            comparison,
2281                            operand,
2282                        )?;
2283                        self.record_dropped_histograms(dropped_histogram, comparison);
2284                        output
2285                    }
2286                }
2287            }
2288        };
2289        match output {

2350/// The bytes the histogram table holds for one row of `view`: its scalar columns, the offset each
2351/// bucket list pays for it, and the values those lists hold.
2352fn histogram_bytes(view: &HistogramView<'_>) -> usize {
2353    let custom = view
2354        .custom_values
2355        .len()
2356        .saturating_mul(std::mem::size_of::<f64>());
2357    let sides = [view.positive, view.negative]
2358        .iter()
2359        .map(|side| {
2360            side.span_offsets
2361                .len()
2362                .saturating_mul(2)
2363                .saturating_mul(std::mem::size_of::<i32>())
2364                .saturating_add(
2365                    side.buckets
2366                        .len()
2367                        .saturating_mul(std::mem::size_of::<f64>()),
2368                )
2369        })
2370        .fold(0usize, usize::saturating_add);

2390/// The position of the grid point at `timestamp`, which the selection reported as one of its own.
2391///
2392/// A timestamp off that lattice, or one past the end of the grid, is a disagreement between the
2393/// selection and its binding rather than a sample to skip, so it is reported instead of dropped.
2394fn grid_position(grid: &EvalGrid, timestamp: i64) -> Result<usize> {
2395    let spec = grid.spec();
2396    // This runs once per reported point, so the common case stays in `i64`; a grid whose numbers do not
2397    // fit, or a timestamp off the lattice, falls through to the `i128` path below.
2398    if spec.step_ms > 0
2399        && let Some(offset) = timestamp.checked_sub(spec.start_ms)
2400        && offset >= 0
2401    {
2402        let (position, remainder) = (offset / spec.step_ms, offset % spec.step_ms);
2403        if remainder == 0 && (position as u64) < spec.len {
2404            return Ok(position as usize);
2405        }
2406    }
2407    let offset = i128::from(timestamp) - i128::from(spec.start_ms);
2408    let step = i128::from(spec.step_ms);
2409    let position = offset.div_euclid(step);
2410    if offset.rem_euclid(step) != 0 || position < 0 || position >= i128::from(spec.len) {
2411        return Err(DataFusionError::Execution(format!(
2412            "the instant selection reported the timestamp {timestamp}ms, which is not a point of \
2413             its grid {spec:?}"
2414        )));
2415    }
2416    Ok(position as usize)
2417}

2456/// An upper bound of what the operations of a map allocate for the series they write over `series`.
2457///
2458/// Each operation reads the series the one before it wrote, which stays alive until the next one is
2459/// written, so the bound is the largest of the first one's workspace and of two consecutive ones'.
2460/// Every such operation writes floats only, at no more rows than it read.
2461fn map_workspace_bytes(ops: &[MapOp], series: &Series) -> Result<usize> {
2462    let (mut previous, mut bound) = (0usize, 0usize);
2463    let mut first = true;
2464    for op in ops {
2465        let workspace = match op {
2466            MapOp::Label(_) => continue,
2467            MapOp::Math(_) if !first || is_float_series(series) => mapped_workspace_bytes(series)?,
2468            _ if first => built_series_workspace_bytes(series)?,
2469            _ => built_workspace_bytes(series.time.len(), series.time.len(), series.fields.len())?,
2470        };
2471        let live = previous
2472            .checked_add(workspace)
2473            .ok_or_else(|| workspace_overflow(series.time.len(), series.fields.len()))?;
2474        bound = bound.max(live);
2475        previous = workspace;
2476        first = false;
2477    }
2478    Ok(bound)
2479}

2481/// An upper bound of what a kernel that maps each dense segment of `series` into a new array of its
2482/// length allocates: float math over a series of floats, and arithmetic against a constant, which also
2483/// rebuilds a histogram field sample by sample.
2484///
2485/// A mapped segment holds its values behind the header of the allocation they share, rounded up to
2486/// Arrow's 64-byte alignment. A histogram row is not one `f64`: the kernel copies the whole payload,
2487/// buckets included, so the largest row in the field is the one to charge.
2488fn mapped_workspace_bytes(series: &Series) -> Result<usize> {
2489    let segments = series
2490        .fields
2491        .iter()
2492        .map(|field| field.segments().len())
2493        .fold(0usize, usize::saturating_add);
2494    let bound = || {
2495        let mut bytes = assembly_bytes(series.fields.len(), segments, &series.metric)?;
2496        for field in series.fields.iter() {
2497            let row = histogram_row_bytes(field).map_or(std::mem::size_of::<f64>(), |histogram| {
2498                histogram.max(std::mem::size_of::<f64>())
2499            });
2500            for segment in field.segments() {
2501                bytes = segment
2502                    .len()
2503                    .checked_mul(row)?
2504                    .checked_add(64 + BUFFER_HEADER_BYTES)?
2505                    .checked_add(bytes)?;
2506            }
2507        }
2508        Some(bytes)
2509    };
2510    bound().ok_or_else(|| workspace_overflow(series.time.len(), series.fields.len()))
2511}

2513/// An upper bound of what a kernel that pushes a subset of the rows of `series` into a
2514/// [`GridSeriesBuilder`] allocates: [`built_workspace_bytes`] over the series' rows and runs, and
2515/// the result's assembly ([`assembly_bytes`]), one segment per field.
2516fn built_series_workspace_bytes(series: &Series) -> Result<usize> {
2517    let fields = series.fields.len();

2519    assembly_bytes(fields, fields, &series.metric)

2524/// The bytes a result of `fields` fields over `segments` segments is assembled from, beside its values:
2525/// the descriptor lists and the label set rebuilt from `metric`.
2526///
2527/// A list collected or pushed without knowing its length starts at four entries and doubles, so it
2528/// holds at most `2 * n + 4` descriptors for `n` of them.
2529fn assembly_bytes(fields: usize, segments: usize, metric: &SeriesLabels) -> Option<usize> {
2530    let descriptors = segments
2531        .checked_mul(6)?
2532        .checked_add(fields.checked_mul(16)?)?
2533        .checked_add(8)?
2534        .checked_mul(std::mem::size_of::<Samples>())?;
2535    descriptors
2536        .checked_add(std::mem::size_of_val(metric.tags()))?
2537        .checked_add(BUFFER_HEADER_BYTES + std::mem::size_of::<Series>())
2538}

2543/// An upper bound of the bytes a [`GridSeriesBuilder`] allocates for a result of `fields` float fields
2544/// whose positions are a subset of `rows` positions making up `runs` runs.
2545///
2546/// For a result of `kept` rows in `split` runs, the builder's row indices and each field's values grow
2547/// by doubling, so each holds at most `2 * kept + 4` entries, and finishing turns each run into a span
2548/// the axis copies once. Dropping one position splits at most one run in two, so
2549/// `split <= min(kept, runs + rows - kept)`, and the bound is the larger of the two ends of that
2550/// region. A caller that cannot say the result's positions are the input's passes `runs = rows`.
2551fn built_workspace_bytes(rows: usize, runs: usize, fields: usize) -> Result<usize> {
2552    let cost = |kept: usize, split: usize| {
2553        let capacity = kept.checked_mul(2)?.checked_add(4)?;
2554        let indices = capacity.checked_mul(std::mem::size_of::<usize>())?;
2555        let validity = kept.div_ceil(8).checked_mul(2)?.checked_add(64)?;
2556        let field = capacity
2557            .checked_mul(std::mem::size_of::<f64>())?
2558            .checked_add(validity)?;
2559        let spans = split
2560            .checked_mul(2 * std::mem::size_of::<GridSpan>() + 2 * std::mem::size_of::<usize>())?;
2561        fields
2562            .checked_mul(field)?
2563            .checked_add(indices)?
2564            .checked_add(spans)
2565    };
2566    let runs = runs.min(rows);
2567    let fragmented = rows.checked_add(runs).map(|total| total.div_ceil(2));
2568    fragmented
2569        .and_then(|fragmented| Some(cost(rows, runs)?.max(cost(fragmented, fragmented)?)))
2570        .ok_or_else(|| workspace_overflow(rows, fields))
2571}

2576    ))
2577}

2579/// An upper bound of what `window`'s next push allocates for the rows it hands out, for input whose
2580/// last sample is at `last_ts`, closing its series when `last`, with `fields`, of the series `metric`
2581/// names.
2582///
2583/// The rows are bounded by [`RangeEvaluator::output_rows_bound`] and written as
2584/// [`built_workspace_bytes`] counts them; the points they sit on are the output grid's, so each may be a
2585/// run of its own. A function that can answer with a native histogram also writes an index per field
2586/// and row and a table row grown by doubling, bounded by the widest histogram the evaluator retains or
2587/// `fields` carries. A combination of disjoint layouts can exceed that row; it is not bounded here.
2588fn window_workspace_bytes(
2589    window: &RangeEvaluator,
2590    last_ts: Option<i64>,
2591    last: bool,
2592    fields: &SeriesFields,
2593    metric: &SeriesLabels,
2594) -> Result<usize> {
2595    let rows = window.output_rows_bound(last_ts, last);
2596    let width = window.output_schema().fields.len();
2597    let floats = assembly_bytes(width, width, metric)
2598        .ok_or_else(|| workspace_overflow(rows, width))?
2599        .checked_add(built_workspace_bytes(rows, rows, width)?)
2600        .ok_or_else(|| workspace_overflow(rows, width))?;
2601    if !window.yields_histograms() {
2602        return Ok(floats);
2603    }
2604    let retained = window
2605        .widest_retained_histogram()
2606        .map(|heap| heap.saturating_add(HISTOGRAM_ROW_HEADER_BYTES));
2607    let incoming = fields.iter().filter_map(histogram_row_bytes).max();
2608    let Some(widest) = retained.max(incoming) else {
2609        return Ok(floats);
2610    };
2611    let histograms = || {
2612        let row = widest
2613            .checked_mul(2)?
2614            .checked_add(2 * std::mem::size_of::<Option<u32>>() + std::mem::size_of::<u32>() + 1)?;
2615        let in_flight = widest.checked_add(std::mem::size_of::<FloatHistogram>())?;
2616        rows.checked_mul(row)?
2617            .checked_add(in_flight)?
2618            .checked_mul(width)?
2619            .checked_add(floats)
2620    };
2621    histograms().ok_or_else(|| {
2622        DataFusionError::Execution(format!(
2623            "the histogram workspace for {rows} rows of {width} fields does not fit one address \
2624             space"
2625        ))
2626    })
2627}

2901/// Stack one [`RuntimeNode::build`] or [`RuntimeNode::pull`] level needs below it before the recursion
2902/// grows its stack.
2903///
2904/// Both walks recurse once per tree level and a debug build's frame is large (about 140 KiB for a
2905/// `build` level and 80 KiB for a `pull` level, measured on a 100-term chain), which can abort an 8 MiB
2906/// runtime worker. There is no depth limit, so the recursion grows its stack instead.
2907const NATIVE_STACK_RED_ZONE: usize = 512 * 1024;
2908
2909/// Stack a growth adds when the reserve above is spent. It holds several levels of the same walk.
2910const NATIVE_STACK_GROWTH: usize = 4 * 1024 * 1024;

2917        pool: &Arc<dyn MemoryPool>,
2918        inputs: &mut Vec<InputSlot>,
2919        binary_before: &mut usize,

2921        duplicate_evidence: Option<Arc<DuplicateEvidence>>,
2922    ) -> Result<Self> {
2923        stacker::maybe_grow(NATIVE_STACK_RED_ZONE, NATIVE_STACK_GROWTH, || {
2924            Self::build_within_stack(
2925                node,
2926                pool,
2927                inputs,
2928                binary_before,
2929                location,
2930                duplicate_evidence,
2931            )
2932        })
2933    }

3833    /// Pulls the next chunk this node hands over, taking what its children owe it.
3834    ///
3835    /// `inputs_finished` is the caller's word that no more source chunks will arrive: a node that
3836    /// decides nothing until its whole input is there reads it to tell "nothing more" from "nothing
3837    /// yet".
3838    ///
3839    /// The pull recurses once per tree level, so it grows its stack as it descends
3840    /// ([`NATIVE_STACK_RED_ZONE`]).
3841    fn pull(
3842        &mut self,
3843        inputs: &mut [InputSlot],
3844        inputs_finished: bool,
3845    ) -> Result<Option<NativeChunk>> {
3846        stacker::maybe_grow(NATIVE_STACK_RED_ZONE, NATIVE_STACK_GROWTH, || {
3847            self.pull_within_stack(inputs, inputs_finished)
3848        })
3849    }

3851    fn pull_within_stack(
3852        &mut self,
3853        inputs: &mut [InputSlot],
3854        inputs_finished: bool,
3855    ) -> Result<Option<NativeChunk>> {
3856        match &mut self.op {
3857            // A source hands over what the caller queued at it. The id was assigned when the tree
3858            // was built, so an id past the leaves means this call and the tree disagree rather than
3859            // that the chunk belongs somewhere else.
3860            RuntimeOp::Input(id) => {
3861                let input = *id;
3862                let count = inputs.len();
3863                let slot = inputs
3864                    .get_mut(input.index())
3865                    .ok_or_else(|| no_input(count, input))?;
3866                slot.pop_source()
3867            }
3868            // A scalar owes its consumer no chunk: it produces one value and no series, and the
3869            // value is the constant on the tree, which the consumer reads off its own binding. A
3870            // pull of it is therefore empty by construction and says nothing has stalled.
3871            RuntimeOp::Scalar => Ok(None),
3872            RuntimeOp::ScalarOf {
3873                input,
3874                input_schema,
3875                grid,
3876                retained,
3877                produced,
3878                state,
3879            } => {
3880                while let Some(chunk) = input.pull(inputs, inputs_finished)? {
3881                    let NativeChunk::Evaluated(chunk) = chunk else {
3882                        return Err(DataFusionError::Execution(
3883                            "scalar() consumes evaluated chunks, not raw samples".into(),
3884                        ));
3885                    };
3886                    validate_binary_operand(input_schema, std::slice::from_ref(&chunk.series))
3887                        .map_err(|error| {
3888                            DataFusionError::Execution(format!(
3889                                "scalar() rejected its input: {error}"
3890                            ))
3891                        })?;
3892                    let pending = retained.pending(&chunk);
3893                    state.buffer.grow(pending).map_err(|e| state.context(e))?;
3894                    retained
3895                        .retain(chunk, Reader::Vector)
3896                        .map_err(|e| state.context(e))?;
3897                    state
3898                        .buffer
3899                        .resize(retained.held)
3900                        .map_err(|e| state.context(e))?;
3901                }
3902                if !inputs_finished || !input.settled(inputs) {
3903                    return Ok(None);
3904                }
3905                if !*produced {
3906                    let len = grid.spec().len as usize;
3907                    let bytes = grid_chunk_bytes(grid).map_err(|e| state.context(e))?;
3908                    // The counts the kernel decides its answer by are read while it walks the
3909                    // assembled input, so they are reserved on top of the chunk it hands over.
3910                    let counts_bytes = len.saturating_mul(std::mem::size_of::<u8>());
3911                    let assembly = retained
3912                        .assembly_bytes(input_schema)
3913                        .ok_or_else(vector_overflow)?;
3914                    state
3915                        .buffer
3916                        .grow(assembly.saturating_add(bytes).saturating_add(counts_bytes))
3917                        .map_err(|e| state.context(e))?;
3918                    let assembled = retained
3919                        .assemble(input_schema, Reader::Vector)
3920                        .map_err(|e| state.context(e))?;
3921                    retained.release();
3922                    // `scalar()` reads a whole vector, which is what the assembled input is, so
3923                    // the kernel Prometheus' `funcScalar` is answers here too: one value per grid
3924                    // point, NaN where the point does not hold exactly one float sample, and the
3925                    // histogram samples of the input ignored.
3926                    let answer =
3927                        select::scalar(input_schema, &assembled).map_err(|e| state.context(e))?;
3928                    // The assembled input is consumed, so what this node holds from here on is the
3929                    // one chunk the answer becomes.
3930                    drop(assembled);
3931                    state.buffer.resize(bytes).map_err(|e| state.context(e))?;
3932                    // Only the layout the answer is named by changes: a scalar sits on this node's
3933                    // grid, in the one field a scalar's carrier has.
3934                    let schema = scalar_schema(Arc::clone(grid));
3935                    let series =
3936                        Series::try_new(&schema, answer.metric, answer.time, answer.fields)
3937                            .map_err(series_error)?;
3938                    state.queued.push_back((
3939                        NativeChunk::Evaluated(EvaluatedChunk::try_new(series, true)?),
3940                        bytes,
3941                    ));
3942                    *produced = true;
3943                }
3944                state.pop_queued(evaluated_values)
3945            }
3946            RuntimeOp::EvalTime {
3947                grid,
3948                date,
3949                produced,
3950                state,
3951            } => {

3953                    let schema = scalar_schema(Arc::clone(grid));
3954                    state
3955                        .produce(grid, || match date {
3956                            None => map_time::time(&schema),
3957                            Some(date) => map_time::date_of_eval_time(*date, &schema),
3958                        })
3959                        .map_err(|error| state.context(error))?;
3960                }
3961                state.pop_queued(evaluated_values)
3962            }
3963            RuntimeOp::VectorOf {
3964                input,
3965                literal,
3966                produced,
3967                schema,
3968                state,
3969            } => {

3971                    if !std::mem::replace(produced, true) {
3972                        let grid = schema.eval_grid.clone().ok_or_else(|| {
3973                            DataFusionError::Execution(
3974                                "vector() produces values without a grid".to_string(),
3975                            )
3976                        })?;
3977                        state
3978                            .produce(&grid, || {
3979                                Ok(scalar_chunk_on(schema, Arc::clone(&grid), *value)?.series)
3980                            })
3981                            .map_err(|error| state.context(error))?;
3982                    }

3984                }

3986                let Some(chunk) = input.pull(inputs, inputs_finished)? else {
3987                    return Ok(None);
3988                };

3995                let series = Series::try_new(schema, series.metric, series.time, series.fields)

4214                // binary node only retains the chunks, because a pair is not decided until both

4228                // a node that reports itself settled reports it for everything below it too.

4312                vector,

4521    /// Whether this node has produced everything it ever will.
4522    ///
4523    /// A pull that finds nothing does not answer it: a collecting window reports nothing until it is
4524    /// flushed, and a binary node nothing until both nodes it reads from are finished. Each node
4525    /// therefore answers from its own state and from its children's.
4526    fn settled(&self, inputs: &[InputSlot]) -> bool {
4527        match &self.op {
4528            // A source has handed over everything it ever will once nothing is queued at it: what
4529            // it hands over is what the caller queued.
4530            RuntimeOp::Input(id) => inputs
4531                .get(id.index())
4532                .is_some_and(|slot| slot.state.queued.is_empty()),
4533            // A scalar produces its value without reading anything, so it is settled from the
4534            // start: there is no chunk it owes its consumer.
4535            RuntimeOp::Scalar => true,
4536            RuntimeOp::EvalTime {
4537                produced, state, ..
4538            } => *produced && state.queued.is_empty(),
4539            RuntimeOp::ScalarOf {
4540                input,
4541                produced,
4542                retained,
4543                state,
4544                ..
4545            } => {
4546                *produced && state.queued.is_empty() && retained.held == 0 && input.settled(inputs)
4547            }
4548            RuntimeOp::ParameterizedMap {
4549                input,
4550                parameters,
4551                state,
4552                ..
4553            } => {
4554                state.queued.is_empty()
4555                    && input.settled(inputs)
4556                    && parameters.iter().all(|parameter| parameter.settled(inputs))
4557            }
4558            RuntimeOp::ParameterizedHistogram {
4559                input,
4560                vector,
4561                state,
4562                ..
4563            } => {
4564                // Parameter nodes are released after their values are installed; `computed`
4565                // already implies those values were consumed.
4566                state.queued.is_empty() && vector.computed && input.settled(inputs)
4567            }
4568            RuntimeOp::ParameterizedTopK {
4569                input,
4570                vector,
4571                state,
4572                ..
4573            } => {
4574                // Parameter nodes are released after their values are installed; `computed`
4575                // already implies the parameter was consumed.
4576                state.queued.is_empty() && vector.computed && input.settled(inputs)
4577            }
4578            RuntimeOp::VectorOf {
4579                input,
4580                literal,
4581                produced,
4582                state,
4583                ..
4584            } => {
4585                state.queued.is_empty() && (*produced || literal.is_none()) && input.settled(inputs)
4586            }
4587            // These report as they are handed chunks, so a child with nothing left is all it takes
4588            // for them to have nothing left either.
4589            RuntimeOp::Instant { input, state, .. }
4590            | RuntimeOp::Map { input, state, .. }
4591            | RuntimeOp::Timestamp { input, state, .. } => {
4592                state.queued.is_empty() && input.settled(inputs)
4593            }
4594            // A window that collects its input has not answered until it has flushed it, so a merge
4595            // still held is an answer still owed.
4596            RuntimeOp::Window {
4597                input,
4598                merge,
4599                state,
4600                ..
4601            } => state.queued.is_empty() && merge.is_none() && input.settled(inputs),
4602            RuntimeOp::Merge {
4603                input,
4604                merge,
4605                state,
4606            } => state.queued.is_empty() && merge.is_none() && input.settled(inputs),
4607            // An aggregate answers one series per group and only once its input ended, so its own
4608            // fold has to have run as well: a child that has handed everything up is not enough,
4609            // because every one of those chunks was folded and none of them answered yet.
4610            RuntimeOp::Aggregate {
4611                input,
4612                aggregate,
4613                state,
4614            } => state.queued.is_empty() && aggregate.computed && input.settled(inputs),
4615            RuntimeOp::Vector {
4616                input,
4617                vector,
4618                state,
4619            } => state.queued.is_empty() && vector.computed && input.settled(inputs),
4620            // A vector match is settled once its kernel ran and both operands are: what one side
4621            // answers at a point depends on the other side.
4622            RuntimeOp::Binary {
4623                left,
4624                right,
4625                binary,
4626                state,
4627            } => {
4628                state.queued.is_empty()
4629                    && binary.computed
4630                    && left.settled(inputs)
4631                    && right.settled(inputs)
4632            }
4633            // A broadcast owes what its sides have not handed over yet. A constant side is settled
4634            // from the start.
4635            RuntimeOp::Broadcast {
4636                left, right, state, ..
4637            } => state.queued.is_empty() && left.settled(inputs) && right.settled(inputs),
4638        }
4639    }

4641    /// Merges the collected input of every window that collects it, deepest first.
4642    ///
4643    /// A window above another collects the chunks the one below hands up, so flushing the shallow one
4644    /// first would merge an input it has not received yet. A window whose input has not settled is left
4645    /// for a later call.
4646    fn flush_ready_merges(&mut self, inputs: &[InputSlot]) -> Result<bool> {
4647        match &mut self.op {
4648            RuntimeOp::Input(_) | RuntimeOp::Scalar | RuntimeOp::EvalTime { .. } => Ok(false),
4649            RuntimeOp::ScalarOf { input, .. } => input.flush_ready_merges(inputs),
4650            RuntimeOp::ParameterizedMap {
4651                input, parameters, ..
4652            } => {
4653                let mut flushed = input.flush_ready_merges(inputs)?;
4654                for parameter in parameters {
4655                    flushed |= parameter.flush_ready_merges(inputs)?;
4656                }
4657                Ok(flushed)
4658            }
4659            RuntimeOp::Instant { input, .. }
4660            | RuntimeOp::Map { input, .. }
4661            | RuntimeOp::Timestamp { input, .. }
4662            | RuntimeOp::Aggregate { input, .. }
4663            | RuntimeOp::Vector { input, .. }
4664            | RuntimeOp::VectorOf { input, .. } => input.flush_ready_merges(inputs),
4665            RuntimeOp::ParameterizedHistogram {
4666                input, parameters, ..
4667            } => {
4668                let mut flushed = input.flush_ready_merges(inputs)?;
4669                for p in parameters.iter_mut().flatten() {
4670                    flushed |= p.flush_ready_merges(inputs)?;
4671                }
4672                Ok(flushed)
4673            }
4674            RuntimeOp::ParameterizedTopK {
4675                input, parameter, ..
4676            } => {
4677                let mut flushed = input.flush_ready_merges(inputs)?;
4678                if let Some(parameter) = parameter {
4679                    flushed |= parameter.flush_ready_merges(inputs)?;
4680                }
4681                Ok(flushed)
4682            }
4683            RuntimeOp::Binary { left, right, .. } | RuntimeOp::Broadcast { left, right, .. } => {
4684                let left_flushed = left.flush_ready_merges(inputs)?;
4685                let right_flushed = right.flush_ready_merges(inputs)?;
4686                Ok(left_flushed || right_flushed)
4687            }
4688            RuntimeOp::Merge {
4689                input,
4690                merge,
4691                state,
4692            } => {
4693                let mut flushed = input.flush_ready_merges(inputs)?;
4694                if merge.is_some() && input.settled(inputs) {
4695                    state.flush_canonical_merge(merge)?;
4696                    flushed = true;
4697                }
4698                Ok(flushed)
4699            }
4700            RuntimeOp::Window {
4701                input,
4702                window,
4703                merge,
4704                state,
4705            } => {
4706                let mut flushed = input.flush_ready_merges(inputs)?;
4707                // A window collects its input only when it can hold two series under one canonical label
4708                // set, and that input has to be finished before it is merged.
4709                if merge.is_some() && input.settled(inputs) {
4710                    let Some(evaluator) = window.as_mut() else {
4711                        return Err(no_window());
4712                    };
4713                    state
4714                        .flush_merge(evaluator, merge)
4715                        .map_err(|error| state.context(error))?;
4716                    flushed = true;
4717                }
4718                Ok(flushed)
4719            }
4720        }
4721    }

4725        match &self.op {
4726            RuntimeOp::Input(_) | RuntimeOp::Scalar | RuntimeOp::EvalTime { .. } => false,
4727            RuntimeOp::ScalarOf { input, .. } => input.any_merge_pending(),
4728            RuntimeOp::ParameterizedMap {
4729                input, parameters, ..
4730            } => input.any_merge_pending() || parameters.iter().any(RuntimeNode::any_merge_pending),
4731            RuntimeOp::Instant { input, .. }
4732            | RuntimeOp::Map { input, .. }
4733            | RuntimeOp::Timestamp { input, .. }
4734            | RuntimeOp::Aggregate { input, .. }
4735            | RuntimeOp::Vector { input, .. }
4736            | RuntimeOp::VectorOf { input, .. } => input.any_merge_pending(),
4737            RuntimeOp::ParameterizedHistogram {
4738                input, parameters, ..
4739            } => {
4740                input.any_merge_pending()
4741                    || parameters
4742                        .iter()
4743                        .flatten()
4744                        .any(RuntimeNode::any_merge_pending)
4745            }
4746            RuntimeOp::ParameterizedTopK {
4747                input, parameter, ..
4748            } => {
4749                input.any_merge_pending()
4750                    || parameter.as_ref().is_some_and(|p| p.any_merge_pending())
4751            }
4752            RuntimeOp::Binary { left, right, .. } | RuntimeOp::Broadcast { left, right, .. } => {
4753                left.any_merge_pending() || right.any_merge_pending()
4754            }
4755            // Only a window that collects its input keeps collected input to merge.
4756            RuntimeOp::Window { input, merge, .. } => merge.is_some() || input.any_merge_pending(),
4757            RuntimeOp::Merge { input, merge, .. } => merge.is_some() || input.any_merge_pending(),
4758        }

4761    /// Refuses a node that ended inside a series, in every node that would read the remainder.
4762    ///
4763    /// The assembly above would read half a vector as a whole one, and a window that collected its input
4764    /// would merge the pieces it has into a series that looks finished.
4765    fn unclosed(&self) -> Result<()> {
4766        match &self.op {
4767            RuntimeOp::Input(_) | RuntimeOp::Scalar | RuntimeOp::EvalTime { .. } => Ok(()),
4768            RuntimeOp::ScalarOf {
4769                input,
4770                retained,
4771                state,
4772                ..
4773            } => {
4774                retained
4775                    .ensure_closed(Reader::Vector)
4776                    .map_err(|error| state.context(error))?;
4777                input.unclosed()
4778            }
4779            RuntimeOp::ParameterizedMap {
4780                input, parameters, ..
4781            } => {
4782                input.unclosed()?;
4783                parameters.iter().try_for_each(RuntimeNode::unclosed)
4784            }
4785            RuntimeOp::Instant { input, .. }
4786            | RuntimeOp::Map { input, .. }
4787            | RuntimeOp::Timestamp { input, .. }
4788            | RuntimeOp::Window { input, .. }
4789            | RuntimeOp::Aggregate { input, .. }
4790            | RuntimeOp::VectorOf { input, .. }
4791            | RuntimeOp::Merge { input, .. } => input.unclosed(),
4792            RuntimeOp::ParameterizedTopK {
4793                input,
4794                parameter,
4795                vector,
4796                state,
4797            } => {
4798                vector
4799                    .input
4800                    .ensure_closed(Reader::Vector)
4801                    .map_err(|e| state.context(e))?;
4802                input.unclosed()?;
4803                if let Some(parameter) = parameter {
4804                    parameter.unclosed()?;
4805                }
4806                Ok(())
4807            }
4808            RuntimeOp::ParameterizedHistogram {
4809                input,
4810                parameters,
4811                vector,
4812                state,
4813                ..
4814            } => {
4815                vector
4816                    .input
4817                    .ensure_closed(Reader::Vector)
4818                    .map_err(|e| state.context(e))?;
4819                input.unclosed()?;
4820                parameters
4821                    .iter()
4822                    .flatten()
4823                    .try_for_each(RuntimeNode::unclosed)
4824            }
4825            RuntimeOp::Binary {
4826                left,
4827                right,
4828                binary,
4829                state,
4830            } => {
4831                binary.closed().map_err(|error| state.context(error))?;
4832                left.unclosed()?;
4833                right.unclosed()
4834            }
4835            RuntimeOp::Broadcast { left, right, .. } => {
4836                left.unclosed()?;
4837                right.unclosed()
4838            }
4839            RuntimeOp::Vector {
4840                input,
4841                vector,
4842                state,
4843            } => {
4844                vector
4845                    .input
4846                    .ensure_closed(Reader::Vector)
4847                    .map_err(|error| state.context(error))?;
4848                input.unclosed()
4849            }
4850        }
4851    }

4853    /// Finishes every window's evaluator that is still here, and reports the first failure.
4854    ///
4855    /// Every evaluator is finished even after one fails -- the first error is the one reported -- so that
4856    /// a failure leaves nothing to be read on the way out.
4857    fn finish_windows(&mut self) -> Result<()> {
4858        match &mut self.op {
4859            RuntimeOp::Input(_) | RuntimeOp::Scalar | RuntimeOp::EvalTime { .. } => Ok(()),
4860            RuntimeOp::ScalarOf { input, .. } => input.finish_windows(),
4861            RuntimeOp::ParameterizedMap {
4862                input, parameters, ..
4863            } => {
4864                // Every evaluator is finished even after one fails, as for a binary node.
4865                let first = input.finish_windows();
4866                parameters
4867                    .iter_mut()
4868                    .map(RuntimeNode::finish_windows)
4869                    .fold(first, Result::and)
4870            }
4871            RuntimeOp::Instant { input, .. }
4872            | RuntimeOp::Map { input, .. }
4873            | RuntimeOp::Timestamp { input, .. }
4874            | RuntimeOp::Aggregate { input, .. }
4875            | RuntimeOp::Vector { input, .. }
4876            | RuntimeOp::VectorOf { input, .. }
4877            | RuntimeOp::Merge { input, .. } => input.finish_windows(),
4878            RuntimeOp::ParameterizedTopK {
4879                input, parameter, ..
4880            } => {
4881                let first = input.finish_windows();
4882                let parameter = parameter.as_mut().map_or(Ok(()), |p| p.finish_windows());
4883                first.and(parameter)
4884            }
4885            RuntimeOp::ParameterizedHistogram {
4886                input, parameters, ..
4887            } => {
4888                let first = input.finish_windows();
4889                parameters
4890                    .iter_mut()
4891                    .flatten()
4892                    .map(RuntimeNode::finish_windows)
4893                    .fold(first, Result::and)
4894            }
4895            RuntimeOp::Binary { left, right, .. } | RuntimeOp::Broadcast { left, right, .. } => {
4896                // Both sides are finished even when the first one fails: the failure is reported
4897                // from here, and an evaluator left holding a series would keep its bytes.
4898                let first = left.finish_windows();
4899                let second = right.finish_windows();
4900                first.and(second)
4901            }
4902            RuntimeOp::Window {
4903                input,
4904                window,
4905                state,
4906                ..
4907            } => {
4908                let child = input.finish_windows();
4909                let own = window
4910                    .take()
4911                    .map_or(Ok(()), RangeEvaluator::finish)
4912                    .map_err(|error| state.context(error));
4913                child.and(own)
4914            }
4915        }
4916    }

5063    /// Queues a source chunk, holding what it is worth until it leaves.
5064    ///
5065    /// What this node holds is its queue, so the reservation is the sum of the estimates in it.
5066    fn enqueue(&mut self, chunk: NativeChunk) -> Result<()> {
5067        let positions = chunk.positions();
5068        let bytes = chunk.bytes();
5069        // The chunk is held from here on, so the pool admits it before the queue keeps it: a
5070        // rejected chunk leaves nothing behind.
5071        self.buffer.grow(bytes)?;
5072        self.queued.push_back((chunk, bytes));
5073        self.buffer.metrics.record_input(1, positions);
5074        Ok(())
5075    }
5076
5077    /// Hands over the next queued chunk, if there is one, giving back what it was queued with.
5078    ///
5079    /// It is the one way a chunk leaves this node's queue, for both the caller's chunks and a kernel's
5080    /// results. `values` is how that node counts what it hands over, and the two do not count the same
5081    /// thing: an input's figure is its positions, a kernel's result holds one value per field.
5082    fn pop_queued(&mut self, values: fn(&NativeChunk) -> usize) -> Result<Option<NativeChunk>> {
5083        let Some((mut chunk, bytes)) = self.queued.pop_front() else {
5084            return Ok(None);
5085        };
5086        // A producer hands its state to a consumer that packs rows from the batch across polls, so
5087        // that chunk's charge travels with it instead of being returned here.
5088        let moved = match &mut chunk {
5089            NativeChunk::AggregateState(StateChunk::Series { reservation, .. }) => {
5090                debug_assert!(
5091                    reservation.is_none(),
5092                    "a handed-over state chunk was charged to a queue twice"
5093                );
5094                let transferred = self.buffer.transfer(bytes)?;
5095                let moved = transferred.size();
5096                *reservation = Some(transferred);
5097                moved
5098            }
5099            _ => 0,
5100        };
5101        let metrics = Arc::clone(&self.buffer.metrics);
5102        {
5103            // An input node's own work is handing the chunk over, so its self time covers that and
5104            // nothing its producer did. Whatever the chunk did not take with it, the pool no longer
5105            // has to hold.
5106            let _timer = metrics.timer();
5107            self.buffer.shrink(bytes - moved)?;
5108            self.release_queued_tables()?;
5109        }
5110        metrics.record_output(1, values(&chunk));
5111        // An evaluated chunk is addressed on its grid, so its decided positions are grid points; a
5112        // raw source carries timestamps, which are not positions on any grid.
5113        if let NativeChunk::Evaluated(chunk) = &chunk {
5114            metrics.record_grid_points(chunk.series.time.len());
5115        }
5116        Ok(Some(chunk))
5117    }

5119    /// Maps one evaluated chunk, keeping its axis and its closing.
5120    ///
5121    /// `None` means the operation emptied every position and the chunk does not close its series.
5122    fn map(&mut self, map: &MapRuntime, chunk: EvaluatedChunk) -> Result<Option<EvaluatedChunk>> {
5123        let (positions, closing) = (chunk.series.time.len(), chunk.is_last);
5124        // The chunk is held for as long as this node works on it, beside the series its
5125        // operations write.
5126        let workspace = map_workspace_bytes(&map.ops, &chunk.series)?;
5127        self.buffer.grow(evaluated_bytes(&chunk))?;
5128        self.buffer.grow(workspace)?;
5129        let metrics = Arc::clone(&self.buffer.metrics);
5130        let outcome = {
5131            // The timer covers this node's own work only: it is dropped before the driver polls
5132            // the input again.
5133            let _timer = metrics.timer();
5134            map_chunk(&map.schema, &chunk, &map.ops, &mut |bytes| {
5135                self.buffer.grow(bytes)
5136            })
5137        };
5138        metrics.record_input(1, positions);
5139        if closing {
5140            metrics.record_completed_series(1);
5141        }
5142        // The chunk and the workspace are this call's, not the node's, so a failed map gives them
5143        // back here rather than holding the pool's budget until the stopped executor is dropped.
5144        let output = match outcome {
5145            Ok(output) => output,
5146            Err(error) => {
5147                self.buffer.resize(self.emitted_bytes())?;
5148                return Err(error);
5149            }
5150        };
5151        if let Some(output) = &output {
5152            metrics.record_grid_points(output.series.time.len());
5153            metrics.record_output(1, valid_values(output));
5154            // What it handed out is what the label sets of this node now hold: `map_chunk` has
5155            // already rewritten the output's label set, so the label set of the output is the one
5156            // two input series can share.
5157            self.track(output)?;
5158        }
5159        // The chunk is gone and the label sets this node has handed out are what is left.
5160        let emitted = self.emitted_bytes();
5161        self.buffer.resize(emitted)?;
5162        Ok(output)
5163    }

5218    /// Projects evaluation timestamps from one evaluated chunk, preserving its closing.
5219    fn timestamp(
5220        &mut self,
5221        schema: &Arc<SeriesSchema>,
5222        chunk: EvaluatedChunk,
5223    ) -> Result<Option<EvaluatedChunk>> {

5225        let closing = chunk.is_last;
5226        // The kernel reads the grid's timestamps, which it materializes when they are deferred.
5227        let grid_points = schema.eval_grid.as_ref().map_or(0, |grid| {
5228            usize::try_from(grid.spec().len).unwrap_or(usize::MAX)
5229        });

5231        let scratch = grid_points
5232            .checked_mul(std::mem::size_of::<i64>())
5233            .and_then(|timestamps| timestamps.checked_add(workspace))
5234            .ok_or_else(|| {
5235                DataFusionError::Execution(
5236                    "the workspace of timestamp does not fit one address space".to_string(),
5237                )

5240        self.buffer.grow(scratch)?;
5241        let labels = drop_metric_name(&chunk.series.metric)?;
5242        let metrics = Arc::clone(&self.buffer.metrics);
5243        let result = {
5244            let _timer = metrics.timer();

5246        };
5247        metrics.record_input(1, positions);
5248        if closing {
5249            metrics.record_completed_series(1);
5250        }
5251        // As in `Self::map`, a failed kernel gives back the chunk and the workspace here.
5252        let result = match result {
5253            Ok(result) => result,
5254            Err(error) => {

5256                return Err(error);
5257            }
5258        };

5260            Some(series) => Some(EvaluatedChunk::try_new(series, closing)?),
5261            None if closing => Some(no_row_chunk(schema, labels)?),

5263        };

5265            metrics.record_grid_points(output.series.time.len());

5402        chunk: NativeChunk,
5403    ) -> Result<Option<EvaluatedChunk>> {
5404        if let Some(merge) = merge.as_mut() {
5405            let NativeChunk::Evaluated(chunk) = chunk else {
5406                return Err(DataFusionError::Execution(
5407                    "the window collects evaluated fragments, but it was handed raw samples"
5408                        .to_string(),
5409                ));
5410            };
5411            return self.collect(merge, window, chunk);
5412        }
5413        let (positions, closing) = (chunk.positions(), chunk.is_last());
5414        // The chunk is held for as long as the evaluator consumes it, beside the rows the push
5415        // hands out, which are counted on the output grid: a sparse input can decide far more
5416        // points than it carries, and a dense one far fewer.
5417        let (fields, metric) = match &chunk {
5418            NativeChunk::Raw(chunk) => (&chunk.fields, chunk.metric.as_ref()),
5419            NativeChunk::Evaluated(chunk) => (&chunk.series.fields, Some(&chunk.series.metric)),
5420            NativeChunk::AggregateState(_) => {
5421                return Err(DataFusionError::Execution(
5422                    "an aggregate state chunk reached a node that reads values or samples"
5423                        .to_string(),
5424                ));
5425            }
5426        };
5427        let workspace = match metric {
5428            Some(metric) => {
5429                window_workspace_bytes(window, last_sample_ms(&chunk)?, closing, fields, metric)?
5430            }
5431            // The evaluator refuses a raw chunk without labels before it writes a row.
5432            None => 0,
5433        };
5434        self.buffer.grow(chunk.bytes())?;
5435        self.buffer.grow(workspace)?;
5436        let metrics = Arc::clone(&self.buffer.metrics);
5437        let outcome = {
5438            let _timer = metrics.timer();
5439            match &chunk {
5440                // Raw samples are read window by window by the evaluator itself.
5441                NativeChunk::Raw(chunk) => window.push(chunk),
5442                // An evaluated input is read as computed samples: the evaluator carries the window state of
5443                // one series across its chunks and closes it on the chunk that says so, so a decision can
5444                // be reported before the series ends.
5445                NativeChunk::Evaluated(chunk) => window.push_series(&chunk.series, chunk.is_last),
5446                NativeChunk::AggregateState(_) => {
5447                    return Err(DataFusionError::Execution(
5448                        "an aggregate state chunk reached a node that reads values or samples"
5449                            .to_string(),
5450                    ));
5451                }
5452            }
5453        };
5454        metrics.record_input(1, positions);
5455        if closing {
5456            // A series closes exactly once, on the chunk carrying `is_last`. A window that decided
5457            // nothing for it still closed it, so it is counted here rather than on its output.
5458            metrics.record_completed_series(1);
5459        }
5460        // As in `Self::map`, a failed push gives back the chunk and the workspace here.
5461        let output = match outcome {
5462            Ok(output) => output,
5463            Err(error) => {
5464                self.buffer
5465                    .resize(window.retained_bytes().saturating_add(self.emitted_bytes()))?;
5466                return Err(error);
5467            }
5468        };
5469        if let Some(output) = &output {
5470            metrics.record_grid_points(output.series.time.len());
5471            metrics.record_output(1, valid_values(output));
5472            self.track(output)?;
5473        }
5474        // The chunk has been consumed: what this node still holds is what the evaluator kept of it and the
5475        // label sets it has already handed out. Neither is charged per chunk consumed, which would make a
5476        // fixed window over a long series read as an allocation failure.
5477        let retained = window.retained_bytes();
5478        let emitted = self.emitted_bytes();
5479        self.buffer.resize(retained.saturating_add(emitted))?;
5480        Ok(output)
5481    }

5587    /// Queues the series a kernel that reads its whole input produced, and settles the reservation on the
5588    /// queue.
5589    ///
5590    /// `shares_input` says the series are the assembled input itself, as a sort hands it back: each is then
5591    /// measured the way the input was ([`operand_chunk_bytes`]) and its tables billed once to the queue.
5592    ///
5593    /// `beside` is what the node keeps past the queue, such as a parameterized operator's per-point
5594    /// parameter values.
5595    fn queue_results(
5596        &mut self,
5597        output: Vec<Series>,
5598        shares_input: bool,
5599        beside: usize,
5600    ) -> Result<()> {
5601        let mut tables = HashSet::new();
5602        // The results are queued the way a node that produces a chunk per input queues what it
5603        // hands on: the queue is the one source of chunk order and of the bytes queued chunks hold.
5604        let mut closed = 0usize;
5605        for series in output {
5606            // The kernel decided every point of a result series at once, so the one chunk it is handed
5607            // over as closes it. The result is already admitted within the kernel's workspace.
5608            let bytes = if shares_input {
5609                let (fresh, table_bytes) = unbilled_tables(&series, &tables);
5610                tables.extend(fresh);
5611                self.queued_tables = self.queued_tables.saturating_add(table_bytes);
5612                operand_chunk_bytes(&series)
5613            } else {
5614                series_bytes(&series)
5615            };
5616            let chunk = NativeChunk::Evaluated(EvaluatedChunk::try_new(series, true)?);
5617            self.queued.push_back((chunk, bytes));
5618            closed += 1;
5619        }
5620        // This node's own accounting of closed series is the results it closed, one per result
5621        // series: an input chunk that closed its series is a closure of the node below, and
5622        // counting it here as well would report the same work twice.
5623        self.buffer.metrics.record_completed_series(closed);
5624        // The input the reservation covered is gone and the queue holds what the kernel returned,
5625        // so the reservation settles on the queue: what a queued result is given back with as it
5626        // leaves is exactly the figure it is held for here.
5627        let queued = self
5628            .queued
5629            .iter()
5630            .fold(self.queued_tables, |total, (_, bytes)| {
5631                total.saturating_add(*bytes)
5632            });
5633        self.buffer.resize(queued.saturating_add(beside))
5634    }

5636    /// Folds one chunk into this node's aggregate.
5637    ///
5638    /// The chunk is reserved for before it is folded, and the reservation settles afterwards on the fold
5639    /// that is left: the fold keeps its states, so a chunk that closes its series leaves nothing behind.
5640    fn retain_aggregate(
5641        &mut self,
5642        aggregate: &mut AggregateRuntime,
5643        chunk: NativeChunk,
5644    ) -> Result<()> {
5645        // A `Final` aggregate merges state; the others fold values. Which one a chunk is has
5646        // already been checked against the leaf that queued it, so an unexpected kind here is a
5647        // tree the compiler should have refused.
5648        if let NativeChunk::AggregateState(chunk) = chunk {
5649            let pending = aggregate.pending_state(&chunk);
5650            let positions = match &chunk {
5651                StateChunk::Series { series, .. } => {
5652                    series.iter().map(|series| series.time.len()).sum()
5653                }
5654                StateChunk::Row(row) => row.time.len(),
5655            };
5656            self.buffer.grow(pending)?;
5657            let metrics = Arc::clone(&self.buffer.metrics);
5658            {
5659                let _timer = metrics.timer();
5660                aggregate.accumulate_state(chunk, &mut self.buffer, pending)?;
5661            }
5662            metrics.record_input(1, positions);
5663            return Ok(());
5664        }
5665        let NativeChunk::Evaluated(chunk) = chunk else {
5666            return Err(DataFusionError::Execution(
5667                "the aggregate consumes evaluated chunks, not raw samples".to_string(),
5668            ));
5669        };
5670        // The figure is read before the chunk is folded, because reserving after the fact would be
5671        // reserving for memory the fold already read.
5672        let pending = aggregate.pending(&chunk);
5673        let positions = chunk.series.time.len();
5674        self.buffer.grow(pending)?;
5675        let metrics = Arc::clone(&self.buffer.metrics);
5676        {
5677            let _timer = metrics.timer();
5678            aggregate.accumulate(chunk, &mut self.buffer, pending)?;
5679        }
5680        metrics.record_input(1, positions);
5681        // A chunk that closed its series closed it for the node below: a group is not a series, so
5682        // the fold completing nothing on it is why nothing is counted here. The series this node
5683        // closes are the ones it hands out, counted when it queues them.
5684        Ok(())
5685    }

5687    /// Finishes this node's fold and queues one chunk per group it produced.
5688    ///
5689    /// Each group is one series, decided in full by the fold, so the one chunk it is handed over as closes
5690    /// it.
5691    fn compute_aggregate(&mut self, aggregate: &mut AggregateRuntime) -> Result<()> {
5692        let metrics = Arc::clone(&self.buffer.metrics);
5693        let _timer = metrics.timer();
5694        let groups = aggregate.groups();
5695        let queue_capacity = self.queued.capacity();
5696        let queue_target = self.queued.len().saturating_add(groups);
5697        let queue_growth = queue_target
5698            .saturating_sub(queue_capacity)
5699            .saturating_mul(std::mem::size_of::<(NativeChunk, usize)>());
5700        self.buffer.grow(queue_growth)?;
5701        if groups > 0 {
5702            self.queued.try_reserve_exact(groups).map_err(|error| {
5703                DataFusionError::Execution(format!(
5704                    "cannot reserve aggregate output queue: {error}"
5705                ))
5706            })?;
5707        }
5708        let queue_bytes = queue_target
5709            .max(queue_capacity)
5710            .saturating_mul(std::mem::size_of::<(NativeChunk, usize)>());
5711        let output = aggregate.compute(&mut self.buffer, queue_bytes)?;
5712        // The fold has returned its series; reserve their handover before appending them to the queue.
5713        let Some(output) = output else {
5714            return Ok(());
5715        };
5716        let mut closed = 0usize;
5717        match output {
5718            AggregateOutput::Samples(series) => {
5719                for series in series {
5720                    let bytes = series_bytes(&series);
5721                    let chunk = NativeChunk::Evaluated(EvaluatedChunk::try_new(series, true)?);
5722                    self.queued.push_back((chunk, bytes));
5723                    closed += 1;
5724                }
5725            }
5726            // A partial aggregate hands its whole partition over as one state chunk: the groups in
5727            // it are the groups this execution folded, and the consumer closes each when its own
5728            // input ends rather than when a chunk arrives.
5729            AggregateOutput::State(chunk) => {
5730                let StateChunk::Series { series, .. } = &chunk else {
5731                    return Err(DataFusionError::Internal(
5732                        "Partial aggregate produced row state".into(),
5733                    ));
5734                };
5735                closed = series.len();
5736                let bytes = state_bytes(&chunk);
5737                self.queued
5738                    .push_back((NativeChunk::AggregateState(chunk), bytes));
5739            }
5740        }
5741        // A group is one series and this node closes each of them exactly once, on the one chunk it
5742        // hands it over as.
5743        self.buffer.metrics.record_completed_series(closed);
5744        // The output Vec has been consumed; the node now holds only queue backing and chunks.
5745        let queued_bytes = self
5746            .queued
5747            .iter()
5748            .fold(0usize, |total, (_, bytes)| total.saturating_add(*bytes));
5749        let queue_backing = self
5750            .queued
5751            .capacity()
5752            .saturating_mul(std::mem::size_of::<(NativeChunk, usize)>());
5753        self.buffer
5754            .resize(queued_bytes.saturating_add(queue_backing))
5755    }

5757    /// Feeds one chunk to this node's instant selection. `None` means the series has not ended yet.
5758    fn instant(
5759        &mut self,
5760        instant: &mut InstantRuntime,
5761        chunk: NativeChunk,
5762    ) -> Result<Option<EvaluatedChunk>> {
5763        let NativeChunk::Raw(chunk) = chunk else {
5764            return Err(DataFusionError::Execution(
5765                "the instant selection consumes raw samples, not an evaluated chunk".to_string(),
5766            ));
5767        };
5768        let (positions, closing) = (chunk.time.len(), chunk.is_last);
5769        // The chunk is held from here on: a selection keeps every chunk of a series until the one
5770        // that closes it, so what this node reserves for is the series it has buffered.
5771        self.buffer.grow(raw_bytes(&chunk))?;
5772        // The chunk that closes the series gathers it in one go, and the gather allocates a workspace of
5773        // its own beside the buffered chunks: the axis it concatenates, the descriptors that read it, the
5774        // two arrays the selection returns and the rows the builder keeps. It is reserved here, before it
5775        // exists, so a pool too small for it refuses the gather instead of learning about it afterwards.
5776        let scratch = if closing {
5777            instant.gather_bytes(&chunk)
5778        } else {
5779            0
5780        };
5781        self.buffer.grow(scratch)?;
5782        let metrics = Arc::clone(&self.buffer.metrics);
5783        let outcome = {
5784            // The timer covers this node's own work only: it is dropped before the driver polls
5785            // the input again.
5786            let _timer = metrics.timer();
5787            instant.push(chunk)
5788        };
5789        metrics.record_input(1, positions);
5790        if closing {
5791            // A series closes exactly once, on the chunk carrying `is_last`: the selection gathers
5792            // the whole series on it, so it is counted here rather than on the chunk handed out.
5793            metrics.record_completed_series(1);
5794        }
5795        // A failed gather gives its workspace back here rather than leaving it to the driver: the workspace
5796        // is this call's, not something the node keeps.
5797        let output = match outcome {
5798            Ok(output) => output,
5799            Err(error) => {
5800                self.settle_buffered(instant)?;
5801                return Err(error);
5802            }
5803        };
5804        if let Some(output) = &output {
5805            metrics.record_grid_points(output.series.time.len());
5806            metrics.record_output(1, valid_values(output));
5807            if instant.selection.value == InstantValue::Timestamp {
5808                self.track(output)?;
5809            }
5810        }
5811        // What this node still holds is the chunks of the series it is selecting: a whole series is held
5812        // until it ends. The chunk just consumed and the gather's workspace are both gone, so this gives
5813        // them back.
5814        self.settle_buffered(instant)?;
5815        Ok(output)
5816    }

5882        // evaluator has read none of them yet.

5890    /// Merges this window's collected input and evaluates it, one merged series at a time.
5891    ///
5892    /// Each merged series is handed to the evaluator as one closing series, so the window reports on it
5893    /// once. `merge` is taken here, so a window whose collected input failed to merge does not merge it
5894    /// again.
5895    fn flush_merge(
5896        &mut self,
5897        window: &mut RangeEvaluator,
5898        merge: &mut Option<MergeInput>,
5899    ) -> Result<()> {
5900        let Some(merge) = merge.take() else {
5901            return Ok(());
5902        };
5903        if merge.fragments.is_empty() {
5904            // Nothing arrived, so there is nothing to merge and nothing to answer: the window's
5905            // evaluator never read a series and the charge goes back.
5906            return self.buffer.shrink(merge.bytes);
5907        }
5908        // The workspace the merge allocates is charged before it runs, from the rows, segments and
5909        // fields of the fragments as they were collected.
5910        let workspace = merge.workspace_bytes();
5911        let held = merge.bytes.saturating_add(workspace);
5912        let schema = Arc::clone(&merge.schema);
5913        self.buffer.grow(workspace)?;
5914        // What the queue holds before this flush adds to it: a queued answer is charged until the pull that
5915        // hands it up returns its estimate, so every resize below has to leave it charged.
5916        let mut queued_bytes = self
5917            .queued
5918            .iter()
5919            .fold(self.queued_tables, |total, (_, bytes)| {
5920                total.saturating_add(*bytes)
5921            });
5922        // The fragments' charge stands for the merged series too -- a merged run is a slice of a
5923        // fragment's own segment, so no value is copied -- and the workspace stays charged until
5924        // the flush is over.
5925        let merged = merge_same_labelset(&schema, merge.fragments)?;
5926        for series in merged {
5927            let workspace = window_workspace_bytes(
5928                window,
5929                last_series_ms(&series),
5930                true,
5931                &series.fields,
5932                &series.metric,
5933            )?;
5934            self.buffer.grow(workspace)?;
5935            let output = {
5936                let metrics = Arc::clone(&self.buffer.metrics);
5937                let _timer = metrics.timer();
5938                window.push_series(&series, true)?
5939            };
5940            let Some(output) = output else {
5941                self.buffer.shrink(workspace)?;
5942                continue;
5943            };
5944            self.buffer
5945                .metrics
5946                .record_grid_points(output.series.time.len());
5947            self.buffer.metrics.record_output(1, valid_values(&output));
5948            self.track(&output)?;
5949            let retained = window.retained_bytes();
5950            let emitted = self.emitted_bytes();
5951            self.buffer.resize(
5952                retained
5953                    .saturating_add(emitted)
5954                    .saturating_add(held)
5955                    .saturating_add(queued_bytes),
5956            )?;
5957            let bytes = evaluated_bytes(&output);
5958            self.enqueue_output(output)?;
5959            queued_bytes = queued_bytes.saturating_add(bytes);
5960        }
5961        // The fragments and the merge's workspace go back, and whatever the flush produced stays
5962        // charged in the node's queue until the caller pulls it.
5963        self.buffer.shrink(held)

5966    /// Queues one output this node computed while flushing its collected input, charged as it is queued and
5967    /// given back as it leaves.

5971        self.queued
5972            .push_back((NativeChunk::Evaluated(chunk), bytes));
5973        Ok(())
5974    }

5977    fn take_output(&mut self) -> Result<Option<NativeChunk>> {
5978        let Some((chunk, bytes)) = self.queued.pop_front() else {
5979            return Ok(None);
5980        };
5981        self.buffer.shrink(bytes)?;
5982        self.release_queued_tables()?;
5983        Ok(Some(chunk))
5984    }

5987    fn release_queued_tables(&mut self) -> Result<()> {
5988        if self.queued.is_empty() && self.queued_tables > 0 {
5989            self.buffer
5990                .shrink(std::mem::take(&mut self.queued_tables))?;
5991        }
5992        Ok(())
5993    }

5995    /// Records what this node hands out under its output label set.
5996    ///
5997    /// A node that rewrites labels is where two input series become indistinguishable, so this is where a
5998    /// second one is refused: checking at the root would be too late for a window above, which would
5999    /// already have evaluated the two as two series.
6000    fn track(&mut self, output: &EvaluatedChunk) -> Result<()> {
6001        match self.emitted.as_mut() {
6002            Some(emitted) => emitted.occupy(output),
6003            // A node that keeps its input's labels cannot make two series one.
6004            None => Ok(()),
6005        }
6006    }

6013    }

6017    /// Queues one source chunk at this input, checking it against the binding the leaf was built with.
6018    ///
6019    /// A chunk of the other kind has no address here, so it is refused rather than queued and read as
6020    /// though it were the kind this leaf promised.

6035    ///
6036    /// The handover is what [`Self::open_source_series`] follows: after this call the flag says whether
6037    /// the series the consumer just received a chunk of is still open, which is what closing the execution
6038    /// reads to tell a truncated input from one that ended cleanly.
6039    fn pop_source(&mut self) -> Result<Option<NativeChunk>> {
6040        let Some(chunk) = self.state.pop_queued(NativeChunk::positions)? else {

6042        };

6044        // open any more, so the flag follows the handover and not the history of handovers.
6045        self.open_source_series = !chunk.is_last();

6047            self.state.buffer.metrics.record_completed_series(1);
6048        }

6050    }

6053/// Drives one execution of a bound tree.
6054///
6055/// The driver owns the state and nothing else: every decision a node makes was bound before the tree
6056/// existed, so a pull only reads the position and runs the bound operation.
6057///
6058/// A window directly over a source, a map at the root and a binary node over a constant all stream. A
6059/// window whose input can hold two series under one canonical label set collects that input instead and
6060/// reports at [`Self::finish`].
6061pub struct NativeExecutor {
6062    state: NativeExecutionState,
6063    phase: Phase,
6064    /// Whether the caller ended the input.
6065    ///
6066    /// Until then a pull that finds nothing means nothing has settled yet, and a binary node retains what
6067    /// its operands hand over without computing. [`NativeExecutor::finish`] drains once, checks that every
6068    /// stream closed, and only then sets this flag and drains again.
6069    inputs_finished: bool,
6070    /// Whether the caller ended the input.
6071    ///
6072    /// Until then a pull that finds nothing means nothing has settled yet. After it, the same pull runs the
6073    /// next stage of the ending, and only a pull that finds every stage done closes the nodes and returns
6074    /// their reservations.
6075    ended: bool,
6076}
6077
6078/// Whether the execution can still produce chunks.
6079#[derive(Clone, Debug, PartialEq, Eq)]
6080enum Phase {
6081    /// Chunks may still be queued and pulled.
6082    Running,
6083    /// The input ended and every node closed.
6084    Finished,
6085    /// A chunk failed, so the tree stopped: the evaluators must not be reused after an error.
6086    Failed(String),
6087}

6187    /// The next chunk the root produces, whatever kind it is: the tree's root output decides whether this
6188    /// is a value, a raw chunk or state.
6189    fn next_output(&mut self) -> Result<Option<NativeChunk>> {
6190        self.check_running()?;
6191        // The end of the input is walked here rather than in one call, so a chunk is handed over as
6192        // soon as the tree settles it: the caller pulls the answer instead of receiving it as one
6193        // collected value, and what it has not pulled yet stays charged to the node holding it.
6194        loop {
6195            // The state's tree is the one this execution was built from, and a pull is asked of its
6196            // root; the nodes below it read their own children from there.
6197            let inputs_finished = self.inputs_finished;
6198            match self.live_mut()?.pull(inputs_finished) {
6199                Ok(Some(chunk)) => return Ok(Some(chunk)),
6200                // Nothing has settled: the stages below decide whether that is a stall or an end.
6201                Ok(None) => {}
6202                Err(error) => {
6203                    self.stop(&error);
6204                    return Err(error);
6205                }
6206            }
6207            // Nothing has settled. While the caller is still pushing, that is a stall and the
6208            // answer is simply not ready; once it has ended the input, each stage below runs on the
6209            // first pull that finds the tree quiet, and the loop looks for what it produced.
6210            if !self.ended {
6211                return Ok(None);
6212            }
6213            if !self.inputs_finished {
6214                // The source queues are empty and every node has consumed what it was given, so a series the
6215                // caller never closed is still open here and nowhere else. Half a vector is not a vector, so
6216                // it is refused before a binary node computes.
6217                if let Err(error) = self.live()?.unclosed() {
6218                    self.stop(&error);
6219                    return Err(error);
6220                }
6221                self.inputs_finished = true;
6222                continue;
6223            }
6224            // A window that collects its input answers only once its own input is finished, and what a
6225            // binary node above it reads is that answer, so the deepest ready window is flushed first.
6226            match self.live_mut()?.flush_ready_merges() {
6227                // A window answered: what it queued is pulled on the next turn, and that pull is
6228                // also what carries its answer to a node above it.
6229                Ok(true) => continue,
6230                Ok(false) => {}
6231                Err(error) => {
6232                    self.stop(&error);
6233                    return Err(error);
6234                }
6235            }
6236            if self.live()?.any_merge_pending() {
6237                // A window still holds collected input, so something below it has not finished:
6238                // pulling again is what gives that node its turn.
6239                continue;
6240            }
6241            // Every node has answered everything it can. Closing them is what hands back the
6242            // reservations they were charged, so pulling to exhaustion leaves the pool as it found
6243            // it.
6244            let live = self.state.live.take().ok_or_else(no_live_execution)?;
6245            return match live.close() {
6246                Ok(()) => {
6247                    self.phase = Phase::Finished;
6248                    Ok(None)
6249                }
6250                Err(error) => {
6251                    self.stop(&error);
6252                    Err(error)
6253                }
6254            };
6255        }
6256    }

6345/// Maps one chunk with `ops`, keeping its axis and its closing.
6346///
6347/// Mapping is point-wise over valued positions: a float chunk keeps the rows it came with under math, a
6348/// chunk holding native histograms loses the rows only histograms filled, and a histogram function keeps
6349/// only the rows histograms filled. `None` means the operation emptied every position, which yields no
6350/// chunk at all unless this chunk closed its series: a closing chunk is still handed over, holding no
6351/// position.
6352///
6353/// `admit` is handed the bytes a label change can write before the change builds them; an error from it
6354/// stops the map.
6355fn map_chunk(
6356    schema: &Arc<SeriesSchema>,
6357    chunk: &EvaluatedChunk,
6358    ops: &[MapOp],
6359    admit: &mut dyn FnMut(usize) -> Result<()>,
6360) -> Result<Option<EvaluatedChunk>> {
6361    let mut series = chunk.series.clone();
6362    for (index, op) in ops.iter().enumerate() {
6363        let mapped = match op {
6364            MapOp::Label(rewrite) => {
6365                admit(rewrite.written_bytes(&series.metric))?;
6366                rewrite.apply(&mut series.metric)?;
6367                continue;
6368            }
6369            MapOp::Math(math) => map_math(schema, &series, *math)?,
6370            MapOp::Histogram(function) => map_native_histogram(*function, schema, &series)?,
6371        };
6372        if let Some(mapped) = mapped {
6373            series = mapped;
6374            continue;
6375        }
6376        // `clamp` with `max < min` is the empty vector, math skips native histograms and a histogram function
6377        // skips floats, so a series left without a value of the kind it reads has no row to hand over.
6378        if !chunk.is_last {
6379            return Ok(None);
6380        }
6381        // The closing chunk names the series the map would have produced, so the operations not
6382        // run still decide its labels.
6383        let mut metric = series.metric.clone();
6384        for op in &ops[index..] {
6385            match op {
6386                MapOp::Label(rewrite) => {
6387                    admit(rewrite.written_bytes(&metric))?;
6388                    rewrite.apply(&mut metric)?;
6389                }
6390                MapOp::Math(_) => metric = drop_metric_name(&metric)?,
6391                // `simpleHistogramFunc` drops only the name, not the type and unit.
6392                MapOp::Histogram(_) => metric.remove("__name__"),
6393            }
6394        }
6395        let empty = no_row_series(schema, &series, metric)?;
6396        return EvaluatedChunk::try_new(empty, true).map(Some);
6397    }
6398    EvaluatedChunk::try_new(series, chunk.is_last).map(Some)
6399}

6404/// operation produces -- a map drops the name, a filtering comparison keeps it -- so the caller says which.
6405fn no_row_series(schema: &SeriesSchema, series: &Series, metric: SeriesLabels) -> Result<Series> {
6406    let grid = series.time.grid().ok_or_else(|| {

6415    Series::try_new(schema, metric, TimeAxis::Grid(time), fields).map_err(series_error)
6416}

6550/// The broadcast a binary node with one scalar operand is, read off its binding.
6551///
6552/// One evaluated vector and one scalar is a broadcast over the vector, and two scalars of which one varies
6553/// is a broadcast over the varying one. A constant comes from the scalar node's own binding and
6554/// [`ScalarOperand`] preserves its side; a varying scalar is read from the chunk its node hands over.
6555///
6556/// [`NativeNode::validate`] refuses a binary operand that produces neither evaluated values nor a scalar
6557/// before any of this runs, so the last arm below is an exhaustiveness backstop.
6558fn broadcast_runtime(
6559    left: &NativeNode,
6560    right: &NativeNode,
6561    matching: &BinaryMatching,
6562) -> Result<BroadcastRuntime> {
6563    let literal = |node: &NativeNode| match &node.expr {
6564        NativeExpr::Scalar(BoundScalar { value }) => Some(*value),
6565        _ => None,
6566    };
6567    let (vector_left, sample_schema) = match (&left.output, &right.output) {
6568        (NativeType::Samples(vector), NativeType::Scalar(_)) if vector.eval_grid.is_some() => {
6569            (true, Arc::clone(vector))
6570        }
6571        (NativeType::Scalar(_), NativeType::Samples(vector)) if vector.eval_grid.is_some() => {
6572            (false, Arc::clone(vector))
6573        }
6574        (NativeType::Scalar(grid), NativeType::Scalar(_)) => {
6575            // Two constants are folded by the compiler; one that was not has no series to carry
6576            // the result. The executor test `an_unimplemented_node_is_reported_not_panicked`
6577            // builds this pair directly.
6578            if literal(left).is_some() && literal(right).is_some() {
6579                return Err(unsupported_binary(
6580                    "two constants, which broadcast over no vector",
6581                ));
6582            }
6583            (
6584                literal(left).is_none(),
6585                Arc::new(scalar_schema(Arc::clone(grid))),
6586            )
6587        }
6588        (_, _) => {
6589            return Err(unsupported_binary(
6590                "an operand that is neither evaluated samples nor a scalar, which no validated \
6591                 tree binds as a binary operand",
6592            ));
6593        }
6594    };
6595    let scalar_side = if vector_left { right } else { left };
6596    let scalar = match literal(scalar_side) {
6597        Some(value) if vector_left => BroadcastScalar::Literal(ScalarOperand::Right(value)),
6598        Some(value) => BroadcastScalar::Literal(ScalarOperand::Left(value)),
6599        None => BroadcastScalar::Dynamic {
6600            left: !vector_left,
6601            value: None,
6602        },
6603    };
6604    // A scalar has no labels, so the modifiers a vector–vector operation would use have no
6605    // meaning here: a tree that carries them describes another operation than the one it computes.
6606    if matching.labels != MatchingLabels::Ignoring(Vec::new())
6607        || matching.cardinality != Cardinality::OneToOne
6608        || matching.fill != FillValues::default()
6609    {
6610        return Err(matched_scalar());
6611    }
6612    Ok(BroadcastRuntime {
6613        scalar,
6614        vector_left,
6615        op: broadcast_op(matching)?,
6616        sample_schema,
6617        scalar_bytes: 0,
6618        annotations: None,
6619    })
6620}

6622/// Whether the [`Emitted`] table `broadcast` would keep only repeats the one its vector side keeps.
6623///
6624/// Literal arithmetic leaves missing samples missing and floats at their positions, so its output occupancy
6625/// is a subset of the window's, and the second label drop removes only empty-valued labels that
6626/// [`label_set`] ignores. Arithmetic therefore cannot create a collision the window did not already refuse.
6627fn elides_emitted_table(broadcast: &BroadcastRuntime, vector: &NativeNode) -> bool {
6628    matches!(broadcast.scalar, BroadcastScalar::Literal(_))
6629        && matches!(broadcast.op, BroadcastOp::Arithmetic(_))
6630        && matches!(
6631            &vector.expr,
6632            NativeExpr::Window { op, .. } if op.function.drops_metric_name()
6633        )
6634}

6636/// The operation a binary node's `matching` describes.
6637///
6638/// A comparison is a filter unless it was bound as `bool`, and a filter keeps the label set it read while
6639/// every other operation drops the metric name. Arithmetic has no `bool` form, so one bound that way is
6640/// refused.
6641fn broadcast_op(matching: &BinaryMatching) -> Result<BroadcastOp> {
6642    let arithmetic = |op: ArithmeticOp| {
6643        if matching.return_bool {
6644            Err(DataFusionError::Execution(
6645                "the binary node carries `bool` on an arithmetic operator, which has no bool form"
6646                    .to_string(),
6647            ))
6648        } else {
6649            Ok(BroadcastOp::Arithmetic(op))
6650        }
6651    };
6652    let comparison = |op: ComparisonOp| {
6653        if matching.return_bool {
6654            Ok(BroadcastOp::Compare(Comparison::Bool(op)))
6655        } else {
6656            Ok(BroadcastOp::Compare(Comparison::Filter(op)))
6657        }
6658    };
6659    match matching.op.kind() {
6660        BinaryKind::Arithmetic(op) => arithmetic(op),
6661        BinaryKind::Comparison(op) => comparison(op),
6662    }
6663}
