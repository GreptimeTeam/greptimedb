28/// The rule a repeated match group violates.
29#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
30pub enum MatchGroupViolation {
31    /// Several series share a match group on the side that must hold one series per group.
32    DuplicateOnOneSide { one_side_is_left: bool },
33    /// One-to-one matching found several matches for the same group.
34    ImplicitManyToOne,
35    /// A group modifier left several matches with the same result label set.
36    AmbiguousGroupLabels,
37}
38
39impl MatchGroupViolation {
40    fn message(&self, group: &str) -> String {
41        match self {
42            Self::DuplicateOnOneSide { one_side_is_left } => {
43                let side = if *one_side_is_left { "left" } else { "right" };
44                format!(
45                    "found duplicate series for the match group {group} on the {side} hand-side \
46                     of the operation; many-to-many matching not allowed: matching labels must \
47                     be unique on one side"
48                )
49            }
50            Self::ImplicitManyToOne => format!(
51                "multiple matches for labels {group}: many-to-one matching must be explicit \
52                 (group_left/group_right)"
53            ),
54            Self::AmbiguousGroupLabels => format!(
55                "multiple matches for labels {group}: grouping labels must ensure unique matches"
56            ),
57        }
58    }
59}
60
61/// Rejects a vector matching whose match groups are not unique.
62///
63/// Takes the per-group row count as first argument and the label columns that form the group
64/// as the remaining ones. Returns `true` when every group holds a single row, and fails the
65/// query otherwise; PromQL has no way to express the duplicate series it would produce.
66pub struct UniqueMatchGroup;
67
68impl UniqueMatchGroup {
69    pub const fn name() -> &'static str {
70        "prom_assert_unique_match_group"
71    }
72
73    pub fn scalar_udf(labels: Vec<String>, violation: MatchGroupViolation) -> ScalarUDF {
74        ScalarUDF::new_from_impl(AssertUniqueMatchGroup {
75            signature: Signature::variadic_any(Volatility::Volatile),
76            labels,
77            violation,
78        })
79    }
80}
81
82#[derive(Debug, Clone, PartialEq, Eq, Hash)]
83struct AssertUniqueMatchGroup {
84    signature: Signature,
85    /// Label names of the group, in the same order as the label arguments.
86    labels: Vec<String>,
87    violation: MatchGroupViolation,
88}
89
90impl AssertUniqueMatchGroup {
91    fn render_group(&self, args: &[ColumnarValue], row: usize) -> DfResult<String> {
92        let mut rendered = Vec::with_capacity(self.labels.len());
93        for (label, arg) in self.labels.iter().zip(args) {
94            let array = extract_array(arg)?;
95            // A scalar argument was expanded to a single row above.
96            let row = if array.len() == 1 { 0 } else { row };
97            if row >= array.len() || array.is_null(row) {
98                continue;
99            }
100            let value = array_value_to_string(&array, row)
101                .map_err(|e| DataFusionError::ArrowError(Box::new(e), None))?;
102            if value.is_empty() {
103                continue;
104            }
105            rendered.push(format!("{label}=\"{value}\""));
106        }
107        Ok(format!("{{{}}}", rendered.join(", ")))
108    }
109}
110
111impl ScalarUDFImpl for AssertUniqueMatchGroup {
112    fn name(&self) -> &str {
113        UniqueMatchGroup::name()
114    }
115
116    fn signature(&self) -> &Signature {
117        &self.signature
118    }
119
120    fn return_type(&self, _arg_types: &[DataType]) -> DfResult<DataType> {
121        Ok(DataType::Boolean)
122    }
123
124    fn invoke_with_args(&self, args: ScalarFunctionArgs) -> DfResult<ColumnarValue> {
125        let Some((counts, labels)) = args.args.split_first() else {
126            return Err(DataFusionError::Execution(format!(
127                "{} expects the match group row count as first argument",
128                UniqueMatchGroup::name()
129            )));
130        };
131        let counts = extract_array(counts)?;
132        let counts = counts
133            .as_any()
134            .downcast_ref::<Int64Array>()
135            .ok_or_else(|| {
136                DataFusionError::Execution(format!(
137                    "{} expects an Int64 match group row count, found {}",
138                    UniqueMatchGroup::name(),
139                    counts.data_type()
140                ))
141            })?;
142
143        if let Some(row) =
144            (0..counts.len()).find(|row| !counts.is_null(*row) && counts.value(*row) > 1)
145        {
146            let group = self.render_group(labels, row)?;
147            return Err(DataFusionError::Execution(self.violation.message(&group)));
148        }
149
150        Ok(ColumnarValue::Scalar(ScalarValue::Boolean(Some(true))))
151    }
152}

154#[cfg(test)]
155mod tests {
156    use std::sync::Arc;
157
158    use datafusion::arrow::array::StringArray;
159    use datafusion::arrow::datatypes::Field;
160
161    use super::*;
162
163    fn invoke(counts: Vec<i64>, hosts: Vec<Option<&str>>) -> DfResult<ColumnarValue> {
164        let udf = UniqueMatchGroup::scalar_udf(
165            vec!["host".to_string()],
166            MatchGroupViolation::ImplicitManyToOne,
167        );
168        let number_rows = counts.len();
169        udf.invoke_with_args(ScalarFunctionArgs {
170            args: vec![
171                ColumnarValue::Array(Arc::new(Int64Array::from(counts))),
172                ColumnarValue::Array(Arc::new(StringArray::from(hosts))),
173            ],
174            arg_fields: vec![
175                Arc::new(Field::new("count", DataType::Int64, true)),
176                Arc::new(Field::new("host", DataType::Utf8, true)),
177            ],
178            number_rows,
179            return_field: Arc::new(Field::new("assert", DataType::Boolean, false)),
180            config_options: Arc::new(Default::default()),
181        })
182    }
183
184    #[test]
185    fn unique_groups_pass() {
186        let result = invoke(vec![1, 1], vec![Some("a"), Some("b")]).unwrap();
187        assert!(matches!(
188            result,
189            ColumnarValue::Scalar(ScalarValue::Boolean(Some(true)))
190        ));
191    }
192
193    #[test]
194    fn duplicate_group_reports_its_labels() {
195        let err = invoke(vec![1, 2], vec![Some("a"), Some("b")]).unwrap_err();
196        assert!(
197            err.to_string()
198                .contains("multiple matches for labels {host=\"b\"}"),
199            "{err}"
200        );
201    }
202
203    #[test]
204    fn null_label_is_omitted_from_the_group() {
205        let err = invoke(vec![2], vec![None]).unwrap_err();
206        assert!(err.to_string().contains("labels {}"), "{err}");
207    }
208}
