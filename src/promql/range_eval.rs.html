47/// The samples a range selector reads at evaluation time `T`: source timestamps in
48/// `(E - range_ms, E]`, where `E` is `T - offset_ms`, or `anchor_ms - offset_ms` with `@`. A
49/// modifier widens that: `anchored` also reads the lookback before `E - range_ms`, `smoothed` the
50/// lookback on both sides, and a scan feeding the window has to read that much more.
51#[derive(Clone, Copy, Debug, serde::Serialize, serde::Deserialize, PartialEq, Hash, Eq)]
52pub struct RangeWindow {
53    /// Positive window length in milliseconds.
54    pub range_ms: i64,
55    /// Selector offset in milliseconds; positive reads earlier samples.
56    pub offset_ms: i64,
57    /// The resolved `@` timestamp in milliseconds, with `start()` and `end()` already replaced
58    /// by the statement's start and end.
59    pub anchor_ms: Option<i64>,
60    /// The experimental `anchored` or `smoothed` selector modifier, which also reads samples
61    /// around the window.
62    pub modifier: Option<RangeModifier>,
63}
64
65/// The experimental range selector modifiers of Prometheus, which read samples beyond the
66/// window to pin its boundary values. Not to be confused with an `@` anchor, which fixes the
67/// window's end.
68#[derive(Clone, Copy, Debug, PartialEq, Eq, serde::Serialize, serde::Deserialize, Hash)]
69pub enum RangeModifier {
70    /// `anchored`: the window also reads the last sample at or before its start, up to the
71    /// statement's lookback before it, and takes it as the value at the start.
72    Anchored { lookback_ms: i64 },
73    /// `smoothed`: the window also reads samples up to the lookback before its start and after its
74    /// end, and interpolates the values at both boundaries.
75    Smoothed { lookback_ms: i64 },
76}
77
78impl RangeModifier {
79    /// How far before the window start the window also reads.
80    pub(crate) fn before_ms(self) -> i64 {
81        match self {
82            Self::Anchored { lookback_ms } | Self::Smoothed { lookback_ms } => lookback_ms,
83        }
84    }
85
86    /// How far after the window end the window also reads.
87    pub(crate) fn after_ms(self) -> i64 {
88        match self {
89            Self::Anchored { .. } => 0,
90            Self::Smoothed { lookback_ms } => lookback_ms,
91        }
92    }
93
94    pub(crate) fn name(self) -> &'static str {
95        match self {
96            Self::Anchored { .. } => "anchored",
97            Self::Smoothed { .. } => "smoothed",
98        }
99    }
100}

120/// The windows of one field at consecutive grid points, as ranges into the field's retained
121/// samples.
122pub(crate) struct FloatWindows<'a> {
123    /// The retained samples, in time order.
124    pub timestamps: &'a [i64],
125    pub values: &'a [f64],
126    /// Counter resets over the retained samples, ascending by position.
127    pub resets: &'a [CounterReset],
128    /// Absolute position of `timestamps[0]`: a reset position is `base` plus a sample index.
129    pub base: u64,
130    /// One range per window, ascending. Each is a half-open index range `[start, end)`; the time
131    /// window it was derived from is left-open, `(T - range_ms, T]`.
132    pub windows: &'a [Range<usize>],
133    /// The evaluation timestamp of each window.
134    pub eval_ts: &'a [i64],
135    pub range_ms: i64,
136}

138impl FloatWindows<'_> {
139    /// The resets inside window `index`.
140    ///
141    /// `windows` index the retained arrays as they stand in this batch, not absolutely: each
142    /// span is relative to `timestamps` and `values`, which is why `base` plus a span start is
143    /// the absolute position of the sample the span starts at.
144    pub(crate) fn resets_of(&self, index: usize) -> &[CounterReset] {
145        let span = &self.windows[index];
146        resets_between(
147            self.resets,
148            self.base + span.start as u64,
149            self.base + span.end as u64,
150        )
151    }

161/// Evaluates range windows over raw series, one series at a time, handing out its results as they
162/// become final.
163///
164/// Each series is pushed as its chunks in order. A push evaluates the grid points whose windows
165/// the pushed samples complete, and hands them out as one [`EvaluatedChunk`], so a consumer can
166/// stream a long series instead of waiting for its last chunk. One chunk may hold several runs of
167/// positions, separated by stretches holding no value, and the axis names them. The series closes
168/// on a chunk marked `is_last`, which is handed out even when it holds no position.
169///
170/// Fields are evaluated independently over their own valid samples; a row holding a stale
171/// marker in any field is dropped, as the ordinary range selector does. Only the samples later
172/// windows can still use are retained. Native histogram samples need a [`RangeFunction`]; an
173/// evaluator built from a bare kernel rejects them.
174pub struct RangeEvaluator {
175    grid: Arc<EvalGrid>,
176    /// The window with `offset_ms` replaced by the offset samples are shifted by.
177    window: RangeWindow,
178    replay: Replay,
179    kernel: WindowKernel,
180    function: Option<RangeFunction>,
181    annotations: Option<PromqlAnnotationCollector>,
182    output_schema: Arc<SeriesSchema>,
183    output: GridSeriesBuilder,
184    /// The grid of evaluated input series, such as the inner result of a subquery; `None` for
185    /// raw input.
186    input_grid: Option<GridSpec>,
187    series: Option<SeriesState>,
188    fields: Vec<FieldSamples>,
189    row: Vec<Option<GridSample>>,
190    drops_metric_name: bool,
191    /// Raised by the first non-empty window, where Prometheus validates function arguments.
192    window_error: Option<String>,
193    /// Chunks appended by the bulk path, so a test can prove that path is reached rather than
194    /// only that the output is right, which the per-sample path would also produce.
195    #[cfg(test)]
196    bulk_ingests: usize,
197    /// The batch form of `function`, when it has one and the input allows it. Present only for
198    /// `Replay::PerPoint`, whose windows advance with the grid; an anchored selector replays one
199    /// window and gains nothing from batching.
200    batch: Option<BatchKernel>,
201    /// Windows accumulated for the next batch call.
202    pending: PendingBatch,
203    /// Batch kernel calls made so far. Test-only: both dispatch shapes produce the same output,
204    /// so a test comparing only output could not tell which one ran.
205    #[cfg(test)]
206    batch_calls: usize,
207}

209/// Windows accumulated for the next batch call: one list per field.
210struct PendingBatch {
211    /// Grid index of each accumulated point, ascending.
212    ///
213    /// Not assumed contiguous: `evaluate` skips the grid points whose window ends before the
214    /// series has a sample, so a batch can span such a gap and must write each point to its own
215    /// row.
216    indices: Vec<usize>,
217    /// Evaluation timestamp of each accumulated point.
218    eval_ts: Vec<i64>,
219    /// The retained-sample range of each point's window, one list per field.
220    ///
221    /// A span is relative to the retained arrays at the time it was recorded, not an absolute
222    /// index, so [`FieldSamples::advance`] invalidates it: that is why `evaluate` flushes the
223    /// batch before advancing.
224    spans: Vec<Vec<Range<usize>>>,
225    /// One output column per field, written by the batch kernel.
226    columns: Vec<Vec<Option<f64>>>,
227    /// Cursor into each field's retained timestamps; valid across batch flushes, but invalidated
228    /// when `FieldSamples::advance` drains the retained arrays.
229    ends: Vec<usize>,
230    /// Whether each cursor has been positioned by binary search since the last drain.
231    positioned: Vec<bool>,
232}
233
234impl PendingBatch {
235    fn with_fields(fields: usize) -> Self {
236        Self {
237            indices: Vec::new(),
238            eval_ts: Vec::new(),
239            spans: vec![Vec::new(); fields],
240            columns: vec![Vec::new(); fields],
241            ends: vec![0; fields],
242            positioned: vec![false; fields],
243        }
244    }
245
246    fn len(&self) -> usize {
247        self.eval_ts.len()
248    }
249
250    fn is_empty(&self) -> bool {
251        self.eval_ts.is_empty()
252    }
253
254    /// The bytes this batch's own vectors hold, by capacity.
255    fn bytes(&self) -> usize {
256        let spans: usize = self.spans.iter().map(|spans| spans.capacity()).sum();
257        let columns: usize = self.columns.iter().map(|column| column.capacity()).sum();
258        self.indices.capacity() * std::mem::size_of::<usize>()
259            + self.eval_ts.capacity() * std::mem::size_of::<i64>()
260            + spans * std::mem::size_of::<Range<usize>>()
261            + columns * std::mem::size_of::<Option<f64>>()
262            + self.ends.capacity() * std::mem::size_of::<usize>()
263            + self.positioned.capacity() * std::mem::size_of::<bool>()
264    }
265
266    /// Drops the accumulated points, keeping the allocations.
267    fn clear_points(&mut self) {
268        self.indices.clear();
269        self.eval_ts.clear();
270        for spans in &mut self.spans {
271            spans.clear();
272        }
273    }
274
275    fn reset_cursors(&mut self) {
276        self.ends.fill(0);
277        self.positioned.fill(false);
278    }
279}

284#[inline]
285fn timestamp_le(timestamp: i64, eval_ts: i64) -> bool {
286    #[cfg(test)]
287    POSITION_COMPARISONS.fetch_add(1, Ordering::Relaxed);
288    timestamp <= eval_ts
289}

291/// Records one point for the next batch call: its evaluation timestamp and, per field, the range
292/// of retained samples its window covers.
293fn accumulate(pending: &mut PendingBatch, index: usize, eval_ts: i64, fields: &[FieldSamples]) {
294    debug_assert_eq!(pending.spans.len(), fields.len());
295    pending.indices.push(index);
296    pending.eval_ts.push(eval_ts);
297    for (field_index, (spans, field)) in pending.spans.iter_mut().zip(fields).enumerate() {
298        // Cursors refer to retained arrays, which are unchanged by flushes. `head` only moves
299        // forward; never revisit samples already behind every remaining window.
300        pending.ends[field_index] = pending.ends[field_index].max(field.head);
301        if !pending.positioned[field_index] {
302            pending.ends[field_index] = field.timestamps[field.head..]
303                .partition_point(|&timestamp| timestamp_le(timestamp, eval_ts))
304                + field.head;
305            pending.positioned[field_index] = true;
306        } else {
307            while pending.ends[field_index] < field.timestamps.len()
308                && timestamp_le(field.timestamps[pending.ends[field_index]], eval_ts)
309            {
310                pending.ends[field_index] += 1;
311            }
312        }
313        spans.push(field.head..pending.ends[field_index]);
314    }
315}

317/// Runs the batch kernel over the accumulated points and appends its columns.
318fn flush_batch(
319    kernel: &mut BatchKernel,
320    pending: &mut PendingBatch,
321    fields: &[FieldSamples],
322    range_ms: i64,
323    row: &mut Vec<Option<GridSample>>,
324    output: &mut GridSeriesBuilder,
325    #[cfg(test)] batch_calls: &mut usize,
326) -> Result<()> {
327    if pending.is_empty() {
328        return Ok(());
329    }
330    // Counted here rather than at the call site, so the number a test reads is by construction
331    // the number of kernel calls that happened.
332    #[cfg(test)]
333    {
334        *batch_calls += 1;
335    }
336    for (index, field) in fields.iter().enumerate() {
337        let batch = FloatWindows {
338            timestamps: &field.timestamps,
339            values: &field.values,
340            resets: &field.resets,
341            base: field.base,
342            windows: &pending.spans[index],
343            eval_ts: &pending.eval_ts,
344            range_ms,
345        };
346        kernel(&batch, &mut pending.columns[index]);
347    }
348    for (offset, &grid_index) in pending.indices.iter().enumerate() {
349        row.clear();
350        for column in &pending.columns {
351            row.push(column[offset].map(GridSample::Float));
352        }
353        output.push_samples(grid_index, row)?;
354    }
355    pending.clear_points();
356    Ok(())
357}

375impl SeriesState {
376    /// The bytes this series' identity holds, which the rows handed out do not cover.
377    fn identity_bytes(&self) -> usize {
378        // A scan slices the primary key out of a decoded buffer, so the bytes the key shows are not
379        // the bytes it keeps: a key of a few bytes pins the allocation it was sliced out of.
380        // Charging the slice length instead would show a handful of bytes for an allocation the
381        // size of a decoded batch. One evaluator holds one series at a time, and an evaluated input
382        // holds no key at all, so there is no second slice of the same allocation to dedupe.
383        let primary_key = self
384            .primary_key
385            .as_ref()
386            .map_or(0, |key| key.capacity().max(key.len()));
387        let metric: usize = self.metric.name().map_or(0, |name| name.len())
388            + self
389                .metric
390                .tags()
391                .iter()
392                .map(|tag| tag.name.len() + tag.value.len())
393                .sum::<usize>();
394        std::mem::size_of::<SeriesState>() + primary_key + metric
395    }
396}

398/// How the windows of an `@`-anchored selector map to the grid.
399///
400/// Like Prometheus, an anchored selector is read once, as the window of the first grid point
401/// with the offset rewritten to `start - (anchor - offset)`.
402#[derive(Clone, Copy, Debug, PartialEq, Eq)]
403enum Replay {
404    /// Not anchored: every grid point reads its own window.
405    PerPoint,
406    /// The first point's result is reported at every point (step-invariant functions).
407    Once,
408    /// The first point's window is evaluated at every point (`predict_linear`).
409    EveryPoint,
410}

412#[derive(Default)]
413struct FieldSamples {
414    timestamps: Vec<i64>,
415    values: Vec<f64>,
416    /// Samples before `head` are behind every remaining window.
417    head: usize,
418    /// Native histogram samples, kept apart from the floats as Prometheus does.
419    histograms: Vec<(i64, FloatHistogram)>,
420    /// Histograms before `histogram_head` are behind every remaining window; their payloads are
421    /// released by [`advance_histograms`](Self::advance_histograms).
422    histogram_head: usize,
423    /// Estimated payload of the histograms this field still holds: the bucket lists and other
424    /// buffers the samples own, which no `size_of` of a slot shows.
425    ///
426    /// Measured for a histogram when it is admitted and released with it, so a retained sample is
427    /// not walked again on every push. A histogram is never modified once admitted, so the payload
428    /// its release subtracts is the one its admission added. Samples already behind every window
429    /// keep their payload in the figure until the drain releases them.
430    histogram_bytes: usize,
431    /// Sequence number of `values[0]`. Absolute and monotonic across the series this evaluator
432    /// evaluates, so a reset keeps its position when `head` samples are dropped instead of every
433    /// index being renumbered. An empty field holds the number of samples already dropped, which
434    /// keeps positions increasing when the next series starts.
435    base: u64,
436    /// Resets between consecutive retained samples, ascending by position: exactly the ones a
437    /// window over these samples can still contain. Maintained only for the counter-correcting
438    /// functions, and released as the samples they reach back to are dropped.
439    resets: Vec<CounterReset>,
440    /// Whether `resets` is maintained for this field.
441    index_resets: bool,
442}

444/// One counter reset: a sample that dropped below the sample before it.
445#[derive(Clone, Copy, Debug)]
446pub(crate) struct CounterReset {
447    /// Position of the later sample, as an absolute sequence number. Its predecessor is
448    /// `position - 1`, so a window holding the samples `[start, end)` contains this reset
449    /// exactly when `start < position < end`.
450    pub position: u64,
451    /// The value the reset adds back: the sample at `position - 1`, which is inside the window
452    /// whenever the reset is.
453    pub previous_value: f64,
454}
455
456/// The resets inside the window over the absolute positions `[start, end)`.
457///
458/// Left-open: a reset at `start` would add back a sample behind the window.
459fn resets_between(resets: &[CounterReset], start: u64, end: u64) -> &[CounterReset] {
460    let first = resets.partition_point(|reset| reset.position <= start);
461    let count = resets[first..].partition_point(|reset| reset.position < end);
462    &resets[first..first + count]
463}
464
465impl FieldSamples {
466    /// The bytes this field's buffers hold, by capacity: the samples, the history behind `head`
467    /// included, plus the estimated payload of the histograms still held.
468    fn bytes(&self) -> usize {
469        self.timestamps.capacity() * std::mem::size_of::<i64>()
470            + self.values.capacity() * std::mem::size_of::<f64>()
471            + self.histograms.capacity() * std::mem::size_of::<(i64, FloatHistogram)>()
472            + self.resets.capacity() * std::mem::size_of::<CounterReset>()
473            + self.histogram_bytes
474    }
475
476    fn new(index_resets: bool) -> Self {
477        Self {
478            index_resets,
479            ..Self::default()
480        }
481    }
482
483    /// Appends one valid sample, recording a counter reset when it dropped below the sample
484    /// before it.
485    ///
486    /// "Before it" means the previous entry of `values`, which leaves native histogram samples
487    /// out. A drop across an intervening histogram is therefore recorded as a reset of two
488    /// samples that are not neighbours. That is unobservable: a window holding both floats also
489    /// holds the histogram between them, so it is a mixed window, whose counter functions report
490    /// nothing; and a window starting after the earlier float excludes a reset on its first
491    /// sample. Should the mixed-window rules ever start reading the float resets, this has to
492    /// compare timestamps instead.
493    fn push(&mut self, timestamp: i64, value: f64) {
494        if self.index_resets
495            && let Some(&previous_value) = self.values.last()
496            && value < previous_value
497        {
498            self.resets.push(CounterReset {
499                position: self.base + self.values.len() as u64,
500                previous_value,
501            });
502        }
503        self.timestamps.push(timestamp);
504        self.values.push(value);
505    }
506
507    /// Appends samples that are all present, in bulk.
508    ///
509    /// Same result as calling [`push`](Self::push) once per sample, including the reset scan,
510    /// which stays per-sample for a field that maintains the index; the two appends become two
511    /// `memcpy`s.
512    fn push_bulk(&mut self, timestamps: &[i64], values: &[f64]) {
513        debug_assert_eq!(timestamps.len(), values.len());
514        if self.index_resets {
515            let start = self.values.len();
516            let mut previous = self.values.last().copied();
517            for (offset, &value) in values.iter().enumerate() {
518                if let Some(previous_value) = previous
519                    && value < previous_value
520                {
521                    self.resets.push(CounterReset {
522                        position: self.base + (start + offset) as u64,
523                        previous_value,
524                    });
525                }
526                previous = Some(value);
527            }
528        }
529        self.timestamps.extend_from_slice(timestamps);
530        self.values.extend_from_slice(values);
531    }
532
533    /// Drops the samples before `head` and the resets whose predecessor is one of them.
534    fn advance(&mut self) {
535        let head = std::mem::take(&mut self.head);
536        if head == 0 {
537            return;
538        }
539        self.base += head as u64;
540        self.timestamps.drain(..head);
541        self.values.drain(..head);
542        // A reset at `base` would add back the last dropped sample, so it goes with it.
543        let dropped = self
544            .resets
545            .partition_point(|reset| reset.position <= self.base);
546        self.resets.drain(..dropped);
547    }
548
549    /// Drops the histograms before `histogram_head`, releasing the payload they held.
550    ///
551    /// The dropped samples are measured as they go, which costs one measurement per dropped
552    /// histogram in total: the retained ones are never scanned again.
553    fn advance_histograms(&mut self) {
554        let head = std::mem::take(&mut self.histogram_head);
555        if head == 0 {
556            return;
557        }
558        let dropped: usize = self.histograms[..head]
559            .iter()
560            .map(|(_, histogram)| histogram.heap_bytes())
561            .sum();
562        self.histogram_bytes = self.histogram_bytes.saturating_sub(dropped);
563        self.histograms.drain(..head);
564    }
565
566    /// The resets inside the window holding `len` samples from `head`.
567    fn resets_in(&self, len: usize) -> &[CounterReset] {
568        let start = self.base + self.head as u64;
569        resets_between(&self.resets, start, start + len as u64)
570    }
571}

573/// The bytes the histograms a row holds box, beyond the slots they sit in.
574///
575/// A slot is an `Option<GridSample>`: a pointer and a tag. A result that is a native histogram
576/// points at a [`FloatHistogram`] on the heap, which owns its bucket lists in turn, and both stay
577/// alive between pushes. Only slots the last evaluated point left in place are counted: a push
578/// clears the row.
579fn boxed_bytes(row: &[Option<GridSample>]) -> usize {
580    row.iter()
581        .flatten()
582        .map(|sample| match sample {
583            GridSample::Histogram(histogram) => {
584                std::mem::size_of::<FloatHistogram>() + histogram.heap_bytes()
585            }
586            GridSample::Float(_) => 0,
587        })
588        .sum()
589}

592    /// Creates an evaluator of `function` for the series of `input` onto `grid`.
593    pub fn try_new(
594        input: &SeriesSchema,
595        grid: Arc<EvalGrid>,
596        window: RangeWindow,
597        function: RangeFunction,
598    ) -> Result<Self> {
599        let mut window_error = None;
600        if let RangeFunction::DoubleExponentialSmoothing { smoothing, trend } = function {
601            // Prometheus rejects factors outside (0, 1); NaN passes its check.
602            window_error = [("smoothing", smoothing), ("trend", trend)]
603                .into_iter()
604                .find(|(_, factor)| *factor <= 0.0 || *factor >= 1.0)
605                .map(|(name, factor)| {
606                    format!("invalid {name} factor. Expected: 0 < factor < 1, got: {factor}")
607                });
608        }
609        if let Some(modifier) = window.modifier {
610            check_modifier(input, modifier, function)?;
611        }
612        let fold_offset = fold_offset(grid.spec(), window)?;
613        let step_invariant = !matches!(function, RangeFunction::PredictLinear(_));
614        if !step_invariant {
615            predict_linear_instants(grid.spec(), fold_offset)?;
616        }
617        // A modified window corrects counters over the samples it holds rather than through the
618        // reset index, and has no batch form.
619        let (kernel, index_resets) = match window.modifier {
620            Some(modifier) => (function.modified_kernel(modifier), false),
621            None => (
622                function.kernel(fold_offset),
623                function.indexes_counter_resets(),
624            ),
625        };
626        let mut evaluator =
627            Self::with_kernel(input, grid, window, kernel, step_invariant, index_resets)?;
628        evaluator.drops_metric_name = function.drops_metric_name();
629        evaluator.window_error = window_error;
630        evaluator.function = Some(function);
631        // An anchored selector replays one window at every point, so its windows do not advance
632        // with the grid and a batch would only repeat the same call.
633        if matches!(evaluator.replay, Replay::PerPoint) && window.modifier.is_none() {
634            evaluator.batch = function.batch_kernel();
635        }
636        Ok(evaluator)
637    }

640    /// `index_resets` tells whether the kernel corrects counters, which is the only case where
641    /// the per-field reset index is worth maintaining.
642    pub(crate) fn with_kernel(
643        input: &SeriesSchema,
644        grid: Arc<EvalGrid>,

714    /// An estimate of the bytes this evaluator keeps between pushes.
715    ///
716    /// Counted by capacity, so it is what the storage holds rather than what it uses: the retained
717    /// samples per field (timestamps, values, the reset index and the native histograms with the
718    /// payload each of them owns), the row a push fills with the histograms it boxes, the identity
719    /// of the open series and the pending batch's own vectors. Left out are the kernel's scratch
720    /// buffers, this struct's `Vec` headers and the grid, which every chunk addressed on it shares.
721    ///
722    /// The figure is an estimate, not a ledger of every allocation: a histogram's payload is
723    /// measured once when it is admitted and released with it, so neither the retained samples nor
724    /// the transient allocations of a push are walked again.
725    ///
726    /// A push is the only caller that also holds the incoming chunk, and it charges that itself for
727    /// as long as the call runs. The figure must not be per-chunk either: charging every chunk
728    /// consumed would make the charge grow with the chunk count, so a fixed window over a long
729    /// series would read as an allocation failure even though the window holds no more samples.
730    pub(crate) fn retained_bytes(&self) -> usize {
731        let fields: usize = self.fields.iter().map(FieldSamples::bytes).sum();
732        let row = self.row.capacity() * std::mem::size_of::<Option<GridSample>>();
733        fields
734            + row
735            + boxed_bytes(&self.row)
736            + self.series.as_ref().map_or(0, SeriesState::identity_bytes)
737            + self.pending.bytes()
738    }

740    /// An upper bound of the rows the next push hands out, for input whose last sample is at
741    /// `last_ts` (`None` when it holds none) and which closes its series when `last`.
742    ///
743    /// A push writes at most one row per point it decides, from the series' cursor -- the first
744    /// point for a series not started yet. Unless the series closes, it decides no point past the
745    /// time the input has reached; and a point holds a row only while its window still reaches the
746    /// latest sample, so the points after that one's window are decided empty. A replayed `@`
747    /// result fills every grid point from the one window it evaluates first.
748    pub(crate) fn output_rows_bound(&self, last_ts: Option<i64>, last: bool) -> usize {
749        let spec = self.grid.spec();
750        let len = usize::try_from(spec.len).unwrap_or(usize::MAX);
751        let state = self.series.as_ref();
752        let Some(latest) = last_ts.or_else(|| state.and_then(|state| state.last_ts)) else {
753            return 0;
754        };
755        let cursor = state.map_or(0, |state| state.cursor);
756        if !matches!(self.replay, Replay::PerPoint) {
757            return if cursor == 0 { len } else { 0 };
758        }
759        // The grid points whose evaluation time is before `bound`.
760        let (start, step) = (i128::from(spec.start_ms), i128::from(spec.step_ms));
761        let points_before = |bound: i128| {
762            let offset = bound - start;
763            if offset <= 0 {
764                return 0;
765            }
766            usize::try_from((offset + step - 1) / step).map_or(len, |points| points.min(len))
767        };
768        let shifted = i128::from(latest) + i128::from(self.window.offset_ms);
769        let before = self
770            .window
771            .modifier
772            .map_or(0, |modifier| i128::from(modifier.before_ms()));
773        let mut end = points_before(shifted + i128::from(self.window.range_ms) + before);
774        if !last && last_ts.is_some() {
775            end = end.min(points_before(shifted + 1));
776        }
777        end.saturating_sub(cursor)
778    }

800    /// Consumes the next chunk of the current series, or starts a new series.
801    ///
802    /// Returns every position this chunk decided, in one chunk addressed absolutely on the
803    /// evaluation grid; `None` when it decided no position holding a value. A decided position
804    /// without a result is a row of no chunk, so the positions between two runs of one chunk hold
805    /// no value and cost nothing. One push hands out at most one chunk.
806    ///
807    /// A series always closes exactly once: on the chunk holding its last rows, or on an empty
808    /// chunk after it when nothing is left. The evaluator must not be reused after an error.
809    pub fn push(&mut self, chunk: &RawSeriesChunk) -> Result<Option<EvaluatedChunk>> {
810        if self.input_grid.is_some() {
811            return Err(DataFusionError::Execution(
812                "raw series chunk pushed to an evaluator of evaluated series".into(),
813            ));
814        }
815        self.check_chunk(chunk)?;
816        self.ingest(chunk.time.as_buffer(), &chunk.fields, true)?;
817        let last_ts = chunk.time.as_buffer()[chunk.time.len() - 1];
818        if !chunk.is_last {
819            // Later chunks only hold samples after `last_ts`.
820            self.evaluate(Some(
821                i128::from(last_ts) + i128::from(self.window.offset_ms),
822            ))?;
823            return self.emit();
824        }
825        self.close_series()
826    }

828    /// Evaluates one run of an evaluated series, such as the inner result of a subquery: the
829    /// value at each grid point is a sample at that time. Stale markers are kept as values,
830    /// since the input is computed.
831    ///
832    /// `last` says whether this delivery closes its series. A stream hands over the same logical
833    /// series more than once -- a prefix now, its closing chunk later -- and the windows of a later
834    /// delivery read the samples of the earlier ones, so the state a series retains lives on across
835    /// them: only the `last` call ends it. Handing the deliveries over without saying which is last
836    /// would close the series early and lose exactly the samples a window across the boundary
837    /// needs.
838    ///
839    /// One delivery may hold several runs of positions, separated by stretches holding no value;
840    /// they all reach the caller in one chunk, addressed by the axis. A delivery that holds no
841    /// position at all is a legal close when it is `last`, and an error otherwise.
842    pub fn push_series(&mut self, series: &Series, last: bool) -> Result<Option<EvaluatedChunk>> {
843        let time = &series.time;
844        let Some(grid) = time.grid() else {
845            return Err(DataFusionError::Execution(
846                "evaluated range input must be a grid series".into(),
847            ));
848        };
849        if self.input_grid != Some(grid.spec())
850            || series.fields.len() != self.fields.len()
851            || series.fields.iter().any(|field| field.len() != time.len())
852        {
853            return Err(DataFusionError::Execution(
854                "evaluated series does not match the input schema".into(),
855            ));
856        }
857        // The input may skip positions: it is ingested by the positions it carries, and the points
858        // it skips hold no sample, which is exactly what a hole means.
859        let timestamps = time.timestamps().map_err(external)?;
860        let last_ts = timestamps.last().copied();
861        if !last && last_ts.is_none() {
862            // An empty delivery inside a series decides nothing: it is indistinguishable from a
863            // stall, so only the closing `is_last` one is a legal close.
864            return Err(DataFusionError::Execution(
865                "evaluated range input holds no position before its last run".into(),
866            ));
867        }
868        match &mut self.series {
869            None => {
870                self.output.begin_series();
871                self.series = Some(SeriesState {
872                    primary_key: None,
873                    metric: series.metric.clone(),
874                    last_ts,
875                    cursor: 0,
876                    closing: false,
877                    emitted_end: 0,
878                });
879            }
880            Some(state) => {
881                // A run continues its series; the earlier runs' samples stay retained for it.
882                if state.primary_key.is_some() {
883                    return Err(DataFusionError::Execution(
884                        "evaluated series pushed inside a raw series".into(),
885                    ));
886                }
887                if state.closing {
888                    return Err(DataFusionError::Execution(
889                        "evaluated series pushed after it closed".into(),
890                    ));
891                }
892                if state.metric != series.metric {
893                    return Err(DataFusionError::Execution(
894                        "evaluated series changed identity before its last run".into(),
895                    ));
896                }
897                if let Some(last_ts) = last_ts {
898                    if let Some(previous) = state.last_ts
899                        && timestamps[0] <= previous
900                    {
901                        return Err(DataFusionError::Execution(format!(
902                            "evaluated series runs are not increasing: {} after {previous}",
903                            timestamps[0]
904                        )));
905                    }
906                    state.last_ts = Some(last_ts);
907                }
908            }
909        }
910        if let Some(last_ts) = last_ts {
911            self.ingest(&timestamps, &series.fields, false)?;
912            if last {
913                return self.close_series();
914            }
915            // Later runs only hold samples after `last_ts`, so every point up to it is decided.
916            // The samples are shifted onto the evaluation timeline by the selector offset, so the
917            // ready point is on that timeline too.
918            self.evaluate(Some(
919                i128::from(last_ts) + i128::from(self.window.offset_ms),
920            ))?;
921            return self.emit();
922        }
923        // A closing chunk holding no position: there is nothing to ingest, but the samples the
924        // earlier deliveries left retained still decide the windows that follow them, so the close
925        // finishes them rather than ending the series on the spot.
926        debug_assert!(last, "an empty chunk that is not last is rejected above");
927        self.close_series()
928    }

930    /// Evaluates the rest of the series of the last input chunk and closes it.
931    ///
932    /// The rows still buffered are handed out here as one chunk, and the chunk that sees nothing
933    /// behind it carries `is_last`.
934    fn close_series(&mut self) -> Result<Option<EvaluatedChunk>> {
935        self.evaluate(None)?;
936        self.series
937            .as_mut()
938            .expect("every path that closes a series started one")
939            .closing = true;
940        self.emit()
941    }

943    /// Hands out the rows buffered so far as one chunk, and closes the series when nothing is left.
944    ///
945    /// The chunk holds every row decided since the last hand-out, in one go and in one [`Series`]:
946    /// several runs of positions may sit inside it, separated by stretches holding no value, and
947    /// the axis names them. A position decided to hold none costs no row, so it lies between the
948    /// runs rather than inside one. `is_last` closes the series, and a series always closes exactly
949    /// once: on the chunk that carries its last rows, or on an empty chunk after it when no row is
950    /// left.
951    fn emit(&mut self) -> Result<Option<EvaluatedChunk>> {
952        let Some(state) = self.series.as_ref() else {
953            return Ok(None);
954        };
955        let closing = state.closing;
956        let emitted_end = state.emitted_end;
957        let mut metric = state.metric.clone();
958        if self.drops_metric_name {
959            metric.remove_all(&METADATA_LABELS);
960        }
961        let Some(series) = self.output.take_buffered(metric)? else {
962            return if closing {
963                self.empty_chunk(emitted_end)
964            } else {
965                Ok(None)
966            };
967        };
968        // The builder addresses every position it hands out, so the chunk's bounds come from the
969        // axis itself: `first_position` is where it starts, and `last_position` where it ends.
970        let start = series
971            .time
972            .first_position()
973            .expect("the builder addresses every row on the grid");
974        let end = series.time.last_position().map_or(start, |last| last + 1);
975        debug_assert!(start >= emitted_end, "a chunk never goes back");
976        if closing {
977            // The buffer was drained into this chunk, so nothing of the series is left behind it.
978            self.series = None;
979            EvaluatedChunk::try_new(series, true).map(Some)
980        } else {
981            self.series
982                .as_mut()
983                .expect("checked at the entry of emit")
984                .emitted_end = end;
985            EvaluatedChunk::try_new(series, false).map(Some)
986        }
987    }

1016    fn check_chunk(&mut self, chunk: &RawSeriesChunk) -> Result<()> {
1017        let primary_key = Some(chunk.primary_key.as_slice());
1018        let time = chunk.time.as_buffer();
1019        if time.is_empty()
1020            || chunk.fields.len() != self.fields.len()
1021            || chunk.fields.iter().any(|field| field.len() != time.len())
1022        {
1023            return Err(DataFusionError::Execution(format!(
1024                "raw series chunk with {} timestamps and {} fields does not match {} schema fields",
1025                time.len(),
1026                chunk.fields.len(),
1027                self.fields.len()
1028            )));
1029        }
1030        match &mut self.series {
1031            None => {
1032                let metric = chunk.metric.clone().ok_or_else(|| {
1033                    DataFusionError::Execution(
1034                        "range evaluation requires materialized series labels".into(),
1035                    )
1036                })?;
1037                self.output.begin_series();
1038                self.series = Some(SeriesState {
1039                    primary_key: Some(chunk.primary_key.clone()),
1040                    metric,
1041                    last_ts: Some(time[time.len() - 1]),
1042                    cursor: 0,
1043                    closing: false,
1044                    emitted_end: 0,
1045                });
1046            }
1047            Some(state) => {
1048                if state.primary_key.as_ref().map(|key| key.as_slice()) != primary_key
1049                    || chunk.metric.as_ref() != Some(&state.metric)
1050                {
1051                    return Err(DataFusionError::Execution(
1052                        "raw series identity changed before its last chunk".into(),
1053                    ));
1054                }
1055                if let Some(last_ts) = state.last_ts
1056                    && time[0] <= last_ts
1057                {
1058                    return Err(DataFusionError::Execution(format!(
1059                        "raw series timestamps are not increasing across chunks: {} after {last_ts}",
1060                        time[0]
1061                    )));
1062                }
1063                state.last_ts = Some(time[time.len() - 1]);
1064            }
1065        }
1066        Ok(())
1067    }

1069    /// Appends the chunk in bulk when every field is an entirely present, unshifted float column.
1070    ///
1071    /// Returns `false` when any condition of the per-sample path applies, so the caller can run
1072    /// it. A chunk carrying a stale marker takes that path too: the marker drops the whole row,
1073    /// and reproducing that here would cost the same scan the loop already makes.
1074    ///
1075    /// A field mixing floats and native histograms never qualifies: `dense()` reports
1076    /// [`DenseSamplesRef::Mixed`] for it, keeping its windows on the histogram path where the
1077    /// mixed-window rules decide the result.
1078    fn ingest_in_bulk(
1079        &mut self,
1080        timestamps: &[i64],
1081        fields: &SeriesFields,
1082        drop_stale: bool,
1083    ) -> bool {
1084        // Lengths are checked by `check_chunk` for raw input and by `push_series` for evaluated
1085        // input, so only the shape of the columns is decided here.
1086        if self.window.offset_ms != 0 {
1087            return false;
1088        }
1089        let mut columns = Vec::with_capacity(fields.len());
1090        for field in fields.iter() {
1091            let Some(DenseSamplesRef::Float(values)) = field.dense() else {
1092                return false;
1093            };
1094            if values.null_count() != 0 {
1095                return false;
1096            }
1097            columns.push(values);
1098        }
1099        // A kept sample's shifted time must fit `i64`; with no offset it is the source timestamp,
1100        // which `RawTimestamps` already validated as increasing and which the frame widened into
1101        // this array without narrowing, so there is nothing left to check here.
1102        if drop_stale
1103            && columns.iter().any(|values| {
1104                values
1105                    .values()
1106                    .iter()
1107                    .any(|value| is_prometheus_stale_nan(*value))
1108            })
1109        {
1110            return false;
1111        }
1112        for (field, values) in self.fields.iter_mut().zip(columns) {
1113            field.push_bulk(timestamps, values.values());
1114        }
1115        #[cfg(test)]
1116        {
1117            self.bulk_ingests += 1;
1118        }
1119        true
1120    }

1122    /// Retains the samples of `fields` at `timestamps`. From `raw` input a row holding a stale
1123    /// marker in any field is dropped; evaluated input keeps it as a value.
1124    fn ingest(&mut self, timestamps: &[i64], fields: &SeriesFields, raw: bool) -> Result<()> {
1125        if self.ingest_in_bulk(timestamps, fields, raw) {
1126            return Ok(());
1127        }
1128        let mut samples: Vec<_> = fields.iter().map(|field| field.iter()).collect();
1129        let mut row = Vec::with_capacity(samples.len());
1130        for &timestamp in timestamps {
1131            row.clear();
1132            let mut stale = false;
1133            for field in &mut samples {
1134                let sample = field.next().expect("field length checked by the caller");
1135                match &sample {
1136                    SampleRef::Missing => {}
1137                    SampleRef::Float(value) => stale |= is_prometheus_stale_nan(*value),
1138                    SampleRef::Histogram(view) => {
1139                        if self.function.is_none() {
1140                            return Err(DataFusionError::NotImplemented(
1141                                "range evaluation of histogram samples".into(),
1142                            ));
1143                        }
1144                        stale |= is_prometheus_stale_nan(view.sum);
1145                    }
1146                }
1147                row.push(sample);
1148            }
1149            if stale && raw {
1150                continue;
1151            }
1152            // Like RangeManipulate, any kept sample whose shifted time leaves i64 fails the series.
1153            let shifted = timestamp
1154                .checked_add(self.window.offset_ms)
1155                .ok_or_else(|| {
1156                    DataFusionError::Execution(format!(
1157                        "sample timestamp {timestamp} overflows with offset {} ms",
1158                        self.window.offset_ms
1159                    ))
1160                })?;
1161            // Refused here rather than when a point is evaluated: a smoothed point can be decided
1162            // before the rest of its window arrives, so whether a later histogram is seen would
1163            // depend on how the input is chunked.
1164            if let Some(modifier) = self.window.modifier
1165                && row
1166                    .iter()
1167                    .any(|sample| matches!(sample, SampleRef::Histogram(_)))
1168                && self.decided_window_reads(shifted)
1169            {
1170                return Err(DataFusionError::NotImplemented(format!(
1171                    "the {} range modifier over native histogram samples",
1172                    modifier.name()
1173                )));
1174            }
1175            for (field, sample) in self.fields.iter_mut().zip(&row) {
1176                match sample {
1177                    SampleRef::Missing => {}
1178                    SampleRef::Float(value) => {
1179                        field.push(shifted, *value);
1180                    }
1181                    SampleRef::Histogram(view) => {
1182                        let mut h = FloatHistogram::try_from_view(view)?;
1183                        // Like Prometheus' `evalSubquery`: an explicit counter reset hint
1184                        // describes the sample's raw neighbours, not its evaluated ones.
1185                        if !raw && h.reset_hint != CounterResetHint::Gauge {
1186                            h.reset_hint = CounterResetHint::Unknown;
1187                        }
1188                        field.histogram_bytes =
1189                            field.histogram_bytes.saturating_add(h.heap_bytes());
1190                        field.histograms.push((shifted, h));
1191                    }
1192                }
1193            }
1194        }
1195        Ok(())
1196    }

1198    /// Whether a sample at `timestamp` on the evaluation timeline lies in the window of a point
1199    /// this evaluator decides: a window ending at `E` reads `(E - range - before, E + after]`, and
1200    /// only the first point's window is read when the result is replayed.
1201    fn decided_window_reads(&self, timestamp: i64) -> bool {
1202        let (before, after) = self.window.modifier.map_or((0, 0), |modifier| {
1203            (
1204                i128::from(modifier.before_ms()),
1205                i128::from(modifier.after_ms()),
1206            )
1207        });
1208        let spec = self.grid.spec();
1209        let ends = match self.replay {
1210            Replay::PerPoint => i128::from(spec.len),
1211            Replay::Once | Replay::EveryPoint => 1,
1212        };
1213        let (start, step) = (i128::from(spec.start_ms), i128::from(spec.step_ms));
1214        // The window ends that read the sample are those in `[timestamp - after, timestamp +
1215        // range + before)`; the first grid point at or after the lower bound decides.
1216        let lowest = i128::from(timestamp) - after;
1217        let highest = i128::from(timestamp) + i128::from(self.window.range_ms) + before;
1218        let first = if lowest <= start {
1219            0
1220        } else {
1221            (lowest - start + step - 1) / step
1222        };
1223        first < ends && start + first * step < highest
1224    }

1226    /// Evaluates grid points up to `ready` on the evaluation timeline, or all remaining points
1227    /// when `ready` is `None`.
1228    fn evaluate(&mut self, ready: Option<i128>) -> Result<()> {
1229        let points_read = self.points_read();
1230        let state = self.series.as_mut().expect("checked by check_chunk");
1231        let spec = self.grid.spec();
1232        let grid = self.grid.timestamps().map_err(external)?;
1233        // A modifier also reads samples up to the lookback ahead of the window start, so they are
1234        // kept that much longer.
1235        let before = self
1236            .window
1237            .modifier
1238            .map_or(0, |modifier| i128::from(modifier.before_ms()));
1239        // `smoothed` interpolates the window end with the first sample at or after it, which can
1240        // arrive up to the lookback later. A point is decided once every field holds a sample at
1241        // or after it, or once the lookback past it has arrived. The input's last row does not say
1242        // that alone: a stale row is dropped, and a field can hold no value in it.
1243        let ready = match (ready, self.window.modifier) {
1244            (Some(ready), Some(RangeModifier::Smoothed { lookback_ms })) => {
1245                let held = self
1246                    .fields
1247                    .iter()
1248                    .map(|field| {
1249                        field
1250                            .timestamps
1251                            .last()
1252                            .map_or(i128::MIN, |&t| i128::from(t))
1253                    })
1254                    .min()
1255                    .unwrap_or(i128::MIN);
1256                Some(held.max(ready - i128::from(lookback_ms)))
1257            }
1258            (ready, _) => ready,
1259        };
1260        let range = i128::from(self.window.range_ms) + before;
1261        let mut empty_tail = false;
1262        while state.cursor < points_read {
1263            let eval_ts = grid[state.cursor];
1264            let lower = i128::from(eval_ts) - range;
1265            let mut earliest = None;
1266            // Later windows start no earlier, so samples behind this one go even when the point
1267            // itself is not ready yet.
1268            for field in &mut self.fields {
1269                field.head += field.timestamps[field.head..]
1270                    .partition_point(|&timestamp| i128::from(timestamp) <= lower);
1271                field.histogram_head += field.histograms[field.histogram_head..]
1272                    .partition_point(|(timestamp, _)| i128::from(*timestamp) <= lower);
1273                let first_histogram = field.histograms.get(field.histogram_head);
1274                for &timestamp in field
1275                    .timestamps
1276                    .get(field.head)
1277                    .into_iter()
1278                    .chain(first_histogram.map(|(timestamp, _)| timestamp))
1279                {
1280                    earliest = Some(earliest.map_or(timestamp, |min: i64| min.min(timestamp)));
1281                }
1282            }
1283            if ready.is_some_and(|ready| i128::from(eval_ts) > ready) {
1284                break;
1285            }
1286            // No retained sample: every point from here on is decided too. A sample that still
1287            // has to arrive lies beyond the window of every one of them, so none can gain a value.
1288            let Some(earliest) = earliest else {
1289                empty_tail = true;
1290                break;
1291            };
1292            if earliest > eval_ts {
1293                // Skip the points whose windows end before the earliest sample. A smoothed window
1294                // reads past its end, but without a sample at or before its end it has no result,
1295                // as Prometheus returns nothing when the first sample is after the end.
1296                let step = i128::from(spec.step_ms);
1297                let steps = (i128::from(earliest) - i128::from(spec.start_ms) + step - 1) / step;
1298                state.cursor = usize::try_from(steps).map_or(grid.len(), |i| i.min(grid.len()));
1299                continue;
1300            }
1301            if let Some(error) = &self.window_error {
1302                return Err(DataFusionError::Execution(error.clone()));
1303            }
1304
1305            let mut evaluator = RowEvaluator {
1306                fields: &self.fields,
1307                kernel: &mut self.kernel,
1308                function: self.function,
1309                annotations: self.annotations.as_ref(),
1310                range_ms: self.window.range_ms,
1311                offset_ms: self.window.offset_ms,
1312                modifier: self.window.modifier,
1313            };
1314            match self.replay {
1315                Replay::PerPoint => {
1316                    // A window holding a native histogram is decided by the mixed-window rules,
1317                    // which the batch kernel -- reading floats alone -- cannot reproduce. Flush
1318                    // what is pending and decide this point on the per-window path.
1319                    //
1320                    // Histograms are retained in time order, so the first one still ahead of the
1321                    // window head decides: a later one cannot be inside this window if this one
1322                    // is not. The check runs only when a batch kernel is set, so a function
1323                    // without one never pays for it.
1324                    let histogram_window = self.batch.is_some()
1325                        && self.fields.iter().any(|field| {
1326                            field.histograms.get(field.histogram_head).is_some_and(
1327                                |(timestamp, _)| i128::from(*timestamp) <= i128::from(eval_ts),
1328                            )
1329                        });
1330                    if let Some(kernel) = self.batch.as_mut()
1331                        && !histogram_window
1332                    {
1333                        accumulate(&mut self.pending, state.cursor, eval_ts, &self.fields);
1334                        if self.pending.len() >= EVALUATION_BATCH {
1335                            flush_batch(
1336                                kernel,
1337                                &mut self.pending,
1338                                &self.fields,
1339                                self.window.range_ms,
1340                                &mut self.row,
1341                                &mut self.output,
1342                                #[cfg(test)]
1343                                &mut self.batch_calls,
1344                            )?;
1345                        }
1346                        state.cursor += 1;
1347                        continue;
1348                    }
1349                    if let Some(kernel) = self.batch.as_mut() {
1350                        flush_batch(
1351                            kernel,
1352                            &mut self.pending,
1353                            &self.fields,
1354                            self.window.range_ms,
1355                            &mut self.row,
1356                            &mut self.output,
1357                            #[cfg(test)]
1358                            &mut self.batch_calls,
1359                        )?;
1360                    }
1361                    evaluator.evaluate(eval_ts, eval_ts, &mut self.row)?;
1362                    self.output.push_samples(state.cursor, &self.row)?;
1363                }
1364                Replay::Once => {
1365                    evaluator.evaluate(eval_ts, eval_ts, &mut self.row)?;
1366                    for index in 0..grid.len() {
1367                        self.output.push_samples(index, &self.row)?;
1368                    }
1369                }
1370                Replay::EveryPoint => {
1371                    for (index, &point) in grid.iter().enumerate() {
1372                        evaluator.evaluate(eval_ts, point, &mut self.row)?;
1373                        self.output.push_samples(index, &self.row)?;
1374                    }
1375                }
1376            }
1377            state.cursor += 1;
1378        }
1379        if let Some(kernel) = self.batch.as_mut() {
1380            flush_batch(
1381                kernel,
1382                &mut self.pending,
1383                &self.fields,
1384                self.window.range_ms,
1385                &mut self.row,
1386                &mut self.output,
1387                #[cfg(test)]
1388                &mut self.batch_calls,
1389            )?;
1390        }
1391        // A point is decided once it has been evaluated, once its window can no longer gain a
1392        // sample, or -- at the end of the series or of what the input has reached -- once no
1393        // retained sample reaches it. The last case is what keeps a long empty stretch cheap: the
1394        // loop stops at the first such point instead of walking the rest of the grid, and none of
1395        // the points behind it can gain a value, so the cursor skips them.
1396        if empty_tail && matches!(self.replay, Replay::PerPoint) {
1397            let mut decided = state.cursor;
1398            if let Some(ready) = ready {
1399                while decided < points_read && i128::from(grid[decided]) <= ready {
1400                    decided += 1;
1401                }
1402            } else {
1403                decided = points_read;
1404            }
1405            state.cursor = decided;
1406        }
1407        debug_assert!(
1408            self.pending.is_empty(),
1409            "`evaluate` flushes the batch before `advance` rewrites the retained arrays a \
1410             pending span indexes"
1411        );
1412        let mut drains_samples = false;
1413        for field in &mut self.fields {
1414            // After the last window, samples are still ingested so that the overflow and
1415            // histogram checks do not depend on how the series is chunked, but none are kept.
1416            // An anchored selector can skip past the one point it reads.
1417            if state.cursor >= points_read {
1418                field.head = field.timestamps.len();
1419                field.histogram_head = field.histograms.len();
1420            }
1421            drains_samples |= field.head > 0;
1422            field.advance();
1423            field.advance_histograms();
1424        }
1425        // A drain shifts retained-array indices. Batch flushes do not, so they leave cursors valid.
1426        if drains_samples {
1427            self.pending.reset_cursors();
1428        }
1429        Ok(())
1430    }
1431
1432    /// The number of leading grid points whose windows are read; an anchored selector reads
1433    /// only the first.
1434    fn points_read(&self) -> usize {
1435        match self.replay {
1436            Replay::PerPoint => self.grid.spec().len as usize,
1437            Replay::Once | Replay::EveryPoint => 1,
1438        }
1439    }
1440}

1442/// Computes the result of every field's window at one grid point.
1443struct RowEvaluator<'a> {
1444    fields: &'a [FieldSamples],
1445    kernel: &'a mut WindowKernel,
1446    function: Option<RangeFunction>,
1447    annotations: Option<&'a PromqlAnnotationCollector>,
1448    range_ms: i64,
1449    offset_ms: i64,
1450    modifier: Option<RangeModifier>,
1451}
1452
1453impl RowEvaluator<'_> {
1454    /// Fills `row` with the result over each field's window ending at `window_end`, evaluated
1455    /// at `eval_ts`.
1456    fn evaluate(
1457        &mut self,
1458        window_end: i64,
1459        eval_ts: i64,
1460        row: &mut Vec<Option<GridSample>>,
1461    ) -> Result<()> {
1462        row.clear();
1463        // A smoothed window also reads the samples after its end.
1464        let window_end = window_end + self.modifier.map_or(0, RangeModifier::after_ms);
1465        for field in self.fields {
1466            let timestamps = &field.timestamps[field.head..];
1467            let len = timestamps.partition_point(|&timestamp| timestamp <= window_end);
1468            let floats = (len > 0).then(|| FloatWindow {
1469                timestamps: &timestamps[..len],
1470                values: &field.values[field.head..field.head + len],
1471                counter_resets: field.resets_in(len),
1472                eval_ts,
1473                range_ms: self.range_ms,
1474            });
1475            let histograms = &field.histograms[field.histogram_head..];
1476            let histograms = &histograms
1477                [..histograms.partition_point(|(timestamp, _)| *timestamp <= window_end)];
1478            row.push(match self.function {
1479                Some(function) if !histograms.is_empty() => histogram::evaluate(
1480                    function,
1481                    floats,
1482                    histograms,
1483                    eval_ts,
1484                    self.range_ms,
1485                    self.offset_ms,
1486                    &mut *self.kernel,
1487                    self.annotations,
1488                )?,
1489                _ => floats.and_then(|window| (self.kernel)(window).map(GridSample::Float)),
1490            });
1491        }
1492        Ok(())
1493    }
1494}

1496/// Rejects a modifier on a function Prometheus does not allow it with, as Prometheus rejects the
1497/// query (`AnchoredSafeFunctions`, `SmoothedSafeFunctions`), and on evaluated input, which reads
1498/// no raw samples around its windows.
1499///
1500/// `pub(crate)` because it is a rule about the input's schema and the window alone, so the native
1501/// route's admission predicate (`native_tree_supported`) asks it rather than a second copy of it,
1502/// the way it asks [`fold_offset`].
1503pub(crate) fn check_modifier(
1504    input: &SeriesSchema,
1505    modifier: RangeModifier,
1506    function: RangeFunction,
1507) -> Result<()> {
1508    if input.eval_grid.is_some() {
1509        return Err(DataFusionError::NotImplemented(format!(
1510            "the {} range modifier over evaluated series",
1511            modifier.name()
1512        )));
1513    }
1514    let allowed = match modifier {
1515        RangeModifier::Anchored { .. } => matches!(
1516            function,
1517            RangeFunction::Rate
1518                | RangeFunction::Increase
1519                | RangeFunction::Delta
1520                | RangeFunction::Resets
1521                | RangeFunction::Changes
1522        ),
1523        RangeModifier::Smoothed { .. } => matches!(
1524            function,
1525            RangeFunction::Rate | RangeFunction::Increase | RangeFunction::Delta
1526        ),
1527    };
1528    if !allowed {
1529        let list = match modifier {
1530            RangeModifier::Anchored { .. } => "changes, delta, increase, rate, resets",
1531            RangeModifier::Smoothed { .. } => "delta, increase, rate",
1532        };
1533        return Err(DataFusionError::Plan(format!(
1534            "{} modifier can only be used with: {list} - not with {function:?}",
1535            modifier.name()
1536        )));
1537    }
1538    if modifier.before_ms() < 0 {
1539        return Err(DataFusionError::Plan(format!(
1540            "lookback must not be negative, got {} ms",
1541            modifier.before_ms()
1542        )));
1543    }
1544    Ok(())
1545}

1547/// The offset samples are shifted by: the selector offset, or for `@` the offset that places
1548/// the anchored window at the first grid point.
1549///
1550/// `pub(crate)` because whether it fits is decided from the grid and the window alone, so the
1551/// native route's admission predicate asks the same arithmetic rather than a second copy of it.
1552pub(crate) fn fold_offset(grid: GridSpec, window: RangeWindow) -> Result<i64> {
1553    let Some(anchor) = window.anchor_ms else {
1554        return Ok(window.offset_ms);
1555    };
1556    anchor
1557        .checked_sub(window.offset_ms)
1558        .and_then(|end| grid.start_ms.checked_sub(end))
1559        .ok_or_else(|| {
1560            DataFusionError::Plan(format!(
1561                "@ {anchor} ms with offset {} ms is out of range",
1562                window.offset_ms
1563            ))
1564        })
1565}

1567/// Checks that `predict_linear` can read the instant of every point of `grid` once shifted by
1568/// `fold_offset`.
1569///
1570/// Unlike every other range function, which reads the window a grid point opens alone,
1571/// `predict_linear` reads the instant it reports as well, and reports it shifted by the fold
1572/// offset. The first and the last point are the ones checked: the shift is the same at every point
1573/// and the grid's points are ordered, so what fits at both ends fits between them.
1574///
1575/// The refusal is arithmetic rather than shape, but it is decided by the grid and the window alone,
1576/// which is why the native route's admission predicate reaches it too
1577/// (`native_tree_supported`): a query refused only here has to keep the plan it had instead of
1578/// failing while its runtime is built.
1579pub(crate) fn predict_linear_instants(grid: GridSpec, fold_offset: i64) -> Result<()> {
1580    let last = i128::from(grid.start_ms) + i128::from(grid.step_ms) * (i128::from(grid.len) - 1);
1581    for eval_ts in [i128::from(grid.start_ms), last] {
1582        if i64::try_from(eval_ts + i128::from(fold_offset)).is_err() {
1583            return Err(DataFusionError::Plan(format!(
1584                "predict_linear evaluation time {eval_ts} overflows with offset {fold_offset} ms"
1585            )));
1586        }
1587    }
1588    Ok(())
1589}

1591/// The inner grid of a subquery read by `window` from the outer grid `outer`: the multiples of
1592/// `step_ms` after `start - F - range` up to the last window end, where `F` is the fold offset
1593/// of `window`; with `@`, the only window end is `start - F`. `None` when no multiple falls in
1594/// the windows.
1595pub fn subquery_grid(
1596    outer: GridSpec,
1597    window: RangeWindow,
1598    step_ms: i64,
1599) -> Result<Option<GridSpec>> {
1600    if step_ms <= 0 {
1601        return Err(DataFusionError::Plan(format!(
1602            "subquery step must be positive, got {step_ms} ms"
1603        )));
1604    }
1605    let fold_offset = i128::from(fold_offset(outer, window)?);
1606    let step = i128::from(step_ms);
1607    let start = i128::from(outer.start_ms);
1608    let last_eval = match window.anchor_ms {
1609        Some(_) => start,
1610        None => start + i128::from(outer.step_ms) * (i128::from(outer.len) - 1),
1611    };
1612    let first = ((start - fold_offset - i128::from(window.range_ms)).div_euclid(step) + 1) * step;
1613    let last = last_eval - fold_offset;
1614    if first > last {
1615        return Ok(None);
1616    }
1617    let len = u64::try_from((last - first) / step + 1)
1618        .map_err(|_| DataFusionError::Plan("subquery grid has too many points".into()))?;
1619    let first = i64::try_from(first).map_err(|_| {
1620        DataFusionError::Plan(format!("subquery grid start {first} ms is out of range"))
1621    })?;
1622    GridSpec::try_new(first, step_ms, len)
1623        .map(Some)
1624        .map_err(|error| DataFusionError::External(Box::new(error)))
1625}
