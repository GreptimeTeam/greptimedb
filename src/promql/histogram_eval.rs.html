55/// still groups by metadata before removing it, so distinct histograms are not mixed together.

59/// `histogram_quantile` or `histogram_fraction` with its constant arguments.
60#[derive(Debug, Clone, Copy, serde::Serialize, serde::Deserialize)]
61pub enum HistogramFunction {
62    /// `histogram_quantile(q, v)`.
63    Quantile(f64),
64    /// `histogram_fraction(lower, upper, v)`.
65    Fraction { lower: f64, upper: f64 },
66}
67
68/// `histogram_quantiles(v, label, q...)`: [`evaluate_histograms`] with `Quantile(q)` for each
69/// `q`, each result labelled with `label` set to `q` as Prometheus' `FormatOpenMetricsFloat`
70/// writes it. Results that then share a label set merge as [`merge_same_labelset`] does, so a
71/// repeated `q`, or a `label` that overwrites the label telling two histograms apart, fails at a
72/// point both hold.
73///
74/// Each `q` is a constant. Prometheus evaluates the quantile arguments at every step, so a
75/// quantile that varies over time is not this kernel's to evaluate.
76pub fn histogram_quantiles(
77    label: &str,
78    quantiles: &[f64],
79    schema: &Arc<SeriesSchema>,
80    series: &[Series],
81) -> Result<Vec<Series>> {
82    let mut output = Vec::new();
83    for &quantile in quantiles {
84        let value: Arc<str> = Arc::from(format_open_metrics_float(quantile));
85        for mut result in
86            evaluate_histograms(HistogramFunction::Quantile(quantile), schema, series)?
87        {
88            result.metric.set(label, value.clone()).map_err(external)?;
89            output.push(result);
90        }
91    }
92    merge_same_labelset(schema, output)
93}

95/// Evaluates `function` over the classic and native histograms in `series`.
96///
97/// Float series with the same labels other than `le` form a classic histogram; a float series
98/// whose `le` does not parse is ignored. Native histogram samples are evaluated one by one. At a
99/// point where a native histogram and classic buckets with the same labels (`le` aside) are both
100/// present, neither is evaluated. Each field is evaluated on its own. Classic results drop
101/// `le`; all results drop `__name__`, `__type__` and `__unit__`. Series that then share a label
102/// set are merged as [`merge_same_labelset`] does.
103pub fn evaluate_histograms(
104    function: HistogramFunction,
105    schema: &Arc<SeriesSchema>,
106    series: &[Series],
107) -> Result<Vec<Series>> {
108    evaluate_histograms_with_annotations(function, schema, series, None, None)
109}

111/// Per-grid-point arguments to a classic or native histogram function.
112#[derive(Debug, Clone, Copy)]
113pub struct HistogramParameters<'a> {
114    pub quantile: Option<&'a [Option<f64>]>,
115    pub lower: Option<&'a [Option<f64>]>,
116    pub upper: Option<&'a [Option<f64>]>,
117}

119/// Evaluates `function` with optional per-point scalar arguments, reporting what it drops to
120/// `collector`. A missing parameter value suppresses the output at that point, matching aggregate
121/// quantile's missing-q behavior.
122pub(crate) fn evaluate_histograms_with_annotations(
123    function: HistogramFunction,
124    schema: &Arc<SeriesSchema>,
125    series: &[Series],
126    parameters: Option<HistogramParameters<'_>>,
127    collector: Option<&PromqlAnnotationCollector>,
128) -> Result<Vec<Series>> {
129    // The groups in the order their first member arrived, so the result is in one order on every
130    // run, as `merge_same_labelset` keeps it.
131    let mut index: HashMap<LabelSet, usize> = HashMap::new();
132    let mut groups: Vec<Group<'_>> = Vec::new();
133    let mut group_of = |metric: SeriesLabels| {
134        *index.entry(label_set(&metric)).or_insert_with(|| {
135            groups.push(Group::new(metric));
136            groups.len() - 1
137        })
138    };
139    let mut members: Vec<(usize, Option<f64>, &Series)> = Vec::new();
140    for series in series {
141        grid_axis(series)?;
142        if let Some(upper_bound) = series
143            .metric
144            .get(BUCKET_LABEL)
145            .and_then(classic::parse_bound)
146        {
147            let mut metric = series.metric.clone();
148            metric.remove(BUCKET_LABEL);
149            members.push((group_of(metric), Some(upper_bound), series));
150        }
151        if has_histograms(series) {
152            members.push((group_of(series.metric.clone()), None, series));
153        }
154    }
155    for (slot, upper_bound, series) in members {
156        let group = &mut groups[slot];
157        match upper_bound {
158            Some(upper_bound) => group.classic.push((upper_bound, series)),
159            None => {
160                // A label set carries at most one native histogram series: a second one would
161                // overwrite the first and drop it from the result.
162                debug_assert!(
163                    group.native.is_none(),
164                    "one label set carries at most one native histogram series; a second one for \
165                     the same label set means the input handed one label set over as several \
166                     series, which a gap in the axis is there to avoid"
167                );
168                group.native = Some(series);
169            }
170        }
171    }
172
173    let mut output = Vec::with_capacity(groups.len());
174    let mut builder = GridSeriesBuilder::try_new(schema.clone())?;
175    let mut classic_buckets = Vec::new();
176    let mut native_buckets = Vec::new();
177    let mut values = vec![None; schema.fields.len()];
178    for mut group in groups {
179        // One builder serves every group, so each result series starts explicitly.
180        builder.begin_series();
181        // The positions any member carries, merged once: the holes no member holds are never
182        // visited, so a group of sparse series costs its rows and not its envelope.
183        let axes = group
184            .classic
185            .iter()
186            .map(|member| &member.1.time)
187            .chain(group.native.map(|native| &native.time));
188        // One forward cursor per member and field. The merged positions only increase, so each
189        // member's rows do too: reading them costs one pass over the field, not a chunk scan per
190        // row. The classic members come first, then the native one.
191        let members = group
192            .classic
193            .iter()
194            .map(|&(_, series)| series)
195            .chain(group.native);
196        let mut cursors: Vec<Vec<RowCursor<'_>>> = members
197            .map(|series| series.fields.iter().map(|field| field.cursor()).collect())
198            .collect();
199        let classic_members = group.classic.len();
200        for point in union_positions(axes) {
201            for (field, value) in values.iter_mut().enumerate() {
202                classic_buckets.clear();
203                for (member, &(upper_bound, series)) in group.classic.iter().enumerate() {
204                    let count = series.time.row_of(point).and_then(|row| {
205                        match cursors[member][field]
206                            .at(row)
207                            .expect("a row never goes back")
208                        {
209                            SampleRef::Float(count) => Some(count),
210                            _ => None,
211                        }
212                    });
213                    if let Some(count) = count {
214                        classic_buckets.push(classic::Bucket { upper_bound, count });
215                    }
216                }
217                // Go sorts the buckets present at each point by `<`: +0 and -0 tie and NaN
218                // ties with everything, so which of them coalescing keeps and where a NaN
219                // bound lands follow its sort algorithm.
220                go_sort::sort_by(&mut classic_buckets, |a, b| a.upper_bound < b.upper_bound);
221                let native = group.native.and_then(|series| {
222                    let row = series.time.row_of(point)?;
223                    match cursors[classic_members][field]
224                        .at(row)
225                        .expect("a row never goes back")
226                    {
227                        SampleRef::Histogram(h) => Some(h),
228                        _ => None,
229                    }
230                });
231                let argument = |values: Option<&[Option<f64>]>, bound: f64| -> Option<f64> {
232                    match values {
233                        Some(values) => values.get(point).copied().flatten(),
234                        None => Some(bound),
235                    }
236                };
237                let function = match function {
238                    HistogramFunction::Quantile(q) => {
239                        let Some(q) = argument(parameters.and_then(|p| p.quantile), q) else {
240                            *value = None;
241                            continue;
242                        };
243                        HistogramFunction::Quantile(q)
244                    }
245                    HistogramFunction::Fraction { lower, upper } => {
246                        let (Some(lower), Some(upper)) = (
247                            argument(parameters.and_then(|p| p.lower), lower),
248                            argument(parameters.and_then(|p| p.upper), upper),
249                        ) else {
250                            *value = None;
251                            continue;
252                        };
253                        HistogramFunction::Fraction { lower, upper }
254                    }
255                };
256                // A native histogram and classic buckets of one label set conflict.
257                *value = match (native, classic_buckets.is_empty()) {
258                    (Some(_), false) => {
259                        if let Some(collector) = collector {
260                            collector.record_warning(format!("{}: native histogram and classic buckets both present; dropping both", match function {
261                                HistogramFunction::Quantile(_) => "histogram_quantile",
262                                HistogramFunction::Fraction { .. } => "histogram_fraction",
263                            }));
264                        }
265                        None
266                    }
267                    (None, true) => None,
268                    (Some(h), true) => Some(match function {
269                        HistogramFunction::Quantile(q) => {
270                            native::quantile(q, &h, &mut native_buckets)?
271                        }
272                        HistogramFunction::Fraction { lower, upper } => {
273                            native::fraction(lower, upper, &h, &mut native_buckets)?
274                        }
275                    }),
276                    (None, false) => Some(match function {
277                        HistogramFunction::Quantile(q) => {
278                            classic::bucket_quantile(q, &mut classic_buckets)
279                        }
280                        HistogramFunction::Fraction { lower, upper } => {
281                            classic::bucket_fraction(lower, upper, &mut classic_buckets)
282                        }
283                    }),
284                };
285            }
286            builder.push(point, &values)?;
287        }
288        group.metric.remove_all(&METADATA_LABELS);
289        output.extend(builder.finish(group.metric)?);
290    }
291    merge_same_labelset(schema, output)
292}

294/// The series of one label set (`le` aside): classic buckets, and the series of exactly that
295/// label set when it holds native histograms.
296struct Group<'a> {
297    metric: SeriesLabels,
298    classic: Vec<(f64, &'a Series)>,
299    native: Option<&'a Series>,
300}
301
302impl Group<'_> {
303    fn new(metric: SeriesLabels) -> Self {
304        Self {
305            metric,
306            classic: Vec::new(),
307            native: None,
308        }
309    }
310}

312/// Checks that `series` is addressed on a grid, which is what a position lookup needs.
313fn grid_axis(series: &Series) -> Result<&TimeAxis> {
314    if series.time.grid().is_none() {
315        return Err(DataFusionError::Execution(
316            "histogram functions require evaluated series".into(),
317        ));
318    }
319    Ok(&series.time)
320}

322fn has_histograms(series: &Series) -> bool {
323    series.fields.iter().any(|samples| {
324        samples
325            .segments()
326            .iter()
327            .any(|segment| !matches!(segment.dense(), Some(DenseSamplesRef::Float(_))))
328    })
329}

337/// A function of one native histogram.
338#[derive(Debug, Clone, Copy, PartialEq, Eq, serde::Serialize, serde::Deserialize, Hash)]
339pub enum NativeHistogramFunction {
340    Count,
341    Sum,
342    Avg,
343    StdDev,
344    StdVar,
345}
346
347/// Applies `function` to the native histogram samples of `series`; float samples are dropped and
348/// so is `__name__`. `None` when the series has no histogram sample.
349///
350/// Histogram samples may leave holes, but the holes only skip positions: the result is one series
351/// over the runs of consecutive positions that hold a value.
352///
353/// Series that share a label set once their names are dropped must then go through
354/// [`merge_same_labelset`].
355pub fn map_native_histogram(
356    function: NativeHistogramFunction,
357    schema: &Arc<SeriesSchema>,
358    series: &Series,
359) -> Result<Option<Series>> {
360    validate_grid_series(schema, series)?;
361    let mut builder = GridSeriesBuilder::try_new(schema.clone())?;
362    let mut buckets = Vec::new();
363    let mut values = vec![None; schema.fields.len()];
364    // One cursor per field, driven by the positions the axis carries: a series of several runs
365    // costs its rows, and the positions it skips are never visited.
366    let mut cursors = series
367        .fields
368        .iter()
369        .map(|field| field.cursor())
370        .collect::<Vec<_>>();
371    for (point, row) in series.time.positions() {
372        for (field, value) in values.iter_mut().enumerate() {
373            *value = match cursors[field]
374                .at(row)
375                .expect("the axis and the fields are aligned")
376            {
377                SampleRef::Histogram(h) => Some(function.apply(&h, &mut buckets)?),
378                _ => None,
379            };
380        }
381        builder.push(point, &values)?;
382    }
383    let mut metric = series.metric.clone();
384    // Prometheus `simpleHistogramFunc` drops only `__name__`, unlike `histogram_quantile`.
385    metric.remove("__name__");
386    builder.finish(metric)
387}
388
389impl NativeHistogramFunction {
390    /// The PromQL name of the function.
391    pub fn name(self) -> &'static str {
392        match self {
393            Self::Count => "histogram_count",
394            Self::Sum => "histogram_sum",
395            Self::Avg => "histogram_avg",
396            Self::StdDev => "histogram_stddev",
397            Self::StdVar => "histogram_stdvar",
398        }
399    }
400
401    fn apply(self, h: &HistogramView<'_>, buckets: &mut Vec<native::NativeBucket>) -> Result<f64> {
402        Ok(match self {
403            Self::Count => h.count,
404            Self::Sum => h.sum,
405            Self::Avg => h.sum / h.count,
406            Self::StdDev => native::variance(h, buckets)?.sqrt(),
407            Self::StdVar => native::variance(h, buckets)?,
408        })
409    }
410}
411
412impl PartialEq for HistogramFunction {
413    fn eq(&self, o: &Self) -> bool {
414        match (self, o) {
415            (Self::Quantile(a), Self::Quantile(b)) => a.to_bits() == b.to_bits(),
416            (Self::Fraction { lower: a, upper: b }, Self::Fraction { lower: c, upper: d }) => {
417                a.to_bits() == c.to_bits() && b.to_bits() == d.to_bits()
418            }
419            _ => false,
420        }
421    }
422}
423impl Eq for HistogramFunction {}
424impl std::hash::Hash for HistogramFunction {
425    fn hash<H: std::hash::Hasher>(&self, h: &mut H) {
426        match self {
427            Self::Quantile(x) => {
428                0u8.hash(h);
429                x.to_bits().hash(h)
430            }
431            Self::Fraction { lower, upper } => {
432                1u8.hash(h);
433                lower.to_bits().hash(h);
434                upper.to_bits().hash(h)
435            }
436        }
437    }
