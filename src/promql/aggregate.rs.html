68/// Aggregates each field at each absolute evaluation instant.
69///
70/// Each field is aggregated independently; an output row exists where any field has a value,
71/// and a field without input values at that row is null. Ordinary NaNs count as values.
72///
73/// Inputs are folded in slice order. Prometheus folds them in storage postings order, and the
74/// compensated `sum`/`avg` depend on that order in the last bits, so callers must keep it.
75/// Groups are returned in the order their first input appears, as in Prometheus: a nested
76/// aggregation folds them in that order.
77///
78/// Evaluated inputs on the shared schema/grid are admitted. An operation that reads a value takes
79/// float samples only; `count` and `group` ask whether a sample is present, so they also accept a
80/// histogram. `by` on a value field needs per-point labels and an output schema adapter, and is
81/// rejected. There is no fallback.
82pub fn aggregate_float(
83    schema: &Arc<SeriesSchema>,
84    input: &[Series],
85    op: AggregateOp,
86    grouping: Grouping<'_>,
87) -> Result<Vec<Series>> {
88    let mut aggregator = FloatAggregator::try_new(schema, op, grouping)?;
89    for series in input {
90        aggregator.update(series)?;
91    }
92    aggregator.finish()
93}

111/// What an aggregate does with a native histogram sample.
112#[derive(Clone, Copy, Debug, PartialEq, Eq)]
113pub(crate) enum HistogramSamples {
114    /// Counts it as one present sample: `count` counts every present sample, and `group` answers
115    /// `1` for a non-empty group, whatever the sample's type is.
116    Presence,
117    /// Ignores it and reports that it did: `min`, `max`, `stddev`, `stdvar` and `quantile` read a
118    /// float value, and Prometheus skips the histogram samples instead of refusing the statement.
119    Ignored,
120    /// Folds the payload and answers a histogram: `sum` adds native histograms bucket by bucket
121    /// and `avg` adds them and divides, as Prometheus does, so one group's result is a native
122    /// histogram.
123    Folded,
124}
125
126/// What `op` does with a native histogram sample.
127pub(crate) fn histogram_samples(op: AggregateOp) -> HistogramSamples {
128    match op {
129        AggregateOp::Count | AggregateOp::Group => HistogramSamples::Presence,
130        AggregateOp::Min | AggregateOp::Max | AggregateOp::Stddev | AggregateOp::Stdvar => {
131            HistogramSamples::Ignored
132        }
133        AggregateOp::Sum | AggregateOp::Avg => HistogramSamples::Folded,
134    }
135}

137/// Checks a series for the aggregate `op` folds, and returns its evaluated axis.
138///
139/// Every operation this route folds takes a histogram series: it counts the sample, skips it, or
140/// folds the payload. What a series must satisfy is therefore its axis, not its sample type -- the
141/// fold itself is where a payload an operation cannot use is refused or dropped.
142pub(crate) fn validate_aggregate_series<'a>(
143    schema: &SeriesSchema,
144    series: &'a Series,
145    op: AggregateOp,
146) -> Result<&'a TimeAxis> {
147    match histogram_samples(op) {
148        HistogramSamples::Presence | HistogramSamples::Ignored | HistogramSamples::Folded => {
149            validate_grid_series(schema, series)
150        }
151    }
152}

163    pub(crate) fn try_new_budgeted(
164        schema: &Arc<SeriesSchema>,
165        op: AggregateOp,
166        grouping: Grouping<'_>,
167        mut budget: impl FnMut(usize) -> Result<()>,
168    ) -> Result<Self> {
169        validate_input(schema, &[], grouping)?;
170        let (by, labels) = match grouping {
171            Grouping::By(labels) => (true, labels),
172            Grouping::Without(labels) => (false, labels),
173        };
174        let label_storage =
175            grouping_bytes(grouping).saturating_add(labels_capacity_bytes(grouping));
176        budget(label_storage)?;
177        let mut owned_labels: Vec<Arc<str>> = Vec::new();
178        owned_labels
179            .try_reserve_exact(labels.len())
180            .map_err(|error| {
181                DataFusionError::Execution(format!(
182                    "cannot reserve aggregate grouping labels: {error}"
183                ))
184            })?;
185        owned_labels.extend(labels.iter().map(|label| Arc::from(*label)));
186        let retained = owned_labels
187            .capacity()
188            .saturating_mul(size_of::<Arc<str>>())
189            .saturating_add(owned_labels.iter().fold(0usize, |bytes, label| {
190                bytes.saturating_add(label.len() + 2 * size_of::<usize>())
191            }));
192        Ok(Self {
193            schema: schema.clone(),
194            op,
195            by,
196            labels: owned_labels,
197            group_index: HashMap::new(),
198            groups: Vec::new(),
199            retained,
200        })
201    }

203    /// Folds `series` into its group. A series that fails validation leaves the state unchanged.
204    ///
205    /// A budget refusal is not a validation failure: the fold may have opened a group or filled
206    /// blocks before the allocation it could not admit, so a failed [`Self::update_budgeted`]
207    /// leaves the state half folded and its retained figure stale. The runtime stops the execution
208    /// and drops the aggregator on that error, which is why no rollback is kept; a caller holding
209    /// one must drop it rather than fold into it again.
210    pub fn update(&mut self, series: &Series) -> Result<()> {
211        self.update_budgeted(series, |_| Ok(()))
212    }

214    pub(crate) fn update_budgeted(
215        &mut self,
216        series: &Series,
217        mut budget: impl FnMut(usize) -> Result<()>,
218    ) -> Result<()> {
219        validate_aggregate_series(&self.schema, series, self.op)?;
220        let existing_bytes = self.state_bytes();
221        let grouping_workspace = grouping_workspace_bytes(&series.metric);
222        let mut workspace_bytes =
223            grouping_workspace.saturating_add(pending_bytes(self.schema.fields.len()));
224        budget(existing_bytes.saturating_add(workspace_bytes))?;
225        workspace_bytes = workspace_bytes.saturating_add(
226            self.labels
227                .len()
228                .saturating_mul(std::mem::size_of::<&str>()),
229        );
230        budget(existing_bytes.saturating_add(workspace_bytes))?;
231        let labels: Vec<&str> = self.labels.iter().map(|label| &**label).collect();
232        let grouping = if self.by {
233            Grouping::By(&labels)
234        } else {
235            Grouping::Without(&labels)
236        };
237        let metric = grouping.apply(&series.metric)?;
238        let key_bytes = metric
239            .tags()
240            .len()
241            .saturating_add(1)
242            .saturating_mul(std::mem::size_of::<(Arc<str>, Arc<str>)>());
243        workspace_bytes = workspace_bytes
244            .saturating_add(grouping_workspace_bytes(&metric))
245            .saturating_add(key_bytes);
246        budget(existing_bytes.saturating_add(workspace_bytes))?;
247        let key = label_key(&metric);
248        let key_retained = key
249            .capacity()
250            .saturating_mul(size_of::<(Arc<str>, Arc<str>)>());
251        let new_group = !self.group_index.contains_key(&key);
252        let group_capacity = self.groups.capacity();
253        let map_capacity = self.group_index.capacity();
254        let old_map_backing = hash_map_backing_bytes(map_capacity);
255        if new_group {
256            let next_group_capacity = if self.groups.len() == group_capacity {
257                group_capacity.saturating_mul(2).max(4)
258            } else {
259                group_capacity
260            };
261            let group_realloc_bytes = if next_group_capacity > group_capacity {
262                next_group_capacity.saturating_mul(size_of::<Group>())
263            } else {
264                0
265            };
266            budget(
267                existing_bytes
268                    .saturating_add(metric_bytes(&metric))
269                    .saturating_add(key_bytes)
270                    // A realloc that cannot grow in place keeps both buffers alive while it
271                    // copies, so a growing step admits the old buffer beside the new one --
272                    // the same reason the map below admits both tables. The old buffer alone
273                    // is already in `existing_bytes`; what is added here is the new one.
274                    .saturating_add(group_realloc_bytes)
275                    .saturating_add(if self.group_index.len() == map_capacity {
276                        hash_map_backing_bytes(next_hash_capacity(map_capacity))
277                    } else {
278                        0
279                    })
280                    .saturating_add(workspace_bytes),
281            )?;
282        }
283        let group_capacity_growth = if new_group && self.groups.len() == group_capacity {
284            group_capacity.saturating_mul(2).max(4)
285        } else {
286            group_capacity
287        };
288        if new_group && group_capacity_growth > group_capacity {
289            self.groups.reserve(1);
290        }
291        if new_group && self.group_index.len() == self.group_index.capacity() {
292            self.group_index.reserve(1);
293        }
294        let index = match self.group_index.entry(key) {
295            Entry::Occupied(entry) => *entry.get(),
296            Entry::Vacant(entry) => {
297                self.groups.push(Group {
298                    metric: metric.clone(),
299                    blocks: BTreeMap::new(),
300                    histograms: BTreeMap::new(),
301                });
302                let index = self.groups.len() - 1;
303                entry.insert(index);
304                index
305            }
306        };
307        if new_group {
308            self.retained = self
309                .retained
310                .saturating_add(metric_bytes(&metric))
311                .saturating_add(key_retained)
312                .saturating_add(
313                    self.groups
314                        .capacity()
315                        .saturating_sub(group_capacity)
316                        .saturating_mul(size_of::<Group>()),
317                )
318                .saturating_add(
319                    hash_map_backing_bytes(self.group_index.capacity())
320                        .saturating_sub(old_map_backing),
321                );
322        }
323        drop(labels);
324        drop(metric);
325        self.retained = self.groups[index].accumulate(
326            series,
327            self.op,
328            &mut budget,
329            self.retained,
330            self.schema.fields.len(),
331        )?;
332        Ok(())
333    }

397            .saturating_add(hash_map_backing_bytes(self.group_index.capacity()))
398            .saturating_add(self.group_index.keys().fold(0usize, |bytes, key| {
399                bytes.saturating_add(

403            }))
404            .saturating_add(
405                self.labels
406                    .iter()
407                    .map(|label| label.len() + 2 * std::mem::size_of::<usize>())
408                    .sum::<usize>(),
409            )
410            .saturating_add(
411                self.groups
412                    .capacity()
413                    .saturating_mul(std::mem::size_of::<Group>()),

431/// Turns the state of `groups` into result series, in group order, admitting as it builds.
432fn finish_groups_budgeted(
433    schema: Arc<SeriesSchema>,
434    op: AggregateOp,
435    groups: Vec<Group>,
436    state: usize,
437    mut budget: impl FnMut(usize) -> Result<()>,
438) -> Result<Vec<Series>> {
439    let width = schema.fields.len();
440    let group_count = groups.len();
441    if group_count == 0 {
442        return Ok(Vec::new());
443    }
444    let groups = groups.into_iter();
445    let schema_arc = Arc::clone(&schema);
446    let mut builder = GridSeriesBuilder::try_new(schema)?;
447    let output_capacity_bytes = group_count.saturating_mul(std::mem::size_of::<Series>());
448    let row_capacity_bytes = width.saturating_mul(std::mem::size_of::<Option<f64>>());
449    let base = state
450        .saturating_add(output_capacity_bytes)
451        .saturating_add(row_capacity_bytes);
452    budget(base)?;
453    let mut output = Vec::new();
454    output.try_reserve_exact(group_count).map_err(|error| {
455        DataFusionError::Execution(format!("cannot reserve aggregate outputs: {error}"))
456    })?;
457    let mut row = Vec::new();
458    row.try_reserve_exact(width).map_err(|error| {
459        DataFusionError::Execution(format!("cannot reserve aggregate row: {error}"))
460    })?;
461    let mut completed_output_bytes = 0usize;
462    for group in groups {
463        // A group that folded histogram samples answers histogram columns; one that folded floats
464        // keeps the float path below, so a group without a single histogram payload is built
465        // exactly as before.
466        if !group.histograms.is_empty() {
467            let finish_bytes = histogram_finish_bytes(&group, width);
468            budget(
469                state
470                    .saturating_add(output_capacity_bytes)
471                    .saturating_add(row_capacity_bytes)
472                    .saturating_add(completed_output_bytes)
473                    .saturating_add(finish_bytes),
474            )?;
475            if let Some(finished) = finish_histogram_group(op, &schema_arc, group, width)? {
476                completed_output_bytes = completed_output_bytes
477                    .saturating_add(crate::grid_series::series_bytes(&finished));
478                output.push(finished);
479            }
480            budget(
481                state
482                    .saturating_add(output_capacity_bytes)
483                    .saturating_add(row_capacity_bytes)
484                    .saturating_add(completed_output_bytes),
485            )?;
486            continue;
487        }
488        // One builder serves every group, so each result series starts explicitly.
489        builder.begin_series();
490        let (rows, runs) = group_rows_runs(&group, width);
491        let finish_bytes = finish_working_bytes(rows, runs, width);
492        budget(
493            state
494                .saturating_add(output_capacity_bytes)
495                .saturating_add(row_capacity_bytes)
496                .saturating_add(completed_output_bytes)
497                .saturating_add(finish_bytes),
498        )?;
499        builder.reserve_rows(rows)?;
500        for (&block, states) in &group.blocks {
501            // The last block can extend past the grid; only points with a sample are pushed.
502            for (offset, point) in states.chunks_exact(width).enumerate() {
503                if !point.iter().any(|state| state.seen) {
504                    continue;
505                }
506                row.clear();
507                row.extend(point.iter().map(|state| state.finish(op)));
508                builder.push(block * BLOCK_POINTS + offset, &row)?;
509            }
510        }
511        let finished = builder.finish(group.metric)?;
512        completed_output_bytes = completed_output_bytes.saturating_add(
513            finished
514                .iter()
515                .map(crate::grid_series::series_bytes)
516                .sum::<usize>(),
517        );
518        output.extend(finished);
519        budget(
520            state
521                .saturating_add(output_capacity_bytes)
522                .saturating_add(row_capacity_bytes)
523                .saturating_add(completed_output_bytes),
524        )?;
525    }
526    Ok(output)
527}

529/// Builds one group's histogram result, or `None` when it has nothing left to answer.
530///
531/// A position's answer is its live sum plus the compensation paired with it -- what the kernel's own
532/// `sum` does at the end of its fold -- with the start timestamp cleared and the empty buckets
533/// dropped. A poisoned position was removed, so it contributes no value; a position that is
534/// poisoned in one field and answered in another still carries the row.
535fn finish_histogram_group(
536    op: AggregateOp,
537    schema: &Arc<SeriesSchema>,
538    group: Group,
539    width: usize,
540) -> Result<Option<Series>> {
541    // A field's column is one type. The fold already refuses a histogram where it had folded a
542    // float at that position; a float folded after a histogram is caught here, where the column
543    // type is chosen for the whole group.
544    if group
545        .blocks
546        .values()
547        .any(|states| states.iter().any(|state| state.seen))
548    {
549        return Err(DataFusionError::Execution(
550            "a native aggregate cannot mix float and histogram samples in one field".into(),
551        ));
552    }
553    let grid = schema.eval_grid.clone().ok_or_else(|| {
554        DataFusionError::Execution("histogram aggregation requires an evaluation grid".into())
555    })?;
556    let mut positions: Vec<usize> = group
557        .histograms
558        .keys()
559        .map(|&(position, _)| position)
560        .collect();
561    positions.dedup();
562    let mut builders: Vec<HistogramBatchBuilder> =
563        (0..width).map(|_| HistogramBatchBuilder::new()).collect();
564    let mut indices: Vec<Vec<Option<u32>>> = vec![Vec::new(); width];
565    let mut spans: Vec<GridSpan> = Vec::new();
566    let mut row: Vec<Option<FloatHistogram>> = vec![None; width];
567    for position in positions {
568        // A poisoned position has no answer, so the row it would have made is dropped whole: every
569        // column is written once per *answered* position, or the columns and the axis would drift.
570        let mut answered = false;
571        for (field, value) in row.iter_mut().enumerate() {
572            *value = match group.histograms.get(&(position, field)) {
573                Some(accumulator) => {
574                    let value = finished_histogram(op, accumulator)?;
575                    answered |= value.is_some();
576                    value
577                }
578                None => None,
579            };
580        }
581        if !answered {
582            continue;
583        }
584        // A gap between two answered positions ends a run, and a row belongs to the run it opens.
585        match spans.last_mut() {
586            Some(span) if span.start() + span.len() == position => {
587                *span = GridSpan::new(span.start(), span.len() + 1);
588            }
589            _ => spans.push(GridSpan::new(position, 1)),
590        }
591        for ((builder, indices), value) in builders.iter_mut().zip(&mut indices).zip(&row) {
592            indices.push(match value {
593                Some(value) => Some(value.push_to(builder)?),
594                None => None,
595            });
596        }
597    }
598    if spans.is_empty() {
599        return Ok(None);
600    }
601    let mut samples = Vec::with_capacity(width);
602    for (builder, indices) in builders.into_iter().zip(indices) {
603        let batch = builder.finish().map_err(external)?;
604        samples.push(
605            Samples::try_from_histogram(UInt32Array::from(indices), batch).map_err(external)?,
606        );
607    }
608    // The result is addressed by the runs that carry a histogram, not by the span between the first
609    // and the last one: the nulls padding them stay inside a run, never between two.
610    let time = TimeAxis::from_grid_spans(grid, spans).map_err(external)?;
611    Series::try_new(
612        schema,
613        group.metric,
614        time,
615        SeriesFields::try_from_vec(samples).map_err(external)?,
616    )
617    .map(Some)
618    .map_err(external)
619}

621/// One position's answer, compacted, or `None` when the point was poisoned.
622///
623/// A sum answers its live value with the compensation folded in. An average in mean mode answers its
624/// mean with the compensation folded in; one that never switched divides the sum and the compensation
625/// by the sample count and then folds the compensation in, which is what `sum_or_avg` does at the end
626/// of both paths.
627fn finished_histogram(
628    op: AggregateOp,
629    accumulator: &HistogramAccumulator,
630) -> Result<Option<FloatHistogram>> {
631    if accumulator.evidence.poisoned {
632        return Ok(None);
633    }
634    let mut value = match (op, accumulator.mean.as_ref()) {
635        (AggregateOp::Avg, Some(mean)) => {
636            let mut mean = mean.clone();
637            if let Some(compensation) = accumulator.compensation.as_ref() {
638                mean.add(compensation)?;
639            }
640            mean
641        }
642        (AggregateOp::Avg, None) => {
643            let mut value = accumulator.value.clone();
644            value.div(accumulator.count);
645            if let Some(mut compensation) = accumulator.compensation.clone() {
646                compensation.div(accumulator.count);
647                value.add(&compensation)?;
648            }
649            value
650        }
651        _ => {
652            let mut value = accumulator.value.clone();
653            if let Some(compensation) = accumulator.compensation.as_ref() {
654                value.add(compensation)?;
655            }
656            value
657        }
658    };
659    // The answer's empty buckets are dropped the way the kernel's own aggregate drops them. It also
660    // carries no start timestamp of its own, which every derivation above clears along with
661    // `compact`.
662    value.compact();
663    Ok(Some(value))
664}

666/// Bytes [`finish_histogram_group`] works in while it builds one group's result.
667fn histogram_finish_bytes(group: &Group, width: usize) -> usize {
668    let payload = group
669        .histograms
670        .values()
671        .fold(0usize, |bytes, accumulator| {
672            bytes.saturating_add(accumulator_bytes(accumulator))
673        });
674    let rows = group.histograms.len();
675    let fields = width.saturating_mul(
676        size_of::<HistogramBatchBuilder>()
677            .saturating_add(size_of::<Vec<Option<u32>>>())
678            .saturating_add(size_of::<Samples>()),
679    );
680    // Twice the fold's payload: the answer is one derived histogram per position, and an average
681    // divides a copy of its live value before it adds the compensation in.
682    payload
683        .saturating_mul(2)
684        .saturating_add(rows.saturating_mul(size_of::<Option<u32>>()))
685        .saturating_add(rows.saturating_mul(size_of::<GridSpan>()))
686        .saturating_add(fields)
687        .saturating_add(size_of::<Series>().saturating_add(size_of::<TimeAxis>()))
688}

700fn group_rows_runs(group: &Group, width: usize) -> (usize, usize) {
701    if width == 0 {
702        return (0, 0);
703    }
704    let mut rows = 0usize;
705    let mut runs = 0usize;
706    let mut previous: Option<usize> = None;
707    for (&block, states) in &group.blocks {
708        for (offset, point) in states.chunks_exact(width).enumerate() {
709            if point.iter().any(|state| state.seen) {
710                let index = block.saturating_mul(BLOCK_POINTS).saturating_add(offset);
711                rows = rows.saturating_add(1);
712                if previous.is_none_or(|previous| previous.checked_add(1) != Some(index)) {
713                    runs = runs.saturating_add(1);
714                }
715                previous = Some(index);
716            }
717        }
718    }
719    (rows, runs)
720}

755/// `quantile(q, v)`, with `q` read at each grid point from an unlabelled single-field scalar on
756/// the same grid, as Prometheus evaluates the parameter at every step. A point where `q` has no
757/// value has no result; `q` outside `[0, 1]` or NaN gives what Prometheus' `quantile` gives.
758pub fn aggregate_quantile(
759    schema: &Arc<SeriesSchema>,
760    input: &[Series],
761    q: &Series,
762    grouping: Grouping<'_>,
763) -> Result<Vec<Series>> {
764    validate_input(schema, input, grouping)?;
765    let q_axis = validate_scalar(schema, q)?;
766    let q: Vec<(usize, f64)> = q_axis
767        .positions()
768        .zip(q.fields.as_slice()[0].iter())
769        .filter_map(|((point, _), sample)| match sample {
770            SampleRef::Float(value) => Some((point, value)),
771            _ => None,
772        })
773        .collect();
774    // Members keep input order, the order Prometheus appends to its per-step heap.
775    let mut group_index = HashMap::new();
776    let mut groups: Vec<(SeriesLabels, Vec<&Series>)> = Vec::new();
777    for series in input {
778        let metric = grouping.apply(&series.metric)?;
779        let index = *group_index.entry(label_key(&metric)).or_insert_with(|| {
780            groups.push((metric, Vec::new()));
781            groups.len() - 1
782        });
783        groups[index].1.push(series);
784    }
785    let width = schema.fields.len();
786    let mut builder = GridSeriesBuilder::try_new(schema.clone())?;
787    let mut output = Vec::with_capacity(groups.len());
788    let (mut values, mut sorted, mut row) = (Vec::new(), Vec::new(), Vec::with_capacity(width));
789    for (metric, members) in groups {
790        builder.begin_series();
791        // Each member's next position waits in a min-heap, so a point costs the members present
792        // there and not the whole group: a group of short-lived series stays linear in its
793        // samples. Ties pop in member order, the input order Prometheus appends values in.
794        let mut positions: Vec<_> = members
795            .iter()
796            .map(|series| series.time.positions())
797            .collect();
798        // Indexed by field, then member.
799        let mut cursors: Vec<Vec<_>> = (0..width)
800            .map(|field| {
801                members
802                    .iter()
803                    .map(|series| series.fields.as_slice()[field].cursor())
804                    .collect()
805            })
806            .collect();
807        let mut next: BinaryHeap<Reverse<(usize, usize, usize)>> = positions
808            .iter_mut()
809            .enumerate()
810            .filter_map(|(member, positions)| {
811                positions
812                    .next()
813                    .map(|(point, row)| Reverse((point, member, row)))
814            })
815            .collect();
816        let mut present = Vec::new();
817        while let Some(&Reverse((point, _, _))) = next.peek() {
818            present.clear();
819            while let Some(&Reverse((at, member, row))) = next.peek()
820                && at == point
821            {
822                next.pop();
823                present.push((member, row));
824                if let Some((point, row)) = positions[member].next() {
825                    next.push(Reverse((point, member, row)));
826                }
827            }
828            let q = q
829                .binary_search_by_key(&point, |&(point, _)| point)
830                .ok()
831                .map(|index| q[index].1);
832            row.clear();
833            for field in &mut cursors {
834                values.clear();
835                for &(member, member_row) in &present {
836                    if let Some(SampleRef::Float(value)) = field[member].at(member_row) {
837                        values.push(value);
838                    }
839                }
840                row.push(match q {
841                    Some(q) if !values.is_empty() => Some(quantile(q, &values, &mut sorted)),
842                    _ => None,
843                });
844            }
845            builder.push(point, &row)?;
846        }
847        output.extend(builder.finish(metric)?);
848    }
849    Ok(output)
850}

881/// `count_values(label, v)`: at each grid point, the number of samples in each group of series
882/// that share a value, with the value written to `label` in Prometheus' `FormatFloat(v, 'f', -1)`
883/// form, or for a native histogram as its `FloatHistogram.String`. With `by`, `label` is part of
884/// the grouping; with `without`, it is dropped if listed.
885pub fn count_values(
886    schema: &Arc<SeriesSchema>,
887    input: &[Series],
888    label: &str,
889    grouping: Grouping<'_>,
890) -> Result<Vec<Series>> {
891    validate_count_values(schema, label, grouping)?;
892    // Unlike the other aggregations, this one reads histogram samples too.
893    for series in input {
894        validate_grid_series(schema, series)?;
895    }
896    let by_labels: Vec<&str>;
897    let grouping = match grouping {
898        Grouping::By(labels) => {
899            by_labels = labels.iter().copied().chain([label]).collect();
900            Grouping::By(&by_labels)
901        }
902        without => without,
903    };
904    let width = schema.fields.len();
905    let mut group_index = HashMap::new();
906    // Counts are kept per point that has one, not in blocks: a gauge with distinct values makes
907    // about one group per sample.
908    let mut groups: Counts = Vec::new();
909    for series in input {
910        // Equal bits format to the same label value, so each distinct value of a series
911        // resolves its group once. A histogram is keyed by the text it is written as.
912        let mut group_of_value: HashMap<Value, usize> = HashMap::new();
913        for (field_index, field) in series.fields.iter().enumerate() {
914            for ((point, _), sample) in series.time.positions().zip(field.iter()) {
915                let key = match sample {
916                    SampleRef::Missing => continue,
917                    SampleRef::Float(value) => Value::Float(value.to_bits()),
918                    SampleRef::Histogram(h) => Value::Histogram(histogram_string(&h)?),
919                };
920                let group = match group_of_value.entry(key) {
921                    Entry::Occupied(entry) => *entry.get(),
922                    Entry::Vacant(entry) => {
923                        // Prometheus writes a float in Go's `'f'` form and a histogram as
924                        // `FloatHistogram.String` does.
925                        let text = match entry.key() {
926                            Value::Float(bits) => format_float_f(f64::from_bits(*bits)),
927                            Value::Histogram(text) => text.clone(),
928                        };
929                        let mut metric = series.metric.clone();
930                        metric.set(label, Arc::from(text)).map_err(external)?;
931                        let metric = grouping.apply(&metric)?;
932                        let index = *group_index.entry(label_key(&metric)).or_insert_with(|| {
933                            groups.push((metric, BTreeMap::new()));
934                            groups.len() - 1
935                        });
936                        *entry.insert(index)
937                    }
938                };
939                groups[group]
940                    .1
941                    .entry(point)
942                    .or_insert_with(|| vec![0.0; width])[field_index] += 1.0;
943            }
944        }
945    }
946    count_series(schema, groups)
947}

1070/// Conservative backing estimate for Rust 1.98.1's insert-only `HashMap` (hashbrown 0.17.1).
1071/// Bucket count follows its 7/8 load factor; entries, one control byte per bucket and the
1072/// 16-byte control-group tail are counted. This models requested table storage, not allocator slack.
1073/// An upper bound on the bytes the group index holds for `capacity` entries.
1074///
1075/// The table keeps its entries in a power-of-two number of buckets below `16/7` of the usable
1076/// capacity, with one control byte beside each bucket. Three buckets per entry therefore covers the
1077/// table whatever exact growth the map uses, which is what keeps this a bound rather than a model of
1078/// one implementation: a toolchain that changes how the map grows cannot make it under-count.
1079fn hash_map_backing_bytes(capacity: usize) -> usize {
1080    capacity
1081        .saturating_mul(3)
1082        .saturating_mul(size_of::<(LabelKey, usize)>() + 1)
1083}

1091///
1092/// A node keeps at most eleven key/value pairs and twelve child pointers, plus a header of a few
1093/// words; the leaf form is smaller than that. The header allowance is generous on purpose, so the
1094/// bound does not depend on how a toolchain lays the node out.

1096    11 * size_of::<(usize, Box<[FloatState]>)>() + 16 * size_of::<usize>()

1147/// Grid points per state block. A block is allocated only once a sample falls in it, so sparse
1148/// input never expands to the full grid and dense input costs one allocation per block.

1154    /// each of its points, point by point.
1155    blocks: BTreeMap<usize, Box<[FloatState]>>,
1156    /// The histogram states of this group, keyed by absolute grid position and field.
1157    ///
1158    /// Only an aggregate that folds a payload allocates one, and only where it saw a histogram
1159    /// sample, so a float fold never touches this map and its state stays the blocks alone.
1160    histograms: BTreeMap<(usize, usize), HistogramAccumulator>,
1161}

1163/// One position's histogram fold: the live payload, the compensation paired with it, the running
1164/// mean an average switched to, and the evidence the fold recorded beside them.
1165///
1166/// The payload here is the kernel's own owned histogram rather than the state frame's row
1167/// representation: a fold adds bucket by bucket, and rebuilding a frame row per sample would
1168/// re-encode the layout it just added into. The state export builds that row from this.
1169///
1170/// The fields are the `HistogramAvg` state layout of the design: `value` is the state's
1171/// `histogram_value`, `mean` is its `incremental_mean` mode, `count` is its `count`, and
1172/// `compensation` is its `histogram_compensation`. A `sum` fold leaves `mean` at `None` and only
1173/// counts for the average's sake.
1174struct HistogramAccumulator {
1175    /// The live value: the running sum, or -- once an average switched -- the sum the switch started
1176    /// from, which the mean is derived from.
1177    value: FloatHistogram,
1178    /// The compensation paired with `value`, carrying `value`'s bucket layout.
1179    ///
1180    /// It is `None` only while a [`FloatHistogram::kahan_add`] call owns it, and stays `None` once
1181    /// the position is poisoned: a poisoned point folds nothing more, so there is nothing left to
1182    /// compensate.
1183    compensation: Option<FloatHistogram>,
1184    /// The running mean an average switched to once its sum overflowed; `None` while it still sums.
1185    mean: Option<FloatHistogram>,
1186    /// Samples folded into this position, which is the divisor an average's mean uses.
1187    count: f64,
1188    evidence: HistogramEvidence,
1189}

1191/// The evidence bits a histogram fold records beside its payload.
1192///
1193/// These are the Boolean components of the histogram state layout: the fold records them per point
1194/// so a merge of two partitions can report what the inputs disagreed about. Only `poisoned` steers
1195/// the fold's own output today -- a poisoned point is removed -- and the rest are carried for that
1196/// merge, except `counter_reset_collision`, which is decided from them when the answer is finished.
1197#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
1198pub(crate) struct HistogramEvidence {
1199    /// The point cannot be answered because its samples disagreed on their bucket layout.
1200    pub(crate) poisoned: bool,
1201    /// The disagreement was exponential buckets meeting custom buckets.
1202    pub(crate) incompatible_layout: bool,
1203    /// Two custom-bucket samples were reconciled into the intersection of their bounds.
1204    pub(crate) custom_bounds_reconciled: bool,
1205    /// Some sample reported a counter reset.
1206    pub(crate) counter_reset_seen: bool,
1207    /// Some sample reported that no counter reset happened.
1208    pub(crate) not_counter_reset_seen: bool,
1209    /// One point saw both a reset hint and its negation, which cannot both hold of its samples.
1210    ///
1211    /// The two bits above are recorded per point and travel through the state, so a summary of them
1212    /// is a claim about the fold rather than about one point. This bit is the per-point decision:
1213    /// [`evidence_summary`] sets it for a point that holds both bits and is not poisoned, so two
1214    /// groups that each saw one of the hints do not contradict each other.

1218/// Which field of which layout a fold is writing, and how it folds it.
1219///
1220/// One value carried through the per-block work instead of three separate arguments: the three
1221/// never vary independently within one series, and the block loop is the only place that reads
1222/// them.
1223#[derive(Clone, Copy)]
1224struct FieldFold {
1225    width: usize,
1226    field_index: usize,
1227    op: AggregateOp,
1228}
1229
1230impl Group {
1231    /// Folds one series' float samples into this group's blocks, one field at a time.
1232    ///
1233    /// Within a field the samples arrive in position order, so the first sample of a block admits
1234    /// that block and allocates its states, and every following sample of the same block is written
1235    /// into the states just placed. Each block needs one acquisition per field run, and a field
1236    /// with no value in a block leaves that block unallocated.
1237    fn accumulate(
1238        &mut self,
1239        series: &Series,
1240        op: AggregateOp,
1241        budget: &mut impl FnMut(usize) -> Result<()>,
1242        mut retained: usize,
1243        width: usize,
1244    ) -> Result<usize> {
1245        // The grouping workspace the caller admitted for this series is gone once the fold starts,
1246        // so only the retained state stays charged while the samples are written.
1247        budget(retained)?;
1248        for (field_index, field) in series.fields.iter().enumerate() {
1249            let fold = FieldFold {
1250                width,
1251                field_index,
1252                op,
1253            };
1254            // A payload-folding aggregate keeps its own per-position histogram state, addressed by
1255            // the absolute grid position: histogram samples never enter the block buffer the float
1256            // samples share, and a field that carries them cannot also carry floats. A float field
1257            // of the same aggregate still folds through that buffer, so the branch is per field.
1258            let folds_histograms = histogram_samples(op) == HistogramSamples::Folded
1259                && field
1260                    .iter()
1261                    .any(|sample| matches!(sample, SampleRef::Histogram(_)));
1262            if folds_histograms {
1263                for ((index, _), sample) in series.time.positions().zip(field.iter()) {
1264                    match sample {
1265                        SampleRef::Missing => {}
1266                        SampleRef::Histogram(view) => {
1267                            retained = self.fold_histogram(fold, index, &view, budget, retained)?;
1268                        }
1269                        // The finish step refuses the other order (a float folded after a
1270                        // histogram); refusing this one here keeps the column single-typed.
1271                        SampleRef::Float(_) => {
1272                            return Err(DataFusionError::Execution(
1273                                "a native aggregate cannot mix float and histogram samples in one field"
1274                                    .into(),
1275                            ));
1276                        }
1277                    }
1278                }
1279                continue;
1280            }
1281            // The axis says which absolute positions the field holds and the field iterator
1282            // supplies them in order, so a sparse axis never turns into its envelope here. Samples
1283            // that hold nothing never reach a state, and `validate_aggregate_series` has already
1284            // decided how this aggregate treats histogram samples.
1285            let mut samples = series
1286                .time
1287                .positions()
1288                .zip(field.iter())
1289                .filter_map(|((index, _), sample)| match sample {
1290                    SampleRef::Float(sample) => Some((index, sample)),
1291                    SampleRef::Histogram(_) => match histogram_samples(op) {
1292                        // `count` and `group` ask whether a sample is present, not which payload
1293                        // it carries, so a histogram is one contribution. The value is a
1294                        // placeholder: `FloatState` counts the contributions for `count` and only
1295                        // records that it saw one for `group`.
1296                        HistogramSamples::Presence => Some((index, 1.0)),
1297                        // Prometheus skips a histogram sample where the aggregate reads a float
1298                        // value; the runtime reports that it did.
1299                        HistogramSamples::Ignored => None,
1300                        // A payload-folding aggregate never reaches this loop: its samples were
1301                        // folded in the histogram branch above, and a field carrying both kinds is
1302                        // refused there.
1303                        HistogramSamples::Folded => {
1304                            unreachable!("a payload-folding aggregate folds histograms above")
1305                        }
1306                    },
1307                    _ => None,
1308                })
1309                .peekable();
1310            while let Some((index, sample)) = samples.next() {
1311                let block = index / BLOCK_POINTS;
1312                let (states, admitted) = self.block_states(block, fold, budget, retained)?;
1313                retained = admitted;
1314                states[(index % BLOCK_POINTS) * fold.width + fold.field_index]
1315                    .update(fold.op, sample);
1316                while let Some(&(at, sample)) = samples.peek()
1317                    && at / BLOCK_POINTS == block
1318                {
1319                    samples.next();
1320                    states[(at % BLOCK_POINTS) * fold.width + fold.field_index]
1321                        .update(fold.op, sample);
1322                }
1323            }
1324        }
1325        Ok(retained)
1326    }
1327
1328    /// The states of `block` for this field, admitting and allocating the block if this field has
1329    /// not written there yet, along with the retained bytes after that admission.
1330    ///
1331    /// A new block is admitted before the tree holds it, so a refused allocation leaves no block
1332    /// behind and the retained figure never counts an allocation that was not made.
1333    fn block_states<'a>(
1334        &'a mut self,
1335        block: usize,
1336        fold: FieldFold,
1337        budget: &mut impl FnMut(usize) -> Result<()>,
1338        retained: usize,
1339    ) -> Result<(&'a mut [FloatState], usize)> {
1340        let new_block = !self.blocks.contains_key(&block);
1341        let (block_bytes, backing_delta) = if new_block {
1342            (
1343                BLOCK_POINTS
1344                    .saturating_mul(fold.width)
1345                    .saturating_mul(size_of::<FloatState>()),
1346                btree_backing_bytes(self.blocks.len() + 1)
1347                    .saturating_sub(btree_backing_bytes(self.blocks.len())),
1348            )
1349        } else {
1350            (0, 0)
1351        };
1352        let retained = retained
1353            .saturating_add(backing_delta)
1354            .saturating_add(block_bytes);
1355        if new_block {
1356            budget(retained)?;
1357        }
1358        let states = self.blocks.entry(block).or_insert_with(|| {
1359            vec![FloatState::default(); BLOCK_POINTS * fold.width].into_boxed_slice()
1360        });
1361        Ok((&mut states[..], retained))
1362    }
1363}

1365impl Group {
1366    /// Folds one histogram sample into its position's state, admitting what the fold allocates.
1367    ///
1368    /// The first sample of a position establishes the state: it is the live value, and its
1369    /// compensation is the layout-carrying zero `new_compensation` builds -- never an `empty_like`
1370    /// histogram, which has no bucket indices for the compensated add to index. Every later sample
1371    /// is one [`FloatHistogram::kahan_add`] into the live value (an average's mean once it switched),
1372    /// carrying the compensation this position already holds.
1373    ///
1374    /// Two samples whose bucket layouts cannot be added (exponential meeting custom buckets) poison
1375    /// the position instead of failing the statement, and a poisoned position folds nothing more:
1376    /// Prometheus drops the point, so the evidence is what survives. Numeric overflow is not a reason
1377    /// to poison a point -- a saturated sum answers `Inf`, as the kernel's own sum does, and an
1378    /// average switches to its mean instead.
1379    fn fold_histogram(
1380        &mut self,
1381        fold: FieldFold,
1382        position: usize,
1383        view: &HistogramView<'_>,
1384        budget: &mut impl FnMut(usize) -> Result<()>,
1385        retained: usize,
1386    ) -> Result<usize> {
1387        // A field's column is either float or histogram, so a position that already folded a float
1388        // cannot also fold a histogram. The other order is caught when the fold is finished, which
1389        // is where the column type is chosen.
1390        if let Some(states) = self.blocks.get(&(position / BLOCK_POINTS))
1391            && states[(position % BLOCK_POINTS) * fold.width + fold.field_index].seen
1392        {
1393            return Err(DataFusionError::Execution(
1394                "a native aggregate cannot mix float and histogram samples in one field".into(),
1395            ));
1396        }
1397        let sample = FloatHistogram::try_from_view(view)?;
1398        let key = (position, fold.field_index);
1399        let vacant = !self.histograms.contains_key(&key);
1400        let backing = if vacant {
1401            histogram_backing_bytes(self.histograms.len() + 1)
1402                .saturating_sub(histogram_backing_bytes(self.histograms.len()))
1403        } else {
1404            0
1405        };
1406        if vacant {
1407            // Admit the new position before allocating it: a live value, the compensation that
1408            // starts beside it, and the map node holding both.
1409            budget(
1410                retained
1411                    .saturating_add(size_of::<HistogramAccumulator>())
1412                    .saturating_add(sample.heap_bytes().saturating_mul(2))
1413                    .saturating_add(backing),
1414            )?;
1415        }
1416        let accumulator = self.histograms.entry(key).or_insert_with(|| {
1417            let compensation = sample.new_compensation();
1418            HistogramAccumulator {
1419                value: sample.clone(),
1420                compensation: Some(compensation),
1421                mean: None,
1422                count: 1.0,
1423                evidence: HistogramEvidence::default(),
1424            }
1425        });
1426        // The evidence follows the sample's own reset hint, whether or not the point still folds.
1427        match sample.reset_hint {
1428            CounterResetHint::CounterReset => accumulator.evidence.counter_reset_seen = true,
1429            CounterResetHint::NotCounterReset => accumulator.evidence.not_counter_reset_seen = true,
1430            _ => {}
1431        }
1432        if accumulator.evidence.poisoned {
1433            return Ok(retained);
1434        }
1435        if vacant {
1436            // The admitted estimate is an upper bound; what the fold keeps is the real figure.
1437            return Ok(retained
1438                .saturating_add(accumulator_bytes(accumulator))
1439                .saturating_add(backing));
1440        }
1441        // An add can grow both bucket lists, so the incoming sample's payload on top of what the
1442        // position already holds is what is admitted; the figure kept afterwards is the real one.
1443        let before = accumulator_bytes(accumulator);
1444        budget(
1445            retained
1446                .saturating_add(before)
1447                .saturating_add(sample.heap_bytes()),
1448        )?;
1449        if accumulator.value.uses_custom_buckets() != sample.uses_custom_buckets() {
1450            // Exponential buckets cannot be added to custom ones: the point is removed rather than
1451            // answered, and the evidence says why.
1452            accumulator.evidence.poisoned = true;
1453            accumulator.evidence.incompatible_layout = true;
1454            accumulator.compensation = None;
1455            return Ok(retained);
1456        }
1457        match fold.op {
1458            AggregateOp::Sum => accumulator.fold_sum(&sample)?,
1459            AggregateOp::Avg => accumulator.fold_avg(&sample)?,
1460            op => unreachable!("{op:?} does not fold a histogram payload"),
1461        }
1462        Ok(retained
1463            .saturating_sub(before)
1464            .saturating_add(accumulator_bytes(accumulator)))
1465    }
1466}

1468impl HistogramAccumulator {
1469    /// Adds one sample into the running sum, one compensated add into the live value.
1470    fn fold_sum(&mut self, sample: &FloatHistogram) -> Result<()> {
1471        let (compensation, outcome) = self.value.kahan_add(sample, self.compensation.take())?;
1472        self.compensation = Some(compensation);
1473        self.evidence.custom_bounds_reconciled |= outcome.custom_bounds_reconciled;
1474        Ok(())
1475    }
1476
1477    /// Averages one sample in, switching from the running sum to a running mean when the sum would
1478    /// overflow.
1479    ///
1480    /// Each step is Prometheus v3.15's `aggregation`/`sum_or_avg` average in the same order: the
1481    /// trial add is what decides the mode, and once it overflows the trial is discarded, the sum so
1482    /// far becomes the mean of the samples before this one, and every later sample enters as
1483    /// `sample / count` weighted by `(count - 1) / count`. Dividing before adding is what keeps a
1484    /// mean of finite inputs finite: two histograms of `1e308` at one observation each average to
1485    /// `1e308`, while `(1e308 + 1e308) / 2` would be `Inf`.
1486    fn fold_avg(&mut self, sample: &FloatHistogram) -> Result<()> {
1487        self.count += 1.0;
1488        if self.mean.is_none() {
1489            // The trial runs on a copy, so an overflowing add leaves the sum it started from intact
1490            // -- that sum is what the mean begins at.
1491            let mut trial = self.value.clone();
1492            let (compensation, outcome) = trial.kahan_add(sample, self.compensation.clone())?;
1493            // A reconciliation counts whether or not the trial is kept: the two sides' bounds meet
1494            // in it either way.
1495            self.evidence.custom_bounds_reconciled |= outcome.custom_bounds_reconciled;
1496            if !trial.has_overflow() {
1497                self.value = trial;
1498                self.compensation = Some(compensation);
1499                return Ok(());
1500            }
1501            let mut started = self.value.clone();
1502            started.div(self.count - 1.0);
1503            if let Some(compensation) = self.compensation.as_mut() {
1504                compensation.div(self.count - 1.0);
1505            }
1506            self.mean = Some(started);
1507        }
1508        // The mean's unit follows the value's: the compensation is scaled by the same weight the
1509        // value is, and the incoming sample is divided before it is added.
1510        let q = (self.count - 1.0) / self.count;
1511        if let Some(compensation) = self.compensation.as_mut() {
1512            compensation.mul(q);
1513        }
1514        let mut sample = sample.clone();
1515        sample.div(self.count);
1516        let mean = self.mean.as_mut().expect("set above");
1517        mean.mul(q);
1518        let (compensation, outcome) = mean.kahan_add(&sample, self.compensation.take())?;
1519        self.compensation = Some(compensation);
1520        self.evidence.custom_bounds_reconciled |= outcome.custom_bounds_reconciled;
1521        Ok(())
1522    }
1523}

1526    /// This position's state as the frame payload a `Partial` hands over.
1527    ///
1528    /// The payload is the fold's own state, not its answer: a sum exports the running sum and the
1529    /// compensation beside it, and folding the compensation in belongs to the merge that finishes the
1530    /// answer. An average exports whichever value it is keeping -- the sum it still adds to, or the
1531    /// mean it switched to -- with the count and the mode flag that say how to read them.
1532    ///
1533    /// A poisoned position exports the empty placeholders the state layout reserves for it: it
1534    /// answers nothing, and the merge reads its evidence, not its buckets.
1535    fn payload(&self, op: AggregateOp) -> Result<StatePayload> {
1536        if self.evidence.poisoned {
1537            let empty = self.value.empty_like();
1538            return Ok(StatePayload::Histogram(Box::new(HistogramState {
1539                value: histogram_batch(&empty)?,
1540                compensation: histogram_batch(&empty)?,
1541                count: 0.0,
1542                incremental_mean: false,
1543                poisoned: true,
1544                incompatible_layout: self.evidence.incompatible_layout,
1545                custom_bounds_reconciled: self.evidence.custom_bounds_reconciled,
1546                counter_reset_seen: self.evidence.counter_reset_seen,
1547                not_counter_reset_seen: self.evidence.not_counter_reset_seen,
1548            })));
1549        }
1550        let switched = self.mean.is_some();
1551        let value = match op {
1552            // A sum's state is its running sum: adding the compensation in is the merge's job.
1553            AggregateOp::Sum => &self.value,
1554            AggregateOp::Avg if switched => self.mean.as_ref().expect("checked above"),
1555            AggregateOp::Avg => &self.value,
1556            op => unreachable!("{op:?} does not fold a histogram payload"),
1557        };
1558        let compensation = self
1559            .compensation
1560            .clone()
1561            .unwrap_or_else(|| value.new_compensation());
1562        Ok(StatePayload::Histogram(Box::new(HistogramState {
1563            value: histogram_batch(value)?,
1564            compensation: histogram_batch(&compensation)?,

1566            // payload cannot claim a component its kind does not carry.
1567            count: if op == AggregateOp::Avg {
1568                self.count
1569            } else {
1570                0.0
1571            },
1572            incremental_mean: switched && op == AggregateOp::Avg,
1573            poisoned: false,
1574            incompatible_layout: self.evidence.incompatible_layout,
1575            custom_bounds_reconciled: self.evidence.custom_bounds_reconciled,
1576            counter_reset_seen: self.evidence.counter_reset_seen,
1577            not_counter_reset_seen: self.evidence.not_counter_reset_seen,
1578        })))
1579    }

1581    /// The state a decoded payload carries, as the merge keeps it.
1582    fn from_payload(op: AggregateOp, payload: &HistogramState) -> Result<Self> {
1583        let value = decoded_histogram(&payload.value)?;
1584        let compensation = decoded_histogram(&payload.compensation)?;
1585        let switched = op == AggregateOp::Avg && payload.incremental_mean;
1586        Ok(Self {
1587            // An average that switched keeps its mean in the mean slot; the sum it switched from is
1588            // not part of the state it exported, so the value slot holds the mean too.
1589            value: value.clone(),
1590            compensation: Some(compensation),
1591            mean: switched.then_some(value),
1592            count: payload.count,
1593            // A decoded state is one point's payload, so whether its two hint bits contradict is
1594            // decided where the whole answer is, in `evidence_summary`: the state layout carries the
1595            // bits, not the decision.
1596            evidence: HistogramEvidence {
1597                poisoned: payload.poisoned,
1598                incompatible_layout: payload.incompatible_layout,
1599                custom_bounds_reconciled: payload.custom_bounds_reconciled,
1600                counter_reset_seen: payload.counter_reset_seen,
1601                not_counter_reset_seen: payload.not_counter_reset_seen,
1602                counter_reset_collision: false,
1603            },
1604        })
1605    }

1607    /// Merges another partition's state into this position's.
1608    ///
1609    /// The rules are the design's: a sum adds both sides' values and then both sides' compensations
1610    /// into the live value, and an average either adds its sums the same way -- when that stays
1611    /// finite -- or continues in mean mode from the means and the counts, dividing before it adds.
1612    /// Poison is sticky, and both sides' evidence is kept whatever the outcome of the arithmetic.
1613    fn merge_state(&mut self, op: AggregateOp, other: &HistogramState) -> Result<()> {
1614        self.evidence.poisoned |= other.poisoned;
1615        self.evidence.incompatible_layout |= other.incompatible_layout;
1616        self.evidence.custom_bounds_reconciled |= other.custom_bounds_reconciled;
1617        self.evidence.counter_reset_seen |= other.counter_reset_seen;
1618        self.evidence.not_counter_reset_seen |= other.not_counter_reset_seen;
1619        if self.evidence.poisoned {
1620            // A poisoned point answers nothing, so there is no value to merge into it.
1621            return Ok(());
1622        }
1623        let right = decoded_histogram(&other.value)?;
1624        let right_compensation = decoded_histogram(&other.compensation)?;
1625        // Two partitions can each be valid on their own and still disagree: one folded exponential
1626        // buckets, the other custom ones. The fold removes such a point when both layouts meet
1627        // inside one partition, so the merge reports it the same way instead of letting the add
1628        // refuse the layouts and fail the whole statement.
1629        let left_uses_custom = match &self.mean {
1630            Some(mean) => mean.uses_custom_buckets(),
1631            None => self.value.uses_custom_buckets(),
1632        };
1633        if left_uses_custom != right.uses_custom_buckets() {
1634            self.evidence.poisoned = true;
1635            self.evidence.incompatible_layout = true;
1636            return Ok(());
1637        }
1638        match op {
1639            AggregateOp::Sum => self.merge_sum(right, right_compensation),
1640            AggregateOp::Avg => self.merge_avg(
1641                right,
1642                right_compensation,
1643                other.count,
1644                other.incremental_mean,
1645            ),
1646            op => unreachable!("{op:?} does not merge a histogram payload"),
1647        }
1648    }

1650    /// Merges two sums: both values and then both compensations go into the live value.
1651    ///
1652    /// The receiver stays the live value for both adds -- the compensation's unit follows the value's
1653    /// -- so the second add folds the other side's compensation into the sum this state carries, not
1654    /// into the compensation the first add returned.
1655    fn merge_sum(
1656        &mut self,
1657        right: FloatHistogram,
1658        right_compensation: FloatHistogram,
1659    ) -> Result<()> {
1660        let left_compensation = self
1661            .compensation
1662            .take()
1663            .expect("a folding position keeps its compensation");
1664        let (compensation, first) = self.value.kahan_add(&right, Some(left_compensation))?;
1665        let (compensation, second) = self
1666            .value
1667            .kahan_add(&right_compensation, Some(compensation))?;
1668        self.compensation = Some(compensation);
1669        self.evidence.custom_bounds_reconciled |=
1670            first.custom_bounds_reconciled || second.custom_bounds_reconciled;
1671        Ok(())
1672    }

1674    /// Merges an average: the sums add while they stay finite, and otherwise the means and their
1675    /// counts do.
1676    fn merge_avg(
1677        &mut self,
1678        right: FloatHistogram,
1679        right_compensation: FloatHistogram,
1680        right_count: f64,
1681        right_is_mean: bool,
1682    ) -> Result<()> {
1683        let left_is_mean = self.mean.is_some();
1684        let left_count = self.count;
1685        let total = left_count + right_count;
1686        let left_value = if left_is_mean {
1687            self.mean.clone().expect("checked above")
1688        } else {
1689            self.value.clone()
1690        };
1691        let left_compensation = self
1692            .compensation
1693            .clone()
1694            .unwrap_or_else(|| left_value.new_compensation());
1695        if !left_is_mean && !right_is_mean {
1696            // Sum and sum: steps 1-2 on a copy of the left sum. The trial is what decides the mode,
1697            // and a reconciliation it reports counts whether or not it is kept.
1698            let mut trial = left_value.clone();
1699            let (compensation, first) = trial.kahan_add(&right, Some(left_compensation.clone()))?;
1700            let (compensation, second) =
1701                trial.kahan_add(&right_compensation, Some(compensation))?;
1702            self.evidence.custom_bounds_reconciled |=
1703                first.custom_bounds_reconciled || second.custom_bounds_reconciled;
1704            if !trial.has_overflow() {
1705                self.value = trial;
1706                self.compensation = Some(compensation);
1707                self.count = total;
1708                return Ok(());
1709            }
1710        }
1711        // The mean path, written so that no intermediate product or sum can overflow: each side
1712        // becomes a mean, its count becomes a weight, and the weighted means are scaled before they
1713        // are added -- with their compensations scaled the same way, because the compensation's unit
1714        // is the value's.
1715        let weight_left = left_count / total;
1716        let weight_right = right_count / total;
1717        let mut left_mean = left_value;
1718        let mut left_mean_compensation = left_compensation;
1719        if !left_is_mean {
1720            left_mean.div(left_count);
1721            left_mean_compensation.div(left_count);
1722        }
1723        let mut right_mean = right;
1724        let mut right_mean_compensation = right_compensation;
1725        if !right_is_mean {
1726            right_mean.div(right_count);
1727            right_mean_compensation.div(right_count);
1728        }
1729        left_mean.mul(weight_left);
1730        left_mean_compensation.mul(weight_left);
1731        right_mean.mul(weight_right);
1732        right_mean_compensation.mul(weight_right);
1733        let (compensation, first) =
1734            left_mean.kahan_add(&right_mean, Some(left_mean_compensation))?;
1735        let (compensation, second) =
1736            left_mean.kahan_add(&right_mean_compensation, Some(compensation))?;
1737        self.value = left_mean.clone();
1738        self.mean = Some(left_mean);
1739        self.compensation = Some(compensation);
1740        self.count = total;
1741        self.evidence.custom_bounds_reconciled |=
1742            first.custom_bounds_reconciled || second.custom_bounds_reconciled;
1743        Ok(())
1744    }
1745}

1762/// The evidence a fold or a merge has to report, once, wherever it finishes.
1763///
1764/// A poisoned point suppresses its own reconciliation bit: the point answers nothing, so reporting
1765/// that its bounds were reconciled would report a reconciliation of an answer that is not there.
1766///
1767/// The reset hints are per point too: a point whose samples report a reset and no reset contradicts
1768/// itself -- Prometheus skips such a point -- while two points that each report one of the two do
1769/// not contradict each other, and neither does one series whose hint changes from one point to the
1770/// next. The summary therefore ORs the per-point collision rather than the two bits themselves: a
1771/// `sum by (job)` whose `job` a only ever resets and whose `job` b never does is two ordinary
1772/// points, and a point cannot be both poisoned and in contradiction.
1773///
1774/// Everything else is reported if any point recorded it, which is what makes the message set the same
1775/// whichever order the partitions were merged in.
1776fn evidence_summary(groups: &[Group]) -> HistogramEvidence {
1777    let mut summary = HistogramEvidence::default();
1778    for accumulator in groups.iter().flat_map(|group| group.histograms.values()) {
1779        let evidence = accumulator.evidence;
1780        summary.poisoned |= evidence.poisoned;
1781        summary.incompatible_layout |= evidence.incompatible_layout;
1782        summary.counter_reset_seen |= evidence.counter_reset_seen;
1783        summary.not_counter_reset_seen |= evidence.not_counter_reset_seen;
1784        summary.counter_reset_collision |=
1785            evidence.counter_reset_seen && evidence.not_counter_reset_seen && !evidence.poisoned;
1786        summary.custom_bounds_reconciled |= evidence.custom_bounds_reconciled && !evidence.poisoned;
1787    }
1788    summary
1789}

1822/// The float part of Prometheus' `groupedAggregation`.
1823#[derive(Clone, Copy, Default)]
1824struct FloatState {
1825    seen: bool,
1826    value: f64,
1827    mean: f64,
1828    compensation: f64,
1829    count: f64,
1830    incremental_mean: bool,
1831}
1832
1833impl FloatState {
1834    fn update(&mut self, op: AggregateOp, sample: f64) {
1835        if !self.seen {
1836            *self = Self {
1837                seen: true,
1838                value: match op {
1839                    // The running sum of squared deviations; Prometheus starts it at NaN for a
1840                    // non-finite first sample.
1841                    AggregateOp::Stddev | AggregateOp::Stdvar if !sample.is_finite() => f64::NAN,
1842                    AggregateOp::Stddev | AggregateOp::Stdvar => 0.0,
1843                    _ => sample,
1844                },
1845                mean: sample,
1846                count: 1.0,
1847                ..Self::default()
1848            };
1849            return;
1850        }
1851        match op {
1852            AggregateOp::Sum => {
1853                (self.value, self.compensation) = kahan_inc(sample, self.value, self.compensation);
1854            }
1855            AggregateOp::Avg => self.update_avg(sample),
1856            AggregateOp::Count => self.count += 1.0,
1857            AggregateOp::Max => {
1858                if self.value < sample || self.value.is_nan() {
1859                    self.value = sample;
1860                }
1861            }
1862            AggregateOp::Min => {
1863                if self.value > sample || self.value.is_nan() {
1864                    self.value = sample;
1865                }
1866            }
1867            AggregateOp::Group => {}
1868            // Welford's update, uncompensated as in Prometheus.
1869            AggregateOp::Stddev | AggregateOp::Stdvar => {
1870                self.count += 1.0;
1871                let delta = sample - self.mean;
1872                self.mean += delta / self.count;
1873                self.value += delta * (sample - self.mean);
1874            }
1875        }
1876    }
1877
1878    /// Sums directly until the running sum would overflow, then switches to an incremental mean.
1879    fn update_avg(&mut self, sample: f64) {
1880        self.count += 1.0;
1881        if !self.incremental_mean {
1882            let (value, compensation) = kahan_inc(sample, self.value, self.compensation);
1883            if !value.is_infinite() {
1884                (self.value, self.compensation) = (value, compensation);
1885                return;
1886            }
1887            self.incremental_mean = true;
1888            self.mean = self.value / (self.count - 1.0);
1889            self.compensation /= self.count - 1.0;
1890        }
1891        let q = (self.count - 1.0) / self.count;
1892        (self.mean, self.compensation) =
1893            kahan_inc(sample / self.count, q * self.mean, q * self.compensation);
1894    }
1895
1896    fn finish(self, op: AggregateOp) -> Option<f64> {
1897        if !self.seen {
1898            return None;
1899        }
1900        Some(match op {
1901            AggregateOp::Sum => self.value + self.compensation,
1902            AggregateOp::Avg if self.incremental_mean => self.mean + self.compensation,
1903            AggregateOp::Avg => self.value / self.count + self.compensation / self.count,
1904            AggregateOp::Count => self.count,
1905            AggregateOp::Min | AggregateOp::Max => self.value,
1906            AggregateOp::Group => 1.0,
1907            AggregateOp::Stdvar => self.value / self.count,
1908            AggregateOp::Stddev => (self.value / self.count).sqrt(),
1909        })
1910    }
1911}

1913/// Prometheus `kahansum.Inc`, with the Neumaier swap.
1914fn kahan_inc(inc: f64, sum: f64, compensation: f64) -> (f64, f64) {
1915    let t = sum + inc;
1916    let compensation = if t.is_infinite() {
1917        0.0
1918    } else if sum.abs() >= inc.abs() {
1919        compensation + ((sum - t) + inc)
1920    } else {
1921        compensation + ((inc - t) + sum)
1922    };
1923    (t, compensation)
1924}
