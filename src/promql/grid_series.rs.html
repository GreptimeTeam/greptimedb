38pub(crate) fn series_bytes(series: &Series) -> usize {
39    let runs = series.time.runs().count();
40    let segments: usize = series
41        .fields
42        .iter()
43        .map(|field| field.segments().len())
44        .sum();
45    std::mem::size_of::<Series>()
46        .saturating_add(std::mem::size_of::<TimeAxis>())
47        .saturating_add(runs.saturating_mul(std::mem::size_of::<(usize, usize)>()))
48        .saturating_add(
49            segments.saturating_mul(std::mem::size_of::<Samples>() + std::mem::size_of::<usize>()),
50        )
51        .saturating_add(fields_bytes(&series.fields))
52        .saturating_add(series.metric.name().map_or(0, str::len))
53        .saturating_add(std::mem::size_of_val(series.metric.tags()))
54        .saturating_add(
55            series
56                .metric
57                .tags()
58                .iter()
59                .map(|tag| tag.name.len() + tag.value.len())
60                .sum::<usize>(),
61        )
62}

64pub(crate) fn fields_bytes(fields: &SeriesFields) -> usize {
65    fields
66        .iter()
67        .map(|field| {
68            field.segments().iter().fold(0usize, |total, segment| {
69                total.saturating_add(segment_bytes(segment))
70            })
71        })
72        .fold(0usize, usize::saturating_add)
73}

75/// The bytes one dense segment of samples holds.
76pub(crate) fn segment_bytes(segment: &Samples) -> usize {
77    match segment.dense() {
78        Some(DenseSamplesRef::Float(values)) => values.get_array_memory_size(),
79        Some(DenseSamplesRef::Histogram(histograms)) => histograms
80            .indices()
81            .get_array_memory_size()
82            .saturating_add(table_bytes(histograms.batch())),
83        Some(DenseSamplesRef::Mixed(mixed)) => mixed
84            .floats()
85            .get_array_memory_size()
86            .saturating_add(mixed.histograms().indices().get_array_memory_size())
87            .saturating_add(table_bytes(mixed.histograms().batch())),
88        // `segments` yields dense segments only.
89        None => 0,
90    }
91}

93pub(crate) fn table_bytes(table: &Arc<HistogramBatch>) -> usize {
94    std::mem::size_of::<HistogramBatch>().saturating_add(table.as_arrow().get_array_memory_size())
95}

101/// Builds one grid series from rows pushed at increasing absolute grid indices.
102///
103/// A row is kept only when at least one field holds a value; a field without a value at a kept row
104/// gets a null. The rows are handed out as one [`Series`] whose [`TimeAxis`] names their maximal
105/// runs, so positions without a value cost nothing.
106///
107/// A builder serves one series at a time; only [`GridSeriesBuilder::begin_series`] resets it.
108pub struct GridSeriesBuilder {
109    schema: Arc<SeriesSchema>,
110    grid: Arc<EvalGrid>,
111    /// Smallest index the next push may use.
112    next: usize,
113    /// Grid index of each retained row, ascending.
114    indices: Vec<usize>,
115    /// Empty until the first retained row of a hand-out.
116    fields: Vec<FieldBuilder>,
117}
118
119struct FieldBuilder {
120    values: Vec<f64>,
121    nulls: NullBufferBuilder,
122    /// Created at the first histogram value; holds an index per row from then on.
123    histograms: Option<HistogramColumn>,
124}
125
126struct HistogramColumn {
127    batch: HistogramBatchBuilder,
128    indices: Vec<Option<u32>>,
129}

134    /// Boxed so that buffered float results stay as small as `Option<f64>`.

140/// `sample` as an owned value; `None` when missing.
141pub(crate) fn owned(sample: SampleRef<'_>) -> Result<Option<GridSample>> {
142    Ok(match sample {
143        SampleRef::Missing => None,
144        SampleRef::Float(value) => Some(GridSample::Float(value)),
145        SampleRef::Histogram(view) => Some(GridSample::Histogram(Box::new(
146            FloatHistogram::try_from_view(&view)?,
147        ))),
148    })

151impl GridSeriesBuilder {
152    /// Errors when `schema` has no evaluation grid.
153    pub fn try_new(schema: Arc<SeriesSchema>) -> Result<Self> {
154        let grid = schema.eval_grid.clone().ok_or_else(|| {
155            DataFusionError::Execution("grid series schema requires an evaluation grid".into())
156        })?;
157        Ok(Self {
158            schema,
159            grid,
160            next: 0,
161            indices: Vec::new(),
162            fields: Vec::new(),
163        })
164    }
165
166    /// Reserves capacity for the exact number of valued rows in the next series.
167    ///
168    /// Call after `begin_series` and before its first push; callers that skip it rely on normal
169    /// growth.
170    pub(crate) fn reserve_rows(&mut self, rows: usize) -> Result<()> {
171        self.indices
172            .try_reserve_exact(rows.saturating_sub(self.indices.len()))
173            .map_err(|error| {
174                DataFusionError::Execution(format!("cannot reserve grid row indices: {error}"))
175            })?;
176        if rows == 0 {
177            return Ok(());
178        }
179        if self.fields.is_empty() {
180            self.fields = (0..self.schema.fields.len())
181                .map(|_| FieldBuilder::new(rows))
182                .collect();
183        }
184        for field in &mut self.fields {
185            field
186                .values
187                .try_reserve_exact(rows.saturating_sub(field.values.len()))
188                .map_err(|error| {
189                    DataFusionError::Execution(format!("cannot reserve grid values: {error}"))
190                })?;
191            field.nulls = NullBufferBuilder::new(rows);
192        }
193        Ok(())
194    }
195
196    /// Pushes the field values at absolute grid `index`; `None` is a missing value.
197    ///
198    /// Errors unless indices strictly increase within the grid and `values` has one entry per field.
199    pub fn push(&mut self, index: usize, values: &[Option<f64>]) -> Result<()> {
200        if !self.start_row(index, values.len(), values.iter().all(Option::is_none))? {
201            return Ok(());
202        }
203        for (field, value) in self.fields.iter_mut().zip(values) {
204            field.push_float(*value);
205        }
206        Ok(())
207    }
208
209    /// Like [`GridSeriesBuilder::push`], with native histogram values.
210    pub(crate) fn push_samples(
211        &mut self,
212        index: usize,
213        values: &[Option<GridSample>],
214    ) -> Result<()> {
215        if !self.start_row(index, values.len(), values.iter().all(Option::is_none))? {
216            return Ok(());
217        }
218        for (field, value) in self.fields.iter_mut().zip(values) {
219            match value {
220                Some(GridSample::Histogram(h)) => {
221                    let column = field.start_histogram();
222                    column.indices.push(Some(h.push_to(&mut column.batch)?));
223                }
224                Some(GridSample::Float(value)) => field.push_float(Some(*value)),
225                None => field.push_float(None),
226            }
227        }
228        Ok(())
229    }
230
231    /// Like [`GridSeriesBuilder::push_samples`], copying histograms from their views without an
232    /// owned intermediate.
233    pub(crate) fn push_refs(&mut self, index: usize, values: &[SampleRef<'_>]) -> Result<()> {
234        if !self.start_row(
235            index,
236            values.len(),
237            values
238                .iter()
239                .all(|value| matches!(value, SampleRef::Missing)),
240        )? {
241            return Ok(());
242        }
243        for (field, value) in self.fields.iter_mut().zip(values) {
244            match value {
245                SampleRef::Histogram(view) => {
246                    let column = field.start_histogram();
247                    column
248                        .indices
249                        .push(Some(column.batch.push(*view).map_err(external)?));
250                }
251                SampleRef::Float(value) => field.push_float(Some(*value)),
252                SampleRef::Missing => field.push_float(None),
253            }
254        }
255        Ok(())
256    }
257
258    /// Validates the row and retains it; returns whether it holds a value to write.
259    fn start_row(&mut self, index: usize, len: usize, empty: bool) -> Result<bool> {
260        if len != self.schema.fields.len() {
261            return Err(DataFusionError::Execution(format!(
262                "grid series row has {len} fields, schema has {}",
263                self.schema.fields.len()
264            )));
265        }
266        if index < self.next || index >= self.grid.spec().len as usize {
267            return Err(DataFusionError::Execution(format!(
268                "grid series index {index} is not increasing or exceeds grid length {}",
269                self.grid.spec().len as usize
270            )));
271        }
272        self.next = index + 1;
273        if empty {
274            return Ok(false);
275        }
276        if self.fields.is_empty() {
277            self.fields = (0..len).map(|_| FieldBuilder::new(0)).collect();
278        }
279        self.indices.push(index);
280        Ok(true)
281    }
282
283    /// Whether the next [`Self::finish`] or [`Self::take_buffered`] has rows to return.
284    pub fn has_pending(&self) -> bool {
285        !self.indices.is_empty()
286    }
287
288    /// Starts the series the following pushes belong to and resets the index cursor.
289    ///
290    /// Panics in debug builds when rows of the previous series are still pending.
291    pub fn begin_series(&mut self) {
292        debug_assert!(
293            !self.has_pending(),
294            "a new series starts with rows of the previous one still pending"
295        );
296        self.next = 0;
297        self.indices.clear();
298        self.fields.clear();
299    }
300
301    /// Hands out the rows pushed since the last hand-out as one [`Series`], or `None` when there
302    /// are none. The index cursor is kept, so later pushes continue the same series.
303    ///
304    /// Same as [`Self::finish`]; the name marks a prefix flush at the call site.
305    pub fn take_buffered(&mut self, metric: SeriesLabels) -> Result<Option<Series>> {
306        self.collect(metric)
307    }
308
309    /// Hands out the rows pushed since the last hand-out as one [`Series`], or `None` when there
310    /// are none. Same as [`Self::take_buffered`]: neither resets the index cursor, so a caller
311    /// starting another series calls [`Self::begin_series`].
312    pub fn finish(&mut self, metric: SeriesLabels) -> Result<Option<Series>> {
313        self.collect(metric)
314    }
315
316    fn collect(&mut self, metric: SeriesLabels) -> Result<Option<Series>> {
317        if self.indices.is_empty() {
318            return Ok(None);
319        }
320        let fields = std::mem::take(&mut self.fields);
321        let indices = std::mem::take(&mut self.indices);
322        let mut samples = Vec::new();
323        samples.try_reserve_exact(fields.len()).map_err(|error| {
324            DataFusionError::Execution(format!("cannot reserve grid field descriptions: {error}"))
325        })?;
326        for mut field in fields {
327            let floats = Float64Array::new(field.values.into(), field.nulls.finish());
328            let sample = if let Some(column) = field.histograms {
329                let batch = column.batch.finish().map_err(external)?;
330                let histograms =
331                    HistogramSamples::try_new(UInt32Array::from(column.indices), batch)
332                        .map_err(external)?;
333                Samples::try_from_mixed(floats, histograms).map_err(external)?
334            } else {
335                Samples::from_float(floats)
336            };
337            samples.push(sample);
338        }
339        let fields = SeriesFields::try_from_vec(samples).map_err(external)?;
340        let run_count = count_runs(&indices);
341        let spans = spans_of(&indices, run_count)?;
342        let time = TimeAxis::from_grid_spans(self.grid.clone(), spans).map_err(external)?;
343        Series::try_new(&self.schema, metric, time, fields)
344            .map(Some)
345            .map_err(external)
346    }
347}

349/// Counts the maximal runs of consecutive positions in ascending `indices`.
350fn count_runs(indices: &[usize]) -> usize {
351    indices
352        .windows(2)
353        .filter(|pair| pair[1] != pair[0] + 1)
354        .count()
355        .saturating_add(usize::from(!indices.is_empty()))
356}
357
358fn spans_of(indices: &[usize], runs: usize) -> Result<Vec<GridSpan>> {
359    let mut spans: Vec<GridSpan> = Vec::new();
360    spans.try_reserve_exact(runs).map_err(|error| {
361        DataFusionError::Execution(format!("cannot reserve grid spans: {error}"))
362    })?;
363    let mut start = 0;
364    while let Some(&position) = indices.get(start) {
365        let mut len = 1;
366        while indices.get(start + len) == Some(&(position + len)) {
367            len += 1;
368        }
369        spans.push(GridSpan::new(position, len));
370        start += len;
371    }
372    Ok(spans)
373}

375impl FieldBuilder {
376    fn new(rows: usize) -> Self {
377        Self {
378            values: Vec::new(),
379            nulls: NullBufferBuilder::new(rows),
380            histograms: None,
381        }
382    }
383
384    fn push_float(&mut self, value: Option<f64>) {
385        self.values.push(value.unwrap_or_default());
386        self.nulls.append(value.is_some());
387        if let Some(column) = &mut self.histograms {
388            column.indices.push(None);
389        }
390    }
391
392    /// Writes the float side of a histogram row and returns the column its index goes into.
393    fn start_histogram(&mut self) -> &mut HistogramColumn {
394        self.values.push(0.0);
395        self.nulls.append(false);
396        let rows = self.values.len() - 1;
397        self.histograms.get_or_insert_with(|| HistogramColumn {
398            batch: HistogramBatchBuilder::new(),
399            indices: vec![None; rows],
400        })
401    }
402}

407pub(crate) type LabelSet = Vec<(Arc<str>, Arc<str>)>;

409/// A label set as Prometheus prints one in errors: `{name="value", ...}`.
410pub(crate) fn render_label_set(labels: &LabelSet) -> String {
411    let labels: Vec<String> = labels
412        .iter()
413        .map(|(name, value)| format!("{name}=\"{value}\""))
414        .collect();
415    format!("{{{}}}", labels.join(", "))
416}

418/// The label set Prometheus compares, which has no empty values.
419pub(crate) fn label_set(metric: &SeriesLabels) -> LabelSet {
420    let name = metric
421        .name()
422        .filter(|name| !name.is_empty())
423        .map(|name| (Arc::from("__name__"), Arc::from(name)));
424    name.into_iter()
425        .chain(
426            metric
427                .tags()
428                .iter()
429                .filter(|tag| !tag.value.is_empty())
430                .map(|tag| (tag.name.clone(), tag.value.clone())),
431        )
432        .collect()
433}

435/// Merges series that share a label set into one, as Prometheus does after a function that
436/// rewrites labels or drops `__name__`. Errors when two of them hold the same field at one point.
437///
438/// Values are sliced from their segments rather than copied. Merged series keep the order of their
439/// first member; the first member's metric names the result.
440pub fn merge_same_labelset(schema: &Arc<SeriesSchema>, series: Vec<Series>) -> Result<Vec<Series>> {
441    let mut groups: HashMap<LabelSet, Vec<usize>> = HashMap::with_capacity(series.len());
442    let mut collided = false;
443    for (index, series) in series.iter().enumerate() {
444        match groups.entry(label_set(&series.metric)) {
445            Entry::Occupied(mut entry) => {
446                entry.get_mut().push(index);
447                collided = true;
448            }
449            Entry::Vacant(entry) => {
450                entry.insert(vec![index]);
451            }
452        }
453    }
454    if !collided {
455        return Ok(series);
456    }
457
458    let grid = schema.eval_grid.clone().ok_or_else(|| {
459        DataFusionError::Execution("grid series schema requires an evaluation grid".into())
460    })?;
461    let mut series: Vec<Option<Series>> = series.into_iter().map(Some).collect();
462    let mut output = Vec::with_capacity(groups.len());
463    // Order by first member so the output order is deterministic across runs.
464    let mut groups: Vec<(LabelSet, Vec<usize>)> = groups.into_iter().collect();
465    groups.sort_unstable_by_key(|(_, members)| members[0]);
466    // `(absolute position, member slot, compact row)`.
467    let mut rows: Vec<(usize, usize, usize)> = Vec::new();
468    let mut positions: Vec<usize> = Vec::new();
469    for (labels, members) in groups {
470        if let [member] = members[..] {
471            output.extend(series[member].take());
472            continue;
473        }
474        let mut fragments: Vec<&Series> = Vec::with_capacity(members.len());
475        for &member in &members {
476            let fragment = series[member].as_ref().ok_or_else(|| {
477                DataFusionError::Execution("each series is in exactly one label set".into())
478            })?;
479            let Some(fragment_grid) = fragment.time.grid() else {
480                return Err(DataFusionError::Execution(
481                    "merging requires evaluated series".into(),
482                ));
483            };
484            if fragment_grid.spec() != grid.spec() {
485                return Err(DataFusionError::Execution(
486                    "merging requires one evaluation grid".into(),
487                ));
488            }
489            if fragment.fields.len() != schema.fields.len() {
490                return Err(DataFusionError::Execution(format!(
491                    "merging requires {} fields, a fragment holds {}",
492                    schema.fields.len(),
493                    fragment.fields.len()
494                )));
495            }
496            let time_len = fragment.time.len();
497            if let Some((field, samples)) = fragment
498                .fields
499                .iter()
500                .enumerate()
501                .find(|(_, samples)| samples.len() != time_len)
502            {
503                return Err(DataFusionError::Execution(format!(
504                    "merging requires every field to hold the {time_len} rows of its axis, field \
505                     {field} holds {}",
506                    samples.len()
507                )));
508            }
509            fragments.push(fragment);
510        }
511        rows.clear();
512        for (member_slot, fragment) in fragments.iter().enumerate() {
513            let mut readers: Vec<FieldReader<'_>> =
514                fragment.fields.iter().map(FieldReader::new).collect();
515            for (position, row) in fragment.time.positions() {
516                let mut present = false;
517                for reader in &mut readers {
518                    let (segment, offset) = reader.locate(row)?;
519                    present |= !matches!(segment.sample(offset), SampleRef::Missing);
520                }
521                if present {
522                    rows.push((position, member_slot, row));
523                }
524            }
525        }
526        // Within one member, rows stay ascending, so its `FieldReader`s only move forward.
527        rows.sort_unstable();
528        positions.clear();
529        positions.extend(rows.iter().map(|&(position, _, _)| position));
530        positions.dedup();
531        let mut fields = Vec::with_capacity(schema.fields.len());
532        for field in 0..schema.fields.len() {
533            let mut readers: Vec<FieldReader<'_>> = fragments
534                .iter()
535                .map(|fragment| FieldReader::new(&fragment.fields.as_slice()[field]))
536                .collect();
537            fields.push(merge_field(&mut readers, &rows, &labels)?);
538        }
539        let time =
540            TimeAxis::from_grid_spans(grid.clone(), spans_of(&positions, count_runs(&positions))?)
541                .map_err(external)?;
542        let fields = SeriesFields::try_from_vec(fields).map_err(external)?;
543        // Members may differ in empty-valued labels; the first member's metric is kept.
544        let metric = fragments[0].metric.clone();
545        output.push(Series::try_new(schema, metric, time, fields).map_err(external)?);
546    }
547    Ok(output)
548}

550/// Builds one field of the merged series, one sample per merged position: slices of member segments
551/// for values, null runs for positions only other fields fill.
552fn merge_field(
553    readers: &mut [FieldReader<'_>],
554    rows: &[(usize, usize, usize)],
555    labels: &LabelSet,
556) -> Result<Samples> {
557    let mut chunks: Vec<Samples> = Vec::new();
558    let mut run: Option<FieldRun<'_>> = None;
559    let mut missing = 0;
560    let mut start = 0;
561    while start < rows.len() {
562        let position = rows[start].0;
563        let mut end = start;
564        let mut value: Option<(usize, &Samples, usize)> = None;
565        while end < rows.len() && rows[end].0 == position {
566            let (_, member, row) = rows[end];
567            let (segment, offset) = readers[member].locate(row)?;
568            if !matches!(segment.sample(offset), SampleRef::Missing) {
569                if value.is_some() {
570                    let labels: Vec<_> = labels
571                        .iter()
572                        .map(|(name, value)| format!("{name}=\"{value}\""))
573                        .collect();
574                    return Err(DataFusionError::Execution(format!(
575                        "vector cannot contain metrics with the same labelset {{{}}}",
576                        labels.join(", ")
577                    )));
578                }
579                value = Some((member, segment, offset));
580            }
581            end += 1;
582        }
583        match value {
584            Some((member, segment, offset)) => {
585                if let Some(run) = &mut run
586                    && run.continues(member, segment, offset)
587                {
588                    run.len += 1;
589                } else {
590                    if let Some(run) = run.take() {
591                        run.push(&mut chunks)?;
592                    }
593                    push_missing(&mut chunks, &mut missing);
594                    run = Some(FieldRun {
595                        member,
596                        segment,
597                        offset,
598                        len: 1,
599                    });
600                }
601            }
602            None => {
603                if let Some(run) = run.take() {
604                    run.push(&mut chunks)?;
605                }
606                missing += 1;
607            }
608        }
609        start = end;
610    }
611    if let Some(run) = run {
612        run.push(&mut chunks)?;
613    }
614    push_missing(&mut chunks, &mut missing);
615    Samples::try_from_chunks(chunks).map_err(external)
616}

618/// Consecutive rows of one member's dense segment, handed out as one [`Samples::slice`].
619struct FieldRun<'a> {
620    member: usize,
621    segment: &'a Samples,
622    offset: usize,
623    len: usize,
624}
625
626impl FieldRun<'_> {
627    fn continues(&self, member: usize, segment: &Samples, offset: usize) -> bool {
628        self.member == member
629            && std::ptr::eq(self.segment, segment)
630            && self.offset + self.len == offset
631    }
632
633    fn push(self, chunks: &mut Vec<Samples>) -> Result<()> {
634        chunks.push(
635            self.segment
636                .slice(self.offset, self.len)
637                .map_err(external)?,
638        );
639        Ok(())
640    }
641}

643/// Forward-only reader over a chunked field. [`Samples::sample`] rescans segments from the first on
644/// every call; this walks them once.
645struct FieldReader<'a> {
646    segments: &'a [Samples],
647    segment: usize,
648    /// Logical row the current segment starts at.
649    start: usize,
650}
651
652impl<'a> FieldReader<'a> {
653    fn new(samples: &'a Samples) -> Self {
654        Self {
655            segments: samples.segments(),
656            segment: 0,
657            start: 0,
658        }
659    }
660
661    /// Returns the dense segment holding `row` and the offset inside it. `row` must not go
662    /// backwards.
663    fn locate(&mut self, row: usize) -> Result<(&'a Samples, usize)> {
664        while let Some(segment) = self.segments.get(self.segment) {
665            if let Some(offset) = row
666                .checked_sub(self.start)
667                .filter(|offset| *offset < segment.len())
668            {
669                return Ok((segment, offset));
670            }
671            self.segment += 1;
672            self.start += segment.len();
673        }
674        Err(DataFusionError::Execution(
675            "merging requires each field to be row-aligned with its time axis".into(),
676        ))
677    }
678}
679
680fn push_missing(chunks: &mut Vec<Samples>, missing: &mut usize) {
681    if *missing != 0 {
682        chunks.push(Samples::from_float(Float64Array::new_null(*missing)));
683        *missing = 0;
684    }
685}
686
687pub(crate) fn external(error: datatypes::error::Error) -> DataFusionError {
688    DataFusionError::External(Box::new(error))
689}
