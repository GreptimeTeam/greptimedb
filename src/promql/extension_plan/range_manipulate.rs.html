57/// Time series manipulator for range function.
58///
59/// This plan will "fold" time index and value columns into [RangeArray]s, and truncate
60/// other columns to the same length with the "folded" [RangeArray] column.
61///
62/// To pass runtime information to the execution plan (or the range function), This plan
63/// will add those extra columns:
64/// - timestamp range with type [RangeArray], which is the folded timestamp column.
65/// - end of current range with the same type as the timestamp column. (todo)
66#[derive(Debug, PartialEq, Eq, Hash)]
67pub struct RangeManipulate {
68    start: Millisecond,
69    end: Millisecond,
70    interval: Millisecond,
71    range: Millisecond,
72    offset: Millisecond,
73    time_index: String,
74    field_columns: Vec<String>,
75    input: LogicalPlan,
76    output_schema: DFSchemaRef,
77    unfix: Option<UnfixIndices>,
78}

130    fn calculate_output_schema(
131        input_schema: &DFSchemaRef,
132        time_index: &str,
133        field_columns: &[String],
134    ) -> DataFusionResult<DFSchemaRef> {
135        let columns = input_schema.fields();
136        let mut new_columns = Vec::with_capacity(columns.len() + 1);
137        for i in 0..columns.len() {
138            let x = input_schema.qualified_field(i);
139            new_columns.push((x.0.cloned(), x.1.clone()));
140        }
141
142        // process time index column
143        // the raw timestamp field is preserved. And a new timestamp_range field is appended to the last.
144        let Some(ts_col_index) = input_schema.index_of_column_by_name(None, time_index) else {
145            return Err(datafusion::common::field_not_found(
146                None::<TableReference>,
147                time_index,
148                input_schema.as_ref(),
149            ));
150        };
151        let ts_col_field = &columns[ts_col_index];
152        let output_time_field = Arc::new(
153            ts_col_field
154                .as_ref()
155                .clone()
156                .with_data_type(DataType::Timestamp(TimeUnit::Millisecond, None)),
157        );
158        new_columns[ts_col_index] = (
159            input_schema.qualified_field(ts_col_index).0.cloned(),
160            output_time_field.clone(),
161        );
162        let timestamp_range_field = Field::new(
163            Self::build_timestamp_range_name(time_index),
164            RangeArray::convert_field(output_time_field.as_ref())
165                .data_type()
166                .clone(),
167            ts_col_field.is_nullable(),
168        );
169        new_columns.push((None, Arc::new(timestamp_range_field)));
170
171        // process value columns
172        for name in field_columns {
173            let Some(index) = input_schema.index_of_column_by_name(None, name) else {
174                return Err(datafusion::common::field_not_found(
175                    None::<TableReference>,
176                    name,
177                    input_schema.as_ref(),
178                ));
179            };
180            new_columns[index] = (None, Arc::new(RangeArray::convert_field(&columns[index])));
181        }
182
183        Ok(Arc::new(DFSchema::new_with_metadata(
184            new_columns,
185            input_schema.metadata().clone(),
186        )?))
187    }

217        let tag_column_indices = self

235    pub fn deserialize(bytes: &[u8]) -> Result<Self> {
236        let pb_range_manipulate = pb::RangeManipulate::decode(bytes).context(DeserializeSnafu)?;
237        let empty_schema = Arc::new(DFSchema::empty());
238        let placeholder_plan = LogicalPlan::EmptyRelation(EmptyRelation {
239            produce_one_row: false,
240            schema: empty_schema.clone(),
241        });
242
243        let unfix = UnfixIndices {
244            time_index_idx: pb_range_manipulate.time_index_idx,
245            tag_column_indices: pb_range_manipulate.tag_column_indices.clone(),
246        };
247        debug!("RangeManipulate deserialize unfix: {:?}", unfix);
248
249        // Unlike `Self::new()`, this method doesn't check the input schema as it will fail
250        // because the input schema is empty.
251        // But this is Ok since datafusion guarantees to call `with_exprs_and_inputs` for the
252        // deserialized plan.
253        Ok(Self {
254            start: pb_range_manipulate.start,
255            end: pb_range_manipulate.end,
256            interval: pb_range_manipulate.interval,
257            range: pb_range_manipulate.range,
258            offset: 0,
259            time_index: String::new(),
260            field_columns: Vec::new(),
261            input: placeholder_plan,
262            output_schema: empty_schema,
263            unfix: Some(unfix),
264        })
265    }

303impl UserDefinedLogicalNodeCore for RangeManipulate {
304    fn name(&self) -> &str {
305        Self::name()
306    }
307
308    fn inputs(&self) -> Vec<&LogicalPlan> {
309        vec![&self.input]
310    }
311
312    fn schema(&self) -> &DFSchemaRef {
313        &self.output_schema
314    }
315
316    fn expressions(&self) -> Vec<Expr> {
317        if self.unfix.is_some() {
318            return vec![];
319        }
320
321        let mut exprs = Vec::with_capacity(1 + self.field_columns.len());
322        exprs.push(ident(&self.time_index));
323        exprs.extend(self.field_columns.iter().map(ident));
324        exprs
325    }
326
327    fn necessary_children_exprs(&self, output_columns: &[usize]) -> Option<Vec<Vec<usize>>> {
328        if self.unfix.is_some() {
329            return None;
330        }
331
332        let input_schema = self.input.schema();
333        let input_len = input_schema.fields().len();
334        let time_index_idx = input_schema.index_of_column_by_name(None, &self.time_index)?;
335
336        if output_columns.is_empty() {
337            let indices = (0..input_len).collect::<Vec<_>>();
338            return Some(vec![indices]);
339        }
340
341        let mut required = Vec::with_capacity(output_columns.len() + 1 + self.field_columns.len());
342        required.push(time_index_idx);
343        for value_column in &self.field_columns {
344            required.push(input_schema.index_of_column_by_name(None, value_column)?);
345        }
346        for &idx in output_columns {
347            if idx < input_len {
348                required.push(idx);
349            } else if idx == input_len {
350                // Derived timestamp range column.
351                required.push(time_index_idx);
352            } else {
353                warn!(
354                    "Output column index {} is out of bounds for input schema with length {}",
355                    idx, input_len
356                );
357                return None;
358            }
359        }
360
361        required.sort_unstable();
362        required.dedup();
363        Some(vec![required])
364    }
365
366    fn fmt_for_explain(&self, f: &mut std::fmt::Formatter) -> std::fmt::Result {
367        write!(
368            f,
369            "PromRangeManipulate: req range=[{}..{}], interval=[{}], eval range=[{}], time index=[{}], values={:?}",
370            self.start, self.end, self.interval, self.range, self.time_index, self.field_columns
371        )
372    }
373
374    fn with_exprs_and_inputs(
375        &self,
376        _exprs: Vec<Expr>,
377        mut inputs: Vec<LogicalPlan>,
378    ) -> DataFusionResult<Self> {
379        if inputs.len() != 1 {
380            return Err(DataFusionError::Internal(
381                "RangeManipulate should have at exact one input".to_string(),
382            ));
383        }
384
385        let input: LogicalPlan = inputs.pop().unwrap();
386        let input_schema = input.schema();
387
388        if let Some(unfix) = &self.unfix {
389            // transform indices to names
390            let time_index = resolve_column_name(
391                unfix.time_index_idx,
392                input_schema,
393                "RangeManipulate",
394                "time index",
395            )?;
396
397            let field_columns = unfix
398                .tag_column_indices
399                .iter()
400                .map(|idx| resolve_column_name(*idx, input_schema, "RangeManipulate", "tag"))
401                .collect::<DataFusionResult<Vec<String>>>()?;
402
403            let output_schema =
404                Self::calculate_output_schema(input_schema, &time_index, &field_columns)?;
405
406            Ok(Self {
407                start: self.start,
408                end: self.end,
409                interval: self.interval,
410                range: self.range,
411                offset: local_offset(&input, &time_index),
412                time_index,
413                field_columns,
414                input,
415                output_schema,
416                unfix: None,
417            })
418        } else {
419            let output_schema =
420                Self::calculate_output_schema(input_schema, &self.time_index, &self.field_columns)?;
421
422            Ok(Self {
423                start: self.start,
424                end: self.end,
425                interval: self.interval,
426                range: self.range,
427                offset: self.offset,
428                time_index: self.time_index.clone(),
429                field_columns: self.field_columns.clone(),
430                input,
431                output_schema,
432                unfix: None,
433            })
434        }
435    }

651impl Stream for RangeManipulateStream {
652    type Item = DataFusionResult<RecordBatch>;
653
654    fn poll_next(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Option<Self::Item>> {
655        let poll = loop {
656            match ready!(self.input.poll_next_unpin(cx)) {
657                Some(Ok(batch)) => {
658                    let timer = std::time::Instant::now();
659                    let result = self.manipulate(batch);
660                    if let Ok(None) = result {
661                        self.metric.elapsed_compute().add_elapsed(timer);
662                        continue;
663                    } else {
664                        self.num_series.add(1);
665                        self.metric.elapsed_compute().add_elapsed(timer);
666                        break Poll::Ready(result.transpose());
667                    }
668                }
669                None => {
670                    PROMQL_SERIES_COUNT.observe(self.num_series.value() as f64);
671                    break Poll::Ready(None);
672                }
673                Some(Err(e)) => break Poll::Ready(Some(Err(e))),
674            }
675        };
676        self.metric.record_poll(poll)
677    }
678}

680impl RangeManipulateStream {
681    // Prometheus: https://github.com/prometheus/prometheus/blob/e934d0f01158a1d55fa0ebb035346b195fcc1260/promql/engine.go#L1113-L1198
682    // But they are not exactly the same, because we don't eager-evaluate on the data in this plan.
683    // And the generated timestamp is not aligned to the step. It's expected to do later.
684    pub fn manipulate(&self, input: RecordBatch) -> DataFusionResult<Option<RecordBatch>> {
685        let mut other_columns = (0..input.columns().len()).collect::<HashSet<_>>();
686        // calculate the range
687        let (ranges, (start, end)) = self.calculate_range(&input)?;
688        // ignore this if all ranges are empty
689        if ranges.iter().all(|(_, len)| *len == 0) {
690            return Ok(None);
691        }
692
693        // transform columns
694        let mut new_columns = input.columns().to_vec();
695        for index in self.field_columns.iter() {
696            let _ = other_columns.remove(index);
697            let column = input.column(*index);
698            let new_column = Arc::new(
699                RangeArray::from_ranges(column.clone(), ranges.clone())
700                    .map_err(|e| ArrowError::InvalidArgumentError(e.to_string()))?
701                    .into_dict(),
702            );
703            new_columns[*index] = new_column;
704        }
705
706        // The timestamp range payload is always millisecond ABI. A column is reusable as-is
707        // only when its own data type already is that ABI: the output schema fixes the
708        // dictionary value type, so a millisecond column carrying a timezone must be
709        // converted like any other native unit.
710        let ts_column = input.column(self.time_index);
711        let timestamp_values: ArrayRef = if self.offset == 0
712            && matches!(
713                ts_column.data_type(),
714                DataType::Timestamp(TimeUnit::Millisecond, None)
715            ) {
716            ts_column.clone()
717        } else {
718            // Shift in wide native precision before truncating toward zero, preserving
719            // null validity.
720            let scale = nanoseconds_per_native_tick(self.time_unit);
721            let (timestamps, _) = timestamp_array_to_primitive(ts_column).ok_or_else(|| {
722                DataFusionError::Execution("Time index column is not a timestamp".into())
723            })?;
724            // Single pass over the timestamps: an i128 intermediate holds any native
725            // timestamp shift; only the final millisecond result must fit i64. Null-free
726            // batches (the dominant PromQL case) take arrow's infallible `unary` kernel,
727            // which LLVM can vectorize, and report overflow through a shared cell. But
728            // `unary` visits NULL slots' underlying values, so a NULL slot holding an
729            // arbitrary overflowing value would raise a false batch-wide error; nullable
730            // batches therefore take `try_unary`, which applies the closure to valid rows
731            // only.
732            let offset_ns = (self.offset as i128) * 1_000_000;
733            let timestamp_values = if timestamps.null_count() == 0 {
734                let overflow = std::cell::Cell::new(false);
735                let shifted = compute::unary::<Int64Type, _, Int64Type>(&timestamps, |v| {
736                    let shifted_ns = (v as i128) * scale + offset_ns;
737                    let shifted_ms = shifted_ns / 1_000_000;
738                    match i64::try_from(shifted_ms) {
739                        Ok(ms) => ms,
740                        Err(_) => {
741                            overflow.set(true);
742                            0
743                        }
744                    }
745                });
746                if overflow.get() {
747                    return Err(DataFusionError::Execution(
748                        "RangeManipulate timestamp payload overflow".into(),
749                    ));
750                }
751                shifted
752            } else {
753                compute::try_unary::<Int64Type, _, Int64Type>(&timestamps, |v| {
754                    let shifted_ns = (v as i128) * scale + offset_ns;
755                    i64::try_from(shifted_ns / 1_000_000).map_err(|_| {
756                        ArrowError::ComputeError(
757                            "RangeManipulate timestamp payload overflow".into(),
758                        )
759                    })
760                })?
761            };
762            Arc::new(TimestampMillisecondArray::new(
763                timestamp_values.values().clone(),
764                timestamp_values.nulls().cloned(),
765            ))
766        };
767        let ts_range_column = RangeArray::from_ranges(timestamp_values, ranges.clone())
768            .map_err(|e| ArrowError::InvalidArgumentError(e.to_string()))?
769            .into_dict();
770        new_columns.push(Arc::new(ts_range_column));
771
772        // truncate other columns
773        let take_indices = Int64Array::from(vec![0; ranges.len()]);
774        for index in other_columns.into_iter() {
775            new_columns[index] = compute::take(&input.column(index), &take_indices, None)?;
776        }
777        // replace timestamp with the aligned one
778        let new_time_index = if ranges.len() != self.aligned_ts_array.len() {
779            Self::build_aligned_ts_array(start, end, self.interval)
780        } else {
781            self.aligned_ts_array.clone()
782        };
783        new_columns[self.time_index] = new_time_index;
784
785        RecordBatch::try_new(self.output_schema.clone(), new_columns)
786            .map(Some)
787            .map_err(|e| DataFusionError::ArrowError(Box::new(e), None))
788    }

790    fn build_aligned_ts_array(start: i64, end: i64, interval: i64) -> ArrayRef {
791        Arc::new(TimestampMillisecondArray::from_iter_values(
792            (start..=end).step_by(interval as _),
793        ))
794    }

796    /// Return values:
797    /// - A vector of tuples where each tuple contains the start index and length of the range.
798    /// - A tuple of the actual start/end timestamp used to calculate the range.
799    #[allow(clippy::type_complexity)]
800    fn calculate_range(
801        &self,
802        input: &RecordBatch,
803    ) -> DataFusionResult<(Vec<(u32, u32)>, (i64, i64))> {
804        let ts_column = input.column(self.time_index);
805        let scale = nanoseconds_per_native_tick(self.time_unit);
806        let (timestamps, _) = timestamp_array_to_primitive(ts_column).ok_or_else(|| {
807            DataFusionError::Execution("Time index column is not a timestamp".into())
808        })?;
809        let timestamp_null_count = timestamps.null_count();
810        let timestamps = timestamps.values();
811        let timestamp =
812            |index| (timestamps[index] as i128) * scale + (self.offset as i128) * 1_000_000;
813        let len = timestamps.len();
814        if len == 0 {
815            return Ok((vec![], (self.start, self.end)));
816        }
817
818        // Shorten the range using wide arithmetic so timestamps near the native
819        // type limits retain every query-aligned evaluation point.
820        let query_start = self.start as i128;
821        let query_end = self.end as i128;
822        let interval = self.interval as i128;
823        let first_ts = timestamp(0).div_euclid(1_000_000);
824        // Preserve the query's alignment pattern when optimizing start time.
825        let remainder = (first_ts - query_start).rem_euclid(interval);
826        let first_ts_aligned = first_ts + (interval - remainder).rem_euclid(interval);
827        let last_ts_with_range =
828            (timestamp(len - 1) + (self.range as i128) * 1_000_000).div_euclid(1_000_000);
829        let remainder = (last_ts_with_range - query_start).rem_euclid(interval);
830        let last_ts_aligned = last_ts_with_range - remainder;
831        let start = query_start.max(first_ts_aligned);
832        let end = query_end.min(last_ts_aligned);
833        if start > end {
834            return Ok((vec![], (self.start, self.end)));
835        }
836        // The intersection is within the declared i64 query bounds.
837        let start = start as i64;
838        let end = end as i64;
839
840        // Range membership is decided on shifted native ticks, before the
841        // timestamp-range payload is converted to its millisecond ABI. This
842        // keeps sub-millisecond samples distinct in a range; equal millisecond
843        // payload values are not a reason to deduplicate input samples.
844        //
845        // Calculate for every aligned timestamp (`curr_ts`), assuming ordered
846        // timestamps: the scan compares shifted samples in `i64` nanoseconds
847        // whenever they fit the batch, and in exact `i128` nanoseconds otherwise.
848        let narrow = (timestamp_null_count == 0)
849            .then(|| {
850                narrow_scan(
851                    self.time_unit,
852                    self.offset,
853                    self.range,
854                    timestamps,
855                    start,
856                    end,
857                )
858            })
859            .flatten();
860        let ranges = match narrow {
861            Some(scan) => scan_ranges::<i64>(self.interval, timestamps, start, end, &scan),
862            None => {
863                let scan = ScanNanoseconds::<i128>::wide(self.time_unit, self.offset, self.range);
864                scan_ranges::<i128>(self.interval, timestamps, start, end, &scan)
865            }
866        };
867
868        Ok((ranges, (start, end)))
869    }

915/// Cursor scan over the aligned evaluation points, in `Ns` nanoseconds.
916fn scan_ranges<Ns: Nanoseconds>(
917    interval: Millisecond,
918    timestamps: &[i64],
919    start: Millisecond,
920    end: Millisecond,
921    scan: &ScanNanoseconds<Ns>,
922) -> Vec<(u32, u32)> {
923    let len = timestamps.len();
924    let mut ranges = Vec::new();
925    let mut left = 0usize;
926    let mut right = 0usize;
927    for curr_ts in (start..=end).step_by(interval as _) {
928        let curr_ts_ns = scan.instant(curr_ts);
929        let start_ts = scan.window_start(curr_ts_ns);
930
931        while left < len && scan.shift(timestamps[left]) <= start_ts {
932            left += 1;
933        }
934        right = right.max(left);
935        while right < len && scan.shift(timestamps[right]) <= curr_ts_ns {
936            right += 1;
937        }
938
939        if left == right {
940            ranges.push((0, 0));
941        } else {
942            ranges.push((left as _, (right - left) as _));
943        }
944    }
945    ranges
946}

948/// Cursor scan baselines in nanoseconds, shared with `InstantManipulate`.
949///
950/// The narrow `i64` baselines come from [`narrow_scan`], the exact ones from
951/// [`ScanNanoseconds::wide`].
952#[derive(Debug, Clone, Copy)]
953pub(super) struct ScanNanoseconds<Ns> {
954    /// Nanoseconds per timestamp tick.
955    scale: Ns,
956    /// The offset shift in nanoseconds.
957    offset_ns: Ns,
958    /// The range or lookback window in nanoseconds.
959    window_ns: Ns,
960}
961
962/// Nanoseconds carrier of the cursor scans: `i64` where [`narrow_scan`] proved that
963/// the batch fits, `i128` for the exact fallback.
964pub(super) trait Nanoseconds:
965    Copy + PartialOrd + From<i64> + Add<Output = Self> + Sub<Output = Self> + Mul<Output = Self>
966{
967}
968
969impl<T> Nanoseconds for T where
970    T: Copy + PartialOrd + From<i64> + Add<Output = Self> + Sub<Output = Self> + Mul<Output = Self>
971{
972}
973
974impl ScanNanoseconds<i128> {
975    /// The `i128` baselines, exact for every input.
976    pub(super) fn wide(time_unit: TimeUnit, offset: Millisecond, window: Millisecond) -> Self {
977        Self {
978            scale: nanoseconds_per_native_tick(time_unit),
979            offset_ns: i128::from(offset) * 1_000_000,
980            window_ns: i128::from(window) * 1_000_000,
981        }
982    }
983}
984
985impl<Ns: Nanoseconds> ScanNanoseconds<Ns> {
986    /// Shifted nanoseconds of an input timestamp.
987    pub(super) fn shift(&self, timestamp: i64) -> Ns {
988        Ns::from(timestamp) * self.scale + self.offset_ns
989    }
990
991    /// Nanoseconds of an evaluation instant given in milliseconds.
992    pub(super) fn instant(&self, millis: Millisecond) -> Ns {
993        Ns::from(millis) * Ns::from(1_000_000)
994    }
995
996    /// The exclusive lower boundary of the window ending at `instant`.
997    pub(super) fn window_start(&self, instant: Ns) -> Ns {
998        instant - self.window_ns
999    }
1000}

1002/// The narrow `i64` baselines of a sorted batch, or `None` when they would overflow.
1003pub(super) fn narrow_scan(
1004    time_unit: TimeUnit,
1005    offset: Millisecond,
1006    window: Millisecond,
1007    timestamps: &[i64],
1008    start: Millisecond,
1009    end: Millisecond,
1010) -> Option<ScanNanoseconds<i64>> {
1011    let scale = i64::try_from(nanoseconds_per_native_tick(time_unit)).ok()?;
1012    let offset_ns = offset.checked_mul(1_000_000)?;
1013    let window_ns = window.checked_mul(1_000_000)?;
1014    // Sorted samples, so the checked endpoints bound every interior shift and product.
1015    let shift = |timestamp: i64| timestamp.checked_mul(scale)?.checked_add(offset_ns);
1016    shift(*timestamps.first()?)?;
1017    shift(*timestamps.last()?)?;
1018    // The checked `[start, end]` bounds every instant and every `instant - window`.
1019    start.checked_mul(1_000_000)?.checked_sub(window_ns)?;
1020    end.checked_mul(1_000_000)?.checked_sub(window_ns)?;
1021    Some(ScanNanoseconds {
1022        scale,
1023        offset_ns,
1024        window_ns,
1025    })
1026}

1949    /// Calculates exact offsets directly from the range predicate used by PromQL.
1950    ///
1951    /// Input timestamps are sorted and non-null. Interval is positive, range is
1952    /// nonnegative, and test values are chosen to avoid `i64` overflow.
1953    fn calculate_range_oracle(
1954        timestamps: &[i64],
1955        start: i64,
1956        end: i64,
1957        interval: i64,
1958        range: i64,
1959    ) -> Vec<(u32, u32)> {
1960        // Match `calculate_range`'s explicit empty-input early return.
1961        if timestamps.is_empty() || start > end {
1962            return vec![];
1963        }
1964
1965        (start..=end)
1966            .step_by(interval as usize)
1967            .map(|curr| {
1968                let mut offset = None;
1969                let mut length = 0;
1970                for (index, &ts) in timestamps.iter().enumerate() {
1971                    if ts > curr - range && ts <= curr {
1972                        offset.get_or_insert(index);
1973                        length += 1;
1974                    }
1975                }
1976                (offset.unwrap_or(0) as u32, length)
1977            })
1978            .collect()
1979    }

2248    /// Exact expectations of the cursor scan, formulated outside the production loops.
2249    ///
2250    /// Every evaluation instant keeps the samples of `(instant - range, instant]` in
2251    /// shifted nanoseconds: the lower boundary is exclusive, the upper one inclusive, and
2252    /// native ticks below a millisecond stay distinct.
2253    fn scan_oracle(
2254        time_unit: TimeUnit,
2255        offset: Millisecond,
2256        range: Millisecond,
2257        timestamps: &[i64],
2258        start: Millisecond,
2259        end: Millisecond,
2260        interval: Millisecond,
2261    ) -> Vec<(u32, u32)> {

2264            .iter()

2266            .collect::<Vec<_>>();

2268            .step_by(interval as usize)

2375    /// Replays the cursor scan in checked `i64` arithmetic.
2376    ///
2377    /// Reports whether the narrow carrier overflows on an operation the scan evaluates,
2378    /// which is what makes the `i128` fallback unavoidable for an input.
2379    fn narrow_carrier_overflows(
2380        time_unit: TimeUnit,
2381        offset: Millisecond,
2382        range: Millisecond,
2383        timestamps: &[i64],
2384        start: Millisecond,
2385        end: Millisecond,
2386        interval: Millisecond,
2387    ) -> bool {
2388        let Ok(scale) = i64::try_from(nanoseconds_per_native_tick(time_unit)) else {
2389            return true;
2390        };
2391        let Some(offset_ns) = offset.checked_mul(1_000_000) else {
2392            return true;
2393        };
2394        let Some(window_ns) = range.checked_mul(1_000_000) else {
2395            return true;
2396        };
2397        let shift = |timestamp: i64| timestamp.checked_mul(scale)?.checked_add(offset_ns);
2398        let mut left = 0usize;
2399        let mut right = 0usize;
2400        for instant in (start..=end).step_by(interval as usize) {
2401            let Some(instant_ns) = instant.checked_mul(1_000_000) else {
2402                return true;
2403            };
2404            let Some(window_start) = instant_ns.checked_sub(window_ns) else {
2405                return true;
2406            };
2407            while left < timestamps.len() {
2408                let Some(shifted) = shift(timestamps[left]) else {
2409                    return true;
2410                };
2411                if shifted <= window_start {
2412                    left += 1;
2413                } else {
2414                    break;
2415                }
2416            }
2417            right = right.max(left);
2418            while right < timestamps.len() {
2419                let Some(shifted) = shift(timestamps[right]) else {
2420                    return true;
2421                };
2422                if shifted <= instant_ns {
2423                    right += 1;
2424                } else {
2425                    break;
2426                }
2427            }
2428        }
2429        false
2430    }

2432    /// The checked `i64` operation that makes [`narrow_scan`] reject an input, if any.
2433    ///
2434    /// Mirrors the rejection criteria of [`narrow_scan`] so the deterministic cases below
2435    /// can pin which boundary drives the `i128` fallback.
2436    fn narrow_carrier_rejection(
2437        time_unit: TimeUnit,
2438        offset: Millisecond,
2439        range: Millisecond,
2440        timestamps: &[i64],
2441        start: Millisecond,
2442        end: Millisecond,
2443    ) -> Option<&'static str> {
2444        let Ok(scale) = i64::try_from(nanoseconds_per_native_tick(time_unit)) else {
2445            return Some("nanoseconds per native tick do not fit i64");
2446        };
2447        let Some(offset_ns) = offset.checked_mul(1_000_000) else {
2448            return Some("offset in nanoseconds does not fit i64");
2449        };
2450        let Some(window_ns) = range.checked_mul(1_000_000) else {
2451            return Some("range in nanoseconds does not fit i64");
2452        };
2453        let shift = |timestamp: i64| timestamp.checked_mul(scale)?.checked_add(offset_ns);
2454        let (Some(first), Some(last)) = (timestamps.first(), timestamps.last()) else {
2455            return Some("empty batch has no endpoints");
2456        };
2457        if shift(*first).is_none() || shift(*last).is_none() {
2458            return Some("sample nanoseconds do not fit i64");
2459        }
2460        if start
2461            .checked_mul(1_000_000)
2462            .and_then(|nanoseconds| nanoseconds.checked_sub(window_ns))
2463            .is_none()
2464        {
2465            return Some("start bound in nanoseconds does not fit i64");
2466        }
2467        if end
2468            .checked_mul(1_000_000)
2469            .and_then(|nanoseconds| nanoseconds.checked_sub(window_ns))
2470            .is_none()
2471        {
2472            return Some("end bound in nanoseconds do not fit i64");
2473        }
2474        None
