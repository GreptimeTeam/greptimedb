15//! Arithmetic on owned native histograms, following Prometheus `FloatHistogram`
16//! (`model/histogram/float_histogram.go`): `Add`, `KahanAdd`, `Sub`, `Mul`, `Div`, `Compact`,
17//! `CopyToSchema`, `DetectReset` and `Equals`, including zero bucket reconciliation, resolution
18//! reduction and the intersection of mismatched custom bounds.
19//!
20//! Each side is a list of `(bucket index, count)` sorted by index. Go splices span layouts in
21//! place; the bucket indices and the order of additions to each bucket do not depend on the
22//! layout, so the results are the same.
23
24use datafusion::error::{DataFusionError, Result};
25use datatypes::series::histogram::{
26    BucketSideView, CounterResetHint, HistogramBatchBuilder, HistogramView,
27};
28
29use crate::functions::kahan_inc;
30use crate::histogram_eval::native::{CUSTOM_BUCKETS_SCHEMA, bound};

32type Buckets = Vec<(i32, f64)>;
33
34/// A native histogram with absolute bucket counts.
35#[derive(Debug, Clone)]
36pub(crate) struct FloatHistogram {
37    pub(crate) reset_hint: CounterResetHint,
38    /// The start timestamp of the sample this histogram was read from, in milliseconds.
39    ///
40    /// It is metadata of GreptimeDB's histogram row, not part of the histogram value that
41    /// Prometheus's `FloatHistogram` models. This owned form preserves it -- [`Self::try_from_view`]
42    /// reads it off the sample and [`Self::push_to`] writes it back -- while every operation that
43    /// derives a histogram from its input ([`Self::add`], [`Self::kahan_add`], [`Self::mul`],
44    /// [`Self::div`], [`Self::compact`], [`Self::copy_to_schema`], with the constructors beside
45    /// them) clears it. [`Self::equals`] does not compare it.
46    ///
47    /// Both halves are this module's own policy and say nothing about how Prometheus propagates
48    /// these timestamps: its `FloatHistogram` does not carry the field, while its rate and reset
49    /// evaluation receives start timestamps through a separate channel. This crate's native range
50    /// kernel does not read the field; the legacy extrapolation path does, to synthesize a zero
51    /// point. So a selection that keeps the field preserves what was stored, and does not by itself
52    /// make a native rate agree with the legacy one.
53    pub(crate) start_timestamp: Option<i64>,
54    pub(crate) schema: i32,
55    pub(crate) zero_threshold: f64,
56    pub(crate) zero_count: f64,
57    pub(crate) count: f64,
58    pub(crate) sum: f64,
59    pub(crate) positive: Buckets,
60    pub(crate) negative: Buckets,
61    pub(crate) custom_values: Vec<f64>,
62    /// The positive and negative spans of a histogram read from a view, kept until its buckets
63    /// change, because Go `Equals` compares span layouts. Otherwise consecutive indices form one
64    /// span, the layout Go `Compact` produces.
65    layout: Option<[Vec<(i32, i32)>; 2]>,
66}
67
68/// What an addition or subtraction reports besides its result.
69#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
70pub(crate) struct AddOutcome {
71    /// One side was a counter reset and the other was not.
72    pub(crate) counter_reset_collision: bool,
73    /// Custom bounds differed and the result uses their intersection.
74    pub(crate) custom_bounds_reconciled: bool,
75}

78    pub(crate) fn try_from_view(view: &HistogramView<'_>) -> Result<Self> {
79        Ok(Self {
80            reset_hint: view.reset_hint,
81            start_timestamp: view.start_timestamp,
82            schema: view.schema,
83            zero_threshold: view.zero_threshold,
84            zero_count: view.zero_count,
85            count: view.count,
86            sum: view.sum,
87            positive: side_from_view(view.positive)?,
88            negative: side_from_view(view.negative)?,
89            custom_values: view.custom_values.to_vec(),
90            layout: Some([view.positive, view.negative].map(|side| {
91                side.span_offsets
92                    .iter()
93                    .copied()
94                    .zip(side.span_lengths.iter().copied())
95                    .collect()
96            })),
97        })
98    }

100    /// Estimates the heap held by the bucket lists, custom bounds and preserved span layout.
101    ///
102    /// A histogram can own tens of kilobytes of buckets that its struct size does not include.
103    /// Capacity slack and small header overestimates are acceptable for window reservations.
104    pub(crate) fn heap_bytes(&self) -> usize {
105        fn buckets(side: &Buckets) -> usize {
106            side.capacity() * std::mem::size_of::<(i32, f64)>()
107        }
108        let layout = self.layout.as_ref().map_or(0, |sides| {
109            std::mem::size_of::<[Vec<(i32, i32)>; 2]>().saturating_add(
110                sides
111                    .iter()
112                    .map(|side| side.capacity() * std::mem::size_of::<(i32, i32)>())
113                    .fold(0, usize::saturating_add),
114            )
115        });
116        buckets(&self.positive)
117            .saturating_add(buckets(&self.negative))
118            .saturating_add(
119                self.custom_values
120                    .capacity()
121                    .saturating_mul(std::mem::size_of::<f64>()),
122            )
123            .saturating_add(layout)
124    }

126    /// The spans of each side: the layout a view was read with, or the compacted layout of a
127    /// histogram this crate derived.
128    ///
129    /// [`crate::histogram_eval::legacy_bridge`] reads them to build the legacy
130    /// `NativeHistogram`, whose spans are explicit where this type keeps them in the bucket
131    /// indices.
132    pub(crate) fn spans(&self) -> [Vec<(i32, i32)>; 2] {
133        self.layout
134            .clone()
135            .unwrap_or_else(|| [&self.positive, &self.negative].map(canonical_spans))
136    }

138    /// Appends the histogram to `builder`; consecutive bucket indices share a span.
139    pub(crate) fn push_to(&self, builder: &mut HistogramBatchBuilder) -> Result<u32> {
140        let [positive_spans, negative_spans] = self.spans();
141        let (positive_offsets, positive_lengths): (Vec<_>, Vec<_>) =
142            positive_spans.into_iter().unzip();
143        let (negative_offsets, negative_lengths): (Vec<_>, Vec<_>) =
144            negative_spans.into_iter().unzip();
145        let positive_counts: Vec<f64> = self.positive.iter().map(|&(_, count)| count).collect();
146        let negative_counts: Vec<f64> = self.negative.iter().map(|&(_, count)| count).collect();
147        builder
148            .push(HistogramView {
149                schema: self.schema,
150                zero_threshold: self.zero_threshold,
151                sum: self.sum,
152                count: self.count,
153                zero_count: self.zero_count,
154                reset_hint: self.reset_hint,
155                start_timestamp: self.start_timestamp,
156                custom_values: &self.custom_values,
157                positive: BucketSideView {
158                    span_offsets: &positive_offsets,
159                    span_lengths: &positive_lengths,
160                    buckets: &positive_counts,
161                },
162                negative: BucketSideView {
163                    span_offsets: &negative_offsets,
164                    span_lengths: &negative_lengths,
165                    buckets: &negative_counts,
166                },
167            })
168            .map_err(|error| DataFusionError::External(Box::new(error)))
169    }

193    /// Go `Mul`.
194    pub(crate) fn mul(&mut self, factor: f64) {
195        // A result this operation derived, so it carries no start timestamp of its own.
196        self.start_timestamp = None;
197        self.zero_count *= factor;
198        self.count *= factor;
199        self.sum *= factor;
200        for (_, count) in self.positive.iter_mut().chain(self.negative.iter_mut()) {
201            *count *= factor;
202        }
203        if factor < 0.0 {
204            self.reset_hint = CounterResetHint::Gauge;
205        }
206    }
207
208    /// Go `Div`; dividing by zero removes every bucket.
209    pub(crate) fn div(&mut self, scalar: f64) {
210        // A result this operation derived, so it carries no start timestamp of its own.
211        self.start_timestamp = None;
212        self.zero_count /= scalar;
213        self.count /= scalar;
214        self.sum /= scalar;
215        if scalar == 0.0 {
216            self.positive.clear();
217            self.negative.clear();
218            self.layout = None;
219            return;
220        }
221        for (_, count) in self.positive.iter_mut().chain(self.negative.iter_mut()) {
222            *count /= scalar;
223        }
224        if scalar < 0.0 {
225            self.reset_hint = CounterResetHint::Gauge;
226        }
227    }

229    /// Go `HasOverflow`: whether any count, the sum or a custom bound is infinite.
230    pub(crate) fn has_overflow(&self) -> bool {
231        [self.zero_count, self.count, self.sum]
232            .into_iter()
233            .chain(
234                self.positive
235                    .iter()
236                    .chain(&self.negative)
237                    .map(|&(_, count)| count),
238            )
239            .chain(self.custom_values.iter().copied())
240            .any(f64::is_infinite)
241    }

243    /// Go `Compact(0)`: drops empty buckets.
244    pub(crate) fn compact(&mut self) {
245        // A result this operation derived, so it carries no start timestamp of its own.
246        self.start_timestamp = None;
247        self.positive.retain(|&(_, count)| count != 0.0);
248        self.negative.retain(|&(_, count)| count != 0.0);
249        self.layout = None;
250    }

252    /// Go `CopyToSchema`; `target` must not exceed the schema.
253    pub(crate) fn copy_to_schema(&self, target: i32) -> Result<Self> {
254        if target == self.schema {
255            // The copy is a histogram this call produced, like the reduced one below: a start
256            // timestamp belongs to the sample that carried it, not to a value derived from it.
257            return Ok(Self {
258                start_timestamp: None,
259                ..self.clone()
260            });
261        }
262        if self.uses_custom_buckets() || target == CUSTOM_BUCKETS_SCHEMA || target > self.schema {
263            return Err(incompatible_schema());
264        }
265        Ok(Self {
266            reset_hint: CounterResetHint::Unknown,
267            start_timestamp: None,
268            schema: target,
269            zero_threshold: self.zero_threshold,
270            zero_count: self.zero_count,
271            count: self.count,
272            sum: self.sum,
273            positive: reduce_resolution(&self.positive, self.schema, target),
274            negative: reduce_resolution(&self.negative, self.schema, target),
275            custom_values: Vec::new(),
276            layout: None,
277        })
278    }

290    fn add_or_sub(&mut self, other: &Self, subtract: bool) -> Result<AddOutcome> {
291        check_schemas(self, other)?;
292        self.layout = None;
293        // A result this operation derived, so it carries no start timestamp of its own.
294        self.start_timestamp = None;
295        let mut outcome = AddOutcome {
296            counter_reset_collision: self.adjust_counter_reset(other),
297            custom_bounds_reconciled: false,
298        };
299        let sign = if subtract { -1.0 } else { 1.0 };
300        if !self.uses_custom_buckets() {
301            let (other_zero_count, _) = self.reconcile_zero_buckets(other, None)?;
302            if subtract {
303                self.zero_count -= other_zero_count;
304            } else {
305                self.zero_count += other_zero_count;
306            }
307        }
308        if subtract {
309            self.count -= other.count;
310            self.sum -= other.sum;
311        } else {
312            self.count += other.count;
313            self.sum += other.sum;
314        }
315
316        if self.uses_custom_buckets() {
317            if self.custom_values == other.custom_values {
318                add_buckets(self, &other.positive, sign, None, None, true)?;
319            } else {
320                outcome.custom_bounds_reconciled = true;
321                let bounds = intersect_custom_bounds(&self.custom_values, &other.custom_values);
322                (self.positive, _) = add_custom_buckets_with_mismatches(
323                    subtract,
324                    &self.positive,
325                    &self.custom_values,
326                    &other.positive,
327                    &other.custom_values,
328                    None,
329                    &bounds,
330                );
331                self.custom_values = bounds;
332            }
333            return Ok(outcome);
334        }
335
336        let reduced;
337        let (other_positive, other_negative) = if other.schema < self.schema {
338            self.positive = reduce_resolution(&self.positive, self.schema, other.schema);
339            self.negative = reduce_resolution(&self.negative, self.schema, other.schema);
340            self.schema = other.schema;
341            (&other.positive, &other.negative)
342        } else if other.schema > self.schema {
343            reduced = [&other.positive, &other.negative]
344                .map(|side| reduce_resolution(side, other.schema, self.schema));
345            (&reduced[0], &reduced[1])
346        } else {
347            (&other.positive, &other.negative)
348        };
349        add_buckets(self, other_positive, sign, None, None, true)?;
350        add_buckets(self, other_negative, sign, None, None, false)?;
351        Ok(outcome)
352    }

354    /// Go `KahanAdd`: adds `other` with compensation `c`, created when `None`, and returns the
355    /// updated compensation.
356    pub(crate) fn kahan_add(
357        &mut self,
358        other: &Self,
359        c: Option<Self>,
360    ) -> Result<(Self, AddOutcome)> {
361        check_schemas(self, other)?;
362        self.layout = None;
363        // A result this operation derived, so it carries no start timestamp of its own.
364        self.start_timestamp = None;
365        let mut outcome = AddOutcome {
366            counter_reset_collision: self.adjust_counter_reset(other),
367            custom_bounds_reconciled: false,
368        };
369        let mut c = c.unwrap_or_else(|| self.new_compensation());
370        if !self.uses_custom_buckets() {
371            let (other_zero_count, other_c_zero_count) =
372                self.reconcile_zero_buckets(other, Some(&mut c))?;
373            (self.zero_count, c.zero_count) =
374                kahan_inc(other_zero_count, self.zero_count, c.zero_count);
375            (self.zero_count, c.zero_count) =
376                kahan_inc(other_c_zero_count, self.zero_count, c.zero_count);
377        }
378        (self.count, c.count) = kahan_inc(other.count, self.count, c.count);
379        (self.sum, c.sum) = kahan_inc(other.sum, self.sum, c.sum);
380
381        if self.uses_custom_buckets() {
382            if self.custom_values == other.custom_values {
383                add_buckets(self, &other.positive, 1.0, Some(&mut c), None, true)?;
384            } else {
385                outcome.custom_bounds_reconciled = true;
386                let bounds = intersect_custom_bounds(&self.custom_values, &other.custom_values);
387                let compensation: Vec<f64> = c.positive.iter().map(|&(_, c)| c).collect();
388                let (positive, positive_c) = add_custom_buckets_with_mismatches(
389                    false,
390                    &self.positive,
391                    &self.custom_values,
392                    &other.positive,
393                    &other.custom_values,
394                    Some(&compensation),
395                    &bounds,
396                );
397                self.positive = positive;
398                c.positive = positive_c;
399                self.custom_values = bounds.clone();
400                c.custom_values = bounds;
401            }
402            return Ok((c, outcome));
403        }
404
405        let mut other_positive = None;
406        let mut other_negative = None;
407        if other.schema < self.schema {
408            (self.positive, c.positive) =
409                kahan_reduce_resolution(&self.positive, &c.positive, self.schema, other.schema);
410            (self.negative, c.negative) =
411                kahan_reduce_resolution(&self.negative, &c.negative, self.schema, other.schema);
412            self.schema = other.schema;
413        } else if other.schema > self.schema {
414            if !other.positive.is_empty() {
415                let zeros = zero_compensation(&other.positive);
416                other_positive = Some(kahan_reduce_resolution(
417                    &other.positive,
418                    &zeros,
419                    other.schema,
420                    self.schema,
421                ));
422            }
423            if !other.negative.is_empty() {
424                let zeros = zero_compensation(&other.negative);
425                other_negative = Some(kahan_reduce_resolution(
426                    &other.negative,
427                    &zeros,
428                    other.schema,
429                    self.schema,
430                ));
431            }
432        }
433        let (positive, positive_c) = match &other_positive {
434            Some((buckets, c)) => (buckets, Some(c)),
435            None => (&other.positive, None),
436        };
437        add_buckets(self, positive, 1.0, Some(&mut c), positive_c, true)?;
438        let (negative, negative_c) = match &other_negative {
439            Some((buckets, c)) => (buckets, Some(c)),
440            None => (&other.negative, None),
441        };
442        add_buckets(self, negative, 1.0, Some(&mut c), negative_c, false)?;
443        c.schema = self.schema;
444        c.zero_threshold = self.zero_threshold;
445        Ok((c, outcome))
446    }

473    /// Go `adjustCounterReset`.
474    fn adjust_counter_reset(&mut self, other: &Self) -> bool {
475        use CounterResetHint::{Gauge, Unknown};
476        match (self.reset_hint, other.reset_hint) {
477            (mine, theirs) if mine == theirs => false,
478            (Gauge, _) => false,
479            (_, Gauge) => {
480                self.reset_hint = Gauge;
481                false
482            }
483            (Unknown, _) => false,
484            (_, Unknown) => {
485                self.reset_hint = Unknown;
486                false
487            }
488            _ => {
489                self.reset_hint = Unknown;
490                true
491            }
492        }
493    }

495    /// Go `zeroCountForLargerThreshold`: the zero count after widening the zero bucket to at
496    /// least `larger_threshold`, the threshold actually needed so no bucket straddles it, and
497    /// the Kahan compensation of the count.
498    fn zero_count_for_larger_threshold(
499        &self,
500        mut larger_threshold: f64,
501        c: Option<&Self>,
502    ) -> Result<(f64, f64, f64)> {
503        let mut c_zero_count = c.map_or(0.0, |c| c.zero_count);
504        if larger_threshold == self.zero_threshold {
505            return Ok((self.zero_count, larger_threshold, c_zero_count));
506        }
507        'outer: loop {
508            // Go keeps the compensation accumulated by an abandoned pass.
509            let mut zero_count = self.zero_count;
510            for (position, &(idx, count)) in self.positive.iter().enumerate() {
511                let (lower, upper) = (self.bound(idx - 1)?, self.bound(idx)?);
512                if lower >= larger_threshold {
513                    break;
514                }
515                (zero_count, c_zero_count) = kahan_inc(count, zero_count, c_zero_count);
516                if let Some(c) = c {
517                    (zero_count, c_zero_count) =
518                        kahan_inc(c.positive[position].1, zero_count, c_zero_count);
519                }
520                if upper > larger_threshold {
521                    if count != 0.0 {
522                        larger_threshold = upper;
523                    }
524                    break;
525                }
526            }
527            for (position, &(idx, count)) in self.negative.iter().enumerate() {
528                let (lower, upper) = (-self.bound(idx)?, -self.bound(idx - 1)?);
529                if upper <= -larger_threshold {
530                    break;
531                }
532                (zero_count, c_zero_count) = kahan_inc(count, zero_count, c_zero_count);
533                if let Some(c) = c {
534                    (zero_count, c_zero_count) =
535                        kahan_inc(c.negative[position].1, zero_count, c_zero_count);
536                }
537                if lower < -larger_threshold {
538                    if count != 0.0 {
539                        larger_threshold = -lower;
540                        continue 'outer;
541                    }
542                    break;
543                }
544            }
545            return Ok((zero_count, larger_threshold, c_zero_count));
546        }
547    }

549    /// Go `trimBucketsInZeroBucket`: empties the buckets inside the zero bucket, then compacts.
550    fn trim_buckets_in_zero_bucket(&mut self, c: Option<&mut Self>) -> Result<()> {
551        let mut trimmed_positive = 0;
552        for &(idx, _) in &self.positive {
553            if self.bound(idx - 1)? >= self.zero_threshold {
554                break;
555            }
556            trimmed_positive += 1;
557        }
558        let mut trimmed_negative = 0;
559        for &(idx, _) in &self.negative {
560            if -self.bound(idx - 1)? <= -self.zero_threshold {
561                break;
562            }
563            trimmed_negative += 1;
564        }
565        for bucket in &mut self.positive[..trimmed_positive] {
566            bucket.1 = 0.0;
567        }
568        for bucket in &mut self.negative[..trimmed_negative] {
569            bucket.1 = 0.0;
570        }
571        match c {
572            Some(c) => {
573                for bucket in &mut c.positive[..trimmed_positive] {
574                    bucket.1 = 0.0;
575                }
576                for bucket in &mut c.negative[..trimmed_negative] {
577                    bucket.1 = 0.0;
578                }
579                compact_with(&mut self.positive, &mut c.positive);
580                compact_with(&mut self.negative, &mut c.negative);
581            }
582            None => self.compact(),
583        }
584        Ok(())
585    }

587    /// Go `reconcileZeroBuckets`: widens whichever zero bucket is narrower until both match, and
588    /// returns the zero count of `other` for that threshold with its compensation.
589    fn reconcile_zero_buckets(
590        &mut self,
591        other: &Self,
592        mut c: Option<&mut Self>,
593    ) -> Result<(f64, f64)> {
594        let mut other_zero_count = other.zero_count;
595        let mut other_zero_threshold = other.zero_threshold;
596        let mut other_c_zero_count = 0.0;
597        while other_zero_threshold != self.zero_threshold {
598            if self.zero_threshold > other_zero_threshold {
599                (other_zero_count, other_zero_threshold, other_c_zero_count) =
600                    other.zero_count_for_larger_threshold(self.zero_threshold, None)?;
601            }
602            if other_zero_threshold > self.zero_threshold {
603                let (zero_count, threshold, c_zero_count) =
604                    self.zero_count_for_larger_threshold(other_zero_threshold, c.as_deref())?;
605                self.zero_count = zero_count;
606                self.zero_threshold = threshold;
607                if let Some(c) = c.as_deref_mut() {
608                    c.zero_count = c_zero_count;
609                }
610                self.trim_buckets_in_zero_bucket(c.as_deref_mut())?;
611            }
612        }
613        Ok((other_zero_count, other_c_zero_count))
614    }

616    fn bound(&self, idx: i32) -> Result<f64> {
617        bound(idx, self.schema, &self.custom_values)
618    }

620    /// The reset a pair of start timestamps witnesses, given the window timestamps they are
621    /// compared against.
622    ///
623    /// Separate from the payload so a caller whose window timestamps have been moved by an
624    /// `offset` can shift the stored timestamps onto the same axis without cloning either
625    /// histogram: `range_eval` keeps its samples at `timestamp + offset_ms` while the stored start
626    /// timestamp stays where the sender put it, and comparing across those two axes reports a
627    /// reset for a pair that never restarted.
628    pub(crate) fn start_timestamp_reset(
629        current_start: i64,
630        previous_start: i64,
631        previous_ts: i64,
632        current_ts: i64,
633    ) -> bool {
634        if current_start == 0 || current_start >= current_ts || current_start < previous_ts {
635            return false;
636        }
637        if current_start > previous_ts {
638            return true;
639        }
640
641        // The previous sample's own start timestamp witnesses a restart between it and the one
642        // before, which the current sample inherits. Requiring it to be at or before `previous_ts`
643        // keeps a start later than its own sample — which no sender can have observed — from being
644        // read as one.
645        //
646        // What this does NOT implement is the rule that two adjacent samples sharing the same
647        // non-zero start timestamp never restarted between them: here only the previous sample's
648        // start is compared with its own time, so equal starts are not excluded. The existing
649        // planner's `detect_start_timestamp_reset` has the same gap, so the two routes agree, which
650        // is what this route owes it; closing the gap is a change to both.
651        previous_start <= previous_ts && previous_start != 0 && previous_start != previous_ts
652    }
653
654    /// Detects a counter reset for a window whose timestamps an `offset` has already moved.
655    ///
656    /// `range_eval` hands its windows at `timestamp + offset_ms` while a histogram's start
657    /// timestamp stays on the axis the sender wrote it on, so one of the two has to move. The
658    /// window timestamps are moved back here rather than the start timestamps forward, because a
659    /// start of `0` means "the sender stated none" and moving it forward can cancel the offset
660    /// into a real instant: a start of 2500ms under `offset -2.5s` would become that `0` and stop
661    /// being read as a reset. Restoring the sample times keeps the absence meaning what it means.
662    ///
663    /// The subtraction wraps, which is what Go's `int64` arithmetic does and what
664    /// [`crate::range_eval::function::extrapolation_factor`] relies on for the same reason: real
665    /// timestamps are nowhere near the range where the two disagree, and wrapping cannot turn a
666    /// missing start timestamp into a present one.
667    pub(crate) fn detect_counter_reset_shifted(
668        &self,
669        previous: &Self,
670        previous_ts: i64,
671        current_ts: i64,
672        offset_ms: i64,
673    ) -> bool {
674        Self::start_timestamp_reset(
675            self.start_timestamp.unwrap_or_default(),
676            previous.start_timestamp.unwrap_or_default(),
677            previous_ts.wrapping_sub(offset_ms),
678            current_ts.wrapping_sub(offset_ms),
679        ) || self.detect_reset(previous).unwrap_or(false)
680    }
681
682    pub(crate) fn detect_reset(&self, previous: &Self) -> Result<bool> {
683        match self.reset_hint {
684            CounterResetHint::CounterReset => return Ok(true),
685            CounterResetHint::NotCounterReset => return Ok(false),
686            _ => {}
687        }
688        if self.count < previous.count {
689            return Ok(true);
690        }
691        if self.uses_custom_buckets() {
692            if !previous.uses_custom_buckets() {
693                return Ok(true);
694            }
695            if self.custom_values != previous.custom_values {
696                return self.detect_reset_with_mismatched_custom_bounds(previous);
697            }
698        }
699        if self.schema > previous.schema || self.zero_threshold < previous.zero_threshold {
700            return Ok(true);
701        }
702        let (previous_zero_count, threshold, _) =
703            previous.zero_count_for_larger_threshold(self.zero_threshold, None)?;
704        if threshold != self.zero_threshold || self.zero_count < previous_zero_count {
705            return Ok(true);
706        }
707        for (current, earlier) in [
708            (&self.positive, &previous.positive),
709            (&self.negative, &previous.negative),
710        ] {
711            let current = iterate(current, self.schema, self.schema, self.zero_threshold)?;
712            let earlier = iterate(earlier, previous.schema, self.schema, self.zero_threshold)?;
713            if side_reset(&current, &earlier) {
714                return Ok(true);
715            }
716        }
717        Ok(false)
718    }

720    /// Go `detectResetWithMismatchedCustomBounds`: compares the counts rolled up to each bound
721    /// both histograms share.
722    fn detect_reset_with_mismatched_custom_bounds(&self, previous: &Self) -> Result<bool> {
723        let current_bounds = &self.custom_values;
724        let previous_bounds = &previous.custom_values;
725        let mut current = RollUp::new(self);
726        let mut earlier = RollUp::new(previous);
727        let (mut i, mut j) = (0, 0);
728        while i <= current_bounds.len() && j <= previous_bounds.len() {
729            let current_bound = current_bounds.get(i).copied().unwrap_or(f64::INFINITY);
730            let previous_bound = previous_bounds.get(j).copied().unwrap_or(f64::INFINITY);
731            if current_bound == previous_bound {
732                let current_sum = current.sum_up_to(current_bound)?;
733                let previous_sum = earlier.sum_up_to(current_bound)?;
734                if current_sum < previous_sum {
735                    return Ok(true);
736                }
737                i += 1;
738                j += 1;
739            } else if current_bound < previous_bound {
740                i += 1;
741            } else {
742                j += 1;
743            }
744        }
745        Ok(false)
746    }

748    /// Go `Equals`: bit equality of counts and of the bucket layout.
749    pub(crate) fn equals(&self, other: &Self) -> bool {
750        let same_side = |a: &Buckets, b: &Buckets| {
751            a.len() == b.len()
752                && a.iter()
753                    .zip(b)
754                    .all(|(a, b)| a.0 == b.0 && a.1.to_bits() == b.1.to_bits())
755        };
756        self.schema == other.schema
757            && self.count.to_bits() == other.count.to_bits()
758            && self.sum.to_bits() == other.sum.to_bits()
759            && (!self.uses_custom_buckets() || self.custom_values == other.custom_values)
760            && self.zero_threshold == other.zero_threshold
761            && self.zero_count.to_bits() == other.zero_count.to_bits()
762            && same_side(&self.negative, &other.negative)
763            && same_side(&self.positive, &other.positive)
764            && {
765                let ([positive, negative], [other_positive, other_negative]) =
766                    (self.spans(), other.spans());
767                spans_match(&negative, &other_negative) && spans_match(&positive, &other_positive)
768            }
769    }
770}

772/// The rolling sum of custom buckets up to a bound, as Go's `rollupSumForBound` closure keeps
773/// it across calls.
774struct RollUp<'a> {
775    histogram: &'a FloatHistogram,
776    next: usize,
777    started: bool,
778    has_more: bool,
779}
780
781impl<'a> RollUp<'a> {
782    fn new(histogram: &'a FloatHistogram) -> Self {
783        Self {
784            histogram,
785            next: 0,
786            started: false,
787            has_more: false,
788        }
789    }
790
791    fn sum_up_to(&mut self, bound: f64) -> Result<f64> {
792        if self.started && !self.has_more {
793            return Ok(0.0);
794        }
795        self.started = true;
796        let mut sum = 0.0;
797        while let Some(&(idx, count)) = self.histogram.positive.get(self.next) {
798            if self.histogram.bound(idx)? > bound {
799                self.has_more = true;
800                return Ok(sum);
801            }
802            sum += count;
803            self.next += 1;
804        }
805        self.has_more = false;
806        Ok(sum)
807    }
808}

823/// Go `floatBucketIterator`: the buckets of `side` merged down to `target`, without the leading
824/// ones whose upper bound is at most `threshold`.
825fn iterate(side: &Buckets, schema: i32, target: i32, threshold: f64) -> Result<Buckets> {
826    let mut out: Buckets = Vec::with_capacity(side.len());
827    for &(idx, count) in side {
828        let target_idx = target_index(idx, schema, target);
829        match out.last_mut() {
830            Some(last) if schema != target && last.0 == target_idx => last.1 += count,
831            _ => out.push((target_idx, count)),
832        }
833    }
834    // Go starts skipping only for a nonzero start value.
835    if target != CUSTOM_BUCKETS_SCHEMA && threshold != 0.0 {
836        let mut skipped = 0;
837        for &(idx, _) in &out {
838            if bound(idx, target, &[])? > threshold {
839                break;
840            }
841            skipped += 1;
842        }
843        out.drain(..skipped);
844    }
845    Ok(out)
846}

848/// Go `targetIdx`.
849fn target_index(idx: i32, schema: i32, target: i32) -> i32 {
850    if schema == target {
851        return idx;
852    }
853    ((idx - 1) >> (schema - target)) + 1
854}

857fn reduce_resolution(side: &Buckets, schema: i32, target: i32) -> Buckets {
858    let mut out: Buckets = Vec::with_capacity(side.len());

863            _ => out.push((target_idx, count)),

865    }

869/// Go `kahanReduceResolution`.
870fn kahan_reduce_resolution(
871    side: &Buckets,
872    compensation: &Buckets,
873    schema: i32,
874    target: i32,
875) -> (Buckets, Buckets) {
876    let mut out: Buckets = Vec::with_capacity(side.len());
877    let mut out_c: Buckets = Vec::with_capacity(side.len());
878    for (&(idx, count), &(_, c)) in side.iter().zip(compensation) {
879        let target_idx = target_index(idx, schema, target);
880        match (out.last_mut(), out_c.last_mut()) {
881            (Some(last), Some(last_c)) if last.0 == target_idx => {
882                (last.1, last_c.1) = kahan_inc(count, last.1, last_c.1);
883                (last.1, last_c.1) = kahan_inc(c, last.1, last_c.1);
884            }
885            _ => {
886                out.push((target_idx, count));
887                out_c.push((target_idx, c));
888            }
889        }
890    }
891    (out, out_c)
892}

898/// Go `kahanCompact(0)`: drops the buckets whose count and compensation are both zero.
899fn compact_with(side: &mut Buckets, compensation: &mut Buckets) {
900    let keep: Vec<bool> = side
901        .iter()

904        .collect();

906    side.retain(|_| *flags.next().unwrap());
907    let mut flags = keep.iter();
908    compensation.retain(|_| *flags.next().unwrap());

911/// Go `addBuckets` and `kahanAddBuckets`: adds `sign * other` to one side of `h`, inserting the
912/// buckets `h` lacks. Leading buckets of `other` inside the zero bucket are skipped.
913fn add_buckets(
914    h: &mut FloatHistogram,
915    other: &Buckets,
916    sign: f64,
917    c: Option<&mut FloatHistogram>,
918    other_c: Option<&Buckets>,
919    positive: bool,
920) -> Result<()> {
921    let exponential = !h.uses_custom_buckets();
922    let threshold = h.zero_threshold;
923    let schema = h.schema;
924    let side = if positive {
925        &mut h.positive
926    } else {
927        &mut h.negative
928    };
929    let mut side_c = c.map(|c| {
930        if positive {
931            &mut c.positive
932        } else {
933            &mut c.negative
934        }
935    });
936    let mut below_threshold = true;
937    let mut position = 0;
938    for (k, &(idx, count)) in other.iter().enumerate() {
939        if below_threshold && exponential && bound(idx, schema, &[])? <= threshold {
940            continue;
941        }
942        below_threshold = false;
943        let value = count * sign;
944        let value_c = other_c.map_or(0.0, |other_c| other_c[k].1) * sign;
945        while position < side.len() && side[position].0 < idx {
946            position += 1;
947        }
948        if position < side.len() && side[position].0 == idx {
949            match side_c.as_deref_mut() {
950                Some(side_c) => {
951                    let (sum, comp) = (&mut side[position].1, &mut side_c[position].1);
952                    (*sum, *comp) = kahan_inc(value, *sum, *comp);
953                    if value_c != 0.0 {
954                        (*sum, *comp) = kahan_inc(value_c, *sum, *comp);
955                    }
956                }
957                None => side[position].1 += value,
958            }
959        } else {
960            side.insert(position, (idx, value));
961            if let Some(side_c) = side_c.as_deref_mut() {
962                side_c.insert(position, (idx, value_c));
963            }
964        }
965    }
966    Ok(())
967}

969/// Go `intersectCustomBucketBounds`.
970fn intersect_custom_bounds(a: &[f64], b: &[f64]) -> Vec<f64> {
971    let mut out = Vec::new();
972    let (mut i, mut j) = (0, 0);
973    while i < a.len() && j < b.len() {
974        if a[i] == b[j] {
975            out.push(a[i]);
976            i += 1;
977            j += 1;
978        } else if a[i] < b[j] {
979            i += 1;
980        } else {
981            j += 1;
982        }
983    }
984    out
985}

987/// Go `addCustomBucketsWithMismatches`: maps both sides onto the intersected bounds, adding
988/// with Kahan summation, and drops buckets left empty. Returns the buckets and their
989/// compensation.
990fn add_custom_buckets_with_mismatches(
991    subtract: bool,
992    a: &Buckets,
993    a_bounds: &[f64],
994    b: &Buckets,
995    b_bounds: &[f64],
996    a_compensation: Option<&[f64]>,
997    bounds: &[f64],
998) -> (Buckets, Buckets) {
999    let mut target = vec![0.0; bounds.len() + 1];
1000    let mut target_c = vec![0.0; bounds.len() + 1];
1001    let mut map = |side: &Buckets, side_bounds: &[f64], negate: bool, c: Option<&[f64]>| {
1002        let mut intersect_idx = 0;
1003        for (k, &(idx, value)) in side.iter().enumerate() {
1004            let mut target_idx = target.len() - 1;
1005            if let Some(&source_bound) = usize::try_from(idx).ok().and_then(|i| side_bounds.get(i))
1006            {
1007                while intersect_idx < bounds.len() {
1008                    if bounds[intersect_idx] >= source_bound {
1009                        target_idx = intersect_idx;
1010                        break;
1011                    }
1012                    intersect_idx += 1;
1013                }
1014            }
1015            let (sum, comp) = (&mut target[target_idx], &mut target_c[target_idx]);
1016            if negate {
1017                (*sum, *comp) = kahan_inc(-value, *sum, *comp);
1018            } else {
1019                (*sum, *comp) = kahan_inc(value, *sum, *comp);
1020                if let Some(c) = c {
1021                    (*sum, *comp) = kahan_inc(c[k], *sum, *comp);
1022                }
1023            }
1024        }
1025    };
1026    map(a, a_bounds, false, a_compensation);
1027    map(b, b_bounds, subtract, None);
1028    let mut out = Vec::new();
1029    let mut out_c = Vec::new();
1030    for (i, (&value, &c)) in target.iter().zip(&target_c).enumerate() {
1031        if value == 0.0 && c == 0.0 {
1032            continue;
1033        }
1034        out.push((i as i32, value));
1035        out_c.push((i as i32, c));
1036    }
1037    (out, out_c)
1038}

1040/// Go `detectReset` over two sides with matching indices.
1041fn side_reset(current: &Buckets, previous: &Buckets) -> bool {
1042    let mut current = current.iter().peekable();
1043    for &(idx, count) in previous {
1044        while current.next_if(|bucket| bucket.0 < idx).is_some() {}
1045        match current.peek() {
1046            Some(&&(current_idx, current_count)) if current_idx == idx => {
1047                if current_count < count {
1048                    return true;
1049                }
1050            }
1051            _ if count != 0.0 => return true,
1052            _ => {}
1053        }
1054    }
1055    false
1056}

1058fn side_from_view(side: BucketSideView<'_>) -> Result<Buckets> {
1059    let invalid = || DataFusionError::Execution("native histogram has an invalid layout".into());
1060    if side.span_offsets.len() != side.span_lengths.len() {
1061        return Err(invalid());
1062    }
1063    let mut out = Vec::with_capacity(side.buckets.len());
1064    let mut counts = side.buckets.iter();
1065    let mut idx = 0i64;
1066    for (&offset, &length) in side.span_offsets.iter().zip(side.span_lengths) {
1067        if length < 0 {
1068            return Err(invalid());
1069        }
1070        idx += i64::from(offset);
1071        for _ in 0..length {
1072            let count = *counts.next().ok_or_else(invalid)?;
1073            out.push((i32::try_from(idx).map_err(|_| invalid())?, count));
1074            idx += 1;
1075        }
1076    }
1077    if counts.next().is_some() {
1078        return Err(invalid());
1079    }
1080    Ok(out)
1081}

1083/// Spans with one span per run of consecutive indices.
1084fn canonical_spans(side: &Buckets) -> Vec<(i32, i32)> {
1085    let mut spans: Vec<(i32, i32)> = Vec::new();
1086    let mut next = None;
1087    for &(idx, _) in side {
1088        match next {
1089            Some(expected) if expected == idx => spans.last_mut().unwrap().1 += 1,
1090            _ => spans.push((next.map_or(idx, |expected| idx - expected), 1)),
1091        }
1092        next = Some(idx + 1);
1093    }
1094    spans
1095}

1097/// Go `spansMatch`: equal layouts once zero-length spans are folded into the next span.
1098fn spans_match(a: &[(i32, i32)], b: &[(i32, i32)]) -> bool {
1099    fn fold(spans: &[(i32, i32)]) -> Vec<(i32, i32)> {
1100        let mut out = Vec::new();
1101        let mut offset = 0;
1102        for &(span_offset, length) in spans {
1103            offset += span_offset;
1104            if length != 0 {
1105                out.push((offset, length));
1106                offset = 0;
1107            }
1108        }
1109        out
1110    }
1111    fold(a) == fold(b)
1112}
