33/// A native histogram aggregation.
34#[derive(Debug, Clone, Copy, PartialEq, Eq)]
35pub enum HistogramAggregate {
36    Sum,
37    Avg,
38}
39
40/// Aggregates the native histograms of one group at every point, adding them in the order of
41/// `group`. A point yields no histogram when the group also has a float sample there, or when
42/// its histograms mix exponential and custom buckets. Float-only points are left to the float
43/// aggregation. The result carries `metric`.
44pub fn aggregate_histograms(
45    op: HistogramAggregate,
46    schema: &Arc<SeriesSchema>,
47    group: &[Series],
48    metric: SeriesLabels,
49) -> Result<Option<Series>> {
50    let grid = schema.eval_grid.clone().ok_or_else(|| {
51        DataFusionError::Execution("histogram aggregation requires an evaluation grid".into())
52    })?;
53    for series in group {
54        grid_axis(series)?;
55    }
56
57    let fields = schema.fields.len();
58    // Only the positions some member carries can hold a result, and the union visits each of
59    // them once: a hole between two members is never scanned, so nothing densifies here.
60    let mut builders: Vec<HistogramBatchBuilder> =
61        (0..fields).map(|_| HistogramBatchBuilder::new()).collect();
62    let mut indices: Vec<Vec<Option<u32>>> = vec![Vec::new(); fields];
63    // The maximal runs of positions that carried a result; one row is written per run row, so a
64    // result over few scattered positions costs few rows.
65    let mut spans: Vec<GridSpan> = Vec::new();
66    let mut row: Vec<Option<FloatHistogram>> = (0..fields).map(|_| None).collect();
67    // One forward cursor per member and field: the merged positions only increase, so a member's
68    // rows do too, and reading them costs one pass over the field instead of a chunk scan per row.
69    // A member that skips a position is asked for nothing, so its cursor stays where it was.
70    let mut cursors: Vec<Vec<RowCursor<'_>>> = group
71        .iter()
72        .map(|series| series.fields.iter().map(|field| field.cursor()).collect())
73        .collect();
74    let mut rows: Vec<Option<usize>> = vec![None; group.len()];
75    for point in union_positions(group.iter().map(|series| &series.time)) {
76        for (member, series) in group.iter().enumerate() {
77            rows[member] = series.time.row_of(point);
78        }
79        for (field, value) in row.iter_mut().enumerate() {
80            let samples = group.iter().enumerate().filter_map(|(member, _)| {
81                rows[member].map(|row| {
82                    cursors[member][field]
83                        .at(row)
84                        .expect("a row never goes back")
85                })
86            });
87            *value = aggregate_point(op, samples)?;
88        }
89        if row.iter().all(Option::is_none) {
90            continue;
91        }
92        // A gap in the results ends a run, and a row always belongs to the run it opens.
93        match spans.last_mut() {
94            Some(span) if span.start() + span.len() == point => {
95                *span = GridSpan::new(span.start(), span.len() + 1);
96            }
97            _ => spans.push(GridSpan::new(point, 1)),
98        }
99        for ((value, builder), indices) in row.iter().zip(&mut builders).zip(&mut indices) {
100            indices.push(match value {
101                Some(h) => Some(h.push_to(builder)?),
102                None => None,
103            });
104        }
105    }
106    if spans.is_empty() {
107        return Ok(None);
108    }
109    let mut samples = Vec::with_capacity(fields);
110    for (builder, indices) in builders.into_iter().zip(indices) {
111        let batch = builder.finish().map_err(external)?;
112        samples.push(
113            Samples::try_from_histogram(UInt32Array::from(indices), batch).map_err(external)?,
114        );
115    }
116    // The result is addressed by the runs that carry a histogram, not by the span between the
117    // first and the last one: the nulls padding them stay inside a run, never between two.
118    let time = TimeAxis::from_grid_spans(grid, spans).map_err(external)?;
119    debug_assert_eq!(time.len(), samples[0].len());
120    Series::try_new(
121        schema,
122        metric,
123        time,
124        SeriesFields::try_from_vec(samples).map_err(external)?,
125    )
126    .map(Some)
127    .map_err(external)
128}

130/// One point of one field.
131fn aggregate_point<'a>(
132    op: HistogramAggregate,
133    samples: impl Iterator<Item = SampleRef<'a>>,
134) -> Result<Option<FloatHistogram>> {
135    let mut has_float = false;
136    let mut histograms = samples.filter_map(|sample| match sample {
137        SampleRef::Missing => None,
138        SampleRef::Float(_) => {
139            has_float = true;
140            None
141        }
142        SampleRef::Histogram(view) => Some(FloatHistogram::try_from_view(&view)),
143    });
144    let (value, _) = sum_or_avg(op, histograms.by_ref())?;
145    // A fold stopped by mixed bucket kinds still reports conversion errors of later samples.
146    for histogram in histograms {
147        histogram?;
148    }
149    let Some(mut value) = value.filter(|_| !has_float) else {
150        return Ok(None);
151    };
152    value.compact();
153    Ok(Some(value))
154}

156/// The Kahan sum or the mean of `histograms` in order, without compaction; `None` when there is
157/// none or they mix exponential and custom buckets. Prometheus `sum_over_time` and
158/// `avg_over_time` run the same steps as the aggregations. Histograms are consumed one at a
159/// time, so the first error ends the fold. The result is derived, so it carries no start
160/// timestamp.
161///
162/// The flag reports whether any add had to reconcile two different custom-bucket sets, which is
163/// what Prometheus annotates with an info. It is carried out of the fold rather than recomputed:
164/// the fold already calls [`FloatHistogram::kahan_add`] and receives that outcome.
165pub(crate) fn sum_or_avg(
166    op: HistogramAggregate,
167    histograms: impl IntoIterator<Item = Result<FloatHistogram>>,
168) -> Result<(Option<FloatHistogram>, bool)> {
169    let mut value: Option<FloatHistogram> = None;
170    let mut compensation: Option<FloatHistogram> = None;
171    let mut mean: Option<FloatHistogram> = None;
172    let mut count = 0.0;
173    let mut custom_reconciled = false;
174    for h in histograms {
175        let h = h?;
176        count += 1.0;
177        let Some(current) = value.as_mut() else {
178            value = Some(h);
179            continue;
180        };
181        // Go drops the result with a warning for this addition error only; others propagate.
182        if current.uses_custom_buckets() != h.uses_custom_buckets() {
183            return Ok((None, custom_reconciled));
184        }
185        match op {
186            HistogramAggregate::Sum => {
187                let (c, outcome) = current.kahan_add(&h, compensation.take())?;
188                custom_reconciled |= outcome.custom_bounds_reconciled;
189                compensation = Some(c);
190            }
191            HistogramAggregate::Avg => {
192                if mean.is_none() {
193                    let mut sum = current.clone();
194                    let (c, outcome) = sum.kahan_add(&h, compensation.clone())?;
195                    custom_reconciled |= outcome.custom_bounds_reconciled;
196                    if !sum.has_overflow() {
197                        *current = sum;
198                        compensation = Some(c);
199                        continue;
200                    }
201                    let mut started = current.clone();
202                    started.div(count - 1.0);
203                    if let Some(c) = compensation.as_mut() {
204                        c.div(count - 1.0);
205                    }
206                    mean = Some(started);
207                }
208                // Go switches to an incremental mean once the plain sum overflows.
209                let q = (count - 1.0) / count;
210                if let Some(c) = compensation.as_mut() {
211                    c.mul(q);
212                }
213                let mut to_add = h;
214                to_add.div(count);
215                let mean = mean.as_mut().expect("set above");
216                mean.mul(q);
217                let (c, outcome) = mean.kahan_add(&to_add, compensation.take())?;
218                custom_reconciled |= outcome.custom_bounds_reconciled;
219                compensation = Some(c);
220            }
221        }
222    }
223    let Some(mut value) = value else {
224        return Ok((None, custom_reconciled));
225    };
226    match op {
227        HistogramAggregate::Sum => {
228            if let Some(c) = compensation {
229                value.add(&c)?;
230            }
231        }
232        HistogramAggregate::Avg => match mean {
233            Some(mut mean) => {
234                if let Some(c) = compensation {
235                    mean.add(&c)?;
236                }
237                value = mean;
238            }
239            None => {
240                value.div(count);
241                if let Some(mut c) = compensation {
242                    c.div(count);
243                    value.add(&c)?;
244                }
245            }
246        },
247    }
248    // The result was derived from its inputs, so it carries no start timestamp of its own, like
249    // every other operation that builds one. Clearing it here covers the fold that added nothing:
250    // the first histogram of a window becomes the result as it arrived, and the adds above are
251    // what drop the field otherwise, which would make the payload depend on the sample count.
252    // The bucket layout is left as the fold built it -- compaction belongs to the caller.
253    value.start_timestamp = None;
254    Ok((Some(value), custom_reconciled))
255}
