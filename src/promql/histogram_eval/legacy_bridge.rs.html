15//! The legacy native-histogram estimators, delegated to the native kernel.
16//!
17//! `prom_native_histogram_quantile` and `prom_native_histogram_fraction` used to estimate with the
18//! bucket walk inside `NativeHistogram` itself: exponential bucket bounds computed as `2^x`, and a
19//! forward walk from the lowest bucket only. The native kernel (`histogram_eval`) walks the same
20//! histogram the way Prometheus does -- the bound table in `bounds` and, for a quantile above the
21//! median of a histogram whose sum is not NaN, a walk from the highest bucket down -- so keeping
22//! both estimators leaves two answers for one histogram that differ in their last places.
23//!
24//! These functions read a [`NativeHistogram`] through the borrowed [`HistogramView`] that both
25//! representations share, hand it to the kernel as the [`FloatHistogram`] the kernel owns, and
26//! return the answer in the shape the legacy UDFs report it. The scalar arithmetic -- `add`, `sub`,
27//! `scale`, `divide_by` -- takes the same route below the estimators, so Prometheus's rules for
28//! reconciling a zero bucket, reducing a schema and intersecting custom bounds live in one place.
29

39/// A [`NativeHistogram`] borrowed as a [`HistogramView`], the form the native kernel reads.
40///
41/// A view holds its spans as two parallel slices while a `NativeHistogram` holds them as
42/// [`Span`]s, so the offsets and lengths are copied here and the counts, bounds and metadata stay
43/// borrowed for as long as the view lives.
44pub(crate) struct LegacyView<'a> {
45    histogram: &'a NativeHistogram,
46    positive_offsets: Vec<i32>,
47    positive_lengths: Vec<i32>,
48    negative_offsets: Vec<i32>,
49    negative_lengths: Vec<i32>,
50}
51
52impl<'a> LegacyView<'a> {
53    pub(crate) fn new(histogram: &'a NativeHistogram) -> Self {
54        let (positive_offsets, positive_lengths) = span_columns(&histogram.positive_spans);
55        let (negative_offsets, negative_lengths) = span_columns(&histogram.negative_spans);
56        Self {
57            histogram,
58            positive_offsets,
59            positive_lengths,
60            negative_offsets,
61            negative_lengths,
62        }
63    }
64
65    /// The view of the histogram these span columns were taken from.
66    pub(crate) fn view(&self) -> HistogramView<'_> {
67        HistogramView {
68            schema: self.histogram.schema,
69            zero_threshold: self.histogram.zero_threshold,
70            sum: self.histogram.sum,
71            count: self.histogram.count,
72            zero_count: self.histogram.zero_count,
73            reset_hint: self.histogram.reset_hint,
74            start_timestamp: self.histogram.start_timestamp,
75            custom_values: &self.histogram.custom_values,
76            positive: BucketSideView {
77                span_offsets: &self.positive_offsets,
78                span_lengths: &self.positive_lengths,
79                buckets: &self.histogram.positive_buckets,
80            },
81            negative: BucketSideView {
82                span_offsets: &self.negative_offsets,
83                span_lengths: &self.negative_lengths,
84                buckets: &self.histogram.negative_buckets,
85            },
86        }
87    }
88}

90fn span_columns(spans: &[Span]) -> (Vec<i32>, Vec<i32>) {
91    spans.iter().map(|span| (span.offset, span.length)).unzip()
92}

94/// `histogram_quantile(q, histogram)`, with the NaN-observation note the legacy UDF records.
95pub(crate) fn quantile_with_info(
96    histogram: &NativeHistogram,
97    q: f64,
98) -> (f64, Option<NativeHistogramQuantileInfo>) {
99    let view = LegacyView::new(histogram);
100    match native::quantile(q, &view.view(), &mut Vec::new()) {
101        Ok(value) => (value, nan_observation_info(histogram, q)),
102        // The kernel refuses a layout it cannot walk, a custom histogram carrying a zero count
103        // among them. The legacy estimator reported a histogram it could not read as no estimate,
104        // which is the same NaN without a note about NaN observations.
105        Err(_) => (f64::NAN, None),
106    }
107}

109/// `histogram_fraction(lower, upper, histogram)`, with whether NaN observations were excluded.
110pub(crate) fn fraction_with_info(
111    histogram: &NativeHistogram,
112    lower: f64,
113    upper: f64,
114) -> (f64, bool) {
115    let view = LegacyView::new(histogram);
116    match native::fraction(lower, upper, &view.view(), &mut Vec::new()) {
117        Ok(value) => {
118            // The same answer the legacy estimator gave: a fraction is degenerate, and reports
119            // nothing, unless it runs for a non-empty histogram over an increasing pair of bounds.
120            let excluded_nans = histogram.count != 0.0
121                && !lower.is_nan()
122                && !upper.is_nan()
123                && lower < upper
124                && histogram.sum.is_nan()
125                && bucket_total(histogram) < histogram.count;
126            (value, excluded_nans)
127        }
128        Err(_) => (f64::NAN, false),
129    }
130}

132/// The note for a NaN observation, which the kernel accounts for but does not report.
133///
134/// A NaN observation leaves the sum NaN and puts fewer observations in the buckets than the count
135/// holds them (Prometheus marks it the same way). The kernel then walks the buckets forward from
136/// the lowest, so the note distinguishes the two outcomes: the estimate is skewed higher when the
137/// requested rank falls inside the buckets' total, and NaN when it falls past it because the
138/// observations the buckets cannot carry were the highest ones.
139fn nan_observation_info(
140    histogram: &NativeHistogram,
141    q: f64,
142) -> Option<NativeHistogramQuantileInfo> {
143    if histogram.count == 0.0 || !(0.0..=1.0).contains(&q) {
144        return None;
145    }
146    let total = bucket_total(histogram);
147    if !(histogram.sum.is_nan() && total < histogram.count) {
148        return None;
149    }
150    Some(if total < q * histogram.count {
151        NativeHistogramQuantileInfo::NaNResult
152    } else {
153        NativeHistogramQuantileInfo::NaNSkew
154    })
155}

157/// What the populated buckets hold, summed in the order the kernel walks them: the negative side
158/// from its highest index down, the zero bucket, then the positive side upwards.
159fn bucket_total(histogram: &NativeHistogram) -> f64 {
160    let mut total = 0.0;
161    for count in histogram.negative_buckets.iter().rev() {
162        total += count;
163    }
164    total += histogram.zero_count;
165    for count in &histogram.positive_buckets {
166        total += count;
167    }
168    total
169}

171/// The native kernel's owned form of `histogram`.
172///
173/// A view borrows the offsets and lengths it needs from two temporary slices, so the copy happens
174/// here and the `FloatHistogram` owns the counts that result.
175fn float_histogram(histogram: &NativeHistogram) -> Result<FloatHistogram> {
176    if steps_backwards(&histogram.positive_spans) || steps_backwards(&histogram.negative_spans) {
177        return Err(datafusion::error::DataFusionError::Execution(
178            "native histogram has an invalid layout".into(),
179        ));
180    }
181    let view = LegacyView::new(histogram);
182    FloatHistogram::try_from_view(&view.view())
183}

185/// Whether a span list steps backwards, which the legacy `bucket_indices` refused.
186///
187/// The kernel's `side_from_view` adds each offset to a running index and accepts a later span that
188/// steps backwards, so the expanded indices repeat or go negative. Legacy refused such a list
189/// outright, so the bridge refuses it too: `add`/`sub` answer `None` as they did, and
190/// `scale`/`divide_by` keep their field-by-field fallback.
191fn steps_backwards(spans: &[Span]) -> bool {
192    spans.iter().skip(1).any(|span| span.offset < 0)
193}

195/// The legacy representation of `histogram`, with the span layout it carries.
196fn legacy_histogram(histogram: &FloatHistogram) -> NativeHistogram {
197    let [positive, negative] = histogram.spans();
198    NativeHistogram {
199        schema: histogram.schema,
200        zero_threshold: histogram.zero_threshold,
201        sum: histogram.sum,
202        reset_hint: histogram.reset_hint,
203        start_timestamp: histogram.start_timestamp,
204        custom_values: histogram.custom_values.clone(),
205        positive_spans: legacy_spans(positive),
206        negative_spans: legacy_spans(negative),
207        count: histogram.count,
208        zero_count: histogram.zero_count,
209        positive_buckets: bucket_counts(&histogram.positive),
210        negative_buckets: bucket_counts(&histogram.negative),
211    }
212}

214fn legacy_spans(spans: Vec<(i32, i32)>) -> Vec<Span> {
215    spans

225/// `left + right`, or `None` for a pair the kernel refuses to combine.
226pub(crate) fn add(left: &NativeHistogram, right: &NativeHistogram) -> Option<NativeHistogram> {
227    combine(left, right, false)
228}
229
230/// `left - right`, a gauge histogram the way Prometheus's `-` leaves it.
231pub(crate) fn sub(left: &NativeHistogram, right: &NativeHistogram) -> Option<NativeHistogram> {
232    combine(left, right, true)
233}
234
235fn combine(
236    left: &NativeHistogram,
237    right: &NativeHistogram,
238    subtract: bool,
239) -> Option<NativeHistogram> {
240    let mut combined = float_histogram(left).ok()?;
241    let other = float_histogram(right).ok()?;
242    let outcome = if subtract {
243        combined.sub(&other)
244    } else {
245        combined.add(&other)
246    }
247    .ok()?;
248    // The outcome reports what the kernel already folded into the result, and the legacy callers
249    // report both from the operands before the call: `custom_bounds_reconciled` is the intersection
250    // the kernel keeps in `custom_values`, which `needs_custom_reconciliation` announces, and a
251    // `counter_reset_collision` is the contradiction the kernel answers with an unknown hint, which
252    // is what the legacy arithmetic derived from two hints that disagree. Holding the kernel to
253    // both keeps a change there from dropping either.
254    debug_assert_eq!(
255        outcome.custom_bounds_reconciled,
256        left.needs_custom_reconciliation(right)
257    );
258    if outcome.counter_reset_collision {
259        debug_assert_eq!(combined.reset_hint, CounterResetHint::Unknown);
260    }
261    if subtract {
262        combined.reset_hint = CounterResetHint::Gauge;
263    }
264    // The legacy arithmetic compacted a sum, so a bucket the two sides cancel out of leaves no
265    // span behind; the kernel keeps the bucket until a caller drops it.
266    combined.compact();
267    Some(legacy_histogram(&combined))
268}

270/// `histogram * factor`, a gauge histogram for a negative factor.
271pub(crate) fn scale(histogram: &NativeHistogram, factor: f64) -> NativeHistogram {
272    let Ok(mut scaled) = float_histogram(histogram) else {
273        // A layout the kernel cannot read -- spans that disagree with the counts, or a bucket index
274        // that leaves `i32` once expanded -- keeps the field-by-field scaling the legacy caller
275        // had, so such a histogram is still multiplied instead of dropped, and reports its layout
276        // when it is serialized. Scaling has no bounds to reconcile, so nothing is lost here.
277        return field_scaled(histogram, factor);
278    };
279    scaled.mul(factor);
280    legacy_histogram(&scaled)
281}
282
283/// `histogram / divisor`, with no buckets left for a zero divisor.
284pub(crate) fn divide_by(histogram: &NativeHistogram, divisor: f64) -> NativeHistogram {
285    let Ok(mut divided) = float_histogram(histogram) else {
286        return field_divided(histogram, divisor);
287    };
288    divided.div(divisor);
289    legacy_histogram(&divided)
290}
291
292fn field_scaled(histogram: &NativeHistogram, factor: f64) -> NativeHistogram {
293    let mut scaled = histogram.clone();
294    scaled.count *= factor;
295    scaled.zero_count *= factor;
296    scaled.sum *= factor;
297    for count in scaled
298        .positive_buckets
299        .iter_mut()
300        .chain(&mut scaled.negative_buckets)
301    {
302        *count *= factor;
303    }
304    if factor < 0.0 {
305        scaled.reset_hint = CounterResetHint::Gauge;
306    }
307    scaled
308}
309
310fn field_divided(histogram: &NativeHistogram, divisor: f64) -> NativeHistogram {
311    let mut divided = histogram.clone();
312    divided.count /= divisor;
313    divided.zero_count /= divisor;
314    divided.sum /= divisor;
315    if divisor == 0.0 {
316        divided.positive_spans.clear();
317        divided.positive_buckets.clear();
318        divided.negative_spans.clear();
319        divided.negative_buckets.clear();
320    } else {
321        for count in divided
322            .positive_buckets
323            .iter_mut()
324            .chain(&mut divided.negative_buckets)
325        {
326            *count /= divisor;
327        }
328    }
329    if divisor < 0.0 {
330        divided.reset_hint = CounterResetHint::Gauge;
331    }
332    divided
333}

488    /// Prometheus's NaN branch walks the remaining buckets and leaves the last one for the
489    /// interpolation, so a multi-bucket `sum = NaN` histogram answers with the top bucket rather
490    /// than the bucket the rank fell into. Pinning the number here, rather than only comparing
491    /// against the kernel, keeps a future kernel change from silently re-deciding it.
492    #[test]
493    fn quantile_with_nan_sum_uses_the_last_bucket() {
494        let mut histogram = empty(0, 0.0, 10.0, f64::NAN, 0.0);
495        histogram.positive_spans = vec![span(0, 2)];
496        histogram.positive_buckets = vec![8.0, 1.0];
497
498        // The nine observations the buckets hold are fewer than the count of ten and the sum is
499        // NaN, so the note reports the skew: the rank still falls inside what the buckets carry.
500        let (value, info) = quantile_with_info(&histogram, 0.5);
501        assert_eq!(
502            value.to_bits(),
503            32.0f64.to_bits(),
504            "rank 5 lands in (0.5, 1] (0.7711), but the NaN walk leaves (1, 2] behind and \
505             fraction 5 / 1 puts the answer at 2^5"
506        );
507        assert_eq!(
508            info,
509            Some(NativeHistogramQuantileInfo::NaNSkew),
510            "a rank inside the bucket total is still an estimate"
511        );

547    /// A tie between two buckets that do not touch resolves differently in the two walks, and that
548    /// difference is why the kernel replaced the estimator: the legacy walk goes up from the lowest
549    /// bucket and stops at the first one the rank fits in, while the kernel walks a finite-sum
550    /// histogram at or above the median down from the highest. Both numbers are pinned as literals,
551    /// the legacy one computed through the estimator still in `common/query`, so that a later
552    /// change to either walk shows up as an arithmetic difference instead of a quiet one.
553    #[test]
554    fn exact_tie_answers_one_from_the_legacy_walk_and_two_from_the_kernel() {
555        let mut tie = empty(0, 0.0, 2.0, 3.75, 0.0);
556        // Index 0 and index 2 hold one observation each, and index 1 is empty, so the two buckets
557        // do not touch and nothing interpolates across the gap.
558        tie.positive_spans = vec![span(0, 1), span(1, 1)];
559        tie.positive_buckets = vec![1.0, 1.0];
560
561        // `NativeHistogram::quantile_with_info` is the estimator the legacy UDF used before the
562        // kernel replaced it and is still in the workspace: rank 0.5 * 2 = 1 fills the bucket at
563        // index 0 exactly, whose fraction is 1, so the answer is that bucket's upper bound.
564        assert_eq!(
565            tie.quantile_with_info(0.5),
566            (1.0, None),
567            "the legacy walk answers with the upper bound of (0.5, 1]"
568        );
569
570        // The kernel's backwards walk meets the bucket at index 2 first, and the rank inside it is
571        // 0, so the answer is the lower bound of (2, 4].
572        let (value, info) = quantile_with_info(&tie, 0.5);
573        assert_eq!(
574            value.to_bits(),
575            2.0f64.to_bits(),
576            "the kernel's walk answers with the lower bound of (2, 4]"
577        );
578        assert_eq!(
579            info, None,
580            "a tie between two buckets is not a NaN observation"
581        );
582    }

684    /// The buckets a side holds that carry observations, by index and bit pattern.
685    fn populated_buckets(spans: &[Span], buckets: &[f64]) -> Vec<(i32, u64)> {
686        let (offsets, lengths) = columns(spans);
687        let mut index = 0;
688        let mut out = Vec::new();
689        let mut counts = buckets.iter();
690        for (position, (offset, length)) in offsets.into_iter().zip(lengths).enumerate() {
691            index = if position == 0 {
692                offset
693            } else {
694                index + offset
695            };
696            for _ in 0..length {
697                let count = *counts.next().unwrap();
698                if count != 0.0 {
699                    out.push((index, count.to_bits()));
700                }
701                index += 1;
702            }
703        }
704        out
705    }

708        side.iter()
709            .filter(|(_, count)| *count != 0.0)
710            .map(|(index, count)| (*index, count.to_bits()))
711            .collect()
712    }

732    /// Every field of the legacy payload against the kernel's, span segmentation included: a
733    /// comparison of the populated buckets alone cannot tell one span of two buckets from two
734    /// spans of one bucket each, and says nothing about the metadata beside the counts.
735    ///
736    /// `expected_start_timestamp` states the policy of the operation that produced the payload. One
737    /// that derives a histogram from its input -- `add`, `sub`, `scale`, `divide_by` -- clears the
738    /// start timestamp, because the field names the start of the sample the payload was read from
739    /// and a derived value is not that sample; the kernel clears it in the same step, so both sides
740    /// must carry the value passed here. `scale` and `divide_by` keep a field-by-field fallback for
741    /// a layout the kernel cannot read, which derives nothing and copies the field, and the test
742    /// that exercises it asserts `Some(..)` where it belongs. `quantile_with_info` and
743    /// `fraction_with_info` return no payload and leave their input untouched, so they have no
744    /// start timestamp to compare here.

752        assert_eq!(
753            legacy.zero_threshold.to_bits(),
754            native.zero_threshold.to_bits(),
755            "{name} zero_threshold"
756        );
757        assert_eq!(
758            legacy.count.to_bits(),
759            native.count.to_bits(),
760            "{name} count"
761        );
762        assert_eq!(legacy.sum.to_bits(), native.sum.to_bits(), "{name} sum");
763        assert_eq!(
764            legacy.zero_count.to_bits(),
765            native.zero_count.to_bits(),
766            "{name} zero_count"
767        );
768        assert_eq!(legacy.reset_hint, native.reset_hint, "{name} reset_hint");
769        assert_eq!(
770            legacy.custom_values, native.custom_values,
771            "{name} custom_values"
772        );
773        assert_eq!(
774            legacy.start_timestamp, expected_start_timestamp,
775            "{name} start_timestamp"
776        );
777        assert_eq!(
778            native.start_timestamp, expected_start_timestamp,
779            "{name} kernel start_timestamp"

784            segments(&legacy.positive_spans),

1101        assert_eq!(difference.zero_count.to_bits(), 0.0f64.to_bits());
1102        assert!(
1103            difference.positive_spans.is_empty(),
1104            "a cancelled bucket leaves no span behind, got {:?}",
1105            difference.positive_spans
1106        );
1107        assert!(difference.positive_buckets.is_empty());
1108        assert!(difference.negative_spans.is_empty());
1109        assert!(difference.negative_buckets.is_empty());
1110        // The same histogram as no histogram at all.
1111        assert!(
1112            difference.promql_eq(&empty(2, 0.5, 0.0, 0.0, 0.0)),
1113            "the cancelling subtraction is the empty histogram"
1114        );
1115
1116        // Two spans of one bucket each describe the same buckets as one span of two: the first
1117        // span's offset is the bucket index, and each later offset is the gap after the previous
1118        // span, so `(0, 1), (0, 1)` puts the second bucket at index 1 while `(0, 1), (1, 1)`
1119        // leaves index 1 empty and puts it at index 2.
1120        let mut one_span = empty(0, 0.0, 3.0, 3.0, 0.0);
1121        one_span.positive_spans = vec![span(0, 2)];
1122        one_span.positive_buckets = vec![1.0, 2.0];
1123        let mut two_spans = empty(0, 0.0, 3.0, 3.0, 0.0);
1124        two_spans.positive_spans = vec![span(0, 1), span(0, 1)];
1125        two_spans.positive_buckets = vec![1.0, 2.0];
1126        assert!(
1127            one_span.promql_eq(&two_spans),
1128            "the same indices and counts under a different segmentation"
1129        );
1130
1131        let mut gapped = empty(0, 0.0, 3.0, 3.0, 0.0);
1132        gapped.positive_spans = vec![span(0, 1), span(1, 1)];
1133        gapped.positive_buckets = vec![1.0, 2.0];
1134        assert!(
1135            !one_span.promql_eq(&gapped),
1136            "the same counts at indices 0 and 2 are not the same histogram"
1137        );
1138        let mut other_counts = two_spans.clone();
1139        other_counts.positive_buckets = vec![2.0, 1.0];
1140        assert!(
1141            !one_span.promql_eq(&other_counts),
1142            "the same indices with other counts are not the same histogram"
1143        );
1144    }

1146    /// Payloads the legacy caller could still produce and the kernel refuses to read: a span whose
1147    /// length disagrees with the counts, a span list that steps backwards, an index that leaves
1148    /// `i32` once expanded, and a negative span length. All four carry the same numbers, so the
1149    /// field-by-field answers below are the same for each of them.
1150    fn unreadable_layouts() -> Vec<(&'static str, NativeHistogram, Option<i64>)> {
1151        // The span claims one bucket and the counts hold two.
1152        let mut short_counts = empty(0, 0.0, 3.0, 5.0, 1.0);
1153        short_counts.reset_hint = CounterResetHint::NotCounterReset;
1154        short_counts.positive_spans = vec![span(0, 1)];
1155        short_counts.positive_buckets = vec![1.0, 2.0];
1156        let mut backwards = short_counts.clone();
1157        backwards.positive_spans = vec![span(0, 1), span(-1, 1)];
1158        let mut overflow = short_counts.clone();
1159        overflow.positive_spans = vec![span(i32::MAX, 2)];
1160        let mut negative_length = short_counts.clone();
1161        negative_length.positive_spans = vec![span(0, -1)];
1162
1163        vec![
1164            ("short_counts", short_counts, Some(4_242)),
1165            ("backwards", backwards, Some(4_243)),
1166            ("overflow", overflow, Some(4_244)),
1167            ("negative_length", negative_length, Some(4_245)),
1168        ]
1169        .into_iter()
1170        .map(|(name, mut histogram, start_timestamp)| {
1171            histogram.start_timestamp = start_timestamp;
1172            (name, histogram, start_timestamp)
1173        })
1174        .collect()
1175    }

1177    /// A layout the kernel cannot walk keeps the field-by-field scaling and division the legacy
1178    /// caller had, so such a histogram is still multiplied instead of dropped, and keeps the start
1179    /// timestamp it was stored with.
1180    #[test]
1181    fn scale_and_divide_fall_back_to_the_fields_for_an_unreadable_layout() {
1182        for (name, histogram, start_timestamp) in unreadable_layouts() {
1183            // A negative factor turns the histogram into a gauge; the untouched hint otherwise.
1184            let negated = scale(&histogram, -0.5);
1185            assert_eq!(negated.count, -1.5, "{name} count");
1186            assert_eq!(negated.sum, -2.5, "{name} sum");
1187            assert_eq!(negated.zero_count, -0.5, "{name} zero_count");
1188            assert_eq!(
1189                negated.positive_buckets,
1190                vec![-0.5, -1.0],
1191                "{name} positive"
1192            );
1193            assert_eq!(negated.reset_hint, CounterResetHint::Gauge, "{name} hint");
1194            assert_eq!(
1195                negated.start_timestamp, start_timestamp,
1196                "{name}: the field-by-field fallback copies the payload, so the start timestamp \
1197                 survives where the kernel path clears it"
1198            );
1199            assert_eq!(
1200                negated.positive_spans, histogram.positive_spans,
1201                "{name}: the fallback keeps the layout it cannot expand"
1202            );
1203
1204            let doubled = scale(&histogram, 2.0);
1205            assert_eq!(doubled.count, 6.0, "{name} count");
1206            assert_eq!(doubled.sum, 10.0, "{name} sum");
1207            assert_eq!(doubled.zero_count, 2.0, "{name} zero_count");
1208            assert_eq!(doubled.positive_buckets, vec![2.0, 4.0], "{name} positive");
1209            assert_eq!(
1210                doubled.reset_hint,
1211                CounterResetHint::NotCounterReset,
1212                "{name}: a positive factor leaves a hint alone"
1213            );
1214            assert_eq!(doubled.start_timestamp, start_timestamp, "{name}");
1215
1216            let quartered = divide_by(&histogram, 4.0);
1217            assert_eq!(quartered.count, 0.75, "{name} count");
1218            assert_eq!(quartered.sum, 1.25, "{name} sum");
1219            assert_eq!(quartered.zero_count, 0.25, "{name} zero_count");
1220            assert_eq!(
1221                quartered.positive_buckets,
1222                vec![0.25, 0.5],
1223                "{name} positive"
1224            );
1225            assert_eq!(
1226                quartered.reset_hint,
1227                CounterResetHint::NotCounterReset,
1228                "{name}: a positive divisor leaves a hint alone"
1229            );
1230            assert_eq!(quartered.start_timestamp, start_timestamp, "{name}");
1231
1232            // A zero divisor empties the buckets on this path too.
1233            let over_zero = divide_by(&histogram, 0.0);
1234            assert_eq!(over_zero.count, f64::INFINITY, "{name} count");
1235            assert_eq!(over_zero.sum, f64::INFINITY, "{name} sum");
1236            assert_eq!(over_zero.zero_count, f64::INFINITY, "{name} zero_count");
1237            assert!(over_zero.positive_spans.is_empty(), "{name}");
1238            assert!(over_zero.positive_buckets.is_empty(), "{name}");
1239            assert_eq!(over_zero.start_timestamp, start_timestamp, "{name}");
1240
1241            // The addition and the subtraction refuse the same layout outright, as the legacy
1242            // arithmetic did, rather than answering with a payload the kernel never built.
1243            assert!(
1244                add(&histogram, &histogram).is_none(),
1245                "{name}: a layout the kernel cannot read has no sum"
1246            );
1247            assert!(
1248                sub(&histogram, &histogram).is_none(),
1249                "{name}: a layout the kernel cannot read has no difference"
1250            );
1251        }
1252    }

1254    /// The factors that are not a plain finite number, on a layout the kernel can walk: a zero
1255    /// divisor leaves no bucket, a NaN factor reaches every count, and a negative one turns the
1256    /// histogram into a gauge.
1257    #[test]
1258    fn scale_and_divide_by_the_edge_factors() {
1259        let mut histogram = empty(0, 0.0, 4.0, 6.0, 0.0);
1260        histogram.reset_hint = CounterResetHint::NotCounterReset;
1261        histogram.start_timestamp = Some(7);
1262        histogram.positive_spans = vec![span(0, 2)];
1263        histogram.positive_buckets = vec![3.0, 1.0];
1264        histogram.negative_spans = vec![span(0, 1)];
1265        histogram.negative_buckets = vec![1.0];
1266
1267        let over_zero = divide_by(&histogram, 0.0);
1268        assert_eq!(over_zero.count, f64::INFINITY, "a positive count over zero");
1269        assert_eq!(over_zero.sum, f64::INFINITY, "a positive sum over zero");
1270        assert!(
1271            over_zero.zero_count.is_nan(),
1272            "a zero count over zero is NaN"
1273        );

1275        assert!(over_zero.positive_buckets.is_empty());
1276        assert!(over_zero.negative_spans.is_empty());

1299        assert!(over_negative_zero.positive_spans.is_empty());
1300        assert!(over_negative_zero.positive_buckets.is_empty());
