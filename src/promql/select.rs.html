15//! Operators that depend on the whole set of evaluated series at each grid point: `topk`,
16//! `bottomk`, `limitk`, `limit_ratio`, `scalar`, `absent` and the instant-result sorts.
17//!
18//! Semantics follow Prometheus (`aggregationK`, `HashRatioSampler`, `funcScalar`, `funcAbsent`,
19//! `createLabelsForAbsentFunction`, `funcSort*`). A scalar is a grid series without labels that
20//! has a value at every point, so `vector(s)` is `s` itself and has no kernel.

40/// The labels that form an aggregation group.
41#[derive(Clone, Debug, PartialEq, Eq)]
42pub enum Grouping {
43    /// `by(...)`: only these labels, `__name__` included when listed.
44    By(Vec<Arc<str>>),
45    /// `without(...)`: every label except these and `__name__`.
46    Without(Vec<Arc<str>>),
47}

49/// The order `sort_instant` arranges an instant result in.
50#[derive(Clone, Debug, PartialEq, Eq, serde::Serialize, serde::Deserialize, Hash)]
51pub enum SortOrder {
52    /// `sort`: ascending values, NaN last.
53    Asc,
54    /// `sort_desc`: descending values, NaN last.
55    Desc,
56    /// `sort_by_label`: natural order of the listed label values, then the label sets.
57    ByLabel(Vec<Arc<str>>),
58    /// `sort_by_label_desc`.
59    ByLabelDesc(Vec<Arc<str>>),
60}

62/// `topk(k, ...)` (`top`) or `bottomk(k, ...)` with a constant `k`.
63///
64/// At every point each group keeps the `k` largest (or smallest) values, NaN ranking last for
65/// both; the selected samples keep their series' labels and values. Histogram samples are not
66/// ranked, as Prometheus ignores them here. Prometheus breaks ties by
67/// the order its storage returns series in, which is not a property of the series; here ties
68/// go to the series whose label set sorts first.
69pub fn topk(
70    k: f64,
71    top: bool,
72    grouping: &Grouping,
73    schema: &SeriesSchema,
74    series: &[Series],
75) -> Result<Vec<Series>> {
76    let k = k_param(k, series.len())?;
77    topk_at(|_| Ok(k), top, grouping, schema, series)
78}

80/// `topk`/`bottomk` with `k` evaluated independently at each grid point.
81pub fn topk_at(
82    k_at: impl Fn(usize) -> Result<Option<usize>>,
83    top: bool,
84    grouping: &Grouping,
85    schema: &SeriesSchema,
86    series: &[Series],
87) -> Result<Vec<Series>> {
88    single_field(schema, "topk/bottomk")?;
89    // Validates the series against the schema and its grid; the walk below visits the carried
90    // positions, not the whole grid.
91    grid_points(schema, series)?;
92    let mut order: Vec<usize> = (0..series.len()).collect();
93    order.sort_by(|&a, &b| compare_labels(&series[a].metric, &series[b].metric));
94    let mut groups: HashMap<GroupKey, Vec<usize>> = HashMap::new();
95    for index in order {
96        groups
97            .entry(group_key(&series[index].metric, grouping))
98            .or_default()
99            .push(index);
100    }
101
102    let mut selected: Vec<Vec<(usize, usize)>> = vec![Vec::new(); series.len()];
103    let mut candidates: Vec<(usize, usize, f64)> = Vec::new();
104    for members in groups.values() {
105        // A point no member carries holds no candidate, so only the union of the members'
106        // positions is visited: a hole in the grid never reaches the ranking. Each member reads
107        // through one forward cursor, whose rows only increase as the union walk advances.
108        let mut cursors: Vec<RowCursor<'_>> = members
109            .iter()
110            .map(|&member| series[member].fields.as_slice()[0].cursor())
111            .collect();
112        for point in union_positions(members.iter().map(|&member| &series[member].time)) {
113            candidates.clear();
114            for (slot, &member) in members.iter().enumerate() {
115                let Some(row) = series[member].time.row_of(point) else {
116                    continue;
117                };
118                if let SampleRef::Float(value) = cursors[slot]
119                    .at(row)
120                    .expect("the axis and the fields are aligned")
121                {
122                    candidates.push((member, row, value));
123                }
124            }
125            // Members are in label order, so a stable sort hands ties to the earlier label set.
126            candidates.sort_by(|a, b| rank(a.2, b.2, top));
127            let Some(k) = k_at(point)? else {
128                continue;
129            };
130            for &(member, row, _) in candidates.iter().take(k) {
131                selected[member].push((point, row));
132            }
133        }
134    }
135
136    let output = Arc::new(schema.clone());
137    let mut builder = GridSeriesBuilder::try_new(output)?;
138    let mut result = Vec::new();
139    for (index, points) in selected.into_iter().enumerate() {
140        if points.is_empty() {
141            continue;
142        }
143        let mut sorted = points;
144        sorted.sort_unstable();
145        // One builder serves every result series, so each one starts explicitly: only
146        // `begin_series` resets the positions it accepts.
147        builder.begin_series();
148        // The selected positions of one series only increase, so one cursor reads their rows.
149        let mut cursor = series[index].fields.as_slice()[0].cursor();
150        for (point, row) in sorted {
151            builder.push(point, &[cursor_value(&mut cursor, row)?])?;
152        }
153        result.extend(builder.finish(series[index].metric.clone())?);
154    }
155    Ok(result)
156}

158/// `limitk(k, ...)` with a constant `k`.
159///
160/// At every point each group keeps the first `k` series in input order that hold a sample there,
161/// float or histogram, with their labels and values. Prometheus takes them in the order its
162/// storage returns series: label order once a query reads several queriers (the head and a
163/// block), creation order when the TSDB head alone answers it. A caller that wants Prometheus'
164/// result over persisted data passes the series in label order.
165pub fn limitk(
166    k: f64,
167    grouping: &Grouping,
168    schema: &SeriesSchema,
169    series: &[Series],
170) -> Result<Vec<Series>> {
171    single_field(schema, "limitk")?;
172    grid_points(schema, series)?;
173    let Some(k) = k_param(k, series.len())? else {
174        return Ok(Vec::new());
175    };
176    let mut groups: HashMap<GroupKey, Vec<usize>> = HashMap::new();
177    for (index, item) in series.iter().enumerate() {
178        groups
179            .entry(group_key(&item.metric, grouping))
180            .or_default()
181            .push(index);
182    }
183    let mut selected: Vec<Vec<(usize, usize)>> = vec![Vec::new(); series.len()];
184    for members in groups.values() {
185        let mut cursors: Vec<RowCursor<'_>> = members
186            .iter()
187            .map(|&member| series[member].fields.as_slice()[0].cursor())
188            .collect();
189        for point in union_positions(members.iter().map(|&member| &series[member].time)) {
190            let mut taken = 0;
191            for (slot, &member) in members.iter().enumerate() {
192                if taken == k {
193                    break;
194                }
195                let Some(row) = series[member].time.row_of(point) else {
196                    continue;
197                };
198                if !matches!(
199                    cursors[slot]
200                        .at(row)
201                        .expect("the axis and the fields are aligned"),
202                    SampleRef::Missing
203                ) {
204                    selected[member].push((point, row));
205                    taken += 1;
206                }
207            }
208        }
209    }
210
211    let mut builder = GridSeriesBuilder::try_new(Arc::new(schema.clone()))?;
212    let mut result = Vec::new();
213    for (index, points) in selected.into_iter().enumerate() {
214        if points.is_empty() {
215            continue;
216        }
217        builder.begin_series();
218        // A series belongs to one group, whose points are visited in order.
219        let mut cursor = series[index].fields.as_slice()[0].cursor();
220        for (point, row) in points {
221            let sample = cursor.at(row).expect("the axis and the fields are aligned");
222            builder.push_samples(point, &[owned(sample)?])?;
223        }
224        result.extend(builder.finish(series[index].metric.clone())?);
225    }
226    Ok(result)
227}

229/// `limit_ratio(r, ...)` with a constant `r`.
230///
231/// Keeps whole series: Prometheus derives an offset in `[0, 1)` from the hash of a series' labels
232/// and keeps the series when the offset is below `r`, or, for a negative `r`, at least `1 + r`.
233/// Which series are kept does not depend on the grouping, so it is not a parameter. The hash is
234/// the one of Prometheus' default `stringlabels` build; the other label builds hash differently.
235pub fn limit_ratio(ratio: f64, schema: &SeriesSchema, series: &[Series]) -> Result<Vec<Series>> {
236    grid_points(schema, series)?;
237    if ratio.is_nan() {
238        return Err(DataFusionError::Execution("Ratio value is NaN".into()));
239    }
240    let ratio = ratio.clamp(-1.0, 1.0);
241    let mut builder = GridSeriesBuilder::try_new(Arc::new(schema.clone()))?;
242    let mut result = Vec::new();
243    let mut row = Vec::with_capacity(schema.fields.len());
244    for item in series {
245        let offset = sample_offset(&item.metric);
246        let kept = if ratio >= 0.0 {
247            offset < ratio
248        } else {
249            offset >= 1.0 + ratio
250        };
251        if !kept {
252            continue;
253        }
254        // Rebuilt rather than cloned: an input may carry rows without a value, a result may not.
255        builder.begin_series();
256        let mut cursors: Vec<RowCursor<'_>> =
257            item.fields.iter().map(|field| field.cursor()).collect();
258        for (point, index) in item.time.positions() {
259            row.clear();
260            for cursor in &mut cursors {
261                row.push(owned(
262                    cursor
263                        .at(index)
264                        .expect("the axis and the fields are aligned"),
265                )?);
266            }
267            builder.push_samples(point, &row)?;
268        }
269        result.extend(builder.finish(item.metric.clone())?);
270    }
271    Ok(result)
272}

274/// Prometheus `HashRatioSampler.SampleOffset`: the xxHash64 of the labels as `stringlabels`
275/// encodes them, each name and value prefixed by its length, over `u64::MAX`.
276fn sample_offset(metric: &SeriesLabels) -> f64 {
277    let mut data = Vec::new();
278    for (name, value) in labels_with_name(metric) {
279        // Prometheus label sets hold no empty value.
280        if value.is_empty() {
281            continue;
282        }
283        for text in [&name, &value] {
284            let len = text.len();
285            if len < 255 {
286                data.push(len as u8);
287            } else {
288                data.extend([255, len as u8, (len >> 8) as u8, (len >> 16) as u8]);
289            }
290            data.extend_from_slice(text.as_bytes());
291        }
292    }
293    twox_hash::XxHash64::oneshot(0, &data) as f64 / u64::MAX as f64
294}

296/// `scalar(v)`: the value of the only float sample present at each point, NaN otherwise.
297///
298/// Native histogram samples are ignored, as Prometheus' `funcScalar` ignores them: they hold no
299/// float value, and they are not an element either, so they never make a point hold more than one
300/// sample. A point whose only sample is a histogram therefore answers NaN, a histogram beside one
301/// float answers that float, and two floats answer NaN.
302///
303/// A scalar holds a value at every point of the grid, so its value array is written where the input
304/// is read rather than staged and rebuilt: the one buffer of values becomes the result's array, and
305/// the counts that decide between a value and a NaN are dropped before it is handed over. Building
306/// it therefore costs one `f64` and one count per point and nothing else per point -- no optional
307/// value, no row index, no second copy of the answer -- which is what the caller reserves for it.
308pub fn scalar(schema: &SeriesSchema, series: &[Series]) -> Result<Series> {
309    single_field(schema, "scalar")?;
310    let (grid, points) = grid_and_points(schema, series)?;
311    // A point no float has reached answers the Go NaN, which is where it starts. The first float of
312    // a point writes its value there; the second takes the point back to NaN, and so does any float
313    // after it, so a point holding more than one float never reads as holding one.
314    let mut values = vec![GO_NAN; points];
315    let mut counts = vec![0u8; points];
316    for item in series {
317        // Walk the rows this series carries: a sparse axis must not be read as its envelope, and
318        // a forward cursor keeps the read proportional to them.
319        let mut field = item.fields.as_slice()[0].cursor();
320        for (point, row) in item.time.positions() {
321            match field.at(row).expect("the axis and the fields are aligned") {
322                SampleRef::Float(value) => {
323                    let count = &mut counts[point];
324                    values[point] = if *count == 0 { value } else { GO_NAN };
325                    *count = count.saturating_add(1);
326                }
327                // Neither a histogram nor a position holding no sample is an element of the
328                // vector, so neither decides what the point answers.
329                SampleRef::Histogram(_) | SampleRef::Missing => {}
330            }
331        }
332    }
333    // The counts have decided every value above; handing them back first leaves the values as the
334    // only per-point buffer alive when the answer is built.
335    drop(counts);
336    let fields = SeriesFields::one(Samples::from_float(Float64Array::from(values)));
337    // Every point of the grid holds a value, so the axis is the whole grid as one run.
338    let time = TimeAxis::Grid(GridRange::try_new(grid, 0, points).map_err(external)?);
339    let empty = SeriesLabels::try_new(None, Vec::new()).map_err(external)?;
340    Series::try_new(schema, empty, time, fields).map_err(external)
341}

343/// `absent(v)`: 1 with `labels` at every point where no series is present.
344///
345/// `absent_over_time(v[r])` is this over the result of `present_over_time(v[r])`. When `v[r]` is
346/// a range selector, `labels` come from its matchers as for `absent(v)`; for a subquery
347/// Prometheus derives no labels.
348///
349/// All of it is one label set, so the result is one series. A point where one field is present and
350/// another is absent keeps its row, with a null in the field that is present; a point where every
351/// field is present contributes no row, and the axis names the runs of points that do. `None` means
352/// no point is absent in any field, so no row survives.
353pub fn absent(
354    schema: &SeriesSchema,
355    series: &[Series],
356    labels: SeriesLabels,
357) -> Result<Option<Series>> {
358    let points = grid_points(schema, series)?;
359    let fields = schema.fields.len();
360    // Each field is absent on its own.
361    let mut present = vec![false; points * fields];
362    for item in series {
363        for (field, samples) in item.fields.iter().enumerate() {
364            // One cursor per field, driven by the positions the axis carries.
365            let mut cursor = samples.cursor();
366            for (point, row) in item.time.positions() {
367                if !matches!(
368                    cursor.at(row).expect("the axis and the fields are aligned"),
369                    SampleRef::Missing
370                ) {
371                    present[point * fields + field] = true;
372                }
373            }
374        }
375    }
376    let mut builder = GridSeriesBuilder::try_new(Arc::new(schema.clone()))?;
377    let mut row = vec![None; fields];
378    for point in 0..points {
379        for (field, value) in row.iter_mut().enumerate() {
380            *value = (!present[point * fields + field]).then_some(1.0);
381        }
382        builder.push(point, &row)?;
383    }
384    builder.finish(labels)
385}

387/// A label matcher of the selector `absent` is applied to.
388#[derive(Clone, Debug, PartialEq, Eq)]
389pub struct LabelMatcher {
390    pub name: Arc<str>,
391    pub op: MatchOp,
392    pub value: Arc<str>,
393    /// Whether the matcher accepts an empty value, i.e. a missing label; the caller evaluates
394    /// regular expressions.
395    pub matches_empty: bool,
396}
397
398/// The operator of a [`LabelMatcher`].
399#[derive(Clone, Copy, Debug, PartialEq, Eq, serde::Serialize, serde::Deserialize, Hash)]
400pub enum MatchOp {
401    Equal,
402    NotEqual,
403    Regex,
404    NotRegex,
405}

407/// The labels of an `absent` result, as Prometheus `createLabelsForAbsentFunction` derives them:
408/// the first equality matcher of a name sets that label, any other matcher of the name removes
409/// it, and `__name__` is skipped.
410///
411/// Prometheus reads the matchers after TSDB's `PostingsForMatchers` has reordered them in place,
412/// moving the matchers that narrow the result before those that only subtract from it; that is
413/// the order users see whenever the storage holds data, and the order used here. It changes the
414/// result only when a name has both an equality and another matcher.
415pub fn absent_labels(matchers: &[LabelMatcher]) -> Result<SeriesLabels> {
416    let must_be_set: Vec<&str> = matchers
417        .iter()
418        .filter(|matcher| !matcher.matches_empty)
419        .map(|matcher| &*matcher.name)
420        .collect();
421    let subtracting = |matcher: &LabelMatcher| {
422        !must_be_set.contains(&&*matcher.name)
423            || (matches!(matcher.op, MatchOp::NotEqual | MatchOp::NotRegex)
424                && matcher.matches_empty)
425    };
426    let ordered = matchers
427        .iter()
428        .filter(|matcher| !subtracting(matcher))
429        .chain(matchers.iter().filter(|matcher| subtracting(matcher)));
430
431    let mut labels: Vec<Label> = Vec::new();
432    let mut set: Vec<&str> = Vec::new();
433    for matcher in ordered {
434        if &*matcher.name == "__name__" {
435            continue;
436        }
437        if matcher.op == MatchOp::Equal && !set.contains(&&*matcher.name) {
438            labels.retain(|label| label.name != matcher.name);
439            labels.push(Label {
440                name: matcher.name.clone(),
441                value: matcher.value.clone(),
442            });
443            set.push(&matcher.name);
444        } else {
445            labels.retain(|label| label.name != matcher.name);
446        }
447    }
448    labels.retain(|label| !label.value.is_empty());
449    SeriesLabels::try_new(None, labels).map_err(external)
450}

452/// Orders the series of an instant result. A range result keeps its order: Prometheus does not
453/// sort matrices.
454///
455/// `sort` and `sort_desc` drop histogram samples, in a range result too: Prometheus filters them
456/// out at every step (`filterFloats`) before it sorts. The label sorts keep every sample.
457pub fn sort_instant(
458    order: &SortOrder,
459    schema: &SeriesSchema,
460    mut series: Vec<Series>,
461) -> Result<Vec<Series>> {
462    let points = grid_points(schema, &series)?;
463    if matches!(order, SortOrder::Asc | SortOrder::Desc) {
464        single_field(schema, "sort")?;
465        let mut floats = Vec::with_capacity(series.len());
466        for item in series {
467            floats.extend(float_samples(schema, item)?);
468        }
469        series = floats;
470    }
471    if points != 1 {
472        return Ok(series);
473    }
474    match order {
475        SortOrder::Asc | SortOrder::Desc => {
476            let mut keyed = Vec::with_capacity(series.len());
477            for item in series {
478                let value = value_at(&item, 0)?;
479                keyed.push((value.unwrap_or(f64::NAN), item));
480            }
481            let descending = *order == SortOrder::Desc;
482            // `sort.Sort(sort.Reverse(h))` with Prometheus' heap `Less`, which holds whenever
483            // its first value is NaN.
484            go_sort::sort_by(&mut keyed, |(a, _), (b, _)| {
485                b.is_nan() || if descending { b < a } else { b > a }
486            });
487            Ok(keyed.into_iter().map(|(_, item)| item).collect())
488        }
489        SortOrder::ByLabel(labels) | SortOrder::ByLabelDesc(labels) => {
490            let descending = matches!(order, SortOrder::ByLabelDesc(_));
491            go_sort::sort_by(&mut series, |a, b| {
492                for label in labels {
493                    let (a, b) = (
494                        a.metric.get(label).unwrap_or(""),
495                        b.metric.get(label).unwrap_or(""),
496                    );
497                    if a == b {
498                        continue;
499                    }
500                    return natural_less(a, b) != descending;
501                }
502                let order = compare_labels(&a.metric, &b.metric);
503                if descending {
504                    order == Ordering::Greater
505                } else {
506                    order == Ordering::Less
507                }
508            });
509            Ok(series)
510        }
511    }
512}

514/// `series` without its histogram samples, `None` when nothing else is left. One field, as `sort`
515/// reads.
516fn float_samples(schema: &SeriesSchema, series: Series) -> Result<Option<Series>> {
517    let field = &series.fields.as_slice()[0];
518    if !field
519        .iter()
520        .any(|sample| matches!(sample, SampleRef::Histogram(_)))
521    {
522        return Ok(Some(series));
523    }
524    let mut builder = GridSeriesBuilder::try_new(Arc::new(schema.clone()))?;
525    let mut cursor = field.cursor();
526    for (point, row) in series.time.positions() {
527        let value = match cursor.at(row).expect("the axis and the fields are aligned") {
528            SampleRef::Float(value) => Some(value),
529            SampleRef::Missing | SampleRef::Histogram(_) => None,
530        };
531        builder.push(point, &[value])?;
532    }
533    builder.finish(series.metric)
534}

536/// Go's `math.NaN()` bit pattern, which Prometheus returns for an undefined scalar.
537const GO_NAN: f64 = f64::from_bits(0x7FF8_0000_0000_0001);
538
539/// The order of `a` and `b` for `topk` (`top`) or `bottomk`; NaN ranks last for both.
540fn rank(a: f64, b: f64, top: bool) -> Ordering {
541    match (a.is_nan(), b.is_nan()) {
542        (true, true) => Ordering::Equal,
543        (true, false) => Ordering::Greater,
544        (false, true) => Ordering::Less,
545        _ if top => b.partial_cmp(&a).expect("not NaN"),
546        _ => a.partial_cmp(&b).expect("not NaN"),
547    }
548}

550/// The `k` of `topk`, `bottomk` and `limitk` as Prometheus reads it: nothing for `k` below 1, an
551/// error for NaN or a value out of int64, otherwise `k` truncated and capped at the number of
552/// input series.
553pub(crate) fn k_param(k: f64, series: usize) -> Result<Option<usize>> {
554    if k < 1.0 {
555        return Ok(None);
556    }
557    if k.is_nan() {
558        return Err(DataFusionError::Execution("Parameter value is NaN".into()));
559    }
560    // Prometheus' `maxInt64`, the largest float64 below 2^63.
561    if k >= 9_223_372_036_854_774_784.0 {
562        return Err(DataFusionError::Execution(format!(
563            "Scalar value {k} overflows int64"
564        )));
565    }
566    Ok(Some(k.trunc().min(series as f64) as usize))
567}

569/// Validates the complete parameter series for Prometheus topk/bottomk/limitk semantics.
570///
571/// The values are read where the node holds them, in one pass. The caller keeps both the series the
572/// parameter arrived as and the values it was read into, so a second copy of the present ones would
573/// be memory neither of them is charged for.
574pub(crate) fn validate_topk_parameters(values: &[Option<f64>]) -> Result<()> {
575    // A position the series decided nothing at is no parameter value, so only the present ones are
576    // read. NaN is tracked rather than folded into the bounds: it is a value Prometheus refuses
577    // outright, not the largest one, and it is refused even where every value is below 1.
578    let (mut max, mut min, mut nan) = (f64::NEG_INFINITY, f64::INFINITY, false);
579    for value in values.iter().flatten() {
580        if value.is_nan() {
581            nan = true;
582        } else {
583            max = max.max(*value);
584            min = min.min(*value);
585        }
586    }
587    if nan {
588        return Err(DataFusionError::Execution("Parameter value is NaN".into()));
589    }
590    // A `k` below 1 selects nothing, so a value that would not fit an int64 is never read.
591    if max < 1.0 {
592        return Ok(());
593    }
594    if min <= -9_223_372_036_854_775_808.0 {
595        return Err(DataFusionError::Execution(format!(
596            "Scalar value {min} underflows int64"
597        )));
598    }
599    if max >= 9_223_372_036_854_774_784.0 {
600        return Err(DataFusionError::Execution(format!(
601            "Scalar value {max} overflows int64"
602        )));
603    }
604    Ok(())
605}

607/// Label pairs identifying a group; empty values are absent, as in Prometheus.
608type GroupKey = Vec<(Arc<str>, Arc<str>)>;
609
610fn group_key(metric: &SeriesLabels, grouping: &Grouping) -> GroupKey {
611    let mut key: Vec<_> = labels_with_name(metric)
612        .into_iter()
613        .filter(|(name, value)| {
614            !value.is_empty()
615                && match grouping {
616                    Grouping::By(names) => names.contains(name),
617                    Grouping::Without(names) => &**name != "__name__" && !names.contains(name),
618                }
619        })
620        .collect();
621    key.sort();
622    key
623}
624
625/// The label pairs of `metric` with `__name__` among them, sorted by name.
626fn labels_with_name(metric: &SeriesLabels) -> Vec<(Arc<str>, Arc<str>)> {
627    let mut labels: Vec<_> = metric
628        .tags()
629        .iter()
630        .map(|tag| (tag.name.clone(), tag.value.clone()))
631        .collect();
632    if let Some(name) = metric.name() {
633        labels.push((Arc::from("__name__"), Arc::from(name)));
634    }
635    labels.sort();
636    labels
637}

639/// Prometheus `labels.Compare`: label pairs in name order, then the shorter set first.
640fn compare_labels(a: &SeriesLabels, b: &SeriesLabels) -> Ordering {
641    labels_with_name(a).cmp(&labels_with_name(b))
642}

644/// `github.com/facette/natsort.Compare`, which `sort_by_label` uses: digit runs compare as
645/// integers when both parse, other runs as strings.
646fn natural_less(a: &str, b: &str) -> bool {
647    let (chunks_a, chunks_b) = (chunks(a), chunks(b));
648    for (i, chunk_a) in chunks_a.iter().enumerate() {
649        let Some(chunk_b) = chunks_b.get(i) else {
650            return false;
651        };
652        let equal = match (chunk_a.parse::<i64>(), chunk_b.parse::<i64>()) {
653            (Ok(a), Ok(b)) if a != b => return a < b,
654            (Ok(_), Ok(_)) => true,
655            _ => chunk_a == chunk_b,
656        };
657        if !equal {
658            return chunk_a < chunk_b;
659        }
660        if i == chunks_a.len() - 1 {
661            return true;
662        } else if i == chunks_b.len() - 1 {
663            return false;
664        }
665    }
666    false
667}
668
669/// Maximal runs of ASCII digits and of other characters, like the regexp `(\d+|\D+)`.
670fn chunks(s: &str) -> Vec<&str> {
671    let mut chunks = Vec::new();
672    let mut start = 0;
673    let bytes = s.as_bytes();
674    for i in 1..=bytes.len() {
675        if i == bytes.len() || bytes[i].is_ascii_digit() != bytes[start].is_ascii_digit() {
676            chunks.push(&s[start..i]);
677            start = i;
678        }
679    }
680    chunks
681}

693/// Checks every series against `schema` and returns the grid length.
694fn grid_points(schema: &SeriesSchema, series: &[Series]) -> Result<usize> {
695    Ok(grid_and_points(schema, series)?.1)
696}

698/// Checks every series against `schema` and returns the grid it validated them on, with its length.
699///
700/// An operator that addresses points on the grid needs the grid itself, not just its length, and
701/// resolving it twice would let the two answers disagree.
702fn grid_and_points(schema: &SeriesSchema, series: &[Series]) -> Result<(Arc<EvalGrid>, usize)> {
703    let grid = schema.eval_grid.as_ref().ok_or_else(|| {
704        DataFusionError::Execution("select operators need an evaluation grid".into())
705    })?;
706    for item in series {
707        // Both grid axes are addressed on a grid; what matters is that it is this schema's.
708        let matches = match item.time.grid() {
709            Some(item_grid) => {
710                item_grid.spec() == grid.spec() && item.fields.len() == schema.fields.len()
711            }
712            None => false,
713        };
714        if !matches {
715            return Err(DataFusionError::Execution(
716                "series does not match the select operator schema".into(),
717            ));
718        }
719    }
720    Ok((Arc::clone(grid), grid.spec().len as usize))
721}

723/// The first field's value of `series` at grid `point`, looked up by position.
724///
725/// A point read: its only caller, `sort_instant`, reads one point of each instant series, so there
726/// is no walk for a cursor to follow. The row of a position is searched for, not counted from the
727/// first one, and a position the axis does not carry answers `None` rather than a row read from
728/// another offset.
729fn value_at(series: &Series, point: usize) -> Result<Option<f64>> {
730    let Some(row) = series.time.row_of(point) else {
731        return Ok(None);
732    };
733    // `Samples::sample` walks the chunk descriptors, not the rows, which is what a random read
734    // costs; a cursor would have to replay every row before this one.
735    into_value(series.fields.as_slice()[0].sample(row))
736}
737
738/// A sample as a value for an operator that ranks or sorts reads it.
739///
740/// The callers read floats only -- `sort` drops the histogram samples before it keys a series by a
741/// value, and `topk` collects float candidates -- so a histogram here is a broken invariant rather
742/// than an input.
743fn into_value(sample: SampleRef<'_>) -> Result<Option<f64>> {
744    match sample {
745        SampleRef::Missing => Ok(None),
746        SampleRef::Float(value) => Ok(Some(value)),
747        SampleRef::Histogram(_) => Err(DataFusionError::NotImplemented(
748            "select operators over histogram samples".into(),
749        )),
750    }
751}
752
753/// The value a forward cursor reads at compact `row`; the rows asked for only increase.
754fn cursor_value(cursor: &mut RowCursor<'_>, row: usize) -> Result<Option<f64>> {
755    into_value(cursor.at(row).expect("the axis and the fields are aligned"))
756}
