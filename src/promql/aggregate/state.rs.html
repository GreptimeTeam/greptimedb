15//! Partial aggregation: the state one execution partition folds its series into, and the merge
16//! of such states into the final result.
17//!
18//! A merge regroups the operations of a serial fold, so floating-point results differ from
19//! [`aggregate_float`](crate::aggregate::aggregate_float) by rounding. The same reassociation can
20//! also move an overflow of finite values: a serial `sum` of `[1e308, 1e308, -1e308]` is `+Inf`,
21//! while the merge of `[1e308]` and `[1e308, -1e308]` is finite. `stddev` and `stdvar` differ
22//! more where the samples differ by little more than the rounding of their mean: Prometheus'
23//! uncompensated Welford update rounds the mean on every sample, and no pairwise merge reproduces
24//! those roundings. For `[1e16]` and `[1e16 + 2]` Prometheus gives a variance of 2 and the merge
25//! gives the exact 1. Explicit NaN and infinite inputs follow the same rules as the serial fold. Where the serial fold keeps the first of two equal
26//! values, as `min` and `max` do for `0` and `-0`, a merge keeps the one merged first.
27

51/// A component of [`AccumulatorState`].
52#[derive(Clone, Copy, Debug, PartialEq, Eq)]
53pub enum StateField {
54    /// The running sum (`sum`, and `avg` until it switches to a mean), the extreme (`min`,
55    /// `max`), or the sum of squared deviations (`stddev`, `stdvar`).
56    Value,
57    /// The running mean (`avg` once switched, `stddev`, `stdvar`).
58    Mean,
59    /// The Kahan compensation of `Value`, or of `Mean` once `avg` has switched.
60    Compensation,
61    /// The number of samples folded in.
62    Count,
63    /// Whether `avg` has switched from a sum to a mean, which it does when the sum overflows.
64    IncrementalMean,
65}
66
67impl AggregateOp {
68    /// The components the state of this aggregation carries; the others are unused and need not
69    /// be stored. `group` has none: a state that exists has a value of 1.
70    pub fn state_fields(self) -> &'static [StateField] {
71        use StateField::*;
72        match self {
73            Self::Sum => &[Value, Compensation],
74            Self::Count => &[Count],
75            Self::Avg => &[Value, Mean, Compensation, Count, IncrementalMean],
76            Self::Min | Self::Max => &[Value],
77            Self::Group => &[],
78            Self::Stddev | Self::Stdvar => &[Value, Mean, Count],
79        }
80    }
81}

110    /// The state folded so far, for a merge with the state of other partitions.
111    pub fn finish_partial(self) -> Result<PartialAggregate> {
112        self.finish_partial_inner()
113    }
114
115    /// [`Self::finish_partial`] that admits what it builds first: the fold state and the groups it
116    /// is exported as, at the exact capacity [`Self::finish_partial_inner`] gives them. A refusal
117    /// consumes the aggregator, so the caller drops the execution.
118    pub(crate) fn finish_partial_budgeted(
119        self,
120        mut budget: impl FnMut(usize) -> Result<()>,
121    ) -> Result<PartialAggregate> {
122        let width = self.schema.fields.len();
123        let mut partial_bytes = self.groups.len().saturating_mul(size_of::<StateSeries>());
124        for group in &self.groups {
125            // A group that folded histograms keeps them in its own map instead of the blocks, so
126            // the points it exports and what each of their payloads builds come from there: the
127            // export builds two one-row Arrow batches per point, which the chunk they are handed
128            // to is charged for only once that chunk exists.
129            let (points, payloads) = if group.histograms.is_empty() {
130                (seen_points(group, width), 0)
131            } else {
132                let payloads = group
133                    .histograms
134                    .values()
135                    .fold(0usize, |bytes, accumulator| {
136                        bytes.saturating_add(exported_payload_bytes(accumulator))
137                    });
138                (group.histograms.len(), payloads)
139            };
140            partial_bytes = partial_bytes
141                .saturating_add(size_of::<TimeAxis>())
142                .saturating_add(
143                    points.saturating_mul(size_of::<GridSpan>().saturating_add(size_of::<usize>())),
144                )
145                .saturating_add(width.saturating_mul(size_of::<Vec<Option<StatePayload>>>()))
146                .saturating_add(
147                    points
148                        .saturating_mul(width)
149                        .saturating_mul(size_of::<Option<StatePayload>>()),
150                )
151                .saturating_add(payloads);
152        }
153        budget(self.state_bytes().saturating_add(partial_bytes))?;
154        self.finish_partial_inner()
155    }
156
157    /// Turns the fold into the frame's own groups: field-major states on sparse absolute positions.
158    ///
159    /// The blocks are walked once, in ascending position order, so a position's states are appended
160    /// to every field at the same time and adjacent positions coalesce into one span as they come.
161    /// What comes out is the representation the frame writes; no per-point vector is built.
162    fn finish_partial_inner(self) -> Result<PartialAggregate> {
163        let width = self.schema.fields.len();
164        let grid = self.schema.eval_grid.as_ref().ok_or_else(|| {
165            DataFusionError::Execution("aggregate state is indexed on an evaluated grid".into())
166        })?;
167        let grid_len = grid.spec().len as usize;
168        let mut groups: Vec<StateSeries> = Vec::with_capacity(self.groups.len());
169        for group in self.groups {
170            let points = if group.histograms.is_empty() {
171                seen_points(&group, width)
172            } else {
173                // A field's state is one family, so a group that kept histogram payloads exports
174                // those: a float state beside them would be one column carrying two kinds.
175                if group
176                    .blocks
177                    .values()
178                    .any(|states| states.iter().any(|state| state.seen))
179                {
180                    return Err(DataFusionError::Execution(
181                        "an aggregate field cannot carry both float and histogram state".into(),
182                    ));
183                }
184                group.histograms.len()
185            };
186            let mut spans: Vec<GridSpan> = Vec::with_capacity(points.min(1));
187            let mut states: Vec<Vec<Option<StatePayload>>> =
188                (0..width).map(|_| Vec::with_capacity(points)).collect();
189            if !group.histograms.is_empty() {
190                // The histogram fold is keyed by absolute position, and its keys are ordered, so one
191                // pass visits each carried position once with its fields' payloads.
192                let mut position = None;
193                for (&(point, field), accumulator) in &group.histograms {
194                    if position != Some(point) {
195                        if point >= grid_len {
196                            return Err(DataFusionError::Execution(format!(
197                                "aggregate state point {point} exceeds a grid of {grid_len}"
198                            )));
199                        }
200                        extend_spans(&mut spans, point);
201                        for field_states in &mut states {
202                            field_states.push(None);
203                        }
204                        position = Some(point);
205                    }
206                    // The row just opened is the one this position's payloads belong to. It is the
207                    // row the field's states just grew, not the one the span *count* names: adjacent
208                    // points share one span while each of them still opens its own row, so a run
209                    // index would write every point of the run into the run's first row, leave the
210                    // rest empty, and the frame would drop them.
211                    let row = states[field].len() - 1;
212                    states[field][row] = Some(accumulator.payload(self.op)?);
213                }
214                let time = TimeAxis::from_grid_spans(Arc::clone(grid), spans)
215                    .map_err(|error| DataFusionError::Execution(error.to_string()))?;
216                groups.push(StateSeries {
217                    metric: group.metric,
218                    time,
219                    states,
220                });
221                continue;
222            }
223            for (&block, block_states) in &group.blocks {
224                for (offset, point) in block_states.chunks_exact(width).enumerate() {
225                    if !point.iter().any(|state| state.seen) {
226                        continue;
227                    }
228                    let position = block * BLOCK_POINTS + offset;
229                    if position >= grid_len {
230                        return Err(DataFusionError::Execution(format!(
231                            "aggregate state point {position} exceeds a grid of {grid_len}"
232                        )));
233                    }
234                    extend_spans(&mut spans, position);
235                    for (field, state) in point.iter().enumerate() {
236                        states[field].push(state.partial().map(StatePayload::Float));
237                    }
238                }
239            }
240            let time = TimeAxis::from_grid_spans(Arc::clone(grid), spans)
241                .map_err(|error| DataFusionError::Execution(error.to_string()))?;
242            groups.push(StateSeries {
243                metric: group.metric,
244                time,
245                states,
246            });
247        }
248        Ok(PartialAggregate {
249            op: self.op,
250            groups,
251        })
252    }
253}

255/// Appends `position` to `spans`, extending the last run when it continues.
256fn extend_spans(spans: &mut Vec<GridSpan>, position: usize) {
257    match spans.last_mut() {
258        Some(span) if span.start() + span.len() == position => {
259            *span = GridSpan::new(span.start(), span.len() + 1);
260        }
261        _ => spans.push(GridSpan::new(position, 1)),
262    }
263}

265/// Bytes one one-row batch of the frame's histogram layout costs before its bucket data.
266///
267/// Every column of the layout owns an Arrow array node with its own buffers, and that cost does not
268/// shrink for a histogram with one bucket: one row measures 2.6 KB whatever the bucket count, which
269/// is what a payload reports once the batch exists to be measured. It is rounded up here, so a batch
270/// whose bucket lists land at the top of their buffer capacity is covered too.
271pub(super) const HISTOGRAM_LAYOUT_ROW_BYTES: usize = 3 * 1024;
272
273/// Bytes one exported point's payload builds: the boxed state and the two one-row batches
274/// [`HistogramAccumulator::payload`] makes from the accumulator's live histogram and its
275/// compensation.
276///
277/// It is counted from the histograms the payload is built from rather than from the batches, which
278/// is the point: this is what the export admits before those batches exist. The bucket data a batch
279/// writes is charged at the width of the padded Rust pair, which bounds the `f64` each bucket costs,
280/// and the layout is charged per batch.
281pub(super) fn exported_payload_bytes(accumulator: &HistogramAccumulator) -> usize {
282    // The batch the payload builds from the live value is the mean once an average switched, and
283    // the other one is the compensation paired with it: the widest of the three bounds both.
284    let widest = [
285        Some(&accumulator.value),
286        accumulator.mean.as_ref(),
287        accumulator.compensation.as_ref(),
288    ]
289    .into_iter()
290    .flatten()
291    .map(|histogram| histogram.heap_bytes())
292    .max()
293    .unwrap_or(0);
294    size_of::<HistogramState>().saturating_add(
295        HISTOGRAM_LAYOUT_ROW_BYTES
296            .saturating_add(widest)
297            .saturating_mul(2),
298    )
299}

301/// The points of `group` that carry a state: the ones a partial of it holds.
302fn seen_points(group: &Group, width: usize) -> usize {
303    group.blocks.values().fold(0usize, |count, states| {
304        count.saturating_add(
305            states
306                .chunks_exact(width)
307                .filter(|point| point.iter().any(|state| state.seen))
308                .count(),
309        )
310    })
311}

313/// Bytes a group's label key holds: a pair per capacity slot, and the `__name__` entry, whose
314/// label and value `label_key` copies, each with its own `Arc` header. The other label strings are
315/// the metric's, already counted with it.
316fn key_bytes(key: &LabelKey) -> usize {
317    let name = key
318        .iter()
319        .find(|(label, _)| &**label == "__name__")
320        .map_or(0, |(label, value)| {
321            label
322                .len()
323                .saturating_add(value.len())
324                .saturating_add(4 * size_of::<usize>())
325        });
326    key.capacity()
327        .saturating_mul(size_of::<(Arc<str>, Arc<str>)>())
328        .saturating_add(name)
329}
330
331/// Coarse bytes the key of `metric` holds while it is built: a pair per slot, the `__name__` label
332/// and a copy of the metric name, each with its `Arc` header.
333fn key_workspace_bytes(metric: &SeriesLabels) -> usize {
334    metric
335        .tags()
336        .len()
337        .saturating_add(1)
338        .saturating_mul(size_of::<(Arc<str>, Arc<str>)>())
339        .saturating_add("__name__".len())
340        .saturating_add(metric.name().map_or(0, str::len))
341        .saturating_add(4 * size_of::<usize>())
342        .saturating_add(size_of::<LabelKey>())
343}

382    /// Merges one Arrow-backed state row directly, without materializing a partial aggregate.
383    pub(crate) fn merge_row(
384        &mut self,
385        row: &StateRow,
386        mut budget: impl FnMut(usize) -> Result<()>,
387    ) -> Result<()> {
388        if row.fields.kind().is_histogram() {
389            return self.merge_histogram_row(row, budget);
390        }
391        let op = crate::native::state::op_of_state(row.kind);
392        if op != self.op {
393            return Err(DataFusionError::Execution(format!(
394                "cannot merge {:?} state into {:?}",
395                op, self.op
396            )));
397        }
398        let width = self.schema.fields.len();
399        if row.fields.len() != width {
400            return Err(DataFusionError::Execution(
401                "aggregate state does not match the grid or fields of the merge".into(),
402            ));
403        }
404        let grid_len = self
405            .schema
406            .eval_grid
407            .as_ref()
408            .expect("checked at construction")
409            .spec()
410            .len as usize;
411        let mut opened = None;
412        let key_workspace = key_workspace_bytes(&row.metric);
413        let mut live = None;
414        for (point, position) in row.time.positions() {
415            if point >= grid_len {
416                return Err(DataFusionError::Execution(
417                    "aggregate state does not match the grid or fields of the merge".into(),
418                ));
419            }
420            let mut present = false;
421            for field in 0..width {
422                present |= row
423                    .fields
424                    .is_present(field, position)
425                    .map_err(crate::grid_series::external)?;
426            }
427            if !present {
428                return Err(DataFusionError::Execution(
429                    "aggregate state names a position with no contribution".into(),
430                ));
431            }
432            let index = match opened {
433                Some(index) => index,
434                None => {
435                    budget(self.retained.saturating_add(key_workspace))?;
436                    let key = label_key(&row.metric);
437                    let live_bytes = key_bytes(&key);
438                    let index = match self.group_index.get(&key).copied() {
439                        Some(index) => index,
440                        None => {
441                            self.open_group(row.metric.clone(), key, live_bytes, &mut budget)?
442                        }
443                    };
444                    live = Some(live_bytes);
445                    opened = Some(index);
446                    index
447                }
448            };
449            let mut field = 0;
450            let mut state_error = None;
451            let states = std::iter::from_fn(|| {
452                if field == width || state_error.is_some() {
453                    return None;
454                }
455                let current = field;
456                field += 1;
457                match state_at(&row.fields, current, position) {
458                    Ok(state) => Some(state),
459                    Err(error) => {
460                        state_error = Some(error);
461                        None
462                    }
463                }
464            });
465            self.write_point(index, point, states, live.unwrap_or(0), &mut budget)?;
466            if let Some(error) = state_error {
467                return Err(error);
468            }
469        }
470        Ok(())
471    }

473    /// Merges one Arrow-backed histogram state row: every present field's payload is merged into
474    /// that position's own state.
475    ///
476    /// A histogram state is sparse in both directions - a position is carried when any field has a
477    /// contribution there, and a field may be absent at a carried position - so the row is walked
478    /// position by position and each present component is merged where it belongs.
479    fn merge_histogram_row(
480        &mut self,
481        row: &StateRow,
482        mut budget: impl FnMut(usize) -> Result<()>,
483    ) -> Result<()> {
484        if crate::native::state::op_of_state(row.kind) != self.op {
485            return Err(DataFusionError::Execution(format!(
486                "cannot merge {:?} state into {:?}",
487                row.kind, self.op
488            )));
489        }
490        let width = self.schema.fields.len();
491        if row.fields.len() != width {
492            return Err(DataFusionError::Execution(
493                "aggregate state does not match the grid or fields of the merge".into(),
494            ));
495        }
496        let grid_len = self
497            .schema
498            .eval_grid
499            .as_ref()
500            .expect("checked at construction")
501            .spec()
502            .len as usize;
503        let mut opened = None;
504        for (point, position) in row.time.positions() {
505            if point >= grid_len {
506                return Err(DataFusionError::Execution(
507                    "aggregate state does not match the grid or fields of the merge".into(),
508                ));
509            }
510            let index = match opened {
511                Some(index) => index,
512                None => {
513                    let index = self.open_row_group(&row.metric, &mut budget)?;
514                    opened = Some(index);
515                    index
516                }
517            };
518            let mut present = false;
519            for field in 0..width {
520                let Some(payload) = histogram_state_at(&row.fields, field, position)? else {
521                    continue;
522                };
523                present = true;
524                self.merge_payload(index, point, field, &payload, &mut budget)?;
525            }
526            if !present {
527                return Err(DataFusionError::Execution(
528                    "aggregate state names a position with no contribution".into(),
529                ));
530            }
531        }
532        Ok(())
533    }

535    /// The group this row's labels name, opening it (and its key) on first sight.
536    fn open_row_group(
537        &mut self,
538        metric: &SeriesLabels,
539        budget: &mut impl FnMut(usize) -> Result<()>,
540    ) -> Result<usize> {
541        budget(self.retained.saturating_add(key_workspace_bytes(metric)))?;
542        let key = label_key(metric);
543        let live = key_bytes(&key);
544        match self.group_index.get(&key).copied() {
545            Some(index) => Ok(index),
546            None => self.open_group(metric.clone(), key, live, budget),
547        }
548    }

550    /// Merges one field's payload into the state of `(point, field)`, admitting what it allocates.
551    fn merge_payload(
552        &mut self,
553        index: usize,
554        point: usize,
555        field: usize,
556        payload: &StatePayload,
557        budget: &mut impl FnMut(usize) -> Result<()>,
558    ) -> Result<()> {
559        let StatePayload::Histogram(payload) = payload else {
560            return Err(DataFusionError::Execution(
561                "a histogram state carries a numeric payload".into(),
562            ));
563        };
564        let key = (point, field);
565        let histograms = &mut self.groups[index].histograms;
566        let vacant = !histograms.contains_key(&key);
567        let backing = if vacant {
568            histogram_backing_bytes(histograms.len() + 1)
569                .saturating_sub(histogram_backing_bytes(histograms.len()))
570        } else {
571            0
572        };
573        if vacant {
574            // Admit the state this position is about to hold before allocating it: both histograms
575            // the payload carries and the map node that holds them.
576            let incoming = payload.value.as_arrow().get_array_memory_size()
577                + payload.compensation.as_arrow().get_array_memory_size();
578            budget(
579                self.retained
580                    .saturating_add(size_of::<HistogramAccumulator>())
581                    .saturating_add(incoming)
582                    .saturating_add(backing),
583            )?;
584            let accumulator = HistogramAccumulator::from_payload(self.op, payload)?;
585            let bytes = accumulator_bytes(&accumulator).saturating_add(backing);
586            histograms.insert(key, accumulator);
587            self.retained = self.retained.saturating_add(bytes);
588            return Ok(());
589        }
590        let accumulator = histograms.get_mut(&key).expect("checked as occupied above");
591        let before = accumulator_bytes(accumulator);
592        let incoming = payload.value.as_arrow().get_array_memory_size()
593            + payload.compensation.as_arrow().get_array_memory_size();
594        budget(
595            self.retained
596                .saturating_add(before)
597                .saturating_add(incoming),
598        )?;
599        accumulator.merge_state(self.op, payload)?;
600        self.retained = self
601            .retained
602            .saturating_sub(before)
603            .saturating_add(accumulator_bytes(accumulator));
604        Ok(())
605    }

607    /// Adds a group and its key, admitting the group vector and index growth first.
608    fn open_group(
609        &mut self,
610        metric: SeriesLabels,
611        key: LabelKey,
612        live: usize,
613        budget: &mut impl FnMut(usize) -> Result<()>,
614    ) -> Result<usize> {
615        let capacity = self.groups.capacity();
616        let next_capacity = if self.groups.len() == capacity {
617            capacity.saturating_mul(2).max(4)
618        } else {
619            capacity
620        };
621        let map_capacity = self.group_index.capacity();
622        let next_map_capacity = if self.group_index.len().saturating_add(1) > map_capacity {
623            next_hash_capacity(map_capacity)
624        } else {
625            map_capacity
626        };
627        let key_live = key_bytes(&key);
628        let metric_live = metric_bytes(&metric);
629        // A realloc that cannot grow in place keeps both buffers alive while it copies, so a
630        // growing step admits the new buffer beside the old one, which `retained` already holds.
631        budget(
632            self.retained
633                .saturating_add(live)
634                .saturating_add(key_live)
635                .saturating_add(metric_live)
636                .saturating_add(if next_capacity > capacity {
637                    next_capacity.saturating_mul(size_of::<Group>())
638                } else {
639                    0
640                })
641                .saturating_add(if next_map_capacity > map_capacity {
642                    hash_map_backing_bytes(next_map_capacity)
643                } else {
644                    0
645                }),
646        )?;
647        self.groups.try_reserve(1).map_err(|error| {
648            DataFusionError::Execution(format!("cannot reserve aggregate groups: {error}"))
649        })?;
650        self.groups.push(Group {
651            metric,
652            blocks: BTreeMap::new(),
653            histograms: BTreeMap::new(),
654        });
655        let index = self.groups.len() - 1;
656        self.group_index.insert(key, index);
657        self.retained = self
658            .retained
659            .saturating_add(
660                self.groups
661                    .capacity()
662                    .saturating_sub(capacity)
663                    .saturating_mul(size_of::<Group>()),
664            )
665            .saturating_add(
666                hash_map_backing_bytes(self.group_index.capacity())
667                    .saturating_sub(hash_map_backing_bytes(map_capacity)),
668            )
669            .saturating_add(metric_live)
670            .saturating_add(key_live);
671        Ok(index)
672    }
673
674    /// Writes one point's states, admitting a missing block before allocating it.
675    fn write_point(
676        &mut self,
677        index: usize,
678        point: usize,
679        states: impl Iterator<Item = Option<AccumulatorState>>,
680        live: usize,
681        budget: &mut impl FnMut(usize) -> Result<()>,
682    ) -> Result<()> {
683        let width = self.schema.fields.len();
684        let block_bytes = BLOCK_POINTS
685            .saturating_mul(width)
686            .saturating_mul(size_of::<FloatState>());
687        let blocks = &mut self.groups[index].blocks;
688        let block_index = point / BLOCK_POINTS;
689        let entries = blocks.len();
690        let block = match blocks.entry(block_index) {
691            BTreeEntry::Occupied(entry) => entry.into_mut(),
692            BTreeEntry::Vacant(entry) => {
693                budget(
694                    self.retained
695                        .saturating_add(live)
696                        .saturating_add(block_bytes)
697                        .saturating_add(btree_backing_bytes(entries.saturating_add(1))),
698                )?;
699                self.retained = self
700                    .retained
701                    .saturating_add(block_bytes)
702                    .saturating_add(btree_backing_bytes(entries.saturating_add(1)))
703                    .saturating_sub(btree_backing_bytes(entries));
704                entry.insert(vec![FloatState::default(); BLOCK_POINTS * width].into_boxed_slice())
705            }
706        };
707        let offset = (point % BLOCK_POINTS) * width;
708        for (slot, state) in block[offset..offset + width].iter_mut().zip(states) {
709            if let Some(state) = state {
710                slot.merge(self.op, FloatState::from_partial(state));
711            }
712        }
713        Ok(())
714    }
715
716    /// The result of every group, in the order the groups first arrived.
717    pub fn finish(self) -> Result<Vec<Series>> {
718        finish_groups(self.schema, self.op, self.groups)
719    }
720
721    /// [`Self::finish`] that admits the result it builds, on top of what the merge still holds. The
722    /// merge is consumed either way, so a caller that budgets drops it on a refusal.
723    pub(crate) fn finish_budgeted(
724        self,
725        budget: impl FnMut(usize) -> Result<()>,
726    ) -> Result<Vec<Series>> {
727        finish_groups_budgeted(self.schema, self.op, self.groups, self.retained, budget)
728    }

731/// One present field's histogram state at one position, read from the row's component arrays.
732///
733/// The two payload components are decoded into the kernel's own histograms, and the components a
734/// kind may omit - an average's count and mode flag - default to what the merge would fold from.
735fn histogram_state_at(
736    fields: &StateFieldsBatch,
737    field: usize,
738    position: usize,
739) -> Result<Option<StatePayload>> {
740    if !fields
741        .is_present(field, position)
742        .map_err(crate::grid_series::external)?
743    {
744        return Ok(None);
745    }
746    let mut state = datatypes::series::aggregate_state::HistogramState {
747        value: Arc::new(payload_histogram(
748            fields,
749            field,
750            StateComponent::HistogramValue,
751            position,
752        )?),
753        compensation: Arc::new(payload_histogram(
754            fields,
755            field,
756            StateComponent::HistogramCompensation,
757            position,
758        )?),
759        count: 0.0,
760        incremental_mean: false,
761        poisoned: false,
762        incompatible_layout: false,
763        custom_bounds_reconciled: false,
764        counter_reset_seen: false,
765        not_counter_reset_seen: false,
766    };
767    for &component in fields.components() {
768        match component {
769            StateComponent::HistogramValue | StateComponent::HistogramCompensation => {}
770            StateComponent::Count => {
771                state.count = payload_float(fields, field, component, position)?
772            }
773            StateComponent::IncrementalMean
774            | StateComponent::Poisoned
775            | StateComponent::IncompatibleLayout
776            | StateComponent::CustomBoundsReconciled
777            | StateComponent::CounterResetSeen
778            | StateComponent::NotCounterResetSeen => {
779                let flag = payload_bool(fields, field, component, position)?;
780                match component {
781                    StateComponent::IncrementalMean => state.incremental_mean = flag,
782                    StateComponent::Poisoned => state.poisoned = flag,
783                    StateComponent::IncompatibleLayout => state.incompatible_layout = flag,
784                    StateComponent::CustomBoundsReconciled => state.custom_bounds_reconciled = flag,
785                    StateComponent::CounterResetSeen => state.counter_reset_seen = flag,
786                    _ => state.not_counter_reset_seen = flag,
787                }
788            }
789            StateComponent::Value | StateComponent::Mean | StateComponent::Compensation => {
790                return Err(DataFusionError::Execution(
791                    "a histogram state declares a numeric component".into(),
792                ));
793            }
794        }
795    }
796    Ok(Some(StatePayload::Histogram(Box::new(state))))
797}

841/// One present field's state at one position, read from the row's component arrays.
842///
843/// The components come from the row's own declaration; the native aggregate caller checks
844/// compatibility before this runs, so a component list that disagrees is caught before this runs.
845fn state_at(
846    fields: &StateFieldsBatch,
847    field: usize,
848    position: usize,
849) -> Result<Option<AccumulatorState>> {
850    if !fields
851        .is_present(field, position)
852        .map_err(crate::grid_series::external)?
853    {
854        return Ok(None);
855    }
856    let mut state = AccumulatorState::default();
857    for &component in fields.components() {
858        match component {
859            StateComponent::IncrementalMean => {
860                state.incremental_mean = fields
861                    .field_bool_values(field, component)
862                    .map_err(crate::grid_series::external)?
863                    .and_then(|array| (!array.is_null(position)).then(|| array.value(position)))
864                    .ok_or_else(|| {
865                        DataFusionError::Execution(
866                            "present aggregate state has a missing component".into(),
867                        )
868                    })?;
869            }
870            // A histogram state's payload is not numeric components, so this merge cannot read one:
871            // [`FloatMerger::merge_row`] sends a histogram row to `merge_histogram_row`, so a
872            // histogram component reaching here is a row whose kind says numeric and whose payload
873            // says otherwise. Refusing is what keeps a histogram from being folded as the numeric
874            // state it is not.
875            StateComponent::HistogramValue
876            | StateComponent::HistogramCompensation
877            | StateComponent::Poisoned
878            | StateComponent::IncompatibleLayout
879            | StateComponent::CustomBoundsReconciled
880            | StateComponent::CounterResetSeen
881            | StateComponent::NotCounterResetSeen => {
882                return Err(DataFusionError::Execution(
883                    "histogram aggregate state reached the numeric merge".into(),
884                ));
885            }
886            _ => {
887                let value = fields
888                    .field_values(field, component)
889                    .map_err(crate::grid_series::external)?
890                    .and_then(|array| (!array.is_null(position)).then(|| array.value(position)))
891                    .ok_or_else(|| {
892                        DataFusionError::Execution(
893                            "present aggregate state has a missing component".into(),
894                        )
895                    })?;
896                match component {
897                    StateComponent::Value => state.value = value,
898                    StateComponent::Mean => state.mean = value,
899                    StateComponent::Compensation => state.compensation = value,
900                    StateComponent::Count => state.count = value,
901                    _ => unreachable!("every other component is handled above"),
902                }
903            }
904        }
905    }
906    Ok(Some(state))
907}

909impl FloatState {
910    fn partial(&self) -> Option<AccumulatorState> {
911        self.seen.then_some(AccumulatorState {
912            value: self.value,
913            mean: self.mean,
914            compensation: self.compensation,
915            count: self.count,
916            incremental_mean: self.incremental_mean,
917        })
918    }
919
920    fn from_partial(state: AccumulatorState) -> Self {
921        Self {
922            seen: true,
923            value: state.value,
924            mean: state.mean,
925            compensation: state.compensation,
926            count: state.count,
927            incremental_mean: state.incremental_mean,
928        }
929    }
930
931    /// Folds the state of another partition into this one.
932    fn merge(&mut self, op: AggregateOp, other: Self) {
933        if !self.seen {
934            *self = other;
935            return;
936        }
937        match op {
938            AggregateOp::Sum => {
939                (self.value, self.compensation) = add_compensated(*self, other);
940            }
941            AggregateOp::Count => self.count += other.count,
942            AggregateOp::Group => {}
943            AggregateOp::Max => {
944                if self.value < other.value || self.value.is_nan() {
945                    self.value = other.value;
946                }
947            }
948            AggregateOp::Min => {
949                if self.value > other.value || self.value.is_nan() {
950                    self.value = other.value;
951                }
952            }
953            AggregateOp::Avg => self.merge_avg(other),
954            AggregateOp::Stddev | AggregateOp::Stdvar => {
955                // Chan et al.'s pairwise update. `delta * (delta * weight)` keeps the product
956                // finite where it is: `delta²` alone overflows for `[0]` and `[1.5e154]`, whose
957                // merged sum of squares is finite.
958                let count = self.count + other.count;
959                let delta = other.mean - self.mean;
960                let weight = self.count * other.count / count;
961                self.mean += delta * (other.count / count);
962                self.value = self.value + other.value + delta * (delta * weight);
963                self.count = count;
964            }
965        }
966    }
967
968    /// Adds the sums while they stay finite, and otherwise continues with the count-weighted
969    /// mean, as the serial update switches to an incremental mean once its sum overflows.
970    fn merge_avg(&mut self, other: Self) {
971        let count = self.count + other.count;
972        if !self.incremental_mean && !other.incremental_mean {
973            let (value, compensation) = add_compensated(*self, other);
974            if !value.is_infinite() {
975                self.value = value;
976                self.compensation = compensation;
977                self.count = count;
978                return;
979            }
980        }
981        let (mean, compensation) = self.mean_form();
982        let (other_mean, other_compensation) = other.mean_form();
983        let (weight, other_weight) = (self.count / count, other.count / count);
984        (self.mean, self.compensation) = kahan_inc(
985            other_mean * other_weight,
986            mean * weight,
987            compensation * weight + other_compensation * other_weight,
988        );
989        self.incremental_mean = true;
990        self.count = count;
991    }
992
993    /// The mean and its compensation, converting a sum as the serial update does on switching.
994    fn mean_form(&self) -> (f64, f64) {
995        if self.incremental_mean {
996            (self.mean, self.compensation)
997        } else {
998            (self.value / self.count, self.compensation / self.count)
999        }
1000    }
1001}
1002
1003/// The compensated sum of two partial sums. Both the other sum and its compensation are added as
1004/// Kahan addends: adding the compensation plainly would drop what the compensation already
1005/// holds once later partials cancel the sums, as merging `[1e100, 1]`, `[-1e100, 1e84]` and
1006/// `[-1e84]` would lose the `1`.
1007fn add_compensated(this: FloatState, other: FloatState) -> (f64, f64) {
1008    let (value, compensation) = kahan_inc(other.value, this.value, this.compensation);
1009    kahan_inc(other.compensation, value, compensation)
1010}

1012/// Merges `count_values` results of several partitions. Each partial already carries its full
1013/// output label set, value label included, so series with one label set add their counts point
1014/// by point and field by field.
1015pub fn merge_count_values(
1016    schema: &Arc<SeriesSchema>,
1017    partials: impl IntoIterator<Item = Series>,
1018) -> Result<Vec<Series>> {
1019    let width = schema.fields.len();
1020    let mut group_index: HashMap<LabelKey, usize> = HashMap::new();
1021    let mut groups: Counts = Vec::new();
1022    for series in partials {
1023        validate_float_series(schema, &series)?;
1024        let index = match group_index.entry(label_key(&series.metric)) {
1025            Entry::Occupied(entry) => *entry.get(),
1026            Entry::Vacant(entry) => {
1027                groups.push((series.metric.clone(), BTreeMap::new()));
1028                *entry.insert(groups.len() - 1)
1029            }
1030        };
1031        for (field_index, field) in series.fields.iter().enumerate() {
1032            for ((point, _), sample) in series.time.positions().zip(field.iter()) {
1033                if let SampleRef::Float(count) = sample {
1034                    groups[index]
1035                        .1
1036                        .entry(point)
1037                        .or_insert_with(|| vec![0.0; width])[field_index] += count;
1038                }
1039            }
1040        }
1041    }
1042    count_series(schema, groups)
1043}
