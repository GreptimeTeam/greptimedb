28/// A time-varying, unlabelled single-field scalar, with explicit operand direction.
29#[derive(Clone, Copy, Debug)]
30pub enum ScalarSeriesOperand<'a> {
31    Left(&'a Series),
32    Right(&'a Series),
33}
34
35/// Applies arithmetic between vector fields and a time-varying scalar on the same grid, and drops
36/// the metric name.
37///
38/// Only positions both operands carry are evaluated; a missing scalar point yields a missing
39/// output, and `None` means no output row. The scalar must be unlabelled and single-field, and both
40/// inputs must hold floats only, even outside the overlap. Two scalars also work. Same-labelset
41/// merging is left to the caller.
42pub fn map_arithmetic_series(
43    schema: &Arc<SeriesSchema>,
44    vector: &Series,
45    op: ArithmeticOp,
46    scalar: ScalarSeriesOperand<'_>,
47) -> Result<Option<Series>> {
48    let metric = drop_metric_name(&vector.metric)?;
49    let (output, _) = map_grid_scalar(
50        schema,
51        vector,
52        scalar,
53        metric,
54        NonFloatSample::Impossible,
55        |left, right, _| Some(op.calculate(left, right)),
56    )?;
57    Ok(output)
58}

60/// Compares against a time-varying scalar. Filter keeps vector values and labels; bool emits
61/// 0/1 and drops the metric name.
62///
63/// Aligns like [`map_arithmetic_series`]. Each field is compared independently; a row is kept
64/// where any field passes. Histogram samples are dropped, as in
65/// [`crate::map::map_compare_literal`], and the second element reports whether one was.
66pub fn map_compare_series(
67    schema: &Arc<SeriesSchema>,
68    vector: &Series,
69    comparison: Comparison,
70    scalar: ScalarSeriesOperand<'_>,
71) -> Result<(Option<Series>, bool)> {
72    let metric = if matches!(comparison, Comparison::Bool(_)) {
73        drop_metric_name(&vector.metric)?
74    } else {
75        vector.metric.clone()
76    };
77    map_grid_scalar(
78        schema,
79        vector,
80        scalar,
81        metric,
82        NonFloatSample::Drop,
83        |left, right, sample| comparison.apply(left, right, sample),
84    )
85}

87/// `max_of(a, b)` and `min_of(a, b)`, which Prometheus evaluates with Go's `math.Max` and
88/// `math.Min`.
89#[derive(Clone, Copy, Debug, PartialEq, Eq)]
90pub(crate) enum ScalarFunction {
91    MaxOf,
92    MinOf,
93}
94
95impl ScalarFunction {
96    /// Go's `math.Max` / `math.Min`: an infinity of the function's own sign wins over NaN, NaN
97    /// over any other value, and `+0` is greater than `-0`. Symmetric in its arguments.
98    pub(crate) fn apply(self, a: f64, b: f64) -> f64 {
99        let (wins, signed_zero_wins) = match self {
100            Self::MaxOf => (f64::INFINITY, false),
101            Self::MinOf => (f64::NEG_INFINITY, true),
102        };
103        if a == wins || b == wins {
104            wins
105        } else if a.is_nan() || b.is_nan() {
106            f64::NAN
107        } else if a == 0.0 && a == b {
108            if a.is_sign_negative() == signed_zero_wins {
109                a
110            } else {
111                b
112            }
113        } else if (a > b) == (self == Self::MaxOf) {
114            a
115        } else {
116            b
117        }
118    }
119}

121/// A math function whose parameters are time-varying scalars. For `max_of` and `min_of` the
122/// mapped series is the first scalar operand.
123#[derive(Clone, Copy, Debug)]
124pub enum ParameterizedMath<'a> {
125    /// `round(v, to_nearest)`.
126    Round { to_nearest: &'a Series },
127    /// `clamp(v, min, max)`.
128    Clamp { min: &'a Series, max: &'a Series },
129    /// `clamp_min(v, min)`.
130    ClampMin { min: &'a Series },
131    /// `clamp_max(v, max)`.
132    ClampMax { max: &'a Series },
133    /// `max_of(a, b)`: the mapped series is `a`, `other` is `b`.
134    MaxOf { other: &'a Series },
135    /// `min_of(a, b)`: the mapped series is `a`, `other` is `b`.
136    MinOf { other: &'a Series },
137}
138
139impl<'a> ParameterizedMath<'a> {
140    fn parameters(self) -> Vec<&'a Series> {
141        match self {
142            Self::Round { to_nearest } => vec![to_nearest],
143            Self::Clamp { min, max } => vec![min, max],
144            Self::ClampMin { min } => vec![min],
145            Self::ClampMax { max } => vec![max],
146            Self::MaxOf { other } | Self::MinOf { other } => vec![other],
147        }
148    }
149
150    /// Binds one point's parameter values, given in argument order.
151    fn at(self, values: &[f64]) -> PointMath {
152        match self {
153            Self::Round { .. } => PointMath::Math(MathOp::Round { nearest: values[0] }),
154            Self::Clamp { .. } => PointMath::Math(MathOp::Clamp {
155                min: values[0],
156                max: values[1],
157            }),
158            Self::ClampMin { .. } => PointMath::Math(MathOp::ClampMin { min: values[0] }),
159            Self::ClampMax { .. } => PointMath::Math(MathOp::ClampMax { max: values[0] }),
160            Self::MaxOf { .. } => PointMath::Scalar {
161                function: ScalarFunction::MaxOf,
162                other: values[0],
163            },
164            Self::MinOf { .. } => PointMath::Scalar {
165                function: ScalarFunction::MinOf,
166                other: values[0],
167            },
168        }
169    }
170}

172/// A parameterized math function with one point's parameters bound.
173#[derive(Clone, Copy, Debug)]
174enum PointMath {
175    Math(MathOp),
176    Scalar {
177        function: ScalarFunction,
178        other: f64,
179    },
180}
181
182impl PointMath {
183    /// `None` for a `clamp` with `max < min`, which has no result at that point.
184    fn calculate(self, value: f64) -> Option<f64> {
185        match self {
186            Self::Math(op) => (!op.is_empty_clamp()).then(|| op.calculate(value)),
187            Self::Scalar { function, other } => Some(function.apply(value, other)),
188        }
189    }
190}

192/// Applies `function` to every float field with parameters read per point, and drops the metric
193/// name, like [`crate::map::map_math`] with constant parameters; histogram samples are skipped.
194///
195/// A point where a parameter has no value, or where `clamp` has `max < min` (Prometheus returns an
196/// empty vector for that step), has no result. `None` means no point has one. Same-labelset merging
197/// is left to the caller.
198pub fn map_math_with_parameters(
199    schema: &Arc<SeriesSchema>,
200    series: &Series,
201    function: ParameterizedMath<'_>,
202) -> Result<Option<Series>> {
203    let time = validate_grid_series(schema, series)?;
204    let parameters = function.parameters();
205    let mut parameter_axes = Vec::with_capacity(parameters.len());
206    for parameter in &parameters {
207        parameter_axes.push(validate_scalar(schema, parameter)?);
208    }
209    // Points only increase, so one forward cursor per parameter suffices.
210    let mut parameter_cursors: Vec<_> = parameters
211        .iter()
212        .map(|parameter| parameter.fields.as_slice()[0].cursor())
213        .collect();
214    let mut field_cursors: Vec<_> = series.fields.iter().map(|field| field.cursor()).collect();
215    let mut values = Vec::with_capacity(parameters.len());
216    let mut row = Vec::with_capacity(field_cursors.len());
217    let mut builder = GridSeriesBuilder::try_new(schema.clone())?;
218    for (point, index) in time.positions() {
219        values.clear();
220        for (axis, cursor) in parameter_axes.iter().zip(&mut parameter_cursors) {
221            let value = axis.row_of(point).and_then(|row| {
222                float_sample(cursor.at(row).expect("parameter rows only increase"))
223            });
224            values.extend(value);
225        }
226        let op = (values.len() == parameters.len()).then(|| function.at(&values));
227        row.clear();
228        for cursor in &mut field_cursors {
229            let sample = cursor
230                .at(index)
231                .expect("the axis and the fields are aligned");
232            row.push(match (op, sample) {
233                (Some(op), SampleRef::Float(value)) => op.calculate(value),
234                _ => None,
235            });
236        }
237        builder.push(point, &row)?;
238    }
239    builder.finish(drop_metric_name(&series.metric)?)
240}

242/// How [`map_grid_scalar`] treats a histogram sample in the vector.
243#[derive(Clone, Copy, Debug)]
244enum NonFloatSample {
245    /// The vector is validated as float-only.
246    Impossible,
247    /// The position yields nothing and the drop is reported, as Prometheus does for an operator
248    /// undefined on histograms.
249    Drop,
250}
251
252/// Calls `calculate(left, right, vector_sample)` at each position where both operands exist.
253fn map_grid_scalar(
254    schema: &Arc<SeriesSchema>,
255    vector: &Series,
256    scalar: ScalarSeriesOperand<'_>,
257    metric: SeriesLabels,
258    non_float: NonFloatSample,
259    calculate: impl Fn(f64, f64, f64) -> Option<f64>,
260) -> Result<(Option<Series>, bool)> {
261    let vector_time = match non_float {
262        NonFloatSample::Impossible => validate_float_series(schema, vector)?,
263        NonFloatSample::Drop => validate_grid_series(schema, vector)?,
264    };
265    let (scalar_series, scalar_left) = match scalar {
266        ScalarSeriesOperand::Left(s) => (s, true),
267        ScalarSeriesOperand::Right(s) => (s, false),
268    };
269    let scalar_time = validate_scalar(schema, scalar_series)?;
270    let mut builder = GridSeriesBuilder::try_new(schema.clone())?;
271    // Each operand reads its own row from the shared run; positions only one side carries yield
272    // nothing, so they are skipped.
273    let mut scalar_samples = scalar_series.fields.as_slice()[0].cursor();
274    let mut field_samples = vector
275        .fields
276        .iter()
277        .map(|field| field.cursor())
278        .collect::<Vec<_>>();
279    let mut row = Vec::with_capacity(field_samples.len());
280    let mut dropped_histogram = false;
281    for pair in vector_time.intersect_runs(scalar_time) {
282        for offset in 0..pair.len {
283            let scalar = float_sample(
284                scalar_samples
285                    .at(pair.other_row + offset)
286                    .expect("the shared run lies inside the scalar axis"),
287            );
288            row.clear();
289            for field in &mut field_samples {
290                let sample = match field
291                    .at(pair.row + offset)
292                    .expect("the shared run lies inside the vector axis")
293                {
294                    SampleRef::Float(value) => Some(value),
295                    SampleRef::Missing => None,
296                    SampleRef::Histogram(_) => {
297                        match non_float {
298                            NonFloatSample::Drop => dropped_histogram = true,
299                            NonFloatSample::Impossible => {
300                                unreachable!("validated as float")
301                            }
302                        }
303                        None
304                    }
305                };
306                row.push(scalar.zip(sample).and_then(|(scalar, sample)| {
307                    if scalar_left {
308                        calculate(scalar, sample, sample)
309                    } else {
310                        calculate(sample, scalar, sample)
311                    }
312                }));
313            }
314            builder.push(pair.position + offset, &row)?;
315        }
316    }
317    Ok((builder.finish(metric)?, dropped_histogram))
318}

320pub(crate) fn validate_scalar<'a>(
321    schema: &SeriesSchema,
322    scalar: &'a Series,
323) -> Result<&'a TimeAxis> {
324    if scalar.metric.name().is_some() || !scalar.metric.tags().is_empty() {
325        return Err(DataFusionError::Execution(
326            "grid scalar must be unlabelled".into(),
327        ));
328    }
329    let schema = SeriesSchema {
330        fields: vec![FieldSpec {
331            name: "scalar".into(),
332        }]
333        .into(),
334        eval_grid: schema.eval_grid.clone(),
335    };
336    validate_float_series(&schema, scalar)
337}
