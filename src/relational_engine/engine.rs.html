169struct EngineInner {
170    /// All regions opened by the engine.
171    ///
172    /// Writing to `regions` should also hold the `region_mutex`.
173    regions: RwLock<HashMap<RegionId, RelationalRegionRef>>,
174
175    /// Region mutex is used to protect the operations such as creating/opening/closing
176    /// a region, to avoid things like opening the same region simultaneously.
177    region_mutex: Mutex<()>,
178
179    object_store: ObjectStore,
180}
181
182type EngineInnerRef = Arc<EngineInner>;
183
184fn ensure_region_requirements(
185    requirements: RegionRequirements,
186    object_store: &ObjectStore,
187) -> EngineResult<()> {
188    if !requirements.object_storage {
189        return Ok(());
190    }
191
192    ensure!(
193        object_store::util::is_object_storage(object_store),
194        UnsupportedSnafu {
195            operation: "create or open region with object storage requirement on non-object storage"
196        }
197    );
198
199    Ok(())
200}
201
202impl EngineInner {
203    fn new(object_store: ObjectStore) -> Self {
204        Self {
205            regions: RwLock::new(HashMap::new()),
206            region_mutex: Mutex::new(()),
207            object_store,
208        }
209    }
210
211    async fn handle_request(
212        &self,
213        region_id: RegionId,
214        request: RegionRequest,
215    ) -> EngineResult<RegionResponse> {
216        let result = match request {
217            RegionRequest::Create(req) => self.handle_create(region_id, req).await,
218            RegionRequest::Drop(req) => self.handle_drop(region_id, req).await,
219            RegionRequest::Open(req) => self.handle_open(region_id, req).await,
220            RegionRequest::Close(req) => self.handle_close(region_id, req).await,
221            _ => UnsupportedSnafu {
222                operation: request.to_string(),
223            }
224            .fail(),
225        };
226        result.map(RegionResponse::new)
227    }
228
229    async fn stop(&self) -> EngineResult<()> {
230        let _lock = self.region_mutex.lock().await;
231        self.regions.write().unwrap().clear();
232        Ok(())
233    }
234
235    fn set_region_role(&self, _region_id: RegionId, _region_role: RegionRole) -> EngineResult<()> {
236        // TODO(zhongzc): Improve the semantics and implementation of this API.
237        Ok(())
238    }
239
240    fn state(&self, region_id: RegionId) -> Option<RegionRole> {
241        if self.regions.read().unwrap().get(&region_id).is_some() {
242            Some(RegionRole::Leader)
243        } else {
244            None
245        }
246    }
247}

249impl EngineInner {
250    async fn handle_create(
251        &self,
252        region_id: RegionId,
253        request: RegionCreateRequest,
254    ) -> EngineResult<AffectedRows> {
255        ensure!(
256            request.engine == RELATIONAL_ENGINE,
257            UnexpectedEngineSnafu {
258                engine: request.engine
259            }
260        );
261
262        // If the region is already open, validate the definition before treating the
263        // create as idempotent: a mismatched create must fail, not silently succeed.
264        if let Some(region) = self.get_region(region_id).await {
265            region.check_create_request(region_id, &request)?;
266            return Ok(0);
267        }
268
269        info!("Try to create region, region_id: {}", region_id);
270
271        let _lock = self.region_mutex.lock().await;
272        // Check again after acquiring the lock
273        if let Some(region) = self.get_region(region_id).await {
274            region.check_create_request(region_id, &request)?;
275            return Ok(0);
276        }
277
278        ensure_region_requirements(request.requirements, &self.object_store)?;
279
280        // The manifest is the source of truth for whether a region already
281        // exists. The in-memory `regions` map only tracks regions opened in
282        // this process, so after a restart it is empty even though the
283        // manifest was already persisted. Recover the region from the manifest
284        // instead of failing with `ManifestExists`, keeping `CREATE EXTERNAL
285        // TABLE` idempotent.
286        let region_dir = RelationalRegion::region_dir(&request.table_dir, region_id);
287        if RelationalRegionManifest::exists(&region_dir, &self.object_store).await? {
288            let manifest =
289                RelationalRegionManifest::load(region_id, &region_dir, &self.object_store).await?;
290            let region = RelationalRegion::from_manifest(&region_dir, manifest)?;
291            // The persisted definition must match the incoming request, otherwise a
292            // retried CREATE with a different schema would silently reuse the old
293            // manifest. Reject the mismatch instead.
294            region.check_create_request(region_id, &request)?;
295            self.regions.write().unwrap().insert(region_id, region);
296            info!(
297                "Region recovered from existing manifest, region_id: {}",
298                region_id
299            );
300            return Ok(0);
301        }
302
303        let res = RelationalRegion::create(region_id, request, &self.object_store).await;
304        let region = res.inspect_err(|err| {
305            error!(
306                err;
307                "Failed to create region, region_id: {}",
308                region_id
309            );
310        })?;
311        self.regions.write().unwrap().insert(region_id, region);
312
313        info!("A new region is created, region_id: {}", region_id);
314        Ok(0)
315    }
316
317    async fn handle_open(
318        &self,
319        region_id: RegionId,
320        request: RegionOpenRequest,
321    ) -> EngineResult<AffectedRows> {
322        if self.exists(region_id).await {
323            return Ok(0);
324        }
325
326        info!("Try to open region, region_id: {}", region_id);
327
328        let _lock = self.region_mutex.lock().await;
329        // Check again after acquiring the lock
330        if self.exists(region_id).await {
331            return Ok(0);
332        }
333
334        ensure_region_requirements(request.requirements, &self.object_store)?;
335
336        let res = RelationalRegion::open(region_id, request, &self.object_store).await;
337        let region = res.inspect_err(|err| {
338            error!(
339                err;
340                "Failed to open region, region_id: {}",
341                region_id
342            );
343        })?;
344        self.regions.write().unwrap().insert(region_id, region);
345
346        info!("Region opened, region_id: {}", region_id);
347        Ok(0)
348    }
349
350    async fn handle_close(
351        &self,
352        region_id: RegionId,
353        _request: RegionCloseRequest,
354    ) -> EngineResult<AffectedRows> {
355        let _lock = self.region_mutex.lock().await;
356
357        let mut regions = self.regions.write().unwrap();
358        if regions.remove(&region_id).is_some() {
359            info!("Region closed, region_id: {}", region_id);
360        }
361
362        Ok(0)
363    }
364
365    async fn handle_drop(
366        &self,
367        region_id: RegionId,
368        _request: RegionDropRequest,
369    ) -> EngineResult<AffectedRows> {
370        // Serialize against create/open/close so the region can't be closed
371        // between the existence check and the manifest deletion. A region that
372        // is no longer in memory (e.g. closed concurrently) is reported as not
373        // found instead of silently returning success with a leaked manifest.
374        let _lock = self.region_mutex.lock().await;
375
376        let region = self.get_region(region_id).await;
377        let Some(region) = region else {
378            return RegionNotFoundSnafu { region_id }.fail();
379        };
380
381        info!("Try to drop region, region_id: {}", region_id);
382
383        let res = RelationalRegion::drop(&region, &self.object_store).await;
384        res.inspect_err(|err| {
385            error!(
386                err;
387                "Failed to drop region, region_id: {}",
388                region_id
389            );
390        })?;
391
392        let _ = self.regions.write().unwrap().remove(&region_id);
393
394        info!("Region dropped, region_id: {}", region_id);
395        Ok(0)
396    }
397
398    async fn get_region(&self, region_id: RegionId) -> Option<RelationalRegionRef> {
399        self.regions.read().unwrap().get(&region_id).cloned()
400    }
401
402    async fn exists(&self, region_id: RegionId) -> bool {
403        self.regions.read().unwrap().contains_key(&region_id)
404    }
405}
