52#[derive(Debug, Clone, Serialize)]
53pub enum AllocateRegion {
54    Build(BuildPlan),
55    Execute(ExecutePlan),
56}
57
58impl<'de> Deserialize<'de> for AllocateRegion {
59    fn deserialize<D>(deserializer: D) -> std::result::Result<Self, D::Error>
60    where
61        D: Deserializer<'de>,
62    {
63        #[derive(Deserialize)]
64        enum CurrentAllocateRegion {
65            Build(BuildPlan),
66            Execute(ExecutePlan),
67        }
68
69        #[derive(Deserialize)]
70        struct LegacyAllocateRegion {
71            plan_entries: Vec<AllocationPlanEntry>,
72        }
73
74        #[derive(Deserialize)]
75        #[serde(untagged)]
76        enum AllocateRegionRepr {
77            Current(CurrentAllocateRegion),
78            Legacy(LegacyAllocateRegion),
79        }
80
81        match AllocateRegionRepr::deserialize(deserializer)? {
82            AllocateRegionRepr::Current(CurrentAllocateRegion::Build(build_plan)) => {
83                Ok(Self::Build(build_plan))
84            }
85            AllocateRegionRepr::Current(CurrentAllocateRegion::Execute(execute_plan)) => {
86                Ok(Self::Execute(execute_plan))
87            }
88            AllocateRegionRepr::Legacy(legacy) => Ok(Self::Build(BuildPlan {
89                plan_entries: legacy.plan_entries,
90            })),
91        }
92    }
93}

100impl BuildPlan {
101    async fn next(
102        &mut self,
103        ctx: &mut Context,
104        _procedure_ctx: &ProcedureContext,
105    ) -> Result<(Box<dyn State>, Status)> {
106        let timer = Instant::now();
107        let table_id = ctx.persistent_ctx.table_id;
108        let table_route_value = ctx.get_table_route_value().await?;
109        let mut next_region_number =
110            AllocateRegion::get_next_region_number(table_route_value.max_region_number().unwrap());
111
112        // Converts allocation plan to repartition plan.
113        let repartition_plan_entries = AllocateRegion::convert_to_repartition_plans(
114            table_id,
115            &mut next_region_number,
116            &self.plan_entries,
117            table_route_value.region_routes().unwrap(),
118        )?;
119        let plan_count = repartition_plan_entries.len();
120        let to_allocate = AllocateRegion::count_regions_to_allocate(&repartition_plan_entries);
121        info!(
122            "Repartition allocate regions start, table_id: {}, groups: {}, regions_to_allocate: {}",
123            table_id, plan_count, to_allocate
124        );
125
126        // If no region to allocate, directly dispatch the plan.
127        if AllocateRegion::count_regions_to_allocate(&repartition_plan_entries) == 0 {
128            ctx.persistent_ctx.plans = repartition_plan_entries;
129            ctx.update_allocate_region_elapsed(timer.elapsed());
130            return Ok((Box::new(Dispatch), Status::executing(true)));
131        }
132
133        ctx.persistent_ctx.plans = repartition_plan_entries;
134        debug!(
135            "Repartition allocate regions build plan completed, table_id: {}, elapsed: {:?}",
136            table_id,
137            timer.elapsed()
138        );
139        Ok((
140            Box::new(AllocateRegion::Execute(ExecutePlan)),
141            Status::executing(true),
142        ))
143    }
144}

149impl ExecutePlan {
150    async fn next(
151        &mut self,
152        ctx: &mut Context,
153        procedure_ctx: &ProcedureContext,
154    ) -> Result<(Box<dyn State>, Status)> {
155        let timer = Instant::now();
156        let table_id = ctx.persistent_ctx.table_id;
157        let allocate_regions = AllocateRegion::collect_allocate_regions(&ctx.persistent_ctx.plans);
158        let region_number_and_partition_exprs =
159            AllocateRegion::prepare_region_allocation_data(&allocate_regions)?;
160        let table_info_value = ctx.get_table_info_value().await?;
161        let affinity = if let Some(resolver) = &ctx.affinity_resolver {
162            Some(resolve_allocation_affinity(ctx, resolver.as_ref()).await?)
163        } else {
164            None
165        };
166        let groups = if let Some(affinity) = &affinity {
167            group_allocations(table_id, &region_number_and_partition_exprs, affinity)
168        } else {
169            vec![AllocationGroup {
170                affinity: None,
171                regions: region_number_and_partition_exprs
172                    .iter()
173                    .map(|(number, partition)| (*number, partition.as_str()))
174                    .collect(),
175            }]
176        };
177        let mut new_allocated_region_routes = Vec::with_capacity(allocate_regions.len());
178        for AllocationGroup { affinity, regions } in groups {
179            let routes = ctx
180                .region_routes_allocator
181                .allocate(
182                    table_id,
183                    &regions,
184                    &PeerAllocContext {
185                        affinity: affinity.cloned(),
186                    },
187                )
188                .await
189                .context(error::AllocateRegionRoutesSnafu { table_id })?;
190            new_allocated_region_routes.extend(routes);
191        }
192        let skip_wal = table_info_value.table_info.meta.options.skip_wal;
193        let allocate_noop = if skip_wal {
194            let table_route_value = ctx.get_table_route_value().await?;
195            let region_wal_options =
196                get_region_wal_options(&ctx.table_metadata_manager, &table_route_value, table_id)
197                    .await
198                    .context(error::AllocateWalOptionsSnafu { table_id })?;
199            // A table disabled after creation keeps its real providers. A table
200            // created with skip_wal has Noop providers and must keep them.
201            !table_route_value
202                .region_routes()
203                .unwrap()
204                .iter()
205                .all(|route| {
206                    region_wal_options
207                        .get(&route.region.id.region_number())
208                        .is_none_or(|option| {
209                            matches!(
210                                option,
211                                WalOptions::RaftEngine
212                                    | WalOptions::Kafka(_)
213                                    | WalOptions::ObjectStore(_)
214                            )
215                        })
216                })
217        } else {
218            false
219        };
220        let mut wal_options = ctx
221            .wal_options_allocator
222            .allocate(
223                &allocate_regions
224                    .iter()
225                    .map(|r| r.region_id.region_number())
226                    .collect::<Vec<_>>(),
227                allocate_noop,
228            )
229            .await
230            .context(error::AllocateWalOptionsSnafu { table_id })?;
231        let _remote_wal_lock_guards =
232            acquire_remote_wal_read_locks(procedure_ctx, &wal_options).await;
233        refresh_initial_pruned_entry_ids(&ctx.table_metadata_manager, &mut wal_options)
234            .await
235            .context(error::AllocateWalOptionsSnafu { table_id })?;
236
237        let new_region_count = new_allocated_region_routes.len();
238        let new_regions_brief: Vec<_> = new_allocated_region_routes
239            .iter()
240            .map(|route| {
241                let region_id = route.region.id;
242                let peer = route.leader_peer.as_ref().map(|p| p.id).unwrap_or_default();
243                format!("region_id: {}, peer: {}", region_id, peer)
244            })
245            .collect();
246        info!(
247            "Allocated regions for repartition, table_id: {}, new_region_count: {}, new_regions: {:?}",
248            table_id, new_region_count, new_regions_brief
249        );
250
251        // The table route metadata is not updated yet; register it in memory for region lease renewal.
252        let _operating_guards = Context::register_operating_regions(
253            &ctx.memory_region_keeper,
254            &new_allocated_region_routes,
255        )?;
256        // Allocates the regions on datanodes.
257        AllocateRegion::allocate_regions(
258            &ctx.node_manager,
259            &table_info_value.table_info,
260            &new_allocated_region_routes,
261            &wal_options,
262        )
263        .await?;
264
265        // Updates the table routes.
266        let table_lock = TableLock::Write(table_id).into();
267        let _guard = procedure_ctx.provider.acquire_lock(&table_lock).await;
268        // MUST refresh the table route value after acquiring the lock to avoid lost update.
269        // Otherwise, the new allocated regions might be overridden by concurrent repartition procedures.
270        let table_route_value = ctx.get_table_route_value().await?;
271        // Safety: it is physical table route value.
272        let region_routes = table_route_value.region_routes().unwrap();
273        let new_region_routes =
274            AllocateRegion::generate_region_routes(region_routes, &new_allocated_region_routes);
275        ctx.update_table_route(&table_route_value, new_region_routes, wal_options)
276            .await?;
277        ctx.invalidate_table_cache().await?;
278
279        ctx.update_allocate_region_elapsed(timer.elapsed());
280        Ok((Box::new(Dispatch), Status::executing(true)))
281    }
282}

303impl AllocateRegion {
304    pub fn new(plan_entries: Vec<AllocationPlanEntry>) -> Self {
305        AllocateRegion::Build(BuildPlan { plan_entries })
306    }
307
308    fn generate_region_routes(
309        region_routes: &[RegionRoute],
310        new_allocated_region_ids: &[RegionRoute],
311    ) -> Vec<RegionRoute> {
312        let region_ids = region_routes
313            .iter()
314            .map(|r| r.region.id)
315            .collect::<HashSet<_>>();
316        let mut new_region_routes = region_routes.to_vec();
317        for new_allocated_region_id in new_allocated_region_ids {
318            if !region_ids.contains(&new_allocated_region_id.region.id) {
319                new_region_routes.push(new_allocated_region_id.clone());
320            }
321        }
322        new_region_routes
323    }
324
325    /// Converts allocation plan entries to repartition plan entries.
326    ///
327    /// This method converts allocation intents into concrete repartition plans,
328    /// updates `next_region_number` for newly allocated regions, and captures
329    /// each plan's `original_target_routes` from the current table-route view.
330    ///
331    /// This also persists each plan's pre-staging target routes for rollback.
332    fn convert_to_repartition_plans(
333        table_id: TableId,
334        next_region_number: &mut RegionNumber,
335        plan_entries: &[AllocationPlanEntry],
336        current_region_routes: &[RegionRoute],
337    ) -> Result<Vec<RepartitionPlanEntry>> {
338        let region_routes_map = current_region_routes
339            .iter()
340            .map(|route| (route.region.id, route))
341            .collect::<HashMap<_, _>>();
342
343        plan_entries
344            .iter()
345            .map(|plan_entry| {
346                let mut plan = convert_allocation_plan_to_repartition_plan(
347                    table_id,
348                    next_region_number,
349                    plan_entry,
350                );
351                Self::capture_plan_original_target_routes(&mut plan, &region_routes_map)?;
352                Ok(plan)
353            })
354            .collect()
355    }
356
357    fn capture_plan_original_target_routes(
358        plan: &mut RepartitionPlanEntry,
359        region_routes_map: &HashMap<RegionId, &RegionRoute>,
360    ) -> Result<()> {
361        // Persist the pre-staging target-route view on the parent plan.
362        // Newly allocated targets are skipped because rollback deletes their
363        // route metadata rather than restoring an original target route.
364        let mut original_target_routes = Vec::with_capacity(plan.target_regions.len());
365        for target in &plan.target_regions {
366            if plan.allocated_region_ids.contains(&target.region_id) {
367                // This target region is to be allocated, so it doesn't exist in current routes.
368                continue;
369            }
370            let route = region_routes_map.get(&target.region_id).context(
371                error::RepartitionTargetRegionMissingSnafu {
372                    group_id: plan.group_id,
373                    region_id: target.region_id,
374                },
375            )?;
376            {
377                original_target_routes.push((*route).clone());
378            }
379        }
380
381        plan.original_target_routes = original_target_routes;
382        Ok(())
383    }
384
385    /// Collects all regions that need to be allocated from the repartition plan entries.
386    fn collect_allocate_regions(
387        repartition_plan_entries: &[RepartitionPlanEntry],
388    ) -> Vec<&TargetRegionDescriptor> {
389        repartition_plan_entries
390            .iter()
391            .flat_map(|p| p.allocate_regions())
392            .collect()
393    }
394
395    /// Prepares region allocation data: region numbers and their partition expressions.
396    fn prepare_region_allocation_data(
397        allocate_regions: &[&TargetRegionDescriptor],
398    ) -> Result<Vec<(RegionNumber, String)>> {
399        allocate_regions
400            .iter()
401            .map(|r| {
402                Ok((
403                    r.region_id.region_number(),
404                    r.partition_expr
405                        .as_json_str()
406                        .context(error::SerializePartitionExprSnafu)?,
407                ))
408            })
409            .collect()
410    }
411
412    /// Calculates the total number of regions that need to be allocated.
413    fn count_regions_to_allocate(repartition_plan_entries: &[RepartitionPlanEntry]) -> usize {
414        repartition_plan_entries
415            .iter()
416            .map(|p| p.allocated_region_ids.len())
417            .sum()
418    }
419
420    /// Gets the next region number from the physical table route.
421    fn get_next_region_number(max_region_number: RegionNumber) -> RegionNumber {
422        max_region_number + 1
423    }
424
425    async fn allocate_regions(
426        node_manager: &NodeManagerRef,
427        raw_table_info: &TableInfo,
428        region_routes: &[RegionRoute],
429        wal_options: &RegionWalOptions,
430    ) -> Result<()> {
431        let table_ref = TableReference::full(
432            &raw_table_info.catalog_name,
433            &raw_table_info.schema_name,
434            &raw_table_info.name,
435        );
436        let table_id = raw_table_info.ident.table_id;
437        // Repartition allocation targets physical regions, so exclude metric internal columns
438        // and derive primary keys from tag semantics.
439        let request = build_template_from_raw_table_info_for_physical_table(raw_table_info)
440            .context(error::BuildCreateRequestSnafu { table_id })?;
441        common_telemetry::debug!(
442            "Allocating regions request, table_id: {}, request: {:?}",
443            table_id,
444            request
445        );
446        let builder = CreateRequestBuilder::new(request, None)
447            .with_requirements(RegionRequirements::object_storage());
448        let region_count = region_routes.len();
449        let wal_region_count = wal_options.len();
450        info!(
451            "Allocating regions on datanodes, table_id: {}, region_count: {}, wal_regions: {}",
452            table_id, region_count, wal_region_count
453        );
454        let executor = CreateTableExecutor::new(table_ref.into(), false, builder);
455        executor
456            .on_create_regions(node_manager, table_id, region_routes, wal_options)
457            .await
458            .context(error::AllocateRegionsSnafu { table_id })?;
459
460        Ok(())
461    }
462}

464/// Selects the lowest-ID contributing source for a target spanning multiple sources.
465/// Legacy plans without a transition map fall back to the lowest-ID source in the group.
466fn collect_allocation_sources(
467    plans: &[RepartitionPlanEntry],
468    routes: &[RegionRoute],
469) -> Vec<(RegionId, Option<u64>)> {
470    let leaders: HashMap<_, _> = routes
471        .iter()
472        .filter_map(|route| {
473            route
474                .leader_peer
475                .as_ref()
476                .map(|peer| (route.region.id, peer.id))
477        })
478        .collect();
479    let mut sources = Vec::new();
480    for plan in plans {
481        let allocated: HashSet<_> = plan.allocated_region_ids.iter().copied().collect();
482        let mut target_sources = vec![None; plan.target_regions.len()];
483        if plan.transition_map.is_empty() {
484            let source = plan
485                .source_regions
486                .iter()
487                .map(|source| source.region_id())
488                .min();
489            target_sources.fill(source);
490        } else {
491            for (source, targets) in plan.source_regions.iter().zip(&plan.transition_map) {
492                for target_index in targets {
493                    if let Some(target_source) = target_sources.get_mut(*target_index) {
494                        let source_id = source.region_id();
495                        *target_source =
496                            Some(target_source.map_or(source_id, |current| current.min(source_id)));
497                    }
498                }
499            }
500        }
501        for (target, source) in plan.target_regions.iter().zip(target_sources) {
502            if allocated.contains(&target.region_id) {
503                sources.push((
504                    target.region_id,
505                    source.and_then(|source| leaders.get(&source)).copied(),
506                ));
507            }
508        }
509    }
510    sources
511}

513/// Owns requirements once per source; targets refer to them by index.
514#[derive(Debug)]
515struct AllocationAffinity {
516    requirements: Vec<Option<AffinityRequirements>>,
517    target_sources: HashMap<RegionId, usize>,
518}
519
520async fn resolve_allocation_affinity(
521    ctx: &Context,
522    resolver: &dyn AffinityResolver,
523) -> Result<AllocationAffinity> {
524    let table_id = ctx.persistent_ctx.table_id;
525    let routes = ctx.get_table_route_value().await?;
526    let sources =
527        collect_allocation_sources(&ctx.persistent_ctx.plans, routes.region_routes().unwrap());
528    let mut source_indices = HashMap::new();
529    let mut references = Vec::new();
530    for (_, source_id) in &sources {
531        if let Some(source_id) = source_id {
532            source_indices.entry(*source_id).or_insert_with(|| {
533                let index = references.len();
534                references.push(AffinityReference::Datanode(*source_id));
535                index
536            });
537        }
538    }
539    let resolved = resolver
540        .resolve(&references)
541        .await
542        .context(error::AllocateRegionRoutesSnafu { table_id })?;
543    snafu::ensure!(
544        resolved.len() == source_indices.len(),
545        error::UnexpectedSnafu {
546            violated: "affinity resolution must preserve the number and order of references",
547        }
548    );
549    snafu::ensure!(
550        ctx.affinity_fallback == AffinityFallback::Unconstrained
551            || sources.iter().all(|(_, source)| {
552                source.is_some_and(|id| resolved[source_indices[&id]].is_some())
553            }),
554        error::UnexpectedSnafu {
555            violated: "source affinity requirements are unavailable",
556        }
557    );
558    Ok(AllocationAffinity {
559        requirements: resolved,
560        target_sources: sources
561            .into_iter()
562            .filter_map(|(region_id, source_id)| {
563                source_id.map(|source_id| (region_id, source_indices[&source_id]))
564            })
565            .collect(),
566    })
567}

569#[derive(Debug, PartialEq, Eq)]
570struct AllocationGroup<'a> {
571    affinity: Option<&'a AffinityRequirements>,
572    regions: Vec<(RegionNumber, &'a str)>,
573}
574
575fn group_allocations<'a>(
576    table_id: TableId,
577    regions: &'a [(RegionNumber, String)],
578    affinity: &'a AllocationAffinity,
579) -> Vec<AllocationGroup<'a>> {
580    let mut groups: Vec<AllocationGroup<'a>> = Vec::new();
581    for (number, partition) in regions {
582        let requirements = affinity
583            .target_sources
584            .get(&RegionId::new(table_id, *number))
585            .and_then(|index| affinity.requirements[*index].as_ref());
586        if let Some(group) = groups
587            .iter_mut()
588            .find(|group| group.affinity == requirements)
589        {
590            group.regions.push((*number, partition.as_str()));
591        } else {
592            groups.push(AllocationGroup {
593                affinity: requirements,
594                regions: vec![(*number, partition.as_str())],
595            });
596        }
597    }
598    groups
599}

661        table_id: TableId,
662        source_region_numbers: &[u32],
663        target_ranges: &[(i64, i64)],
664    ) -> AllocationPlanEntry {
665        let source_regions = source_region_numbers
666            .iter()
667            .enumerate()
668            .map(|(i, &n)| {
669                let start = i as i64 * 100;
670                let end = (i + 1) as i64 * 100;
671                create_region_descriptor(table_id, n, "x", start, end)
672            })
673            .collect();
674
675        let target_partition_exprs = target_ranges
676            .iter()
677            .map(|&(start, end)| range_expr("x", start, end))
678            .collect();
679
680        AllocationPlanEntry {
681            group_id: Uuid::new_v4(),
682            source_regions,
683            target_partition_exprs,
684            transition_map: vec![],
685        }
686    }
