209    pub async fn build(self) -> Result<Metasrv> {
210        let MetasrvBuilder {
211            election,
212            meta_peer_client,
213            options,
214            kv_backend,
215            in_memory,
216            selector,
217            handler_group_builder,
218            node_manager,
219            plugins,
220            table_metadata_allocator,
221        } = self;
222
223        let options = options.unwrap_or_default();
224        options.gc.validate()?;
225
226        let kv_backend = kv_backend.unwrap_or_else(|| Arc::new(MemoryKvBackend::new()));
227        let in_memory = in_memory.unwrap_or_else(|| Arc::new(MemoryKvBackend::new()));
228
229        let state = Arc::new(RwLock::new(match election {
230            None => State::leader(options.grpc.server_addr.clone(), true),
231            Some(_) => State::follower(options.grpc.server_addr.clone()),
232        }));
233
234        let leader_cached_kv_backend = Arc::new(LeaderCachedKvBackend::new(
235            state.clone(),
236            kv_backend.clone(),
237        ));
238
239        let meta_peer_client = meta_peer_client
240            .unwrap_or_else(|| build_default_meta_peer_client(&election, &in_memory));
241        let database_operator = Arc::new(DatabaseOperator::new(meta_peer_client.clone()));
242
243        let event_inserter = Box::new(InsertForwarder::new(
244            database_operator.clone(),
245            Some(InsertOptions {
246                ttl: options.event_recorder.ttl,
247                append_mode: true,
248                twcs_compaction_time_window: Some(DEFAULT_COMPACTION_TIME_WINDOW),
249            }),
250        ));
251        // Builds the event recorder to record important events and persist them as the system table.
252        let event_recorder = Arc::new(EventRecorderImpl::with_event_type_filter(
253            Box::new(EventHandlerImpl::new(event_inserter)),
254            options.event_recorder.event_types.clone(),
255            options.event_recorder.flush_interval,
256        ));
257
258        let selector = selector.unwrap_or_else(|| Arc::new(LeaseBasedSelector));
259        let pushers = Pushers::default();
260        let mailbox = build_mailbox(&kv_backend, &pushers);
261        let runtime_switch_manager = Arc::new(RuntimeSwitchManager::new(kv_backend.clone()));
262        let repartition_gc_requirement_manager =
263            Arc::new(RepartitionGcRequirementManager::new(kv_backend.clone()));
264        let procedure_manager = build_procedure_manager(
265            &options,
266            &kv_backend,
267            &runtime_switch_manager,
268            event_recorder,
269        );
270
271        let table_metadata_manager = Arc::new(TableMetadataManager::new(
272            leader_cached_kv_backend.clone() as _,
273        ));
274        let flow_metadata_manager = Arc::new(FlowMetadataManager::new(
275            leader_cached_kv_backend.clone() as _,
276        ));
277
278        let selector_ctx = SelectorContext {
279            affinity_fallback: plugins
280                .as_ref()
281                .and_then(|plugins| plugins.get::<AffinityFallback>())
282                .unwrap_or_default(),
283            affinity: plugins
284                .as_ref()
285                .and_then(|plugins| plugins.get::<AffinityRef>())
286                .unwrap_or_else(|| Arc::new(NoopAffinity)),
287            peer_discovery: meta_peer_client.clone(),
288        };
289
290        let wal_provider = build_wal_provider(&options.wal, kv_backend.clone())
291            .await
292            .context(BuildWalProviderSnafu)?;
293        let wal_provider = Arc::new(wal_provider);
294        let is_remote_wal = wal_provider.is_remote_wal();
295        let table_metadata_allocator = table_metadata_allocator.unwrap_or_else(|| {
296            let sequence = Arc::new(
297                SequenceBuilder::new(TABLE_ID_SEQ, kv_backend.clone())
298                    .initial(MIN_USER_TABLE_ID as u64)
299                    .step(10)
300                    .build(),
301            );
302            let peer_allocator = Arc::new(
303                MetasrvPeerAllocator::new(selector_ctx.clone(), selector.clone())
304                    .with_max_items(MAX_REGION_SEQ),
305            );
306            Arc::new(TableMetadataAllocator::with_peer_allocator(
307                sequence,
308                wal_provider.clone(),
309                peer_allocator,
310            ))
311        });
312        let table_id_allocator = table_metadata_allocator.table_id_allocator();
313
314        let flow_selector =
315            Arc::new(RoundRobinSelector::new(SelectTarget::Flownode)) as SelectorRef;
316
317        let flow_metadata_allocator = {
318            // for now flownode just use round-robin selector
319            let flow_selector_ctx = selector_ctx.clone();
320            let peer_allocator = Arc::new(MetasrvPeerAllocator::new(
321                flow_selector_ctx,
322                flow_selector.clone(),
323            ));
324            let seq = Arc::new(
325                SequenceBuilder::new(FLOW_ID_SEQ, kv_backend.clone())
326                    .initial(MIN_USER_FLOW_ID as u64)
327                    .step(10)
328                    .build(),
329            );
330
331            Arc::new(FlowMetadataAllocator::with_peer_allocator(
332                seq,
333                peer_allocator,
334            ))
335        };
336        let flow_state_handler =
337            FlowStateHandler::new(FlowStateManager::new(in_memory.clone().as_kv_backend_ref()));
338
339        let memory_region_keeper = Arc::new(MemoryRegionKeeper::default());
340        let node_manager = node_manager.unwrap_or_else(|| {
341            let datanode_client_channel_config = options.datanode.client.channel_config();
342            Arc::new(NodeClients::new(datanode_client_channel_config))
343        });
344        let cache_invalidator = Arc::new(MetasrvCacheInvalidator::new(
345            mailbox.clone(),
346            MetasrvInfo {
347                server_addr: options.grpc.server_addr.clone(),
348            },
349        ));
350
351        if !is_remote_wal && options.enable_region_failover {
352            ensure!(
353                options.allow_region_failover_on_local_wal,
354                error::UnexpectedSnafu {
355                    violated: "Region failover is not supported in the local WAL implementation!
356                    If you want to enable region failover for local WAL, please set `allow_region_failover_on_local_wal` to true.",
357                }
358            );
359            if options.allow_region_failover_on_local_wal {
360                warn!(
361                    "Region failover is force enabled in the local WAL implementation! This may lead to data loss during failover!"
362                );
363            }
364        }
365
366        let (tx, rx) = RegionSupervisor::channel();
367        let (region_failure_detector_controller, region_supervisor_ticker): (
368            RegionFailureDetectorControllerRef,
369            Option<std::sync::Arc<RegionSupervisorTicker>>,
370        ) = if options.enable_region_failover {
371            (
372                Arc::new(RegionFailureDetectorControl::new(tx.clone())) as _,
373                Some(Arc::new(RegionSupervisorTicker::new(
374                    DEFAULT_TICK_INTERVAL,
375                    options.region_failure_detector_initialization_delay,
376                    DEFAULT_INITIALIZATION_RETRY_PERIOD,
377                    tx.clone(),
378                ))),
379            )
380        } else {
381            (Arc::new(NoopRegionFailureDetectorControl) as _, None as _)
382        };
383
384        // region migration manager
385        let region_migration_manager = Arc::new(RegionMigrationManager::new(
386            procedure_manager.clone(),
387            DefaultContextFactory::new(
388                in_memory.clone(),
389                table_metadata_manager.clone(),
390                memory_region_keeper.clone(),
391                region_failure_detector_controller.clone(),
392                mailbox.clone(),
393                options.grpc.server_addr.clone(),
394                cache_invalidator.clone(),
395            ),
396        ));
397        region_migration_manager.try_start()?;
398        let affinity_resolver = plugins
399            .as_ref()
400            .and_then(|plugins| plugins.get::<SelectorFactoryRef>())
401            .and_then(|factory| factory.build_affinity_resolver(meta_peer_client.clone()));
402        let region_supervisor_selector = plugins
403            .as_ref()
404            .and_then(|plugins| plugins.get::<RegionStatAwareSelectorRef>());
405
406        let supervisor_selector = match region_supervisor_selector {
407            Some(selector) => {
408                info!("Using region stat aware selector");
409                RegionSupervisorSelector::RegionStatAwareSelector(selector)
410            }
411            None => RegionSupervisorSelector::NaiveSelector(selector.clone()),
412        };
413
414        let region_failover_handler = if options.enable_region_failover {
415            let region_supervisor = RegionSupervisor::new(
416                rx,
417                options.failure_detector,
418                selector_ctx.clone(),
419                supervisor_selector,
420                region_migration_manager.clone(),
421                runtime_switch_manager.clone(),
422                meta_peer_client.clone(),
423                leader_cached_kv_backend.clone(),
424            )
425            .with_state(state.clone());
426            let region_supervisor = if let Some(resolver) = &affinity_resolver {
427                region_supervisor.with_affinity_resolver(resolver.clone())
428            } else {
429                region_supervisor
430            };
431
432            Some(RegionFailureHandler::new(
433                region_supervisor,
434                HeartbeatAcceptor::new(tx),
435            ))
436        } else {
437            None
438        };
439
440        let leader_region_registry = Arc::new(LeaderRegionRegistry::default());
441        let topic_stats_registry = Arc::new(TopicStatsRegistry::default());
442
443        let ddl_context = DdlContext {
444            peer_alloc_context_provider: None,
445            affinity_fallback: selector_ctx.affinity_fallback,
446            affinity_resolver,
447            node_manager: node_manager.clone(),
448            cache_invalidator: cache_invalidator.clone(),
449            memory_region_keeper: memory_region_keeper.clone(),
450            leader_region_registry: leader_region_registry.clone(),
451            table_metadata_manager: table_metadata_manager.clone(),
452            table_metadata_allocator: table_metadata_allocator.clone(),
453            flow_metadata_manager: flow_metadata_manager.clone(),
454            flow_metadata_allocator: flow_metadata_allocator.clone(),
455            region_failure_detector_controller,
456            soft_drop_enabled: ddl_soft_drop_enabled(&options),
457            soft_drop_retention: ddl_soft_drop_retention(&options),
458            create_database_metadata_committer: None,
459        };
460        let procedure_manager_c = procedure_manager.clone();
461        let repartition_procedure_factory: RepartitionProcedureFactoryRef = if options.gc.enable {
462            Arc::new(DefaultRepartitionProcedureFactory::new(
463                mailbox.clone(),
464                options.grpc.server_addr.clone(),
465                repartition_gc_requirement_manager.clone(),
466            ))
467        } else {
468            Arc::new(GcDisabledRepartitionProcedureFactory::new(
469                mailbox.clone(),
470                options.grpc.server_addr.clone(),
471                repartition_gc_requirement_manager.clone(),
472            ))
473        };
474        let ddl_manager = DdlManager::new(
475            ddl_context,
476            procedure_manager_c,
477            repartition_procedure_factory,
478        );
479
480        let ddl_manager = if let Some(configurator) = plugins
481            .as_ref()
482            .and_then(|p| p.get::<DdlManagerConfiguratorRef<DdlManagerConfigureContext>>())
483        {
484            let ctx = DdlManagerConfigureContext {
485                kv_backend: kv_backend.clone(),
486                meta_peer_client: meta_peer_client.clone(),
487            };
488            configurator
489                .configure(ddl_manager, ctx)
490                .await
491                .context(OtherSnafu)?
492        } else {
493            ddl_manager
494        };
495        ddl_manager
496            .register_loaders()
497            .context(error::InitDdlManagerSnafu)?;
498
499        let ddl_manager = Arc::new(ddl_manager);
500
501        let region_flush_ticker = if is_remote_wal {
502            let remote_wal_options = options.wal.remote_wal_options().unwrap();
503            let (region_flush_trigger, region_flush_ticker) = RegionFlushTrigger::new(
504                table_metadata_manager.clone(),
505                leader_region_registry.clone(),
506                topic_stats_registry.clone(),
507                mailbox.clone(),
508                options.grpc.server_addr.clone(),
509                remote_wal_options.flush_trigger_size,
510                remote_wal_options.checkpoint_trigger_size,
511                remote_wal_options.region_flush_trigger_interval,
512                remote_wal_options.periodic_checkpoint_persist_interval,
513            );
514            region_flush_trigger.try_start()?;
515
516            Some(Arc::new(region_flush_ticker))
517        } else {
518            None
519        };
520
521        // remote WAL prune ticker and manager
522        let wal_prune_ticker = if is_remote_wal && options.wal.enable_active_wal_pruning() {
523            let (tx, rx) = WalPruneManager::channel();
524            // Safety: Must be remote WAL.
525            let remote_wal_options = options.wal.remote_wal_options().unwrap();
526            let kafka_client = build_kafka_client(&remote_wal_options.connection)
527                .await
528                .context(error::BuildKafkaClientSnafu)?;
529            let wal_prune_context = WalPruneContext {
530                client: Arc::new(kafka_client),
531                table_metadata_manager: table_metadata_manager.clone(),
532                leader_region_registry: leader_region_registry.clone(),
533            };
534            let wal_prune_manager = WalPruneManager::new(
535                remote_wal_options.auto_prune_parallelism,
536                remote_wal_options.auto_prune_logical_delete,
537                rx,
538                procedure_manager.clone(),
539                wal_prune_context,
540            );
541            // Start manager in background. Ticker will be started in the main thread to send ticks.
542            wal_prune_manager.try_start().await?;
543            let wal_prune_ticker = Arc::new(WalPruneTicker::new(
544                remote_wal_options.auto_prune_interval,
545                tx.clone(),
546            ));
547            Some(wal_prune_ticker)
548        } else {
549            None
550        };
551
552        let gc_ticker = if options.gc.enable {
553            let gc_scheduler_ctx = DefaultGcSchedulerCtx::try_new(
554                table_metadata_manager.clone(),
555                procedure_manager.clone(),
556                runtime_switch_manager.clone(),
557                #[cfg(feature = "enterprise")]
558                ddl_manager.clone(),
559                meta_peer_client.clone(),
560                mailbox.clone(),
561                options.grpc.server_addr.clone(),
562            )?;
563            let (gc_scheduler, gc_ticker) = GcScheduler::new_with_config(
564                gc_scheduler_ctx,
565                runtime_switch_manager.clone(),
566                options.gc.clone(),
567            )?;
568            gc_scheduler.try_start()?;
569
570            Some(Arc::new(gc_ticker))
571        } else {
572            None
573        };
574
575        let customized_region_lease_renewer = plugins
576            .as_ref()
577            .and_then(|plugins| plugins.get::<CustomizedRegionLeaseRenewerRef>());
578
579        let persist_region_stats_handler = if !options.stats_persistence.ttl.is_zero() {
580            let inserter = Box::new(InsertForwarder::new(
581                database_operator.clone(),
582                Some(InsertOptions {
583                    ttl: options.stats_persistence.ttl,
584                    append_mode: true,
585                    twcs_compaction_time_window: Some(
586                        REGION_STATS_TABLE_TWCS_COMPACTION_TIME_WINDOW,
587                    ),
588                }),
589            ));
590
591            Some(PersistStatsHandler::new(
592                inserter,
593                options.stats_persistence.interval,
594            ))
595        } else {
596            None
597        };
598
599        let handler_group_builder = match handler_group_builder {
600            Some(handler_group_builder) => handler_group_builder,
601            None => {
602                let region_lease_handler = RegionLeaseHandler::new(
603                    default_distributed_time_constants().region_lease.as_secs(),
604                    table_metadata_manager.clone(),
605                    memory_region_keeper.clone(),
606                    customized_region_lease_renewer,
607                );
608
609                HeartbeatHandlerGroupBuilder::new(pushers)
610                    .with_plugins(plugins.clone())
611                    .with_region_failure_handler(region_failover_handler)
612                    .with_region_lease_handler(Some(region_lease_handler))
613                    .with_flush_stats_factor(Some(options.flush_stats_factor))
614                    .with_flow_state_handler(Some(flow_state_handler))
615                    .with_persist_stats_handler(persist_region_stats_handler)
616                    .add_default_handlers()
617            }
618        };
619
620        let enable_telemetry = options.enable_telemetry;
621        let metasrv_home = Path::new(&options.data_home)
622            .join(METASRV_DATA_DIR)
623            .to_string_lossy()
624            .to_string();
625
626        let reconciliation_manager = Arc::new(ReconciliationManager::new(
627            node_manager.clone(),
628            table_metadata_manager.clone(),

695fn build_procedure_manager(
696    options: &MetasrvOptions,
697    kv_backend: &KvBackendRef,
698    runtime_switch_manager: &RuntimeSwitchManagerRef,
699    event_recorder: EventRecorderRef,
700) -> ProcedureManagerRef {
701    let manager_config = ManagerConfig {
702        max_retry_times: options.procedure.max_retry_times,
703        retry_delay: options.procedure.retry_delay,
704        max_running_procedures: options.procedure.max_running_procedures,
705        ..Default::default()
706    };
707    let kv_state_store = Arc::new(
708        KvStateStore::new(kv_backend.clone()).with_max_value_size(
709            options
710                .procedure
711                .max_metadata_value_size
712                .map(|v| v.as_bytes() as usize),
713        ),
714    );
715
716    Arc::new(LocalManager::new(
717        manager_config,
718        kv_state_store.clone(),
719        kv_state_store,
720        Some(runtime_switch_manager.clone()),
721        Some(event_recorder),
722    ))
723}

725/// Resolves if soft-drop is enabled from metasrv options.
726///
727/// Soft drop is an enterprise-only feature; it is always disabled in
728/// non-enterprise builds regardless of the configuration.
729fn ddl_soft_drop_enabled(options: &MetasrvOptions) -> bool {
730    cfg!(feature = "enterprise") && options.gc.experimental_soft_drop.enable
731}
732
733/// Returns soft-drop retention for recovering persisted procedures.
734fn ddl_soft_drop_retention(options: &MetasrvOptions) -> Option<Duration> {
735    Some(options.gc.experimental_soft_drop.retention)
736}
