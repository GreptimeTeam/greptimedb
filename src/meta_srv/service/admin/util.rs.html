26/// Returns a 200 response with a text body.
27pub fn to_text_response(text: &str) -> Result<http::Response<String>> {
28    http::Response::builder()
29        .header("Content-Type", "text/plain")
30        .status(http::StatusCode::OK)
31        .body(text.to_string())
32        .context(error::InvalidHttpBodySnafu)
33}
34
35/// Returns a 200 response with a JSON body.
36pub fn to_json_response<T>(response: T) -> Result<http::Response<String>>
37where
38    T: Serialize + Debug,
39{
40    let response = serde_json::to_string(&response).context(error::SerializeToJsonSnafu {
41        input: format!("{response:?}"),
42    })?;
43    http::Response::builder()
44        .header("Content-Type", "application/json")
45        .status(http::StatusCode::OK)
46        .body(response)
47        .context(error::InvalidHttpBodySnafu)
48}
49
50/// Returns a 404 response with an empty body.
51pub fn to_not_found_response() -> Result<http::Response<String>> {
52    http::Response::builder()
53        .status(http::StatusCode::NOT_FOUND)
54        .body("".to_string())
55        .context(error::InvalidHttpBodySnafu)
56}

58/// A wrapper for error handling in admin services.
59pub(crate) struct ErrorHandler<E>(E)
60where
61    E: std::error::Error + Send + Sync + Sized;
62
63impl<E> ErrorHandler<E>
64where
65    E: std::error::Error + Send + Sync + Sized,
66{
67    pub(crate) fn new(error: E) -> Self {
68        Self(error)
69    }
70}
71
72impl<E> IntoResponse for ErrorHandler<E>
73where
74    E: std::error::Error + Send + Sync + Sized,
75{
76    fn into_response(self) -> Response {
77        (StatusCode::INTERNAL_SERVER_ERROR, self.0.to_string()).into_response()
78    }
79}

81/// Uses the same leadership predicate as the metasrv gRPC procedure service.
82pub(crate) fn ensure_leader(metasrv: &Metasrv) -> Result<()> {
83    ensure!(
84        metasrv.is_leader(),
85        error::IsNotLeaderSnafu {
86            node_addr: metasrv.options().grpc.server_addr.clone(),
87        }
88    );
89    Ok(())
90}
