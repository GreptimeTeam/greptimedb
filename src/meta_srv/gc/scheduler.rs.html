46/// Report for a GC job.
47#[derive(Debug)]
48pub enum GcJobReport {
49    PerDatanode {
50        per_datanode_reports: HashMap<DatanodeId, GcReport>,
51        failed_datanodes: HashMap<DatanodeId, Vec<Error>>,
52    },
53    Combined {
54        report: GcReport,
55    },
56}
57
58impl Default for GcJobReport {
59    fn default() -> Self {
60        Self::PerDatanode {
61            per_datanode_reports: HashMap::new(),
62            failed_datanodes: HashMap::new(),
63        }
64    }
65}
66
67impl GcJobReport {
68    pub fn combined(report: GcReport) -> Self {
69        Self::Combined { report }
70    }
71
72    pub fn merge_to_report(self) -> GcReport {
73        match self {
74            GcJobReport::Combined { report } => report,
75            GcJobReport::PerDatanode {
76                per_datanode_reports,
77                ..
78            } => {
79                let mut combined = GcReport::default();
80                for (_datanode_id, report) in per_datanode_reports {
81                    combined.merge(report);
82                }
83                combined
84            }
85        }
86    }
87}

89/// [`Event`] represents various types of events that can be processed by the gc ticker.
90///
91/// Variants:
92/// - `Tick`: This event is used to trigger gc periodically.
93/// - `Manually`: This event is used to trigger a manual gc run and provides a channel
94///   to send back the result for that run.
95///   Optional parameters allow specifying target regions and GC behavior.
96pub enum Event {
97    Tick,
98    Manually {
99        /// Channel sender to return the GC job report or error
100        sender: oneshot::Sender<Result<GcJobReport>>,
101        /// Optional specific region IDs to GC. If None, scheduler will select candidates automatically.
102        region_ids: Option<Vec<RegionId>>,
103        /// Optional override for full file listing. If None, uses scheduler config.
104        full_file_listing: Option<bool>,
105        /// Optional override for timeout. If None, uses scheduler config.
106        timeout: Option<Duration>,
107    },
108}

120/// [`GcScheduler`] is used to periodically trigger garbage collection on datanodes.
121pub struct GcScheduler {
122    pub(crate) ctx: Arc<dyn SchedulerCtx>,
123    /// Runtime switch manager to check maintenance mode.
124    pub(crate) runtime_switch_manager: RuntimeSwitchManagerRef,
125    /// The receiver of events.
126    pub(crate) receiver: Receiver<Event>,
127    /// GC configuration.
128    pub(crate) config: GcSchedulerOptions,
129    /// Tracks the last GC time for regions.
130    pub(crate) region_gc_tracker: Arc<Mutex<RegionGcTracker>>,
131    /// Last time the tracker was cleaned up.
132    pub(crate) last_tracker_cleanup: Arc<Mutex<Instant>>,
133}

135impl GcScheduler {
136    /// Creates a new [`GcScheduler`] with custom configuration.
137    pub(crate) fn new_with_config(
138        ctx: impl SchedulerCtx + 'static,
139        runtime_switch_manager: RuntimeSwitchManagerRef,
140        config: GcSchedulerOptions,
141    ) -> Result<(Self, GcTicker)> {
142        // Validate configuration before creating the scheduler
143        config.validate()?;
144
145        let (tx, rx) = Self::channel();
146        let gc_ticker = GcTicker::new(TICKER_INTERVAL, tx);
147        let gc_trigger = Self {
148            ctx: Arc::new(ctx),
149            runtime_switch_manager,
150            receiver: rx,
151            config,
152            region_gc_tracker: Arc::new(Mutex::new(HashMap::new())),
153            last_tracker_cleanup: Arc::new(Mutex::new(Instant::now())),
154        };
155        Ok((gc_trigger, gc_ticker))
156    }

158    pub(crate) fn channel() -> (Sender<Event>, Receiver<Event>) {
159        tokio::sync::mpsc::channel(8)
160    }

162    /// Starts the gc trigger.
163    pub fn try_start(mut self) -> Result<()> {
164        common_runtime::spawn_global(async move { self.run().await });
165        info!("GC trigger started");
166        Ok(())
167    }

169    pub(crate) async fn run(&mut self) {
170        while let Some(event) = self.receiver.recv().await {
171            match event {
172                Event::Tick => {
173                    info!("Received gc tick");
174                    let span =
175                        common_telemetry::tracing::info_span!("meta_gc_tick", trigger = "ticker");
176                    if let Err(e) = self.handle_tick().instrument(span).await {
177                        error!(e; "Failed to handle gc tick");
178                    }
179                }
180                Event::Manually {
181                    sender,
182                    region_ids,
183                    full_file_listing,
184                    timeout,
185                } => {
186                    info!("Received manually gc request");
187                    let span =
188                        common_telemetry::tracing::info_span!("meta_gc_tick", trigger = "manual");
189                    let result = self
190                        .handle_manual_gc(region_ids, full_file_listing, timeout)
191                        .instrument(span)
192                        .await;
193                    if let Err(e) = &result {
194                        if matches!(e, Error::ManualGcRejectedByMaintenanceMode { .. }) {
195                            info!("Rejected manual gc request: {}", e);
196                        } else {
197                            error!(e; "Failed to handle manual gc");
198                        }
199                    }
200                    let _ = sender.send(result);
201                }
202            }
203        }
204    }

206    pub(crate) async fn handle_tick(&self) -> Result<GcJobReport> {
207        METRIC_META_GC_SCHEDULER_CYCLES_TOTAL.inc();
208        let _timer = METRIC_META_GC_SCHEDULER_DURATION_SECONDS.start_timer();
209        info!("Start to trigger gc");
210        if self.is_maintenance_mode_enabled().await? {
211            info!("Skip gc trigger because maintenance mode is enabled");
212            return Ok(GcJobReport::default());
213        }
214        #[cfg(feature = "enterprise")]
215        if self.config.experimental_soft_drop.enable {
216            self.purge_expired_soft_dropped_tables(common_time::util::current_time_millis())
217                .await;
218        }
219        let span = common_telemetry::tracing::info_span!("meta_gc_handle_tick");
220        let report = self
221            .trigger_gc(EventContext::new(TriggerReason::ScheduledGc))
222            .instrument(span)
223            .await?;
224
225        // Periodically clean up stale tracker entries
226        self.cleanup_tracker_if_needed().await?;
227
228        info!("Finished gc trigger");
229
230        Ok(report)
231    }

233    #[cfg(feature = "enterprise")]
234    async fn purge_expired_soft_dropped_tables(&self, now_millis: i64) {
235        // The scheduler is only constructed after GcSchedulerOptions::validate().
236        let retention_millis =
237            i64::try_from(self.config.experimental_soft_drop.retention.as_millis())
238                .unwrap_or(i64::MAX);
239        let dropped_tables = match self.ctx.list_dropped_tables().await {
240            Ok(dropped_tables) => dropped_tables,
241            Err(error) => {
242                error!(error; "Failed to list soft-dropped tables for GC");
243                return;
244            }
245        };
246        let table_count = dropped_tables.len();
247        let scan_start = self.ctx.next_purge_scan_start(table_count);
248
249        for table in dropped_tables
250            .iter()
251            .cycle()
252            .skip(scan_start)
253            .take(table_count)
254            .filter(|table| {
255                table
256                    .retention_expires_at
257                    .or_else(|| {
258                        table
259                            .dropped_at
260                            .and_then(|dropped_at| dropped_at.checked_add(retention_millis))
261                    })
262                    .is_some_and(|expires_at| expires_at <= now_millis)
263            })
264        {
265            let table_id = table.table_id;
266            let Some(reservation) = self
267                .ctx
268                .try_reserve_purge(table_id, self.config.max_concurrent_tables)
269            else {
270                continue;
271            };
272            METRIC_META_GC_SOFT_DROP_PURGES_TOTAL
273                .with_label_values(&["submitted"])
274                .inc();
275            let ctx = self.ctx.clone();
276            common_runtime::spawn_global(async move {
277                match ctx.purge_dropped_table(table_id).await {
278                    Ok(()) => reservation.record_outcome(PurgeOutcome::Succeeded),
279                    Err(error) => {
280                        reservation.record_outcome(PurgeOutcome::Failed);
281                        error!(error; "Failed to purge expired soft-dropped table {}", table_id);
282                    }
283                }
284            });
285        }
286    }

288    /// Handles a manual GC request with optional specific parameters.
289    ///
290    /// If `region_ids` is specified, GC will be performed only on those regions.
291    /// Otherwise, falls back to automatic candidate selection.
292    pub(crate) async fn handle_manual_gc(
293        &self,
294        region_ids: Option<Vec<RegionId>>,
295        full_file_listing: Option<bool>,
296        timeout: Option<Duration>,
297    ) -> Result<GcJobReport> {
298        info!("Start to handle manual gc request");
299
300        if self.is_maintenance_mode_enabled().await? {
301            info!("Skip manual gc request because maintenance mode is enabled");
302            return error::ManualGcRejectedByMaintenanceModeSnafu {}.fail();
303        }
304
305        // No specific regions, use default tick behavior
306        let Some(regions) = region_ids else {
307            let report = self
308                .trigger_gc(EventContext::new(TriggerReason::Manual))
309                .await?;
310            info!("Finished manual gc request");
311            return Ok(report);
312        };
313
314        // Empty regions list, return empty report
315        if regions.is_empty() {
316            info!("Finished manual gc request");
317            return Ok(GcJobReport::combined(GcReport::default()));
318        }
319
320        let full_listing = full_file_listing.unwrap_or(false);
321        let gc_timeout = timeout.unwrap_or(self.config.mailbox_timeout);
322
323        let region_set: HashSet<RegionId> = regions.iter().copied().collect();
324        let table_reparts = self.ctx.get_table_reparts().await?;
325        let dropped_collector =
326            DroppedRegionCollector::new(self.ctx.as_ref(), &self.config, &self.region_gc_tracker);
327        let dropped_assignment = dropped_collector
328            .collect_and_assign_with_cooldown(&table_reparts, false)
329            .await?;
330
331        let mut dropped_region_set = HashSet::new();
332        let mut dropped_routes_override = Region2Peers::new();
333        for overrides in dropped_assignment.region_routes_override.into_values() {
334            for (region_id, route) in overrides {
335                if region_set.contains(&region_id) {
336                    dropped_region_set.insert(region_id);
337                    dropped_routes_override.insert(region_id, route);
338                }
339            }
340        }
341
342        let (dropped_regions, active_regions): (Vec<_>, Vec<_>) = regions
343            .into_iter()
344            .partition(|region_id| dropped_region_set.contains(region_id));
345
346        let mut combined_report = GcReport::default();
347
348        if !active_regions.is_empty() {
349            let report = self
350                .ctx
351                .gc_regions(
352                    &active_regions,
353                    full_listing,
354                    gc_timeout,
355                    Region2Peers::new(),
356                    EventContext::new(TriggerReason::Manual),
357                )
358                .await?;
359            combined_report.merge(report);
360        }
361
362        if !dropped_regions.is_empty() {
363            let report = self
364                .ctx
365                .gc_regions(
366                    &dropped_regions,
367                    true,
368                    gc_timeout,
369                    dropped_routes_override,
370                    EventContext::new(TriggerReason::Manual),
371                )
372                .await?;
373            combined_report.merge(report);
374        }
375
376        let report = GcJobReport::combined(combined_report);
377
378        info!("Finished manual gc request");
379        Ok(report)
380    }
