15use std::time::Duration;
16
17use common_base::readable_size::ReadableSize;
18use serde::{Deserialize, Serialize};
19
20/// Generation of the node prefix in standalone mode.
21///
22/// Standalone runs a single datanode that never changes generation, so the
23/// prefix layout carries the constant. Distributed mode will take both the
24/// node id and the generation from the metasrv.
25pub const STANDALONE_GENERATION: u64 = 0;
26
27/// What a read does with a segment that still does not decode after a
28/// second fetch, because its checksum does not match or its content
29/// disagrees with its footer entry.
30#[derive(Debug, Clone, Copy, Default, Serialize, Deserialize, PartialEq, Eq)]
31#[serde(rename_all = "snake_case")]
32pub enum CorruptedSegmentAction {
33    /// The segment is skipped and recorded as a hole of its region; the other
34    /// regions of the object are unaffected.
35    #[default]
36    Skip,
37    /// The read fails.
38    Fail,
39}
40
41/// Object store wal configurations for datanode.
42#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
43#[serde(default)]
44pub struct ObjectStoreWalConfig {
45    /// Name of the storage provider that holds the WAL objects.
46    /// An empty name selects the default object store.
47    pub storage_provider: String,
48    /// Path prefix of the WAL objects inside the storage provider.
49    ///
50    /// The store runs under a node prefix derived from it, see
51    /// [`node_prefix`](Self::node_prefix).
52    pub prefix: String,
53    /// Interval of flushing buffered entries to the object store, at least
54    /// 10ms, defaults to 100ms.
55    #[serde(with = "humantime_serde")]
56    pub flush_interval: Duration,
57    /// The max size of a single batch object, defaults to 8MiB.
58    pub max_batch_bytes: ReadableSize,
59    /// What a read does with a segment that still does not decode after a
60    /// second fetch, because its checksum does not match or its content
61    /// disagrees with its footer entry, defaults to `skip`.
62    pub on_corrupted_segment: CorruptedSegmentAction,
63}
64
65impl Default for ObjectStoreWalConfig {
66    fn default() -> Self {
67        Self {
68            storage_provider: String::new(),
69            prefix: "wal".to_string(),
70            flush_interval: Duration::from_millis(100),
71            max_batch_bytes: ReadableSize::mb(8),
72            on_corrupted_segment: CorruptedSegmentAction::Skip,
73        }
74    }
75}
76
77impl ObjectStoreWalConfig {
78    /// Returns the prefix the store of `node_id` runs under in `generation`:
79    /// `<prefix>/datanodes/<node_id>/epochs/<generation>`.
80    ///
81    /// Every place that names the store's prefix, whether to open the store or
82    /// to allocate the WAL options of a region, must derive it here so the
83    /// prefix a region persists matches the one its store runs under.
84    pub fn node_prefix(&self, node_id: u64, generation: u64) -> String {
85        format!("{}/datanodes/{node_id}/epochs/{generation}", self.prefix)
86    }
87}

89#[cfg(test)]
90mod tests {
91    use super::*;
92
93    #[test]
94    fn test_node_prefix() {
95        let config = ObjectStoreWalConfig::default();
96        assert_eq!(
97            config.node_prefix(0, STANDALONE_GENERATION),
98            "wal/datanodes/0/epochs/0"
99        );
100
101        let config = ObjectStoreWalConfig {
102            prefix: "cluster-a/wal".to_string(),
103            ..Default::default()
104        };
105        assert_eq!(
106            config.node_prefix(3, 7),
107            "cluster-a/wal/datanodes/3/epochs/7"
108        );
109    }
110}
