33/// Wal configurations for metasrv.
34#[derive(Debug, Serialize, Deserialize, Clone, PartialEq, Default)]
35#[serde(tag = "provider", rename_all = "snake_case")]
36#[allow(clippy::large_enum_variant)]
37pub enum MetasrvWalConfig {
38    #[default]
39    RaftEngine,
40    Kafka(MetasrvKafkaConfig),
41}
42
43#[allow(clippy::large_enum_variant)]
44/// Wal configurations for datanode.
45#[derive(Debug, Serialize, Deserialize, Clone, PartialEq)]
46#[serde(tag = "provider", rename_all = "snake_case")]
47pub enum DatanodeWalConfig {
48    RaftEngine(RaftEngineConfig),
49    Kafka(DatanodeKafkaConfig),
50    Noop,
51    #[serde(rename = "experimental_object_store")]
52    ObjectStore(ObjectStoreWalConfig),
53}
54
55impl Default for DatanodeWalConfig {
56    fn default() -> Self {
57        Self::RaftEngine(RaftEngineConfig::default())
58    }
59}
60
61impl TryFrom<DatanodeWalConfig> for MetasrvWalConfig {
62    type Error = Error;
63
64    fn try_from(config: DatanodeWalConfig) -> Result<Self, Self::Error> {
65        match config {
66            DatanodeWalConfig::RaftEngine(_) => Ok(Self::RaftEngine),
67            DatanodeWalConfig::Kafka(config) => Ok(Self::Kafka(MetasrvKafkaConfig {
68                connection: config.connection,
69                kafka_topic: config.kafka_topic,
70                auto_create_topics: config.auto_create_topics,
71                // This field won't be used in standalone mode
72                auto_prune_interval: DEFAULT_AUTO_PRUNE_INTERVAL,
73                // This field won't be used in standalone mode
74                auto_prune_logical_delete: DEFAULT_AUTO_PRUNE_LOGICAL_DELETE,
75                // This field won't be used in standalone mode
76                auto_prune_parallelism: DEFAULT_AUTO_PRUNE_PARALLELISM,
77                // This field won't be used in standalone mode
78                flush_trigger_size: DEFAULT_FLUSH_TRIGGER_SIZE,
79                // This field won't be used in standalone mode
80                checkpoint_trigger_size: DEFAULT_CHECKPOINT_TRIGGER_SIZE,
81                // This field won't be used in standalone mode
82                region_flush_trigger_interval: DEFAULT_REGION_FLUSH_TRIGGER_INTERVAL,
83                // This field won't be used in standalone mode
84                periodic_checkpoint_persist_interval: DEFAULT_PERIODIC_CHECKPOINT_PERSIST_INTERVAL,
85            })),
86            DatanodeWalConfig::Noop => UnsupportedWalProviderSnafu {
87                provider: "noop".to_string(),
88            }
89            .fail(),
90            DatanodeWalConfig::ObjectStore(_) => UnsupportedWalProviderSnafu {
91                provider: "experimental_object_store".to_string(),
92            }
93            .fail(),
94        }
95    }
96}

98impl MetasrvWalConfig {
99    /// Returns if active wal pruning is enabled.
100    pub fn enable_active_wal_pruning(&self) -> bool {
101        match self {
102            MetasrvWalConfig::RaftEngine => false,
103            MetasrvWalConfig::Kafka(config) => config.auto_prune_interval > Duration::ZERO,
104        }
105    }
106
107    /// Gets the kafka connection config.
108    pub fn remote_wal_options(&self) -> Option<&MetasrvKafkaConfig> {
109        match self {
110            MetasrvWalConfig::RaftEngine => None,
111            MetasrvWalConfig::Kafka(config) => Some(config),
112        }
113    }
114}
115
116impl From<MetasrvWalConfig> for DatanodeWalConfig {
117    fn from(config: MetasrvWalConfig) -> Self {
118        match config {
119            MetasrvWalConfig::RaftEngine => Self::RaftEngine(RaftEngineConfig::default()),
120            MetasrvWalConfig::Kafka(config) => Self::Kafka(DatanodeKafkaConfig {
121                connection: config.connection,
122                kafka_topic: config.kafka_topic,
123                ..Default::default()
124            }),
125        }
126    }
127}

281    #[test]
282    fn test_toml_object_store() {
283        let toml_str = r#"
284            provider = "experimental_object_store"
285        "#;
286        let datanode_wal_config: DatanodeWalConfig = toml::from_str(toml_str).unwrap();
287        assert_eq!(
288            datanode_wal_config,
289            DatanodeWalConfig::ObjectStore(ObjectStoreWalConfig::default())
290        );
291        let DatanodeWalConfig::ObjectStore(config) = &datanode_wal_config else {
292            unreachable!()
293        };
294        assert_eq!(config.prefix, "wal");
295        assert_eq!(config.flush_interval, Duration::from_millis(100));
296        assert_eq!(config.max_batch_bytes, ReadableSize::mb(8));
297        assert_eq!(config.on_corrupted_segment, CorruptedSegmentAction::Skip);
298        assert!(matches!(
299            MetasrvWalConfig::try_from(datanode_wal_config).unwrap_err(),
300            Error::UnsupportedWalProvider { .. }
301        ));
302
303        let toml_str = r#"
304            provider = "experimental_object_store"
305            storage_provider = "s3"
306            prefix = "cluster-a/wal"
307            flush_interval = "500ms"
308            max_batch_bytes = "4MB"
309            on_corrupted_segment = "fail"
310        "#;
311        let datanode_wal_config: DatanodeWalConfig = toml::from_str(toml_str).unwrap();
312        let expected = ObjectStoreWalConfig {
313            storage_provider: "s3".to_string(),
314            prefix: "cluster-a/wal".to_string(),
315            flush_interval: Duration::from_millis(500),
316            max_batch_bytes: ReadableSize::mb(4),
317            on_corrupted_segment: CorruptedSegmentAction::Fail,
318        };
319        assert_eq!(
320            datanode_wal_config,
321            DatanodeWalConfig::ObjectStore(expected)
322        );
323
324        // The persisted tag is not accepted as a config provider.
325        let toml_str = r#"
326            provider = "object_store"
327        "#;
328        assert!(toml::from_str::<DatanodeWalConfig>(toml_str).is_err());
329    }

354#[cfg(test)]
355mod rskafka_contract_tests {
356    use std::sync::Arc;
357    use std::time::Duration;
358
359    use rskafka::build_info::DEFAULT_CLIENT_ID;
360    use rskafka::messenger::{Messenger, RequestError, SyncVersionsError};
361
362    #[tokio::test]
363    async fn test_send_timeout_poisoned_messenger() {
364        let (_peer, stream) = tokio::io::duplex(1);
365        let mut messenger = Messenger::new(
366            stream,
367            1_000,
368            Arc::from(DEFAULT_CLIENT_ID),
369            Some(Duration::from_millis(50)),
370        );
371
372        let error = messenger.sync_versions().await.unwrap_err();
373        assert!(matches!(
374            error,
375            SyncVersionsError::RequestError(RequestError::Poisoned(error))
376                if matches!(
377                    error.as_ref(),
378                    RequestError::IO(error) if error.kind() == std::io::ErrorKind::TimedOut
379                )
380        ));
381    }
