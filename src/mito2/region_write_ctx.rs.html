34/// Notifier to notify write result on drop.
35struct WriteNotify {
36    /// Error to send to the waiter.
37    err: Option<Arc<Error>>,
38    /// Sender to send write result to the waiter for this mutation.
39    sender: OptionOutputTx,
40    /// Number of rows to be written.
41    num_rows: usize,
42}
43
44impl WriteNotify {
45    /// Creates a new notify from the `sender`.
46    fn new(sender: OptionOutputTx, num_rows: usize) -> WriteNotify {
47        WriteNotify {
48            err: None,
49            sender,
50            num_rows,
51        }
52    }
53
54    /// Send result to the waiter.
55    fn notify_result(&mut self) {
56        if let Some(err) = &self.err {
57            // Try to send the error to waiters.
58            self.sender
59                .send_mut(Err(err.clone()).context(WriteGroupSnafu));
60        } else {
61            // Send success result.
62            self.sender.send_mut(Ok(self.num_rows));
63        }
64    }
65}
66
67impl Drop for WriteNotify {
68    fn drop(&mut self) {
69        self.notify_result();
70    }
71}

73/// Context to keep region metadata and buffer write requests.
74pub(crate) struct RegionWriteCtx {
75    /// Id of region to write.
76    region_id: RegionId,
77    /// Version of the region while creating the context.
78    version: VersionRef,
79    /// VersionControl of the region.
80    version_control: VersionControlRef,
81    /// Next sequence number to write.
82    ///
83    /// The context assigns a unique sequence number for each row.
84    next_sequence: SequenceNumber,
85    /// Next entry id of WAL to write.
86    next_entry_id: EntryId,
87    /// Valid WAL entry to write.
88    ///
89    /// We keep [WalEntry] instead of mutations to avoid taking mutations
90    /// out of the context to construct the wal entry when we write to the wal.
91    wal_entry: WalEntry,
92    /// Mutations that skip WAL, paired with their write notifiers.
93    memtable_mutations: Vec<(Mutation, WriteNotify)>,
94    /// Wal options of the region being written to.
95    provider: Provider,
96    /// Notifiers to send write results to waiters.
97    ///
98    /// The i-th notify is for the i-th mutation in `wal_entry`.
99    wal_notifiers: Vec<WriteNotify>,
100    /// Notifiers for bulk requests.
101    bulk_notifiers: Vec<WriteNotify>,
102    /// Pending bulk write requests
103    pub(crate) bulk_parts: Vec<BulkPart>,
104    /// The write operation is failed and we should not write to the mutable memtable.
105    failed: bool,
106
107    // Metrics:
108    /// Rows to put.
109    pub(crate) put_num: usize,
110    /// Rows to delete.
111    pub(crate) delete_num: usize,
112    /// The total bytes written to the region.
113    pub(crate) written_bytes: Option<Arc<AtomicU64>>,
114}

136            next_entry_id: last_entry_id + 1,
137            wal_entry: WalEntry::default(),
138            memtable_mutations: Vec::new(),
139            provider,
140            wal_notifiers: Vec::new(),
141            bulk_notifiers: vec![],

143            put_num: 0,
144            delete_num: 0,

150    /// Push mutation to the context.
151    /// This method adopts the sequence number in parameters if present.
152    pub(crate) fn push_mutation(
153        &mut self,
154        op_type: i32,
155        rows: Option<Rows>,
156        write_hint: Option<WriteHint>,
157        tx: OptionOutputTx,
158        sequence: Option<SequenceNumber>,
159        skip_wal: bool,
160    ) {
161        if let Some(sequence) = sequence {
162            self.next_sequence = sequence;
163        }
164        let num_rows = rows.as_ref().map(|rows| rows.rows.len()).unwrap_or(0);
165        let mutation = Mutation {
166            op_type,
167            sequence: self.next_sequence,
168            rows,
169            write_hint,
170        };
171
172        // Assign sequences before routing so concurrent memtable writes retain
173        // their logical order regardless of the WAL policy.
174        let notify = WriteNotify::new(tx, num_rows);
175        if skip_wal {
176            self.memtable_mutations.push((mutation, notify));
177        } else {
178            self.wal_entry.mutations.push(mutation);
179            self.wal_notifiers.push(notify);
180        }
181
182        // Increase sequence number.
183        self.next_sequence += num_rows as u64;
184
185        // Update metrics.
186        match OpType::try_from(op_type) {
187            Ok(OpType::Delete) => self.delete_num += num_rows,
188            Ok(OpType::Put) => self.put_num += num_rows,
189            Err(_) => (),
190        }
191    }

217    /// Returns whether writes in this context should skip WAL.
218    pub(crate) fn skip_wal(&self) -> bool {
219        self.provider == Provider::Noop
220            || self.version.options.skip_wal
221            || (self.wal_entry.mutations.is_empty() && self.wal_entry.bulk_entries.is_empty())
222    }

224    /// Sets error and marks all write operations are failed.
225    pub(crate) fn set_error(&mut self, err: Arc<Error>) {
226        // Set error for all notifiers.
227        for notify in self
228            .wal_notifiers
229            .iter_mut()
230            .chain(self.memtable_mutations.iter_mut().map(|(_, notify)| notify))
231        {
232            notify.err = Some(err.clone());
233        }
234        for notify in &mut self.bulk_notifiers {
235            notify.err = Some(err.clone());
236        }
237
238        // Fail the whole write operation.
239        self.failed = true;
240    }

243    pub(crate) fn is_failed(&self) -> bool {
244        self.failed
245    }

258    /// Consumes mutations and writes them into mutable memtable.
259    pub(crate) async fn write_memtable(&mut self) {
260        debug_assert_eq!(self.wal_notifiers.len(), self.wal_entry.mutations.len());
261
262        if self.failed {
263            return;
264        }
265
266        let mutable_memtable = self.version.memtables.mutable.clone();
267        let prev_memory_usage = if self.written_bytes.is_some() {
268            Some(mutable_memtable.memory_usage())
269        } else {
270            None
271        };
272
273        let mut mutations = mem::take(&mut self.wal_entry.mutations)
274            .into_iter()
275            .zip(&mut self.wal_notifiers)
276            .chain(
277                self.memtable_mutations
278                    .iter_mut()
279                    // Keep notifiers in the context until all writes complete.
280                    .map(|(mutation, notify)| (mem::take(mutation), notify)),
281            )
282            .filter_map(|(mutation, notify)| {
283                let kvs = KeyValues::new(&self.version.metadata, mutation)?;
284                Some((notify, kvs))
285            })
286            .collect::<Vec<_>>();
287
288        if mutations.len() == 1 {
289            if let Err(err) = mutable_memtable.write(&mutations[0].1) {
290                mutations[0].0.err = Some(Arc::new(err));
291            }
292        } else {
293            let mut tasks = FuturesUnordered::new();
294            for (notify, kvs) in mutations {
295                let mutable = mutable_memtable.clone();
296                // use tokio runtime to schedule tasks.
297                let task = common_runtime::spawn_blocking_global(move || mutable.write(&kvs));
298                tasks.push(async move { (notify, task.await) });
299            }
300
301            while let Some((notify, result)) = tasks.next().await {
302                // First unwrap the result from `spawn` above.
303                if let Err(err) = result.unwrap() {
304                    notify.err = Some(Arc::new(err));
305                }
306            }
307        }
308
309        if let Some(written_bytes) = &self.written_bytes {
310            let new_memory_usage = mutable_memtable.memory_usage();
311            let bytes = new_memory_usage.saturating_sub(prev_memory_usage.unwrap_or_default());
312            written_bytes.fetch_add(bytes as u64, Ordering::Relaxed);
313        }
314    }

316    pub(crate) fn push_bulk(
317        &mut self,
318        sender: OptionOutputTx,
319        mut bulk: BulkPart,
320        sequence: Option<SequenceNumber>,
321    ) -> bool {
322        if let Some(sequence) = sequence {
323            self.next_sequence = sequence;
324        }
325        bulk.sequence = self.next_sequence;
326        let entry = match BulkWalEntry::try_from(&bulk) {
327            Ok(entry) => entry,
328            Err(e) => {
329                sender.send(Err(e));
330                return false;
331            }
332        };
333
334        self.bulk_notifiers
335            .push(WriteNotify::new(sender, bulk.num_rows()));
336
337        // Add bulk wal entry
338        self.wal_entry.bulk_entries.push(entry);
339        self.next_sequence += bulk.num_rows() as u64;
340        self.bulk_parts.push(bulk);
341        true
342    }

344    pub(crate) async fn write_bulk(&mut self) {
345        if self.failed || self.bulk_parts.is_empty() {
346            return;
347        }
348        #[cfg(test)]
349        test_hooks::pause_before_bulk_install(self.region_id, &self.version_control).await;
350        let _timer = metrics::REGION_WORKER_HANDLE_WRITE_ELAPSED
351            .with_label_values(&["write_bulk"])
352            .start_timer();
353
354        let mutable_memtable = &self.version.memtables.mutable;
355        let prev_memory_usage = if self.written_bytes.is_some() {
356            Some(mutable_memtable.memory_usage())
357        } else {
358            None
359        };
360
361        if self.bulk_parts.len() == 1 {
362            let part = self.bulk_parts.swap_remove(0);
363            let num_rows = part.num_rows();
364            if let Err(e) = self.version.memtables.mutable.write_bulk(part) {
365                self.bulk_notifiers[0].err = Some(Arc::new(e));
366            } else {
367                self.put_num += num_rows;
368            }
369            return;
370        }
371
372        let mut tasks = FuturesUnordered::new();
373        for (i, part) in self.bulk_parts.drain(..).enumerate() {
374            let mutable = mutable_memtable.clone();
375            tasks.push(common_runtime::spawn_blocking_global(move || {
376                let num_rows = part.num_rows();
377                (i, mutable.write_bulk(part), num_rows)
378            }));
379        }
380        while let Some(result) = tasks.next().await {
381            // first unwrap the result from `spawn` above
382            let (i, result, num_rows) = result.unwrap();
383            if let Err(err) = result {
384                self.bulk_notifiers[i].err = Some(Arc::new(err));
385            } else {
386                self.put_num += num_rows;
387            }
388        }
389
390        if let Some(written_bytes) = &self.written_bytes {
391            let new_memory_usage = mutable_memtable.memory_usage();
392            let bytes = new_memory_usage.saturating_sub(prev_memory_usage.unwrap_or_default());
393            written_bytes.fetch_add(bytes as u64, Ordering::Relaxed);
394        }
395    }

397    /// Publishes the assigned sequences and entry id to the region's committed
398    /// watermark. Only call after both [`write_memtable`](Self::write_memtable)
399    /// and [`write_bulk`](Self::write_bulk) have completed; a failed context
400    /// must not publish.
401    pub(crate) fn publish_sequence_and_entry_id(&self) {
402        if self.failed {
403            return;
404        }
405        self.version_control
406            .set_sequence_and_entry_id(self.next_sequence - 1, self.next_entry_id - 1);
407    }

411#[cfg(test)]
412pub(crate) mod test_hooks {
413    use std::sync::Mutex;

421    /// Channels of an armed bulk-install barrier; dropping the senders (by
422    /// disarming) unblocks writes paused on it.
423    struct ActiveBarrier {
424        id: u64,
425        /// Only bulk writes for this region on this version control (Arc
426        /// identity) pause at the barrier.
427        target_region_id: RegionId,
428        target_version_control: VersionControlRef,
429        reached: watch::Sender<bool>,
430        release: watch::Sender<bool>,
431    }
432
433    static ACTIVE_BARRIER: Mutex<Option<ActiveBarrier>> = Mutex::new(None);
434    static NEXT_BARRIER_ID: AtomicU64 = AtomicU64::new(1);
435
436    fn lock_active_barrier() -> std::sync::MutexGuard<'static, Option<ActiveBarrier>> {
437        // Never let a poisoned mutex (e.g. a panic in another test while
438        // holding the lock) hang or break unrelated tests.
439        ACTIVE_BARRIER
440            .lock()
441            .unwrap_or_else(|poisoned| poisoned.into_inner())
442    }
443
444    /// RAII guard: releasing (or dropping) it unblocks a paused write and
445    /// disarms the barrier.
446    pub(crate) struct BulkInstallBarrier {
447        id: u64,
448        reached_rx: watch::Receiver<bool>,
449        release_tx: watch::Sender<bool>,
450        released: bool,
451    }
452
453    impl BulkInstallBarrier {
454        pub(crate) async fn wait_until_reached(&mut self) {
455            if !*self.reached_rx.borrow() {
456                let _ = self.reached_rx.wait_for(|reached| *reached).await;
457            }
458        }
459
460        pub(crate) fn release(&mut self) {
461            if self.released {
462                return;
463            }
464            self.released = true;
465            let _ = self.release_tx.send(true);
466            disarm_barrier(self.id);
467        }
468    }
469
470    impl Drop for BulkInstallBarrier {
471        fn drop(&mut self) {
472            self.release();
473        }
474    }
475
476    /// Arms the bulk-install barrier and returns the owning guard; any
477    /// previously armed barrier is replaced.
478    pub(crate) fn arm_bulk_install_barrier(
479        target_region_id: RegionId,
480        target_version_control: VersionControlRef,
481    ) -> BulkInstallBarrier {
482        let (reached_tx, reached_rx) = watch::channel(false);
483        let (release_tx, _release_rx) = watch::channel(false);
484        let id = NEXT_BARRIER_ID.fetch_add(1, Ordering::Relaxed);
485        let mut active = lock_active_barrier();
486        *active = Some(ActiveBarrier {
487            id,
488            target_region_id,
489            target_version_control,
490            reached: reached_tx,
491            release: release_tx.clone(),
492        });
493        BulkInstallBarrier {
494            id,
495            reached_rx,
496            release_tx,
497            released: false,
498        }
499    }
500
501    fn disarm_barrier(id: u64) {
502        let mut active = lock_active_barrier();
503        if active.as_ref().is_some_and(|barrier| barrier.id == id) {
504            *active = None;
505        }
506    }
507
508    /// Pauses a bulk write before installing its parts until the barrier is
509    /// released or disarmed.
510    pub(crate) async fn pause_before_bulk_install(
511        region_id: RegionId,
512        version_control: &VersionControlRef,
513    ) {
514        let (reached_tx, release_rx) = {
515            let active = lock_active_barrier();
516            match active.as_ref() {
517                Some(barrier)
518                    if barrier.target_region_id == region_id
519                        && std::sync::Arc::ptr_eq(
520                            &barrier.target_version_control,
521                            version_control,
522                        ) =>
523                {
524                    (barrier.reached.clone(), barrier.release.subscribe())
525                }
526                _ => return,
527            }
528        };
529        let _ = reached_tx.send(true);
530        let mut release_rx = release_rx;
531        if !*release_rx.borrow() {
532            // The sender is dropped when the barrier is disarmed, which makes
533            // `wait_for` return an error instead of hanging forever.
534            let _ = release_rx.wait_for(|released| *released).await;
535        }
536    }
537}

562    fn check_request_skip_wal_preserves_sequences_and_other_writes(skip_wal: bool) {
563        let builder = VersionControlBuilder::new();
564        let region_id = builder.region_id();
565        let version_control = Arc::new(builder.build());
566        let mut ctx = RegionWriteCtx::new(

568            &version_control,
569            Provider::raft_engine_provider(region_id.as_u64()),
570            None,
571        );
572        for (op_type, skip, num_rows) in [
573            (OpType::Put, skip_wal, 2),
574            (OpType::Put, false, 3),
575            (OpType::Delete, false, 1),
576        ] {

578                op_type as i32,

581                    rows: vec![api::v1::Row::default(); num_rows],

583                None,
584                OptionOutputTx::none(),

586                skip,
587            );
588        }

590        for (mutation, notify) in ctx
591            .wal_entry
592            .mutations
593            .iter()
594            .zip(&ctx.wal_notifiers)
595            .chain(
596                ctx.memtable_mutations
597                    .iter()
598                    .map(|(mutation, notify)| (mutation, notify)),
599            )
600        {
601            assert_eq!(mutation.rows.as_ref().unwrap().rows.len(), notify.num_rows);
602        }
603        assert!(ctx.push_bulk(OptionOutputTx::none(), new_bulk_part(), None));
604        assert!(!ctx.skip_wal());
605        assert_eq!(ctx.next_sequence, 9);
606        assert_eq!(ctx.wal_entry.bulk_entries.len(), 1);
607        assert_eq!(ctx.bulk_parts[0].sequence, 7);
608        let sequences: Vec<_> = ctx.wal_entry.mutations.iter().map(|m| m.sequence).collect();
609        assert_eq!(sequences, if skip_wal { vec![3, 6] } else { vec![1, 3, 6] });
610        // Check the actual WAL bytes after routing the mutations.
611        let encoded = crate::wal::encoder::WalEntryEncoder::new().encode_to_vec(&ctx.wal_entry);

617                .map(|m| m.sequence)
618                .collect::<Vec<_>>(),
619            sequences
620        );
621        assert_eq!(decoded.bulk_entries, ctx.wal_entry.bulk_entries);
622        assert_eq!(ctx.wal_entry.mutations.len(), if skip_wal { 2 } else { 3 });
623        assert_eq!(ctx.memtable_mutations.len(), usize::from(skip_wal));
624        if skip_wal {
625            assert_eq!(ctx.memtable_mutations[0].0.sequence, 1);
626        }
627        assert_eq!(
628            ctx.wal_entry.mutations.last().unwrap().op_type,
629            OpType::Delete as i32
630        );
631    }
