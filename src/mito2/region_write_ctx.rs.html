34/// Notifier to notify write result on drop.
35struct WriteNotify {
36    /// Error to send to the waiter.
37    err: Option<Arc<Error>>,
38    /// Sender to send write result to the waiter for this mutation.
39    sender: OptionOutputTx,
40    /// Number of rows to be written.
41    num_rows: usize,
42}
43
44impl WriteNotify {
45    /// Creates a new notify from the `sender`.
46    fn new(sender: OptionOutputTx, num_rows: usize) -> WriteNotify {
47        WriteNotify {
48            err: None,
49            sender,
50            num_rows,
51        }
52    }
53
54    /// Send result to the waiter.
55    fn notify_result(&mut self) {
56        if let Some(err) = &self.err {
57            // Try to send the error to waiters.
58            self.sender
59                .send_mut(Err(err.clone()).context(WriteGroupSnafu));
60        } else {
61            // Send success result.
62            self.sender.send_mut(Ok(self.num_rows));
63        }
64    }
65}
66
67impl Drop for WriteNotify {
68    fn drop(&mut self) {
69        self.notify_result();
70    }
71}

80/// Context to keep region metadata and buffer write requests.
81pub(crate) struct RegionWriteCtx {
82    /// Source of the buffered writes.
83    source: WriteSource,
84    /// Whether accepted non-empty writes bypass WAL. WAL replay never sets this flag.
85    has_skip_wal_writes: bool,
86    /// Id of region to write.
87    region_id: RegionId,
88    /// Version of the region while creating the context.
89    version: VersionRef,
90    /// VersionControl of the region.
91    version_control: VersionControlRef,
92    /// Next sequence number to write.
93    ///
94    /// The context assigns a unique sequence number for each row.
95    next_sequence: SequenceNumber,
96    /// Next entry id of WAL to write.
97    next_entry_id: EntryId,
98    /// Valid WAL entry to write.
99    ///
100    /// We keep [WalEntry] instead of mutations to avoid taking mutations
101    /// out of the context to construct the wal entry when we write to the wal.
102    wal_entry: WalEntry,
103    /// Mutations that skip WAL, paired with their write notifiers.
104    memtable_mutations: Vec<(Mutation, WriteNotify)>,
105    /// Wal options of the region being written to.
106    provider: Provider,
107    /// Notifiers to send write results to waiters.
108    ///
109    /// The i-th notify is for the i-th mutation in `wal_entry`.
110    wal_notifiers: Vec<WriteNotify>,
111    /// Notifiers for bulk requests.
112    bulk_notifiers: Vec<WriteNotify>,
113    /// Pending bulk write requests
114    pub(crate) bulk_parts: Vec<BulkPart>,
115    /// The write operation is failed and we should not write to the mutable memtable.
116    failed: bool,
117
118    // Metrics:
119    /// Rows to put.
120    pub(crate) put_num: usize,
121    /// Rows to delete.
122    pub(crate) delete_num: usize,
123    /// The total bytes written to the region.
124    pub(crate) written_bytes: Option<Arc<AtomicU64>>,
125}

148            version_control: version_control.clone(),
149            next_sequence: committed_sequence + 1,
150            next_entry_id: last_entry_id + 1,
151            wal_entry: WalEntry::default(),
152            memtable_mutations: Vec::new(),
153            provider,
154            wal_notifiers: Vec::new(),
155            bulk_notifiers: vec![],
156            failed: false,
157            put_num: 0,
158            delete_num: 0,
159            bulk_parts: vec![],
160            written_bytes,
161        }

164    /// Push mutation to the context.
165    /// This method adopts the sequence number in parameters if present.
166    pub(crate) fn push_mutation(
167        &mut self,
168        op_type: i32,
169        rows: Option<Rows>,
170        write_hint: Option<WriteHint>,
171        tx: OptionOutputTx,
172        sequence: Option<SequenceNumber>,
173        skip_wal: bool,
174    ) {
175        if let Some(sequence) = sequence {
176            self.next_sequence = sequence;
177        }
178        let num_rows = rows.as_ref().map(|rows| rows.rows.len()).unwrap_or(0);
179        self.has_skip_wal_writes |= num_rows > 0 && self.is_skip_wal_write(skip_wal);
180        let mutation = Mutation {
181            op_type,
182            sequence: self.next_sequence,
183            rows,
184            write_hint,
185        };
186
187        // Assign sequences before routing so concurrent memtable writes retain
188        // their logical order regardless of the WAL policy.
189        let notify = WriteNotify::new(tx, num_rows);
190        if skip_wal {
191            self.memtable_mutations.push((mutation, notify));
192        } else {
193            self.wal_entry.mutations.push(mutation);
194            self.wal_notifiers.push(notify);
195        }
196
197        // Increase sequence number.
198        self.next_sequence += num_rows as u64;
199
200        // Update metrics.
201        match OpType::try_from(op_type) {
202            Ok(OpType::Delete) => self.delete_num += num_rows,
203            Ok(OpType::Put) => self.put_num += num_rows,
204            Err(_) => (),
205        }
206    }

232    /// Returns whether writes in this context should skip WAL.
233    pub(crate) fn skip_wal(&self) -> bool {
234        self.provider == Provider::Noop
235            || self.version.options.skip_wal
236            || (self.wal_entry.mutations.is_empty() && self.wal_entry.bulk_entries.is_empty())
237    }

239    /// Sets error and marks all write operations are failed.
240    pub(crate) fn set_error(&mut self, err: Arc<Error>) {
241        // Set error for all notifiers.
242        for notify in self
243            .wal_notifiers
244            .iter_mut()
245            .chain(self.memtable_mutations.iter_mut().map(|(_, notify)| notify))
246        {
247            notify.err = Some(err.clone());
248        }
249        for notify in &mut self.bulk_notifiers {
250            notify.err = Some(err.clone());
251        }
252
253        // Fail the whole write operation.
254        self.failed = true;
255    }

259        self.failed

273    /// Installs ordinary and bulk writes, then publishes their sequence and entry ID.
274    pub(crate) async fn write_memtables(&mut self) {
275        if self.failed {
276            return;
277        }
278        // A failed installation may leave some rows in memory.
279        if self.has_skip_wal_writes {
280            self.version.memtables.mutable.mark_skip_wal_writes();
281        }
282        self.write_mutations().await;
283        self.write_bulk().await;
284        self.publish_sequence_and_entry_id();
285    }
286
287    fn is_skip_wal_write(&self, request_skip_wal: bool) -> bool {
288        self.source == WriteSource::Request
289            && (request_skip_wal
290                || self.provider == Provider::Noop
291                || self.version.options.skip_wal)
292    }

294    /// Consumes mutations and writes them into mutable memtables.
295    async fn write_mutations(&mut self) {
296        debug_assert_eq!(self.wal_notifiers.len(), self.wal_entry.mutations.len());
297
298        if self.failed {
299            return;
300        }
301
302        let mutable_memtable = self.version.memtables.mutable.clone();
303        let prev_memory_usage = if self.written_bytes.is_some() {
304            Some(mutable_memtable.memory_usage())
305        } else {
306            None
307        };
308
309        let mut mutations = mem::take(&mut self.wal_entry.mutations)
310            .into_iter()
311            .zip(&mut self.wal_notifiers)
312            .chain(
313                self.memtable_mutations
314                    .iter_mut()
315                    // Keep notifiers in the context until all writes complete.
316                    .map(|(mutation, notify)| (mem::take(mutation), notify)),
317            )
318            .filter_map(|(mutation, notify)| {
319                let kvs = KeyValues::new(&self.version.metadata, mutation)?;
320                Some((notify, kvs))
321            })
322            .collect::<Vec<_>>();
323
324        if mutations.len() == 1 {
325            if let Err(err) = mutable_memtable.write(&mutations[0].1) {
326                mutations[0].0.err = Some(Arc::new(err));
327            }
328        } else {
329            let mut tasks = FuturesUnordered::new();
330            for (notify, kvs) in mutations {
331                let mutable = mutable_memtable.clone();
332                // use tokio runtime to schedule tasks.
333                let task = common_runtime::spawn_blocking_global(move || mutable.write(&kvs));
334                tasks.push(async move { (notify, task.await) });
335            }
336
337            while let Some((notify, result)) = tasks.next().await {
338                // First unwrap the result from `spawn` above.
339                if let Err(err) = result.unwrap() {
340                    notify.err = Some(Arc::new(err));
341                }
342            }
343        }
344
345        if let Some(written_bytes) = &self.written_bytes {
346            let new_memory_usage = mutable_memtable.memory_usage();
347            let bytes = new_memory_usage.saturating_sub(prev_memory_usage.unwrap_or_default());
348            written_bytes.fetch_add(bytes as u64, Ordering::Relaxed);
349        }
350    }

352    pub(crate) fn push_bulk(
353        &mut self,
354        sender: OptionOutputTx,
355        mut bulk: BulkPart,
356        sequence: Option<SequenceNumber>,
357        skip_wal: bool,
358    ) -> bool {
359        if let Some(sequence) = sequence {
360            self.next_sequence = sequence;
361        }
362        bulk.sequence = self.next_sequence;
363        bulk.min_sequence = self.next_sequence;
364        if !skip_wal {
365            let entry = match BulkWalEntry::try_from(&bulk) {
366                Ok(entry) => entry,
367                Err(e) => {
368                    sender.send(Err(e));
369                    return false;
370                }
371            };
372            self.wal_entry.bulk_entries.push(entry);
373        }
374
375        self.has_skip_wal_writes |= bulk.num_rows() > 0 && self.is_skip_wal_write(skip_wal);
376        self.bulk_notifiers
377            .push(WriteNotify::new(sender, bulk.num_rows()));
378
379        self.next_sequence += bulk.num_rows() as u64;
380        self.bulk_parts.push(bulk);
381        true
382    }
383
384    async fn write_bulk(&mut self) {
385        if self.failed || self.bulk_parts.is_empty() {
386            return;
387        }
388        #[cfg(test)]
389        test_hooks::pause_before_bulk_install(self.region_id, &self.version_control).await;
390        let _timer = metrics::REGION_WORKER_HANDLE_WRITE_ELAPSED
391            .with_label_values(&["write_bulk"])
392            .start_timer();
393
394        let mutable_memtable = &self.version.memtables.mutable;
395        let prev_memory_usage = if self.written_bytes.is_some() {
396            Some(mutable_memtable.memory_usage())
397        } else {
398            None
399        };
400
401        if self.bulk_parts.len() == 1 {
402            let part = self.bulk_parts.swap_remove(0);
403            let num_rows = part.num_rows();
404            if let Err(e) = self.version.memtables.mutable.write_bulk(part) {
405                self.bulk_notifiers[0].err = Some(Arc::new(e));
406            } else {
407                self.put_num += num_rows;
408            }
409            return;
410        }
411
412        let mut tasks = FuturesUnordered::new();
413        for (i, part) in self.bulk_parts.drain(..).enumerate() {
414            let mutable = mutable_memtable.clone();
415            tasks.push(common_runtime::spawn_blocking_global(move || {
416                let num_rows = part.num_rows();
417                (i, mutable.write_bulk(part), num_rows)
418            }));
419        }
420        while let Some(result) = tasks.next().await {
421            // first unwrap the result from `spawn` above
422            let (i, result, num_rows) = result.unwrap();
423            if let Err(err) = result {
424                self.bulk_notifiers[i].err = Some(Arc::new(err));
425            } else {
426                self.put_num += num_rows;
427            }
428        }
429
430        if let Some(written_bytes) = &self.written_bytes {
431            let new_memory_usage = mutable_memtable.memory_usage();
432            let bytes = new_memory_usage.saturating_sub(prev_memory_usage.unwrap_or_default());
433            written_bytes.fetch_add(bytes as u64, Ordering::Relaxed);
434        }
435    }

437    /// Publishes the assigned sequences and entry id to the region's committed
438    /// watermark. Only call after both [`write_mutations`](Self::write_mutations)
439    /// and [`write_bulk`](Self::write_bulk) have completed; a failed context
440    /// must not publish.
441    fn publish_sequence_and_entry_id(&self) {
442        if self.failed {
443            return;
444        }
445        self.version_control
446            .set_sequence_and_entry_id(self.next_sequence - 1, self.next_entry_id - 1);
447    }

450/// Test-only hooks to make write ordering races deterministic.
451#[cfg(test)]
452pub(crate) mod test_hooks {
453    use std::sync::Mutex;
454    use std::sync::atomic::{AtomicU64, Ordering};
455
456    use store_api::storage::RegionId;
457    use tokio::sync::watch;

459    use crate::region::version::VersionControlRef;

461    /// Channels of an armed bulk-install barrier; dropping the senders (by
462    /// disarming) unblocks writes paused on it.
463    struct ActiveBarrier {
464        id: u64,
465        /// Only bulk writes for this region on this version control (Arc
466        /// identity) pause at the barrier.
467        target_region_id: RegionId,
468        target_version_control: VersionControlRef,
469        reached: watch::Sender<bool>,
470        release: watch::Sender<bool>,
471    }
472
473    static ACTIVE_BARRIER: Mutex<Option<ActiveBarrier>> = Mutex::new(None);
474    static NEXT_BARRIER_ID: AtomicU64 = AtomicU64::new(1);
475
476    fn lock_active_barrier() -> std::sync::MutexGuard<'static, Option<ActiveBarrier>> {
477        // Never let a poisoned mutex (e.g. a panic in another test while
478        // holding the lock) hang or break unrelated tests.
479        ACTIVE_BARRIER
480            .lock()
481            .unwrap_or_else(|poisoned| poisoned.into_inner())
482    }
483
484    /// RAII guard: releasing (or dropping) it unblocks a paused write and
485    /// disarms the barrier.
486    pub(crate) struct BulkInstallBarrier {
487        id: u64,
488        reached_rx: watch::Receiver<bool>,
489        release_tx: watch::Sender<bool>,
490        released: bool,
491    }
492
493    impl BulkInstallBarrier {
494        pub(crate) async fn wait_until_reached(&mut self) {
495            if !*self.reached_rx.borrow() {
496                let _ = self.reached_rx.wait_for(|reached| *reached).await;
497            }
498        }
499
500        pub(crate) fn release(&mut self) {
501            if self.released {
502                return;
503            }
504            self.released = true;
505            let _ = self.release_tx.send(true);
506            disarm_barrier(self.id);
507        }
508    }
509
510    impl Drop for BulkInstallBarrier {
511        fn drop(&mut self) {
512            self.release();
513        }
514    }
515
516    /// Arms the bulk-install barrier and returns the owning guard; any
517    /// previously armed barrier is replaced.
518    pub(crate) fn arm_bulk_install_barrier(
519        target_region_id: RegionId,
520        target_version_control: VersionControlRef,
521    ) -> BulkInstallBarrier {
522        let (reached_tx, reached_rx) = watch::channel(false);
523        let (release_tx, _release_rx) = watch::channel(false);
524        let id = NEXT_BARRIER_ID.fetch_add(1, Ordering::Relaxed);
525        let mut active = lock_active_barrier();
526        *active = Some(ActiveBarrier {
527            id,
528            target_region_id,
529            target_version_control,
530            reached: reached_tx,
531            release: release_tx.clone(),
532        });
533        BulkInstallBarrier {
534            id,
535            reached_rx,
536            release_tx,
537            released: false,
538        }
539    }
540
541    fn disarm_barrier(id: u64) {
542        let mut active = lock_active_barrier();
543        if active.as_ref().is_some_and(|barrier| barrier.id == id) {
544            *active = None;
545        }
546    }
547
548    /// Pauses a bulk write before installing its parts until the barrier is
549    /// released or disarmed.
550    pub(crate) async fn pause_before_bulk_install(
551        region_id: RegionId,
552        version_control: &VersionControlRef,
553    ) {
554        let (reached_tx, release_rx) = {
555            let active = lock_active_barrier();
556            match active.as_ref() {
557                Some(barrier)
558                    if barrier.target_region_id == region_id
559                        && std::sync::Arc::ptr_eq(
560                            &barrier.target_version_control,
561                            version_control,
562                        ) =>
563                {
564                    (barrier.reached.clone(), barrier.release.subscribe())
565                }
566                _ => return,
567            }
568        };
569        let _ = reached_tx.send(true);
570        let mut release_rx = release_rx;
571        if !*release_rx.borrow() {
572            // The sender is dropped when the barrier is disarmed, which makes
573            // `wait_for` return an error instead of hanging forever.
574            let _ = release_rx.wait_for(|released| *released).await;
575        }
576    }
577}

604    fn check_request_skip_wal_preserves_sequences_and_other_writes(
605        skip_wal: bool,
606        bulk_skip_wal: bool,
607    ) {
608        let builder = VersionControlBuilder::new();
609        let region_id = builder.region_id();
610        let version_control = Arc::new(builder.build());
611        let mut ctx = RegionWriteCtx::new(
612            region_id,
613            &version_control,
614            Provider::raft_engine_provider(region_id.as_u64()),
615            None,
616            WriteSource::Request,
617        );
618        for (op_type, skip, num_rows) in [
619            (OpType::Put, skip_wal, 2),
620            (OpType::Put, false, 3),
621            (OpType::Delete, false, 1),
622        ] {
623            ctx.push_mutation(
624                op_type as i32,
625                Some(Rows {
626                    schema: vec![],
627                    rows: vec![api::v1::Row::default(); num_rows],
628                }),
629                None,
630                OptionOutputTx::none(),
631                None,
632                skip,
633            );
634        }
635        // Unequal row counts make a notifier/mutation pairing mismatch visible.
636        for (mutation, notify) in ctx
637            .wal_entry
638            .mutations
639            .iter()
640            .zip(&ctx.wal_notifiers)
641            .chain(
642                ctx.memtable_mutations
643                    .iter()
644                    .map(|(mutation, notify)| (mutation, notify)),
645            )
646        {
647            assert_eq!(mutation.rows.as_ref().unwrap().rows.len(), notify.num_rows);
648        }
649        assert!(ctx.push_bulk(OptionOutputTx::none(), new_bulk_part(), None, bulk_skip_wal));
650        assert!(!ctx.skip_wal());
651        assert_eq!(ctx.next_sequence, 9);
652        assert_eq!(
653            ctx.wal_entry.bulk_entries.len(),
654            usize::from(!bulk_skip_wal)
655        );
656        assert_eq!(ctx.bulk_parts[0].sequence, 7);
657        assert_eq!(ctx.bulk_parts[0].min_sequence, 7);
658        let sequences: Vec<_> = ctx.wal_entry.mutations.iter().map(|m| m.sequence).collect();
659        assert_eq!(sequences, if skip_wal { vec![3, 6] } else { vec![1, 3, 6] });
660        // Check the actual WAL bytes after routing the mutations.
661        let encoded = crate::wal::encoder::WalEntryEncoder::new().encode_to_vec(&ctx.wal_entry);
662        let decoded = WalEntry::decode(encoded.as_slice()).unwrap();
663        assert_eq!(
664            decoded
665                .mutations
666                .iter()
667                .map(|m| m.sequence)
668                .collect::<Vec<_>>(),
669            sequences
670        );
671        assert_eq!(decoded.bulk_entries, ctx.wal_entry.bulk_entries);
672        assert_eq!(ctx.wal_entry.mutations.len(), if skip_wal { 2 } else { 3 });
