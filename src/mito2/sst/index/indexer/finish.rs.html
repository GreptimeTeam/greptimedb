18use common_telemetry::warn;
19use puffin::puffin_manager::{PuffinManager, PuffinWriter};

26impl Indexer {
27    pub(crate) async fn do_finish(&mut self) -> IndexOutput {
28        self.dense_pk_decoder = None;
29        if self.is_empty() {
30            return IndexOutput::default();
31        }
32        let Some(mut writer) = self.build_puffin_writer().await else {
33            self.do_abort().await;
34            return IndexOutput::default();
35        };
36        let mut output = IndexOutput::default();
37        while !self.creators.is_empty() {
38            let mut active = self.creators.remove(0);
39            match active.creator.finish(&mut writer).await {
40                Ok(entries) => {
41                    if let Err(err) = entries.iter().try_for_each(|entry| entry.validate()) {
42                        warn!(err; "Invalid index completion, region_id: {}, file_id: {}", self.region_id, self.file_id);
43                        self.failure.get_or_insert_with(|| Arc::new(err));
44                        drop(active);
45                        self.do_abort().await;
46                        return IndexOutput::default();
47                    }
48                    output.entries.extend(entries);
49                }
50                Err(err) => {
51                    if active.is_builtin() && cfg!(any(test, feature = "test")) {
52                        panic!("Failed to finish index: {err:?}");
53                    }
54                    warn!(err; "Failed to finish index, region_id: {}, file_id: {}, kinds: {:?}", self.region_id, self.file_id, active.kinds);
55                    self.record_failure(err);
56                    drop(active);
57                    self.do_abort().await;
58                    return IndexOutput::default();
59                }
60            }
61        }
62        self.do_prune_intm_sst_dir().await;
63        output.file_size = self.do_finish_puffin_writer(writer).await;
64        if output.file_size == 0 {
65            return IndexOutput::default();
66        }
67        output.version = self.index_version;
68        output
69    }
70
71    async fn build_puffin_writer(&mut self) -> Option<SstPuffinWriter> {
72        let puffin_manager = self.puffin_manager.clone()?;
73
74        let err = match puffin_manager
75            .writer(&RegionIndexId::new(
76                RegionFileId::new(self.physical_region_id, self.file_id),
77                self.index_version,
78            ))
79            .await
80        {
81            Ok(writer) => return Some(writer),
82            Err(err) => err,
83        };
84
85        if cfg!(any(test, feature = "test")) {
86            panic!(
87                "Failed to create puffin writer, region_id: {}, file_id: {}, err: {:?}",
88                self.region_id, self.file_id, err
89            );
90        } else {
91            warn!(
92                err; "Failed to create puffin writer, region_id: {}, file_id: {}",
93                self.region_id, self.file_id,
94            );
95        }
96
97        self.record_failure(BoxedError::new(err));
98        None
99    }
100
101    async fn do_finish_puffin_writer(&mut self, writer: SstPuffinWriter) -> ByteCount {
102        let err = match writer.finish().await {
103            Ok(size) => return size,
104            Err(err) => err,
105        };
106
107        // Use the production best-effort path in tests as well, so injected
108        // close failures can verify that no index descriptors are published.
109        warn!(
110            err; "Failed to finish puffin writer, region_id: {}, file_id: {}",
111            self.region_id, self.file_id,
112        );
113
114        self.record_failure(BoxedError::new(err));
115        0
116    }
117
118    pub(crate) async fn do_prune_intm_sst_dir(&mut self) {
119        if let Some(manager) = self.intermediate_manager.take()
120            && let Err(e) = manager.prune_sst_dir(&self.region_id, &self.file_id).await
121        {
122            warn!(e; "Failed to prune intermediate SST directory, region_id: {}, file_id: {}", self.region_id, self.file_id);
123        }
124    }
125}
