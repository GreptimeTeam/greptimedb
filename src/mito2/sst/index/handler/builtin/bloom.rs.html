36/// Resolved column targets and configuration shared across output files.
37#[derive(Debug)]
38pub(crate) struct BloomPlan {
39    config: BloomFilterConfig,
40    entries: Vec<PlanEntry>,
41    targets: Vec<(IndexTarget, SkippingIndexOptions)>,
42}
43
44impl BloomPlan {
45    /// Resolves targets without opening SST data or creating output files.
46    pub(crate) fn plan(
47        config: &BloomFilterConfig,
48        context: &PlanContext<'_>,
49    ) -> Result<Option<Arc<dyn PlannedIndex>>, BoxedError> {
50        let create = match context.build_type {
51            IndexBuildType::Flush => config.create_on_flush.auto(),
52            IndexBuildType::Compact => config.create_on_compaction.auto(),
53            _ => true,
54        };
55        if !create {
56            return Ok(None);
57        }
58        let mut targets = Vec::new();
59        let mut entries = Vec::new();
60        for column in &context.metadata.column_metadatas {
61            for (target, options) in
62                BloomFilterIndexer::targets(column, &context.build_type).map_err(BoxedError::new)?
63            {
64                entries.push(PlanEntry {
65                    column_id: target_column_id(&target),
66                    kind: IndexKind::Bloom,
67                    format_version: 0,
68                });
69                targets.push((target, options));
70            }
71        }
72        normalize_entries(&mut entries);
73        if entries.is_empty() {
74            return Ok(None);
75        }
76        Ok(Some(Arc::new(Self {
77            config: config.clone(),
78            entries,
79            targets,
80        })))
81    }
82}
83
84#[async_trait]
85impl PlannedIndex for BloomPlan {
86    fn kinds(&self) -> &[IndexKind] {
87        &[IndexKind::Bloom]
88    }
89
90    fn entries(&self) -> &[PlanEntry] {
91        &self.entries
92    }
93
94    async fn build(
95        &self,
96        context: &BuildContext<'_>,
97    ) -> Result<Option<Box<dyn IndexCreator>>, BoxedError> {
98        BloomFilterIndexer::new_with_targets(
99            context.output.file_id(),
100            context.metadata,
101            context.intermediate_manager.clone(),
102            self.config.mem_threshold_on_create(),
103            self.targets.clone(),
104        )
105        .map(|creator| creator.map(|c| Box::new(c) as Box<dyn IndexCreator>))
106        .map_err(BoxedError::new)
107    }
108}
109
110#[async_trait]
111impl IndexCreator for BloomFilterIndexer {
112    async fn update(&mut self, batch: &mut Batch) -> Result<(), BoxedError> {
113        Self::update(self, batch).await.map_err(BoxedError::new)
114    }
115
116    async fn update_flat(&mut self, batch: &RecordBatch) -> Result<(), BoxedError> {
117        Self::update_flat(self, batch)
118            .await
119            .map_err(BoxedError::new)
120    }
121
122    async fn finish(
123        &mut self,
124        writer: &mut SstPuffinWriter,
125    ) -> Result<Vec<CompletedEntry>, BoxedError> {
126        let mut columns: Vec<_> = Self::indexed_targets(self)
127            .filter_map(column_target_id)
128            .map(Some)
129            .collect();
130        let (_, bytes) = Self::finish(self, writer).await.map_err(BoxedError::new)?;
131        if bytes == 0 {
132            return Ok(Vec::new());
133        }
134        if columns.is_empty() {
135            columns.push(None);
136        }
137        columns.sort_unstable();
138        columns.dedup();
139        Ok(columns
140            .into_iter()
141            .map(|column_id| CompletedEntry {
142                column_id,
143                kind: IndexKind::Bloom,
144                format_version: 0,
145            })
146            .collect())
147    }
148
149    async fn abort(&mut self) -> Result<(), BoxedError> {
150        Self::abort(self).await.map_err(BoxedError::new)
151    }
152
153    fn memory_usage(&self) -> usize {
154        Self::memory_usage(self)
155    }
156
157    fn indexed_targets(&self) -> Vec<IndexTarget> {
158        Self::indexed_targets(self).cloned().collect()
159    }
160
161    fn needs_dense_pk(&self) -> bool {
162        true
163    }
164}
