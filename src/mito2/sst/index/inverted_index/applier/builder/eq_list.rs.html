26impl InvertedIndexApplierBuilder<'_> {
27    /// Collects an eq expression in the form of `column = lit`.
28    pub(crate) fn collect_eq(&mut self, left: &DfExpr, right: &DfExpr) -> Result<()> {
29        let resolved = match self.expr_to_index_target(left)? {
30            Some(target) => Some(target),
31            None => self.expr_to_index_target(right)?,
32        };
33        let Some((target, data_type)) = resolved else {
34            return Ok(());
35        };
36        let Some(lit) = Self::nonnull_lit(right).or_else(|| Self::nonnull_lit(left)) else {
37            return Ok(());
38        };
39
40        let predicate = Predicate::InList(InListPredicate {
41            list: BTreeSet::from_iter([Self::encode_lit(lit, data_type)?]),
42        });
43        self.add_predicate(target, predicate);
44        Ok(())
45    }
46
47    /// Collects eq list in the form of `column = lit OR column = lit OR ...`.
48    pub(crate) fn collect_or_eq_list(&mut self, eq_expr: &DfExpr, or_list: &DfExpr) -> Result<()> {
49        let DfExpr::BinaryExpr(BinaryExpr {
50            left,
51            op: Operator::Eq,
52            right,
53        }) = eq_expr
54        else {
55            return Ok(());
56        };
57
58        let resolved = match self.expr_to_index_target(left)? {
59            Some(target) => Some(target),
60            None => self.expr_to_index_target(right)?,
61        };
62        let Some((target, data_type)) = resolved else {
63            return Ok(());
64        };
65        let Some(lit) = Self::nonnull_lit(right).or_else(|| Self::nonnull_lit(left)) else {
66            return Ok(());
67        };
68
69        let bytes = Self::encode_lit(lit, data_type.clone())?;
70        let mut inlist = BTreeSet::from_iter([bytes]);
71
72        if self.collect_eq_list_inner(&target, &data_type, or_list, &mut inlist)? {
73            let predicate = Predicate::InList(InListPredicate { list: inlist });
74            self.add_predicate(target, predicate);
75        }
76
77        Ok(())
78    }
79
80    /// Recursively collects eq list.
81    ///
82    /// Returns false if the expression doesn't match the form then
83    /// caller can safely ignore the expression.
84    fn collect_eq_list_inner(
85        &self,
86        target: &IndexTarget,
87        data_type: &ConcreteDataType,
88        expr: &DfExpr,
89        inlist: &mut BTreeSet<Bytes>,
90    ) -> Result<bool> {
91        let DfExpr::BinaryExpr(BinaryExpr {
92            left,
93            op: op @ (Operator::Eq | Operator::Or),
94            right,
95        }) = expr
96        else {
97            return Ok(false);
98        };
99
100        if op == &Operator::Or {
101            let r = self
102                .collect_eq_list_inner(target, data_type, left, inlist)?
103                .then(|| self.collect_eq_list_inner(target, data_type, right, inlist))
104                .transpose()?
105                .unwrap_or(false);
106            return Ok(r);
107        }
108
109        if op == &Operator::Eq {
110            let resolved = match self.expr_to_index_target(left)? {
111                Some(target) => Some(target),
112                None => self.expr_to_index_target(right)?,
113            };
114            let Some((candidate, _)) = resolved else {
115                return Ok(false);
116            };
117            if target != &candidate {
118                return Ok(false);
119            }
120            let Some(lit) = Self::nonnull_lit(right).or_else(|| Self::nonnull_lit(left)) else {
121                return Ok(false);
122            };
123
124            inlist.insert(Self::encode_lit(lit, data_type.clone())?);
125            return Ok(true);
126        }
127
128        Ok(false)
129    }
130}

146    #[test]
147    fn test_collect_eq_basic() {
148        let (_d, factory) = PuffinManagerFactory::new_for_test_block("test_collect_eq_basic_");
149        let metadata = test_region_metadata();
150        let mut builder = InvertedIndexApplierBuilder::new(
151            "test".to_string(),
152            PathType::Bare,
153            test_object_store(),
154            &metadata,
155            HashSet::from_iter([1, 2, 3]),
156            factory,
157        );
158
159        builder
160            .collect_eq(&tag_column(), &string_lit("foo"))
161            .unwrap();
162        builder
163            .collect_eq(&string_lit("bar"), &tag_column())
164            .unwrap();
165
166        let predicates = builder
167            .output
168            .get(&index::target::IndexTarget::ColumnId(1))
169            .unwrap();
170        assert_eq!(predicates.len(), 2);
171        assert_eq!(
172            predicates[0],
173            Predicate::InList(InListPredicate {
174                list: BTreeSet::from_iter([encoded_string("foo")])
175            })
176        );
177        assert_eq!(
178            predicates[1],
179            Predicate::InList(InListPredicate {
180                list: BTreeSet::from_iter([encoded_string("bar")])
181            })
182        );
183    }
