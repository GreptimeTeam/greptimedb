62/// `InvertedIndexer` creates inverted index for SST files.
63pub struct InvertedIndexer {

65    index_creator: Box<dyn InvertedIndexCreator>,
66    /// The provider of intermediate files.

91impl InvertedIndexer {
92    /// Creates a new `InvertedIndexer`.
93    pub fn new(
94        sst_file_id: FileId,
95        metadata: &RegionMetadataRef,
96        intermediate_manager: IntermediateManager,
97        memory_usage_threshold: Option<usize>,
98        segment_row_count: NonZeroUsize,
99        index_targets: Vec<IndexTarget>,
100    ) -> Self {
101        let temp_file_provider = Arc::new(TempFileProvider::new(
102            IntermediateLocation::new(&metadata.region_id, &sst_file_id),
103            intermediate_manager,
104        ));
105
106        let memory_usage = Arc::new(AtomicUsize::new(0));
107
108        let sorter = ExternalSorter::factory(
109            temp_file_provider.clone() as _,
110            Some(MIN_MEMORY_USAGE_THRESHOLD_PER_COLUMN),
111            memory_usage.clone(),
112            memory_usage_threshold,
113        );
114        let index_creator = Box::new(SortIndexCreator::new(sorter, segment_row_count));
115
116        let codec = IndexValuesCodec::from_tag_columns(
117            metadata.primary_key_encoding,
118            metadata.primary_key_columns(),
119        );
120        let index_targets = index_targets
121            .into_iter()
122            .map(|target| {
123                let key = target.encode();
124                (target, key)
125            })
126            .collect();
127        Self {
128            codec,
129            index_creator,
130            temp_file_provider,
131            value_buf: vec![],
132            pk_offsets: SparseOffsetsCache::new(),
133            stats: Statistics::new(TYPE_INVERTED_INDEX),
134            aborted: false,
135            memory_usage,
136            index_targets,
137            metadata: metadata.clone(),
138        }
139    }

141    /// Updates index with a batch of rows.
142    /// Garbage will be cleaned up if failed to update.
143    pub async fn update(&mut self, batch: &mut Batch) -> Result<()> {
144        ensure!(!self.aborted, OperateAbortedIndexSnafu);
145
146        if batch.is_empty() {
147            return Ok(());
148        }
149
150        if let Err(update_err) = self.do_update(batch).await {
151            // clean up garbage if failed to update
152            if let Err(err) = self.do_cleanup().await {
153                if cfg!(any(test, feature = "test")) {
154                    panic!("Failed to clean up index creator, err: {err}",);
155                } else {
156                    warn!(err; "Failed to clean up index creator");
157                }
158            }
159            return Err(update_err);
160        }
161
162        Ok(())
163    }

165    /// Updates the inverted index with the given flat format RecordBatch.
166    pub async fn update_flat(&mut self, batch: &RecordBatch) -> Result<()> {
167        ensure!(!self.aborted, OperateAbortedIndexSnafu);
168
169        if batch.num_rows() == 0 {
170            return Ok(());
171        }
172
173        self.do_update_flat(batch).await
174    }
175
176    async fn do_update_flat(&mut self, batch: &RecordBatch) -> Result<()> {
177        let mut guard = self.stats.record_update();
178
179        guard.inc_row_count(batch.num_rows());
180
181        let is_sparse = self.metadata.primary_key_encoding == PrimaryKeyEncoding::Sparse;
182        let mut sparse_columns: SmallVec<[(ColumnId, &str); 8]> = SmallVec::new();
183
184        for (target, target_key) in &self.index_targets {
185            let (vector, sort_field, semantic_type) = match target {
186                IndexTarget::ColumnId(col_id) => {
187                    let Some(col_meta) = self.metadata.column_by_id(*col_id) else {
188                        debug!(
189                            "Column {} not found in the metadata during building inverted index",
190                            col_id
191                        );
192                        continue;
193                    };
194                    let col_name = &col_meta.column_schema.name;
195                    let Some(col_array) = batch.column_by_name(col_name) else {
196                        if is_sparse && col_meta.semantic_type == SemanticType::Tag {
197                            if self.codec.pk_col_info(*col_id).is_some() {
198                                sparse_columns.push((*col_id, target_key));
199                            }
200                        } else {
201                            debug!(
202                                "Column {} not found in the batch during building inverted index",
203                                col_id
204                            );
205                        }
206                        continue;
207                    };
208                    let vector =
209                        Helper::try_into_vector(col_array.clone()).context(ConvertVectorSnafu)?;
210                    let sort_field = SortField::new(vector.data_type());
211                    (vector, sort_field, col_meta.semantic_type)
212                }
213                IndexTarget::JsonPath(target) => {
214                    let col_id = target.column_id();
215                    let col = self.metadata.column_by_id(col_id).with_context(|| {
216                        InvalidRecordBatchSnafu {
217                            reason: format!("Missing JSON index column {col_id}"),
218                        }
219                    })?;
220                    let root =
221                        batch
222                            .column_by_name(&col.column_schema.name)
223                            .with_context(|| InvalidRecordBatchSnafu {
224                                reason: format!("Missing JSON index column {col_id} in batch"),
225                            })?;
226                    let leaf = extract_json_type_hint_values(root, target.path())?;
227                    let field = SortField::new(leaf.data_type());
228                    (leaf, field, SemanticType::Field)
229                }
230            };
231            for (row, count) in column_index_rows(batch, semantic_type) {
232                let value = IndexValueCodec::encode_value(
233                    vector.get_ref(row),
234                    &sort_field,
235                    &mut self.value_buf,
236                )
237                .context(EncodeSnafu)?;
238                if self
239                    .index_creator
240                    .push_with_name_n(target_key, value, count)
241                {
242                    self.index_creator
243                        .spill()
244                        .await
245                        .context(PushIndexValueSnafu)?;
246                }
247            }
248        }
249
250        if !sparse_columns.is_empty() {
251            for (pk, count) in PrimaryKeyRuns::try_new(batch)? {
252                let mut view =
253                    SparsePrimaryKeyView::new(pk, &mut self.pk_offsets).context(DecodeSnafu)?;
254                // Visit all needed tags before moving to the next PK so offset discovery is shared.
255                for &(col_id, target_key) in &sparse_columns {
256                    let value = IndexValueCodec::encode_sparse_value(
257                        &mut view,
258                        col_id,
259                        &mut self.value_buf,
260                    )
261                    .context(DecodeSnafu)?;
262                    if self
263                        .index_creator
264                        .push_with_name_n(target_key, value, count)
265                    {
266                        self.index_creator
267                            .spill()
268                            .await
269                            .context(PushIndexValueSnafu)?;
270                    }
271                }
272            }
273        }
274
275        Ok(())
276    }

278    /// Finishes index creation and cleans up garbage.
279    /// Returns the number of rows and bytes written.
280    pub(crate) async fn finish(
281        &mut self,
282        puffin_writer: &mut SstPuffinWriter,
283    ) -> Result<(RowCount, ByteCount)> {
284        ensure!(!self.aborted, OperateAbortedIndexSnafu);
285
286        if self.stats.row_count() == 0 {
287            // no IO is performed, no garbage to clean up, just return
288            return Ok((0, 0));
289        }
290
291        let finish_res = self.do_finish(puffin_writer).await;
292        // clean up garbage no matter finish successfully or not
293        if let Err(err) = self.do_cleanup().await {
294            if cfg!(any(test, feature = "test")) {
295                panic!("Failed to clean up index creator, err: {err}",);
296            } else {
297                warn!(err; "Failed to clean up index creator");
298            }
299        }
300
301        finish_res.map(|_| (self.stats.row_count(), self.stats.byte_count()))
302    }
303
304    /// Aborts index creation and clean up garbage.
305    pub async fn abort(&mut self) -> Result<()> {
306        if self.aborted {
307            return Ok(());
308        }
309        self.aborted = true;
310
311        self.do_cleanup().await
312    }

314    async fn do_update(&mut self, batch: &mut Batch) -> Result<()> {
315        let mut guard = self.stats.record_update();
316
317        let n = batch.num_rows();
318        guard.inc_row_count(n);
319
320        for (target, target_key) in &self.index_targets {
321            let (vector, sort_field) = match target {
322                IndexTarget::ColumnId(col_id) => {
323                    if let Some(col_info) = self.codec.pk_col_info(*col_id) {
324                        let pk_idx = col_info.idx;
325                        let field = &col_info.field;
326                        let value = batch
327                            .pk_col_value(self.codec.decoder(), pk_idx, *col_id)?
328                            .filter(|v| !v.is_null())
329                            .map(|v| {
330                                self.value_buf.clear();
331                                IndexValueCodec::encode_nonnull_value(
332                                    v.as_value_ref(),
333                                    field,
334                                    &mut self.value_buf,
335                                )
336                                .context(EncodeSnafu)?;
337                                Ok(self.value_buf.as_slice())
338                            })
339                            .transpose()?;
340
341                        if self.index_creator.push_with_name_n(target_key, value, n) {
342                            self.index_creator
343                                .spill()
344                                .await
345                                .context(PushIndexValueSnafu)?;
346                        }
347                        continue;
348                    }
349                    let Some(values) = batch.field_col_value(*col_id) else {
350                        debug!(
351                            "Column {} not found in the batch during building inverted index",
352                            col_id
353                        );
354                        continue;
355                    };
356                    let sort_field = SortField::new(values.data.data_type());
357                    (values.data.clone(), sort_field)
358                }
359                IndexTarget::JsonPath(target) => {
360                    let col_id = target.column_id();
361                    let root = batch
362                        .field_col_value(col_id)
363                        .with_context(|| InvalidRecordBatchSnafu {
364                            reason: format!("Missing JSON index column {col_id} in batch"),
365                        })?
366                        .data
367                        .to_arrow_array();
368                    let leaf = extract_json_type_hint_values(&root, target.path())?;
369                    let field = SortField::new(leaf.data_type());
370                    (leaf, field)
371                }
372            };
373            for row in 0..n {
374                let value = IndexValueCodec::encode_value(
375                    vector.get_ref(row),
376                    &sort_field,
377                    &mut self.value_buf,
378                )
379                .context(EncodeSnafu)?;
380                if self.index_creator.push_with_name(target_key, value) {
381                    self.index_creator
382                        .spill()
383                        .await
384                        .context(PushIndexValueSnafu)?;
385                }
386            }
387        }
388
389        Ok(())
390    }

392    /// Data flow of finishing index:
393    ///
394    /// ```text
395    ///                               (In Memory Buffer)
396    ///                                    ┌──────┐
397    ///  ┌─────────────┐                   │ PIPE │
398    ///  │             │ write index data  │      │
399    ///  │ IndexWriter ├──────────────────►│ tx   │
400    ///  │             │                   │      │
401    ///  └─────────────┘                   │      │
402    ///                  ┌─────────────────┤ rx   │
403    ///  ┌─────────────┐ │ read as blob    └──────┘
404    ///  │             │ │
405    ///  │ PuffinWriter├─┤
406    ///  │             │ │ copy to file    ┌──────┐
407    ///  └─────────────┘ └────────────────►│ File │
408    ///                                    └──────┘
409    /// ```
410    async fn do_finish(&mut self, puffin_writer: &mut SstPuffinWriter) -> Result<()> {
411        let mut guard = self.stats.record_finish();
412
413        let (tx, rx) = duplex(PIPE_BUFFER_SIZE_FOR_SENDING_BLOB);
414        let mut index_writer = InvertedIndexBlobWriter::new(tx.compat_write());
415
416        let (index_finish, puffin_add_blob) = futures::join!(
417            // TODO(zhongzc): config bitmap type
418            self.index_creator
419                .finish(&mut index_writer, index::bitmap::BitmapType::Roaring),
420            puffin_writer.put_blob(
421                INDEX_BLOB_TYPE,
422                rx.compat(),
423                PutOptions::default(),
424                Default::default(),
425            )
426        );
427
428        match (
429            puffin_add_blob.context(PuffinAddBlobSnafu),
430            index_finish.context(IndexFinishSnafu),
431        ) {
432            (Err(e1), Err(e2)) => BiErrorsSnafu {
433                first: Box::new(e1),
434                second: Box::new(e2),
435            }
436            .fail()?,
437
438            (Ok(_), e @ Err(_)) => e?,
439            (e @ Err(_), Ok(_)) => e.map(|_| ())?,
440            (Ok(written_bytes), Ok(_)) => {
441                guard.inc_byte_count(written_bytes);
442            }
443        }
444
445        Ok(())
446    }
447
448    async fn do_cleanup(&mut self) -> Result<()> {
449        let _guard = self.stats.record_cleanup();
450
451        self.temp_file_provider.cleanup().await
452    }
