25/// Extracts leaf values for a JSON type hint at the given path.
26///
27/// Returns an error if the path does not exist, traverses a non-object,
28/// or the leaf array cannot be converted to a vector.
29///
30/// # Details
31///
32/// The caller guarantees that the materialized leaf type matches the type hint.
33///
34/// JSON2 builders propagate ancestor nulls to child arrays. SST reads and schema
35/// adaptation preserve this invariant, so no ancestor null bitmap merging is needed.
36pub(crate) fn extract_json_type_hint_values(root: &ArrayRef, path: &[String]) -> Result<VectorRef> {
37    let mut array = root.clone();
38    for segment in path {
39        let object = array
40            .as_any()
41            .downcast_ref::<StructArray>()
42            .with_context(|| InvalidRecordBatchSnafu {
43                reason: format!("JSON index path {path:?} traverses a non-object"),
44            })?;
45        array = object
46            .column_by_name(segment)
47            .with_context(|| InvalidRecordBatchSnafu {
48                reason: format!("JSON index path {path:?} has no materialized field {segment}"),
49            })?
50            .clone();
51    }
52    Helper::try_into_vector(array).context(ConvertVectorSnafu)
53}

55/// Yields (first row, row count) for indexing a materialized column.
56/// Tags are constant within consecutive equal PK dictionary keys. Fields and
57/// timestamps must still be visited row by row. Distinct dictionary entries may
58/// contain equal PK bytes; keeping those runs separate is correct and avoids decoding.
59/// Inputs without a non-null PK dictionary fall back to visiting individual rows.
60pub(crate) fn column_index_rows(
61    batch: &RecordBatch,
62    semantic_type: SemanticType,
63) -> impl Iterator<Item = (usize, usize)> + '_ {
64    let keys = (semantic_type == SemanticType::Tag)
65        .then(|| batch.column_by_name(PRIMARY_KEY_COLUMN_NAME))
66        .flatten()
67        .and_then(|array| array.as_any().downcast_ref::<PrimaryKeyArray>())
68        .filter(|array| array.null_count() == 0 && array.values().null_count() == 0)
69        .map(|array| array.keys().values());
70    let mut row = 0;
71    std::iter::from_fn(move || {
72        if row == batch.num_rows() {
73            return None;
74        }
75        let start = row;
76        row += 1;
77        if let Some(keys) = keys {
78            while row < keys.len() && keys[row] == keys[start] {
79                row += 1;
80            }
81        }
82        Some((start, row - start))
83    })
84}

87mod tests {
88    use std::sync::Arc;

90    use datatypes::arrow::array::{Int64Array, NullArray};

97    #[test]
98    fn test_extract_json_type_hint_values_null_type() {
99        let nulls: ArrayRef = Arc::new(NullArray::new(2));
100        let root: ArrayRef = Arc::new(StructArray::new(
101            vec![Field::new("a", nulls.data_type().clone(), true)].into(),
102            vec![nulls],
103            None,
104        ));
105        // A null leaf can be extracted, but cannot be traversed as an object.
106        assert!(extract_json_type_hint_values(&root, &["a".into(), "b".into()]).is_err());
107        let leaf = extract_json_type_hint_values(&root, &["a".into()]).unwrap();
108        assert_eq!(leaf.len(), 2);
109        assert_eq!(leaf.get(0), Value::Null);
110        assert_eq!(leaf.get(1), Value::Null);
111
112        let values: ArrayRef = Arc::new(Int64Array::from(vec![None, None]));
113        let root: ArrayRef = Arc::new(StructArray::new(
114            vec![Field::new("a", values.data_type().clone(), true)].into(),
115            vec![values],
116            None,
117        ));
118        let leaf = extract_json_type_hint_values(&root, &["a".into()]).unwrap();
119        assert_eq!(leaf.len(), 2);
120        assert_eq!(leaf.get(0), Value::Null);
121        assert_eq!(leaf.get(1), Value::Null);
122    }
123
124    #[test]
125    fn test_extract_json_type_hint_values_nulls_and_slices() {
126        let child = Arc::new(Int64Array::from(vec![Some(11), None, None, None, Some(55)]));
127        let nested = StructArray::new(
128            vec![Field::new("a.b", child.data_type().clone(), true)].into(),
129            vec![child],
130            Some(NullBuffer::from(vec![true, false, false, true, true])),
131        );
132        let root: ArrayRef = Arc::new(StructArray::new(
133            vec![Field::new("nested", nested.data_type().clone(), true)].into(),
134            vec![Arc::new(nested)],
135            Some(NullBuffer::from(vec![true, true, false, true, true])),
136        ));
137        let path = vec!["nested".into(), "a.b".into()];
138        let leaf = extract_json_type_hint_values(&root, &path).unwrap();
139        assert_eq!(
140            (0..leaf.len()).map(|i| leaf.get(i)).collect::<Vec<_>>(),
141            vec![
142                Value::Int64(11),
143                Value::Null,
144                Value::Null,
145                Value::Null,
146                Value::Int64(55)
147            ]
148        );
149        let slice = extract_json_type_hint_values(&root.slice(1, 4), &path).unwrap();
150        assert_eq!(
151            (0..slice.len()).map(|i| slice.get(i)).collect::<Vec<_>>(),
152            vec![Value::Null, Value::Null, Value::Null, Value::Int64(55)]
153        );
154        assert!(extract_json_type_hint_values(&root, &["missing".into()]).is_err());
155        assert!(
156            extract_json_type_hint_values(&root, &["nested".into(), "a.b".into(), "child".into()])
157                .is_err()
158        );
159    }
