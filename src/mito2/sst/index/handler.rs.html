37/// A requested column format; typed implementation parameters stay in its plan.
38#[derive(Debug, Clone, PartialEq, Eq)]
39pub struct PlanEntry {
40    /// Column to index.
41    pub column_id: ColumnId,
42    /// Implementation owning this entry.
43    pub kind: IndexKind,
44    /// Requested format version. The manifest records extension types, not versions.
45    /// A format upgrade requires a forced rebuild; the provider validates Puffin
46    /// properties and payload compatibility before reusing input state.
47    pub format_version: u32,
48}
49
50/// Schema and options used to prepare work without reading input files.
51pub struct PlanContext<'a> {
52    /// Shared region schema.
53    pub metadata: &'a RegionMetadataRef,
54    /// Region index options.
55    pub index_options: &'a IndexOptions,
56    /// Lifecycle event requesting the build.
57    pub build_type: IndexBuildType,
58    /// Existing column descriptors of the target SST during rebuild.
59    /// Empty for new outputs, including compaction; input descriptors are available
60    /// through [`BuildContext::inputs`] when constructing a creator.
61    pub existing_indexes: &'a [ColumnIndexMetadata],
62}
63
64/// Inputs for constructing all of one handler's indexes in a single output file.
65pub struct BuildContext<'a> {
66    /// Shared output schema.
67    pub metadata: &'a RegionMetadataRef,
68    /// Physical output index identity.
69    pub output: RegionIndexId,
70    /// Output manager, possibly backed by a local write cache.
71    pub puffin_manager: &'a SstPuffinManager,
72    /// Existing intermediate-file facility shared by built-in creators.
73    pub intermediate_manager: &'a IntermediateManager,
74    /// Pinned physical input files.
75    pub inputs: &'a [FileHandle],
76    /// Manager for reading persisted input indexes. With no inputs this falls back
77    /// to the output manager; determine input availability from `inputs.is_empty()`.
78    pub input_puffin_manager: &'a SstPuffinManager,
79    /// Permits coarse input-summary reuse only for a complete single output.
80    /// The caller that owns the output plan verifies this scope before construction.
81    pub can_merge_inputs: bool,
82    /// Retained-memory allowance for an injected creator, capped by the remaining
83    /// operation budget and its plan ceiling. `None` applies no limit at this layer.
84    /// Built-in creators use spill budgets.
85    pub memory_limit: Option<usize>,
86    /// Actual output row-group size used to align inverted-index segments.
87    pub row_group_size: Option<usize>,
88    /// Exact row count when rebuilding an existing SST; unknown for compaction outputs.
89    pub output_rows: Option<u64>,
90}

95/// Supplies extension handlers while the engine retains the complete lifecycle.
96pub trait IndexKindProvider: Send + Sync + Debug {
97    /// Related algorithms may share one handler to reuse row traversal.
98    fn handlers(&self) -> Vec<Arc<dyn IndexKindHandler>>;
99}
100
101/// Owns one or more related kinds that can share preparation and row traversal.
102pub trait IndexKindHandler: Send + Sync + Debug {
103    /// Nonempty, unique kinds exclusively owned by this handler. A group must be
104    /// entirely built-in or entirely extension kinds.
105    fn kinds(&self) -> &[IndexKind];
106
107    /// Prepares typed parameters. `None` means no work; an empty entry list may
108    /// still request removal of obsolete descriptors during maintenance.
109    fn plan(&self, context: &PlanContext<'_>) -> Result<Option<Arc<dyn PlannedIndex>>, BoxedError>;
110}
111
112/// An immutable typed plan shared by output files of an operation.
113#[async_trait]
114pub trait PlannedIndex: Send + Sync + Debug {
115    /// Nonempty, unique subset of the handler's kinds, including removal plans.
116    fn kinds(&self) -> &[IndexKind];
117
118    /// Requested column formats. Every entry kind must belong to this plan.
119    fn entries(&self) -> &[PlanEntry];
120
121    /// Prepares input reuse before any output batch is consumed.
122    async fn build(
123        &self,
124        context: &BuildContext<'_>,
125    ) -> Result<Option<Box<dyn IndexCreator>>, BoxedError>;
126
127    /// Whether building this plan can reuse persisted input summaries.
128    /// Row-only plans do not require checking input object existence.
129    fn needs_input_reuse(&self) -> bool {
130        false
131    }
132
133    /// Optional hard ceiling; built-in creators use their existing spill budgets.
134    fn memory_limit(&self) -> Option<usize> {
135        None
136    }
137}

139/// Accumulates rows and writes blobs into the engine-owned Puffin lifecycle.
140/// Implementations release retained state on drop, including cancellation.
141#[async_trait]
142pub trait IndexCreator: Send {
143    /// Whether rebuilding this index requires scanning SST rows after preparation.
144    /// A creator reusing persisted state must account for output rows without updates.
145    fn requires_row_scan(&self) -> bool {
146        true
147    }
148
149    /// Borrows a batch and its reusable primary-key decode cache.
150    async fn update(&mut self, batch: &mut Batch) -> Result<(), BoxedError>;
151
152    /// Borrows array data; cloning shared array handles is allowed.
153    async fn update_flat(&mut self, batch: &RecordBatch) -> Result<(), BoxedError>;
154
155    /// Writes completed entries. Publication waits for the engine to close Puffin.
156    /// The creator is dropped afterward, including on error, without calling its
157    /// `abort` method; implementations must finalize their own temporary resources.
158    async fn finish(
159        &mut self,
160        writer: &mut SstPuffinWriter,
161    ) -> Result<Vec<CompletedEntry>, BoxedError>;
162
163    /// Releases temporary resources; the engine owns final-file cleanup.
164    async fn abort(&mut self) -> Result<(), BoxedError>;
165
166    /// Estimated retained memory, not exact allocator or process RSS accounting.
167    fn memory_usage(&self) -> usize;
168
169    /// Targets needed when preparing the shared dense primary-key decoder.
170    fn indexed_targets(&self) -> Vec<IndexTarget> {
171        Vec::new()
172    }
173
174    /// Whether this creator participates in existing dense-key predecoding.
175    fn needs_dense_pk(&self) -> bool {
176        false
177    }
