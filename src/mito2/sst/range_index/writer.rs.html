54            index_row_group_size: DEFAULT_ROW_GROUP_SIZE,
55        }
56    }

59/// Metrics collected by an [`SstRangeIndexWriter`].
60#[derive(Debug, Clone, Default)]
61pub struct SstRangeIndexWriterMetrics {
62    /// Number of input record batches passed to the writer.
63    pub input_batches: usize,
64    /// Number of source SST rows passed to the writer.
65    pub input_rows: usize,
66    /// Number of non-empty source SST row groups passed to the writer.
67    pub num_source_row_groups: usize,
68    /// Number of series ranges written to the index.
69    pub num_ranges: usize,
70    /// Size of the committed index file. This remains zero for an aborted writer.
71    pub output_bytes: u64,
72    /// Time spent opening the object-store and Parquet writers.
73    pub open_elapsed: Duration,
74    /// Time spent validating input and aggregating ranges.
75    pub aggregate_elapsed: Duration,
76    /// Time spent encoding and writing Parquet batches.
77    pub write_elapsed: Duration,
78    /// Time spent closing a completed file.
79    pub finish_elapsed: Duration,
80    /// Time spent removing incomplete output.
81    pub cleanup_elapsed: Duration,
82    /// Whether this writer was explicitly aborted.
83    pub aborted: bool,
84}
85
86impl SstRangeIndexWriterMetrics {
87    /// Returns time spent by writer-owned work.
88    pub fn total_elapsed(&self) -> Duration {
89        self.open_elapsed
90            + self.aggregate_elapsed
91            + self.write_elapsed
92            + self.finish_elapsed
93            + self.cleanup_elapsed
94    }
95}

106/// Incrementally writes the series row ranges of one SST to a Parquet file.
107///
108/// Each non-empty input batch is tagged with its source SST row-group ID. A row
109/// group may be supplied in multiple batches, but IDs must start at zero and be
110/// contiguous. Input primary keys must preserve the SST's global sort order.
111pub struct SstRangeIndexWriter {
112    codec: SparsePrimaryKeyCodec,
113    schema: SchemaRef,
114    writer: ParquetIndexWriter,
115    current_row_group_id: Option<u32>,
116    current_row_group_offset: i64,
117    last_primary_key: Option<Vec<u8>>,
118    current_row: Option<RangeIndexRow>,
119    buffered_rows: Vec<RangeIndexRow>,
120    metrics: SstRangeIndexWriterMetrics,
121    failed: bool,
122}
123
124impl SstRangeIndexWriter {
125    /// Creates a writer for `path` in `object_store`.
126    ///
127    /// The file name in `path` must be unique in the object store so aborting
128    /// the writer only removes temporary files that belong to this writer.
129    pub async fn try_new(
130        metadata: RegionMetadataRef,
131        object_store: ObjectStore,
132        path: &str,
133        options: SstRangeIndexWriterOptions,
134    ) -> Result<Self> {
135        let open_start = Instant::now();
136        ensure!(
137            options.index_row_group_size > 0,
138            InvalidMetaSnafu {
139                reason: "range index row group size must be greater than zero",
140            }
141        );
142        validate_metadata(&metadata)?;
143        let schema = range_index_schema();
144        let writer = ParquetIndexWriter::try_new(
145            "range index",
146            object_store,
147            path,
148            &schema,
149            options.index_row_group_size,
150        )
151        .await?;
152        let codec = SparsePrimaryKeyCodec::new(&metadata);
153
154        Ok(Self {
155            codec,
156            schema,
157            writer,
158            current_row_group_id: None,
159            current_row_group_offset: 0,
160            last_primary_key: None,
161            current_row: None,
162            buffered_rows: Vec::with_capacity(WRITE_BATCH_SIZE),
163            metrics: SstRangeIndexWriterMetrics {
164                open_elapsed: open_start.elapsed(),
165                ..Default::default()
166            },
167            failed: false,
168        })
169    }
170
171    /// Returns the metrics collected so far.
172    pub fn metrics(&self) -> &SstRangeIndexWriterMetrics {
173        &self.metrics
174    }
175
176    /// Adds one batch from `row_group_id` of the source SST.
177    ///
178    /// The batch may be a primary-key-only projection or a full flat batch. Its
179    /// schema must contain a named `__primary_key` column. After this method
180    /// returns an error, callers must call [`Self::abort`].
181    pub async fn write(&mut self, row_group_id: u32, batch: &RecordBatch) -> Result<()> {
182        ensure!(
183            !self.failed,
184            InvalidRecordBatchSnafu {
185                reason: "cannot write to a failed range index writer",
186            }
187        );
188
189        self.metrics.input_batches += 1;
190        self.metrics.input_rows += batch.num_rows();
191        let aggregate_start = Instant::now();
192        let write_before = self.metrics.write_elapsed;
193        let result = self.write_inner(row_group_id, batch).await;
194        let write_cost = self.metrics.write_elapsed.saturating_sub(write_before);
195        self.metrics.aggregate_elapsed += aggregate_start.elapsed().saturating_sub(write_cost);
196        if result.is_err() {
197            self.failed = true;
198        }
199        result
200    }
201
202    /// Finishes and commits the index file.
203    pub async fn finish(mut self) -> Result<SstRangeIndexWriterMetrics> {
204        if self.failed {
205            let error = InvalidRecordBatchSnafu {
206                reason: "cannot finish a failed range index writer",
207            }
208            .build();
209            self.cleanup().await;
210            return Err(error);
211        }
212
213        let result = self.finish_inner().await;
214        if result.is_err() {
215            self.cleanup().await;
216        }
217        result.map(|_| self.metrics)
218    }
219
220    /// Aborts the writer and removes incomplete output files.
221    pub async fn abort(mut self) -> Result<SstRangeIndexWriterMetrics> {
222        self.metrics.aborted = true;
223        self.cleanup().await;
224        Ok(self.metrics)
225    }
226
227    async fn write_inner(&mut self, row_group_id: u32, batch: &RecordBatch) -> Result<()> {
228        if batch.num_rows() == 0 {
229            return Ok(());
230        }
231
232        self.enter_row_group(row_group_id).await?;
233        let pk_idx = batch
234            .schema()
235            .column_with_name(PRIMARY_KEY_COLUMN_NAME)
236            .map(|(idx, _)| idx)
237            .context(InvalidRecordBatchSnafu {
238                reason: "range index input does not contain __primary_key",
239            })?;
240        let primary_keys = batch.column(pk_idx);
241        let batch_rows = i64::try_from(batch.num_rows()).map_err(|_| {
242            InvalidRecordBatchSnafu {
243                reason: format!(
244                    "range index input batch exceeds Int64: {}",
245                    batch.num_rows()
246                ),
247            }
248            .build()
249        })?;
250        self.current_row_group_offset
251            .checked_add(batch_rows)
252            .context(InvalidRecordBatchSnafu {
253                reason: "source SST row-group offset exceeds Int64",
254            })?;
255
256        if let Some(array) = primary_keys.as_any().downcast_ref::<BinaryArray>() {
257            ensure!(
258                array.null_count() == 0,
259                InvalidRecordBatchSnafu {
260                    reason: "range index input contains null primary keys",
261                }
262            );
263            self.write_binary_primary_keys(array).await
264        } else if let Some(array) = primary_keys
265            .as_any()
266            .downcast_ref::<DictionaryArray<UInt32Type>>()
267        {
268            ensure!(
269                array.null_count() == 0,
270                InvalidRecordBatchSnafu {
271                    reason: "range index input contains null primary keys",
272                }
273            );
274            self.write_dictionary_primary_keys(array).await
275        } else {
276            InvalidRecordBatchSnafu {
277                reason: format!(
278                    "range index requires Binary or Dictionary(UInt32, Binary) primary keys, got {:?}",
279                    primary_keys.data_type()
280                ),
281            }
282            .fail()
283        }
284    }
285
286    async fn enter_row_group(&mut self, row_group_id: u32) -> Result<()> {
287        let Some(current) = self.current_row_group_id else {
288            ensure!(
289                row_group_id == 0,
290                InvalidRecordBatchSnafu {
291                    reason: format!(
292                        "range index row groups must start at zero, got {row_group_id}"
293                    ),
294                }
295            );
296            self.current_row_group_id = Some(row_group_id);
297            self.metrics.num_source_row_groups = 1;
298            return Ok(());
299        };
300
301        ensure!(
302            row_group_id == current || current.checked_add(1) == Some(row_group_id),
303            InvalidRecordBatchSnafu {
304                reason: format!(
305                    "range index row groups must be contiguous, current {current}, got {row_group_id}"
306                ),
307            }
308        );
309        if row_group_id != current {
310            self.finish_current_row().await?;
311            self.current_row_group_id = Some(row_group_id);
312            self.current_row_group_offset = 0;
313            self.metrics.num_source_row_groups += 1;
314        }
315        Ok(())
316    }
317
318    async fn write_binary_primary_keys(&mut self, primary_keys: &BinaryArray) -> Result<()> {
319        let mut start = 0;
320        while start < primary_keys.len() {
321            let primary_key = primary_keys.value(start);
322            let mut end = start + 1;
323            while end < primary_keys.len() && primary_keys.value(end) == primary_key {
324                end += 1;
325            }
326            self.update_primary_key(primary_key, end - start).await?;
327            start = end;
328        }
329        Ok(())
330    }
331
332    async fn write_dictionary_primary_keys(
333        &mut self,
334        primary_keys: &DictionaryArray<UInt32Type>,
335    ) -> Result<()> {
336        let values = primary_keys
337            .values()
338            .as_any()
339            .downcast_ref::<BinaryArray>()
340            .context(InvalidRecordBatchSnafu {
341                reason: "range index primary-key dictionary values are not binary",
342            })?;
343        ensure!(
344            values.null_count() == 0,
345            InvalidRecordBatchSnafu {
346                reason: "range index input contains null primary-key dictionary values",
347            }
348        );
349        let keys = primary_keys.keys().values();
350        let mut start = 0;
351        while start < keys.len() {
352            let key = keys[start];
353            let mut end = start + 1;
354            while end < keys.len() && keys[end] == key {
355                end += 1;
356            }
357            self.update_primary_key(values.value(key as usize), end - start)
358                .await?;
359            start = end;
360        }
361        Ok(())
362    }
363
364    async fn update_primary_key(&mut self, primary_key: &[u8], run_len: usize) -> Result<()> {
365        if let Some(last) = self.last_primary_key.as_deref() {
366            ensure!(
367                primary_key.cmp(last) != Ordering::Less,
368                InvalidRecordBatchSnafu {
369                    reason: "range index input is not sorted by primary key",
370                }
371            );
372        }
373
374        let (table_id, tsid) = self.codec.decode_ids(primary_key).context(DecodeSnafu)?;
375        let run_len = i64::try_from(run_len).map_err(|_| {
376            InvalidRecordBatchSnafu {
377                reason: format!("range index primary-key run is too large: {run_len}"),
378            }
379            .build()
380        })?;
381        let start = self.current_row_group_offset;
382        let end = start
383            .checked_add(run_len)
384            .context(InvalidRecordBatchSnafu {
385                reason: "source SST row-group offset exceeds Int64",
386            })?;
387        let row_group_id = self.current_row_group_id.context(UnexpectedSnafu {
388            reason: "range index writer has no active source row group",
389        })?;
390
391        match self.current_row.as_mut() {
392            Some(row)
393                if row.row_group_id == row_group_id
394                    && row.table_id == table_id
395                    && row.tsid == tsid =>
396            {
397                row.end = end;
398            }
399            _ => {
400                self.finish_current_row().await?;
401                self.current_row = Some(RangeIndexRow {
402                    row_group_id,
403                    table_id,
404                    tsid,
405                    start,
406                    end,
407                });
408            }
409        }
410        self.current_row_group_offset = end;
411        self.last_primary_key = Some(primary_key.to_vec());
412        Ok(())
413    }
414
415    async fn finish_current_row(&mut self) -> Result<()> {
416        if let Some(row) = self.current_row.take() {
417            self.buffered_rows.push(row);
418            self.metrics.num_ranges += 1;
419        }
420        if self.buffered_rows.len() >= WRITE_BATCH_SIZE {
421            self.flush_rows().await?;
422        }
423        Ok(())
424    }
425
426    async fn flush_rows(&mut self) -> Result<()> {
427        if self.buffered_rows.is_empty() {
428            return Ok(());
429        }
430        let batch = rows_to_batch(&self.schema, &self.buffered_rows)?;
431        let start = Instant::now();
432        let result = self.writer.write(&batch).await;
433        self.metrics.write_elapsed += start.elapsed();
434        result?;
435        self.buffered_rows.clear();
436        Ok(())
437    }
438
439    async fn finish_inner(&mut self) -> Result<()> {
440        let aggregate_start = Instant::now();
441        let write_before = self.metrics.write_elapsed;
442        self.finish_current_row().await?;
443        self.flush_rows().await?;
444        let write_cost = self.metrics.write_elapsed.saturating_sub(write_before);
445        self.metrics.aggregate_elapsed += aggregate_start.elapsed().saturating_sub(write_cost);
446
447        let finish_start = Instant::now();
448        self.metrics.output_bytes = self.writer.finish().await?;
449        self.metrics.finish_elapsed += finish_start.elapsed();
450        Ok(())
451    }
452
453    async fn cleanup(&mut self) {
454        let start = Instant::now();
455        self.writer.abort().await;
456        self.current_row = None;
457        self.buffered_rows.clear();
458
459        self.metrics.output_bytes = 0;
460        self.metrics.cleanup_elapsed += start.elapsed();
461    }
462}

464/// Returns the Arrow schema of a per-SST range index.
465pub fn range_index_schema() -> SchemaRef {
466    Arc::new(Schema::new(vec![
467        Field::new(ROW_GROUP_ID_COLUMN, DataType::UInt32, false),
468        Field::new(TABLE_ID_COLUMN, DataType::UInt32, false),
469        Field::new(TSID_COLUMN, DataType::UInt64, false),
470        Field::new(START_COLUMN, DataType::Int64, false),
471        Field::new(END_COLUMN, DataType::Int64, false),
472    ]))
473}
474
475fn validate_metadata(metadata: &RegionMetadataRef) -> Result<()> {
476    ensure!(
477        metadata.primary_key_encoding == PrimaryKeyEncoding::Sparse,
478        InvalidMetaSnafu {
479            reason: "range index only supports sparse primary-key encoding",
480        }
481    );
482    ensure!(
483        metadata
484            .primary_key
485            .starts_with(&[ReservedColumnId::table_id(), ReservedColumnId::tsid()]),
486        InvalidMetaSnafu {
487            reason: "range index requires (__table_id, __tsid) as the primary-key prefix",
488        }
489    );
490    let table_id = metadata
491        .column_by_id(ReservedColumnId::table_id())
492        .context(InvalidMetaSnafu {
493            reason: "range index metadata is missing __table_id",
494        })?;
495    let tsid = metadata
496        .column_by_id(ReservedColumnId::tsid())
497        .context(InvalidMetaSnafu {
498            reason: "range index metadata is missing __tsid",
499        })?;
500    ensure!(
501        table_id.column_schema.data_type == ConcreteDataType::uint32_datatype()
502            && tsid.column_schema.data_type == ConcreteDataType::uint64_datatype(),
503        InvalidMetaSnafu {
504            reason: "range index requires UInt32 __table_id and UInt64 __tsid",
505        }
506    );
507    Ok(())
508}

510fn rows_to_batch(schema: &SchemaRef, rows: &[RangeIndexRow]) -> Result<RecordBatch> {
511    let arrays: Vec<ArrayRef> = vec![
512        Arc::new(UInt32Array::from_iter_values(
513            rows.iter().map(|row| row.row_group_id),
514        )),
515        Arc::new(UInt32Array::from_iter_values(
516            rows.iter().map(|row| row.table_id),
517        )),
518        Arc::new(UInt64Array::from_iter_values(
519            rows.iter().map(|row| row.tsid),
520        )),
521        Arc::new(Int64Array::from_iter_values(
522            rows.iter().map(|row| row.start),
523        )),
524        Arc::new(Int64Array::from_iter_values(rows.iter().map(|row| row.end))),
525    ];
526    RecordBatch::try_new(schema.clone(), arrays).context(NewRecordBatchSnafu)
527}

576    async fn read_index(store: &ObjectStore, path: &str) -> (u64, usize, Vec<RangeIndexRow>) {
577        let bytes = store.read(path).await.unwrap().to_bytes();
578        let output_bytes = bytes.len() as u64;
579        let builder = ParquetRecordBatchReaderBuilder::try_new(bytes).unwrap();
580        let row_groups = builder.metadata().num_row_groups();
581        let batches = builder
582            .build()
583            .unwrap()
584            .collect::<std::result::Result<Vec<_>, _>>()
585            .unwrap();
586        let mut rows = Vec::new();
587        for batch in batches {
588            let row_group_ids = batch
589                .column(0)
590                .as_any()
591                .downcast_ref::<UInt32Array>()
592                .unwrap();
593            let table_ids = batch
594                .column(1)
595                .as_any()
596                .downcast_ref::<UInt32Array>()
597                .unwrap();
598            let tsids = batch
599                .column(2)
600                .as_any()
601                .downcast_ref::<UInt64Array>()
602                .unwrap();
603            let starts = batch
604                .column(3)
605                .as_any()
606                .downcast_ref::<Int64Array>()
607                .unwrap();
608            let ends = batch
609                .column(4)
610                .as_any()
611                .downcast_ref::<Int64Array>()
612                .unwrap();
613            for row in 0..batch.num_rows() {
614                rows.push(RangeIndexRow {
615                    row_group_id: row_group_ids.value(row),
616                    table_id: table_ids.value(row),
617                    tsid: tsids.value(row),
618                    start: starts.value(row),
619                    end: ends.value(row),
620                });
621            }
622        }
623        (output_bytes, row_groups, rows)
624    }
