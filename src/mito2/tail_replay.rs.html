44/// Memtable usage at which replayed rows are written to SSTs.
45const CHUNK_MEMTABLE_USAGE: usize = 64 * 1024 * 1024;
46/// Delay before the first retry of a failed round or an unfinished commit.
47const MIN_BACKOFF: Duration = Duration::from_secs(1);
48/// Longest delay between retries.
49const MAX_BACKOFF: Duration = Duration::from_secs(30);
50/// How often a replay checks whether the region received a Leader lease.
51const LEASE_CHECK_INTERVAL: Duration = Duration::from_millis(500);
52
53/// Returns whether the pending tails of `region` are to be replayed: it has some and
54/// [can replay them](MitoRegion::can_replay_tails).
55pub(crate) fn should_replay_tails(region: &MitoRegion) -> bool {
56    region.can_replay_tails() && region.version().has_pending_tails()
57}
58
59/// Replays the pending tails of a region into SSTs and publishes them in one manifest edit.
60///
61/// Each round reads the tails in order and writes each into its own SSTs: an SST can give all its
62/// rows its largest sequence, so in an SST shared with a later tail, an earlier tail's rows would
63/// take the later tail's sequence and outrank rows that a leader between the two tails flushed. A
64/// tail's 64 MiB chunks only bound memory: writes accepted during the replay already take sequences
65/// above every tail's. The edit adds the SSTs, clears the tails and advances the flushed entry id
66/// and sequence. Another node's listed tail is only a bound, so until the region receives a Leader
67/// lease after a catchup made it writable, the edit keeps its unread part, which a second edit
68/// clears once a lease arrives. A failed round is retried after a back-off while
69/// [should_replay_tails] holds. The task always reports its end to the worker.
70pub(crate) struct TailReplayTask<S> {
71    pub(crate) region: MitoRegionRef,
72    /// Id of this node, whose tails are read from `wal`.
73    pub(crate) node_id: Option<u64>,
74    pub(crate) wal: Wal<S>,
75    /// Writes the chunks. It is never scheduled as a flush.
76    pub(crate) flush_task: RegionFlushTask,
77    pub(crate) state: CancellableTaskState,
78}
79
80impl<S: LogStore> TailReplayTask<S> {
81    pub(crate) async fn run(self) {
82        let region_id = self.region.region_id;
83        let _ended = TailReplayEndGuard {
84            sender: self.flush_task.request_sender.clone(),
85            region_id,
86        };
87        let mut backoff = MIN_BACKOFF;
88        while !self.state.is_cancelled() && should_replay_tails(&self.region) {
89            match self.replay_round().await {
90                Ok(()) => return,
91                Err(_) if self.state.is_cancelled() => break,
92                Err(e) => {
93                    error!(e; "Failed to replay the pending WAL tails of region {region_id}");
94                    self.flush_task
95                        .listener
96                        .on_tail_replay_failed(region_id, &e);
97                }
98            }
99            if !self.sleep(backoff).await {
100                break;
101            }
102            backoff = (backoff * 2).min(MAX_BACKOFF);
103        }
104        info!(
105            "Region {region_id} stopped replaying its pending WAL tails before they were published"
106        );
107    }

109    /// Replays the current tails and publishes the result. An error asks for another round.
110    async fn replay_round(&self) -> Result<()> {
111        // Keeps no version past this point: it would hold the region's memtables, and their
112        // write buffer memory, after a flush released them.
113        let (tails, flushed_entry_id) = {
114            let version = self.region.version();
115            (version.pending_tails.clone(), version.flushed_entry_id)
116        };
117        let uncommitted = UncommittedSsts::new(
118            self.region.region_id,
119            self.region.access_layer.clone(),
120            Some(self.flush_task.cache_manager.clone()),
121        );
122        let (files, last_entry_ids) = match self.replay_tails(&tails, &uncommitted).await {
123            Ok(replayed) => replayed,
124            Err(e) => {
125                uncommitted.cleanup().await;
126                return Err(e);
127            }
128        };
129        let lease_received = self.region.manifest_ctx.leader_lease_received();
130        let remaining = tails
131            .iter()
132            .zip(&last_entry_ids)
133            .filter_map(|(tail, last_entry_id)| {
134                self.remaining_tail(tail, *last_entry_id, lease_received)
135            })
136            .collect::<Vec<_>>();
137        if !files.is_empty() || remaining != tails {
138            let last_entry_id = last_entry_ids.into_iter().flatten().max().unwrap_or(0);
139            let flushed_sequence = files
140                .iter()
141                .filter_map(|file| file.sequence)
142                .map(NonZeroU64::get)
143                .max();
144            let edit = RegionEdit {
145                flushed_sequence,
146                files_to_add: files,
147                files_to_remove: Vec::new(),
148                timestamp_ms: Some(chrono::Utc::now().timestamp_millis()),
149                compaction_time_window: None,
150                flushed_entry_id: (last_entry_id > flushed_entry_id).then_some(last_entry_id),
151                // Later writes of the region take sequences above the replayed rows.
152                committed_sequence: flushed_sequence,
153                wal_recovery: WalRecoveryEdit {
154                    pending_tails: Some(remaining.clone()),
155                    handoff_floor: None,
156                },
157            };
158            if !self.publish(&tails, edit, &uncommitted).await? {
159                return Ok(());
160            }
161        }
162        if remaining.is_empty() || !self.wait_for_leader_lease().await {
163            return Ok(());
164        }
165        let edit = RegionEdit {
166            timestamp_ms: Some(chrono::Utc::now().timestamp_millis()),
167            wal_recovery: WalRecoveryEdit {
168                pending_tails: Some(Vec::new()),
169                handoff_floor: None,
170            },
171            ..Default::default()
172        };
173        self.publish(&remaining, edit, &uncommitted).await?;
174        Ok(())
175    }

177    /// Returns what stays recorded of `tail` after a replay read it up to `last_entry_id`.
178    /// Only another node's listed tail stays, from its last entry read, and only until the
179    /// region receives a Leader lease.
180    fn remaining_tail(
181        &self,
182        tail: &PendingTail,
183        last_entry_id: Option<EntryId>,
184        lease_received: bool,
185    ) -> Option<PendingTail> {
186        if lease_received
187            || tail.upto_kind == UptoKind::Reported
188            || Some(tail.node_id) == self.node_id
189        {
190            return None;
191        }
192        let after = last_entry_id.unwrap_or(tail.after);
193        (after < tail.upto).then(|| PendingTail {
194            after,
195            ..tail.clone()
196        })
197    }

199    /// Replays `tails` into chunks of SSTs. Returns the SSTs and the last entry id read of
200    /// each tail.
201    async fn replay_tails(
202        &self,
203        tails: &[PendingTail],
204        uncommitted: &UncommittedSsts,
205    ) -> Result<(Vec<FileMeta>, Vec<Option<EntryId>>)> {
206        let region_id = self.region.region_id;
207        let provider = &self.region.provider;
208        let mut files = Vec::new();
209        let mut last_entry_ids = Vec::with_capacity(tails.len());
210        for tail in tails {
211            let mut entries = self.unless_cancelled(self.read_tail_entries(tail)).await?;
212            let mut chunk = self.new_chunk_version();
213            let mut last_entry_id = None;
214            while let Some((entry_id, entry)) = self.unless_cancelled(entries.try_next()).await? {
215                if entry_id > tail.upto {
216                    break;
217                }
218                replay_entry(provider, region_id, &chunk, entry, entry_id).await?;
219                last_entry_id = Some(entry_id);
220                if chunk.current().version.memtables.mutable_usage() >= CHUNK_MEMTABLE_USAGE {
221                    files.extend(self.write_chunk_ssts(&chunk, uncommitted).await?);
222                    chunk = self.new_chunk_version();
223                }
224                if entry_id == tail.upto {
225                    break;
226                }
227            }
228            ensure!(
229                last_entry_id == Some(tail.upto) || tail.upto_kind == UptoKind::Listed,
230                TailEntryMissingSnafu {
231                    region_id,
232                    node_id: tail.node_id,
233                    entry_id: tail.upto,
234                }
235            );
236            files.extend(self.write_chunk_ssts(&chunk, uncommitted).await?);
237            last_entry_ids.push(last_entry_id);
238        }
239        Ok((files, last_entry_ids))
240    }

242    /// Streams the region's entries of `tail`, starting after `tail.after`. This node's reported
243    /// tail comes from its own WAL, whose stream may go past `tail.upto`. Every other tail is read
244    /// up to its bound through [Wal::read_node], since a listed bound may name no entry.
245    async fn read_tail_entries(&self, tail: &PendingTail) -> Result<WalEntryStream<'static>> {
246        let (provider, region_id) = (&self.region.provider, self.region.region_id);
247        let upper = match tail.upto_kind {
248            UptoKind::Reported if Some(tail.node_id) == self.node_id => {
249                let mut reader = self.wal.wal_entry_reader(provider, region_id);
250                return reader.read(provider, tail.after + 1);
251            }
252            UptoKind::Reported => WalReadUpperBound::LastEntry(tail.upto),
253            UptoKind::Listed => WalReadUpperBound::Listed(tail.upto),
254        };
255        self.wal
256            .read_node(region_id, tail.node_id, tail.after, upper)
257            .await
258    }

260    /// Waits until the region receives a Leader lease. Returns false if the region stops being
261    /// a writable, altering or editing leader, or the task is cancelled, first.
262    async fn wait_for_leader_lease(&self) -> bool {
263        let region_id = self.region.region_id;
264        info!("Region {region_id} waits for a Leader lease to clear its unread listed tails");
265        while self.region.can_replay_tails() {
266            self.flush_task.listener.on_tail_lease_check(region_id);
267            if self.region.manifest_ctx.leader_lease_received() {
268                return true;
269            }
270            if !self.sleep(LEASE_CHECK_INTERVAL).await {
271                info!("Region {region_id} cancelled its tail replay while waiting for a lease");
272                return false;
273            }
274        }
275        info!("Region {region_id} can no longer replay, keeping its unread listed tails");
276        false
277    }
278
279    /// Returns a scratch version with an empty memtable for one chunk of replayed rows.
280    fn new_chunk_version(&self) -> VersionControlRef {
281        let version = self.region.version();
282        let mutable = TimePartitions::new(
283            version.metadata.clone(),
284            version.memtables.mutable.memtable_builder().clone(),
285            0,
286            Some(version.memtables.mutable.part_duration()),
287        );
288        Arc::new(VersionControl::new(
289            VersionBuilder::new(version.metadata.clone(), mutable)
290                .options(version.options.clone())
291                .build(),
292        ))
293    }
294
295    /// Writes the rows of `chunk` to SSTs.
296    async fn write_chunk_ssts(
297        &self,
298        chunk: &VersionControl,
299        uncommitted: &UncommittedSsts,
300    ) -> Result<Vec<FileMeta>> {
301        chunk.freeze_mutable()?;
302        let version = chunk.current().version;
303        if version.memtables.immutables().is_empty() {
304            return Ok(Vec::new());
305        }
306        self.flush_task
307            .write_ssts(&version, &self.state, uncommitted)
308            .await
309    }

311    /// Commits `edit` for the replayed `tails`. It commits the same content again while the
312    /// result is unknown or the region is busy. Returns whether it committed; if not, the task
313    /// ends. An error asks for another round.
314    async fn publish(
315        &self,
316        tails: &[PendingTail],
317        edit: RegionEdit,
318        uncommitted: &UncommittedSsts,
319    ) -> Result<bool> {
320        let region_id = self.region.region_id;
321        let action_list = RegionMetaActionList::with_action(RegionMetaAction::Edit(edit.clone()));
322        let mut backoff = MIN_BACKOFF;
323        // Whether an attempt may have persisted the edit, which only the same content confirms.
324        let mut uncertain = false;
325        loop {
326            self.flush_task
327                .listener
328                .on_tail_commit_begin(region_id)
329                .await;
330            if self.state.is_cancelled() {
331                info!("Region {region_id} cancelled its tail replay before publishing");
332                uncommitted.cleanup().await;
333                return Ok(false);
334            }
335            match self
336                .region
337                .manifest_ctx
338                .update_manifest_for_tails(tails, action_list.clone())
339                .await
340            {
341                Ok(TailPublication::Committed(manifest_version)) => {
342                    uncommitted.disarm_cleanup();
343                    info!(
344                        "Published the replayed WAL tails of region {region_id} in manifest version {manifest_version}"
345                    );
346                    self.release_wal().await;
347                    let region = self.region.clone();
348                    let tails = tails.to_vec();
349                    let notify = BackgroundNotify::TailReplayCommitted {
350                        region,
351                        edit,
352                        tails,
353                    };
354                    notify_worker(&self.flush_task.request_sender, region_id, notify).await;
355                    return Ok(true);
356                }
357                Ok(TailPublication::Busy) => {}
358                Ok(TailPublication::Stale) => {
359                    info!("Region {region_id} changed its role or tails, dropping the replay");
360                    uncommitted.cleanup().await;
361                    return Ok(false);
362                }
363                // The commit fenced the region, which the worker reloads once this task ends.
364                Err(e @ Error::ManifestVersionConflict { .. }) => {
365                    uncommitted.disarm_cleanup();
366                    warn!(e; "Publishing the replayed WAL tails of region {region_id} conflicted");
367                    return Ok(false);
368                }
369                Err(e) if e.may_have_persisted_manifest_update() => {
370                    uncommitted.disarm_cleanup();
371                    uncertain = true;
372                    error!(e; "Publishing the replayed WAL tails of region {region_id} may have failed, retrying");
373                }
374                Err(Error::RegionStopped { .. }) => {
375                    info!("Region {region_id} stopped, dropping the replay");
376                    uncommitted.cleanup().await;
377                    return Ok(false);
378                }
379                Err(e) if uncertain => {
380                    error!(e; "Failed to publish the replayed WAL tails of region {region_id} again, retrying");
381                }
382                Err(e) => {
383                    uncommitted.cleanup().await;
384                    return Err(e);
385                }
386            }
387            if !self.sleep(backoff).await {
388                info!(
389                    "Region {region_id} cancelled its tail replay while retrying the publication"
390                );
391                uncommitted.cleanup().await;
392                return Ok(false);
393            }
394            backoff = (backoff * 2).min(MAX_BACKOFF);
395        }
396    }

398    /// Releases the WAL up to where the region now replays it on open, keeping this node's
399    /// remaining tails. A failure is only logged.
400    async fn release_wal(&self) {
401        let region = &self.region;
402        let manifest = region.manifest_ctx.manifest().await;
403        let entry_id = cap_wal_release(
404            &manifest.pending_tails,
405            self.node_id,
406            manifest.wal_replay_start(self.node_id),
407        );
408        if let Err(e) = self
409            .wal
410            .obsolete(region.region_id, entry_id, &region.provider)
411            .await
412        {
413            error!(e; "Failed to release the WAL of region {} after publishing its tails", region.region_id);
414        }
415    }
416
417    /// Awaits `future` unless the task is cancelled first.
418    async fn unless_cancelled<T>(&self, future: impl Future<Output = Result<T>>) -> Result<T> {
419        CancellableFuture::new(future, self.state.cancel_handle())
420            .await
421            .unwrap_or_else(|_| FlushCancelledSnafu.fail())
422    }
423
424    /// Sleeps for `delay`. Returns false if the task is cancelled first.
425    async fn sleep(&self, delay: Duration) -> bool {
426        CancellableFuture::new(tokio::time::sleep(delay), self.state.cancel_handle())
427            .await
428            .is_ok()
429    }
430}

432/// Sends `notify` about the region `region_id` to its worker.
433pub(crate) async fn notify_worker(
434    sender: &Sender<WorkerRequestWithTime>,
435    region_id: RegionId,
436    notify: BackgroundNotify,
437) {
438    let request = WorkerRequest::Background { region_id, notify };
439    if sender
440        .send(WorkerRequestWithTime::new(request))
441        .await
442        .is_err()
443    {
444        warn!("The worker of region {region_id} has stopped");
445    }
446}

448/// Reports the end of a tail replay to the worker, however the task ends.
449struct TailReplayEndGuard {
450    sender: Sender<WorkerRequestWithTime>,
451    region_id: RegionId,
452}
453
454impl Drop for TailReplayEndGuard {
455    fn drop(&mut self) {
456        let sender = self.sender.clone();
457        let region_id = self.region_id;
458        common_runtime::spawn_global(async move {
459            notify_worker(&sender, region_id, BackgroundNotify::TailReplayEnded).await;
460        });
461    }
