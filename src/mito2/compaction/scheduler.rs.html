60/// Outcome of cancelling one live compaction task by its exact task identity.
61#[derive(Debug, Clone, Copy, PartialEq, Eq)]
62pub(crate) enum CancelCompactionResult {
63    /// Cancellation was requested; completion is asynchronous.
64    CancelIssued,
65    /// The task is already cancelling.
66    AlreadyCancelling,
67    /// The task has crossed its non-cancellable commit boundary.
68    TooLateToCancel,
69    /// The task is no longer present on this region.
70    NotFound,
71    /// This engine or execution mode does not support cancellation.
72    Unsupported,
73}
74
75impl From<RequestCancelResult> for CancelCompactionResult {
76    fn from(result: RequestCancelResult) -> Self {
77        match result {
78            RequestCancelResult::CancelIssued => Self::CancelIssued,
79            RequestCancelResult::AlreadyCancelling => Self::AlreadyCancelling,
80            RequestCancelResult::TooLateToCancel => Self::TooLateToCancel,
81        }
82    }
83}

85/// Internal testing switch to prevent SST compaction from changing file layouts
86/// during query regression tests. This is not a public configuration option.
87const DISABLE_COMPACTION_ENV: &str = "GREPTIME_DISABLE_COMPACTION";
88
89/// Internal query-regression switch, read once when the first scheduler starts.
90/// Applies to SST compaction only; ingestion and flushing remain enabled.
91static DISABLE_COMPACTION: LazyLock<bool> = LazyLock::new(|| {
92    let value = std::env::var(DISABLE_COMPACTION_ENV).ok();
93    let disabled = parse_disable_compaction(value.as_deref());
94    if disabled {
95        warn!(
96            "SST compaction is disabled by {DISABLE_COMPACTION_ENV}; restart without it to resume"
97        );
98    }
99    disabled
100});
101
102fn parse_disable_compaction(value: Option<&str>) -> bool {
103    match value {
104        None | Some("false") => false,
105        Some("true") => true,
106        Some(_) => {
107            warn!(
108                "Invalid {DISABLE_COMPACTION_ENV}: expected true or false; compaction remains enabled"
109            );
110            false
111        }
112    }
113}

138/// Describes the immediate action produced by a compaction terminal transition.
139#[derive(Debug)]
140pub(crate) enum CompactionTransition {
141    /// No follow-up work was dispatched and no DDL became ready.
142    NoAction,
143    /// An automatic follow-up planning cycle was dispatched.
144    AutomaticFollowupScheduled,
145    /// DDL requests became ready after the compaction lifecycle terminated.
146    DdlReady(Vec<SenderDdlRequest>),
147}
148
149impl CompactionTransition {
150    fn from_pending_ddls(pending_ddls: Vec<SenderDdlRequest>) -> Self {
151        if pending_ddls.is_empty() {
152            Self::NoAction
153        } else {
154            Self::DdlReady(pending_ddls)
155        }
156    }
157
158    #[cfg(test)]
159    fn is_empty(&self) -> bool {
160        !matches!(self, Self::DdlReady(pending_ddls) if !pending_ddls.is_empty())
161    }
162
163    #[cfg(test)]
164    fn len(&self) -> usize {
165        match self {
166            Self::DdlReady(pending_ddls) => pending_ddls.len(),
167            Self::NoAction | Self::AutomaticFollowupScheduled => 0,
168        }
169    }
170}

211    /// Registers a loaded region; replacing its VersionControl invalidates the old incarnation.
212    pub(crate) fn register_region(
213        &mut self,
214        version_control: &VersionControlRef,
215        access_layer: &AccessLayerRef,
216        shared_state: &CompactionStateRef,
217    ) {
218        let region_id = version_control.region_id();
219        if let Some(status) = self.region_status.get_mut(&region_id) {
220            if Arc::ptr_eq(&status.version_control, version_control) {
221                return;
222            }
223            status.on_failure(Arc::new(RegionClosedSnafu { region_id }.build()));
224        }
225        let mut status =
226            CompactionStatus::new(region_id, version_control.clone(), access_layer.clone());
227        status.shared_state = shared_state.clone();
228        self.region_status.insert(region_id, status);
229    }

251    /// Accepts an automatic compaction trigger for a region.
252    ///
253    /// # Effects
254    ///
255    /// Starts planning when the registered region is idle.
256    /// Otherwise, an active status coalesces the trigger into one unrestricted
257    /// follow-up cycle. Returns `true` only when this call dispatches planning.
258    ///
259    /// # Constraints
260    ///
261    /// The owning region worker must call this method serially. Automatic
262    /// triggers have no waiter and may be ignored while a DDL fence is active.
263    /// The region identity comes from `version_control`; `access_layer` and
264    /// `manifest_ctx` must belong to the same region.
265    pub(crate) fn schedule_automatic_compaction(
266        &mut self,
267        compact_options: compact_request::Options,
268        version_control: &VersionControlRef,
269        access_layer: &AccessLayerRef,
270        manifest_ctx: &ManifestContextRef,
271        schema_metadata_manager: SchemaMetadataManagerRef,
272    ) -> Result<bool> {
273        self.schedule_compaction(
274            CompactionTrigger::Automatic,
275            compact_options,
276            version_control,
277            access_layer,
278            OptionOutputTx::none(),
279            manifest_ctx,
280            schema_metadata_manager,
281            self.engine_config.max_background_compactions,
282            None,
283        )
284    }

286    /// Accepts a manual compaction request for a region.
287    ///
288    /// # Effects
289    ///
290    /// Starts planning when the registered region is idle. During
291    /// an automatic compaction, queues the request and replaces any older

295    ///
296    /// # Constraints
297    ///
298    /// The owning region worker must call this method serially. The supplied

302    /// the same region.

311        schema_metadata_manager: SchemaMetadataManagerRef,

324            time_range,

328    /// Applies a background planning result to the current region state.
329    ///
330    /// # Effects
331    ///
332    /// Rejects stale plans, submits a prepared local or remote execution, or
333    /// completes the cycle when planning produced no executable plan. The
334    /// returned transition reports a dispatched automatic follow-up or DDL
335    /// requests released by the resulting terminal transition.
336    ///
337    /// # Constraints
338    ///
339    /// The owning worker must call this method for planning notifications and
340    /// must execute every returned DDL request. The notification may be stale;
341    /// stale notifications intentionally have no effect.
342    pub(crate) async fn handle_compaction_pick_finished(
343        &mut self,
344        finished: CompactionPickFinished,
345        manifest_ctx: &ManifestContextRef,
346        schema_metadata_manager: SchemaMetadataManagerRef,
347    ) -> CompactionTransition {
348        self.handle_compaction_pick_finished_inner(finished, manifest_ctx, schema_metadata_manager)
349            .await
350    }

352    /// Returns whether a terminal notification belongs to the installed execution.
353    ///
354    /// # Effects
355    ///
356    /// Performs a read-only identity check against the region's current plan.
357    ///
358    /// # Constraints
359    ///
360    /// Callers must check this before applying a compaction edit. Matching only
361    /// the region id is insufficient because close/reopen can replace a status.
362    pub(crate) fn is_current_execution(
363        &self,
364        region_id: RegionId,
365        execution: &CompactionExecution,
366    ) -> bool {
367        self.region_status
368            .get(&region_id)
369            .is_some_and(|status| status.matches_execution(execution))
370    }

372    /// Completes the installed execution after its edit has been applied.
373    ///
374    /// # Effects
375    ///
376    /// Notifies waiters, schedules a pending manual or automatic follow-up, or
377    /// returns the region to idle. A follow-up is scheduled whenever the cycle
378    /// latched an automatic trigger, or when `made_progress` is true (the
379    /// execution produced output or reduced the file count): successful compaction
380    /// keeps draining the region until the picker returns no plan. The returned
381    /// transition reports a dispatched automatic follow-up or DDLs that are now
382    /// safe to execute.
383    ///
384    /// # Constraints
385    ///
386    /// The owning worker must call this only after accepting and applying the
387    /// execution's result, and must execute every returned DDL request. Stale
388    /// executions are ignored.
389    pub(crate) async fn on_execution_finished(
390        &mut self,
391        region_id: RegionId,
392        execution: &CompactionExecution,
393        manifest_ctx: &ManifestContextRef,
394        schema_metadata_manager: SchemaMetadataManagerRef,
395        made_progress: bool,
396    ) -> CompactionTransition {
397        if !self.is_current_execution(region_id, execution) {
398            return CompactionTransition::NoAction;
399        }
400        self.on_compaction_finished(
401            region_id,
402            manifest_ctx,
403            schema_metadata_manager,
404            made_progress,
405        )
406        .await
407    }

409    /// Records failure of the installed execution.
410    ///
411    /// # Effects
412    ///
413    /// Resets the matching region to idle and fails all compaction waiters and
414    /// dependent DDL requests with the supplied error.
415    ///
416    /// # Constraints
417    ///
418    /// A stale execution failure is ignored so it cannot tear down replacement

420    pub(crate) fn on_execution_failed(
421        &mut self,
422        region_id: RegionId,
423        execution: &CompactionExecution,
424        err: Arc<Error>,

426        if !self.is_current_execution(region_id, execution) {
427            return;
428        }
429        self.on_compaction_failed(region_id, err);

477    /// Cancels a running compaction and queues its dependent DDL atomically.
478    ///
479    /// # Effects
480    ///
481    /// Requests cancellation when the active phase can still stop, then queues
482    /// the DDL. The queued DDL forms a scheduling fence for subsequent
483    /// compaction triggers. If no compaction is running, returns the sender and
484    /// typed request unchanged.
485    ///
486    /// # Constraints
487    ///
488    /// Production callers use this only for [`DdlRequest::Truncate`] and
489    /// [`DdlRequest::EnterStaging`]. `Ok(())` means the caller must wait for a
490    /// terminal callback to receive and execute the queued DDL; `Err` means the
491    /// caller must execute the returned request directly.
492    pub(crate) fn try_cancel_and_add_ddl<T>(
493        &mut self,
494        region_id: RegionId,
495        sender: OptionOutputTx,
496        request: T,
497        into_ddl_request: impl FnOnce(T) -> DdlRequest,
498    ) -> std::result::Result<(), (OptionOutputTx, T)> {
499        let Some(status) = self.region_status.get_mut(&region_id) else {
500            return Err((sender, request));
501        };
502        if !status.is_compacting() {
503            return Err((sender, request));
504        }
505        status.request_cancel();
506
507        let request = SenderDdlRequest {
508            region_id,
509            sender,
510            request: into_ddl_request(request),
511        };
512        debug!(
513            "Added pending DDL request for region: {}, ddl: {:?}",
514            request.region_id, request.request
515        );
516        status.pending_ddl_requests.push(request);
517        Ok(())
518    }

520    /// Pending and handed-off DDLs retain their turn until the final reply.
521    pub(crate) fn blocks_region_edits(&self, region_id: RegionId) -> bool {
522        self.region_status
523            .get(&region_id)
524            .is_some_and(|status| status.has_ddl())

527    /// Whether a staging request is still waiting for compaction cancellation.
528    pub(crate) fn has_pending_enter_staging(&self, region_id: RegionId) -> bool {

534        })
535    }

538// Internal state-machine helpers.
539impl CompactionScheduler {
540    /// Returns the current plan id and advances the counter.
541    ///
542    /// Takes the counter instead of `&mut self` so callers can bump it while
543    /// holding a mutable borrow of a region status.
544    fn next_plan_id(counter: &mut u64) -> u64 {
545        let plan_id = *counter;
546        *counter = counter.wrapping_add(1);
547        plan_id
548    }

550    #[allow(clippy::too_many_arguments)]
551    fn schedule_compaction(
552        &mut self,
553        trigger: CompactionTrigger,
554        compact_options: compact_request::Options,
555        version_control: &VersionControlRef,
556        access_layer: &AccessLayerRef,
557        waiter: OptionOutputTx,
558        manifest_ctx: &ManifestContextRef,
559        schema_metadata_manager: SchemaMetadataManagerRef,
560        max_parallelism: usize,
561        time_range: Option<TimestampRange>,
562    ) -> Result<bool> {
563        if self.disable_compaction {
564            if matches!(trigger, CompactionTrigger::Manual) {
565                waiter.send(
566                    UnsupportedOperationSnafu {
567                        err_msg: format!("SST compaction is disabled by {DISABLE_COMPACTION_ENV}"),
568                    }
569                    .fail(),
570                );
571            }
572            return Ok(false);
573        }
574        let region_id = version_control.region_id();
575        let status = self.region_status.entry(region_id).or_insert_with(|| {
576            CompactionStatus::new(region_id, version_control.clone(), access_layer.clone())
577        });
578        if status.has_ddl() {
579            info!(
580                "Region {} has pending DDL requests, ignoring compaction: queued={}, executing={}",
581                region_id,
582                status.pending_ddl_requests.len(),
583                status.executing_ddl.is_some(),
584            );
585            waiter.send(CompactionCancelledSnafu.fail());
586            return Ok(false);
587        }
588        let current_state = manifest_ctx.current_state();
589        if current_state.in_staging_session() {
590            info!(
591                "Skipping compaction for region {} in staging mode, options: {:?}",
592                region_id, compact_options
593            );
594            waiter.send(Ok(0));
595            return Ok(false);
596        }
597        if status.has_pending_tails() {
598            info!(
599                "Skipping compaction for region {} with pending WAL tails, options: {:?}",
600                region_id, compact_options
601            );
602            waiter.send(Ok(0));
603            return Ok(false);
604        }
605
606        if status.is_compacting() {
607            match trigger {
608                CompactionTrigger::Automatic => status.mark_automatic_trigger(),
609                CompactionTrigger::Manual if status.is_manual_compaction() => {
610                    waiter.send(ManualCompactionAlreadyRunningSnafu { region_id }.fail());
611                    info!(
612                        "Region {} already has a manually triggered compaction running",
613                        region_id
614                    );
615                }
616                CompactionTrigger::Manual => {
617                    status.set_pending_request(PendingCompaction {
618                        options: compact_options,
619                        waiter,
620                        max_parallelism,
621                        time_range,
622                    });
623                    info!(
624                        "Region {} is running an automatic compaction; manual compaction will be re-scheduled",
625                        region_id
626                    );
627                }
628            }
629            return Ok(false);
630        }
631
632        // Publish the picking phase before dispatching background planning.
633        let plan_id = Self::next_plan_id(&mut self.next_plan_id);
634        status.start_picking_with_trigger(plan_id, trigger);
635        let pick_history = (trigger == CompactionTrigger::Automatic
636            && matches!(compact_options, compact_request::Options::Regular(_)))
637        .then(|| status.pick_history.clone());
638        let request = status.new_compaction_request(
639            self.request_sender.clone(),
640            self.engine_config.clone(),
641            self.cache_manager.clone(),
642            manifest_ctx,
643            self.listener.clone(),
644            schema_metadata_manager,
645            max_parallelism,
646            pick_history,
647        );
648        status.merge_waiter(waiter);
649        self.dispatch_compaction_planning(plan_id, request, compact_options, time_range, trigger);
650        self.listener.on_compaction_scheduled(region_id);
651        Ok(true)
652    }
653
654    // Handle pending manual compaction request for the region.
655    //
656    // Returns true if should early return, false otherwise.
657    fn handle_pending_compaction_request(
658        &mut self,
659        region_id: RegionId,
660        manifest_ctx: &ManifestContextRef,
661        schema_metadata_manager: SchemaMetadataManagerRef,
662    ) -> bool {
663        let Some(status) = self.region_status.get_mut(&region_id) else {
664            return true;
665        };
666
667        // If there is a pending manual compaction request, schedule it.
668        // and defer returning the pending DDL requests to the caller.
669        let Some(pending_request) = std::mem::take(&mut status.pending_request) else {
670            return false;
671        };
672        if status.has_pending_tails() {
673            status.pending_task_state = None;
674            status.publish_state();
675            pending_request.waiter.send(Ok(0));
676            return false;
677        }
678
679        let PendingCompaction {
680            options,
681            waiter,
682            max_parallelism,
683            time_range,
684        } = pending_request;
685
686        let request = status.new_compaction_request(
687            self.request_sender.clone(),
688            self.engine_config.clone(),
689            self.cache_manager.clone(),
690            manifest_ctx,
691            self.listener.clone(),
692            schema_metadata_manager,
693            max_parallelism,
694            None,
695        );
696        status.merge_waiter(waiter);
697        // Bump the counter through a disjoint field borrow so the `status`
698        // borrow stays alive; nothing could have removed the status since it
699        // was fetched above.
700        let plan_id = Self::next_plan_id(&mut self.next_plan_id);
701        status.start_picking_with_trigger(plan_id, CompactionTrigger::Manual);
702        status.active.task_state = status.pending_task_state.take();
703        self.dispatch_compaction_planning(
704            plan_id,
705            request,
706            options,
707            time_range,
708            CompactionTrigger::Manual,
709        );
710        debug!(
711            "Successfully scheduled manual compaction planning for region id: {}",
712            region_id
713        );
714        true
715    }
716
717    /// Notifies the scheduler that the compaction job is finished successfully.
718    async fn on_compaction_finished(
719        &mut self,
720        region_id: RegionId,
721        manifest_ctx: &ManifestContextRef,
722        schema_metadata_manager: SchemaMetadataManagerRef,
723        made_progress: bool,
724    ) -> CompactionTransition {
725        if let Some(ddls) = self.take_ready_ddls(region_id) {
726            return CompactionTransition::DdlReady(ddls);
727        }
728
729        if self.handle_pending_compaction_request(
730            region_id,
731            manifest_ctx,
732            schema_metadata_manager.clone(),
733        ) {
734            return CompactionTransition::NoAction;
735        }
736
737        let Some(status) = self.region_status.get_mut(&region_id) else {
738            return CompactionTransition::NoAction;
739        };
740        for waiter in status.take_waiters() {
741            waiter.send(Ok(0));
742        }
743
744        // Keep draining when the cycle latched an automatic trigger, or when the
745        // execution produced output or reduced the file count. An empty edit stops
746        // here; the next flush trigger resumes.
747        let should_continue = status.active.reset_automatic_followup() || made_progress;
748        if should_continue
749            && self.schedule_automatic_followup(region_id, manifest_ctx, schema_metadata_manager)
750        {
751            return CompactionTransition::AutomaticFollowupScheduled;
752        }
753        if let Some(status) = self.region_status.get_mut(&region_id) {
754            status.become_idle();
755        }
756        CompactionTransition::NoAction
757    }
758
759    fn schedule_automatic_followup(
760        &mut self,
761        region_id: RegionId,
762        manifest_ctx: &ManifestContextRef,
763        schema_metadata_manager: SchemaMetadataManagerRef,
764    ) -> bool {
765        if self.disable_compaction {
766            return false;
767        }
768        let Some(status) = self.region_status.get_mut(&region_id) else {
769            return false;
770        };
771        if status.has_pending_tails() {
772            return false;
773        }
774        // An external automatic trigger requires one unrestricted follow-up pick.
775        let request = status.new_compaction_request(
776            self.request_sender.clone(),
777            self.engine_config.clone(),
778            self.cache_manager.clone(),
779            manifest_ctx,
780            self.listener.clone(),
781            schema_metadata_manager,
782            self.engine_config.max_background_compactions,
783            Some(status.pick_history.clone()),
784        );
785        // Bump the counter through a disjoint field borrow so the `status`
786        // borrow stays alive; nothing could have removed the status since it
787        // was fetched above.
788        let plan_id = Self::next_plan_id(&mut self.next_plan_id);
789        status.start_regular_picking(plan_id);
790        self.dispatch_compaction_planning(
791            plan_id,
792            request,
793            compact_request::Options::Regular(Default::default()),
794            None,
795            CompactionTrigger::Automatic,
796        );
797        debug!(
798            "Successfully scheduled next compaction planning for region id: {}",
799            region_id
800        );
801        true
802    }

862    /// Completes task-specific cancellation without discarding another pending request.
863    fn finish_cancelled_compaction(
864        &mut self,
865        region_id: RegionId,
866        manifest: &ManifestContextRef,
867        schemas: SchemaMetadataManagerRef,
868    ) -> CompactionTransition {
869        if let Some(status) = self.region_status.get_mut(&region_id)
870            && status.finish_task_cancellation()
871        {
872            self.handle_pending_compaction_request(region_id, manifest, schemas);
873            return CompactionTransition::NoAction;
874        }
875        CompactionTransition::from_pending_ddls(self.finish_compaction_on_cancel(region_id))
876    }

878    /// Returns the cycle to idle and records any DDL handoff in the same resident entry.
879    fn finish_compaction_on_cancel(&mut self, region_id: RegionId) -> Vec<SenderDdlRequest> {
880        let Some(status) = self.region_status.get_mut(&region_id) else {
881            return Vec::new();
882        };
883
884        let ddls = status.on_cancel();
885        self.fence_ddl_replies(region_id, ddls)
886    }

888    /// Hands off pending DDLs after compaction ends and cancels remaining compaction waiters.
889    fn take_ready_ddls(&mut self, region_id: RegionId) -> Option<Vec<SenderDdlRequest>> {
890        if self
891            .region_status
892            .get(&region_id)?
893            .pending_ddl_requests
894            .is_empty()
895        {
896            return None;
897        }
898        Some(self.finish_compaction_on_cancel(region_id))
899    }

901    /// Hands off a drained batch of DDLs and fences new compaction until every reply lands.
902    ///
903    /// Installs the fence, intercepts each DDL's reply, and returns the same batch so the caller
904    /// can execute it. The real result of each DDL is still forwarded to its original sender.
905    ///
906    /// # Constraints
907    /// Must only be called from a compaction terminal transition, and only by the worker that owns
908    /// `region_id`.
909    fn fence_ddl_replies(
910        &mut self,
911        region_id: RegionId,
912        mut ddls: Vec<SenderDdlRequest>,
913    ) -> Vec<SenderDdlRequest> {
914        if ddls.is_empty() {
915            return ddls;
916        }
917        let generation = Self::next_plan_id(&mut self.next_plan_id);
918        if let Some(status) = self.region_status.get_mut(&region_id) {
919            // Set region status to "executing DDLs" so future compaction won't be scheduled.
920            status.executing_ddl = Some(DdlExecution::new(generation, ddls.len()));
921        }
922
923        for (request_index, ddl) in ddls.iter_mut().enumerate() {
924            // Swap the DDL's real reply sender for a temporary one so we can observe it. The
925            // spawned task forwards the observed result to the original caller and, in between,
926            // notifies the scheduler that this slot of the batch has completed.
927            let (tx, rx) = tokio::sync::oneshot::channel();
928            let reply = std::mem::replace(&mut ddl.sender, tx.into());
929            let sender = self.request_sender.clone();
930            common_runtime::spawn_global(async move {
931                use snafu::ResultExt;
932                let result = rx
933                    .await
934                    .context(crate::error::RecvSnafu)
935                    .and_then(|result| result);
936                // Route the completion through the worker channel so `on_ddl_complete` sees it
937                // serialized with other region state transitions.
938                //
939                // At most one batch can be outstanding per region: [`CompactionStatus::has_ddl`] blocks
940                // compaction admission while `executing_ddl` is set, so `fence_ddl_replies` cannot run again
941                // for the same region before the current batch drains. A single `(generation, request_index)`
942                // pair therefore identifies every reply. `generation` comes from the scheduler-wide counter,
943                // which outlives region statuses, so a late reply from a previous incarnation cannot clear the
944                // current fence. Inside the batch `request_index` makes completion idempotent: only the last
945                // distinct reply releases the fence.
946                let _ = sender
947                    .send(WorkerRequestWithTime::new(
948                        crate::request::WorkerRequest::Background {
949                            region_id,
950                            notify: crate::request::BackgroundNotify::CompactionDdlComplete {
951                                generation,
952                                request_index,
953                            },
954                        },
955                    ))
956                    .await;
957                reply.send(result);
958            });
959        }
960        ddls
961    }
962
963    /// Releases the DDL fence after its last reply, ignoring replies from an older generation.
964    pub(crate) fn on_ddl_complete(
965        &mut self,
966        region_id: RegionId,
967        generation: u64,
968        request_index: usize,
969    ) {
970        let Some(status) = self.region_status.get_mut(&region_id) else {
971            return;
972        };
973        // `complete` returns false for a stale generation or a duplicate `request_index`, so a late or
974        // repeated notify cannot end the batch early. It only clears the fence on the last distinct
975        // reply, which is what lets `has_ddl` unblock compaction admission again.
976        if let Some(ddl) = &mut status.executing_ddl
977            && ddl.complete(generation, request_index)
978        {
979            status.executing_ddl = None;
980        }
981    }
982}

984impl Drop for CompactionScheduler {
985    fn drop(&mut self) {
986        for (region_id, mut status) in self.region_status.drain() {
987            // We are shutting down so notify all pending tasks.
988            status.on_failure(Arc::new(RegionClosedSnafu { region_id }.build()));
989            self.coordinator.forget_region(region_id);
990        }
991    }
992}

995    /// Reads the same published task state as engine queries.
996    #[cfg(test)]
997    pub(crate) fn inspect_tasks(
998        &self,
999        filters: &MetadataFilter,
1000        node_id: Option<u64>,
1001    ) -> Vec<CompactionTaskEntry> {
1002        let mut entries = Vec::new();
1003        for (region_id, status) in &self.region_status {
1004            if !filters.matches(|name| match name {
1005                compaction_task::REGION_ID => Some(ScalarValue::UInt64(Some(region_id.as_u64()))),
1006                compaction_task::TABLE_ID => Some(ScalarValue::UInt32(Some(region_id.table_id()))),
1007                compaction_task::NODE_ID => Some(ScalarValue::UInt64(node_id)),
1008                _ => None,
1009            }) {
1010                continue;
1011            }
1012            let (active, pending) = CompactionState::current(&status.shared_state);
1013            for task in active.into_iter().chain(pending) {
1014                let entry = task.snapshot(node_id);
1015                if entry.matches(filters) {
1016                    entries.push(entry);
1017                }
1018            }
1019        }
1020        entries
1021    }
