43/// Tracks one unit through its terminal result and subsequent task exit.
44#[derive(Debug)]
45struct UnitExecution {
46    state: CancellableTaskState,
47    /// Observation flags only; they do not publish other fields or form an atomic snapshot.
48    /// The owning worker is their sole writer.
49    finished: AtomicBool,
50    released: AtomicBool,
51}
52
53/// Balances the inflight charge for a polled job on completion, unwind or drop.
54/// An unpolled job never acquires a charge.
55struct InflightCompaction(IntGauge);
56
57impl InflightCompaction {
58    /// Runs the job inside the same accounting scope on every exit path.
59    async fn run(job: impl Future<Output = ()>, gauge: IntGauge) {
60        let _inflight = Self::start(gauge);
61        job.await;
62    }
63
64    fn start(gauge: IntGauge) -> Self {
65        gauge.inc();
66        Self(gauge)
67    }
68}
69
70impl Drop for InflightCompaction {
71    fn drop(&mut self) {
72        self.0.dec();
73    }
74}

76/// A request's local units. All transitions run on the owning region worker.
77#[derive(Debug)]
78pub(crate) struct LocalCompaction {
79    coordinator: Arc<CompactionCoordinator>,
80    region_id: RegionId,
81    /// All pending and submitted units, retained until task exit. Tasks own their input leases.
82    /// Keyed by each unit's execution plan ID, allocated from the scheduler's `next_plan_id`;
83    /// it equals `CompactionExecution::plan_id()` and the `plan_id` in unit notifications.
84    units: HashMap<u64, Arc<UnitExecution>>,
85    /// Built once after admission. No units are added after this set is published.
86    /// Shared handles retain execution flags only, never tasks or SST leases.
87    unit_states: Arc<[Arc<UnitExecution>]>,
88    #[cfg(test)]
89    publication_count: Arc<AtomicUsize>,
90    pending: VecDeque<CompactionTask>,
91    parallelism: usize,
92    error: Option<Arc<Error>>,
93    made_progress: bool,
94    /// Retains accepted cancellation while non-cancellable siblings finish committing.
95    cancellation_requested: bool,
96}
97
98/// Query-visible local progress with shared background execution flags.
99#[derive(Debug, Clone)]
100pub(crate) struct LocalCompactionState {
101    units: Arc<[Arc<UnitExecution>]>,
102    cancellation_requested: bool,
103}
104
105impl LocalCompactionState {
106    /// Aggregates live units: cancellation takes precedence over running and queued states;
107    /// committing takes precedence over executing and waiting phases. A cancelling cycle
108    /// can still be committing while non-cancellable siblings finish.
109    pub(crate) fn inspect(&self, entry: &mut CompactionTaskEntry) {
110        entry.total_units = Some(self.units.len() as u64);
111        let mut completed = 0;
112        let mut cancelling = self.cancellation_requested;
113        let mut committing = false;
114        let mut running = false;
115        for unit in self.units.iter() {
116            completed += u64::from(unit.finished.load(Ordering::Relaxed));
117            if unit.released.load(Ordering::Relaxed) {
118                continue;
119            }
120            cancelling |= unit.state.is_cancelled();
121            committing |= unit.state.commit_started();
122            running |= unit.state.started();
123        }
124        entry.completed_units = Some(completed);
125        running |= committing;
126        entry.state = if cancelling {
127            compaction_task::STATE_CANCELLING
128        } else if running {
129            compaction_task::STATE_RUNNING
130        } else {
131            compaction_task::STATE_QUEUED
132        }
133        .into();
134        entry.phase = if committing {
135            compaction_task::PHASE_COMMITTING
136        } else if running {
137            compaction_task::PHASE_EXECUTING
138        } else {
139            compaction_task::PHASE_WAITING
140        }
141        .into();
142    }
143}

188    /// Requests cancellation for all units that have not started committing.
189    pub(crate) fn request_cancel(&mut self) -> RequestCancelResult {
190        let mut result = RequestCancelResult::TooLateToCancel;
191        for unit in self.units.values() {
192            match unit.state.request_cancel() {
193                RequestCancelResult::CancelIssued => result = RequestCancelResult::CancelIssued,
194                RequestCancelResult::AlreadyCancelling
195                    if result != RequestCancelResult::CancelIssued =>
196                {
197                    result = RequestCancelResult::AlreadyCancelling;
198                }
199                _ => {}
200            }
201        }
202        self.cancellation_requested |= result != RequestCancelResult::TooLateToCancel;
203        self.coordinator.cancel_pending(self.region_id);
204        result
205    }
206
207    /// Counts submitted units that have not exited, including queued jobs.
208    fn num_running(&self) -> usize {
209        self.units.len() - self.pending.len()
210    }
211
212    /// Checks this attempt's phase without depending on any sibling's apply status.
213    fn can_apply(&self, id: u64) -> bool {
214        self.units.get(&id).is_some_and(|unit| {
215            !unit.finished.load(Ordering::Relaxed) && unit.state.commit_started()
216        })
217    }
218
219    /// Removes an exited or rejected unit from scheduling and shared phase aggregation.
220    fn release_unit(&mut self, id: u64) {
221        if let Some(unit) = self.units.remove(&id) {
222            unit.released.store(true, Ordering::Relaxed);
223        }
224    }
225
226    /// Records a terminal result once, but still awaits task exit.
227    fn finish(&mut self, id: u64, result: std::result::Result<bool, Arc<Error>>) {
228        let Some(unit) = self.units.get(&id) else {
229            return;
230        };
231        if unit.finished.load(Ordering::Relaxed) {
232            return;
233        }
234        unit.finished.store(true, Ordering::Relaxed);
235        match result {
236            Ok(progress) => self.made_progress |= progress,
237            Err(err) => {
238                // A real failure must not be hidden by a sibling's cooperative cancellation.
239                if self.error.as_ref().is_none_or(|current| {
240                    matches!(current.as_ref(), Error::CompactionCancelled { .. })
241                }) {
242                    self.error = Some(err);
243                }
244            }
245        }
246    }
247}

251    fn drop(&mut self) {
252        self.request_cancel();
253    }

256impl CompactionScheduler {
257    /// Builds unit tasks with fresh execution IDs and shares their already-reserved inputs.
258    pub(crate) fn prepare_local_units(
259        &mut self,
260        region: CompactionRegion,
261        units: Vec<CompactionUnit>,
262        reserved: CompactingFiles,
263    ) -> CompactionPhase {
264        // Memory admission may have removed oversized units from the picker output.
265        // Report only the inputs that this local cycle will actually execute.
266        if let Some(info) = self
267            .region_status
268            .get_mut(&region.region_id)
269            .and_then(|status| status.active.task_state.as_mut())
270        {
271            info.set_plan_stats(
272                units.iter().flat_map(|unit| &unit.inputs),
273                units.iter().flat_map(|unit| &unit.outputs),
274            );
275        }
276        let mut local = LocalCompaction {
277            coordinator: self.coordinator.clone(),
278            region_id: region.region_id,
279            units: HashMap::new(),
280            unit_states: Arc::from([]),
281            #[cfg(test)]
282            publication_count: Arc::new(AtomicUsize::new(0)),
283            pending: VecDeque::new(),
284            parallelism: region.max_parallelism.max(1),
285            error: None,
286            made_progress: false,
287            cancellation_requested: false,
288        };
289        let context = CompactionContext::from(&region);
290        // Planning is complete; unit tasks must not pin the other units' inputs.
291        drop(region);
292        for unit in units {
293            let files = reserved.for_inputs(&unit.inputs);
294            let id = Self::next_plan_id(&mut self.next_plan_id);
295            let execution = CompactionExecution::new(id, files);
296            let state = CancellableTaskState::new();
297            local.units.insert(
298                id,
299                Arc::new(UnitExecution {
300                    state: state.clone(),
301                    finished: AtomicBool::new(false),
302                    released: AtomicBool::new(false),
303                }),
304            );
305            local.pending.push_back(CompactionTask {
306                state,
307                execution,
308                compaction_context: context.clone(),
309                request_sender: self.request_sender.clone(),
310                listener: self.listener.clone(),
311                estimated_memory_bytes: unit.estimated_memory_bytes(),
312                unit,
313                memory_manager: self.memory_manager.clone(),
314                memory_policy: self.memory_policy,
315                uncommitted: UncommittedSsts::new(
316                    context.region_id,
317                    context.access_layer.clone(),
318                    Some(context.cache_manager.clone()),
319                ),
320            });
321        }
322        local.unit_states = local.units.values().cloned().collect();
323        CompactionPhase::Units(local)
324    }
325
326    /// Returns local unit state only while the region is in a local execution phase.
327    fn local_units_mut(&mut self, region_id: RegionId) -> Option<&mut LocalCompaction> {
328        match &mut self.region_status.get_mut(&region_id)?.active.phase {
329            CompactionPhase::Units(local) => Some(local),
330            _ => None,
331        }
332    }
333
334    /// Submits pending units up to the request limit and arranges task-exit notifications.
335    pub(crate) fn dispatch_local_units(&mut self, region_id: RegionId) {
336        self.submit_local_units(region_id);
337        if let Some(status) = self.region_status.get(&region_id) {
338            status.publish_state();
339        }
340    }
341
342    fn submit_local_units(&mut self, region_id: RegionId) {
343        loop {
344            let Some(local) = self.local_units_mut(region_id) else {
345                return;
346            };
347            if local.num_running() >= local.parallelism {
348                return;
349            }
350            let Some(mut task) = local.pending.pop_front() else {
351                return;
352            };
353            let id = task.execution.plan_id();
354            let state = task.state.clone();
355            let input_bytes = task.unit.estimated_input_bytes();
356            let sender = self.request_sender.clone();
357            let completion = Box::pin(async move {
358                let _ = sender
359                    .send(WorkerRequestWithTime::new(WorkerRequest::Background {
360                        region_id,
361                        notify: BackgroundNotify::CompactionUnit(
362                            CompactionUnitNotification::Released { plan_id: id },
363                        ),
364                    }))
365                    .await;
366            });
367            let job = Box::pin(InflightCompaction::run(
368                async move {
369                    task.run().await;
370                    drop(task);
371                },
372                INFLIGHT_COMPACTION_COUNT.clone(),
373            ));
374            if let Err(err) = self.coordinator.schedule(
375                region_id,
376                ReadyUnit::new(state, input_bytes, job, completion),
377            ) && let Some(local) = self.local_units_mut(region_id)
378            {
379                local.finish(id, Err(Arc::new(err)));
380                local.release_unit(id);
381            }
382        }
383    }
384
385    /// Accepts an edit from any current unit that has started committing and is not finished.
386    pub(crate) fn can_apply_unit(&self, region_id: RegionId, id: u64) -> bool {
387        self.region_status.get(&region_id).is_some_and(|status| {
388            matches!(&status.active.phase, CompactionPhase::Units(local)
389                if local.can_apply(id))
390        })
391    }
392
393    /// Records apply progress or failure without ending sibling executions.
394    pub(crate) fn on_unit_finished(
395        &mut self,
396        region_id: RegionId,
397        id: u64,
398        result: std::result::Result<bool, Arc<Error>>,
399    ) {
400        if let Some(local) = self.local_units_mut(region_id) {
401            local.finish(id, result);
402        }
403        if let Some(status) = self.region_status.get(&region_id) {
404            status.publish_state();
405        }
406    }
407
408    /// Accounts for task exit, starts pending units, and checks request completion.
409    pub(crate) async fn on_unit_released(
410        &mut self,
411        region_id: RegionId,
412        id: u64,
413        manifest: &ManifestContextRef,
414        schemas: SchemaMetadataManagerRef,
415    ) -> CompactionTransition {
416        let Some(local) = self.local_units_mut(region_id) else {
417            return CompactionTransition::NoAction;
418        };
419        if !local.units.contains_key(&id) {
420            return CompactionTransition::NoAction;
421        }
422        let missing_result = if local.units[&id].state.is_cancelled() {
423            error::CompactionCancelledSnafu.build()
424        } else {
425            error::UnexpectedSnafu {
426                reason: "Compaction unit exited without a terminal notification",
427            }
428            .build()
429        };
430        local.finish(id, Err(Arc::new(missing_result)));
431        local.release_unit(id);
432        self.dispatch_local_units(region_id);
433        self.finish_local_units(region_id, manifest, schemas).await
434    }
435
436    /// Completes the request only after all units exit, using the existing terminal paths.
437    pub(crate) async fn finish_local_units(
438        &mut self,
439        region_id: RegionId,
440        manifest: &ManifestContextRef,
441        schemas: SchemaMetadataManagerRef,
442    ) -> CompactionTransition {
443        let Some(local) = self.local_units_mut(region_id) else {
444            return CompactionTransition::NoAction;
445        };
446        if !local.units.is_empty() {
447            return CompactionTransition::NoAction;
448        }
449        if let Some(err) = local.error.take() {
450            if matches!(err.as_ref(), Error::CompactionCancelled { .. }) {
451                return self.finish_cancelled_compaction(region_id, manifest, schemas);
452            }
453            self.on_compaction_failed(region_id, err);
454            return CompactionTransition::NoAction;
455        }
456        let progress = local.made_progress;
457        if let Some(status) = self.region_status.get_mut(&region_id) {
458            for waiter in status.take_waiters() {
459                waiter.send(Ok(0));
460            }
461        }
462        self.on_compaction_finished(region_id, manifest, schemas, progress)
463            .await
464    }

611    #[test]
612    fn test_shared_units_match_live_map_aggregation() {
613        #[derive(Debug, Clone, Copy)]
614        enum Phase {
615            Queued,
616            Running,
617            Committing,
618            Finished,
619            Released,
620            Cancelled,
621            CancelledReleased,
622        }
623        let phases = [
624            Phase::Queued,
625            Phase::Running,
626            Phase::Committing,
627            Phase::Finished,
628            Phase::Released,
629            Phase::Cancelled,
630            Phase::CancelledReleased,
631        ];
632        for first in phases {
633            for second in phases {
634                let mut local = local_compaction();
635                let published = local.current_state();
636                let mut completed = 0;
637                for (id, phase) in [(1, first), (2, second)] {
638                    let state = &local.units[&id].state;
639                    match phase {
640                        Phase::Queued => {}
641                        Phase::Running => state.mark_started(),
642                        Phase::Committing | Phase::Finished | Phase::Released => {
643                            assert!(state.mark_commit_started());
644                        }
645                        Phase::Cancelled | Phase::CancelledReleased => {
646                            assert_eq!(state.request_cancel(), RequestCancelResult::CancelIssued);
647                        }
648                    }
649                    if matches!(
650                        phase,
651                        Phase::Finished | Phase::Released | Phase::CancelledReleased
652                    ) {
653                        local.finish(id, Ok(false));
654                        completed += 1;
655                    }
656                    if matches!(phase, Phase::Released | Phase::CancelledReleased) {
657                        local.release_unit(id);
658                    }
659                }
660                // The former implementation aggregated only units remaining in this map.
661                let cancelling = local.units.values().any(|unit| unit.state.is_cancelled());
662                let committing = local.units.values().any(|unit| unit.state.commit_started());
663                let running = committing || local.units.values().any(|unit| unit.state.started());
664                let expected_state = if cancelling {
665                    "cancelling"
666                } else if running {
667                    "running"
668                } else {
669                    "queued"
670                };
671                let expected_phase = if committing {
672                    "committing"
673                } else if running {
674                    "executing"
675                } else {

721    #[test]
722    fn test_compaction_cancelling_survives_cancelled_sibling_release() {
723        let mut local = local_compaction();
724        let mut entry = task_entry();
725        // One published view observes background phase changes without republishing.
726        let published = local.current_state();
727        published.inspect(&mut entry);
728        assert_eq!((&*entry.state, &*entry.phase), ("queued", "waiting"));
729        local.units[&1].state.mark_started();
730        published.inspect(&mut entry);
731        assert_eq!((&*entry.state, &*entry.phase), ("running", "executing"));
732        assert!(local.units[&1].state.mark_commit_started());
733        published.inspect(&mut entry);
734        assert_eq!((&*entry.state, &*entry.phase), ("running", "committing"));
735        assert_eq!(local.request_cancel(), RequestCancelResult::CancelIssued);
736        published.inspect(&mut entry);
737        assert_eq!((&*entry.state, &*entry.phase), ("cancelling", "committing"));
738
739        // The cancelled sibling exits first; the committing sibling cannot be cancelled.
740        local.finish(2, Err(Arc::new(error::CompactionCancelledSnafu.build())));
741        local.release_unit(2);
742        assert!(!local.units[&1].state.is_cancelled());
743        local.inspect(&mut entry);
744        assert_eq!((&*entry.state, &*entry.phase), ("cancelling", "committing"));
745        assert_eq!(entry.completed_units, Some(1));
746        assert_eq!(entry.total_units, Some(2));
747
748        // A request arriving after every unit crossed the boundary is not accepted.
749        let mut too_late = local_compaction();
750        for unit in too_late.units.values() {
751            assert!(unit.state.mark_commit_started());
752        }
753        assert_eq!(
754            too_late.request_cancel(),
755            RequestCancelResult::TooLateToCancel
756        );
757        too_late.inspect(&mut entry);
758        assert_eq!((&*entry.state, &*entry.phase), ("running", "committing"));
759    }
