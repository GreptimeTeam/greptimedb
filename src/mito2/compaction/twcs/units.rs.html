15//! Unit-based compaction picks for one TWCS window.
16//!
17//! Picks select a contiguous interval of *units* in a fixed order. Grouping
18//! prefers rewriting related files together, but input limits and fallback
19//! paths may split groups or use single-file units. Separate admission checks
20//! preserve each pick's progress guarantees under the assumptions below.
21//!
22//! - [`pick_active_l1`] reduces the L1 file count of the active window.
23//! - [`pick_small_l1`] reduces small L1 files without rewriting large peers.
24//! - [`pick_closed_merge`] only merges intervals of an inactive window whose
25//!   outputs cannot add conflicts. Compaction outputs stay inside the bounding
26//!   box of the inputs, so they can only overlap files intruding on that box.
27//!
28//! When a window's split-unit conflict graph is too dense to materialize,
29//! [`pick_closed_merge`] falls back to a bounded file reduction and admits it
30//! only when the same conflict bound holds.
31//!
32//! # Termination
33//!
34//! Let `P` be the number of conflicting file pairs ([`Extent::conflicts`]) in an
35//! inactive window and `N` its file count. A merge removes every pair among its
36//! inputs. Its outputs are one sorted run with no internal conflicts,
37//! so they only conflict with the `I` files intruding on the inputs' box, at
38//! most `outputs · I` pairs. An inactive pick only merges inputs that had at
39//! least as many pairs with other files, so `P` does not grow; without
40//! intruders the inputs are closed and add no pairs. Every inactive pick either
41//! resolves a pair (`P` drops) or removes a file (`P` does not grow and `N`
42//! drops), so `(P, N)` strictly decreases in lexicographic order until the
43//! window receives new writes. Active-window picks may add conflicts, so each
44//! of them must remove a file: `N` strictly decreases.
45//!
46//! The argument relies on how the compactor writes outputs:
47//!
48//! - Outputs hold a subset of the input rows, so each output's time and key
49//!   bounds lie inside the inputs' bounding box.
50//! - Outputs are one stream in (key, time) order. For regions with a primary
51//!   key, the writer only splits between complete series, so output key ranges
52//!   are disjoint. Without a primary key, outputs have disjoint or touching
53//!   time ranges. Neither is an [`Extent::conflicts`] pair.
54//! - A split only starts a new file once the current one reaches the size
55//!   threshold, and outputs are not larger than inputs in total, so
56//!   [`PickLimits::predicted_outputs`] bounds the output count.
57//!
58//! `test_closed_merges_terminate` and `test_active_l1_picks_terminate` replay
59//! random windows with outputs written under these rules.

70/// An L1 file reduction may rewrite at most this many output files' worth of
71/// bytes per SST it removes.
72const MAX_REWRITE_FILES_PER_REDUCED_FILE: u64 = 4;
73
74/// A small L1 file occupies at most one quarter of the output size budget.
75const SMALL_L1_SIZE_FACTOR: u64 = 4;
76
77/// Maximum number of index nodes visited while building conflict structures for
78/// a pick. Above this, the picker falls back to a bounded file reduction instead
79/// of materializing an unbounded conflict graph.
80const CONFLICT_GRAPH_BUDGET: usize = 1 << 20;
81
82/// Output size limits and input limits shared by both picks.
83#[derive(Debug, Clone, Copy)]
84pub(crate) struct PickLimits {
85    /// Split threshold of compaction outputs; `None` writes a single output.
86    pub(crate) max_output_file_size: Option<u64>,
87    /// Maximum number of input files of one compaction.
88    pub(crate) max_input_files: usize,
89}
90
91impl PickLimits {
92    /// Upper bound of the outputs of compacting an interval: the writer only
93    /// starts a new file after the current one reaches the threshold, and never
94    /// splits a series, so files of a single primary key compact into one file.
95    fn predicted_outputs(&self, stats: &IntervalStats) -> usize {
96        match self.max_output_file_size {
97            Some(max) if max > 0 && stats.single_key.is_none() => {
98                stats.size.div_ceil(max).max(1) as usize
99            }
100            _ => 1,
101        }
102    }
103
104    /// Whether rewriting `size` bytes to remove `reduction` files stays within
105    /// the rewrite budget. An unset output size imposes no bound.
106    fn within_reduction_cost(&self, size: u64, reduction: usize) -> bool {
107        match self.max_output_file_size {
108            Some(max) if max > 0 => {
109                size <= (reduction as u64)
110                    .saturating_mul(MAX_REWRITE_FILES_PER_REDUCED_FILE)
111                    .saturating_mul(max)
112            }
113            _ => true,
114        }
115    }
116
117    /// Bytes a dominant sorted run may exceed its peers by in an overlap merge.
118    fn run_tolerance(&self) -> u64 {
119        self.max_output_file_size.unwrap_or(0)
120    }
121}

136/// Picks L1 files of the active window whose compaction removes files.
137///
138/// Units are groups of conflicting files, so a merge does not leave a file
139/// behind while rewriting its overlapping peers. A group larger than the input
140/// limit is split back into single files. If no interval of groups qualifies,
141/// for example because a large file dominates its group, the groups are split
142/// into single files, so the smaller files can still be merged. Intervals
143/// follow [`sort_units`]; the pick removes the most files, then rewrites the
144/// fewest bytes.
145pub(crate) fn pick_active_l1(
146    l1_files: &[FileHandle],
147    mapper: &PrimaryKeyRangeMapper,
148    options: PickLimits,
149) -> Option<UnitPick> {
150    pick_active_l1_with_budget(l1_files, mapper, options, CONFLICT_GRAPH_BUDGET)
151}
152
153/// [`pick_active_l1`] with an explicit work budget, so tests can exercise the
154/// single-file fallback without a window-sized conflict graph.
155fn pick_active_l1_with_budget(
156    l1_files: &[FileHandle],
157    mapper: &PrimaryKeyRangeMapper,
158    options: PickLimits,
159    work_budget: usize,
160) -> Option<UnitPick> {
161    let extents = extents_of(l1_files, mapper);
162    let index = ExtentIndex::new(&extents);
163    let mut units = match conflict_units(l1_files, &extents, &index, work_budget) {
164        Some(units) => units,
165        None => single_file_units(l1_files, &extents),
166    };
167    split_oversized_units(&mut units, l1_files, &extents, options.max_input_files);
168    if let Some(pick) = pick_reducing_interval(&mut units, l1_files, options) {
169        return Some(pick);
170    }
171    if units.len() == l1_files.len() {
172        return None;
173    }
174    split_oversized_units(&mut units, l1_files, &extents, 1);
175    pick_reducing_interval(&mut units, l1_files, options)
176}

178/// Picks the interval of `units` that removes the most files within the
179/// rewrite budget, then rewrites the fewest bytes.
180fn pick_reducing_interval(
181    units: &mut [Unit],
182    files: &[FileHandle],
183    options: PickLimits,
184) -> Option<UnitPick> {
185    sort_units(units);
186    let range = pick_reducing_interval_where(units, options, |_| true)?;
187    Some(UnitPick {
188        inputs: collect_inputs(&units[range], files),
189        reduces_files: true,
190        units: units.len(),
191        dense_fallback: false,
192    })
193}

195/// Selects a reducing interval of sorted units, with an additional filter so
196/// the dense fallback can retry without duplicating the reduction guards.
197fn pick_reducing_interval_where(
198    units: &[Unit],
199    options: PickLimits,
200    is_eligible: impl Fn(&IntervalStats) -> bool,
201) -> Option<Range<usize>> {
202    let mut best: Option<(usize, u64, Range<usize>)> = None;
203    // File reductions ignore conflicts, so split files are not counted.
204    for_each_interval(units, &[], options.max_input_files, |range, stats| {
205        let reduction = stats
206            .num_files
207            .saturating_sub(options.predicted_outputs(stats));
208        if stats.num_files < 2
209            || reduction == 0
210            || !options.within_reduction_cost(stats.size, reduction)
211            || stats.largest_file_size > stats.size - stats.largest_file_size
212            || !is_eligible(stats)
213        {
214            return;
215        }
216        let better = best.as_ref().is_none_or(|(best_reduction, best_size, _)| {
217            (reduction, Reverse(stats.size)) > (*best_reduction, Reverse(*best_size))
218        });
219        if better {
220            best = Some((reduction, stats.size, range));
221        }
222    });
223
224    best.map(|(_, _, range)| range)
225}

227/// Reduces small L1 files left behind by a closed merge, including in settling
228/// windows. The input bytes fit in one output and no input dominates its peers,
229/// so each merge removes files without repeatedly rewriting an accumulator.
230///
231/// Large and L0 files remain in the external conflict index. Check only the
232/// best reducing interval to bound planning work; rejecting it can miss other
233/// admissible intervals, but cannot increase the window's conflict count.
234pub(crate) fn pick_small_l1(
235    files: &[FileHandle],
236    mapper: &PrimaryKeyRangeMapper,
237    limits: PickLimits,
238) -> Option<UnitPick> {
239    let max_output = limits.max_output_file_size.filter(|size| *size > 0)?;
240    let extents = extents_of(files, mapper);
241    let mut units = single_file_units(files, &extents);
242    units.retain(|unit| {
243        let file = &files[unit.files[0]];
244        file.level() == 1 && file.size() <= max_output / SMALL_L1_SIZE_FACTOR
245    });
246    sort_units(&mut units);
247    let range = pick_reducing_interval_where(&units, limits, |stats| stats.size <= max_output)?;
248    let index = ExtentIndex::new(&extents);
249    if !adds_no_conflicts(&units[range.clone()], &extents, &index, 1) {
250        return None;
251    }
252    Some(UnitPick {
253        inputs: collect_inputs(&units[range], files),
254        reduces_files: true,
255        units: units.len(),
256        dense_fallback: false,
257    })
258}

275/// Picks an interval of units among all files of an inactive window.
276///
277/// Units are the closures of conflicting files: no file outside a unit intrudes
278/// on its bounding box. A closure larger than the input limit is split back into
279/// single files, so files around a late file spanning a large sorted run can
280/// still be merged. Intervals follow [`sort_units`]. An interval is eligible if
281/// its outputs add no more conflicts than its files have with other files (see
282/// [`adds_no_conflicts`]), and it
283///
284/// - removes files within the rewrite budget, or
285/// - resolves conflicts without rewriting a dominant sorted run (unless
286///   [`ClosedMergeOptions::allow_unbalanced`]), and adds at most one tail file.
287///
288/// The pick maximizes resolved conflicts plus removed files per rewritten byte.
289/// A window whose split-unit conflict graph is too dense to materialize falls
290/// back to a bounded file reduction; see [`pick_dense_closed_merge`].
291pub(crate) fn pick_closed_merge(
292    files: &[FileHandle],
293    mapper: &PrimaryKeyRangeMapper,
294    options: ClosedMergeOptions,
295) -> Option<UnitPick> {
296    pick_closed_merge_with_budget(files, mapper, options, CONFLICT_GRAPH_BUDGET)
297}
298
299/// [`pick_closed_merge`] with an explicit work budget, so tests can exercise
300/// the dense fallback without building a window-sized conflict graph.
301fn pick_closed_merge_with_budget(
302    files: &[FileHandle],
303    mapper: &PrimaryKeyRangeMapper,
304    options: ClosedMergeOptions,
305    work_budget: usize,
306) -> Option<UnitPick> {
307    let [best, _] = pick_closed_merge_candidates(files, mapper, options, work_budget, false);
308    best
309}

321/// A scored interval retained until the best candidate(s) are materialized.
322struct ClosedMergeCandidate {
323    benefit: u64,
324    size: u64,
325    range: Range<usize>,
326    reduces_files: bool,
327}
328
329fn pick_closed_merge_candidates(
330    files: &[FileHandle],
331    mapper: &PrimaryKeyRangeMapper,
332    options: ClosedMergeOptions,
333    work_budget: usize,
334    per_class: bool,
335) -> [Option<UnitPick>; 2] {
336    let limits = options.limits;
337    let extents = extents_of(files, mapper);
338    let index = ExtentIndex::new(&extents);
339    let mut dense = false;
340    let mut units = match conflict_units(files, &extents, &index, work_budget) {
341        Some(mut units) => {
342            close_units(&mut units, &extents, &index);
343            split_oversized_units(&mut units, files, &extents, limits.max_input_files);
344            units
345        }
346        None => {
347            dense = true;
348            single_file_units(files, &extents)
349        }
350    };
351    sort_units(&mut units);
352    if dense {
353        return [
354            pick_dense_closed_merge(&units, files, &extents, &index, options),
355            None,
356        ];
357    }
358    let Some(pairs) = split_conflicts(&units, &extents, &index, work_budget) else {
359        return [
360            pick_dense_closed_merge(&units, files, &extents, &index, options),
361            None,
362        ];
363    };
364
365    let mut best: [Option<ClosedMergeCandidate>; 2] = [None, None];
366    for_each_interval(&units, &pairs, limits.max_input_files, |range, stats| {
367        if stats.num_files < 2 {
368            return;
369        }
370        let outputs = limits.predicted_outputs(stats);
371        if outputs > stats.num_files + 1 {
372            return;
373        }
374        let reduction = stats.num_files.saturating_sub(outputs);
375        let reduces_files = reduction > 0
376            && (options.allow_file_reduction || stats.conflicts > 0)
377            && limits.within_reduction_cost(stats.size, reduction);
378        if (stats.conflicts > 0 && !options.allow_conflicts)
379            || (!reduces_files && stats.conflicts == 0)
380        {
381            return;
382        }
383        // Stateless/manual picks compare all intervals in one slot, retaining
384        // their original efficiency ordering and first-interval tie breaker.
385        let best = &mut best[usize::from(per_class && !reduces_files)];
386        let benefit = (reduction + stats.conflicts) as u64;
387        let better = best.as_ref().is_none_or(|candidate| {
388            is_more_efficient(benefit, stats.size, candidate.benefit, candidate.size)
389        });
390        if !better {
391            return;
392        }
393        let units = &units[range.clone()];
394        // Overlap merges must not rewrite a dominant run for a small peer. A
395        // file-reducing merge is already bounded by its rewrite budget.
396        if !reduces_files
397            && !options.allow_unbalanced
398            && !has_balanced_components(units, files, &extents, limits.run_tolerance())
399        {
400            return;
401        }
402        if units.len() > 1 && !adds_no_conflicts(units, &extents, &index, outputs) {
403            return;
404        }
405        *best = Some(ClosedMergeCandidate {
406            benefit,
407            size: stats.size,
408            range,
409            reduces_files,
410        });
411    });
412
413    best.map(|candidate| {
414        candidate.map(|candidate| UnitPick {
415            inputs: collect_inputs(&units[candidate.range], files),
416            reduces_files: candidate.reduces_files,
417            units: units.len(),
418            dense_fallback: false,
419        })
420    })
421}

423/// Picks a bounded file reduction when the split-unit conflict graph is too
424/// dense to materialize.
425///
426/// The candidate comes from [`pick_reducing_interval_where`], which ignores
427/// conflicts. If rejected, it is searched once more for a one-output interval.
428/// Both candidates must pass [`adds_no_conflicts`], so the `(P, N)` termination
429/// argument still holds. Searching only within the original candidate adds at
430/// most O(K²) interval visits for K input files and one extra conflict check,
431/// rather than checking external conflicts for every interval in the window.
432/// This bounded retry can still miss other safely reducible intervals.
433fn pick_dense_closed_merge(
434    units: &[Unit],
435    files: &[FileHandle],
436    extents: &[Extent],
437    index: &ExtentIndex,
438    options: ClosedMergeOptions,
439) -> Option<UnitPick> {
440    let limits = options.limits;
441    let range = pick_reducing_interval_where(units, limits, |_| true)?;
442    let mut selected = &units[range];
443    if !is_admissible_dense_interval(selected, extents, index, options) {
444        let range = pick_reducing_interval_where(selected, limits, |stats| {
445            limits.predicted_outputs(stats) == 1
446        })?;
447        selected = &selected[range];
448        if !is_admissible_dense_interval(selected, extents, index, options) {
449            return None;
450        }
451    }
452    Some(UnitPick {
453        inputs: collect_inputs(selected, files),
454        reduces_files: true,
455        units: units.len(),
456        dense_fallback: true,
457    })
458}

460/// File reduction and rewrite limits are checked by the interval picker;
461/// each dense candidate must also respect the closed-window conflict policy.
462fn is_admissible_dense_interval(
463    units: &[Unit],
464    extents: &[Extent],
465    index: &ExtentIndex,
466    options: ClosedMergeOptions,
467) -> bool {
468    let stats = interval_stats_exact(units, extents);
469    (options.allow_file_reduction || stats.conflicts > 0)
470        && (options.allow_conflicts || stats.conflicts == 0)
471        && adds_no_conflicts(
472            units,
473            extents,
474            index,
475            options.limits.predicted_outputs(&stats),
476        )
477}

479/// Totals of an interval without materializing [`split_conflicts`].
480///
481/// After [`close_units`], files of different units never conflict, so only
482/// split units can conflict across units, and each split unit holds one file.
483/// The dense fallback may instead treat every file as a split unit, and then
484/// the same pairwise loop covers every cross-unit pair. Counting them takes
485/// `O(K²)` for an interval of at most `K` files, instead of building a
486/// window-wide adjacency.
487fn interval_stats_exact<'a>(units: &'a [Unit], extents: &[Extent]) -> IntervalStats<'a> {
488    let mut stats = IntervalStats::default();
489    let mut first = true;
490    for unit in units {
491        stats.num_files += unit.files.len();
492        stats.size += unit.size;
493        stats.largest_file_size = stats.largest_file_size.max(unit.largest_file_size);
494        stats.conflicts += unit.conflicts;
495        stats.single_key = if first {
496            unit.single_key()
497        } else {
498            stats
499                .single_key
500                .filter(|&key| unit.single_key() == Some(key))
501        };
502        first = false;
503    }
504
505    let split_files: Vec<usize> = units
506        .iter()
507        .filter(|unit| unit.split)
508        .map(|unit| unit.files[0])
509        .collect();
510    for (offset, &lhs) in split_files.iter().enumerate() {
511        for &rhs in &split_files[offset + 1..] {
512            if extents[lhs].conflicts(&extents[rhs]) {
513                stats.conflicts += 1;
514            }
515        }
516    }
517    stats
518}

520/// Orders units by minimum primary key, then by time, so adjacent units are
521/// key-adjacent. Compaction outputs and time-sliced flushes split a window by
522/// key ranges, so merging neighbors in this order keeps outputs from covering
523/// the key ranges of files left behind.
524fn sort_units(units: &mut [Unit]) {
525    units.sort_by(|lhs, rhs| {
526        (lhs.first_key(), lhs.extent.start, Reverse(lhs.extent.end))
527            .cmp(&(rhs.first_key(), rhs.extent.start, Reverse(rhs.extent.end)))
528            .then_with(|| lhs.files[0].cmp(&rhs.files[0]))
529    });
530}

532/// Whether `benefit` per `size` bytes beats the best so far; ties prefer fewer
533/// bytes, then the earlier interval.
534fn is_more_efficient(benefit: u64, size: u64, best_benefit: u64, best_size: u64) -> bool {
535    let lhs = u128::from(benefit) * u128::from(best_size.max(1));
536    let rhs = u128::from(best_benefit) * u128::from(size.max(1));
537    lhs > rhs || (lhs == rhs && size < best_size)
538}

540/// Time and primary key bounds of a file or a group of files.
541#[derive(Debug, Clone)]
542pub(crate) struct Extent {
543    start: Timestamp,
544    end: Timestamp,
545    /// Inclusive primary key bounds. `None` is unknown and matches any key.
546    primary_key: Option<(Bytes, Bytes)>,
547}
548
549impl Extent {
550    pub(crate) fn of_file(file: &FileHandle, mapper: &PrimaryKeyRangeMapper) -> Self {
551        let (start, end) = file.time_range();
552        Self {
553            start,
554            end,
555            primary_key: file.primary_key_range(mapper),
556        }
557    }
558
559    fn union(&self, other: &Self) -> Self {
560        Self {
561            start: self.start.min(other.start),
562            end: self.end.max(other.end),
563            primary_key: match (&self.primary_key, &other.primary_key) {
564                (Some((lhs_min, lhs_max)), Some((rhs_min, rhs_max))) => Some((
565                    lhs_min.clone().min(rhs_min.clone()),
566                    lhs_max.clone().max(rhs_max.clone()),
567                )),
568                _ => None,
569            },
570        }
571    }
572
573    /// Time ranges overlap; touching boundaries do not.
574    fn time_overlaps(&self, other: &Self) -> bool {
575        self.start.max(other.start) < self.end.min(other.end)
576    }
577
578    /// Whether two files overlap for the picker.
579    ///
580    /// Shared boundary keys count: independently written files can hold
581    /// different versions of the same row there. Compaction outputs do not
582    /// conflict with each other because the writer never splits a series.
583    pub(crate) fn conflicts(&self, other: &Self) -> bool {
584        self.time_overlaps(other)
585            && match (&self.primary_key, &other.primary_key) {
586                (Some(lhs), Some(rhs)) => {
587                    let (low, high) = ((&lhs.0).max(&rhs.0), (&lhs.1).min(&rhs.1));
588                    low <= high
589                }
590                _ => true,
591            }
592    }
593
594    /// Whether `self` may conflict with a file inside `area`, which is the
595    /// bounding box of a group of files.
596    ///
597    /// A shared boundary key may occur in an output, so the same inclusive
598    /// key overlap used for individual files also applies to the box.
599    fn intrudes(&self, area: &Self) -> bool {
600        self.conflicts(area)
601    }
602}

608/// Upper primary key bound, ordered after every key when unknown.
609#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
610enum UpperKey<'a> {
611    Key(&'a Bytes),
612    Unbounded,
613}
614
615/// Key and time bounds of the files in a subtree of [`ExtentIndex`].
616#[derive(Debug, Clone, Copy)]
617struct SubtreeBounds<'a> {
618    max_upper_key: UpperKey<'a>,
619    min_start: Timestamp,
620    max_end: Timestamp,
621}
622
623impl<'a> SubtreeBounds<'a> {
624    fn of(extent: &'a Extent) -> Self {
625        Self {
626            max_upper_key: match &extent.primary_key {
627                Some((_, max)) => UpperKey::Key(max),
628                None => UpperKey::Unbounded,
629            },
630            min_start: extent.start,
631            max_end: extent.end,
632        }
633    }
634
635    fn merge(self, other: Self) -> Self {
636        Self {
637            max_upper_key: self.max_upper_key.max(other.max_upper_key),
638            min_start: self.min_start.min(other.min_start),
639            max_end: self.max_end.max(other.max_end),
640        }
641    }
642}

644/// Files of a window ordered by their lower primary key, in a segment tree
645/// holding the bounds of each subtree.
646///
647/// A query visits the files whose key and time ranges may overlap an area and
648/// skips subtrees whose keys end before the area or whose times miss it. The
649/// files of a sorted run split by key then only meet their key neighbors,
650/// instead of every file sharing their time range.
651struct ExtentIndex<'a> {
652    extents: &'a [Extent],
653    /// Indices into `extents`, sorted by lower key with unknown keys first.
654    /// Sorted offset `i` is leaf `leaf_base + i`.
655    by_lower_key: Vec<usize>,
656    /// Array-backed tree rooted at index 1; padding leaves are `None`.
657    bounds: Vec<Option<SubtreeBounds<'a>>>,
658    /// First leaf index, a power of two.
659    leaf_base: usize,
660}
661
662impl<'a> ExtentIndex<'a> {
663    fn new(extents: &'a [Extent]) -> Self {
664        let lower_key = |i: usize| extents[i].primary_key.as_ref().map(|(min, _)| min);
665        let mut by_lower_key: Vec<usize> = (0..extents.len()).collect();
666        by_lower_key.sort_by(|&lhs, &rhs| {
667            (lower_key(lhs), extents[lhs].start).cmp(&(lower_key(rhs), extents[rhs].start))
668        });
669        let leaf_base = extents.len().next_power_of_two();
670        let mut bounds = vec![None; 2 * leaf_base];
671        for (offset, &i) in by_lower_key.iter().enumerate() {
672            bounds[leaf_base + offset] = Some(SubtreeBounds::of(&extents[i]));
673        }
674        for node in (1..leaf_base).rev() {
675            bounds[node] = match (bounds[2 * node], bounds[2 * node + 1]) {
676                (Some(lhs), Some(rhs)) => Some(lhs.merge(rhs)),
677                (lhs, rhs) => lhs.or(rhs),
678            };
679        }
680        Self {
681            extents,
682            by_lower_key,
683            bounds,
684            leaf_base,
685        }
686    }
687
688    /// Visits the files whose inclusive key range meets `area`'s and whose time
689    /// range overlaps it, a superset of the files that conflict with or intrude
690    /// on `area`. Takes O(log N) per visited file.
691    fn for_each_near(&self, area: &Extent, visit: impl FnMut(usize)) {
692        let mut budget = usize::MAX;
693        let _ = self.for_each_near_with_budget(area, &mut budget, visit);
694    }
695
696    /// [`for_each_near`] with a budget on visited tree nodes.
697    ///
698    /// Every node reached by the traversal consumes one unit, including nodes
699    /// pruned before any file callback. Returns `false` as soon as the budget is
700    /// exhausted, so a caller can stop a traversal whose pruning is poor even
701    /// when it finds few files.
702    fn for_each_near_with_budget(
703        &self,
704        area: &Extent,
705        budget: &mut usize,
706        mut visit: impl FnMut(usize),
707    ) -> bool {
708        let leaves_end = match &area.primary_key {
709            Some((_, max)) => self.by_lower_key.partition_point(|&i| {
710                self.extents[i]
711                    .primary_key
712                    .as_ref()
713                    .is_none_or(|(min, _)| min <= max)
714            }),
715            None => self.by_lower_key.len(),
716        };
717        self.visit_subtree(1, 0..self.leaf_base, leaves_end, area, budget, &mut visit)
718    }
719
720    fn visit_subtree(
721        &self,
722        node: usize,
723        leaves: Range<usize>,
724        leaves_end: usize,
725        area: &Extent,
726        budget: &mut usize,
727        visit: &mut impl FnMut(usize),
728    ) -> bool {
729        if *budget == 0 {
730            return false;
731        }
732        *budget -= 1;
733        if leaves.start >= leaves_end {
734            return true;
735        }
736        let Some(bounds) = self.bounds[node] else {
737            return true;
738        };
739        if bounds.min_start >= area.end
740            || bounds.max_end <= area.start
741            || area
742                .primary_key
743                .as_ref()
744                .is_some_and(|(min, _)| bounds.max_upper_key < UpperKey::Key(min))
745        {
746            return true;
747        }
748        if leaves.len() == 1 {
749            let i = self.by_lower_key[leaves.start];
750            if self.extents[i].time_overlaps(area) {
751                visit(i);
752            }
753            return true;
754        }
755        let mid = leaves.start + leaves.len() / 2;
756        if !self.visit_subtree(2 * node, leaves.start..mid, leaves_end, area, budget, visit) {
757            return false;
758        }
759        self.visit_subtree(
760            2 * node + 1,
761            mid..leaves.end,
762            leaves_end,
763            area,
764            budget,
765            visit,
766        )
767    }
768}

777/// Files that are compacted together.
778#[derive(Debug)]
779struct Unit {
780    /// Ascending indices into the candidate files.
781    files: Vec<usize>,
782    extent: Extent,
783    size: u64,
784    largest_file_size: u64,
785    /// Number of conflicting file pairs inside the unit.
786    conflicts: usize,
787    /// A single file split from a unit over the input limit. It may conflict
788    /// with other split files, see [`split_conflicts`].
789    split: bool,
790}
791
792impl Unit {
793    fn new(
794        members: Vec<usize>,
795        files: &[FileHandle],
796        extents: &[Extent],
797        conflicts: usize,
798    ) -> Self {
799        let extent = members[1..]
800            .iter()
801            .fold(extents[members[0]].clone(), |extent, &i| {
802                extent.union(&extents[i])
803            });
804        let sizes = members.iter().map(|&i| files[i].size());
805        Self {
806            size: sizes.clone().sum(),
807            largest_file_size: sizes.max().unwrap_or(0),
808            files: members,
809            extent,
810            conflicts,
811            split: false,
812        }
813    }
814
815    /// The minimum primary key, with unknown keys ordered first.
816    fn first_key(&self) -> Option<&Bytes> {
817        self.extent.primary_key.as_ref().map(|(min, _)| min)
818    }
819
820    /// The primary key of a unit whose files all hold that single key.
821    fn single_key(&self) -> Option<&Bytes> {
822        self.extent
823            .primary_key
824            .as_ref()
825            .filter(|range| is_single_key(range))
826            .map(|(min, _)| min)
827    }
828}

830/// Groups files into connected components of [`Extent::conflicts`], each with
831/// its number of conflicting pairs. Each file only checks the files the index
832/// finds near it: O((N + M) log N) for M pairs of files whose key and time
833/// ranges meet.
834///
835/// Returns `None` once the index traversal has visited more than `work_budget`
836/// nodes, so a dense window can fall back before building a quadratic graph.
837fn conflict_units(
838    files: &[FileHandle],
839    extents: &[Extent],
840    index: &ExtentIndex,
841    work_budget: usize,
842) -> Option<Vec<Unit>> {
843    let mut sets = DisjointSet::new(extents.len());
844    let mut conflicts = vec![0; extents.len()];
845    let mut budget = work_budget;
846    for i in 0..extents.len() {
847        let completed = index.for_each_near_with_budget(&extents[i], &mut budget, |j| {
848            // Counts each pair once, from its smaller index.
849            if j > i && extents[i].conflicts(&extents[j]) {
850                sets.union(i, j);
851                conflicts[i] += 1;
852            }
853        });
854        if !completed {
855            return None;
856        }
857    }
858    Some(
859        sets.groups()
860            .into_iter()
861            .map(|members| {
862                let pairs = members.iter().map(|&i| conflicts[i]).sum();
863                Unit::new(members, files, extents, pairs)
864            })
865            .collect(),
866    )
867}

903/// Merges units until no file intrudes on a unit it does not belong to.
904///
905/// Each check only visits the files the index finds near the unit, and each
906/// merge re-checks only the merged unit.
907fn close_units(units: &mut Vec<Unit>, extents: &[Extent], index: &ExtentIndex) {
908    let mut owner = vec![0; extents.len()];
909    for (u, unit) in units.iter().enumerate() {
910        for &i in &unit.files {
911            owner[i] = u;
912        }
913    }
914    let mut alive = vec![true; units.len()];
915    let mut pending: Vec<usize> = (0..units.len()).collect();
916    while let Some(u) = pending.pop() {
917        if !alive[u] {
918            continue;
919        }
920        let area = &units[u].extent;
921        let mut intruders = std::collections::BTreeSet::new();
922        index.for_each_near(area, |i| {
923            if owner[i] != u && extents[i].intrudes(area) {
924                intruders.insert(owner[i]);
925            }
926        });
927        if intruders.is_empty() {
928            continue;
929        }
930        for v in intruders {
931            alive[v] = false;
932            let absorbed = std::mem::take(&mut units[v].files);
933            for &i in &absorbed {
934                owner[i] = u;
935            }
936            let (extent, size, largest, conflicts) = {
937                let other = &units[v];
938                (
939                    other.extent.clone(),
940                    other.size,
941                    other.largest_file_size,
942                    other.conflicts,
943                )
944            };
945            let unit = &mut units[u];
946            unit.files.extend(absorbed);
947            unit.extent = unit.extent.union(&extent);
948            unit.size += size;
949            unit.largest_file_size = unit.largest_file_size.max(largest);
950            unit.conflicts += conflicts;
951        }
952        pending.push(u);
953    }
954    let mut closed = Vec::with_capacity(units.len());
955    for (u, mut unit) in std::mem::take(units).into_iter().enumerate() {
956        if alive[u] {
957            unit.files.sort_unstable();
958            closed.push(unit);
959        }
960    }
961    *units = closed;
962}

964/// Totals of a contiguous interval of units.
965#[derive(Debug, Default)]
966struct IntervalStats<'a> {
967    num_files: usize,
968    size: u64,
969    largest_file_size: u64,
970    conflicts: usize,
971    /// The primary key of an interval whose files all hold that single key.
972    single_key: Option<&'a Bytes>,
973}

975/// Positions of the split units each unit conflicts with, sorted. Files of
976/// different units never conflict, so only split files have any.
977///
978/// Returns `None` once the index traversal has visited more than `work_budget`
979/// nodes, so a dense window cannot materialize an unbounded adjacency.
980fn split_conflicts(
981    units: &[Unit],
982    extents: &[Extent],
983    index: &ExtentIndex,
984    work_budget: usize,
985) -> Option<Vec<Vec<usize>>> {
986    let mut position = vec![None; extents.len()];
987    for (p, unit) in units.iter().enumerate() {
988        if unit.split {
989            position[unit.files[0]] = Some(p);
990        }
991    }
992    let mut budget = work_budget;
993    let mut result = Vec::with_capacity(units.len());
994    for unit in units {
995        if !unit.split {
996            result.push(Vec::new());
997            continue;
998        }
999        let i = unit.files[0];
1000        let mut neighbors = Vec::new();
1001        let completed = index.for_each_near_with_budget(&extents[i], &mut budget, |j| {
1002            if let Some(p) = position[j]
1003                && j != i
1004                && extents[i].conflicts(&extents[j])
1005            {
1006                neighbors.push(p);
1007            }
1008        });
1009        if !completed {
1010            return None;
1011        }
1012        neighbors.sort_unstable();
1013        result.push(neighbors);
1014    }
1015    Some(result)
1016}

1018/// Visits every contiguous interval of units with at most `max_input_files`
1019/// files: O(U · K) intervals for an input limit K. Each interval counts the
1020/// conflicts of its split files from `split_conflicts` (see
1021/// [`split_conflicts`]) with a binary search; an empty `split_conflicts`
1022/// skips them.
1023fn for_each_interval<'a>(
1024    units: &'a [Unit],
1025    split_conflicts: &[Vec<usize>],
1026    max_input_files: usize,
1027    mut visit: impl FnMut(Range<usize>, &IntervalStats<'a>),
1028) {
1029    for left in 0..units.len() {
1030        let mut stats = IntervalStats::default();
1031        for (right, unit) in units.iter().enumerate().skip(left) {
1032            stats.num_files += unit.files.len();
1033            if stats.num_files > max_input_files {
1034                break;
1035            }
1036            stats.size += unit.size;
1037            stats.largest_file_size = stats.largest_file_size.max(unit.largest_file_size);
1038            stats.conflicts += unit.conflicts;
1039            stats.single_key = if right == left {
1040                unit.single_key()
1041            } else {
1042                stats
1043                    .single_key
1044                    .filter(|&key| unit.single_key() == Some(key))
1045            };
1046            if let Some(neighbors) = split_conflicts.get(right) {
1047                // Conflicts with the split files already in the interval.
1048                stats.conflicts += neighbors.partition_point(|&p| p < right)
1049                    - neighbors.partition_point(|&p| p < left);
1050            }
1051            visit(left..right + 1, &stats);
1052        }
1053    }
1054}

1056/// Whether merging `units` into `outputs` files adds no more conflicts than
1057/// their files have with other files. Outputs stay inside the bounding box of
1058/// the inputs, so each of them may conflict with every file intruding on that
1059/// box. An interval without intruders is closed and adds no conflicts.
1060fn adds_no_conflicts(
1061    units: &[Unit],
1062    extents: &[Extent],
1063    index: &ExtentIndex,
1064    outputs: usize,
1065) -> bool {
1066    let area = units[1..]
1067        .iter()
1068        .fold(units[0].extent.clone(), |area, unit| {
1069            area.union(&unit.extent)
1070        });
1071    let selected: std::collections::HashSet<usize> = units
1072        .iter()
1073        .flat_map(|unit| unit.files.iter().copied())
1074        .collect();
1075    let mut intruders = Vec::new();
1076    index.for_each_near(&area, |i| {
1077        if !selected.contains(&i) && extents[i].intrudes(&area) {
1078            intruders.push(i);
1079        }
1080    });
1081    if intruders.is_empty() {
1082        return true;
1083    }
1084    let existing: usize = intruders
1085        .iter()
1086        .map(|&other| {
1087            units
1088                .iter()
1089                .flat_map(|unit| unit.files.iter())
1090                .filter(|&&i| extents[i].conflicts(&extents[other]))
1091                .count()
1092        })
1093        .sum();
1094    outputs.saturating_mul(intruders.len()) <= existing
1095}

1097/// Whether every group of conflicting files among `units` is balanced, see
1098/// [`has_balanced_runs`]. Runs of separate groups never overlap, so pairing a
1099/// dominant run with the peer of another group would hide it. Takes O(K²)
1100/// conflict checks for K files.
1101fn has_balanced_components(
1102    units: &[Unit],
1103    files: &[FileHandle],
1104    extents: &[Extent],
1105    tolerance: u64,
1106) -> bool {
1107    let members: Vec<usize> = units
1108        .iter()
1109        .flat_map(|unit| unit.files.iter().copied())
1110        .collect();
1111    let mut sets = DisjointSet::new(members.len());
1112    for (i, &lhs) in members.iter().enumerate() {
1113        for (j, &rhs) in members.iter().enumerate().skip(i + 1) {
1114            if extents[lhs].conflicts(&extents[rhs]) {
1115                sets.union(i, j);
1116            }
1117        }
1118    }
1119    sets.groups()
1120        .into_iter()
1121        .filter(|group| group.len() > 1)
1122        .all(|group| {
1123            has_balanced_runs(
1124                group.into_iter().map(|i| members[i]),
1125                files,
1126                extents,
1127                tolerance,
1128            )
1129        })
1130}

1132/// Whether the largest sorted run among `files` holds at most `tolerance` bytes
1133/// more than all other runs together. Runs are built greedily in time order,
1134/// which takes O(K²) conflict checks for K files.
1135fn has_balanced_runs(
1136    members: impl Iterator<Item = usize>,
1137    files: &[FileHandle],
1138    extents: &[Extent],
1139    tolerance: u64,
1140) -> bool {
1141    let mut members: Vec<usize> = members.collect();
1142    members.sort_by(|&lhs, &rhs| {
1143        (extents[lhs].start, Reverse(extents[lhs].end), lhs).cmp(&(
1144            extents[rhs].start,
1145            Reverse(extents[rhs].end),
1146            rhs,
1147        ))
1148    });
1149    let mut runs: Vec<(Vec<usize>, u64)> = Vec::new();
1150    for i in members {
1151        let size = files[i].size();
1152        match runs.iter_mut().find(|(run, _)| {
1153            run.iter()
1154                .all(|&other| !extents[i].conflicts(&extents[other]))
1155        }) {
1156            Some((run, run_size)) => {
1157                run.push(i);
1158                *run_size += size;
1159            }
1160            None => runs.push((vec![i], size)),
1161        }
1162    }
1163    let total: u64 = runs.iter().map(|(_, size)| size).sum();
1164    let largest = runs.iter().map(|(_, size)| *size).max().unwrap_or(0);
1165    largest <= (total - largest).saturating_add(tolerance)
1166}

1177/// Union-find over file indices; each root is the smallest index of its set.
1178struct DisjointSet {
1179    parent: Vec<usize>,
1180}
1181
1182impl DisjointSet {
1183    fn new(len: usize) -> Self {
1184        Self {
1185            parent: (0..len).collect(),
1186        }
1187    }
1188
1189    fn find(&mut self, mut i: usize) -> usize {
1190        while self.parent[i] != i {
1191            self.parent[i] = self.parent[self.parent[i]];
1192            i = self.parent[i];
1193        }
1194        i
1195    }
1196
1197    fn union(&mut self, lhs: usize, rhs: usize) {
1198        let (lhs, rhs) = (self.find(lhs), self.find(rhs));
1199        let (root, child) = (lhs.min(rhs), lhs.max(rhs));
1200        self.parent[child] = root;
1201    }
1202
1203    /// Sets in ascending order of their smallest index, each sorted.
1204    fn groups(&mut self) -> Vec<Vec<usize>> {
1205        let mut groups: Vec<Vec<usize>> = Vec::new();
1206        let mut slot = vec![usize::MAX; self.parent.len()];
1207        for i in 0..self.parent.len() {
1208            let root = self.find(i);
1209            if slot[root] == usize::MAX {
1210                slot[root] = groups.len();
1211                groups.push(Vec::new());
1212            }
1213            groups[slot[root]].push(i);
1214        }
1215        groups
1216    }
1217}

1933    fn sim_file(start: i64, end: i64, min: u64, max: u64, size: u64) -> SimFile {
1934        let (min_key, max_key) = (format!("{min:04}"), format!("{max:04}"));

1936            handle: file(start, end, min_key.as_bytes(), max_key.as_bytes(), size),

1942    /// Writes the outputs of merging `inputs` like the compactor: one stream in
1943    /// (key, time) order, split by size into consecutive files. The data is
1944    /// spread evenly over keys, and a series is never split between files.
1945    fn merge(inputs: &[SimFile], rng: &mut Rng) -> Vec<SimFile> {
1946        let min = inputs.iter().map(|file| file.min).min().unwrap();
1947        let max = inputs.iter().map(|file| file.max).max().unwrap();
1948        let start = inputs
1949            .iter()
1950            .map(|file| file.handle.time_range().0.value())
1951            .min()
1952            .unwrap();
1953        let end = inputs
1954            .iter()
1955            .map(|file| file.handle.time_range().1.value())
1956            .max()
1957            .unwrap();
1958        let size: u64 = inputs.iter().map(|file| file.handle.size()).sum();
1959        // Outputs may be fewer than predicted, never more. A single key is
1960        // never split.
1961        let predicted = if min == max {
1962            1
1963        } else {
1964            size.div_ceil(MAX_OUTPUT).max(1)
1965        };
1966        let keys = max - min + 1;
1967        let outputs = (predicted - rng.next(2).min(predicted - 1)).min(keys);
1968        (0..outputs)
1969            .map(|i| {
1970                let first_key = min + keys * i / outputs;
1971                let last_key = min + keys * (i + 1) / outputs - 1;
1972                let output_size = size * (i + 1) / outputs - size * i / outputs;
1973                sim_file(start, end, first_key, last_key, output_size)
1974            })
1975            .collect()
1976    }

1978    fn conflict_pairs(files: &[FileHandle]) -> usize {
1979        let extents = extents_of(files, &primary_key_mapper_for_test());

1982                (i + 1..extents.len())
1983                    .filter(|&j| extents[i].conflicts(&extents[j]))
1984                    .count()
1985            })
1986            .sum()
1987    }

2068        rng: &mut Rng,
2069        pick: impl Fn(&[FileHandle]) -> Option<UnitPick>,
2070        measure: impl Fn(&[FileHandle]) -> (usize, usize),
2071    ) -> usize {

2075    fn run_windows_to_fixpoint(
2076        rng: &mut Rng,
2077        new_window: impl Fn(&mut Rng) -> Vec<SimFile>,
2078        pick: impl Fn(&[FileHandle]) -> Option<UnitPick>,
2079        measure: impl Fn(&[FileHandle]) -> (usize, usize),
2080    ) -> usize {
2081        let mut total_steps = 0;
2082        for window in 0..300 {
2083            let mut files = new_window(rng);
2084            let handles =
2085                |files: &[SimFile]| files.iter().map(|f| f.handle.clone()).collect::<Vec<_>>();
2086            let mut current = measure(&handles(&files));
2087            let mut steps = 0;
2088            while let Some(pick) = pick(&handles(&files)) {
2089                assert!(pick.inputs.len() >= 2);
2090                let picked = ids(&pick.inputs);
2091                let (inputs, kept): (Vec<_>, Vec<_>) = files
2092                    .into_iter()
2093                    .partition(|file| picked.contains(&file.handle.file_id().file_id()));
2094                files = kept;
2095                files.extend(merge(&inputs, rng));
2096                let next = measure(&handles(&files));
2097                assert!(next < current, "window {window}: {current:?} -> {next:?}");
2098                current = next;
2099                steps += 1;
2100            }
2101            total_steps += steps;
2102        }
2103        total_steps
2104    }
