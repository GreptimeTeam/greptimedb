15//! Unit-based compaction picks for one TWCS window.
16//!
17//! Picks select a contiguous interval of *units* in a fixed order. Grouping
18//! prefers rewriting related files together, but input limits and fallback
19//! paths may split groups or use single-file units. Separate admission checks
20//! preserve each pick's progress guarantees under the assumptions below.
21//!
22//! - [`pick_active_l1`] reduces the L1 file count of the active window.
23//! - [`pick_small_l1`] reduces small L1 files without rewriting large peers.
24//! - [`pick_closed_merge`] only merges intervals of an inactive window whose
25//!   outputs cannot add conflicts. Compaction outputs stay inside the bounding
26//!   box of the inputs, so they can only overlap files intruding on that box.
27//!
28//! When a window's split-unit conflict graph is too dense to materialize,
29//! [`pick_closed_merge`] falls back to a bounded file reduction and admits it
30//! only when the same conflict bound holds.
31//!
32//! # Termination
33//!
34//! Let `P` be the number of conflicting file pairs ([`Extent::conflicts`]) in an
35//! inactive window and `N` its file count. A merge removes every pair among its
36//! inputs. Its outputs are one sorted run with no internal conflicts,
37//! so they only conflict with the `I` files intruding on the inputs' box, at
38//! most `outputs · I` pairs. An inactive pick only merges inputs that had at
39//! least as many pairs with other files, so `P` does not grow; without
40//! intruders the inputs are closed and add no pairs. Every inactive pick either
41//! resolves a pair (`P` drops) or removes a file (`P` does not grow and `N`
42//! drops), so `(P, N)` strictly decreases in lexicographic order until the
43//! window receives new writes. Active-window picks may add conflicts, so each
44//! of them must remove a file: `N` strictly decreases.
45//!
46//! The argument relies on how the compactor writes outputs:
47//!
48//! - Outputs hold a subset of the input rows, so each output's time and key
49//!   bounds lie inside the inputs' bounding box.
50//! - Outputs are one stream in (key, time) order. For regions with a primary
51//!   key, the writer only splits between complete series, so output key ranges
52//!   are disjoint. Without a primary key, outputs have disjoint or touching
53//!   time ranges. Neither is an [`Extent::conflicts`] pair.
54//! - A split only starts a new file once the current one reaches the size
55//!   threshold, and outputs are not larger than inputs in total, so
56//!   [`PickLimits::predicted_outputs`] bounds the output count.
57//!
58//! `test_closed_merges_terminate` and `test_active_l1_picks_terminate` replay
59//! random windows with outputs written under these rules.

70/// An L1 file reduction may rewrite at most this many output files' worth of
71/// bytes per SST it removes.
72const MAX_REWRITE_FILES_PER_REDUCED_FILE: u64 = 4;
73
74/// A small L1 file occupies at most one quarter of the output size budget.
75const SMALL_L1_SIZE_FACTOR: u64 = 4;
76
77/// Maximum number of index nodes visited while building conflict structures for
78/// a pick. Above this, the picker falls back to a bounded file reduction instead
79/// of materializing an unbounded conflict graph.
80const CONFLICT_GRAPH_BUDGET: usize = 1 << 20;
81
82/// Output size limits and input limits shared by both picks.
83#[derive(Debug, Clone, Copy)]
84pub(crate) struct PickLimits {
85    /// Split threshold of compaction outputs; `None` writes a single output.
86    pub(crate) max_output_file_size: Option<u64>,
87    /// Maximum number of input files of one compaction.
88    pub(crate) max_input_files: usize,
89}
90
91impl PickLimits {
92    /// Upper bound of the outputs of compacting an interval: the writer only
93    /// starts a new file after the current one reaches the threshold, and never
94    /// splits a series, so files of a single primary key compact into one file.
95    fn predicted_outputs(&self, stats: &IntervalStats) -> usize {
96        match self.max_output_file_size {
97            Some(max) if max > 0 && stats.single_key.is_none() => {
98                stats.size.div_ceil(max).max(1) as usize
99            }
100            _ => 1,
101        }
102    }
103
104    /// Whether rewriting `size` bytes to remove `reduction` files stays within
105    /// the rewrite budget. An unset output size imposes no bound.
106    fn within_reduction_cost(&self, size: u64, reduction: usize) -> bool {
107        match self.max_output_file_size {
108            Some(max) if max > 0 => {
109                size <= (reduction as u64)
110                    .saturating_mul(MAX_REWRITE_FILES_PER_REDUCED_FILE)
111                    .saturating_mul(max)
112            }
113            _ => true,
114        }
115    }
116
117    /// Bytes a dominant sorted run may exceed its peers by in an overlap merge.
118    fn run_tolerance(&self) -> u64 {
119        self.max_output_file_size.unwrap_or(0)
120    }
121}

136/// Picks L1 files of the active window whose compaction removes files.
137///
138/// Units are groups of conflicting files, so a merge does not leave a file
139/// behind while rewriting its overlapping peers. A group larger than the input
140/// limit is split back into single files. If no interval of groups qualifies,
141/// for example because a large file dominates its group, the groups are split
142/// into single files, so the smaller files can still be merged. Intervals
143/// follow [`sort_units`]; the pick removes the most files, then rewrites the
144/// fewest bytes.
145pub(crate) fn pick_active_l1(
146    l1_files: &[FileHandle],
147    mapper: &PrimaryKeyRangeMapper,
148    options: PickLimits,
149) -> Option<UnitPick> {
150    pick_active_l1_with_budget(l1_files, mapper, options, CONFLICT_GRAPH_BUDGET)
151}
152
153/// [`pick_active_l1`] with an explicit work budget, so tests can exercise the
154/// single-file fallback without a window-sized conflict graph.
155fn pick_active_l1_with_budget(
156    l1_files: &[FileHandle],
157    mapper: &PrimaryKeyRangeMapper,
158    options: PickLimits,
159    work_budget: usize,
160) -> Option<UnitPick> {
161    let extents = extents_of(l1_files, mapper);
162    let index = ExtentIndex::new(&extents);
163    let mut units = match conflict_units(l1_files, &extents, &index, work_budget) {
164        Some(units) => units,
165        None => single_file_units(l1_files, &extents),
166    };
167    split_oversized_units(&mut units, l1_files, &extents, options.max_input_files);
168    if let Some(pick) = pick_reducing_interval(&mut units, l1_files, options) {
169        return Some(pick);
170    }
171    if units.len() == l1_files.len() {
172        return None;
173    }
174    split_oversized_units(&mut units, l1_files, &extents, 1);
175    pick_reducing_interval(&mut units, l1_files, options)
176}

178/// Picks the interval of `units` that removes the most files within the
179/// rewrite budget, then rewrites the fewest bytes.
180fn pick_reducing_interval(
181    units: &mut [Unit],
182    files: &[FileHandle],
183    options: PickLimits,
184) -> Option<UnitPick> {
185    sort_units(units);
186    let range = pick_reducing_interval_where(units, options, |_| true)?;
187    Some(UnitPick {
188        inputs: collect_inputs(&units[range], files),
189        reduces_files: true,
190        units: units.len(),
191        dense_fallback: false,
192    })
193}

195/// Selects a reducing interval of sorted units, with an additional filter so
196/// the dense fallback can retry without duplicating the reduction guards.
197fn pick_reducing_interval_where(
198    units: &[Unit],
199    options: PickLimits,
200    is_eligible: impl Fn(&IntervalStats) -> bool,
201) -> Option<Range<usize>> {
202    let mut best: Option<(usize, u64, Range<usize>)> = None;
203    // File reductions ignore conflicts, so split files are not counted.
204    for_each_interval(units, &[], options.max_input_files, |range, stats| {
205        let reduction = stats
206            .num_files
207            .saturating_sub(options.predicted_outputs(stats));
208        if stats.num_files < 2
209            || reduction == 0
210            || !options.within_reduction_cost(stats.size, reduction)
211            || stats.largest_file_size > stats.size - stats.largest_file_size
212            || !is_eligible(stats)
213        {
214            return;
215        }
216        let better = best.as_ref().is_none_or(|(best_reduction, best_size, _)| {
217            (reduction, Reverse(stats.size)) > (*best_reduction, Reverse(*best_size))
218        });
219        if better {
220            best = Some((reduction, stats.size, range));
221        }
222    });
223
224    best.map(|(_, _, range)| range)
225}

227/// Reduces small L1 files left behind by a closed merge, including in settling
228/// windows. The input bytes fit in one output and no input dominates its peers,
229/// so each merge removes files without repeatedly rewriting an accumulator.
230///
231/// Large and L0 files remain in the external conflict index. Check only the
232/// best reducing interval to bound planning work; rejecting it can miss other
233/// admissible intervals, but cannot increase the window's conflict count.
234pub(crate) fn pick_small_l1(
235    files: &[FileHandle],
236    mapper: &PrimaryKeyRangeMapper,
237    limits: PickLimits,
238) -> Option<UnitPick> {
239    let max_output = limits.max_output_file_size.filter(|size| *size > 0)?;
240    let extents = extents_of(files, mapper);
241    let mut units = single_file_units(files, &extents);
242    units.retain(|unit| {
243        let file = &files[unit.files[0]];
244        file.level() == 1 && file.size() <= max_output / SMALL_L1_SIZE_FACTOR
245    });
246    sort_units(&mut units);
247    let range = pick_reducing_interval_where(&units, limits, |stats| stats.size <= max_output)?;
248    let index = ExtentIndex::new(&extents);
249    if !adds_no_conflicts(&units[range.clone()], &extents, &index, 1) {
250        return None;
251    }
252    Some(UnitPick {
253        inputs: collect_inputs(&units[range], files),
254        reduces_files: true,
255        units: units.len(),
256        dense_fallback: false,
257    })
258}

275/// Picks an interval of units among all files of an inactive window.
276///
277/// Units are the closures of conflicting files: no file outside a unit intrudes
278/// on its bounding box. A closure larger than the input limit is split back into
279/// single files, so files around a late file spanning a large sorted run can
280/// still be merged. Intervals follow [`sort_units`]. An interval is eligible if
281/// its outputs add no more conflicts than its files have with other files (see
282/// [`adds_no_conflicts`]), and it
283///
284/// - removes files within the rewrite budget, or
285/// - resolves conflicts without rewriting a dominant sorted run (unless
286///   [`ClosedMergeOptions::allow_unbalanced`]), and adds at most one tail file.
287///
288/// The pick maximizes resolved conflicts plus removed files per rewritten byte.
289/// A window whose split-unit conflict graph is too dense to materialize falls
290/// back to a bounded file reduction; see [`pick_dense_closed_merge`].
291pub(crate) fn pick_closed_merge(
292    files: &[FileHandle],
293    mapper: &PrimaryKeyRangeMapper,
294    options: ClosedMergeOptions,
295) -> Option<UnitPick> {
296    pick_closed_merge_with_budget(files, mapper, options, CONFLICT_GRAPH_BUDGET)
297}

299/// [`pick_closed_merge`] with an explicit work budget, so tests can exercise
300/// the dense fallback without building a window-sized conflict graph.
301fn pick_closed_merge_with_budget(
302    files: &[FileHandle],
303    mapper: &PrimaryKeyRangeMapper,
304    options: ClosedMergeOptions,
305    work_budget: usize,
306) -> Option<UnitPick> {
307    let limits = options.limits;
308    let extents = extents_of(files, mapper);
309    let index = ExtentIndex::new(&extents);
310    let mut dense = false;
311    let mut units = match conflict_units(files, &extents, &index, work_budget) {
312        Some(mut units) => {
313            close_units(&mut units, &extents, &index);
314            split_oversized_units(&mut units, files, &extents, limits.max_input_files);
315            units
316        }
317        None => {
318            dense = true;
319            single_file_units(files, &extents)
320        }
321    };
322    sort_units(&mut units);
323    if dense {
324        return pick_dense_closed_merge(&units, files, &extents, &index, options);
325    }
326    let Some(pairs) = split_conflicts(&units, &extents, &index, work_budget) else {
327        return pick_dense_closed_merge(&units, files, &extents, &index, options);
328    };
329
330    let mut best: Option<(u64, u64, Range<usize>, bool)> = None;
331    for_each_interval(&units, &pairs, limits.max_input_files, |range, stats| {
332        if stats.num_files < 2 {
333            return;
334        }
335        let outputs = limits.predicted_outputs(stats);
336        if outputs > stats.num_files + 1 {
337            return;
338        }
339        let reduction = stats.num_files.saturating_sub(outputs);
340        let reduces_files = reduction > 0
341            && (options.allow_file_reduction || stats.conflicts > 0)
342            && limits.within_reduction_cost(stats.size, reduction);
343        if (stats.conflicts > 0 && !options.allow_conflicts)
344            || (!reduces_files && stats.conflicts == 0)
345        {
346            return;
347        }
348        let benefit = (reduction + stats.conflicts) as u64;
349        let better = best.as_ref().is_none_or(|(best_benefit, best_size, _, _)| {
350            is_more_efficient(benefit, stats.size, *best_benefit, *best_size)
351        });
352        if !better {
353            return;
354        }
355        let units = &units[range.clone()];
356        // Overlap merges must not rewrite a dominant run for a small peer. A
357        // file-reducing merge is already bounded by its rewrite budget.
358        if !reduces_files
359            && !options.allow_unbalanced
360            && !has_balanced_components(units, files, &extents, limits.run_tolerance())
361        {
362            return;
363        }
364        if units.len() > 1 && !adds_no_conflicts(units, &extents, &index, outputs) {
365            return;
366        }
367        best = Some((benefit, stats.size, range, reduces_files));
368    });
369
370    best.map(|(_, _, range, reduces_files)| UnitPick {
371        inputs: collect_inputs(&units[range], files),
372        reduces_files,
373        units: units.len(),
374        dense_fallback: false,
375    })
376}

378/// Picks a bounded file reduction when the split-unit conflict graph is too
379/// dense to materialize.
380///
381/// The candidate comes from [`pick_reducing_interval_where`], which ignores
382/// conflicts. If rejected, it is searched once more for a one-output interval.
383/// Both candidates must pass [`adds_no_conflicts`], so the `(P, N)` termination
384/// argument still holds. Searching only within the original candidate adds at
385/// most O(K²) interval visits for K input files and one extra conflict check,
386/// rather than checking external conflicts for every interval in the window.
387/// This bounded retry can still miss other safely reducible intervals.
388fn pick_dense_closed_merge(
389    units: &[Unit],
390    files: &[FileHandle],
391    extents: &[Extent],
392    index: &ExtentIndex,
393    options: ClosedMergeOptions,
394) -> Option<UnitPick> {
395    let limits = options.limits;
396    let range = pick_reducing_interval_where(units, limits, |_| true)?;
397    let mut selected = &units[range];
398    if !is_admissible_dense_interval(selected, extents, index, options) {
399        let range = pick_reducing_interval_where(selected, limits, |stats| {
400            limits.predicted_outputs(stats) == 1
401        })?;
402        selected = &selected[range];
403        if !is_admissible_dense_interval(selected, extents, index, options) {
404            return None;
405        }
406    }
407    Some(UnitPick {
408        inputs: collect_inputs(selected, files),
409        reduces_files: true,
410        units: units.len(),
411        dense_fallback: true,
412    })
413}

415/// File reduction and rewrite limits are checked by the interval picker;
416/// each dense candidate must also respect the closed-window conflict policy.
417fn is_admissible_dense_interval(
418    units: &[Unit],
419    extents: &[Extent],
420    index: &ExtentIndex,
421    options: ClosedMergeOptions,
422) -> bool {
423    let stats = interval_stats_exact(units, extents);
424    (options.allow_file_reduction || stats.conflicts > 0)
425        && (options.allow_conflicts || stats.conflicts == 0)
426        && adds_no_conflicts(
427            units,
428            extents,
429            index,
430            options.limits.predicted_outputs(&stats),
431        )
432}

434/// Totals of an interval without materializing [`split_conflicts`].
435///
436/// After [`close_units`], files of different units never conflict, so only
437/// split units can conflict across units, and each split unit holds one file.
438/// The dense fallback may instead treat every file as a split unit, and then
439/// the same pairwise loop covers every cross-unit pair. Counting them takes
440/// `O(K²)` for an interval of at most `K` files, instead of building a
441/// window-wide adjacency.
442fn interval_stats_exact<'a>(units: &'a [Unit], extents: &[Extent]) -> IntervalStats<'a> {
443    let mut stats = IntervalStats::default();
444    let mut first = true;
445    for unit in units {
446        stats.num_files += unit.files.len();
447        stats.size += unit.size;
448        stats.largest_file_size = stats.largest_file_size.max(unit.largest_file_size);
449        stats.conflicts += unit.conflicts;
450        stats.single_key = if first {
451            unit.single_key()
452        } else {
453            stats
454                .single_key
455                .filter(|&key| unit.single_key() == Some(key))
456        };
457        first = false;
458    }
459
460    let split_files: Vec<usize> = units
461        .iter()
462        .filter(|unit| unit.split)
463        .map(|unit| unit.files[0])
464        .collect();
465    for (offset, &lhs) in split_files.iter().enumerate() {
466        for &rhs in &split_files[offset + 1..] {
467            if extents[lhs].conflicts(&extents[rhs]) {
468                stats.conflicts += 1;
469            }
470        }
471    }
472    stats
473}

475/// Orders units by minimum primary key, then by time, so adjacent units are
476/// key-adjacent. Compaction outputs and time-sliced flushes split a window by
477/// key ranges, so merging neighbors in this order keeps outputs from covering
478/// the key ranges of files left behind.
479fn sort_units(units: &mut [Unit]) {
480    units.sort_by(|lhs, rhs| {
481        (lhs.first_key(), lhs.extent.start, Reverse(lhs.extent.end))
482            .cmp(&(rhs.first_key(), rhs.extent.start, Reverse(rhs.extent.end)))
483            .then_with(|| lhs.files[0].cmp(&rhs.files[0]))
484    });
485}

487/// Whether `benefit` per `size` bytes beats the best so far; ties prefer fewer
488/// bytes, then the earlier interval.
489fn is_more_efficient(benefit: u64, size: u64, best_benefit: u64, best_size: u64) -> bool {
490    let lhs = u128::from(benefit) * u128::from(best_size.max(1));
491    let rhs = u128::from(best_benefit) * u128::from(size.max(1));
492    lhs > rhs || (lhs == rhs && size < best_size)
493}

495/// Time and primary key bounds of a file or a group of files.
496#[derive(Debug, Clone)]
497pub(crate) struct Extent {
498    start: Timestamp,
499    end: Timestamp,
500    /// Inclusive primary key bounds. `None` is unknown and matches any key.
501    primary_key: Option<(Bytes, Bytes)>,
502}
503
504impl Extent {
505    pub(crate) fn of_file(file: &FileHandle, mapper: &PrimaryKeyRangeMapper) -> Self {
506        let (start, end) = file.time_range();
507        Self {
508            start,
509            end,
510            primary_key: file.primary_key_range(mapper),
511        }
512    }
513
514    fn union(&self, other: &Self) -> Self {
515        Self {
516            start: self.start.min(other.start),
517            end: self.end.max(other.end),
518            primary_key: match (&self.primary_key, &other.primary_key) {
519                (Some((lhs_min, lhs_max)), Some((rhs_min, rhs_max))) => Some((
520                    lhs_min.clone().min(rhs_min.clone()),
521                    lhs_max.clone().max(rhs_max.clone()),
522                )),
523                _ => None,
524            },
525        }
526    }
527
528    /// Time ranges overlap; touching boundaries do not.
529    fn time_overlaps(&self, other: &Self) -> bool {
530        self.start.max(other.start) < self.end.min(other.end)
531    }
532
533    /// Whether two files overlap for the picker.
534    ///
535    /// Shared boundary keys count: independently written files can hold
536    /// different versions of the same row there. Compaction outputs do not
537    /// conflict with each other because the writer never splits a series.
538    pub(crate) fn conflicts(&self, other: &Self) -> bool {
539        self.time_overlaps(other)
540            && match (&self.primary_key, &other.primary_key) {
541                (Some(lhs), Some(rhs)) => {
542                    let (low, high) = ((&lhs.0).max(&rhs.0), (&lhs.1).min(&rhs.1));
543                    low <= high
544                }
545                _ => true,
546            }
547    }
548
549    /// Whether `self` may conflict with a file inside `area`, which is the
550    /// bounding box of a group of files.
551    ///
552    /// A shared boundary key may occur in an output, so the same inclusive
553    /// key overlap used for individual files also applies to the box.
554    fn intrudes(&self, area: &Self) -> bool {
555        self.conflicts(area)
556    }
557}

563/// Upper primary key bound, ordered after every key when unknown.
564#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
565enum UpperKey<'a> {
566    Key(&'a Bytes),
567    Unbounded,
568}

570/// Key and time bounds of the files in a subtree of [`ExtentIndex`].
571#[derive(Debug, Clone, Copy)]
572struct SubtreeBounds<'a> {
573    max_upper_key: UpperKey<'a>,
574    min_start: Timestamp,
575    max_end: Timestamp,
576}
577
578impl<'a> SubtreeBounds<'a> {
579    fn of(extent: &'a Extent) -> Self {
580        Self {
581            max_upper_key: match &extent.primary_key {
582                Some((_, max)) => UpperKey::Key(max),
583                None => UpperKey::Unbounded,
584            },
585            min_start: extent.start,
586            max_end: extent.end,
587        }
588    }
589
590    fn merge(self, other: Self) -> Self {
591        Self {
592            max_upper_key: self.max_upper_key.max(other.max_upper_key),
593            min_start: self.min_start.min(other.min_start),
594            max_end: self.max_end.max(other.max_end),
595        }
596    }
597}

599/// Files of a window ordered by their lower primary key, in a segment tree
600/// holding the bounds of each subtree.
601///
602/// A query visits the files whose key and time ranges may overlap an area and
603/// skips subtrees whose keys end before the area or whose times miss it. The
604/// files of a sorted run split by key then only meet their key neighbors,
605/// instead of every file sharing their time range.
606struct ExtentIndex<'a> {
607    extents: &'a [Extent],
608    /// Indices into `extents`, sorted by lower key with unknown keys first.
609    /// Sorted offset `i` is leaf `leaf_base + i`.
610    by_lower_key: Vec<usize>,
611    /// Array-backed tree rooted at index 1; padding leaves are `None`.
612    bounds: Vec<Option<SubtreeBounds<'a>>>,
613    /// First leaf index, a power of two.
614    leaf_base: usize,
615}
616
617impl<'a> ExtentIndex<'a> {
618    fn new(extents: &'a [Extent]) -> Self {
619        let lower_key = |i: usize| extents[i].primary_key.as_ref().map(|(min, _)| min);
620        let mut by_lower_key: Vec<usize> = (0..extents.len()).collect();
621        by_lower_key.sort_by(|&lhs, &rhs| {
622            (lower_key(lhs), extents[lhs].start).cmp(&(lower_key(rhs), extents[rhs].start))
623        });
624        let leaf_base = extents.len().next_power_of_two();
625        let mut bounds = vec![None; 2 * leaf_base];
626        for (offset, &i) in by_lower_key.iter().enumerate() {
627            bounds[leaf_base + offset] = Some(SubtreeBounds::of(&extents[i]));
628        }
629        for node in (1..leaf_base).rev() {
630            bounds[node] = match (bounds[2 * node], bounds[2 * node + 1]) {
631                (Some(lhs), Some(rhs)) => Some(lhs.merge(rhs)),
632                (lhs, rhs) => lhs.or(rhs),
633            };
634        }
635        Self {
636            extents,
637            by_lower_key,
638            bounds,
639            leaf_base,
640        }
641    }
642
643    /// Visits the files whose inclusive key range meets `area`'s and whose time
644    /// range overlaps it, a superset of the files that conflict with or intrude
645    /// on `area`. Takes O(log N) per visited file.
646    fn for_each_near(&self, area: &Extent, visit: impl FnMut(usize)) {
647        let mut budget = usize::MAX;
648        let _ = self.for_each_near_with_budget(area, &mut budget, visit);
649    }
650
651    /// [`for_each_near`] with a budget on visited tree nodes.
652    ///
653    /// Every node reached by the traversal consumes one unit, including nodes
654    /// pruned before any file callback. Returns `false` as soon as the budget is
655    /// exhausted, so a caller can stop a traversal whose pruning is poor even
656    /// when it finds few files.
657    fn for_each_near_with_budget(
658        &self,
659        area: &Extent,
660        budget: &mut usize,
661        mut visit: impl FnMut(usize),
662    ) -> bool {
663        let leaves_end = match &area.primary_key {
664            Some((_, max)) => self.by_lower_key.partition_point(|&i| {
665                self.extents[i]
666                    .primary_key
667                    .as_ref()
668                    .is_none_or(|(min, _)| min <= max)
669            }),
670            None => self.by_lower_key.len(),
671        };
672        self.visit_subtree(1, 0..self.leaf_base, leaves_end, area, budget, &mut visit)
673    }
674
675    fn visit_subtree(
676        &self,
677        node: usize,
678        leaves: Range<usize>,
679        leaves_end: usize,
680        area: &Extent,
681        budget: &mut usize,
682        visit: &mut impl FnMut(usize),
683    ) -> bool {
684        if *budget == 0 {
685            return false;
686        }
687        *budget -= 1;
688        if leaves.start >= leaves_end {
689            return true;
690        }
691        let Some(bounds) = self.bounds[node] else {
692            return true;
693        };
694        if bounds.min_start >= area.end
695            || bounds.max_end <= area.start
696            || area
697                .primary_key
698                .as_ref()
699                .is_some_and(|(min, _)| bounds.max_upper_key < UpperKey::Key(min))
700        {
701            return true;
702        }
703        if leaves.len() == 1 {
704            let i = self.by_lower_key[leaves.start];
705            if self.extents[i].time_overlaps(area) {
706                visit(i);
707            }
708            return true;
709        }
710        let mid = leaves.start + leaves.len() / 2;
711        if !self.visit_subtree(2 * node, leaves.start..mid, leaves_end, area, budget, visit) {
712            return false;
713        }
714        self.visit_subtree(
715            2 * node + 1,
716            mid..leaves.end,
717            leaves_end,
718            area,
719            budget,
720            visit,
721        )
722    }
723}

732/// Files that are compacted together.
733#[derive(Debug)]
734struct Unit {
735    /// Ascending indices into the candidate files.
736    files: Vec<usize>,
737    extent: Extent,
738    size: u64,
739    largest_file_size: u64,
740    /// Number of conflicting file pairs inside the unit.
741    conflicts: usize,
742    /// A single file split from a unit over the input limit. It may conflict
743    /// with other split files, see [`split_conflicts`].
744    split: bool,
745}
746
747impl Unit {
748    fn new(
749        members: Vec<usize>,
750        files: &[FileHandle],
751        extents: &[Extent],
752        conflicts: usize,
753    ) -> Self {
754        let extent = members[1..]
755            .iter()
756            .fold(extents[members[0]].clone(), |extent, &i| {
757                extent.union(&extents[i])
758            });
759        let sizes = members.iter().map(|&i| files[i].size());
760        Self {
761            size: sizes.clone().sum(),
762            largest_file_size: sizes.max().unwrap_or(0),
763            files: members,
764            extent,
765            conflicts,
766            split: false,
767        }
768    }
769
770    /// The minimum primary key, with unknown keys ordered first.
771    fn first_key(&self) -> Option<&Bytes> {
772        self.extent.primary_key.as_ref().map(|(min, _)| min)
773    }
774
775    /// The primary key of a unit whose files all hold that single key.
776    fn single_key(&self) -> Option<&Bytes> {
777        self.extent
778            .primary_key
779            .as_ref()
780            .filter(|range| is_single_key(range))
781            .map(|(min, _)| min)
782    }
783}

785/// Groups files into connected components of [`Extent::conflicts`], each with
786/// its number of conflicting pairs. Each file only checks the files the index
787/// finds near it: O((N + M) log N) for M pairs of files whose key and time
788/// ranges meet.
789///
790/// Returns `None` once the index traversal has visited more than `work_budget`
791/// nodes, so a dense window can fall back before building a quadratic graph.
792fn conflict_units(
793    files: &[FileHandle],
794    extents: &[Extent],
795    index: &ExtentIndex,
796    work_budget: usize,
797) -> Option<Vec<Unit>> {
798    let mut sets = DisjointSet::new(extents.len());
799    let mut conflicts = vec![0; extents.len()];
800    let mut budget = work_budget;
801    for i in 0..extents.len() {
802        let completed = index.for_each_near_with_budget(&extents[i], &mut budget, |j| {
803            // Counts each pair once, from its smaller index.
804            if j > i && extents[i].conflicts(&extents[j]) {
805                sets.union(i, j);
806                conflicts[i] += 1;
807            }
808        });
809        if !completed {
810            return None;
811        }
812    }
813    Some(
814        sets.groups()
815            .into_iter()
816            .map(|members| {
817                let pairs = members.iter().map(|&i| conflicts[i]).sum();
818                Unit::new(members, files, extents, pairs)
819            })
820            .collect(),
821    )
822}

858/// Merges units until no file intrudes on a unit it does not belong to.
859///
860/// Each check only visits the files the index finds near the unit, and each
861/// merge re-checks only the merged unit.
862fn close_units(units: &mut Vec<Unit>, extents: &[Extent], index: &ExtentIndex) {
863    let mut owner = vec![0; extents.len()];
864    for (u, unit) in units.iter().enumerate() {
865        for &i in &unit.files {
866            owner[i] = u;
867        }
868    }
869    let mut alive = vec![true; units.len()];
870    let mut pending: Vec<usize> = (0..units.len()).collect();
871    while let Some(u) = pending.pop() {
872        if !alive[u] {
873            continue;
874        }
875        let area = &units[u].extent;
876        let mut intruders = std::collections::BTreeSet::new();
877        index.for_each_near(area, |i| {
878            if owner[i] != u && extents[i].intrudes(area) {
879                intruders.insert(owner[i]);
880            }
881        });
882        if intruders.is_empty() {
883            continue;
884        }
885        for v in intruders {
886            alive[v] = false;
887            let absorbed = std::mem::take(&mut units[v].files);
888            for &i in &absorbed {
889                owner[i] = u;
890            }
891            let (extent, size, largest, conflicts) = {
892                let other = &units[v];
893                (
894                    other.extent.clone(),
895                    other.size,
896                    other.largest_file_size,
897                    other.conflicts,
898                )
899            };
900            let unit = &mut units[u];
901            unit.files.extend(absorbed);
902            unit.extent = unit.extent.union(&extent);
903            unit.size += size;
904            unit.largest_file_size = unit.largest_file_size.max(largest);
905            unit.conflicts += conflicts;
906        }
907        pending.push(u);
908    }
909    let mut closed = Vec::with_capacity(units.len());
910    for (u, mut unit) in std::mem::take(units).into_iter().enumerate() {
911        if alive[u] {
912            unit.files.sort_unstable();
913            closed.push(unit);
914        }
915    }
916    *units = closed;
917}

919/// Totals of a contiguous interval of units.
920#[derive(Debug, Default)]
921struct IntervalStats<'a> {
922    num_files: usize,
923    size: u64,
924    largest_file_size: u64,
925    conflicts: usize,
926    /// The primary key of an interval whose files all hold that single key.
927    single_key: Option<&'a Bytes>,
928}

930/// Positions of the split units each unit conflicts with, sorted. Files of
931/// different units never conflict, so only split files have any.
932///
933/// Returns `None` once the index traversal has visited more than `work_budget`
934/// nodes, so a dense window cannot materialize an unbounded adjacency.
935fn split_conflicts(
936    units: &[Unit],
937    extents: &[Extent],
938    index: &ExtentIndex,
939    work_budget: usize,
940) -> Option<Vec<Vec<usize>>> {
941    let mut position = vec![None; extents.len()];
942    for (p, unit) in units.iter().enumerate() {
943        if unit.split {
944            position[unit.files[0]] = Some(p);
945        }
946    }
947    let mut budget = work_budget;
948    let mut result = Vec::with_capacity(units.len());
949    for unit in units {
950        if !unit.split {
951            result.push(Vec::new());
952            continue;
953        }
954        let i = unit.files[0];
955        let mut neighbors = Vec::new();
956        let completed = index.for_each_near_with_budget(&extents[i], &mut budget, |j| {
957            if let Some(p) = position[j]
958                && j != i
959                && extents[i].conflicts(&extents[j])
960            {
961                neighbors.push(p);
962            }
963        });
964        if !completed {
965            return None;
966        }
967        neighbors.sort_unstable();
968        result.push(neighbors);
969    }
970    Some(result)
971}

973/// Visits every contiguous interval of units with at most `max_input_files`
974/// files: O(U · K) intervals for an input limit K. Each interval counts the
975/// conflicts of its split files from `split_conflicts` (see
976/// [`split_conflicts`]) with a binary search; an empty `split_conflicts`
977/// skips them.
978fn for_each_interval<'a>(
979    units: &'a [Unit],
980    split_conflicts: &[Vec<usize>],
981    max_input_files: usize,
982    mut visit: impl FnMut(Range<usize>, &IntervalStats<'a>),
983) {
984    for left in 0..units.len() {
985        let mut stats = IntervalStats::default();
986        for (right, unit) in units.iter().enumerate().skip(left) {
987            stats.num_files += unit.files.len();
988            if stats.num_files > max_input_files {
989                break;
990            }
991            stats.size += unit.size;
992            stats.largest_file_size = stats.largest_file_size.max(unit.largest_file_size);
993            stats.conflicts += unit.conflicts;
994            stats.single_key = if right == left {
995                unit.single_key()
996            } else {
997                stats
998                    .single_key
999                    .filter(|&key| unit.single_key() == Some(key))
1000            };
1001            if let Some(neighbors) = split_conflicts.get(right) {
1002                // Conflicts with the split files already in the interval.
1003                stats.conflicts += neighbors.partition_point(|&p| p < right)
1004                    - neighbors.partition_point(|&p| p < left);
1005            }
1006            visit(left..right + 1, &stats);
1007        }
1008    }
1009}

1011/// Whether merging `units` into `outputs` files adds no more conflicts than
1012/// their files have with other files. Outputs stay inside the bounding box of
1013/// the inputs, so each of them may conflict with every file intruding on that
1014/// box. An interval without intruders is closed and adds no conflicts.
1015fn adds_no_conflicts(
1016    units: &[Unit],
1017    extents: &[Extent],
1018    index: &ExtentIndex,
1019    outputs: usize,
1020) -> bool {
1021    let area = units[1..]
1022        .iter()
1023        .fold(units[0].extent.clone(), |area, unit| {
1024            area.union(&unit.extent)
1025        });
1026    let selected: std::collections::HashSet<usize> = units
1027        .iter()
1028        .flat_map(|unit| unit.files.iter().copied())
1029        .collect();
1030    let mut intruders = Vec::new();
1031    index.for_each_near(&area, |i| {
1032        if !selected.contains(&i) && extents[i].intrudes(&area) {
1033            intruders.push(i);
1034        }
1035    });
1036    if intruders.is_empty() {
1037        return true;
1038    }
1039    let existing: usize = intruders
1040        .iter()
1041        .map(|&other| {
1042            units
1043                .iter()
1044                .flat_map(|unit| unit.files.iter())
1045                .filter(|&&i| extents[i].conflicts(&extents[other]))
1046                .count()
1047        })
1048        .sum();
1049    outputs.saturating_mul(intruders.len()) <= existing
1050}

1052/// Whether every group of conflicting files among `units` is balanced, see
1053/// [`has_balanced_runs`]. Runs of separate groups never overlap, so pairing a
1054/// dominant run with the peer of another group would hide it. Takes O(K²)
1055/// conflict checks for K files.
1056fn has_balanced_components(
1057    units: &[Unit],
1058    files: &[FileHandle],
1059    extents: &[Extent],
1060    tolerance: u64,
1061) -> bool {
1062    let members: Vec<usize> = units
1063        .iter()
1064        .flat_map(|unit| unit.files.iter().copied())
1065        .collect();
1066    let mut sets = DisjointSet::new(members.len());
1067    for (i, &lhs) in members.iter().enumerate() {
1068        for (j, &rhs) in members.iter().enumerate().skip(i + 1) {
1069            if extents[lhs].conflicts(&extents[rhs]) {
1070                sets.union(i, j);
1071            }
1072        }
1073    }
1074    sets.groups()
1075        .into_iter()
1076        .filter(|group| group.len() > 1)
1077        .all(|group| {
1078            has_balanced_runs(
1079                group.into_iter().map(|i| members[i]),
1080                files,
1081                extents,
1082                tolerance,
1083            )
1084        })
1085}

1087/// Whether the largest sorted run among `files` holds at most `tolerance` bytes
1088/// more than all other runs together. Runs are built greedily in time order,
1089/// which takes O(K²) conflict checks for K files.
1090fn has_balanced_runs(
1091    members: impl Iterator<Item = usize>,
1092    files: &[FileHandle],
1093    extents: &[Extent],
1094    tolerance: u64,
1095) -> bool {
1096    let mut members: Vec<usize> = members.collect();
1097    members.sort_by(|&lhs, &rhs| {
1098        (extents[lhs].start, Reverse(extents[lhs].end), lhs).cmp(&(
1099            extents[rhs].start,
1100            Reverse(extents[rhs].end),
1101            rhs,
1102        ))
1103    });
1104    let mut runs: Vec<(Vec<usize>, u64)> = Vec::new();
1105    for i in members {
1106        let size = files[i].size();
1107        match runs.iter_mut().find(|(run, _)| {
1108            run.iter()
1109                .all(|&other| !extents[i].conflicts(&extents[other]))
1110        }) {
1111            Some((run, run_size)) => {
1112                run.push(i);
1113                *run_size += size;
1114            }
1115            None => runs.push((vec![i], size)),
1116        }
1117    }
1118    let total: u64 = runs.iter().map(|(_, size)| size).sum();
1119    let largest = runs.iter().map(|(_, size)| *size).max().unwrap_or(0);
1120    largest <= (total - largest).saturating_add(tolerance)
1121}

1132/// Union-find over file indices; each root is the smallest index of its set.
1133struct DisjointSet {
1134    parent: Vec<usize>,
1135}
1136
1137impl DisjointSet {
1138    fn new(len: usize) -> Self {
1139        Self {
1140            parent: (0..len).collect(),
1141        }
1142    }
1143
1144    fn find(&mut self, mut i: usize) -> usize {
1145        while self.parent[i] != i {
1146            self.parent[i] = self.parent[self.parent[i]];
1147            i = self.parent[i];
1148        }
1149        i
1150    }
1151
1152    fn union(&mut self, lhs: usize, rhs: usize) {
1153        let (lhs, rhs) = (self.find(lhs), self.find(rhs));
1154        let (root, child) = (lhs.min(rhs), lhs.max(rhs));
1155        self.parent[child] = root;
1156    }
1157
1158    /// Sets in ascending order of their smallest index, each sorted.
1159    fn groups(&mut self) -> Vec<Vec<usize>> {
1160        let mut groups: Vec<Vec<usize>> = Vec::new();
1161        let mut slot = vec![usize::MAX; self.parent.len()];
1162        for i in 0..self.parent.len() {
1163            let root = self.find(i);
1164            if slot[root] == usize::MAX {
1165                slot[root] = groups.len();
1166                groups.push(Vec::new());
1167            }
1168            groups[slot[root]].push(i);
1169        }
1170        groups
1171    }
1172}

1813    fn sim_file(start: i64, end: i64, min: u64, max: u64, size: u64) -> SimFile {
1814        let (min_key, max_key) = (format!("{min:04}"), format!("{max:04}"));
1815        SimFile {
1816            handle: file(start, end, min_key.as_bytes(), max_key.as_bytes(), size),
1817            min,
1818            max,
1819        }
1820    }

1822    /// Writes the outputs of merging `inputs` like the compactor: one stream in
1823    /// (key, time) order, split by size into consecutive files. The data is
1824    /// spread evenly over keys, and a series is never split between files.
1825    fn merge(inputs: &[SimFile], rng: &mut Rng) -> Vec<SimFile> {
1826        let min = inputs.iter().map(|file| file.min).min().unwrap();
1827        let max = inputs.iter().map(|file| file.max).max().unwrap();
1828        let start = inputs
1829            .iter()
1830            .map(|file| file.handle.time_range().0.value())
1831            .min()
1832            .unwrap();
1833        let end = inputs
1834            .iter()
1835            .map(|file| file.handle.time_range().1.value())
1836            .max()
1837            .unwrap();
1838        let size: u64 = inputs.iter().map(|file| file.handle.size()).sum();
1839        // Outputs may be fewer than predicted, never more. A single key is
1840        // never split.
1841        let predicted = if min == max {
1842            1
1843        } else {
1844            size.div_ceil(MAX_OUTPUT).max(1)
1845        };
1846        let keys = max - min + 1;
1847        let outputs = (predicted - rng.next(2).min(predicted - 1)).min(keys);
1848        (0..outputs)
1849            .map(|i| {
1850                let first_key = min + keys * i / outputs;
1851                let last_key = min + keys * (i + 1) / outputs - 1;
1852                let output_size = size * (i + 1) / outputs - size * i / outputs;
1853                sim_file(start, end, first_key, last_key, output_size)
1854            })
1855            .collect()
1856    }

1858    fn conflict_pairs(files: &[FileHandle]) -> usize {
1859        let extents = extents_of(files, &primary_key_mapper_for_test());
1860        (0..extents.len())
1861            .map(|i| {
1862                (i + 1..extents.len())
1863                    .filter(|&j| extents[i].conflicts(&extents[j]))
1864                    .count()
1865            })
1866            .sum()
1867    }

1869    fn random_window(rng: &mut Rng) -> Vec<SimFile> {
1870        (0..2 + rng.next(30))
1871            .map(|_| {
1872                let start = rng.next(3_000) as i64;
1873                let end = start + 1 + rng.next(600) as i64;
1874                let min = rng.next(200);
1875                let span = if rng.next(4) == 0 { 200 } else { 20 };
1876                let max = min + rng.next(span);
1877                sim_file(start, end, min, max, 1 + rng.next(2 * MAX_OUTPUT))
1878            })
1879            .collect()

1883    /// spanning many of its keys, which join them into oversized components.

1945    /// Applies `pick` until it picks nothing, asserting that `measure` strictly
1946    /// decreases with every merge. Returns the number of merges.
1947    fn run_to_fixpoint(
1948        rng: &mut Rng,
1949        pick: impl Fn(&[FileHandle]) -> Option<UnitPick>,
1950        measure: impl Fn(&[FileHandle]) -> (usize, usize),
1951    ) -> usize {
1952        run_windows_to_fixpoint(rng, random_window, pick, measure)
1953    }
1954
1955    fn run_windows_to_fixpoint(
1956        rng: &mut Rng,
1957        new_window: impl Fn(&mut Rng) -> Vec<SimFile>,
1958        pick: impl Fn(&[FileHandle]) -> Option<UnitPick>,
1959        measure: impl Fn(&[FileHandle]) -> (usize, usize),
1960    ) -> usize {
1961        let mut total_steps = 0;
1962        for window in 0..300 {
1963            let mut files = new_window(rng);
1964            let handles =
1965                |files: &[SimFile]| files.iter().map(|f| f.handle.clone()).collect::<Vec<_>>();
1966            let mut current = measure(&handles(&files));
1967            let mut steps = 0;
1968            while let Some(pick) = pick(&handles(&files)) {
1969                assert!(pick.inputs.len() >= 2);
1970                let picked = ids(&pick.inputs);
1971                let (inputs, kept): (Vec<_>, Vec<_>) = files
1972                    .into_iter()
1973                    .partition(|file| picked.contains(&file.handle.file_id().file_id()));
1974                files = kept;
1975                files.extend(merge(&inputs, rng));
1976                let next = measure(&handles(&files));
1977                assert!(next < current, "window {window}: {current:?} -> {next:?}");
1978                current = next;
1979                steps += 1;
1980            }
1981            total_steps += steps;
1982        }
1983        total_steps
1984    }
