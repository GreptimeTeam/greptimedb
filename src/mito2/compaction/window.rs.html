35/// Compaction picker that splits the time range of all involved files to windows, and merges
36/// the data segments intersects with those windows of files together so that the output files
37/// never overlaps.
38#[derive(Clone, Debug)]
39pub struct WindowedCompactionPicker {
40    compaction_time_window_seconds: Option<i64>,
41    time_range: Option<TimestampRange>,
42}
43
44impl WindowedCompactionPicker {
45    pub fn new(window_seconds: Option<i64>) -> Self {
46        Self {
47            compaction_time_window_seconds: window_seconds,
48            time_range: None,
49        }
50    }
51
52    /// Sets the time range used to select compaction windows.
53    pub(crate) fn with_time_range(mut self, time_range: Option<TimestampRange>) -> Self {
54        self.time_range = time_range;
55        self
56    }
57
58    // Computes compaction time window. First we respect user specified parameter, then
59    // use persisted window. If persist window is not present, we check the time window
60    // provided while creating table. If all of those are absent, we infer the window
61    // from files in level0.
62    fn calculate_time_window(
63        &self,
64        region_id: RegionId,
65        current_version: &CompactionVersion,
66    ) -> i64 {
67        self.compaction_time_window_seconds
68            .or(current_version
69                .compaction_time_window
70                .map(|t| t.as_secs() as i64))
71            .unwrap_or_else(|| {
72                let levels = current_version.ssts.levels();
73                let inferred = infer_time_bucket(levels[0].files());
74                info!(
75                    "Compaction window for region {} is not present, inferring from files: {:?}",
76                    region_id, inferred
77                );
78                inferred
79            })
80    }
81
82    fn pick_inner(
83        &self,
84        region_id: RegionId,
85        current_version: &CompactionVersion,
86        current_time: Timestamp,
87    ) -> (Vec<CompactionOutput>, Vec<FileHandle>, i64) {
88        let time_window = self.calculate_time_window(region_id, current_version);
89        info!(
90            "Compaction window for region: {} is {} seconds",
91            region_id, time_window
92        );
93
94        let expired_ssts = get_expired_ssts(
95            current_version.ssts.levels(),
96            current_version.options.ttl,
97            current_time,
98        );
99        if !expired_ssts.is_empty() {
100            info!("Expired SSTs in region {}: {:?}", region_id, expired_ssts);
101        }
102        let expired_file_ids = expired_ssts
103            .iter()
104            .map(|file| file.file_id())
105            .collect::<HashSet<_>>();
106
107        let last_non_null = current_version.options.merge_mode() == MergeMode::LastNonNull;
108        // Busy SSTs remain LastNonNull dependencies: skipping an intermediate
109        // version would allow older fields to be promoted above it.
110        let windows = assign_files_to_time_windows(
111            time_window,
112            current_version
113                .ssts
114                .levels()
115                .iter()
116                .flat_map(|level| level.files.values())
117                .filter(|file| !expired_file_ids.contains(&file.file_id()))
118                .filter(|file| last_non_null || !file.compacting()),
119        );
120        let windows = filter_time_windows(windows, self.time_range);
121
122        // One SST can span several outputs. Deferring only an unsafe window
123        // would remove that SST without rewriting all its remaining segments.
124        if last_non_null
125            && windows.values().any(|(_, inputs)| {
126                inputs.iter().any(FileHandle::compacting)
127                    || !inputs_precede_memtables(inputs, current_version.memtable_min_sequence)
128            })
129        {
130            return (vec![], expired_ssts, time_window);
131        }
132
133        (build_output(windows), expired_ssts, time_window)
134    }
135}

137#[async_trait::async_trait]
138impl Picker for WindowedCompactionPicker {
139    async fn pick(&self, compaction_region: &CompactionRegion) -> Result<Option<PickerOutput>> {
140        // TODO: With per-region compaction concurrency, the scheduler must wait
141        // for all running compactions to finish and apply their SST edits, and
142        // drain or cancel queued compactions before capturing current_version
143        // for SWCS. Block new compactions through snapshot capture and picking
144        // so the strict-window plan sees the complete SST view.
145        let picker = self.clone();
146        let region_id = compaction_region.current_version.metadata.region_id;
147        let current_version = compaction_region.current_version.clone();
148        let (outputs, expired_ssts, time_window) =
149            common_runtime::spawn_blocking_compact(move || {
150                picker.pick_inner(region_id, &current_version, Timestamp::current_millis())
151            })
152            .await
153            .context(JoinSnafu)?;
154
155        Ok(Some(PickerOutput {
156            outputs,
157            expired_ssts,
158            time_window_size: time_window,
159            max_file_size: None, // todo (hl): we may need to support `max_file_size` parameter in manual compaction.
160        }))
161    }
162}

164/// Keeps windows that overlap the requested range and their transitive dependencies.
165///
166/// [`assign_files_to_time_windows`] adds an SST to every time window that the SST covers. If a
167/// selected window contains such a cross-window SST, compaction will remove that input SST after
168/// rewriting it. Keeping only the directly selected window would therefore omit the SST's rows in
169/// the other windows. We must include every window covered by the SST, then repeat the process for
170/// other cross-window SSTs in those windows, until the complete dependency closure is selected.
171fn filter_time_windows(
172    mut windows: BTreeMap<i64, (i64, Vec<FileHandle>)>,
173    time_range: Option<TimestampRange>,
174) -> BTreeMap<i64, (i64, Vec<FileHandle>)> {
175    let Some(time_range) = time_range else {
176        return windows;
177    };
178
179    let mut selected_windows = windows
180        .iter()
181        .filter_map(|(lower_bound, (upper_bound, _))| {
182            let window_start = Timestamp::new_second(*lower_bound);
183            let window_end = Timestamp::new_second(*upper_bound);
184            let starts_before_range_end = time_range
185                .end()
186                .is_none_or(|range_end| window_start < range_end);
187            let ends_after_range_start = time_range
188                .start()
189                .is_none_or(|range_start| range_start < window_end);
190            (starts_before_range_end && ends_after_range_start).then_some(*lower_bound)
191        })
192        .collect::<HashSet<_>>();
193
194    let mut file_windows = HashMap::new();
195    for (lower_bound, (_, files)) in &windows {
196        for file in files {
197            file_windows
198                .entry(file.file_id())
199                .or_insert_with(Vec::new)
200                .push(*lower_bound);
201        }
202    }
203
204    let mut pending_windows = selected_windows.iter().copied().collect::<VecDeque<_>>();
205    let mut visited_files = HashSet::new();
206    while let Some(lower_bound) = pending_windows.pop_front() {
207        let (_, files) = &windows[&lower_bound];
208        for file in files {
209            if !visited_files.insert(file.file_id()) {
210                continue;
211            }
212            for dependent_window in &file_windows[&file.file_id()] {
213                if selected_windows.insert(*dependent_window) {
214                    pending_windows.push_back(*dependent_window);
215                }
216            }
217        }
218    }
219
220    windows.retain(|lower_bound, _| selected_windows.contains(lower_bound));
221    windows
222}
223
224fn build_output(windows: BTreeMap<i64, (i64, Vec<FileHandle>)>) -> Vec<CompactionOutput> {
225    let mut outputs = Vec::with_capacity(windows.len());
226    for (lower_bound, (upper_bound, files)) in windows {
227        // safety: the upper bound must > lower bound.
228        let output_time_range = Some(
229            TimestampRange::new(
230                Timestamp::new_second(lower_bound),
231                Timestamp::new_second(upper_bound),
232            )
233            .unwrap(),
234        );
235
236        let output = CompactionOutput {
237            output_level: 1,
238            inputs: files,
239            filter_deleted: false,
240            output_time_range,
241        };
242        outputs.push(output);
243    }
244
245    outputs
246}

248/// Assigns files to time windows. If file does not contain a time range in metadata, it will be
249/// assigned to a special bucket `i64::MAX` (normally no timestamp can be aligned to this bucket)
250/// so that all files without timestamp can be compacted together.
251fn assign_files_to_time_windows<'a>(
252    bucket_sec: i64,
253    files: impl Iterator<Item = &'a FileHandle>,
254) -> BTreeMap<i64, (i64, Vec<FileHandle>)> {
255    let mut buckets = BTreeMap::new();
256
257    for file in files {
258        let (start, end) = file.time_range();
259        let bounds = file_time_bucket_span(
260            // safety: converting whatever timestamp to seconds will not overflow.
261            start.convert_to(TimeUnit::Second).unwrap().value(),
262            end.convert_to(TimeUnit::Second).unwrap().value(),
263            bucket_sec,
264        );
265        for (lower_bound, upper_bound) in bounds {
266            let (_, files) = buckets
267                .entry(lower_bound)
268                .or_insert_with(|| (upper_bound, Vec::new()));
269            files.push(file.clone());
270        }
271    }
272    buckets
273}
274
275/// Calculates timestamp span between start and end timestamp.
276fn file_time_bucket_span(start_sec: i64, end_sec: i64, bucket_sec: i64) -> Vec<(i64, i64)> {
277    assert!(start_sec <= end_sec);
278
279    // if timestamp is between `[i64::MIN, i64::MIN.align_by_bucket(bucket)]`, which cannot
280    // be aligned to a valid i64 bound, simply return `i64::MIN` rather than just underflow.
281    let mut start_aligned = start_sec.align_by_bucket(bucket_sec).unwrap_or(i64::MIN);
282    let end_aligned = end_sec
283        .align_by_bucket(bucket_sec)
284        .unwrap_or(start_aligned + (end_sec - start_sec));
285
286    let mut res = Vec::with_capacity(((end_aligned - start_aligned) / bucket_sec + 1) as usize);
287    while start_aligned <= end_aligned {
288        let window_size = if start_aligned % bucket_sec == 0 {
289            bucket_sec
290        } else {
291            (start_aligned % bucket_sec).abs()
292        };
293        let upper_bound = start_aligned.checked_add(window_size).unwrap_or(i64::MAX);
294        res.push((start_aligned, upper_bound));
295        start_aligned = upper_bound;
296    }
297    res
298}

542    #[case::requested_window(0)]
543    #[case::transitive_window(1)]
544    #[case::unrelated_window(3)]
545    fn test_busy_dependencies_defer_last_non_null_plan_until_released(
546        #[case] busy_window: i64,
547        #[values(MergeMode::LastRow, MergeMode::LastNonNull)] merge_mode: MergeMode,
548    ) {
549        // A and C span both output windows. Skipping a busy intermediate B
550        // would allow an unsafe A+C merge; deferring just one output would
551        // lose the other segment of these shared inputs when they are removed.
552        let [a, b, c, expired] = std::array::from_fn(|_| FileId::random());
553        let files = [
554            (a, 0, 2 * HOUR - 1, 0),
555            (b, busy_window * HOUR, (busy_window + 1) * HOUR - 1, 1),
556            (c, 0, 2 * HOUR - 1, 0),
557            (expired, -2 * HOUR, -2 * HOUR, 0),
558        ];
559        let mut version = build_version(&files, Some(Duration::from_millis(3 * HOUR as u64)));
560        version.options.merge_mode = Some(merge_mode);
561        // No memtable barrier: only the busy SST dependency may defer this plan.
562        assert_eq!(None, version.memtable_min_sequence);
563        let busy_file = version.ssts.levels()[1].files().next().unwrap();
564        let picker =
565            WindowedCompactionPicker::new(Some(HOUR / 1000)).with_time_range(TimestampRange::new(
566                Timestamp::new_millisecond(0),
567                Timestamp::new_millisecond(HOUR),
568            ));
569
570        for busy in [true, false] {
571            busy_file.set_compacting(busy);
572            let (outputs, expired_ssts, _) = picker.pick_inner(
573                version.metadata.region_id,
574                &version,
575                Timestamp::new_millisecond(2 * HOUR),
576            );
577            assert_eq!(1, expired_ssts.len());
578            assert_eq!(expired, expired_ssts[0].meta_ref().file_id);
579            assert!(!expired_ssts[0].compacting());
580
581            if busy && merge_mode == MergeMode::LastNonNull && busy_window < 2 {
582                assert!(outputs.is_empty(), "the entire dependent plan must wait");
583                continue;
584            }
585
586            assert_eq!(2, outputs.len());
587            for (window, output) in outputs.iter().enumerate() {
588                let window = window as i64;
589                let mut expected = HashSet::from([a, c]);
590                if !busy && busy_window == window {
591                    expected.insert(b);
592                }
593                let actual: HashSet<_> = output
594                    .inputs
595                    .iter()
596                    .map(|file| file.meta_ref().file_id)
597                    .collect();
598                assert_eq!(expected, actual);
599                assert_eq!(
600                    TimestampRange::new(
601                        Timestamp::new_millisecond(window * HOUR),
602                        Timestamp::new_millisecond((window + 1) * HOUR),
603                    ),
604                    output.output_time_range
605                );
606            }
607        }
608    }
