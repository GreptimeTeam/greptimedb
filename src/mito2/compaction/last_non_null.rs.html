29/// Picker for [`MergeMode::LastNonNull`](crate::region::options::MergeMode).
30///
31/// LastNonNull fills null fields of a newer row from older non-null values when
32/// rows with the same key and timestamp are merged. That fill is only correct if
33/// every SST that can contribute to the same logical row is merged together:
34/// once a value is promoted into an output, an intermediate version left behind
35/// in a separately compacted SST is invisible, so it can no longer override the
36/// promoted value on a later read.
37///
38/// TWCS alone cannot guarantee this: it schedules independent windows and seeds,
39/// while an L1 or cross-window SST can overlap several of them. This picker
40/// therefore delegates seed selection to [TwcsPicker] and then closes each seed
41/// over the whole snapshot: it absorbs every overlapping file and every seed
42/// connected through those overlaps, so a merge stream is never split across
43/// outputs, and defers a closure whose inputs are already compacting.
44#[derive(Debug)]
45pub(super) struct LastNonNullPicker {
46    base_picker: TwcsPicker,
47}
48
49impl LastNonNullPicker {
50    pub(super) fn new(base_picker: TwcsPicker) -> Self {
51        Self { base_picker }
52    }
53}
54
55#[async_trait::async_trait]
56impl Picker for LastNonNullPicker {
57    async fn pick(&self, region: &CompactionRegion) -> Result<Option<PickerOutput>> {
58        let max_outputs = self.base_picker.max_background_tasks;
59
60        // Find all possible compaction seeds
61        let Some(mut all_seeds) = self
62            .base_picker
63            .pick_with_output_limit(region, None)
64            .await?
65        else {
66            return Ok(None);
67        };
68        let version = region.current_version.clone();
69        common_runtime::spawn_blocking_compact(move || {
70            let expired: HashSet<_> = all_seeds
71                .expired_ssts
72                .iter()
73                .map(FileHandle::file_id)
74                .collect();
75            // Include busy files and all levels/windows. Request ranges and
76            // TWCS size/file-count limits only constrain seeds, not correctness.
77            let all_files = version
78                .ssts
79                .levels()
80                .iter()
81                .flat_map(|level| level.files())
82                .filter(|file| !expired.contains(&file.file_id()))
83                .cloned()
84                .collect();
85            all_seeds.outputs =
86                build_closed_outputs(all_seeds.outputs, all_files, &version.metadata);
87            all_seeds.outputs.retain(|output| {
88                inputs_precede_memtables(&output.inputs, version.memtable_min_sequence)
89            });
90            if let Some(limit) = max_outputs {
91                // Only eligible closures consume the budget. Keep the high-priority
92                // tail because the compactor pops from the end of the output list.
93                let excess = all_seeds.outputs.len().saturating_sub(limit);
94                all_seeds.outputs.drain(..excess);
95            }
96            (!all_seeds.outputs.is_empty() || !all_seeds.expired_ssts.is_empty())
97                .then_some(all_seeds)
98        })
99        .await
100        .context(JoinSnafu)
101    }
102}

104/// Imported SSTs can be newer than pending writes. A merge must not attach an
105/// old field to a sequence above an intermediate version left in a memtable.
106pub(super) fn inputs_precede_memtables(
107    inputs: &[FileHandle],
108    memtable_min_sequence: Option<SequenceNumber>,
109) -> bool {
110    memtable_min_sequence.is_none_or(|min_sequence| {
111        inputs.iter().all(|file| {
112            file.meta_ref()
113                .sequence
114                .is_some_and(|max_sequence| max_sequence.get() < min_sequence)
115        })
116    })
117}

119/// Each seed and snapshot file is consumed at most once. Absorbing a seed also
120/// absorbs its disconnected members: a merge stream must never be split across
121/// independently scheduled outputs.
122fn build_closed_outputs(
123    seeds: Vec<CompactionOutput>,
124    files: Vec<FileHandle>,
125    metadata: &RegionMetadata,
126) -> Vec<CompactionOutput> {
127    // Maps every seed input to its seed, so a closure that reaches that input
128    // can absorb the rest of the seed instead of letting it schedule separately.
129    let seed_by_file: HashMap<_, _> = seeds
130        .iter()
131        .enumerate()
132        .flat_map(|(i, seed)| seed.inputs.iter().map(move |file| (file.file_id(), i)))
133        .collect();
134    // `None` marks a seed already absorbed by an earlier closure.
135    let mut seeds: Vec<_> = seeds.into_iter().map(Some).collect();
136    let mut remaining = FileOverlapIndex::new(files, metadata);
137    let mut outputs = Vec::new();
138    // The compactor pops from the end; retain TWCS's seed priority.
139    for seed_index in (0..seeds.len()).rev() {
140        let Some(output) = seeds[seed_index].take() else {
141            continue;
142        };
143        let output = expand_closure(output, &mut seeds, &seed_by_file, &mut remaining);
144        // Defer the entire closure rather than truncate it around busy files.
145        if !output.inputs.iter().any(FileHandle::compacting) {
146            outputs.push(output);
147        }
148    }
149    outputs.reverse();
150    outputs
151}

153/// Expands one seed into its full merge closure: absorb every connected seed
154/// and every overlapping snapshot file, transitively. Growing `output.inputs`
155/// while iterating is what makes the expansion transitive.
156fn expand_closure(
157    mut output: CompactionOutput,
158    seeds: &mut [Option<CompactionOutput>],
159    seed_by_file: &HashMap<RegionFileId, usize>,
160    remaining_files: &mut FileOverlapIndex<'_>,
161) -> CompactionOutput {
162    let mut selected = HashSet::new();
163    let inputs = std::mem::take(&mut output.inputs);
164    extend_inputs(&mut output, inputs, &mut selected, remaining_files);
165
166    let mut cursor = 0;
167    while cursor < output.inputs.len() {
168        let input = output.inputs[cursor].clone();
169        if let Some(&seed_index) = seed_by_file.get(&input.file_id())
170            && let Some(seed) = seeds[seed_index].take()
171        {
172            // AND the tombstone policies so the merged stream stays conservative.
173            output.filter_deleted &= seed.filter_deleted;
174            extend_inputs(&mut output, seed.inputs, &mut selected, remaining_files);
175        }
176        let dependencies = remaining_files.drain_overlaps(&input);
177        extend_inputs(&mut output, dependencies, &mut selected, remaining_files);
178        cursor += 1;
179    }
180    output
181}
182
183/// Appends never-seen inputs and removes them from the overlap index so the
184/// same file is neither expanded nor scheduled twice.
185fn extend_inputs(
186    output: &mut CompactionOutput,
187    inputs: Vec<FileHandle>,
188    selected: &mut HashSet<RegionFileId>,
189    remaining: &mut FileOverlapIndex<'_>,
190) {
191    for input in inputs {
192        if selected.insert(input.file_id()) {
193            remaining.remove(input.file_id());
194            output.inputs.push(input);
195        }
196    }
197}

583        let dependency = region.current_version.ssts.levels()[1]
584            .files()
585            .next()
586            .unwrap();
587        dependency.set_compacting(true);
588        let picker = new_picker(
589            &compact_request::Options::Regular(Default::default()),
590            &region.region_options,
591            Some(1),
592            None,
593        );
594        let picked = picker.pick(&region).await.unwrap().unwrap();
595        assert!(picked.outputs.is_empty());
596        assert_eq!(1, picked.expired_ssts.len());
597        assert_eq!(
598            Timestamp::new_millisecond(10),
599            picked.expired_ssts[0].time_range().1
600        );
601        assert!(!picked.expired_ssts[0].compacting());
602
603        dependency.set_compacting(false);
604        let picked = picker.pick(&region).await.unwrap().unwrap();
605        assert_eq!(1, picked.outputs.len());
606        assert_eq!(5, picked.outputs[0].inputs.len());
607        assert_eq!(1, picked.expired_ssts.len());
608        assert!(!file_ids(&picked.outputs[0].inputs).contains(&picked.expired_ssts[0].file_id()));
609    }
