50/// Validates every assignment before opening authoritative objects. Missing support is an
51/// error, never permission to read narthex files through the legacy adapter.
52pub(crate) fn file_indices(
53    context: &StreamContext,
54    ranges: &[PartitionRange],
55) -> Result<BTreeSet<usize>> {
56    let mut files = BTreeSet::new();
57    for range in ranges {
58        let range = context
59            .ranges
60            .get(range.identifier)
61            .ok_or_else(|| invalid("invalid narthex prepared scan range"))?;
62        for index in &range.row_group_indices {
63            if context.is_mem_range_index(*index) {
64                continue;
65            }
66            if !context.is_file_range_index(*index) {
67                return Err(invalid(
68                    "narthex preparation requires a prepared memtable source",
69                ));
70            }
71            let i = index.index - context.input.num_memtables();
72            if index.row_group_index != -1 {
73                return Err(invalid(
74                    "narthex preparation requires a complete samples assignment",
75                ));
76            }
77            files.insert(i);
78        }
79    }
80    Ok(files)
81}

83impl SeriesDiscovery {
84    pub(crate) fn fallback_to_samples(
85        &mut self,
86        window: WindowSequence,
87        samples: &[Arc<dyn SamplesFormatReader>],
88        metadata: &RegionMetadataRef,
89        pool: &Arc<dyn datafusion::execution::memory_pool::MemoryPool>,
90    ) -> Result<()> {
91        self.readers
92            .retain(|(_, coverage)| coverage.is_none_or(|w| w.start != window.start));
93        // A sample spanning multiple failed windows needs only one discovery pass.
94        for (_, i) in self
95            .covered_samples
96            .extract_if(.., |(range, _)| window.intersects(*range))
97        {
98            let reader =
99                NarthexSeriesReader::new(metadata.clone(), samples[i].clone(), pool.clone())?;
100            self.readers.push_back((Box::new(reader), None));
101        }
102        Ok(())
103    }
104}

106pub(crate) async fn open_sources(
107    context: &StreamContext,
108    ranges: &[PartitionRange],
109    pool: &Arc<dyn datafusion::execution::memory_pool::MemoryPool>,
110) -> Result<(Vec<Arc<dyn SamplesFormatReader>>, SeriesDiscovery)> {
111    let mut samples = Vec::<Arc<dyn SamplesFormatReader>>::new();
112    let mut series = SeriesDiscovery::default();
113    let indices = file_indices(context, ranges)?;
114    let mut windows = BTreeMap::new();
115    let mut unavailable = BTreeSet::new();
116    for file in &context.input.narthex_series_indexes {
117        let FileFormat::NarthexSeriesIndex(index) = &file.meta_ref().format else {
118            continue;
119        };
120        let window = index.window;
121        if file.region_id() != context.input.region_metadata().region_id
122            || !indices
123                .iter()
124                .any(|i| window.intersects(context.input.files[*i].time_range()))
125        {
126            continue;
127        }
128        let mut reader = match crate::series_index::columnar::reader::ColumnarReader::open_cached(
129            context.input.access_layer.object_store().clone(),
130            crate::series_index::SeriesIndexFileHandle::from_narthex(
131                file.clone(),
132                &context
133                    .input
134                    .access_layer
135                    .build_region_dir(file.meta_ref().region_id),
136            )?,
137            context
138                .input
139                .cache_strategy
140                .columnar_metadata(file.meta_ref().file_id),
141        )
142        .await
143        {
144            Ok(reader) => reader,
145            Err(error)
146                if error.is_object_not_found()
147                    || matches!(
148                        error,
149                        Error::InvalidMeta { .. } | Error::UnsupportedOperation { .. }
150                    ) =>
151            {
152                unavailable.insert(window.start);
153                continue;
154            }
155            Err(error) => return Err(error),
156        };
157        if reader.adapt_tags(context.input.region_metadata()).is_err() {
158            unavailable.insert(window.start);
159            continue;
160        }
161        let current = windows.entry(window.start).or_insert(window);
162        current.max_sequence = current.max_sequence.max(window.max_sequence);
163        series.readers.push_back((Box::new(reader), Some(window)));
164    }
165    windows.retain(|start, _| !unavailable.contains(start));
166    for i in indices {
167        let file = &context.input.files[i];
168        let namespace = file.meta_ref().file_id;
169        let reader = SamplesReader::open_cached(
170            context.input.access_layer.object_store().clone(),
171            SamplesFileHandle::from_narthex(
172                file.clone(),
173                &context
174                    .input
175                    .access_layer
176                    .build_region_dir(file.meta_ref().region_id),
177            )?,
178            context.input.cache_strategy.columnar_metadata(namespace),
179        )
180        .await?;
181        // Match Parquet: foreign SSTs use the destination admission sequence,
182        // while local SSTs retain their encoded row sequences. A local file's
183        // sequence metadata is only a bound, not a replacement for every row.
184        let sequence = if file.region_id() != context.input.region_metadata().region_id {
185            file.meta_ref().sequence.map(|sequence| sequence.get())
186        } else {
187            None
188        };
189        let reader: Arc<dyn SamplesFormatReader> =
190            Arc::new(NarthexSamplesReader { reader, sequence });
191        if !crate::sst::file::series_windows_cover(&windows, file.meta_ref()) {
192            series.readers.push_back((
193                Box::new(NarthexSeriesReader::new(
194                    context.input.region_metadata().clone(),
195                    reader.clone(),
196                    pool.clone(),
197                )?),
198                None,
199            ));
200        } else {
201            series
202                .covered_samples
203                .push((file.time_range(), samples.len()));
204        }
205        samples.push(reader);
206    }
207    let mut seen = BTreeSet::new();
208    for range in ranges {
209        for index in &context.ranges[range.identifier].row_group_indices {
210            if !context.is_mem_range_index(*index) || !seen.insert(index.index) {
211                continue;
212            }
213            for range in context.input.build_mem_ranges(*index) {
214                let reader: Arc<dyn SamplesFormatReader> = Arc::new(NarthexMemtableReader {
215                    range,
216                    schema: context.input.mapper.input_arrow_schema(false),
217                });
218                series.readers.push_back((
219                    Box::new(NarthexSeriesReader::new(
220                        context.input.region_metadata().clone(),
221                        reader.clone(),
222                        pool.clone(),
223                    )?),
224                    None,
225                ));
226                samples.push(reader);
227            }
228        }
229    }
230    Ok((samples, series))
231}

233/// Applies COPY admission sequences before snapshot visibility and deduplication.
234struct NarthexSamplesReader {
235    reader: SamplesReader,
236    sequence: Option<u64>,
237}
238
239impl SamplesFormatReader for NarthexSamplesReader {
240    fn schema(&self) -> SchemaRef {
241        self.reader.schema()
242    }
243
244    fn read(
245        &self,
246        request: SamplesReadRequest,
247    ) -> Result<BoxStream<'static, Result<SamplesBatch>>> {
248        let mut stream = self.reader.read(request)?;
249        let sequence = self.sequence;
250        Ok(Box::pin(async_stream::try_stream! {
251            while let Some(mut samples) = stream.try_next().await? {
252                if let Some(sequence) = sequence
253                    && let Ok(index) = samples.batch.schema().index_of(SEQUENCE_COLUMN_NAME)
254                {
255                    let mut columns = samples.batch.columns().to_vec();
256                    columns[index] = Arc::new(UInt64Array::from(vec![sequence; samples.batch.num_rows()]));
257                    samples.batch = RecordBatch::try_new(samples.batch.schema(), columns).context(NewRecordBatchSnafu)?;
258                }
259                yield samples;
260            }
261        }))
262    }
263}

265/// A series index is optional: sample metadata already owns the full exact key. Empty
266/// projection discovers keys without reading sample payloads. Actual sample reads enforce
267/// time/sequence visibility, so discovery may conservatively retain empty series.
268struct NarthexSeriesReader {
269    pool: Arc<dyn datafusion::execution::memory_pool::MemoryPool>,
270    schema: SchemaRef,
271    tags: Vec<(u32, String)>,
272    samples: Arc<dyn SamplesFormatReader>,
273}
274
275impl NarthexSeriesReader {
276    fn new(
277        metadata: RegionMetadataRef,
278        samples: Arc<dyn SamplesFormatReader>,
279        pool: Arc<dyn datafusion::execution::memory_pool::MemoryPool>,
280    ) -> Result<Self> {
281        let tags: Vec<_> = metadata
282            .primary_key_columns()
283            .filter(|column| {
284                ![ReservedColumnId::table_id(), ReservedColumnId::tsid()]
285                    .contains(&column.column_id)
286            })
287            .map(|column| (column.column_id, column.column_schema.name.clone()))
288            .collect();
289        let mut fields = vec![
290            Field::new(TABLE_ID_COLUMN, DataType::UInt32, false),
291            Field::new(TSID_COLUMN, DataType::UInt64, false),
292            Field::new(ROW_COUNT_COLUMN, DataType::UInt64, false),
293        ];
294        fields.extend(tags.iter().map(|(id, name)| {
295            crate::sst::with_field_id(Field::new(name, DataType::Utf8, true), *id)
296        }));
297        let schema = Arc::new(Schema::new(fields));
298        Ok(Self {
299            schema,
300            tags,
301            samples,
302            pool,
303        })
304    }
305}
306
307impl FormatReader for NarthexSeriesReader {
308    fn schema(&self) -> SchemaRef {
309        self.schema.clone()
310    }
311
312    fn read(&self, request: SeriesReadRequest) -> Result<BoxStream<'static, Result<RecordBatch>>> {
313        let schema = self.schema.clone();
314        let tags = self.tags.clone();
315        let mut stream = self.samples.read(SamplesReadRequest {
316            table_id: request.table_id,
317            memory_pool: Some(self.pool.clone()),
318            ..Default::default()
319        })?;
320        Ok(Box::pin(async_stream::try_stream! {
321            let mut previous: Option<ExactSeriesKey> = None;
322            let mut keys = Vec::with_capacity(1024);
323            while let Some(samples) = stream.try_next().await? {
324                if previous.as_ref() == Some(&samples.key) { continue; }
325                previous = Some(samples.key.clone());
326                if request.ids.as_ref().is_some_and(|ids| !ids.contains(&samples.key.id())) { continue; }
327                keys.push(samples.key);
328                if keys.len() == 1024 {
329                    yield series_batch(&schema, &tags, &keys, &request)?;
330                    keys.clear();
331                }
332            }
333            if !keys.is_empty() { yield series_batch(&schema, &tags, &keys, &request)?; }
334        }))
335    }
336}
337
338fn series_batch(
339    schema: &SchemaRef,
340    tags: &[(u32, String)],
341    keys: &[ExactSeriesKey],
342    request: &SeriesReadRequest,
343) -> Result<RecordBatch> {
344    let codec = SparsePrimaryKeyCodec::schemaless();
345    let mut values = HashMap::<u32, Vec<Option<String>>>::new();
346    for (id, _) in tags {
347        values.insert(*id, Vec::with_capacity(keys.len()));
348    }
349    for key in keys {
350        let CompositeValues::Sparse(decoded) = codec.decode(key.bytes()).context(DecodeSnafu)?
351        else {
352            return Err(invalid("narthex series requires sparse keys"));
353        };
354        for (id, column) in &mut values {
355            column.push(match decoded.get_or_null(*id) {
356                Value::String(value) => Some(value.as_utf8().to_owned()),
357                Value::Null => None,
358                _ => return Err(invalid("narthex series requires string tags")),
359            });
360        }
361    }
362    let mut columns: Vec<ArrayRef> = vec![
363        Arc::new(UInt32Array::from(
364            keys.iter().map(|key| key.id().table_id).collect::<Vec<_>>(),
365        )),
366        Arc::new(UInt64Array::from(
367            keys.iter().map(|key| key.id().tsid).collect::<Vec<_>>(),
368        )),
369        Arc::new(UInt64Array::from(vec![1; keys.len()])),
370    ];
371    for (id, _) in tags {
372        columns.push(Arc::new(StringArray::from(
373            values
374                .remove(id)
375                .ok_or_else(|| invalid("missing narthex series tag"))?,
376        )));
377    }
378    let mut batch = RecordBatch::try_new(schema.clone(), columns).context(NewRecordBatchSnafu)?;
379    for filter in &request.filters {
380        let column = batch
381            .column_by_name(filter.column_name())
382            .ok_or_else(|| invalid("unknown narthex series filter"))?;
383        let mask = filter.evaluate_array(column).context(RecordBatchSnafu)?;
384        batch = filter_record_batch(
385            &batch,
386            &datatypes::arrow::array::BooleanArray::new(mask, None),
387        )
388        .context(ComputeArrowSnafu)?;
389    }
390    let projection = request
391        .projection
392        .iter()
393        .map(|name| {
394            schema
395                .index_of(name)
396                .map_err(|error| invalid(error.to_string()))
397        })
398        .collect::<Result<Vec<_>>>()?;
399    batch.project(&projection).context(ComputeArrowSnafu)
400}

402/// Reopens the pinned memtable range for independent selected-series reads. Discovery
403/// projects only keys at this boundary; no rows are merged or deduplicated before the
404/// prepared source combines memory and persisted samples.
405struct NarthexMemtableReader {
406    range: crate::memtable::MemtableRange,
407    schema: SchemaRef,
408}
409
410impl SamplesFormatReader for NarthexMemtableReader {
411    fn schema(&self) -> SchemaRef {
412        self.schema.clone()
413    }
414
415    fn read(
416        &self,
417        request: SamplesReadRequest,
418    ) -> Result<BoxStream<'static, Result<SamplesBatch>>> {
419        if request.projection.is_empty()
420            && let Some(keys) = self.range.primary_keys()
421        {
422            return Ok(Box::pin(async_stream::try_stream! {
423                let mut offsets = mito_codec::row_converter::SparseOffsetsCache::new();
424                let schema = Arc::new(Schema::empty());
425                for bytes in keys {
426                    let key = ExactSeriesKey::from_sparse_key(&bytes, &mut offsets)?;
427                    if request.table_id.is_none_or(|table| key.id().table_id == table)
428                        && request.keys.as_ref().is_none_or(|keys| keys.contains(&key)) {
429                        let batch = RecordBatch::try_new_with_options(schema.clone(), vec![],
430                            &datatypes::arrow::record_batch::RecordBatchOptions::new().with_row_count(Some(1)))
431                            .context(NewRecordBatchSnafu)?;
432                        yield SamplesBatch { batch, key };
433                    }
434                }
435            }));
436        }
437        let iter = self.range.build_record_batch_iter(None, None)?;
438        Ok(Box::pin(async_stream::try_stream! {
439            use datatypes::arrow::array::{Array, BinaryArray};
440            use crate::sst::parquet::format::PrimaryKeyArray;
441            let mut offsets = mito_codec::row_converter::SparseOffsetsCache::new();
442            for batch in iter {
443                let batch = batch?;
444                let keys = batch.column_by_name(store_api::storage::consts::PRIMARY_KEY_COLUMN_NAME)
445                    .and_then(|array| array.as_any().downcast_ref::<PrimaryKeyArray>())
446                    .ok_or_else(|| invalid("narthex memtable requires dictionary primary keys"))?;
447                let values = keys.values().as_any().downcast_ref::<BinaryArray>()
448                    .ok_or_else(|| invalid("narthex memtable requires binary primary keys"))?;
449                let projection = request.projection.iter().map(|name| batch.schema().index_of(name).map_err(|error| invalid(error.to_string()))).collect::<Result<Vec<_>>>()?;
450                let projected = batch.project(&projection).context(ComputeArrowSnafu)?;
451                let mut start = 0;
452                while start < batch.num_rows() {
453                    let bytes = values.value(keys.keys().value(start) as usize);
454                    let mut end = start + 1;
455                    while end < batch.num_rows() && values.value(keys.keys().value(end) as usize) == bytes { end += 1; }
456                    let key = ExactSeriesKey::from_sparse_key(bytes, &mut offsets)?;
457                    if request.table_id.is_none_or(|table| key.id().table_id == table)
458                        && request.keys.as_ref().is_none_or(|keys| keys.contains(&key)) {
459                        yield SamplesBatch { batch: projected.slice(start, end-start), key };
460                    }
461                    start = end;
462                }
463            }
464        }))
465    }
