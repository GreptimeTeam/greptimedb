88        &self,
89        context: Arc<StreamContext>,
90        ranges: Vec<PartitionRange>,
91        metrics: PartitionMetrics,
92        scan_pool: Arc<dyn MemoryPool>,
93    ) -> Result<Arc<ColumnarPreparedSource>> {
94        let result = self
95            .source
96            .get_or_init(|| async {
97                let raw = RawScanPreparation {
98                    schema: Arc::new(datatypes::series::samples::SeriesSchema {
99                        fields: Vec::new().into(),
100                        eval_grid: None,
101                    }),
102                    fields: Vec::new(),
103                    predicates: Vec::new(),
104                };
105                ColumnarPreparedSource::prepare(
106                    context,
107                    ranges,
108                    RawLabelSource::Explicit,
109                    raw,
110                    metrics,
111                    scan_pool,
112                )
113                .await
114                .map(Arc::new)
115                .map_err(Arc::new)
116            })
117            .await;
118        let source = result
119            .as_ref()
120            .map_err(|error| crate::error::ScanSeriesSnafu.into_error(error.clone()))?;
121        if !matches!(source.labels, RawLabelSource::Explicit) {
122            return Err(invalid("cannot change prepared output representation"));
123        }
124        Ok(source.clone())
125    }

127    pub(crate) async fn prepare(
128        &self,
129        context: Arc<StreamContext>,
130        ranges: Vec<PartitionRange>,
131        labels: RawLabelSource,
132        raw: RawScanPreparation,
133        metrics: PartitionMetrics,
134        scan_pool: Arc<dyn MemoryPool>,
135    ) -> std::result::Result<Arc<dyn PreparedSeriesSource>, BoxedError> {
136        // Per caller, including time awaiting another partition's shared preparation.
137        // This overlaps prepare_wall and must not be added to its component timings.
138        let acquire_timer = crate::metrics::READ_STAGE_ELAPSED
139            .with_label_values(&["columnar_prepare_acquire_wall"])
140            .start_timer();
141        let expected = labels.clone();
142        let result = self
143            .source
144            .get_or_init(|| async {
145                ColumnarPreparedSource::prepare(context, ranges, labels, raw, metrics, scan_pool)
146                    .await
147                    .map(Arc::new)
148                    .map_err(Arc::new)
149            })
150            .await;
151        drop(acquire_timer);
152        let source = result.as_ref().map_err(|error| {
153            BoxedError::new(crate::error::ScanSeriesSnafu.into_error(error.clone()))
154        })?;
155        if !same_labels(&source.labels, &expected) {
156            return Err(BoxedError::new(invalid(
157                "label binding changed after series preparation",
158            )));
159        }
160        Ok(source.clone())
161    }

164pub(super) fn same_labels(left: &RawLabelSource, right: &RawLabelSource) -> bool {
165    match (left, right) {
166        (
167            RawLabelSource::PrimaryKey {
168                metric_name: a,
169                physical_metadata: am,
170                logical_tags: at,
171            },
172            RawLabelSource::PrimaryKey {
173                metric_name: b,
174                physical_metadata: bm,
175                logical_tags: bt,
176            },
177        ) => {
178            a == b
179                && Arc::ptr_eq(am, bm)
180                && at.len() == bt.len()
181                && at
182                    .iter()
183                    .zip(bt.iter())
184                    .all(|(a, b)| a.column_id == b.column_id && a.name == b.name)
185        }
186        _ => false,
187    }
188}

190/// Validates the pinned assignments and coverage without opening artifacts.
191///
192/// Returns whether every assigned source can be read from columnar artifacts. `false` is not an
193/// error: the caller scans those sources with the ordinary reader instead.
194pub(crate) fn validate_inputs(context: &StreamContext, ranges: &[PartitionRange]) -> Result<bool> {
195    if context.input.requires_narthex_preparation() {
196        narthex::file_indices(context, ranges)?;
197        return Ok(true);
198    }
199    collect_inputs(context, ranges).map(|inputs| inputs.is_some())
200}

202/// Collects the assigned SSTs and their covering columnar series indexes.
203///
204/// `None` means the assignments are structurally valid but not ready for columnar reads: an
205/// assigned memtable, an SST without columnar series coverage, or an SST without columnar
206/// samples. Structural corruption is still an error, and every assignment is validated before
207/// either answer is returned, so a later invalid assignment is never masked by missing artifacts.
208fn collect_inputs(
209    context: &StreamContext,
210    ranges: &[PartitionRange],
211) -> Result<Option<(Vec<usize>, HashSet<FileId>)>> {
212    let input = &context.input;
213    let metadata = input.region_metadata();
214    let index = input
215        .series_index
216        .as_ref()
217        .ok_or_else(|| invalid("missing columnar index context"))?;
218    let mut ready = true;
219    let mut files = BTreeMap::<usize, HashSet<i64>>::new();
220    for range in ranges {
221        let range = context
222            .ranges
223            .get(range.identifier)
224            .ok_or_else(|| invalid("invalid prepared scan range"))?;
225        for row_group in &range.row_group_indices {
226            if context.is_mem_range_index(*row_group) {
227                // Artifacts exist for flushed SSTs only.
228                ready = false;
229                continue;
230            }
231            if !context.is_file_range_index(*row_group) {
232                return Err(invalid(
233                    "columnar prepared reads do not support extension ranges",
234                ));
235            }
236            files
237                .entry(row_group.index - input.num_memtables())
238                .or_default()
239                .insert(row_group.row_group_index);
240        }
241    }
242    // Grouped artifacts may also contain sources excluded solely by time pruning.
243    // Sequence-pruned, unassigned, and retired sources remain ineligible.
244    let assigned: HashSet<_> = files
245        .keys()
246        .map(|i| input.files[*i].file_id().file_id())
247        .collect();
248    let allowed = assigned
249        .iter()
250        .chain(&input.time_pruned_files)
251        .copied()
252        .collect();
253    let eligible_samples = index
254        .version
255        .samples
256        .iter()
257        .filter(|(_, handle)| {
258            handle.entry().sources_visible(&allowed)
259                && assigned.iter().any(|id| handle.entry().covers(*id))
260        })
261        .map(|(id, handle)| (*id, handle.clone()))
262        .collect();
263    let mut indexes = HashSet::new();
264    let file_indices = files.keys().copied().collect();
265    for (file_idx, groups) in files {
266        let file = &input.files[file_idx];
267        if !groups.contains(&-1)
268            && (groups.len() as u64 != file.meta_ref().num_row_groups
269                || groups
270                    .iter()
271                    .any(|group| *group < 0 || *group as u64 >= file.meta_ref().num_row_groups))
272        {
273            return Err(invalid(
274                "columnar samples require complete SST range assignments",
275            ));
276        }
277        if file.meta_ref().region_id != metadata.region_id {
278            return Err(invalid(
279                "columnar prepared reads do not support foreign SSTs",
280            ));
281        }
282        let mut covered = false;
283        for handle in index.version.series_indexes.values() {
284            if handle.entry().format() == ArtifactFormat::ColumnarSeries
285                && handle
286                    .entry()
287                    .covers_file(file.meta_ref(), metadata.region_id)
288            {
289                indexes.insert(handle.entry().index_uuid);
290                covered = true;
291            }
292        }
293        if !covered {
294            ready = false;
295        }
296        if !crate::series_index::artifact::samples_cover_file(&eligible_samples, file)
297            || index.version.samples.values().any(|h| {
298                h.entry().covers(file.file_id().file_id())
299                    && h.entry().artifact.format != ArtifactFormat::ColumnarSamples
300            })
301        {
302            ready = false;
303        }
304    }
305    Ok(ready.then_some((file_indices, indexes)))
306}

308/// Pushes the supported conjuncts into the request.
309///
310/// The second element reports, for every input conjunct, whether `ColumnarReader`
311/// applies it exactly, so callers must not evaluate those conjuncts again.
312fn series_read_request(schema: &Schema, conjuncts: &[&Expr]) -> (SeriesReadRequest, Vec<bool>) {
313    let mut request = SeriesReadRequest {
314        projection: schema
315            .fields()
316            .iter()
317            .filter(|field| field.name() != ROW_COUNT_COLUMN)
318            .map(|field| field.name().clone())
319            .collect(),
320        ..Default::default()
321    };
322    let mut accepted = Vec::with_capacity(conjuncts.len());
323    for expr in conjuncts {
324        let Some(filter) = SimpleFilterEvaluator::try_new_with_column_type(expr, &|name| {
325            schema
326                .field_with_name(name)
327                .ok()
328                .map(|field| field.data_type().clone())
329        }) else {
330            accepted.push(false);
331            continue;
332        };
333        if !request
334            .projection
335            .iter()
336            .any(|name| name == filter.column_name())
337        {
338            accepted.push(false);
339            continue;
340        }
341        if filter.column_name() == DATA_SCHEMA_TABLE_ID_COLUMN_NAME
342            && filter.is_eq()
343            && let Some(Value::UInt32(table_id)) = filter.literal_value()
344        {
345            // Keep all equality filters, including contradictory ones, for exact evaluation.
346            request.table_id = Some(table_id);
347        }
348        request.filters.push(filter);
349        accepted.push(true);
350    }
351    (request, accepted)
352}

354/// Flattens the scanner predicate into the conjuncts shared by pushdown and residual filters.
355fn predicate_conjuncts(predicate: Option<&Predicate>) -> Vec<&Expr> {
356    predicate
357        .into_iter()
358        .flat_map(|predicate| predicate.exprs())
359        .flat_map(split_conjunction)
360        .collect()
361}

363struct Candidate {
364    /// Coarse prefix of the exact key; only a hash, never identity.
365    id: MetricSeriesId,
366    /// The canonical sparse key: the identity every comparison, sample attribution and output
367    /// chunk uses. It must not be qualified by region: every region holding a fragment of one
368    /// series has to report the same identity, or the fragments never meet in one owner above
369    /// the regions.
370    key: Buffer,
371    tags: HashMap<u32, Option<String>>,
372    metric: Option<SeriesLabels>,
373}

400/// Opens sidecar readers after validating their complete source coverage.
401async fn open_sidecar_sources(
402    context: &StreamContext,
403    ranges: &[PartitionRange],
404) -> Result<(Vec<Arc<dyn SamplesFormatReader>>, SeriesDiscovery)> {
405    let input = &context.input;
406    let metadata = input.region_metadata();
407    let index = input
408        .series_index
409        .as_ref()
410        .ok_or_else(|| invalid("missing columnar index context"))?;
411    let Some((files, indexes)) = collect_inputs(context, ranges)? else {
412        // Readiness was checked before preparation, so this is a real capability change.
413        return Err(invalid(
414            "columnar series coverage changed after series preparation",
415        ));
416    };
417    let cache = input
418        .cache_strategy
419        .columnar_metadata(index.cache_namespace);
420    let mut samples = Vec::<Arc<dyn SamplesFormatReader>>::new();
421    let assigned: HashSet<_> = files
422        .iter()
423        .map(|i| input.files[*i].file_id().file_id())
424        .collect();
425    let allowed = assigned
426        .iter()
427        .chain(&input.time_pruned_files)
428        .copied()
429        .collect();
430    for handle in index.version.samples.values().filter(|h| {
431        if !h.entry().sources_visible(&allowed) || !assigned.iter().any(|id| h.entry().covers(*id))
432        {
433            return false;
434        }
435        // Coverage validation above still needs every day of each source. Only
436        // opening readers can omit daily artifacts outside the query range.
437        if let (Some(day), Some(range)) = (h.entry().window, input.time_range.as_ref()) {
438            let bounds = day
439                .checked_mul(86_400)
440                .zip(day.checked_add(1).and_then(|next| next.checked_mul(86_400)));
441            if let Some((start, end)) = bounds {
442                return range.intersects(
443                    &(Timestamp::new_second(start)..Timestamp::new_second(end)).into(),
444                );
445            }
446            // Retain extreme days whose bounds cannot be represented in seconds.
447        }
448        true
449    }) {
450        let reader = crate::series_index::samples::reader::SamplesReader::open_cached(
451            index.store.clone(),
452            handle.clone(),
453            cache.clone(),
454        )
455        .await?;
456        format::validate_sample_schema(&reader.schema(), metadata)?;
457        samples.push(Arc::new(reader));
458    }
459    let mut series = SeriesDiscovery::default();
460    for id in indexes {
461        series.readers.push_back((
462            Box::new(
463                crate::series_index::columnar::reader::ColumnarReader::open_cached(
464                    index.store.clone(),
465                    index.version.series_indexes[&id].clone(),
466                    cache.clone(),
467                )
468                .await?,
469            ),
470            None,
471        ));
472    }
473    Ok((samples, series))
474}

476impl ColumnarPreparedSource {
477    async fn prepare(
478        context: Arc<StreamContext>,
479        ranges: Vec<PartitionRange>,
480        labels: RawLabelSource,
481        raw: RawScanPreparation,
482        metrics: PartitionMetrics,
483        scan_pool: Arc<dyn MemoryPool>,
484    ) -> Result<Self> {
485        let _prepare_timer = crate::metrics::READ_STAGE_ELAPSED
486            .with_label_values(&["columnar_prepare_wall"])
487            .start_timer();
488        let mut preparation_metrics = crate::metrics::ColumnarReadMetrics::default();
489        preparation_metrics.counts.insert("prepare_executions", 1);
490        let started = Instant::now();
491        let input = &context.input;
492        let metadata = input.region_metadata();
493        let (samples, mut series) = if context.input.requires_narthex_preparation() {
494            narthex::open_sources(&context, &ranges, &scan_pool).await?
495        } else {
496            open_sidecar_sources(&context, &ranges).await?
497        };
498        preparation_metrics.add_elapsed("columnar_prepare_coverage", started.elapsed());
499        preparation_metrics
500            .counts
501            .insert("prepare_sample_readers", samples.len() as u64);
502        preparation_metrics
503            .counts
504            .insert("prepare_series_indexes", series.readers.len() as u64);
505        let reservation = MemoryConsumer::new("ColumnarPreparedSource").register(&scan_pool);
506        let started = Instant::now();
507        let mut raw = raw;
508        let batch_output = matches!(labels, RawLabelSource::Explicit);
509        let dedup_append_raw =
510            input.requires_narthex_preparation() && input.append_mode && !batch_output;
511        let sample_schema = if batch_output {
512            Some(input.mapper.input_arrow_schema(input.compaction))
513        } else {
514            sample_merge_schema(&context, &mut raw)?
515        };
516        // Sparse compaction omits tag columns, unlike the mapper's query batch.
517        // Derive the offset from the schema that actually reaches the dedup reader.
518        let merge_field_start = sample_schema.as_ref().map_or_else(
519            || input.mapper.field_column_start(),
520            |schema| {
521                schema
522                    .fields()
523                    .iter()
524                    .position(|field| {
525                        metadata
526                            .column_by_name(field.name())
527                            .is_some_and(|column| column.semantic_type == SemanticType::Field)
528                    })
529                    .unwrap_or(0)
530            },
531        );
532        preparation_metrics.add_elapsed("columnar_prepare_schema", started.elapsed());
533        preparation_metrics.bytes.insert("prepare_encoded_keys", 0);
534        preparation_metrics.counts.insert("prepare_batch_series", 0);
535        let direct_labels = !batch_output && sample_schema.is_some();
536        let conjuncts = predicate_conjuncts(input.predicate_group().predicate());
537        let partition = if input.requires_narthex_preparation() {
538            narthex_partition_filter(metadata.partition_expr.as_deref(), None)?
539        } else {
540            None
541        };
542        // Series rows carry tags only, so a rule over any other column, such as the time index,
543        // filters sample rows instead.
544        let (series_partition, sample_partition) = match partition {
545            Some(rule) if partition_on_tags_only(&rule, metadata) => (Some(rule), None),
546            Some(rule) => (None, Some(SamplePartitionFilter::try_new(&rule, metadata)?)),
547            None => (None, None),
548        };
549        let mut retained_rows = 0_usize;
550        let mut candidates = BTreeMap::<ExactSeriesKey, Candidate>::new();
551        while let Some((reader, window)) = series.readers.pop_front() {
552            let started = Instant::now();
553            let schema = reader.schema();
554            format::validate_series_schema(&schema, metadata)?;
555            let mut tags = schema
556                .fields()
557                .iter()
558                .skip(3)
559                .map(|field| {
560                    let id = format::field_id(field)?;
561                    Ok((id, field.name().clone()))
562                })
563                .collect::<Result<Vec<_>>>()?;
564            tags.sort_unstable_by(|a, b| a.1.cmp(&b.1));
565            let (request, accepted) = series_read_request(&schema, &conjuncts);
566            let projection = request
567                .projection
568                .iter()
569                .map(|name| schema.index_of(name).map_err(|e| invalid(e.to_string())))
570                .collect::<Result<Vec<_>>>()?;
571            let projected_schema = Arc::new(
572                schema
573                    .project(&projection)
574                    .map_err(|e| invalid(e.to_string()))?,
575            );
576            let mut filters = Vec::new();
577            for (expr, accepted) in conjuncts.iter().zip(&accepted) {
578                // The reader already applied accepted conjuncts exactly.
579                if *accepted
580                    || !expr
581                        .column_refs()
582                        .iter()
583                        .all(|column| projected_schema.index_of(&column.name).is_ok())
584                {
585                    continue;
586                }
587                filters.push(
588                    Predicate::to_physical_expr(expr, &projected_schema)
589                        .map_err(|e| invalid(e.to_string()))?,
590                );
591            }
592            if let Some(partition) = &series_partition {
593                filters.push(
594                    partition
595                        .try_as_physical_expr(&projected_schema)
596                        .map_err(|error| invalid(error.to_string()))?,
597                );
598            }
599            preparation_metrics.add_elapsed("columnar_prepare_compile", started.elapsed());
600            let mut batches = reader.read(request)?;
601            loop {
602                let timer = preparation_metrics.timer("columnar_prepare_discovery_wall");
603                let batch = batches.try_next().await;
604                drop(timer);
605                let batch = match (batch, window) {
606                    (Ok(batch), _) => batch,
607                    (Err(error), Some(window)) => {
608                        common_telemetry::warn!(
609                            ?error,
610                            ?window,
611                            "Series index read failed, discovering keys from samples"
612                        );
613                        series.fallback_to_samples(window, &samples, metadata, &scan_pool)?;
614                        break;
615                    }
616                    (Err(error), None) => return Err(error),
617                };
618                let Some(mut batch) = batch else {
619                    break;
620                };
621                *preparation_metrics
622                    .counts
623                    .entry("prepare_discovered_rows")
624                    .or_default() += batch.num_rows() as u64;
625                *preparation_metrics
626                    .counts
627                    .entry("prepare_discovery_batches")
628                    .or_default() += 1;
629                let started = Instant::now();
630                for filter in &filters {
631                    batch = common_recordbatch::filter::batch_filter(&batch, filter)
632                        .map_err(|e| invalid(e.to_string()))?;
633                }
634                preparation_metrics.add_elapsed("columnar_prepare_residual", started.elapsed());
635                *preparation_metrics
636                    .counts
637                    .entry("prepare_selected_rows")
638                    .or_default() += batch.num_rows() as u64;
639                let started = Instant::now();
640                let mut duplicate_rows = 0_u64;
641                let table = batch
642                    .column_by_name(DATA_SCHEMA_TABLE_ID_COLUMN_NAME)
643                    .ok_or_else(|| invalid("missing table ID column"))?
644                    .as_any()
645                    .downcast_ref::<UInt32Array>()
646                    .ok_or_else(|| invalid("invalid table ID column"))?;
647                let tsid = batch
648                    .column_by_name(DATA_SCHEMA_TSID_COLUMN_NAME)
649                    .ok_or_else(|| invalid("missing TSID column"))?
650                    .as_any()
651                    .downcast_ref::<UInt64Array>()
652                    .ok_or_else(|| invalid("invalid TSID column"))?;
653                let tag_arrays = tags
654                    .iter()
655                    .map(|(id, name)| {
656                        let strings = batch
657                            .column_by_name(name)
658                            .ok_or_else(|| invalid("missing series tag column"))?
659                            .as_any()
660                            .downcast_ref::<StringArray>()
661                            .ok_or_else(|| invalid("invalid series tag array"))?;
662                        Ok((*id, strings))
663                    })
664                    .collect::<Result<Vec<_>>>()?;
665                // Identity is the exact canonical sparse key. `tsid` is a hash, so rows
666                // that differ only in labels are different series and must both survive:
667                // re-encoding the artifact's own tags reproduces the SST key byte for byte.
668                let codec = SparsePrimaryKeyCodec::schemaless();
669                let mut offsets = mito_codec::row_converter::SparseOffsetsCache::new();
670                let mut identity = Vec::new();
671                let mut previous: Option<Vec<u8>> = None;
672                let mut fresh = Vec::<(ExactSeriesKey, usize)>::new();
673                let mut encoded_key_bytes = 0_u64;
674                for row in 0..batch.num_rows() {
675                    identity.clear();
676                    codec
677                        .encode_internal(table.value(row), tsid.value(row), &mut identity)
678                        .map_err(|e| invalid(e.to_string()))?;
679                    codec
680                        .encode_to_vec(
681                            tag_arrays.iter().map(|(id, strings)| {
682                                (
683                                    *id,
684                                    if strings.is_null(row) {
685                                        ValueRef::Null
686                                    } else {
687                                        ValueRef::String(strings.value(row))
688                                    },
689                                )
690                            }),
691                            &mut identity,
692                        )
693                        .map_err(|e| invalid(e.to_string()))?;
694                    encoded_key_bytes += identity.len() as u64;
695                    // Adjacent duplicates are the common case and never allocate a key.
696                    if previous.as_deref() == Some(identity.as_slice())
697                        || candidates.contains_key(identity.as_slice())
698                    {
699                        duplicate_rows += 1;
700                        continue;
701                    }
702                    let key = ExactSeriesKey::from_sparse_key(&identity, &mut offsets)?;
703                    previous = Some(identity.clone());
704                    fresh.push((key, row));
705                }
706                // Retain only rows that introduce a new identity; the retained batch backs
707                // both explicit labels and the metadata rows of direct-label candidates.
708                let retained = if fresh.len() == batch.num_rows() {
709                    batch.clone()
710                } else {
711                    let selected: Vec<u64> = fresh.iter().map(|(_, row)| *row as u64).collect();
712                    take_record_batch(&batch, &UInt64Array::from(selected))
713                        .context(ComputeArrowSnafu)?
714                };
715                if direct_labels {
716                    // Only explicit-label chunks retain a batch to resolve labels from.
717                    retained_rows += retained.num_rows();
718                }
719                let label_batch = if direct_labels {
720                    let RawLabelSource::PrimaryKey { logical_tags, .. } = &labels else {
721                        return Err(invalid("expected bound labels"));
722                    };
723                    let columns = logical_tags
724                        .iter()
725                        .map(|tag| {
726                            let name = tags
727                                .iter()
728                                .find(|(id, _)| *id == tag.column_id)
729                                .ok_or_else(|| invalid("missing bound tag"))?;
730                            Ok((
731                                tag.name.clone(),
732                                retained
733                                    .schema()
734                                    .index_of(&name.1)
735                                    .map_err(|e| invalid(e.to_string()))?,
736                            ))
737                        })
738                        .collect::<Result<Vec<_>>>()?;
739                    let owner = Arc::new(
740                        MemoryConsumer::new("ColumnarSeriesLabelBatch").register(&scan_pool),
741                    );
742                    let label_batch =
743                        SeriesLabelBatch::try_new(retained.clone(), columns, owner.clone())
744                            .map_err(|e| invalid(e.to_string()))?;
745                    owner
746                        .try_grow(label_batch.estimated_size())
747                        .context(crate::error::ScanMemoryExhaustedSnafu)?;
748                    Some(Arc::new(label_batch))
749                } else {
750                    None
751                };
752                let mut tag_values = 0_u64;
753                let mut batch_series = 0_u64;
754                for (retained_row, (exact, row)) in fresh.into_iter().enumerate() {
755                    let mut values = HashMap::new();
756                    if !direct_labels && (!batch_output || !input.compaction) {
757                        values.reserve(tag_arrays.len());
758                        for (column_id, strings) in &tag_arrays {
759                            values.insert(
760                                *column_id,
761                                (!strings.is_null(row)).then(|| strings.value(row).to_owned()),
762                            );
763                        }
764                    } else if let Some(filter) = &sample_partition {
765                        for column_id in filter.tag_ids() {
766                            let (_, strings) =
767                                tag_arrays
768                                    .iter()
769                                    .find(|(id, _)| *id == column_id)
770                                    .ok_or_else(|| invalid("missing partition tag column"))?;
771                            values.insert(
772                                column_id,
773                                (!strings.is_null(row)).then(|| strings.value(row).to_owned()),
774                            );
775                        }
776                    }
777                    tag_values += values.len() as u64;
778                    let key = Buffer::from(exact.bytes().to_vec());
779                    let bytes = key.len()
780                        + values.values().flatten().map(String::len).sum::<usize>()
781                        + values.capacity() * (std::mem::size_of::<(u32, Option<String>)>() + 16)
782                        + std::mem::size_of::<Candidate>()
783                        + 64;
784                    reservation
785                        .try_grow(bytes)
786                        .context(crate::error::ScanMemoryExhaustedSnafu)?;
787                    let metric = if let Some(label_batch) = &label_batch {
788                        batch_series += 1;
789                        let RawLabelSource::PrimaryKey { metric_name, .. } = &labels else {
790                            return Err(invalid("expected bound labels"));
791                        };
792                        Some(
793                            SeriesLabels::from_batch(
794                                Some(metric_name.clone()),
795                                label_batch.clone(),
796                                retained_row,
797                            )
798                            .map_err(|e| invalid(e.to_string()))?,
799                        )
800                    } else {
801                        None
802                    };
803                    let id = exact.id();
804                    candidates.insert(
805                        exact,
806                        Candidate {
807                            id,
808                            key,
809                            tags: values,
810                            metric,
811                        },
812                    );
813                }
814                *preparation_metrics
815                    .bytes
816                    .entry("prepare_encoded_keys")
817                    .or_default() += encoded_key_bytes;
818                *preparation_metrics
819                    .counts
820                    .entry("prepare_tag_values")
821                    .or_default() += tag_values;
822                *preparation_metrics
823                    .counts
824                    .entry("prepare_duplicate_rows")
825                    .or_default() += duplicate_rows;
826                *preparation_metrics
827                    .counts
828                    .entry("prepare_batch_series")
829                    .or_default() += batch_series;
830                preparation_metrics.add_elapsed("columnar_prepare_candidates", started.elapsed());
831            }
832        }
833        let started = Instant::now();
834        preparation_metrics
835            .counts
836            .insert("prepare_unique_series", candidates.len() as u64);
837        preparation_metrics
838            .counts
839            .insert("prepare_retained_rows", retained_rows as u64);
840        let mut candidates: Vec<_> = candidates.into_values().collect();
841        candidates.sort_unstable_by(|a, b| a.key.as_slice().cmp(b.key.as_slice()));
842        preparation_metrics.add_elapsed("columnar_prepare_sort", started.elapsed());
843        Ok(Self {
844            context,
845            labels,
846            raw: Arc::new(raw),
847            sample_schema,
848            merge_field_start,
849            dedup_append_raw,
850            sample_partition: sample_partition.map(Arc::new),
851            candidates: Arc::new(candidates),
852            samples: Arc::new(samples),
853            metrics,
854            scan_pool,
855            reservation: Arc::new(reservation),
856        })
857    }
858}

860impl ColumnarPreparedSource {
861    pub(crate) fn read_batches(&self, range: Range<usize>) -> Result<BoxedRecordBatchStream> {
862        if range.start > range.end || range.end > self.candidates.len() {
863            return Err(invalid("series range out of bounds"));
864        }
865        let context = self.context.clone();
866        let sample_schema = self.sample_schema.clone();
867        let candidates = self.candidates.clone();
868        let samples = self.samples.clone();
869        let metrics = self.metrics.clone();
870        let reservation = self.reservation.clone();
871        let scan_pool = self.scan_pool.clone();
872        let field_column_start = self.merge_field_start;
873        let dedup_append_raw = self.dedup_append_raw;
874        let sample_partition = self.sample_partition.clone();
875        // Time-only conjuncts drop every version of a (series, time) key alike, so they
876        // commute with dedup and must run before the merge: a row selector inside the
877        // merge would otherwise pick a row they reject. Other conjuncts may depend on
878        // which version dedup keeps and stay after the merge. Volatile conjuncts never get
879        // here: DataFusion's `PushDownFilter` keeps them in the `Filter` above the scan.
880        let mut time_filters = Vec::new();
881        let mut filters = Vec::new();
882        if matches!(self.labels, RawLabelSource::Explicit) && !context.input.compaction {
883            // Batch output keeps this schema before and after the merge.
884            let schema = context.input.mapper.input_arrow_schema(false);
885            let time_index = &context
886                .input
887                .region_metadata()
888                .time_index_column()
889                .column_schema
890                .name;
891            for expr in predicate_conjuncts(context.input.predicate_group().predicate()) {
892                let filter = Predicate::to_physical_expr(expr, &schema)
893                    .map_err(|error| invalid(error.to_string()))?;
894                if expr
895                    .column_refs()
896                    .iter()
897                    .all(|column| &column.name == time_index)
898                {
899                    time_filters.push(filter);
900                } else {
901                    filters.push(filter);
902                }
903            }
904        }
905        let time_filters = Arc::new(time_filters);
906        let stream = async_stream::try_stream! {
907            let _reservation = reservation;
908            if range.is_empty() { return; }
909            let mut sources = Vec::<BoxedRecordBatchStream>::new();
910            for reader in samples.iter() {
911                let candidates = candidates.clone();
912                let range = range.clone();
913                let context = context.clone();
914                let reader = reader.clone();
915                let sample_schema = sample_schema.clone();
916                let scan_pool = scan_pool.clone();
917                let sample_partition = sample_partition.clone();
918                let time_filters = time_filters.clone();
919                sources.push(Box::pin(async_stream::try_stream! {
920                    let reservation = MemoryConsumer::new("ColumnarSamplesRead").register(&scan_pool);
921                    // Account for the selection retained by the samples reader, including
922                    // B-tree node overhead, before allocating it. The exact-key selection and
923                    // the coarse identifiers the reader derives from it both live for the
924                    // lifetime of the sample stream.
925                    let selection_bytes = candidates[range.clone()]
926                        .iter()
927                        .map(|candidate| {
928                            std::mem::size_of::<ExactSeriesKey>()
929                                + candidate.key.len()
930                                + std::mem::size_of::<MetricSeriesId>()
931                                + 64
932                        })
933                        .sum::<usize>();
934                    reservation.try_grow(selection_bytes).context(crate::error::ScanMemoryExhaustedSnafu)?;
935                    // Exact ownership: a block is read only when its own key is selected,
936                    // never because a colliding partner shares the identifier. Coarse
937                    // pruning bounds are derived from these keys inside the reader.
938                    // The keys are built from the one key the candidate retains, so the
939                    // selection costs one owned copy per series rather than a second copy
940                    // alongside every candidate.
941                    let mut offsets = mito_codec::row_converter::SparseOffsetsCache::new();
942                    let keys = candidates[range.clone()]
943                        .iter()
944                        .map(|c| ExactSeriesKey::from_sparse_key(c.key.as_slice(), &mut offsets))
945                        .collect::<Result<BTreeSet<_>>>()?;
946                    let mut stage_metrics = crate::metrics::ColumnarReadMetrics::default();
947                    let schema = reader.schema();
948                    let metadata = context.input.region_metadata();
949                    let name = &metadata.time_index_column().column_schema.name;
950                    let field = schema.field_with_name(name).map_err(|err| invalid(err.to_string()))?;
951                    let unit = match field.data_type() {
952                        datatypes::arrow::datatypes::DataType::Timestamp(unit, _) => Ok(common_time::timestamp::TimeUnit::from(unit)),
953                        _ => Err(invalid(format!("sample time index {name} must be a timestamp"))),
954                    }?;
955                    let bounds = context.input.time_range.as_ref();
956                    let start = bounds.and_then(|r| *r.start()).and_then(|t| t.convert_to_ceil(unit)).map(|t| t.value());
957                    let end = bounds.and_then(|r| *r.end()).and_then(|t| t.convert_to_ceil(unit)).map(|t| t.value());
958                    let mut batches = reader.read(SamplesReadRequest {
959                        projection: sample_projection(&schema, &context),
960                        memory_pool: Some(scan_pool.clone()),
961                        keys: Some(keys),
962                        start,
963                        end,
964                        ..Default::default()
965                    })?;
966                    while let Some(samples) = batches.try_next().await? {
967                        // A block belongs to exactly one series, so the reader returns its
968                        // exact key once and the whole batch is attributed at once.
969                        let SamplesBatch { batch, key } = samples;
970                        let ordinal = candidates[range.clone()]
971                            .binary_search_by(|candidate| {
972                                candidate.key.as_slice().cmp(key.bytes())
973                            })
974                            .map_err(|_| invalid("sample contains an unselected series"))?
975                            + range.start;
976                        let started = std::time::Instant::now();
977                        let flat = sample_to_flat(batch, &candidates[ordinal], &context, sample_schema.as_ref())?;
978                        let flat = match &sample_partition {
979                            Some(filter) => filter.filter(flat, &candidates[ordinal])?,
980                            None => flat,
981                        };
982                        let mut flat = filter_visibility(flat, &context)?;
983                        for filter in time_filters.iter() {
984                            flat = common_recordbatch::filter::batch_filter(&flat, filter)
985                                .map_err(|error| invalid(error.to_string()))?;
986                        }
987                        stage_metrics.add_elapsed("columnar_sample_adapt", started.elapsed());
988                        if flat.num_rows() == 0 { continue; }
989                        let bytes = flat.get_array_memory_size();
990                        reservation.try_grow(bytes).context(crate::error::ScanMemoryExhaustedSnafu)?;
991                        yield flat;
992                        reservation.shrink(bytes);
993                    }
994                }));
995            }
996            let mut merged = if let Some(schema) = sample_schema {
997                SeqScan::merge_flat_sources(&context, sources, Some(&metrics), false, schema, field_column_start).await?
998            } else {
999                SeqScan::build_flat_reader_from_sources(&context, sources, None, Some(&metrics), false, 1).await?
1000            };
1001            // Append mode skips dedup in the merge reader, but raw series reads
1002            // require unique timestamps. Non-append reads use the configured merge
1003            // mode above; batch reads retain append semantics.
1004            if dedup_append_raw {
1005                merged = Box::pin(crate::read::flat_dedup::FlatDedupReader::new(
1006                    merged,
1007                    crate::read::flat_dedup::FlatLastRow::new(context.input.filter_deleted),
1008                    Some(metrics.dedup_metrics_reporter()),
1009                ).into_stream());
1010            }
1011            while let Some(mut batch) = merged.try_next().await? {
1012                for filter in &filters {
1013                    batch = common_recordbatch::filter::batch_filter(&batch, filter)
1014                        .map_err(|error| invalid(error.to_string()))?;
1015                }
1016                if batch.num_rows() > 0 {
1017                    yield batch;
1018                }
1019            }
1020        };
1021        Ok(Box::pin(stream))
1022    }
1023}

1025impl PreparedSeriesSource for ColumnarPreparedSource {
1026    fn num_series(&self) -> usize {
1027        self.candidates.len()
1028    }
1029    fn label_source(&self) -> &RawLabelSource {
1030        &self.labels
1031    }
1032    fn series_info(&self, ordinal: usize) -> std::result::Result<SeriesInfo, BoxedError> {
1033        let candidate = self
1034            .candidates
1035            .get(ordinal)
1036            .ok_or_else(|| BoxedError::new(invalid("series ordinal out of bounds")))?;
1037        Ok(SeriesInfo {
1038            primary_key: candidate.key.clone(),
1039            metric: candidate.metric.clone(),
1040        })
1041    }
1042    fn read_series(
1043        &self,
1044        range: Range<usize>,
1045    ) -> std::result::Result<SendableRawSeriesStream, BoxedError> {
1046        let mut batches = self.read_batches(range).map_err(BoxedError::new)?;
1047        let stream = async_stream::try_stream! {
1048            while let Some(batch) = batches.try_next().await? {
1049                yield ScanBatch::RecordBatch(batch);
1050            }
1051        };
1052        let stream = raw::make_stream(
1053            Box::pin(stream),
1054            self.raw.schema.clone(),
1055            self.labels.clone(),
1056            self.raw.fields.clone(),
1057            self.metrics.clone(),
1058            self.raw.predicates.clone(),
1059            SeriesOrdering::KeyThenTime,
1060        );
1061        if self.sample_schema.is_none() {
1062            return Ok(stream);
1063        }
1064        let candidates = self.candidates.clone();
1065        let reservation = self.reservation.clone();
1066        Ok(raw::with_series_labels(stream, move |key| {
1067            // The resolver can outlive the exhausted sample stream.
1068            let _reservation = &reservation;
1069            let ordinal = candidates
1070                .binary_search_by(|candidate| candidate.key.as_slice().cmp(key.as_slice()))
1071                .map_err(|_| BoxedError::new(invalid("unknown merged series identity")))?;
1072            Ok((
1073                candidates[ordinal].key.clone(),
1074                candidates[ordinal]
1075                    .metric
1076                    .clone()
1077                    .ok_or_else(|| BoxedError::new(invalid("identity merge is missing labels")))?,
1078            ))
1079        }))
1080    }
1081}

1083/// Omits tags only when every predicate is enforced either on the series index
1084/// or on sample columns. Conjunctions can be separated; mixed disjunctions
1085/// retain the ordinary flat adapter.
1086pub(crate) fn sample_merge_schema(
1087    context: &StreamContext,
1088    raw: &mut RawScanPreparation,
1089) -> Result<Option<SchemaRef>> {
1090    let metadata = context.input.region_metadata();
1091    let full = context.input.mapper.input_arrow_schema(false);
1092    let projection: Vec<_> = full
1093        .fields()
1094        .iter()
1095        .enumerate()
1096        .filter_map(|(i, field)| {
1097            let tag = metadata
1098                .column_by_name(field.name())
1099                .is_some_and(|column| column.semantic_type == SemanticType::Tag);
1100            (!tag).then_some(i)
1101        })
1102        .collect();
1103    let schema = Arc::new(full.project(&projection).context(ComputeArrowSnafu)?);
1104    let mut predicates = Vec::new();
1105    if let Some(predicate) = context.input.predicate_group().predicate() {
1106        for expr in predicate.exprs().iter().flat_map(split_conjunction) {
1107            let columns = expr.column_refs();
1108            if columns.iter().all(|column| {
1109                metadata
1110                    .column_by_name(&column.name)
1111                    .is_some_and(|column| column.semantic_type == SemanticType::Tag)
1112            }) {
1113                // prepare() applies this complete expression to every series-index batch.
1114                continue;
1115            }
1116            if !columns
1117                .iter()
1118                .all(|column| schema.index_of(&column.name).is_ok())
1119            {
1120                return Ok(None);
1121            }
1122            predicates.push(
1123                Predicate::to_physical_expr(expr, &schema).map_err(|e| invalid(e.to_string()))?,
1124            );
1125        }
1126    }
1127    let fields = raw
1128        .fields
1129        .iter()
1130        .map(|(index, data_type)| {
1131            projection
1132                .iter()
1133                .position(|old| old == index)
1134                .map(|index| (index, data_type.clone()))
1135                .ok_or_else(|| invalid("sample merge is missing a selected field"))
1136        })
1137        .collect::<Result<Vec<_>>>()?;
1138    raw.fields = fields;
1139    raw.predicates = predicates;
1140    Ok(Some(schema))
1141}

1143fn sample_to_flat(
1144    batch: RecordBatch,
1145    candidate: &Candidate,
1146    context: &StreamContext,
1147    sample_schema: Option<&SchemaRef>,
1148) -> Result<RecordBatch> {
1149    let schema = sample_schema
1150        .cloned()
1151        .unwrap_or_else(|| context.input.mapper.input_arrow_schema(false));
1152    let key = &candidate.key;
1153    let metadata = context.input.region_metadata();
1154    let mut arrays = Vec::<ArrayRef>::new();
1155    for field in schema.fields() {
1156        let array: ArrayRef = match field.name().as_str() {
1157            "__primary_key" => Arc::new(
1158                DictionaryArray::<UInt32Type>::try_new(
1159                    UInt32Array::from(vec![0; batch.num_rows()]),
1160                    Arc::new(BinaryArray::new(
1161                        OffsetBuffer::new(
1162                            vec![
1163                                0,
1164                                i32::try_from(key.len()).map_err(|_| {
1165                                    invalid("series key exceeds Arrow offset limit")
1166                                })?,
1167                            ]
1168                            .into(),
1169                        ),
1170                        key.clone(),
1171                        None,
1172                    )),
1173                )
1174                .context(ComputeArrowSnafu)?,
1175            ),
1176            "__sequence" | "__op_type" => batch
1177                .column_by_name(field.name())
1178                .ok_or_else(|| invalid("missing sample internal column"))?
1179                .clone(),
1180            name => {
1181                let column = metadata
1182                    .column_by_name(name)
1183                    .ok_or_else(|| invalid("unknown flat column"))?;
1184                if column.semantic_type == SemanticType::Tag {
1185                    if column.column_id == ReservedColumnId::table_id() {
1186                        Arc::new(UInt32Array::from(vec![
1187                            candidate.id.table_id;
1188                            batch.num_rows()
1189                        ]))
1190                    } else if column.column_id == ReservedColumnId::tsid() {
1191                        Arc::new(UInt64Array::from(vec![candidate.id.tsid; batch.num_rows()]))
1192                    } else {
1193                        let value = candidate
1194                            .tags
1195                            .get(&column.column_id)
1196                            .ok_or_else(|| invalid("missing series tag"))?;
1197                        Arc::new(StringArray::from(vec![value.as_deref(); batch.num_rows()]))
1198                    }
1199                } else {
1200                    match batch.column_by_name(name) {
1201                        Some(array) => array.clone(),
1202                        None if field.is_nullable() => datatypes::arrow::array::new_null_array(
1203                            field.data_type(),
1204                            batch.num_rows(),
1205                        ),
1206                        None => return Err(invalid("missing non-null sample field")),
1207                    }
1208                }
1209            }
1210        };
1211        arrays.push(if array.data_type() == field.data_type() {
1212            array
1213        } else {
1214            cast(&array, field.data_type()).context(ComputeArrowSnafu)?
1215        });
1216    }
1217    RecordBatch::try_new(schema, arrays).context(NewRecordBatchSnafu)
1218}

1220fn filter_visibility(batch: RecordBatch, context: &StreamContext) -> Result<RecordBatch> {
1221    let sequence_column = batch
1222        .column_by_name(SEQUENCE_COLUMN_NAME)
1223        .ok_or_else(|| invalid("missing sequence column __sequence"))?;
1224    let sequence = sequence_column
1225        .as_any()
1226        .downcast_ref::<UInt64Array>()
1227        .ok_or_else(|| invalid("sequence column __sequence must be UInt64"))?;
1228    let metadata = context.input.region_metadata();
1229    let timestamp = &metadata.time_index_column().column_schema;
1230    let time_column = batch
1231        .column_by_name(&timestamp.name)
1232        .ok_or_else(|| invalid(format!("missing time index column {}", timestamp.name)))?;
1233    let unit = metadata.time_index_type().unit();
1234    let (time, actual_unit) = datatypes::timestamp::timestamp_array_to_primitive(time_column)
1235        .ok_or_else(|| {
1236            invalid(format!(
1237                "time index column {} must be a timestamp",
1238                timestamp.name
1239            ))
1240        })?;
1241    if common_time::timestamp::TimeUnit::from(actual_unit) != unit {
1242        return Err(invalid(format!(
1243            "time index column {} has unit {actual_unit:?}, expected {unit:?}",
1244            timestamp.name
1245        )));
1246    }
1247    // Bounds are computed from the actual arrays: sample metadata does not expose
1248    // sequence bounds here, and format-reader timestamps need not be sorted.
1249    if sequence.null_count() == 0 && time.null_count() == 0 {
1250        let needs_sequence_max = context.input.snapshot_sequence.is_some()
1251            || matches!(
1252                context.input.sequence_range,
1253                Some(SequenceRange::LtEq { .. } | SequenceRange::GtLtEq { .. })
1254            );
1255        let sequence_max = needs_sequence_max.then(|| max(sequence)).flatten();
1256        let snapshot_visible = context
1257            .input
1258            .snapshot_sequence
1259            .is_none_or(|bound| sequence_max.is_some_and(|value| value <= bound));
1260        let range_visible = context
1261            .input
1262            .sequence_range
1263            .is_none_or(|range| match range {
1264                SequenceRange::Gt { min: bound } => {
1265                    min(sequence).is_some_and(|value| value > bound)
1266                }
1267                SequenceRange::LtEq { max: bound } => {
1268                    sequence_max.is_some_and(|value| value <= bound)
1269                }
1270                SequenceRange::GtLtEq {
1271                    min: lower,
1272                    max: upper,
1273                } => {
1274                    sequence_max.is_some_and(|value| value <= upper)
1275                        && min(sequence).is_some_and(|value| value > lower)
1276                }
1277            });
1278        let time_visible = context.input.time_range.as_ref().is_none_or(|range| {
1279            min(&time).zip(max(&time)).is_some_and(|(lower, upper)| {
1280                range.contains(&common_time::Timestamp::new(lower, unit))
1281                    && range.contains(&common_time::Timestamp::new(upper, unit))
1282            })
1283        });
1284        if snapshot_visible && range_visible && time_visible {
1285            return Ok(batch);
1286        }
1287    }
1288    let sequence_filter = context
1289        .input
1290        .sequence_range
1291        .map(|range| range.filter(sequence_column))
1292        .transpose()
1293        .context(ComputeArrowSnafu)?;
1294    let filter = BooleanArray::from(
1295        (0..batch.num_rows())
1296            .map(|i| {
1297                !sequence.is_null(i)
1298                    && !time.is_null(i)
1299                    && context
1300                        .input
1301                        .snapshot_sequence
1302                        .is_none_or(|max| sequence.value(i) <= max)
1303                    && sequence_filter
1304                        .as_ref()
1305                        .is_none_or(|filter| filter.value(i))
1306                    && context.input.time_range.as_ref().is_none_or(|range| {
1307                        range.contains(&common_time::Timestamp::new(time.value(i), unit))
1308                    })
1309            })
1310            .collect::<Vec<_>>(),
1311    );
1312    filter_record_batch(&batch, &filter).context(ComputeArrowSnafu)
1313}

1315/// The mapper includes projected and residual-filter columns. Identity/time and
1316/// merge visibility columns are always needed by the narthex source adapter. The exact
1317/// series key is not projected: the samples reader returns it once per batch.
1318fn sample_projection(
1319    schema: &datatypes::arrow::datatypes::Schema,
1320    context: &StreamContext,
1321) -> Vec<String> {
1322    let input = context.input.mapper.input_arrow_schema(false);
1323    schema
1324        .fields()
1325        .iter()
1326        .enumerate()
1327        .filter(|(index, field)| {
1328            field.name() != store_api::storage::consts::PRIMARY_KEY_COLUMN_NAME
1329                && (*index < 5 || input.index_of(field.name()).is_ok())
1330        })
1331        .map(|(_, field)| field.name().clone())
1332        .collect()
1333}

1363/// A region partition rule that references a column other than a tag, applied to sample rows.
1364///
1365/// Files shared after repartition can hold rows of other partitions, so the rule has to drop them
1366/// row by row; a series may own rows on both sides of the rule.
1367struct SamplePartitionFilter {
1368    /// The columns the rule references, typed as the region declares them.
1369    schema: SchemaRef,
1370    /// Where each schema field's values come from.
1371    inputs: Vec<PartitionInput>,
1372    expr: Arc<dyn datafusion::physical_plan::PhysicalExpr>,
1373}
1374
1375enum PartitionInput {
1376    /// Logical table ID of the series. Repartitioning a metric table may route on it, and the
1377    /// series index keeps it outside the tag columns.
1378    TableId,
1379    /// A tag column, by column ID, valued from the series.
1380    Tag(u32),
1381    /// A column the sample rows carry.
1382    Sample,
1383}
1384
1385impl SamplePartitionFilter {
1386    fn try_new(
1387        rule: &partition::expr::PartitionExpr,
1388        metadata: &store_api::metadata::RegionMetadata,
1389    ) -> Result<Self> {
1390        let mut names = HashSet::new();
1391        rule.collect_column_names(&mut names);
1392        let mut names = names.into_iter().collect::<Vec<_>>();
1393        names.sort_unstable();
1394        let mut fields = Vec::with_capacity(names.len());
1395        let mut inputs = Vec::with_capacity(names.len());
1396        for name in names {
1397            let column = metadata
1398                .column_by_name(&name)
1399                .ok_or_else(|| invalid(format!("unknown partition column {name}")))?;
1400            fields.push(datatypes::arrow::datatypes::Field::new(
1401                name,
1402                datatypes::data_type::DataType::as_arrow_type(&column.column_schema.data_type),
1403                true,
1404            ));
1405            inputs.push(if column.column_id == ReservedColumnId::table_id() {
1406                PartitionInput::TableId
1407            } else if column.semantic_type == SemanticType::Tag {
1408                PartitionInput::Tag(column.column_id)
1409            } else {
1410                PartitionInput::Sample
1411            });
1412        }
1413        let schema = Arc::new(Schema::new(fields));
1414        let expr = rule
1415            .try_as_physical_expr(&schema)
1416            .map_err(|error| invalid(error.to_string()))?;
1417        Ok(Self {
1418            schema,
1419            inputs,
1420            expr,
1421        })
1422    }
1423
1424    fn tag_ids(&self) -> impl Iterator<Item = u32> + '_ {
1425        self.inputs.iter().filter_map(|input| match input {
1426            PartitionInput::Tag(column_id) => Some(*column_id),
1427            _ => None,
1428        })
1429    }
1430
1431    /// Keeps the rows of `flat`, all of `candidate`, that the rule assigns to this region.
1432    fn filter(&self, flat: RecordBatch, candidate: &Candidate) -> Result<RecordBatch> {
1433        let rows = flat.num_rows();
1434        let columns = self
1435            .schema
1436            .fields()
1437            .iter()
1438            .zip(&self.inputs)
1439            .map(|(field, input)| match input {
1440                PartitionInput::TableId => {
1441                    Ok(Arc::new(UInt32Array::from(vec![candidate.id.table_id; rows])) as ArrayRef)
1442                }
1443                PartitionInput::Tag(column_id) => {
1444                    let value = candidate
1445                        .tags
1446                        .get(column_id)
1447                        .ok_or_else(|| invalid("missing partition tag"))?;
1448                    Ok(Arc::new(StringArray::from(vec![value.as_deref(); rows])) as ArrayRef)
1449                }
1450                PartitionInput::Sample => {
1451                    let array = flat
1452                        .column_by_name(field.name())
1453                        .ok_or_else(|| invalid("missing partition column in samples"))?;
1454                    if array.data_type() == field.data_type() {
1455                        Ok(array.clone())
1456                    } else {
1457                        cast(array, field.data_type()).context(ComputeArrowSnafu)
1458                    }
1459                }
1460            })
1461            .collect::<Result<Vec<_>>>()?;
1462        let columns = RecordBatch::try_new_with_options(
1463            self.schema.clone(),
1464            columns,
1465            &datatypes::arrow::record_batch::RecordBatchOptions::new().with_row_count(Some(rows)),
1466        )
1467        .context(NewRecordBatchSnafu)?;
1468        let mask = self
1469            .expr
1470            .evaluate(&columns)
1471            .and_then(|value| value.into_array(rows))
1472            .map_err(|error| invalid(error.to_string()))?;
1473        let mask = mask
1474            .as_any()
1475            .downcast_ref::<BooleanArray>()
1476            .ok_or_else(|| invalid("partition rule must evaluate to booleans"))?;
1477        filter_record_batch(&flat, mask).context(ComputeArrowSnafu)
1478    }
1479}

1481/// Shares narthex discovery across partitions and assigns disjoint exact-series ranges.
1482pub(crate) fn scan_batches(
1483    preparation: Arc<Preparation>,
1484    context: Arc<StreamContext>,
1485    ranges: Vec<PartitionRange>,
1486    partition: usize,
1487    partitions: usize,
1488    metrics: PartitionMetrics,
1489    scan_pool: Arc<dyn MemoryPool>,
1490) -> crate::read::stream::ScanBatchStream {
1491    Box::pin(async_stream::try_stream! {
1492        let source = preparation.prepare_batches(context, ranges, metrics, scan_pool).await?;
1493        let n = source.candidates.len() as u128;
1494        let start = (n * partition as u128 / partitions as u128) as usize;
1495        let end = (n * (partition + 1) as u128 / partitions as u128) as usize;
1496        let mut stream = source.read_batches(start..end)?;
1497        while let Some(batch) = stream.try_next().await? {
1498            yield ScanBatch::RecordBatch(batch);
1499        }
1500    })
1501}
