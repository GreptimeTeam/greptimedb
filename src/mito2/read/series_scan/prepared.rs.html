86impl Preparation {
87    pub(crate) async fn prepare(
88        &self,
89        context: Arc<StreamContext>,
90        ranges: Vec<PartitionRange>,
91        labels: RawLabelSource,
92        raw: RawScanPreparation,
93        metrics: PartitionMetrics,
94    ) -> std::result::Result<Arc<dyn PreparedSeriesSource>, BoxedError> {
95        // Per caller, including time awaiting another partition's shared preparation.
96        // This overlaps prepare_wall and must not be added to its component timings.
97        let acquire_timer = crate::metrics::READ_STAGE_ELAPSED
98            .with_label_values(&["columnar_prepare_acquire_wall"])
99            .start_timer();
100        let expected = labels.clone();
101        let result = self
102            .source
103            .get_or_init(|| async {
104                ColumnarPreparedSource::prepare(context, ranges, labels, raw, metrics)
105                    .await
106                    .map(Arc::new)
107                    .map_err(Arc::new)
108            })
109            .await;
110        drop(acquire_timer);
111        let source = result.as_ref().map_err(|error| {
112            BoxedError::new(crate::error::ScanSeriesSnafu.into_error(error.clone()))
113        })?;
114        if !same_labels(&source.labels, &expected) {
115            return Err(BoxedError::new(invalid(
116                "label binding changed after series preparation",
117            )));
118        }
119        Ok(source.clone())
120    }
121}

123pub(super) fn same_labels(left: &RawLabelSource, right: &RawLabelSource) -> bool {
124    match (left, right) {
125        (
126            RawLabelSource::PrimaryKey {
127                metric_name: a,
128                physical_metadata: am,
129                logical_tags: at,
130            },
131            RawLabelSource::PrimaryKey {
132                metric_name: b,
133                physical_metadata: bm,
134                logical_tags: bt,
135            },
136        ) => {
137            a == b
138                && Arc::ptr_eq(am, bm)
139                && at.len() == bt.len()
140                && at
141                    .iter()
142                    .zip(bt.iter())
143                    .all(|(a, b)| a.column_id == b.column_id && a.name == b.name)
144        }
145        _ => false,
146    }
147}

165/// Collects the assigned SSTs and their covering columnar series indexes.
166///
167/// `None` means the assignments are structurally valid but not ready for columnar reads: an
168/// assigned memtable, an SST without columnar series coverage, or an SST without columnar
169/// samples. Structural corruption is still an error, and every assignment is validated before
170/// either answer is returned, so a later invalid assignment is never masked by missing artifacts.
171fn collect_inputs(
172    context: &StreamContext,
173    ranges: &[PartitionRange],
174) -> Result<Option<(Vec<usize>, HashSet<FileId>)>> {
175    let input = &context.input;
176    let metadata = input.region_metadata();
177    let index = input
178        .series_index
179        .as_ref()
180        .ok_or_else(|| invalid("missing columnar index context"))?;
181    let mut ready = true;
182    let mut files = BTreeMap::<usize, HashSet<i64>>::new();
183    for range in ranges {
184        let range = context
185            .ranges
186            .get(range.identifier)
187            .ok_or_else(|| invalid("invalid prepared scan range"))?;
188        for row_group in &range.row_group_indices {
189            if context.is_mem_range_index(*row_group) {
190                // Artifacts exist for flushed SSTs only.
191                ready = false;
192                continue;
193            }
194            if !context.is_file_range_index(*row_group) {
195                return Err(invalid(
196                    "columnar prepared reads do not support extension ranges",
197                ));
198            }
199            files
200                .entry(row_group.index - input.num_memtables())
201                .or_default()
202                .insert(row_group.row_group_index);
203        }
204    }
205    // Grouped artifacts may also contain sources excluded solely by time pruning.
206    // Sequence-pruned, unassigned, and retired sources remain ineligible.
207    let assigned: HashSet<_> = files
208        .keys()
209        .map(|i| input.files[*i].file_id().file_id())
210        .collect();
211    let allowed = assigned
212        .iter()
213        .chain(&input.time_pruned_files)
214        .copied()
215        .collect();
216    let eligible_samples = index
217        .version
218        .samples
219        .iter()
220        .filter(|(_, handle)| {
221            handle.entry().sources_visible(&allowed)
222                && assigned.iter().any(|id| handle.entry().covers(*id))
223        })
224        .map(|(id, handle)| (*id, handle.clone()))
225        .collect();
226    let mut indexes = HashSet::new();
227    let file_indices = files.keys().copied().collect();
228    for (file_idx, groups) in files {
229        let file = &input.files[file_idx];
230        if !groups.contains(&-1)
231            && (groups.len() as u64 != file.meta_ref().num_row_groups
232                || groups
233                    .iter()
234                    .any(|group| *group < 0 || *group as u64 >= file.meta_ref().num_row_groups))
235        {
236            return Err(invalid(
237                "columnar samples require complete SST range assignments",
238            ));
239        }
240        if file.meta_ref().region_id != metadata.region_id {
241            return Err(invalid(
242                "columnar prepared reads do not support foreign SSTs",
243            ));
244        }
245        let mut covered = false;
246        for handle in index.version.series_indexes.values() {
247            if handle.entry().format() == ArtifactFormat::ColumnarSeries
248                && handle
249                    .entry()
250                    .covers_file(file.meta_ref(), metadata.region_id)
251            {
252                indexes.insert(handle.entry().index_uuid);
253                covered = true;
254            }
255        }
256        if !covered {
257            ready = false;
258        }
259        if !crate::series_index::artifact::samples_cover_file(&eligible_samples, file)
260            || index.version.samples.values().any(|h| {
261                h.entry().covers(file.file_id().file_id())
262                    && h.entry().artifact.format != ArtifactFormat::ColumnarSamples
263            })
264        {
265            ready = false;
266        }
267    }
268    Ok(ready.then_some((file_indices, indexes)))
269}

271/// Pushes the supported conjuncts into the request.
272///
273/// The second element reports, for every input conjunct, whether `ColumnarReader`
274/// applies it exactly, so callers must not evaluate those conjuncts again.
275fn series_read_request(schema: &Schema, conjuncts: &[&Expr]) -> (SeriesReadRequest, Vec<bool>) {
276    let mut request = SeriesReadRequest {
277        projection: schema
278            .fields()
279            .iter()
280            .filter(|field| field.name() != ROW_COUNT_COLUMN)
281            .map(|field| field.name().clone())
282            .collect(),
283        ..Default::default()
284    };
285    let mut accepted = Vec::with_capacity(conjuncts.len());
286    for expr in conjuncts {
287        let Some(filter) = SimpleFilterEvaluator::try_new_with_column_type(expr, &|name| {
288            schema
289                .field_with_name(name)
290                .ok()
291                .map(|field| field.data_type().clone())
292        }) else {
293            accepted.push(false);
294            continue;
295        };
296        if !request
297            .projection
298            .iter()
299            .any(|name| name == filter.column_name())
300        {
301            accepted.push(false);
302            continue;
303        }
304        if filter.column_name() == DATA_SCHEMA_TABLE_ID_COLUMN_NAME
305            && filter.is_eq()
306            && let Some(Value::UInt32(table_id)) = filter.literal_value()
307        {
308            // Keep all equality filters, including contradictory ones, for exact evaluation.
309            request.table_id = Some(table_id);
310        }
311        request.filters.push(filter);
312        accepted.push(true);
313    }
314    (request, accepted)
315}

317/// Flattens the scanner predicate into the conjuncts shared by pushdown and residual filters.
318fn predicate_conjuncts(predicate: Option<&Predicate>) -> Vec<&Expr> {
319    predicate
320        .into_iter()
321        .flat_map(|predicate| predicate.exprs())
322        .flat_map(split_conjunction)
323        .collect()
324}

326struct Candidate {
327    /// Coarse prefix of the exact key; only a hash, never identity.
328    id: MetricSeriesId,
329    /// The one owned key allocation: the region-qualified exact key.
330    key: Buffer,
331    /// The canonical sparse key, a view of `key`'s bytes. This is the identity every
332    /// comparison and sample attribution uses, so it is stored once rather than once per
333    /// representation.
334    merge_key: Buffer,
335    tags: HashMap<u32, Option<String>>,
336    metric: Option<SeriesLabels>,
337}

339struct ColumnarPreparedSource {
340    context: Arc<StreamContext>,
341    labels: RawLabelSource,
342    raw: Arc<RawScanPreparation>,
343    sample_schema: Option<SchemaRef>,
344    candidates: Arc<Vec<Candidate>>,
345    samples: Arc<Vec<Arc<dyn SamplesFormatReader>>>,
346    metrics: PartitionMetrics,
347    // Readers retain this reservation with the candidate set.
348    reservation: Arc<MemoryReservation>,
349}
350
351impl ColumnarPreparedSource {
352    async fn prepare(
353        context: Arc<StreamContext>,
354        ranges: Vec<PartitionRange>,
355        labels: RawLabelSource,
356        raw: RawScanPreparation,
357        metrics: PartitionMetrics,
358    ) -> Result<Self> {
359        let _prepare_timer = crate::metrics::READ_STAGE_ELAPSED
360            .with_label_values(&["columnar_prepare_wall"])
361            .start_timer();
362        let mut preparation_metrics = crate::metrics::ColumnarReadMetrics::default();
363        preparation_metrics.counts.insert("prepare_executions", 1);
364        let started = Instant::now();
365        let input = &context.input;
366        let metadata = input.region_metadata();
367        let index = input
368            .series_index
369            .as_ref()
370            .ok_or_else(|| invalid("missing columnar index context"))?;
371        let Some((files, indexes)) = collect_inputs(&context, &ranges)? else {
372            // Readiness was checked before preparation, so this is a real capability change.
373            return Err(invalid(
374                "columnar series coverage changed after series preparation",
375            ));
376        };
377        preparation_metrics.add_elapsed("columnar_prepare_coverage", started.elapsed());
378        let timer = preparation_metrics.timer("columnar_prepare_samples_open_wall");
379        let cache = input
380            .cache_strategy
381            .columnar_metadata(index.cache_namespace);
382        let mut samples = Vec::<Arc<dyn SamplesFormatReader>>::new();
383        let assigned: HashSet<_> = files
384            .iter()
385            .map(|i| input.files[*i].file_id().file_id())
386            .collect();
387        let allowed = assigned
388            .iter()
389            .chain(&input.time_pruned_files)
390            .copied()
391            .collect();
392        for handle in index.version.samples.values().filter(|h| {
393            if !h.entry().sources_visible(&allowed)
394                || !assigned.iter().any(|id| h.entry().covers(*id))
395            {
396                return false;
397            }
398            // Coverage validation above still needs every day of each source. Only
399            // opening readers can omit daily artifacts outside the query range.
400            if let (Some(day), Some(range)) = (h.entry().window, input.time_range.as_ref()) {
401                let bounds = day
402                    .checked_mul(86_400)
403                    .zip(day.checked_add(1).and_then(|next| next.checked_mul(86_400)));
404                if let Some((start, end)) = bounds {
405                    return range.intersects(
406                        &(Timestamp::new_second(start)..Timestamp::new_second(end)).into(),
407                    );
408                }
409                // Retain extreme days whose bounds cannot be represented in seconds.
410            }
411            true
412        }) {
413            let reader = crate::series_index::samples::reader::SamplesReader::open_cached(
414                index.store.clone(),
415                handle.clone(),
416                cache.clone(),
417            )
418            .await?
419            .with_payload_cache(input.cache_strategy.columnar_payload(index.cache_namespace));
420            format::validate_sample_schema(&reader.schema(), metadata)?;
421            samples.push(Arc::new(reader));
422        }
423        drop(timer);
424        preparation_metrics
425            .counts
426            .insert("prepare_sample_readers", samples.len() as u64);
427        preparation_metrics
428            .counts
429            .insert("prepare_series_indexes", indexes.len() as u64);
430        let scan_pool = unaccounted_scan_pool();
431        let reservation = MemoryConsumer::new("ColumnarPreparedSource").register(&scan_pool);
432        let started = Instant::now();
433        let mut raw = raw;
434        let sample_schema = sample_merge_schema(&context, &mut raw)?;
435        preparation_metrics.add_elapsed("columnar_prepare_schema", started.elapsed());
436        preparation_metrics.bytes.insert("prepare_encoded_keys", 0);
437        preparation_metrics.counts.insert("prepare_batch_series", 0);
438        let direct_labels = sample_schema.is_some();
439        let conjuncts = predicate_conjuncts(input.predicate_group().predicate());
440        let mut retained_rows = 0_usize;
441        let mut candidates = BTreeMap::<ExactSeriesKey, Candidate>::new();
442        for id in indexes {
443            let timer = preparation_metrics.timer("columnar_prepare_series_open_wall");
444            let reader = crate::series_index::columnar::reader::ColumnarReader::open_cached(
445                index.store.clone(),
446                index.version.series_indexes[&id].clone(),
447                cache.clone(),
448            )
449            .await?;
450            drop(timer);
451            let started = Instant::now();
452            let schema = reader.schema();
453            format::validate_series_schema(&schema, metadata)?;
454            let mut tags = schema
455                .fields()
456                .iter()
457                .skip(3)
458                .map(|field| {
459                    let id = format::field_id(field)?;
460                    Ok((id, field.name().clone()))
461                })
462                .collect::<Result<Vec<_>>>()?;
463            tags.sort_unstable_by(|a, b| a.1.cmp(&b.1));
464            let (request, accepted) = series_read_request(&schema, &conjuncts);
465            let projection = request
466                .projection
467                .iter()
468                .map(|name| schema.index_of(name).map_err(|e| invalid(e.to_string())))
469                .collect::<Result<Vec<_>>>()?;
470            let projected_schema = Arc::new(
471                schema
472                    .project(&projection)
473                    .map_err(|e| invalid(e.to_string()))?,
474            );
475            let mut filters = Vec::new();
476            for (expr, accepted) in conjuncts.iter().zip(&accepted) {
477                // The reader already applied accepted conjuncts exactly.
478                if *accepted
479                    || !expr
480                        .column_refs()
481                        .iter()
482                        .all(|column| projected_schema.index_of(&column.name).is_ok())
483                {
484                    continue;
485                }
486                filters.push(
487                    Predicate::to_physical_expr(expr, &projected_schema)
488                        .map_err(|e| invalid(e.to_string()))?,
489                );
490            }
491            preparation_metrics.add_elapsed("columnar_prepare_compile", started.elapsed());
492            let mut batches = reader.read(request)?;
493            loop {
494                let timer = preparation_metrics.timer("columnar_prepare_discovery_wall");
495                let batch = batches.try_next().await?;
496                drop(timer);
497                let Some(mut batch) = batch else {
498                    break;
499                };
500                *preparation_metrics
501                    .counts
502                    .entry("prepare_discovered_rows")
503                    .or_default() += batch.num_rows() as u64;
504                *preparation_metrics
505                    .counts
506                    .entry("prepare_discovery_batches")
507                    .or_default() += 1;
508                let started = Instant::now();
509                for filter in &filters {
510                    batch = common_recordbatch::filter::batch_filter(&batch, filter)
511                        .map_err(|e| invalid(e.to_string()))?;
512                }
513                preparation_metrics.add_elapsed("columnar_prepare_residual", started.elapsed());
514                *preparation_metrics
515                    .counts
516                    .entry("prepare_selected_rows")
517                    .or_default() += batch.num_rows() as u64;
518                let started = Instant::now();
519                let mut duplicate_rows = 0_u64;
520                let table = batch
521                    .column_by_name(DATA_SCHEMA_TABLE_ID_COLUMN_NAME)
522                    .ok_or_else(|| invalid("missing table ID column"))?
523                    .as_any()
524                    .downcast_ref::<UInt32Array>()
525                    .ok_or_else(|| invalid("invalid table ID column"))?;
526                let tsid = batch
527                    .column_by_name(DATA_SCHEMA_TSID_COLUMN_NAME)
528                    .ok_or_else(|| invalid("missing TSID column"))?
529                    .as_any()
530                    .downcast_ref::<UInt64Array>()
531                    .ok_or_else(|| invalid("invalid TSID column"))?;
532                let tag_arrays = tags
533                    .iter()
534                    .map(|(id, name)| {
535                        let strings = batch
536                            .column_by_name(name)
537                            .ok_or_else(|| invalid("missing series tag column"))?
538                            .as_any()
539                            .downcast_ref::<StringArray>()
540                            .ok_or_else(|| invalid("invalid series tag array"))?;
541                        Ok((*id, strings))
542                    })
543                    .collect::<Result<Vec<_>>>()?;
544                // Identity is the exact canonical sparse key. `tsid` is a hash, so rows
545                // that differ only in labels are different series and must both survive:
546                // re-encoding the artifact's own tags reproduces the SST key byte for byte.
547                let codec = SparsePrimaryKeyCodec::schemaless();
548                let mut offsets = mito_codec::row_converter::SparseOffsetsCache::new();
549                let mut identity = Vec::new();
550                let mut previous: Option<Vec<u8>> = None;
551                let mut fresh = Vec::<(ExactSeriesKey, usize)>::new();
552                let mut encoded_key_bytes = 0_u64;
553                for row in 0..batch.num_rows() {
554                    identity.clear();
555                    codec
556                        .encode_internal(table.value(row), tsid.value(row), &mut identity)
557                        .map_err(|e| invalid(e.to_string()))?;
558                    codec
559                        .encode_to_vec(
560                            tag_arrays.iter().map(|(id, strings)| {
561                                (
562                                    *id,
563                                    if strings.is_null(row) {
564                                        ValueRef::Null
565                                    } else {
566                                        ValueRef::String(strings.value(row))
567                                    },
568                                )
569                            }),
570                            &mut identity,
571                        )
572                        .map_err(|e| invalid(e.to_string()))?;
573                    encoded_key_bytes += identity.len() as u64;
574                    // Adjacent duplicates are the common case and never allocate a key.
575                    if previous.as_deref() == Some(identity.as_slice())
576                        || candidates.contains_key(identity.as_slice())
577                    {
578                        duplicate_rows += 1;
579                        continue;
580                    }
581                    let key = ExactSeriesKey::from_sparse_key(&identity, &mut offsets)?;
582                    previous = Some(identity.clone());
583                    fresh.push((key, row));
584                }
585                // Retain only rows that introduce a new identity; the retained batch backs
586                // both explicit labels and the metadata rows of direct-label candidates.
587                let retained = if fresh.len() == batch.num_rows() {
588                    batch.clone()
589                } else {
590                    let selected: Vec<u64> = fresh.iter().map(|(_, row)| *row as u64).collect();
591                    take_record_batch(&batch, &UInt64Array::from(selected))
592                        .context(ComputeArrowSnafu)?
593                };
594                if direct_labels {
595                    // Only explicit-label chunks retain a batch to resolve labels from.
596                    retained_rows += retained.num_rows();
597                }
598                let label_batch = if direct_labels {
599                    let RawLabelSource::PrimaryKey { logical_tags, .. } = &labels else {
600                        return Err(invalid("expected bound labels"));
601                    };
602                    let columns = logical_tags
603                        .iter()
604                        .map(|tag| {
605                            let name = tags
606                                .iter()
607                                .find(|(id, _)| *id == tag.column_id)
608                                .ok_or_else(|| invalid("missing bound tag"))?;
609                            Ok((
610                                tag.name.clone(),
611                                retained
612                                    .schema()
613                                    .index_of(&name.1)
614                                    .map_err(|e| invalid(e.to_string()))?,
615                            ))
616                        })
617                        .collect::<Result<Vec<_>>>()?;
618                    let scan_pool = unaccounted_scan_pool();
619                    let owner = Arc::new(
620                        MemoryConsumer::new("ColumnarSeriesLabelBatch").register(&scan_pool),
621                    );
622                    let label_batch =
623                        SeriesLabelBatch::try_new(retained.clone(), columns, owner.clone())
624                            .map_err(|e| invalid(e.to_string()))?;
625                    owner
626                        .try_grow(label_batch.estimated_size())
627                        .context(crate::error::ScanMemoryExhaustedSnafu)?;
628                    Some(Arc::new(label_batch))
629                } else {
630                    None
631                };
632                let mut tag_values = 0_u64;
633                let mut batch_series = 0_u64;
634                for (retained_row, (exact, row)) in fresh.into_iter().enumerate() {
635                    let mut values = HashMap::new();
636                    if !direct_labels {
637                        values.reserve(tag_arrays.len());
638                        for (column_id, strings) in &tag_arrays {
639                            values.insert(
640                                *column_id,
641                                (!strings.is_null(row)).then(|| strings.value(row).to_owned()),
642                            );
643                        }
644                    }
645                    tag_values += values.len() as u64;
646                    // `merge_key` is always the exact key bytes. Explicit-label chunks keep
647                    // the region-qualified form as their `key`, so `primary_key` stays
648                    // region-scoped while identity and ordering stay exact.
649                    let key = if direct_labels {
650                        let mut buffer = Vec::with_capacity(8 + exact.bytes().len());
651                        buffer.extend_from_slice(&metadata.region_id.as_u64().to_be_bytes());
652                        buffer.extend_from_slice(exact.bytes());
653                        Buffer::from(buffer)
654                    } else {
655                        Buffer::from(exact.bytes().to_vec())
656                    };
657                    let merge_key = if direct_labels {
658                        key.slice_with_length(8, exact.bytes().len())
659                    } else {
660                        key.clone()
661                    };
662                    // The exact key is stored once: `merge_key` is a view of `key`, so the
663                    // reservation charges the single key allocation the candidate retains.
664                    let bytes = key.len()
665                        + values.values().flatten().map(String::len).sum::<usize>()
666                        + values.capacity() * (std::mem::size_of::<(u32, Option<String>)>() + 16)
667                        + std::mem::size_of::<Candidate>()
668                        + 64;
669                    reservation
670                        .try_grow(bytes)
671                        .map_err(|e| invalid(e.to_string()))?;
672                    let metric = if let Some(label_batch) = &label_batch {
673                        batch_series += 1;
674                        let RawLabelSource::PrimaryKey { metric_name, .. } = &labels else {
675                            return Err(invalid("expected bound labels"));
676                        };
677                        Some(
678                            SeriesLabels::from_batch(
679                                Some(metric_name.clone()),
680                                label_batch.clone(),
681                                retained_row,
682                            )
683                            .map_err(|e| invalid(e.to_string()))?,
684                        )
685                    } else {
686                        None
687                    };
688                    let id = exact.id();
689                    candidates.insert(
690                        exact,
691                        Candidate {
692                            id,
693                            key,
694                            merge_key,
695                            tags: values,
696                            metric,
697                        },
698                    );
699                }
700                *preparation_metrics
701                    .bytes
702                    .entry("prepare_encoded_keys")
703                    .or_default() += encoded_key_bytes;
704                *preparation_metrics
705                    .counts
706                    .entry("prepare_tag_values")
707                    .or_default() += tag_values;
708                *preparation_metrics
709                    .counts
710                    .entry("prepare_duplicate_rows")
711                    .or_default() += duplicate_rows;
712                *preparation_metrics
713                    .counts
714                    .entry("prepare_batch_series")
715                    .or_default() += batch_series;
716                preparation_metrics.add_elapsed("columnar_prepare_candidates", started.elapsed());
717            }
718        }
719        let started = Instant::now();
720        preparation_metrics
721            .counts
722            .insert("prepare_unique_series", candidates.len() as u64);
723        preparation_metrics
724            .counts
725            .insert("prepare_retained_rows", retained_rows as u64);
726        let mut candidates: Vec<_> = candidates.into_values().collect();
727        candidates.sort_unstable_by(|a, b| a.key.as_slice().cmp(b.key.as_slice()));
728        preparation_metrics.add_elapsed("columnar_prepare_sort", started.elapsed());
729        Ok(Self {
730            context,
731            labels,
732            raw: Arc::new(raw),
733            sample_schema,
734            candidates: Arc::new(candidates),
735            samples: Arc::new(samples),
736            metrics,
737            reservation: Arc::new(reservation),
738        })
739    }
740}

742impl PreparedSeriesSource for ColumnarPreparedSource {
743    fn num_series(&self) -> usize {
744        self.candidates.len()
745    }
746    fn label_source(&self) -> &RawLabelSource {
747        &self.labels
748    }
749    fn series_info(&self, ordinal: usize) -> std::result::Result<SeriesInfo, BoxedError> {
750        let candidate = self
751            .candidates
752            .get(ordinal)
753            .ok_or_else(|| BoxedError::new(invalid("series ordinal out of bounds")))?;
754        Ok(SeriesInfo {
755            primary_key: candidate.key.clone(),
756            metric: candidate.metric.clone(),
757        })
758    }
759    fn read_series(
760        &self,
761        range: Range<usize>,
762    ) -> std::result::Result<SendableRawSeriesStream, BoxedError> {
763        if range.start > range.end || range.end > self.candidates.len() {
764            return Err(BoxedError::new(invalid("series range out of bounds")));
765        }
766        let context = self.context.clone();
767        let sample_schema = self.sample_schema.clone();
768        let candidates = self.candidates.clone();
769        let samples = self.samples.clone();
770        let metrics = self.metrics.clone();
771        let reservation = self.reservation.clone();
772        let stream = async_stream::try_stream! {
773            let _reservation = reservation;
774            if range.is_empty() { return; }
775            let unit = context.input.region_metadata().time_index_type().unit();
776            let bounds = context.input.time_range.as_ref();
777            let start = bounds.and_then(|r| *r.start()).and_then(|t| t.convert_to_ceil(unit)).map(|t| t.value());
778            let end = bounds.and_then(|r| *r.end()).and_then(|t| t.convert_to_ceil(unit)).map(|t| t.value());
779            let mut sources = Vec::<BoxedRecordBatchStream>::new();
780            for reader in samples.iter() {
781                let candidates = candidates.clone();
782                let range = range.clone();
783                let context = context.clone();
784                let reader = reader.clone();
785                let sample_schema = sample_schema.clone();
786                sources.push(Box::pin(async_stream::try_stream! {
787                    let scan_pool = unaccounted_scan_pool();
788                    let reservation = MemoryConsumer::new("ColumnarSamplesRead").register(&scan_pool);
789                    // Account for the selection retained by the samples reader, including
790                    // B-tree node overhead, before allocating it. The exact-key selection and
791                    // the coarse identifiers the reader derives from it both live for the
792                    // lifetime of the sample stream.
793                    let selection_bytes = candidates[range.clone()]
794                        .iter()
795                        .map(|candidate| {
796                            std::mem::size_of::<ExactSeriesKey>()
797                                + candidate.merge_key.len()
798                                + std::mem::size_of::<MetricSeriesId>()
799                                + 64
800                        })
801                        .sum::<usize>();
802                    reservation.try_grow(selection_bytes).context(crate::error::ScanMemoryExhaustedSnafu)?;
803                    // Exact ownership: a block is read only when its own key is selected,
804                    // never because a colliding partner shares the identifier. Coarse
805                    // pruning bounds are derived from these keys inside the reader.
806                    // The keys are built from the one key the candidate retains, so the
807                    // selection costs one owned copy per series rather than a second copy
808                    // alongside every candidate.
809                    let mut offsets = mito_codec::row_converter::SparseOffsetsCache::new();
810                    let keys = candidates[range.clone()]
811                        .iter()
812                        .map(|c| ExactSeriesKey::from_sparse_key(c.merge_key.as_slice(), &mut offsets))
813                        .collect::<Result<BTreeSet<_>>>()?;
814                    let mut stage_metrics = crate::metrics::ColumnarReadMetrics::default();
815                    let mut batches = reader.read(SamplesReadRequest {
816                        projection: sample_projection(&reader.schema(), &context),
817                        memory_pool: Some(unaccounted_scan_pool()),
818                        keys: Some(keys),
819                        start,
820                        end,
821                        ..Default::default()
822                    })?;
823                    while let Some(samples) = batches.try_next().await? {
824                        // A block belongs to exactly one series, so the reader returns its
825                        // exact key once and the whole batch is attributed at once.
826                        let SamplesBatch { batch, key } = samples;
827                        let ordinal = candidates[range.clone()]
828                            .binary_search_by(|candidate| {
829                                candidate.merge_key.as_slice().cmp(key.bytes())
830                            })
831                            .map_err(|_| invalid("sample contains an unselected series"))?
832                            + range.start;
833                        let started = std::time::Instant::now();
834                        let flat = sample_to_flat(batch, &candidates[ordinal], &context, sample_schema.as_ref())?;
835                        let flat = filter_visibility(flat, &context)?;
836                        stage_metrics.add_elapsed("columnar_sample_adapt", started.elapsed());
837                        if flat.num_rows() == 0 { continue; }
838                        let bytes = flat.get_array_memory_size();
839                        reservation.try_grow(bytes).context(crate::error::ScanMemoryExhaustedSnafu)?;
840                        yield flat;
841                        reservation.shrink(bytes);
842                    }
843                }));
844            }
845            let mut merged = if let Some(schema) = sample_schema {
846                SeqScan::merge_flat_sources(&context, sources, Some(&metrics), false, schema, 0).await?
847            } else {
848                SeqScan::build_flat_reader_from_sources(&context, sources, None, Some(&metrics), false, 1).await?
849            };
850            while let Some(batch) = merged.try_next().await? {
851                if batch.num_rows() > 0 {
852                    yield ScanBatch::RecordBatch(batch);
853                }
854            }
855        };
856        let stream = raw::make_stream(
857            Box::pin(stream),
858            self.raw.schema.clone(),
859            self.labels.clone(),
860            self.raw.fields.clone(),
861            self.metrics.clone(),
862            self.raw.predicates.clone(),
863            SeriesOrdering::KeyThenTime,
864        );
865        if self.sample_schema.is_none() {
866            return Ok(stream);
867        }
868        let candidates = self.candidates.clone();
869        let reservation = self.reservation.clone();
870        Ok(raw::with_series_labels(stream, move |key| {
871            // The resolver can outlive the exhausted sample stream.
872            let _reservation = &reservation;
873            let ordinal = candidates
874                .binary_search_by(|candidate| candidate.merge_key.as_slice().cmp(key.as_slice()))
875                .map_err(|_| BoxedError::new(invalid("unknown merged series identity")))?;
876            Ok((
877                candidates[ordinal].key.clone(),
878                candidates[ordinal]
879                    .metric
880                    .clone()
881                    .ok_or_else(|| BoxedError::new(invalid("identity merge is missing labels")))?,
882            ))
883        }))
884    }
885}

887/// Omits tags only when every predicate is enforced either on the series index
888/// or on sample columns. Conjunctions can be separated; mixed disjunctions
889/// retain the ordinary flat adapter.
890pub(crate) fn sample_merge_schema(
891    context: &StreamContext,
892    raw: &mut RawScanPreparation,
893) -> Result<Option<SchemaRef>> {
894    let metadata = context.input.region_metadata();
895    let full = context.input.mapper.input_arrow_schema(false);
896    let projection: Vec<_> = full
897        .fields()
898        .iter()
899        .enumerate()
900        .filter_map(|(i, field)| {
901            let tag = metadata
902                .column_by_name(field.name())
903                .is_some_and(|column| column.semantic_type == SemanticType::Tag);
904            (!tag).then_some(i)
905        })
906        .collect();
907    let schema = Arc::new(full.project(&projection).context(ComputeArrowSnafu)?);
908    let mut predicates = Vec::new();
909    if let Some(predicate) = context.input.predicate_group().predicate() {
910        for expr in predicate.exprs().iter().flat_map(split_conjunction) {
911            let columns = expr.column_refs();
912            if columns.iter().all(|column| {
913                metadata
914                    .column_by_name(&column.name)
915                    .is_some_and(|column| column.semantic_type == SemanticType::Tag)
916            }) {
917                // prepare() applies this complete expression to every series-index batch.
918                continue;
919            }
920            if !columns
921                .iter()
922                .all(|column| schema.index_of(&column.name).is_ok())
923            {
924                return Ok(None);
925            }
926            predicates.push(
927                Predicate::to_physical_expr(expr, &schema).map_err(|e| invalid(e.to_string()))?,
928            );
929        }
930    }
931    let fields = raw
932        .fields
933        .iter()
934        .map(|(index, data_type)| {
935            projection
936                .iter()
937                .position(|old| old == index)
938                .map(|index| (index, data_type.clone()))
939                .ok_or_else(|| invalid("sample merge is missing a selected field"))
940        })
941        .collect::<Result<Vec<_>>>()?;
942    raw.fields = fields;
943    raw.predicates = predicates;
944    Ok(Some(schema))
945}

947fn sample_to_flat(
948    batch: RecordBatch,
949    candidate: &Candidate,
950    context: &StreamContext,
951    sample_schema: Option<&SchemaRef>,
952) -> Result<RecordBatch> {
953    let schema = sample_schema
954        .cloned()
955        .unwrap_or_else(|| context.input.mapper.input_arrow_schema(false));
956    let key = if sample_schema.is_some() {
957        &candidate.merge_key
958    } else {
959        &candidate.key
960    };
961    let metadata = context.input.region_metadata();
962    let mut arrays = Vec::<ArrayRef>::new();
963    for field in schema.fields() {
964        let array: ArrayRef = match field.name().as_str() {
965            "__primary_key" => Arc::new(
966                DictionaryArray::<UInt32Type>::try_new(
967                    UInt32Array::from(vec![0; batch.num_rows()]),
968                    Arc::new(BinaryArray::new(
969                        OffsetBuffer::new(
970                            vec![
971                                0,
972                                i32::try_from(key.len()).map_err(|_| {
973                                    invalid("series key exceeds Arrow offset limit")
974                                })?,
975                            ]
976                            .into(),
977                        ),
978                        key.clone(),
979                        None,
980                    )),
981                )
982                .context(ComputeArrowSnafu)?,
983            ),
984            "__sequence" | "__op_type" => batch
985                .column_by_name(field.name())
986                .ok_or_else(|| invalid("missing sample internal column"))?
987                .clone(),
988            name => {
989                let column = metadata
990                    .column_by_name(name)
991                    .ok_or_else(|| invalid("unknown flat column"))?;
992                if column.semantic_type == SemanticType::Tag {
993                    if column.column_id == ReservedColumnId::table_id() {
994                        Arc::new(UInt32Array::from(vec![
995                            candidate.id.table_id;
996                            batch.num_rows()
997                        ]))
998                    } else if column.column_id == ReservedColumnId::tsid() {
999                        Arc::new(UInt64Array::from(vec![candidate.id.tsid; batch.num_rows()]))
1000                    } else {
1001                        let value = candidate
1002                            .tags
1003                            .get(&column.column_id)
1004                            .ok_or_else(|| invalid("missing series tag"))?;
1005                        Arc::new(StringArray::from(vec![value.as_deref(); batch.num_rows()]))
1006                    }
1007                } else {
1008                    batch
1009                        .column_by_name(name)
1010                        .ok_or_else(|| invalid("missing sample field"))?
1011                        .clone()
1012                }
1013            }
1014        };
1015        arrays.push(if array.data_type() == field.data_type() {
1016            array
1017        } else {
1018            cast(&array, field.data_type()).context(ComputeArrowSnafu)?
1019        });
1020    }
1021    RecordBatch::try_new(schema, arrays).context(NewRecordBatchSnafu)
1022}

1024fn filter_visibility(batch: RecordBatch, context: &StreamContext) -> Result<RecordBatch> {
1025    let sequence_column = batch
1026        .column_by_name(SEQUENCE_COLUMN_NAME)
1027        .ok_or_else(|| invalid("missing sequence column __sequence"))?;
1028    let sequence = sequence_column
1029        .as_any()
1030        .downcast_ref::<UInt64Array>()
1031        .ok_or_else(|| invalid("sequence column __sequence must be UInt64"))?;
1032    let metadata = context.input.region_metadata();
1033    let timestamp = &metadata.time_index_column().column_schema;
1034    let time_column = batch
1035        .column_by_name(&timestamp.name)
1036        .ok_or_else(|| invalid(format!("missing time index column {}", timestamp.name)))?;
1037    let unit = metadata.time_index_type().unit();
1038    let (time, actual_unit) = datatypes::timestamp::timestamp_array_to_primitive(time_column)
1039        .ok_or_else(|| {
1040            invalid(format!(
1041                "time index column {} must be a timestamp",
1042                timestamp.name
1043            ))
1044        })?;
1045    if common_time::timestamp::TimeUnit::from(actual_unit) != unit {
1046        return Err(invalid(format!(
1047            "time index column {} has unit {actual_unit:?}, expected {unit:?}",
1048            timestamp.name
1049        )));
1050    }
1051    // Bounds are computed from the actual arrays: sample metadata does not expose
1052    // sequence bounds here, and format-reader timestamps need not be sorted.
1053    if sequence.null_count() == 0 && time.null_count() == 0 {
1054        let needs_sequence_max = context.input.snapshot_sequence.is_some()
1055            || matches!(
1056                context.input.sequence_range,
1057                Some(SequenceRange::LtEq { .. } | SequenceRange::GtLtEq { .. })
1058            );
1059        let sequence_max = needs_sequence_max.then(|| max(sequence)).flatten();
1060        let snapshot_visible = context
1061            .input
1062            .snapshot_sequence
1063            .is_none_or(|bound| sequence_max.is_some_and(|value| value <= bound));
1064        let range_visible = context
1065            .input
1066            .sequence_range
1067            .is_none_or(|range| match range {
1068                SequenceRange::Gt { min: bound } => {
1069                    min(sequence).is_some_and(|value| value > bound)
1070                }
1071                SequenceRange::LtEq { max: bound } => {
1072                    sequence_max.is_some_and(|value| value <= bound)
1073                }
1074                SequenceRange::GtLtEq {
1075                    min: lower,
1076                    max: upper,
1077                } => {
1078                    sequence_max.is_some_and(|value| value <= upper)
1079                        && min(sequence).is_some_and(|value| value > lower)
1080                }
1081            });
1082        let time_visible = context.input.time_range.as_ref().is_none_or(|range| {
1083            min(&time).zip(max(&time)).is_some_and(|(lower, upper)| {
1084                range.contains(&common_time::Timestamp::new(lower, unit))
1085                    && range.contains(&common_time::Timestamp::new(upper, unit))
1086            })
1087        });
1088        if snapshot_visible && range_visible && time_visible {
1089            return Ok(batch);
1090        }
1091    }
1092    let sequence_filter = context
1093        .input
1094        .sequence_range
1095        .map(|range| range.filter(sequence_column))
1096        .transpose()
1097        .context(ComputeArrowSnafu)?;
1098    let filter = BooleanArray::from(
1099        (0..batch.num_rows())
1100            .map(|i| {
1101                !sequence.is_null(i)
1102                    && !time.is_null(i)
1103                    && context
1104                        .input
1105                        .snapshot_sequence
1106                        .is_none_or(|max| sequence.value(i) <= max)
1107                    && sequence_filter
1108                        .as_ref()
1109                        .is_none_or(|filter| filter.value(i))
1110                    && context.input.time_range.as_ref().is_none_or(|range| {
1111                        range.contains(&common_time::Timestamp::new(time.value(i), unit))
1112                    })
1113            })
1114            .collect::<Vec<_>>(),
1115    );
1116    filter_record_batch(&batch, &filter).context(ComputeArrowSnafu)
1117}

1119/// The mapper includes projected and residual-filter columns. Identity/time and
1120/// merge visibility columns are always needed by the native source adapter. The exact
1121/// series key is not projected: the samples reader returns it once per batch.
1122fn sample_projection(
1123    schema: &datatypes::arrow::datatypes::Schema,
1124    context: &StreamContext,
1125) -> Vec<String> {
1126    let input = context.input.mapper.input_arrow_schema(false);
1127    schema
1128        .fields()
1129        .iter()
1130        .enumerate()
1131        .filter(|(index, field)| {
1132            field.name() != store_api::storage::consts::PRIMARY_KEY_COLUMN_NAME
1133                && (*index < 5 || input.index_of(field.name()).is_ok())
1134        })
1135        .map(|(_, field)| field.name().clone())
1136        .collect()
