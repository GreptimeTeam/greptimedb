50/// Handles projection and converts batches in flat format with correct schema.
51///
52/// This mapper support duplicate and unsorted projection indices.
53/// The output schema is determined by the projection indices.
54pub struct FlatProjectionMapper {
55    /// Metadata of the region.
56    metadata: RegionMetadataRef,
57    /// Schema for converted [RecordBatch] to return.
58    output_schema: SchemaRef,
59    /// The columns to read from memtables and SSTs.
60    /// The mapper won't deduplicate the column ids.
61    ///
62    /// Note that this doesn't contain the `__table_id` and `__tsid`.
63    read_cols: ReadColumns,
64    /// Ids and DataTypes of columns of the expected batch.
65    /// We can use this to check if the batch is compatible with the expected schema.
66    ///
67    /// It doesn't contain internal columns but always contains the time index column.
68    batch_schema: Vec<(ColumnId, ConcreteDataType)>,
69    /// `true` If the original projection is empty.
70    is_empty_projection: bool,
71    /// The index in flat format [RecordBatch] for each column in the output [RecordBatch].
72    batch_indices: Vec<usize>,
73    /// Precomputed Arrow schema for input batches.
74    input_arrow_schema: datatypes::arrow::datatypes::SchemaRef,
75}
76
77impl FlatProjectionMapper {
78    /// Returns a new mapper with projection.
79    /// If `projection` is empty, it outputs [RecordBatch] without any column but only a row count.
80    /// `SELECT COUNT(*) FROM table` is an example that uses an empty projection. DataFusion accepts
81    /// empty `RecordBatch` and only use its row count in this query.
82    pub fn new(
83        metadata: &RegionMetadataRef,
84        projection: impl IntoIterator<Item = usize>,
85    ) -> Result<Self> {
86        let projection: Vec<_> = projection.into_iter().collect();
87        let read_column_ids = read_column_ids_from_projection(metadata, &projection)?;
88        let read_cols = ReadColumns::new(read_column_ids);
89        Self::new_with_read_columns(metadata, projection, read_cols)
90    }
91
92    /// Returns a new mapper with output projection and explicit read columns.
93    pub fn new_with_read_columns(
94        metadata: &RegionMetadataRef,
95        projection: Vec<usize>,
96        read_cols: ReadColumns,
97    ) -> Result<Self> {
98        // If the original projection is empty.
99        let is_empty_projection = projection.is_empty();
100
101        // Output column schemas for the projection.
102        let mut col_schemas = Vec::with_capacity(projection.len());
103        // Column ids of the output projection without deduplication.
104        let mut output_col_ids = Vec::with_capacity(projection.len());
105        for idx in &projection {
106            let col = metadata
107                .column_metadatas
108                .get(*idx)
109                .with_context(|| InvalidRequestSnafu {
110                    region_id: metadata.region_id,
111                    reason: format!("projection index {} is out of bound", idx),
112                })?;
113            output_col_ids.push(col.column_id);
114
115            let mut schema = col.column_schema.clone();
116            if let Some(data_type) =
117                json2_read_datatype(col.column_id, &schema.data_type, &read_cols)
118            {
119                schema.data_type = data_type;
120            }
121            col_schemas.push(schema);
122        }
123
124        // Creates a map to lookup index.
125        let id_to_index = sst_column_id_indices(metadata);
126
127        // TODO(yingwen): Support different flat schema options.
128        let format_projection = FormatProjection::compute_format_projection(
129            metadata,
130            &id_to_index,
131            // All columns with internal columns.
132            metadata.column_metadatas.len() + 3,
133            read_cols.clone(),
134        );
135
136        let mut batch_schema = flat_projected_columns(metadata, &format_projection);
137
138        for (column_id, data_type) in batch_schema.iter_mut() {
139            if let Some(updated) = json2_read_datatype(*column_id, data_type, &read_cols) {
140                *data_type = updated;
141            }
142        }
143
144        // Safety: We get the column id from the metadata.
145        let input_arrow_schema = compute_input_arrow_schema(metadata, &batch_schema);
146
147        // If projection is empty, we don't output any column.
148        let output_schema = if is_empty_projection {
149            Arc::new(Schema::new(vec![]))
150        } else {
151            // Safety: Columns come from existing schema.
152            Arc::new(Schema::new(col_schemas))
153        };
154
155        let batch_indices = if is_empty_projection {
156            vec![]
157        } else {
158            output_col_ids
159                .iter()
160                .map(|id| {
161                    // Safety: The map is computed from the read projection.
162                    format_projection
163                        .column_id_to_projected_index
164                        .get(id)
165                        .copied()
166                        .with_context(|| {
167                            let name = metadata
168                                .column_by_id(*id)
169                                .map(|column| column.column_schema.name.clone())
170                                .unwrap_or_else(|| id.to_string());
171                            InvalidRequestSnafu {
172                                region_id: metadata.region_id,
173                                reason: format!(
174                                    "output column {} is missing in read projection",
175                                    name
176                                ),
177                            }
178                        })
179                })
180                .collect::<Result<Vec<_>>>()?
181        };
182
183        Ok(FlatProjectionMapper {
184            metadata: metadata.clone(),
185            output_schema,
186            read_cols,
187            batch_schema,
188            is_empty_projection,
189            batch_indices,
190            input_arrow_schema,
191        })
192    }

194    /// Returns a new mapper without projection.
195    pub fn all(metadata: &RegionMetadataRef) -> Result<Self> {
196        FlatProjectionMapper::new(metadata, 0..metadata.column_metadatas.len())
197    }

199    /// Keeps projected string primary-key columns in their flat-format dictionary encoding.
200    pub(crate) fn with_pk_dictionary_encoding(mut self) -> Self {
201        let mut changed = false;
202        let columns = self
203            .output_schema
204            .column_schemas()
205            .iter()
206            .map(|column| {
207                let mut column = column.clone();
208                if column.data_type == ConcreteDataType::string_datatype()
209                    && self
210                        .metadata
211                        .column_by_name(&column.name)
212                        .is_some_and(|metadata| metadata.semantic_type == SemanticType::Tag)
213                {
214                    changed = true;
215                    column.data_type = ConcreteDataType::dictionary_datatype(
216                        ConcreteDataType::uint32_datatype(),
217                        column.data_type.clone(),
218                    );
219                }
220                column
221            })
222            .collect();
223        if changed {
224            self.output_schema = Arc::new(Schema::new_with_version(
225                columns,
226                self.output_schema.version(),
227            ));
228        }
229        self
230    }

241    /// Returns the field column start index in output batch.
242    pub(crate) fn field_column_start(&self) -> usize {
243        for (idx, column_id) in self
244            .batch_schema
245            .iter()
246            .map(|(column_id, _)| column_id)
247            .enumerate()
248        {
249            // Safety: We get the column id from the metadata in new().
250            if self
251                .metadata
252                .column_by_id(*column_id)
253                .unwrap()
254                .semantic_type
255                == SemanticType::Field
256            {
257                return idx;
258            }
259        }
260
261        self.batch_schema.len()
262    }

269    /// Returns the input arrow schema from sources.
270    ///
271    /// The merge reader can use this schema.
272    pub(crate) fn input_arrow_schema(
273        &self,
274        compaction: bool,
275    ) -> datatypes::arrow::datatypes::SchemaRef {
276        if !compaction {
277            self.input_arrow_schema.clone()
278        } else {
279            // For compaction, we need to build a different schema from encoding.
280            let mut options = FlatSchemaOptions::from_encoding(self.metadata.primary_key_encoding);
281            options.concretized_json_types = self
282                .input_arrow_schema
283                .fields()
284                .iter()
285                .filter(|&field| is_json2_extension_type(field))
286                .map(|field| (field.name().clone(), field.data_type().clone()))
287                .collect();
288            to_flat_sst_arrow_schema(&self.metadata, &options)
289        }
290    }

299    /// Converts a flat format [RecordBatch] to a normal [RecordBatch].
300    ///
301    /// The batch must match the `projection` using to build the mapper.
302    pub(crate) fn convert(
303        &self,
304        batch: &datatypes::arrow::record_batch::RecordBatch,
305        cache_strategy: &CacheStrategy,
306    ) -> common_recordbatch::error::Result<RecordBatch> {
307        if self.is_empty_projection {
308            return RecordBatch::new_with_count(self.output_schema.clone(), batch.num_rows());
309        }
310        // Construct output record batch directly from Arrow arrays to avoid
311        // Arrow -> Vector -> Arrow roundtrips in the hot path.
312        let mut arrays = Vec::with_capacity(self.output_schema.num_columns());
313        for (output_idx, index) in self.batch_indices.iter().enumerate() {
314            let mut array = batch.column(*index).clone();
315            // Cast dictionary values to the target type.
316            if let ArrowDataType::Dictionary(_key_type, value_type) = array.data_type()
317                && !matches!(
318                    self.output_schema.arrow_schema().fields()[output_idx].data_type(),
319                    ArrowDataType::Dictionary(_, _)
320                )
321            {
322                // When a string dictionary column contains only a single value, reuse a cached
323                // repeated vector to avoid repeatedly expanding the dictionary.
324                if let Some(dict_array) = single_value_string_dictionary(
325                    &array,
326                    &self.output_schema.column_schemas()[output_idx].data_type,
327                    value_type.as_ref(),
328                ) {
329                    let dict_values = dict_array.values();
330                    let value = if dict_values.is_null(0) {
331                        Value::Null
332                    } else {
333                        Value::from(datatypes::arrow_array::string_array_value(dict_values, 0))
334                    };
335
336                    let repeated = repeated_vector_with_cache(
337                        &self.output_schema.column_schemas()[output_idx].data_type,
338                        &value,
339                        batch.num_rows(),
340                        cache_strategy,
341                    )?;
342                    array = repeated.to_arrow_array();
343                } else {
344                    let casted = datatypes::arrow::compute::cast(&array, value_type)
345                        .context(ArrowComputeSnafu)?;
346                    array = casted;
347                }
348            }
349
350            let field = &self.output_schema.arrow_schema().fields()[output_idx];
351            if is_json2_extension_type(field) {
352                array = JsonArray::from(&array)
353                    .project_to(field.data_type())
354                    .context(DataTypesSnafu)?;
355            }
356
357            arrays.push(array);
358        }
359
360        let df_record_batch =
361            DfRecordBatch::try_new(self.output_schema.arrow_schema().clone(), arrays)
362                .context(NewDfRecordBatchSnafu)?;
363        Ok(RecordBatch::from_df_record_batch(
364            self.output_schema.clone(),
365            df_record_batch,
366        ))
367    }

369    /// Projects columns from the input batch and converts them into vectors.
370    pub(crate) fn project_vectors(
371        &self,
372        batch: &datatypes::arrow::record_batch::RecordBatch,
373    ) -> common_recordbatch::error::Result<Vec<datatypes::vectors::VectorRef>> {
374        let mut columns = Vec::with_capacity(self.output_schema.num_columns());
375        for index in &self.batch_indices {
376            let mut array = batch.column(*index).clone();
377            // Casts dictionary values to the target type.
378            if let datatypes::arrow::datatypes::DataType::Dictionary(_key_type, value_type) =
379                array.data_type()
380            {
381                let casted = datatypes::arrow::compute::cast(&array, value_type)
382                    .context(ArrowComputeSnafu)?;
383                array = casted;
384            }
385            let vector = Helper::try_into_vector(array)
386                .map_err(BoxedError::new)
387                .context(ExternalSnafu)?;
388            columns.push(vector);
389        }
390        Ok(columns)
391    }

394fn json2_read_datatype(
395    column_id: ColumnId,
396    data_type: &ConcreteDataType,
397    read_cols: &ReadColumns,
398) -> Option<ConcreteDataType> {
399    let json_type = data_type.as_json()?;
400    if !json_type.is_json2() {
401        return None;
402    }
403
404    if let Some(concretized) = read_cols.json_target_type(column_id).cloned() {
405        return Some(ConcreteDataType::json2(concretized));
406    }
407
408    if is_empty_json2_type(json_type) {
409        return Some(ConcreteDataType::json2(JsonNativeType::Variant));
410    }
411
412    None
413}
414
415fn is_empty_json2_type(json_type: &JsonType) -> bool {
416    match json_type.native_type() {
417        JsonNativeType::Null => true,
418        JsonNativeType::Object(fields) if fields.is_empty() => true,
419        _ => false,
420    }
421}

423fn single_value_string_dictionary<'a>(
424    array: &'a Arc<dyn Array>,
425    output_type: &ConcreteDataType,
426    value_type: &ArrowDataType,
427) -> Option<&'a datatypes::arrow::array::DictionaryArray<datatypes::arrow::datatypes::UInt32Type>> {
428    if !matches!(
429        value_type,
430        ArrowDataType::Utf8 | ArrowDataType::LargeUtf8 | ArrowDataType::Utf8View
431    ) || !output_type.is_string()
432    {
433        return None;
434    }
435
436    let dict_array = array
437        .as_any()
438        .downcast_ref::<datatypes::arrow::array::DictionaryArray<
439            datatypes::arrow::datatypes::UInt32Type,
440        >>()?;
441
442    (dict_array.values().len() == 1 && dict_array.null_count() == 0).then_some(dict_array)
443}

445/// Returns ids and datatypes of columns of the output batch after applying the `projection`.
446///
447/// It adds the time index column if it doesn't present in the projection.
448pub(crate) fn flat_projected_columns(
449    metadata: &RegionMetadata,
450    format_projection: &FormatProjection,
451) -> Vec<(ColumnId, ConcreteDataType)> {
452    let time_index = metadata.time_index_column();
453    let num_columns = if format_projection
454        .column_id_to_projected_index
455        .contains_key(&time_index.column_id)
456    {
457        format_projection.column_id_to_projected_index.len()
458    } else {
459        format_projection.column_id_to_projected_index.len() + 1
460    };
461    let mut schema = vec![None; num_columns];
462    for (column_id, index) in &format_projection.column_id_to_projected_index {
463        // Safety: FormatProjection ensures the id is valid.
464        schema[*index] = Some((
465            *column_id,
466            metadata
467                .column_by_id(*column_id)
468                .unwrap()
469                .column_schema
470                .data_type
471                .clone(),
472        ));
473    }
474    if num_columns != format_projection.column_id_to_projected_index.len() {
475        schema[num_columns - 1] = Some((
476            time_index.column_id,
477            time_index.column_schema.data_type.clone(),
478        ));
479    }
480
481    // Safety: FormatProjection ensures all indices can be unwrapped.
482    schema.into_iter().map(|id_type| id_type.unwrap()).collect()
483}

485/// Computes the Arrow schema for input batches.
486///
487/// # Panics
488/// Panics if it can't find the column by the column id in the batch_schema.
489pub(crate) fn compute_input_arrow_schema(
490    metadata: &RegionMetadata,
491    batch_schema: &[(ColumnId, ConcreteDataType)],
492) -> datatypes::arrow::datatypes::SchemaRef {
493    let mut new_fields = Vec::with_capacity(batch_schema.len() + 3);
494    for (column_id, data_type) in batch_schema {
495        let column_metadata = metadata.column_by_id(*column_id).unwrap();
496        let field = Field::new(
497            &column_metadata.column_schema.name,
498            data_type.as_arrow_type(),
499            column_metadata.column_schema.is_nullable(),
500        )
501        .with_metadata(column_metadata.column_schema.metadata().clone());
502        let field = with_field_id(field, *column_id);
503        if column_metadata.semantic_type == SemanticType::Tag {
504            new_fields.push(tag_maybe_to_dictionary_field(
505                &column_metadata.column_schema.data_type,
506                &Arc::new(field),
507            ));
508        } else {
509            new_fields.push(Arc::new(field));
510        }
511    }
512    new_fields.extend_from_slice(&internal_fields());
513
514    Arc::new(datatypes::arrow::datatypes::Schema::new(new_fields))
515}

517/// Helper to project compaction batches into flat format columns
518/// (fields + time index + __primary_key + __sequence + __op_type).
519pub(crate) struct CompactionProjectionMapper {
520    mapper: FlatProjectionMapper,
521    assembler: DfBatchAssembler,
522}
523
524impl CompactionProjectionMapper {
525    pub(crate) fn try_new(metadata: &RegionMetadataRef) -> Result<Self> {
526        let projection = metadata
527            .column_metadatas
528            .iter()
529            .enumerate()
530            .filter_map(|(idx, col)| {
531                if matches!(col.semantic_type, SemanticType::Field) {
532                    Some(idx)
533                } else {
534                    None
535                }
536            })
537            .chain([metadata.time_index_column_pos()])
538            .collect::<Vec<_>>();
539
540        let read_col_ids = metadata.column_metadatas.iter().map(|col| col.column_id);
541        let read_cols = ReadColumns::new(read_col_ids);
542        let mapper = FlatProjectionMapper::new_with_read_columns(metadata, projection, read_cols)?;
543        let assembler = DfBatchAssembler::new(mapper.output_schema());
544
545        Ok(Self { mapper, assembler })
546    }
547
548    /// Projects columns and appends internal columns for compaction output.
549    ///
550    /// The input batch is expected to be in flat format with internal columns appended.
551    pub(crate) fn project(&self, batch: DfRecordBatch) -> Result<DfRecordBatch> {
552        let columns = self
553            .mapper
554            .project_vectors(&batch)
555            .context(RecordBatchSnafu)?;
556        self.assembler
557            .build_df_record_batch_with_internal(&batch, columns)
558            .context(RecordBatchSnafu)
559    }
560}
561
562/// Builds [DfRecordBatch] with internal columns appended.
563pub(crate) struct DfBatchAssembler {
564    output_arrow_schema_with_internal: datatypes::arrow::datatypes::SchemaRef,
565}
566
567impl DfBatchAssembler {
568    /// Precomputes the output schema with internal columns.
569    pub(crate) fn new(output_schema: SchemaRef) -> Self {
570        let fields = output_schema
571            .arrow_schema()
572            .fields()
573            .into_iter()
574            .chain(internal_fields().iter())
575            .cloned()
576            .collect::<Vec<_>>();
577        let output_arrow_schema_with_internal =
578            Arc::new(datatypes::arrow::datatypes::Schema::new(fields));
579        Self {
580            output_arrow_schema_with_internal,
581        }
582    }
583
584    /// Builds a [DfRecordBatch] from projected vectors plus internal columns.
585    ///
586    /// Assumes the input batch already contains internal columns as the last three fields
587    /// ("__primary_key", "__sequence", "__op_type").
588    pub(crate) fn build_df_record_batch_with_internal(
589        &self,
590        batch: &datatypes::arrow::record_batch::RecordBatch,
591        mut columns: Vec<datatypes::vectors::VectorRef>,
592    ) -> common_recordbatch::error::Result<DfRecordBatch> {
593        let num_columns = batch.columns().len();
594        // The last 3 columns are the internal columns.
595        let internal_indices = [num_columns - 3, num_columns - 2, num_columns - 1];
596        for index in internal_indices.iter() {
597            let array = batch.column(*index).clone();
598            let vector = Helper::try_into_vector(array)
599                .map_err(BoxedError::new)
600                .context(ExternalSnafu)?;
601            columns.push(vector);
602        }
603        RecordBatch::to_df_record_batch(self.output_arrow_schema_with_internal.clone(), columns)
604    }
605}
