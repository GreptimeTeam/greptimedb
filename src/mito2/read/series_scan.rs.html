81/// Maximum number of candidate series retained before distributing assignments.
82const CANDIDATE_SERIES_ASSIGNMENT_THRESHOLD: usize = 1_000_000;
83
84/// Test-only override of [`CANDIDATE_SERIES_ASSIGNMENT_THRESHOLD`], where `0` means "use the
85/// production constant".
86///
87/// The production build compiles this static, the setter below and the private accessor out, so
88/// the shipped threshold is always the constant above.
89#[cfg(test)]
90static CANDIDATE_SERIES_ASSIGNMENT_THRESHOLD_OVERRIDE: AtomicUsize = AtomicUsize::new(0);
91
92/// Test-only number of candidate assignment rounds sent since the override was last set.
93///
94/// A round is one `send_assignments` call, so a partition that received more than one assignment
95/// sees `>= 2` rounds here. Increments are compiled out of production builds.
96#[cfg(test)]
97static CANDIDATE_ASSIGNMENT_ROUNDS: AtomicUsize = AtomicUsize::new(0);
98
99/// Returns the candidate-series assignment threshold, honouring the test-only override.
100fn candidate_series_assignment_threshold() -> usize {
101    #[cfg(test)]
102    {
103        let threshold = CANDIDATE_SERIES_ASSIGNMENT_THRESHOLD_OVERRIDE.load(Ordering::Relaxed);
104        if threshold > 0 {
105            return threshold;
106        }
107    }
108    CANDIDATE_SERIES_ASSIGNMENT_THRESHOLD
109}
110
111/// Test-only: overrides the candidate-series assignment threshold and resets the round counter.
112///
113/// Pass `0` to restore the production threshold. Tests must reset the override before they
114/// return; a scope guard in `engine::scan_test` does that even on panic.
115#[cfg(test)]
116pub(crate) fn set_candidate_series_assignment_threshold(threshold: usize) {
117    CANDIDATE_SERIES_ASSIGNMENT_THRESHOLD_OVERRIDE.store(threshold, Ordering::Relaxed);
118    CANDIDATE_ASSIGNMENT_ROUNDS.store(0, Ordering::Relaxed);
119}

121/// Test-only: returns the number of assignment rounds sent since the override was last set.
122#[cfg(test)]
123pub(crate) fn candidate_assignment_rounds() -> usize {
124    CANDIDATE_ASSIGNMENT_ROUNDS.load(Ordering::Relaxed)
125}

189/// Scans a region and returns sorted rows of a series in the same partition.
190///
191/// The output order is always order by `(primary key, time index)` inside every
192/// partition.
193/// Always returns the same series (primary key) to the same partition.
194pub struct SeriesScan {
195    /// Implementation used by this scan.
196    mode: SeriesScanMode,
197    prepared: Arc<prepared::Preparation>,
198    /// Properties of the scanner.
199    properties: ScannerProperties,
200    /// Context of streams.
201    stream_ctx: Arc<StreamContext>,
202    /// Shared pruner for file range building.
203    pruner: Arc<Pruner>,
204    /// Legacy data receivers for each partition.
205    legacy_receivers: Mutex<LegacyReceiverList>,
206    /// Candidate assignment receivers for each partition.
207    candidate_receivers: Mutex<CandidateReceiverList>,
208    /// Metrics for each partition.
209    /// The scanner only sets in query and keeps it empty during compaction.
210    metrics_list: Arc<PartitionMetricsList>,
211    /// Memoized result of [`Self::prepare_raw_scan`] for the label binding it was computed for.
212    ///
213    /// Preflight only reads pinned scanner state, and one execution asks for it once per output
214    /// partition, so the first answer serves the rest. Errors are never stored, and every entry
215    /// that mutates the state the answer depends on clears the entry. The guard is taken and
216    /// released inside a synchronous call, so it never spans an await.
217    preflight: Mutex<PreflightCache>,
218    /// Test-only number of preflight computations performed by this scanner.
219    ///
220    /// A cached answer does not increment it, so a test can prove that a repeated preflight over
221    /// the same pinned state is served from [`Self::preflight`].
222    #[cfg(test)]
223    preflight_runs: AtomicUsize,
224}

228    pub(crate) fn new(input: ScanInput, experimental_series_scan_v2: bool) -> Self {
229        let mode = if experimental_series_scan_v2 && Self::supports_two_phase(&input) {
230            SeriesScanMode::TwoPhase
231        } else {
232            SeriesScanMode::Legacy
233        };
234        let mut properties = ScannerProperties::default()
235            .with_append_mode(input.append_mode)
236            .with_total_rows(input.total_rows())
237            .with_total_rows_is_exact(input.append_mode && input.total_rows_is_exact());
238        if let Some(counters) = input.query_stat_counters.clone() {
239            properties.set_query_stat_counters(counters);
240        }
241        let stream_ctx = Arc::new(StreamContext::seq_scan_ctx(input));
242        properties.partitions = vec![stream_ctx.partition_ranges()];
243
244        // Create the shared pruner with number of workers equal to CPU cores.
245        let num_workers = common_stat::get_total_cpu_cores().max(1);
246        let pruner = match mode {
247            SeriesScanMode::Legacy => Arc::new(Pruner::new(stream_ctx.clone(), num_workers)),
248            SeriesScanMode::TwoPhase => Arc::new(Pruner::new_with_options(
249                stream_ctx.clone(),
250                num_workers,
251                PrunerOptions {
252                    retain_builders: true,
253                    enable_predicate_prefilter: false,
254                },
255            )),
256        };
257
258        Self {
259            mode,
260            prepared: Arc::new(prepared::Preparation::default()),
261            properties,
262            stream_ctx,
263            pruner,
264            legacy_receivers: Mutex::new(Vec::new()),
265            candidate_receivers: Mutex::new(Vec::new()),
266            metrics_list: Arc::new(PartitionMetricsList::default()),
267            preflight: Mutex::new(None),
268            #[cfg(test)]
269            preflight_runs: AtomicUsize::new(0),
270        }
271    }

290    /// Returns the memoized preflight, if it was computed for the same label binding.
291    ///
292    /// A binding only matches when its immutable identity matches as well: the preflight validates
293    /// the pinned metadata snapshot and the label binding, so an answer for another binding must
294    /// never be reused even when the labels look alike.
295    fn cached_preflight(
296        &self,
297        label_source: &RawLabelSource,
298    ) -> Option<Option<(RawScanPreparation, bool)>> {
299        let cache = self.preflight.lock().unwrap();
300        let (cached_source, cached) = cache.as_ref()?;
301        if !prepared::same_labels(cached_source, label_source) {
302            return None;
303        }
304        Some(cached.clone())
305    }

315    /// Whether every selected *field* is a `Float64`, which is all the columnar prepared reader
316    /// serves.
317    ///
318    /// Only field columns count: the projection also carries tags and the time index, and those are
319    /// no part of the samples the prepared reader hands over. Checking them here would send an
320    /// ordinary `Float64` query down the ordinary scan just because it projects a timestamp.
321    fn columnar_fields_supported(&self) -> bool {
322        let metadata = self.stream_ctx.input.mapper.metadata();
323        let selected = self.stream_ctx.input.mapper.output_schema();
324        selected
325            .column_schemas()
326            .iter()
327            .filter(|output| {
328                metadata.column_metadatas.iter().any(|column| {
329                    column.column_schema.name == output.name
330                        && column.semantic_type == SemanticType::Field
331                })
332            })
333            .all(|output| output.data_type == ConcreteDataType::float64_datatype())
334    }

336    /// Returns whether raw series come from the columnar prepared source.
337    ///
338    /// Derived only from the pinned series index version and every configured partition range, so
339    /// preflight, the partition reader and scanner-wide preparation agree.
340    /// `false` means the stored sources are read by the ordinary scan.
341    fn uses_prepared_columnar(&self) -> Result<bool, BoxedError> {
342        if !self.uses_columnar() {
343            return Ok(false);
344        }
345        let ranges = self
346            .properties
347            .partitions
348            .iter()
349            .flatten()
350            .copied()
351            .collect::<Vec<_>>();
352        prepared::validate_inputs(&self.stream_ctx, &ranges).map_err(BoxedError::new)
353    }

355    /// Prepares the raw scan and reports whether its data comes from the columnar prepared source.
356    ///
357    /// The answer only depends on pinned scanner state plus the label binding, and an execution
358    /// asks for it once per output partition, so it is memoized per binding. Only successful
359    /// answers are stored: a rejected binding must fail on every call, not replay a rejection or
360    /// hide a later one.
361    fn prepare_raw_scan(
362        &self,
363        label_source: &RawLabelSource,
364    ) -> Result<Option<(RawScanPreparation, bool)>, BoxedError> {
365        if let Some(cached) = self.cached_preflight(label_source) {
366            return Ok(cached);
367        }
368        let computed = self.prepare_raw_scan_uncached(label_source)?;
369        // Both sides of the cache are cheap to clone, and callers own their copy: the merge schema
370        // helper rewrites the field indices and predicates of the preparation it is given, which
371        // must never reach the stored answer.
372        *self.preflight.lock().unwrap() = Some((label_source.clone(), computed.clone()));
373        Ok(computed)
374    }

376    /// Computes a fresh [`Self::prepare_raw_scan`] answer from the pinned scanner state.
377    fn prepare_raw_scan_uncached(
378        &self,
379        label_source: &RawLabelSource,
380    ) -> Result<Option<(RawScanPreparation, bool)>, BoxedError> {
381        #[cfg(test)]
382        self.preflight_runs.fetch_add(1, Ordering::Relaxed);
383        let input = &self.stream_ctx.input;
384        let metadata = input.mapper.metadata();
385        match label_source {
386            RawLabelSource::PrimaryKey {
387                physical_metadata,
388                logical_tags,
389                ..
390            } => {
391                if !Arc::ptr_eq(metadata, physical_metadata) {
392                    debug!(
393                        "Raw series scan preflight rejected metadata snapshot: region_id={}, physical_region_id={}, schema_version={}, physical_schema_version={}",
394                        metadata.region_id,
395                        physical_metadata.region_id,
396                        metadata.schema_version,
397                        physical_metadata.schema_version,
398                    );
399                    return Err(BoxedError::new(
400                        InvalidRecordBatchSnafu {
401                            reason: "raw label source metadata does not match scanner snapshot"
402                                .to_string(),
403                        }
404                        .build(),
405                    ));
406                }
407                let mut ids = HashSet::new();
408                let mut names = HashSet::new();
409                for tag in logical_tags.iter() {
410                    let column = metadata.column_by_id(tag.column_id).filter(|column| {
411                        column.semantic_type == SemanticType::Tag
412                            && metadata.primary_key.contains(&tag.column_id)
413                    });
414                    let valid = column.is_some_and(|column| {
415                        column.column_schema.name == tag.name.as_ref()
416                            && tag.name.as_ref() != "__name__"
417                            && !is_internal_column(tag.name.as_ref())
418                            && !is_metric_engine_internal_column(tag.name.as_ref())
419                            && !ReservedColumnId::is_reserved(tag.column_id)
420                    });
421                    let unique_id = valid && ids.insert(tag.column_id);
422                    let unique_name = unique_id && names.insert(tag.name.as_ref());
423                    if !valid || !unique_id || !unique_name {
424                        debug!(
425                            "Raw series scan preflight rejected label binding: region_id={}, column_id={}, binding_valid={}, unique_id={}, unique_name={}",
426                            metadata.region_id, tag.column_id, valid, unique_id, unique_name,
427                        );
428                        return Err(BoxedError::new(
429                            InvalidRecordBatchSnafu {
430                                reason: "invalid raw label binding".to_string(),
431                            }
432                            .build(),
433                        ));
434                    }
435                }
436            }
437            RawLabelSource::Explicit => {
438                debug!(
439                    "Raw series scan preflight rejected explicit labels: region_id={}",
440                    metadata.region_id,
441                );
442                return Ok(None);
443            }
444        }
445        if self.uses_columnar() && !is_sparse_metric_metadata(metadata) {
446            return Ok(None);
447        }
448        let selected = input.mapper.output_schema();
449        let mut fields = Vec::new();
450        let mut specs = Vec::new();
451        // The columnar prepared reader serves `Float64` fields only. A selection carrying another
452        // kind the raw path supports - a native histogram struct, today - is read by the ordinary
453        // scan instead: the columnar samples are an artifact beside the SST rows, so the same
454        // sources are still reachable, and the frame schema and ordering follow the reader that
455        // actually answers.
456        let columnar_fields_supported = self.columnar_fields_supported();
457        for output in selected.column_schemas() {
458            let Some(column) = metadata.column_metadatas.iter().find(|column| {
459                column.column_schema.name == output.name
460                    && column.semantic_type == SemanticType::Field
461            }) else {
462                continue;
463            };
464            let Some((index, (_, data_type))) = input
465                .mapper
466                .batch_schema()
467                .iter()
468                .enumerate()
469                .find(|(_, (id, _))| *id == column.column_id)
470            else {
471                debug!(
472                    "Raw series scan preflight missing mapper field: region_id={}, column_id={}",
473                    metadata.region_id, column.column_id,
474                );
475                return Err(BoxedError::new(
476                    InvalidRecordBatchSnafu {
477                        reason: format!(
478                            "selected field {} is missing from mapper batch schema",
479                            output.name
480                        ),
481                    }
482                    .build(),
483                ));
484            };
485            // The raw converter casts a numeric field to `Float64` and converts a native histogram
486            // struct into histogram samples; nothing else has a raw representation.
487            if !data_type.is_numeric() && !is_native_histogram_value_type(data_type) {
488                debug!(
489                    "Raw series scan preflight rejected field type: region_id={}, column_id={}",
490                    metadata.region_id, column.column_id,
491                );
492                return Ok(None);
493            }
494            fields.push((index, data_type.clone()));
495            specs.push(FieldSpec {
496                name: Arc::from(output.name.as_str()),
497            });
498        }
499        let time_is_millis = matches!(metadata.time_index_type(), TimestampType::Millisecond(_));
500        if fields.is_empty()
501            || input.append_mode
502            || input.series_row_selector.is_some()
503            || !time_is_millis
504        {
505            debug!(
506                "Raw series scan preflight rejected scan options: region_id={}, selected_fields={}, append_mode={}, row_selector={}, time_is_millis={}",
507                metadata.region_id,
508                fields.len(),
509                input.append_mode,
510                input.series_row_selector.is_some(),
511                time_is_millis,
512            );
513            return Ok(None);
514        }
515        // Structural eligibility errors still reject the raw path; ordinary scans cover sources
516        // without columnar artifacts, so missing coverage is not an error here.
517        let prepared_columnar = self.uses_prepared_columnar()? && columnar_fields_supported;
518        let schema = Arc::new(SeriesSchema {
519            fields: specs.into(),
520            eval_grid: None,
521        });
522        let typed_schema = input.mapper.input_arrow_schema(false);
523        let mut predicates = Vec::new();
524        let predicate = if prepared_columnar {
525            input.predicate_group().predicate()
526        } else {
527            // The ordinary scan applies region predicates before this residual filtering.
528            input.predicate_group().predicate_without_region()
529        };
530        if let Some(predicate) = predicate {
531            if !predicate.dyn_filters().is_empty() {
532                debug!(
533                    "Raw series scan preflight rejected dynamic predicate filters: region_id={}",
534                    metadata.region_id,
535                );
536                return Ok(None);
537            }
538            for expr in predicate.exprs() {
539                // Fresh execution properties cannot preserve Stable or query-dependent semantics.
540                if expr
541                    .exists(|node| {
542                        Ok(matches!(node, Expr::ScalarFunction(function)
543                            if function.func.signature().volatility != Volatility::Immutable)
544                            || matches!(node, Expr::ScalarVariable(..)))
545                    })
546                    .unwrap_or(true)
547                {
548                    debug!(
549                        "Raw series scan preflight rejected predicate volatility or variable: region_id={}",
550                        metadata.region_id,
551                    );
552                    return Ok(None);
553                }
554                // Evaluate residuals with original types before raw conversion to Float64.
555                let Ok(physical) = Predicate::to_physical_expr(expr, &typed_schema) else {
556                    debug!(
557                        "Raw series scan preflight failed predicate compilation: region_id={}",
558                        metadata.region_id,
559                    );
560                    return Ok(None);
561                };
562                if !matches!(
563                    physical.data_type(&typed_schema),
564                    Ok(ArrowDataType::Boolean | ArrowDataType::Null)
565                ) {
566                    debug!(
567                        "Raw series scan preflight rejected predicate type: region_id={}",
568                        metadata.region_id,
569                    );
570                    return Ok(None);
571                }
572                predicates.push(physical);
573            }
574        }
575        Ok(Some((
576            RawScanPreparation {
577                schema,
578                fields,
579                predicates,
580            },
581            prepared_columnar,
582        )))
583    }

585    /// Starts scanner-wide preparation from the preflight already performed by the caller.
586    fn prepare_series_with_raw(
587        &self,
588        ctx: &QueryScanContext,
589        metrics_set: &ExecutionPlanMetricsSet,
590        label_source: &RawLabelSource,
591        preparation: Result<Option<(RawScanPreparation, bool)>, BoxedError>,
592    ) -> futures::future::BoxFuture<
593        'static,
594        Result<Option<Arc<dyn store_api::region_engine::PreparedSeriesSource>>, BoxedError>,
595    > {
596        let state = self.prepared.clone();
597        let stream_ctx = self.stream_ctx.clone();
598        let ranges = self
599            .properties
600            .partitions
601            .iter()
602            .flatten()
603            .copied()
604            .collect();
605        let labels = label_source.clone();
606        let metrics = new_partition_metrics(
607            &self.stream_ctx,
608            ctx.explain_verbose,
609            metrics_set,
610            0,
611            &self.metrics_list,
612        );
613        Box::pin(async move {
614            let Some((preparation, prepared_columnar)) = preparation? else {
615                return Ok(None);
616            };
617            if !prepared_columnar {
618                // Known unsupported coverage: the caller keeps the ordinary scan instead.
619                return Ok(None);
620            }
621            state
622                .prepare(stream_ctx, ranges, labels, preparation, metrics)
623                .await
624                .map(Some)
625        })
626    }

720    fn scan_legacy_batch_in_partition(
721        &self,
722        partition: usize,
723        part_metrics: PartitionMetrics,
724        metrics_set: &ExecutionPlanMetricsSet,
725        explain_verbose: bool,
726    ) -> Result<ScanBatchStream> {
727        self.maybe_start_legacy_distributor(metrics_set, explain_verbose);
728        let mut receiver = self.legacy_receivers.lock().unwrap()[partition]
729            .take()
730            .context(ScanMultiTimesSnafu { partition })?;
731        let stream = try_stream! {
732            part_metrics.on_first_poll();
733
734            let mut fetch_start = Instant::now();
735            let mut metrics = ScannerMetrics::default();
736            while let Some(series) = receiver.recv().await {
737                let series = series?;
738                metrics.scan_cost += fetch_start.elapsed();
739                metrics.num_batches += series.num_batches();
740                metrics.num_rows += series.num_rows();
741
742                let yield_start = Instant::now();
743                yield ScanBatch::Series(series);
744                metrics.yield_cost += yield_start.elapsed();
745                fetch_start = Instant::now();
746            }
747
748            part_metrics.merge_metrics(&metrics);
749            part_metrics.on_finish();
750        };
751        Ok(Box::pin(stream))
752    }

754    fn scan_two_phase_batch_in_partition(
755        &self,
756        ctx: &QueryScanContext,
757        partition: usize,
758        part_metrics: PartitionMetrics,
759        metrics_set: &ExecutionPlanMetricsSet,
760    ) -> Result<ScanBatchStream> {
761        self.maybe_start_candidate_distributor(ctx, metrics_set);
762        // Safety: `scan_batch_in_partition` validates the index, and the receiver
763        // list always contains one entry for each output partition.
764        let mut receiver = self.candidate_receivers.lock().unwrap()[partition]
765            .take()
766            .context(ScanMultiTimesSnafu { partition })?;
767        let stream_ctx = self.stream_ctx.clone();
768        let partition_ranges = self
769            .properties
770            .partitions
771            .iter()
772            .flatten()
773            .copied()
774            .collect::<Vec<_>>();
775        let stream = try_stream! {
776            part_metrics.on_first_poll();
777
778            let mut fetch_start = Instant::now();
779            let mut metrics = ScannerMetrics::default();
780            while let Some(input) = receiver.recv().await {
781                let input = input?;
782                metrics.scan_cost += fetch_start.elapsed();
783
784                let build_start = Instant::now();
785                let reader = SeriesReader::try_new(
786                    stream_ctx.clone(),
787                    partition_ranges.clone(),
788                    input.assigned_series,
789                    input.partition_pruner,
790                    input.range_semaphore,
791                    part_metrics.clone(),
792                )?;
793                let mut reader_stream = reader.build_stream().await?;
794                metrics.scan_cost += build_start.elapsed();
795                fetch_start = Instant::now();
796                while let Some(record_batch) = reader_stream.try_next().await? {
797                    metrics.scan_cost += fetch_start.elapsed();
798                    metrics.num_batches += 1;
799                    metrics.num_rows += record_batch.num_rows();
800
801                    let yield_start = Instant::now();
802                    yield ScanBatch::RecordBatch(record_batch);
803                    metrics.yield_cost += yield_start.elapsed();
804                    fetch_start = Instant::now();
805                }
806            }
807            metrics.scan_cost += fetch_start.elapsed();
808
809            part_metrics.merge_metrics(&metrics);
810            part_metrics.on_finish();
811        };
812        Ok(Box::pin(stream))
813    }

925    /// Checks resource limit for the scanner.
926    pub(crate) fn check_scan_limit(&self) -> Result<()> {
927        // Sum the total number of files across all partitions
928        let total_files: usize = self
929            .properties
930            .partitions
931            .iter()
932            .flat_map(|partition| partition.iter())
933            .map(|part_range| {
934                let range_meta = &self.stream_ctx.ranges[part_range.identifier];
935                range_meta.indices.len()
936            })
937            .sum();
938
939        if !self.stream_ctx.input.compaction {
940            SCAN_MAX_CONCURRENT_FILES
941                .with_label_values(&["query"])
942                .observe(total_files as f64);
943        }
944        if self.stream_ctx.input.uses_scan_memory_limit() {
945            return Ok(());
946        }
947
948        let max_concurrent_files = self.stream_ctx.input.max_concurrent_scan_files;
949        if total_files > max_concurrent_files {
950            return TooManyFilesToReadSnafu {
951                actual: total_files,
952                max: max_concurrent_files,
953            }
954            .fail();
955        }
956
957        Ok(())
958    }

961fn new_legacy_channel_list(num_partitions: usize) -> (SenderList, LegacyReceiverList) {
962    let (senders, receivers): (Vec<_>, Vec<_>) = (0..num_partitions)
963        .map(|_| {
964            let (sender, receiver) = mpsc::channel(1);
965            (Some(sender), Some(receiver))
966        })
967        .unzip();
968    (SenderList::new(senders), receivers)
969}
970
971fn new_candidate_channel_list(
972    num_partitions: usize,
973) -> (CandidateSenderList, CandidateReceiverList, Arc<AtomicUsize>) {
974    let active_receivers = Arc::new(AtomicUsize::new(num_partitions));
975    let (senders, receivers) = (0..num_partitions)
976        .map(|_| {
977            // Partition streams can be consumed sequentially. A bounded channel for an unpolled
978            // partition could block the distributor and prevent it from sending more assignments
979            // to the partition currently being consumed, causing a deadlock.
980            let (sender, receiver) = mpsc::unbounded_channel();
981            let receiver = CandidateReceiver {
982                receiver,
983                active_receivers: active_receivers.clone(),
984            };
985            (Some(sender), Some(receiver))
986        })
987        .unzip();
988    (senders, receivers, active_receivers)
989}

1042    fn scan_series_partition(
1043        &self,
1044        ctx: &QueryScanContext,
1045        metrics_set: &ExecutionPlanMetricsSet,
1046        partition: usize,
1047        label_source: &RawLabelSource,
1048    ) -> Result<Option<SendableRawSeriesStream>, BoxedError> {
1049        let Some((preparation, prepared_columnar)) = self.prepare_raw_scan(label_source)? else {
1050            return Ok(None);
1051        };
1052        if prepared_columnar {
1053            let partitions = self.properties.num_partitions().max(1);
1054            if partition >= partitions {
1055                return Err(BoxedError::new(
1056                    PartitionOutOfRangeSnafu {
1057                        given: partition,
1058                        all: partitions,
1059                    }
1060                    .build(),
1061                ));
1062            }
1063            // Preflight needs the declaration before the stream is polled, while preparation
1064            // needs the original field indices and predicates to derive its own merge schema.
1065            let explicit_labels =
1066                prepared::sample_merge_schema(&self.stream_ctx, &mut preparation.clone())
1067                    .map_err(BoxedError::new)?
1068                    .is_some();
1069            let schema = preparation.schema.clone();
1070            let preparation = self
1071                .check_scan_limit()
1072                .map_err(BoxedError::new)
1073                .map(|_| Some((preparation, prepared_columnar)));
1074            let future = self.prepare_series_with_raw(ctx, metrics_set, label_source, preparation);
1075            let labels = label_source.clone();
1076            let stream = async_stream::try_stream! {
1077                let source = future.await?.ok_or_else(|| BoxedError::new(InvalidRecordBatchSnafu { reason: "columnar prepared capability disappeared" }.build()))?;
1078                let n = source.num_series() as u128;
1079                let start = (n * partition as u128 / partitions as u128) as usize;
1080                let end = (n * (partition + 1) as u128 / partitions as u128) as usize;
1081                let mut reader = source.read_series(start..end)?;
1082                while let Some(chunk) = reader.try_next().await? { yield chunk; }
1083            };
1084            return Ok(Some(raw::from_stream(
1085                schema,
1086                labels,
1087                SeriesOrdering::KeyThenTime,
1088                explicit_labels,
1089                Box::pin(stream),
1090            )));
1091        }
1092        // Stored sources without columnar artifacts (memtables, unindexed SSTs) are read by the
1093        // ordinary scan, so the raw path stays available.
1094        let metrics = new_partition_metrics(
1095            &self.stream_ctx,
1096            ctx.explain_verbose,
1097            metrics_set,
1098            partition,
1099            &self.metrics_list,
1100        );
1101        let batches = self
1102            .scan_batch_in_partition(ctx, partition, metrics.clone(), metrics_set)
1103            .map_err(BoxedError::new)?;
1104        Ok(Some(raw::make_stream(
1105            batches,
1106            preparation.schema,
1107            label_source.clone(),
1108            preparation.fields,
1109            metrics,
1110            preparation.predicates,
1111            SeriesOrdering::KeyThenTime,
1112        )))
1113    }

1261    /// This may double the scan cost of the [SeriesScan] metrics. We can
1262    /// get per-partition metrics in verbose mode to see the metrics of the
1263    /// distributor.

1266    /// Whether to use verbose logging and collect detailed metrics.

1275    )]
1276    async fn execute(&mut self) {
1277        if let Err(e) = self.scan_partitions_flat().await {
1278            self.senders.send_error(e).await;
1279        }
1280    }
1281
1282    /// Scans all parts in flat format using FlatSeriesBatchDivider.
1283    #[tracing::instrument(
1284        skip_all,
1285        fields(region_id = %self.stream_ctx.input.mapper.metadata().region_id)
1286    )]
1287    async fn scan_partitions_flat(&mut self) -> Result<()> {
1288        // Initialize reference counts for all partition ranges.
1289        for partition_ranges in &self.partitions {
1290            self.pruner.add_partition_ranges(partition_ranges);
1291        }
1292
1293        // Create PartitionPruner covering all partitions
1294        let all_partition_ranges: Vec<_> = self.partitions.iter().flatten().cloned().collect();
1295        let needs_merge = all_partition_ranges.len() > 1;
1296        let partition_pruner = Arc::new(PartitionPruner::new(
1297            self.pruner.clone(),
1298            &all_partition_ranges,
1299        ));
1300
1301        let part_metrics = new_partition_metrics(
1302            &self.stream_ctx,
1303            self.explain_verbose,
1304            &self.metrics_set,
1305            self.partitions.len(),
1306            &self.metrics_list,
1307        );
1308        part_metrics.on_first_poll();
1309        // Start fetch time before building sources so scan cost contains
1310        // build part cost.
1311        let mut fetch_start = Instant::now();
1312
1313        // Builds one deduped stream per partition range, then merges across ranges.
1314        let build_start = Instant::now();
1315        let mut tasks = Vec::new();
1316        for partition in &self.partitions {
1317            for part_range in partition {
1318                let stream_ctx = self.stream_ctx.clone();
1319                let part_range = *part_range;
1320                let part_metrics = part_metrics.clone();
1321                let partition_pruner = partition_pruner.clone();
1322                let file_scan_semaphore = self.range_semaphore.clone();
1323                let merge_semaphore = self.range_semaphore.clone();
1324                tasks.push(common_runtime::spawn_query(async move {
1325                    SeqScan::build_flat_partition_range_read(
1326                        &stream_ctx,
1327                        &part_range,
1328                        if needs_merge {
1329                            FlatReadMode::CrossRangeMerge
1330                        } else {
1331                            FlatReadMode::Sequential
1332                        },
1333                        &part_metrics,
1334                        partition_pruner,
1335                        file_scan_semaphore,
1336                        merge_semaphore,
1337                    )
1338                    .await
1339                }));
1340            }
1341        }
1342        let mut range_streams = Vec::with_capacity(tasks.len());
1343        let mut estimated_batch_sizes = Vec::with_capacity(tasks.len());
1344        for task in tasks {
1345            let (stream, estimated_batch_size) = task.await.context(JoinSnafu)??;
1346            range_streams.push(stream);
1347            estimated_batch_sizes.push(estimated_batch_size);
1348        }
1349        let channel_size =
1350            compute_parallel_channel_size(compute_average_batch_size(estimated_batch_sizes));
1351        common_telemetry::debug!(
1352            "SeriesDistributor built {} range_streams, region: {}, build cost: {:?}, channel_size: {}",
1353            range_streams.len(),
1354            self.stream_ctx.input.region_metadata().region_id,
1355            build_start.elapsed(),
1356            channel_size,
1357        );
1358
1359        // Each partition range stream is already deduped, so skip dedup here.
1360        // Use a separate semaphore for the final merge to avoid deadlock with
1361        // range-level merge tasks that share the range_semaphore.
1362        let mut reader = SeqScan::build_flat_reader_from_sources(
1363            &self.stream_ctx,
1364            range_streams,
1365            self.final_merge_semaphore.clone(),
1366            Some(&part_metrics),
1367            true,
1368            channel_size,
1369        )
1370        .await?;
1371        let mut metrics = SeriesDistributorMetrics::default();
1372
1373        let mut divider = FlatSeriesBatchDivider::default();
1374        while let Some(record_batch) = reader.try_next().await? {
1375            metrics.scan_cost += fetch_start.elapsed();
1376            metrics.num_batches += 1;
1377            metrics.num_rows += record_batch.num_rows();
1378
1379            debug_assert!(record_batch.num_rows() > 0);
1380            if record_batch.num_rows() == 0 {
1381                fetch_start = Instant::now();
1382                continue;
1383            }
1384
1385            // Use divider to split series
1386            let divider_start = Instant::now();
1387            let series_batch = divider.push(record_batch);
1388            metrics.divider_cost += divider_start.elapsed();
1389            if let Some(series_batch) = series_batch {
1390                let yield_start = Instant::now();
1391                self.senders
1392                    .send_batch(SeriesBatch::Flat(series_batch))
1393                    .await?;
1394                metrics.yield_cost += yield_start.elapsed();
1395            }
1396            fetch_start = Instant::now();
1397        }
1398
1399        // Send any remaining batch in the divider
1400        let divider_start = Instant::now();
1401        let series_batch = divider.finish();
1402        metrics.divider_cost += divider_start.elapsed();
1403        if let Some(series_batch) = series_batch {
1404            let yield_start = Instant::now();
1405            self.senders
1406                .send_batch(SeriesBatch::Flat(series_batch))
1407                .await?;
1408            metrics.yield_cost += yield_start.elapsed();
1409        }
1410
1411        metrics.scan_cost += fetch_start.elapsed();
1412        metrics.num_series_send_full = self.senders.num_full;
1413        part_metrics.set_distributor_metrics(&metrics);
1414
1415        part_metrics.on_finish();
1416
1417        Ok(())
1418    }
1419}

1421/// Discovers metric series and sends one reader assignment to each partition.
1422struct SeriesCandidateDistributor {
1423    memory_pool: Arc<dyn MemoryPool>,
1424    stream_ctx: Arc<StreamContext>,
1425    range_semaphore: Arc<Semaphore>,
1426    partitions: Vec<Vec<PartitionRange>>,
1427    pruner: Arc<Pruner>,
1428    senders: CandidateSenderList,
1429    active_receivers: Arc<AtomicUsize>,
1430    metrics_set: ExecutionPlanMetricsSet,
1431    metrics_list: Arc<PartitionMetricsList>,
1432    explain_verbose: bool,
1433}
1434
1435impl SeriesCandidateDistributor {
1436    async fn execute(&mut self) {
1437        if let Err(e) = self.distribute().await {
1438            self.send_error(e);
1439        }
1440    }
1441
1442    async fn distribute(&mut self) -> Result<()> {
1443        let part_metrics = new_partition_metrics(
1444            &self.stream_ctx,
1445            self.explain_verbose,
1446            &self.metrics_set,
1447            self.partitions.len(),
1448            &self.metrics_list,
1449        );
1450        part_metrics.on_first_poll();
1451
1452        let candidate_scanner = SeriesCandidateScanner::try_new(
1453            self.stream_ctx.clone(),
1454            self.partitions.clone(),
1455            self.pruner.clone(),
1456            self.range_semaphore.clone(),
1457            self.memory_pool.clone(),
1458            self.metrics_set.clone(),
1459            part_metrics.clone(),
1460        )?;
1461        let partition_pruner = candidate_scanner.partition_pruner();
1462        let mut candidates = candidate_scanner.build_stream().await?;
1463        let mut collector =
1464            SeriesBatchCollector::new(self.partitions.len()).context(InvalidSenderSnafu)?;
1465        let mut chunked = false;
1466        while let Some(batch) = candidates.try_next().await? {
1467            if !self.should_fetch_candidates() {
1468                part_metrics.on_finish();
1469                return Ok(());
1470            }
1471            collector.push(batch);
1472            if collector.len() >= candidate_series_assignment_threshold() {
1473                chunked = true;
1474                self.send_assignments(collector.finish(false), &partition_pruner);
1475                if !self.should_fetch_candidates() {
1476                    part_metrics.on_finish();
1477                    return Ok(());
1478                }
1479                collector =
1480                    SeriesBatchCollector::new(self.partitions.len()).context(InvalidSenderSnafu)?;
1481            }
1482        }
1483
1484        if collector.len() > 0 {
1485            self.send_assignments(collector.finish(!chunked), &partition_pruner);
1486        }
1487        part_metrics.on_finish();
1488        Ok(())
1489    }
1490
1491    fn send_assignments(
1492        &mut self,
1493        assignments: Vec<AssignedSeriesBatch>,
1494        partition_pruner: &Arc<PartitionPruner>,
1495    ) {
1496        // One round of candidate collection; a partition that receives two assignments sees two.
1497        #[cfg(test)]
1498        CANDIDATE_ASSIGNMENT_ROUNDS.fetch_add(1, Ordering::Relaxed);
1499        for (partition, assigned_series) in assignments.into_iter().enumerate() {
1500            if assigned_series.series().is_empty() {
1501                continue;
1502            }
1503            let Some(sender) = self.senders[partition].as_ref() else {
1504                continue;
1505            };
1506            let sent = sender
1507                .send(Ok(SeriesReaderInput {
1508                    assigned_series,
1509                    partition_pruner: partition_pruner.clone(),
1510                    range_semaphore: self.range_semaphore.clone(),
1511                }))
1512                .is_ok();
1513            if !sent {
1514                self.senders[partition] = None;
1515            }
1516        }
1517    }
1518
1519    fn should_fetch_candidates(&self) -> bool {
1520        self.active_receivers.load(Ordering::Relaxed) > 0
1521    }
1522
1523    fn send_error(&mut self, error: Error) {
1524        let error = Arc::new(error);
1525        for sender in self.senders.iter_mut().filter_map(Option::take) {
1526            let result = Err(error.clone()).context(ScanSeriesSnafu);
1527            let _ = sender.send(result);
1528        }
1529    }
1530}

1563/// List of senders.
1564struct SenderList {
1565    senders: Vec<Option<Sender<Result<SeriesBatch>>>>,
1566    /// Number of None senders.
1567    num_nones: usize,
1568    /// Index of the current partition to send.
1569    sender_idx: usize,
1570    /// Number of full senders.
1571    num_full: usize,
1572}
1573
1574impl SenderList {
1575    fn new(senders: Vec<Option<Sender<Result<SeriesBatch>>>>) -> Self {
1576        let num_nones = senders.iter().filter(|sender| sender.is_none()).count();
1577        Self {
1578            senders,
1579            num_nones,
1580            sender_idx: 0,
1581            num_full: 0,
1582        }
1583    }
1584
1585    /// Finds a partition and tries to send the batch to the partition.
1586    /// Returns None if it sends successfully.
1587    fn try_send_batch(&mut self, mut batch: SeriesBatch) -> Result<Option<SeriesBatch>> {
1588        for _ in 0..self.senders.len() {
1589            ensure!(self.num_nones < self.senders.len(), InvalidSenderSnafu);
1590
1591            let sender_idx = self.fetch_add_sender_idx();
1592            let Some(sender) = &self.senders[sender_idx] else {
1593                continue;
1594            };
1595
1596            match sender.try_send(Ok(batch)) {
1597                Ok(()) => return Ok(None),
1598                Err(TrySendError::Full(res)) => {
1599                    self.num_full += 1;
1600                    // Safety: we send Ok.
1601                    batch = res.unwrap();
1602                }
1603                Err(TrySendError::Closed(res)) => {
1604                    self.senders[sender_idx] = None;
1605                    self.num_nones += 1;
1606                    // Safety: we send Ok.
1607                    batch = res.unwrap();
1608                }
1609            }
1610        }
1611
1612        Ok(Some(batch))
1613    }
1614
1615    /// Finds a partition and sends the batch to the partition.
1616    async fn send_batch(&mut self, batch: SeriesBatch) -> Result<()> {
1617        let Some(batch) = self.try_send_batch(batch)? else {
1618            return Ok(());
1619        };
1620
1621        loop {
1622            ensure!(self.num_nones < self.senders.len(), InvalidSenderSnafu);
1623
1624            // Waits on every partition at once and sends to whichever frees a slot first. A
1625            // consumer may leave partitions unpolled for a long time, e.g. one that drains them one
1626            // after another, and waiting on such a partition would stall the scan.
1627            let closed = {
1628                let reservations = self.senders.iter().enumerate().filter_map(|(idx, sender)| {
1629                    sender
1630                        .as_ref()
1631                        .map(|sender| Box::pin(async move { (idx, sender.reserve().await) }))
1632                });
1633                let ((idx, reserved), _, _) = futures::future::select_all(reservations).await;
1634                match reserved {
1635                    Ok(permit) => {
1636                        permit.send(Ok(batch));
1637                        return Ok(());
1638                    }
1639                    Err(_) => idx,
1640                }
1641            };
1642            self.senders[closed] = None;
1643            self.num_nones += 1;
1644        }
1645    }
1646
1647    async fn send_error(&self, error: Error) {
1648        let error = Arc::new(error);
1649        for sender in self.senders.iter().flatten() {
1650            let result = Err(error.clone()).context(ScanSeriesSnafu);
1651            let _ = sender.send(result).await;
1652        }
1653    }
1654
1655    fn fetch_add_sender_idx(&mut self) -> usize {
1656        let sender_idx = self.sender_idx;
1657        self.sender_idx = (self.sender_idx + 1) % self.senders.len();
1658        sender_idx
1659    }
1660}

1682/// A divider to split flat record batches by time series.
1683///
1684/// It only ensures rows of the same series are returned in the same [FlatSeriesBatch].
1685/// However, a [FlatSeriesBatch] may contain rows from multiple series.
1686#[derive(Default)]
1687struct FlatSeriesBatchDivider {
1688    buffer: FlatSeriesBatch,
1689}
1690
1691impl FlatSeriesBatchDivider {
1692    /// Pushes a record batch into the divider.
1693    ///
1694    /// Returns a [FlatSeriesBatch] if we ensure the batch contains all rows of the series in it.
1695    fn push(&mut self, batch: RecordBatch) -> Option<FlatSeriesBatch> {
1696        // If buffer is empty
1697        if self.buffer.batches.is_empty() {
1698            self.buffer.batches.push(batch);
1699            return None;
1700        }
1701
1702        // Gets the primary key column from the incoming batch.
1703        let pk_column_idx = primary_key_column_index(batch.num_columns());
1704        let batch_pk_column = batch.column(pk_column_idx);
1705        let batch_pk_array = batch_pk_column
1706            .as_any()
1707            .downcast_ref::<PrimaryKeyArray>()
1708            .unwrap();
1709        let batch_pk_values = batch_pk_array
1710            .values()
1711            .as_any()
1712            .downcast_ref::<BinaryArray>()
1713            .unwrap();
1714        // Gets the last primary key of the incoming batch.
1715        let batch_last_pk =
1716            primary_key_at(batch_pk_array, batch_pk_values, batch_pk_array.len() - 1);
1717        // Gets the last primary key of the buffer.
1718        // Safety: the buffer is not empty.
1719        let buffer_last_batch = self.buffer.batches.last().unwrap();
1720        let buffer_pk_column = buffer_last_batch.column(pk_column_idx);
1721        let buffer_pk_array = buffer_pk_column
1722            .as_any()
1723            .downcast_ref::<PrimaryKeyArray>()
1724            .unwrap();
1725        let buffer_pk_values = buffer_pk_array
1726            .values()
1727            .as_any()
1728            .downcast_ref::<BinaryArray>()
1729            .unwrap();
1730        let buffer_last_pk =
1731            primary_key_at(buffer_pk_array, buffer_pk_values, buffer_pk_array.len() - 1);
1732
1733        // If last primary key in the batch is the same as last primary key in the buffer.
1734        if batch_last_pk == buffer_last_pk {
1735            self.buffer.batches.push(batch);
1736            return None;
1737        }
1738        // Otherwise, the batch must have a different primary key, we find the first offset of the
1739        // changed primary key.
1740        let batch_pk_keys = batch_pk_array.keys();
1741        let pk_indices = batch_pk_keys.values();
1742        let mut change_offset = 0;
1743        for (i, &key) in pk_indices.iter().enumerate() {
1744            let batch_pk = batch_pk_values.value(key as usize);
1745
1746            if buffer_last_pk != batch_pk {
1747                change_offset = i;
1748                break;
1749            }
1750        }
1751
1752        // Splits the batch at the change offset
1753        let (first_part, remaining_part) = if change_offset > 0 {
1754            let first_part = batch.slice(0, change_offset);
1755            let remaining_part = batch.slice(change_offset, batch.num_rows() - change_offset);
1756            (Some(first_part), Some(remaining_part))
1757        } else {
1758            (None, Some(batch))
1759        };
1760
1761        // Creates the result from current buffer + first part of new batch
1762        let mut result = std::mem::take(&mut self.buffer);
1763        if let Some(first_part) = first_part {
1764            result.batches.push(first_part);
1765        }
1766
1767        // Pushes remaining part to the buffer if it exists
1768        if let Some(remaining_part) = remaining_part {
1769            self.buffer.batches.push(remaining_part);
1770        }
1771
1772        Some(result)
1773    }
1774
1775    /// Returns the final [FlatSeriesBatch].
1776    fn finish(&mut self) -> Option<FlatSeriesBatch> {
1777        if self.buffer.batches.is_empty() {
1778            None
1779        } else {
1780            Some(std::mem::take(&mut self.buffer))
1781        }
1782    }
1783}

1789    index: usize,
