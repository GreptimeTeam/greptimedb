60/// Outcome of admitting a region edit, independent of subsequent region role changes.
61pub(crate) enum RegionEditDispatch {
62    Started,
63    Queued,
64    Rejected,
65}
66
67/// A queue for region edits waiting behind an active edit or ALTER.
68///
69/// Normal writes and bulk inserts that arrive during `Editing` use the stalled-write queue instead.
70/// When an edit completes, those writes are handled before the next queued edit starts, preserving
71/// sequence ordering between direct SST edits and WAL/memtable writes unless global reject
72/// backpressure rejects them first.
73/// Everything is done in the region worker loop.
74pub(crate) struct RegionEditQueue {
75    region_id: RegionId,
76    requests: VecDeque<RegionEditRequest>,
77}
78
79impl RegionEditQueue {
80    const QUEUE_MAX_LEN: usize = 128;
81
82    fn new(region_id: RegionId) -> Self {
83        Self {
84            region_id,
85            requests: VecDeque::new(),
86        }
87    }
88
89    fn enqueue(&mut self, request: RegionEditRequest) -> RegionEditDispatch {
90        if self.requests.len() >= Self::QUEUE_MAX_LEN {
91            request.waiters.reply_with(|| {
92                RegionBusySnafu {
93                    region_id: self.region_id,
94                }
95                .fail()
96            });
97            return RegionEditDispatch::Rejected;
98        };
99        self.requests.push_back(request);
100        RegionEditDispatch::Queued
101    }
102
103    fn dequeue(&mut self) -> Option<RegionEditRequest> {
104        fn can_merge(request: &RegionEditRequest) -> bool {
105            if !matches!(request.kind, RegionEditKind::Import) {
106                return false;
107            }
108            let edit = &request.edit;
109            // Only the `RegionEdit`:
110            // 1. contains the "raw" (file without a sequence) files to add,
111            // 2. and no `committed_sequence`,
112            // 3. and all other fields are empty,
113            // can it be merged.
114            //
115            // However, merging them means they will all share a same sequence, and if there are
116            // overlapping data in the files, the dedup is uncertain. This is a caution that must
117            // be noticed for editing region.
118            edit.files_to_add.iter().all(|f| f.sequence.is_none())
119                && edit.files_to_remove.is_empty()
120                && edit.timestamp_ms.is_none()
121                && edit.compaction_time_window.is_none()
122                && edit.flushed_entry_id.is_none()
123                && edit.flushed_sequence.is_none()
124                && edit.committed_sequence.is_none()
125                && edit.wal_recovery == WalRecoveryEdit::default()
126        }
127
128        let mut merged = self.requests.pop_front()?;
129        if !can_merge(&merged) {
130            return Some(merged);
131        }
132
133        // Merging must preserve the staging binding of every waiter.
134        while let Some(request) = self.requests.pop_front_if(|request| {
135            can_merge(request) && request.staging_partition_expr == merged.staging_partition_expr
136        }) {
137            merged.edit.files_to_add.extend(request.edit.files_to_add);
138            merged.waiters.merge(request.waiters);
139        }
140        debug!(
141            "the files to add: [{}] are merged in one edit",
142            merged
143                .edit
144                .files_to_add
145                .iter()
146                .map(|x| x.file_id.to_string())
147                .collect::<Vec<_>>()
148                .join(", ")
149        );
150        Some(merged)
151    }
152
153    fn is_empty(&self) -> bool {
154        self.requests.is_empty()
155    }
156
157    fn reject_all_as_not_found(mut self) {
158        while let Some(request) = self.requests.pop_front() {
159            request.waiters.reply_with(|| {
160                RegionNotFoundSnafu {
161                    region_id: self.region_id,
162                }
163                .fail()
164            });
165        }
166    }
167}

170    /// Rejects queued DDL and region edit requests as region not found.
171    pub(crate) fn reject_region_requests_as_not_found(&mut self, region_id: RegionId) {
172        if let Some(queue) = self.pending_ddls.remove(region_id) {
173            for ddl in queue {
174                ddl.sender.send(RegionNotFoundSnafu { region_id }.fail());
175            }
176        }
177        self.reject_region_edits_as_not_found(region_id);
178    }
179
180    /// Rejects edits belonging to an abandoned staging without dropping queued controls.
181    pub(crate) fn reject_region_edits_as_not_found(&mut self, region_id: RegionId) {
182        if let Some(edit_queue) = self.region_edit_queues.remove(&region_id) {
183            edit_queue.reject_all_as_not_found();
184        }
185    }

187    /// Handles region change result.
188    pub(crate) async fn handle_manifest_region_change_result(
189        &mut self,
190        change_result: RegionChangeResult,
191    ) {
192        let region = match self.regions.get_region(change_result.region_id) {
193            Some(region) => region,
194            None => {
195                self.reject_region_stalled_requests(&change_result.region_id);
196                change_result.sender.send(
197                    RegionNotFoundSnafu {
198                        region_id: change_result.region_id,
199                    }
200                    .fail(),
201                );
202                return;
203            }
204        };
205
206        if change_result.result.is_ok() {
207            // Updates the region metadata and format.
208            Self::update_region_version(
209                &region.version_control,
210                change_result.new_meta,
211                change_result.new_options,
212                &self.memtable_builder_provider,
213            );
214        }
215
216        // Sets the region as writable.
217        region.switch_state_to_writable(RegionLeaderState::Altering);
218        // Sends the result.
219        change_result.sender.send(change_result.result.map(|_| 0));
220
221        // In async mode, rebuild index after index metadata changed.
222        if self.config.index.build_mode == IndexBuildMode::Async && change_result.need_index {
223            self.handle_rebuild_index(
224                BuildIndexRequest {
225                    region_id: region.region_id,
226                    build_type: IndexBuildType::SchemaChange,
227                    file_metas: Vec::new(),
228                    result_senders: Vec::new(),
229                },
230                OptionOutputTx::new(None),
231            )
232            .await;
233        }
234        // Handles the stalled requests.
235        self.handle_region_stalled_requests(&change_result.region_id, true)
236            .await;
237    }

239    /// Handles region sync request.
240    ///
241    /// Updates the manifest to at least the given version.
242    /// **Note**: The installed version may be greater than the given version.
243    pub(crate) async fn handle_region_sync(&mut self, request: RegionSyncRequest) {
244        let region_id = request.region_id;
245        let sender = request.sender;
246        let region = match self.regions.follower_region(region_id) {
247            Ok(region) => region,
248            Err(e) => {
249                let _ = sender.send(Err(e));
250                return;
251            }
252        };
253
254        let original_manifest_version = region.manifest_ctx.manifest_version().await;
255        let manifest = match region
256            .manifest_ctx
257            .install_manifest_to(request.manifest_version)
258            .await
259        {
260            Ok(manifest) => manifest,
261            Err(e) => {
262                let _ = sender.send(Err(e));
263                return;
264            }
265        };
266        let version = region.version();
267        let mut region_options = version.options.clone();
268        let old_format = region_options.sst_format.unwrap_or_default();
269        // Updates the region options with the manifest.
270        sanitize_region_options(&manifest, &mut region_options);
271        if !version.memtables.is_empty() {
272            let current = region.version_control.current();
273            warn!(
274                "Region {} memtables is not empty, which should not happen, manifest version: {}, last entry id: {}",
275                region.region_id, manifest.manifest_version, current.last_entry_id
276            );
277        }
278
279        // We should sanitize the region options before creating a new memtable.
280        let memtable_builder = if old_format != region_options.sst_format.unwrap_or_default() {
281            // Format changed, also needs to replace the memtable builder.
282            Some(
283                self.memtable_builder_provider
284                    .builder_for_options(&region_options),
285            )
286        } else {
287            None
288        };
289        let new_mutable = version
290            .memtables
291            .mutable
292            .new_with_part_duration(version.compaction_time_window, memtable_builder);
293        // Here it assumes the leader has backfilled the partition_expr of the metadata.
294        let metadata = manifest.metadata.clone();
295
296        let version_builder = version_builder_from_manifest(
297            &manifest,
298            metadata,
299            region.file_purger.clone(),
300            new_mutable,
301            region_options,
302        );
303        let version = version_builder.build();
304        region.version_control.overwrite_current(Arc::new(version));
305        // Restore last sequence from the manifest.
306        if let Some(sequence) = manifest.committed_sequence {
307            region.version_control.update_last_sequence(sequence);
308        }
309
310        let updated = manifest.manifest_version > original_manifest_version;
311        let _ = sender.send(Ok((manifest.manifest_version, updated)));
312    }

316    fn is_import_without_staging_binding(request: &RegionEditRequest) -> bool {
317        matches!(request.kind, RegionEditKind::Import) && request.staging_partition_expr.is_none()
318    }

320    /// Whether EnterStaging is pending in the worker, flush, or compaction queue.
321    fn has_pending_enter_staging(&self, region_id: RegionId) -> bool {
322        self.flush_scheduler.has_pending_enter_staging(region_id)
323            || self
324                .compaction_scheduler
325                .has_pending_enter_staging(region_id)
326            || self.pending_ddls.has_enter_staging(region_id)
327    }

329    /// Handles a newly received edit, rejecting unbound imports while EnterStaging drains
330    /// previously accepted edits. Queued edits bypass this admission check.
331    pub(crate) fn handle_incoming_region_edit(
332        &mut self,
333        request: RegionEditRequest,
334    ) -> RegionEditDispatch {
335        let region_id = request.region_id;
336        if Self::is_import_without_staging_binding(&request)
337            && self.has_pending_enter_staging(region_id)
338        {
339            request
340                .waiters
341                .reply_with(|| RegionBusySnafu { region_id }.fail());
342            return RegionEditDispatch::Rejected;
343        }
344        self.handle_region_edit(request)
345    }

347    /// Executes an admitted edit and reports whether it started, queued, or failed synchronously.
348    fn handle_region_edit(&mut self, request: RegionEditRequest) -> RegionEditDispatch {
349        let region_id = request.region_id;
350        let Some(region) = self.regions.get_region(region_id) else {
351            request
352                .waiters
353                .reply_with(|| RegionNotFoundSnafu { region_id }.fail());
354            return RegionEditDispatch::Rejected;
355        };
356
357        // Unbound imports must not be acknowledged into a manifest that staging can discard.
358        if Self::is_import_without_staging_binding(&request) && region.in_staging_session() {
359            request
360                .waiters
361                .reply_with(|| RejectWriteInStagingSnafu { region_id }.fail());
362            return RegionEditDispatch::Rejected;
363        }
364
365        // A close-time flush ends with the region removed. A close that left staging keeps the
366        // region writable until then, so an edit meant for staging would run as a normal edit.
367        if self.flush_scheduler.has_close_time_flush(region_id) {
368            request
369                .waiters
370                .reply_with(|| RegionBusySnafu { region_id }.fail());
371            return RegionEditDispatch::Rejected;
372        }
373
374        // A close may have abandoned the staging this edit belongs to, even if its flush failed.
375        if let Some(expected) = &request.staging_partition_expr
376            && region
377                .manifest_ctx
378                .staging_partition_info()
379                .as_ref()
380                .and_then(|info| info.partition_expr())
381                != Some(expected.as_str())
382        {
383            let state = region.state();
384            request.waiters.reply_with(|| {
385                RegionStateSnafu {
386                    region_id,
387                    state,
388                    expect: RegionRoleState::Leader(RegionLeaderState::Staging),
389                }
390                .fail()
391            });
392            return RegionEditDispatch::Rejected;
393        }
394
395        if !region.is_writable() || self.scheduler_blocks_region_edits(region_id) {
396            if matches!(
397                region.state(),
398                RegionRoleState::Leader(RegionLeaderState::Editing | RegionLeaderState::Altering)
399            ) || (region.is_writable() && self.scheduler_blocks_region_edits(region_id))
400            {
401                return self
402                    .region_edit_queues
403                    .entry(region_id)
404                    .or_insert_with(|| RegionEditQueue::new(region_id))
405                    .enqueue(request);
406            } else {
407                request
408                    .waiters
409                    .reply_with(|| RegionBusySnafu { region_id }.fail());
410            }
411            return RegionEditDispatch::Rejected;
412        }
413
414        let RegionEditRequest {
415            region_id: _,
416            kind,
417            mut edit,
418            waiters,
419            preload_sst_cache,
420            staging_partition_expr: _,
421        } = request;
422        #[cfg(feature = "enterprise")]
423        if matches!(kind, RegionEditKind::Compaction(_)) && region.is_staging() {
424            waiters.reply_with(|| RegionBusySnafu { region_id }.fail());
425            return RegionEditDispatch::Rejected;
426        }
427        if matches!(kind, RegionEditKind::Import)
428            && let Err(error) = prepare_region_edit(&region, &mut edit)
429        {
430            let error = Arc::new(error);
431            waiters.reply_with(|| Err(error.clone()).context(EditRegionSnafu { region_id }));
432            return RegionEditDispatch::Rejected;
433        }
434
435        // Allow retrieving `is_staging` before spawn the edit region task.
436        let is_staging = region.is_staging();
437        let expect_state = if is_staging {
438            RegionLeaderState::Staging
439        } else {
440            RegionLeaderState::Writable
441        };
442        // Marks the region as editing.
443        if let Err(e) = region.set_editing(expect_state) {
444            let e = Arc::new(e);
445            waiters.reply_with(|| Err(e.clone()).context(EditRegionSnafu { region_id }));
446            return RegionEditDispatch::Rejected;
447        }
448        if let Some(sequence) = edit.committed_sequence {
449            region.version_control.update_last_sequence(sequence);
450        }
451
452        let request_sender = self.sender.clone();
453        let cache_manager = self.cache_manager.clone();
454        let listener = self.listener.clone();
455        // Now the region is in editing state.
456        // Updates manifest in background.
457        common_runtime::spawn_global(async move {
458            let result = match kind {
459                RegionEditKind::Import => {
460                    edit_region(
461                        &region,
462                        edit.clone(),
463                        cache_manager,
464                        listener,
465                        is_staging,
466                        preload_sst_cache,
467                    )
468                    .await
469                }
470                #[cfg(feature = "enterprise")]
471                RegionEditKind::Compaction(context) => {
472                    crate::worker::handle_table_compaction::publish(&region, &context, edit.clone())
473                        .await
474                }
475            }
476            .map_err(Arc::new);
477            let notify = WorkerRequest::Background {
478                region_id,
479                notify: BackgroundNotify::RegionEdit(RegionEditResult {
480                    region_id,
481                    origin: region,
482                    waiters,
483                    edit,
484                    result,
485                    // we always need to restore region state after region edit
486                    update_region_state: true,
487                    is_staging,
488                }),
489            };
490
491            // We don't set state back as the worker loop is already exited.
492            if let Err(res) = request_sender
493                .send(WorkerRequestWithTime::new(notify))
494                .await
495            {
496                warn!(
497                    "Failed to send region edit result back to the worker, region_id: {}, res: {:?}",
498                    region_id, res
499                );
500            }
501        });
502        RegionEditDispatch::Started
503    }

505    /// Handles region edit result.
506    pub(crate) async fn handle_region_edit_result(&mut self, edit_result: RegionEditResult) {
507        let region = match self.regions.get_region(edit_result.region_id) {
508            Some(region) => region,
509            None => {
510                // Fail writes stalled behind this edit if the region was removed before the
511                // edit-completion notification reached the worker.
512                self.fail_region_stalled_requests_as_not_found(&edit_result.region_id);
513                self.reject_region_requests_as_not_found(edit_result.region_id);
514
515                edit_result.waiters.reply_with(|| {
516                    RegionNotFoundSnafu {
517                        region_id: edit_result.region_id,
518                    }
519                    .fail()
520                });
521                return;
522            }
523        };
524
525        if !Arc::ptr_eq(&region, &edit_result.origin) {
526            // The durable edit will be recovered by open. Never replay its old
527            // completion over newer files, or change the new incarnation's queues.
528            edit_result.waiters.reply_with(|| {
529                RegionNotFoundSnafu {
530                    region_id: edit_result.region_id,
531                }
532                .fail()
533            });
534            return;
535        }
536        // A region waiting for a reload recovers its files when the reload replaces it. A fence
537        // stops the manager and sets the mark under the manifest lock, so read both under it.
538        let requires_reopen = edit_result.result.is_err() && {
539            let manager = region.manifest_ctx.manifest_manager.read().await;
540            manager.is_stopped() && !region.manifest_ctx.is_reload_pending()
541        };
542        if let Err(error) = &edit_result.result
543            && requires_reopen
544        {
545            // An uncertain manifest write requires replay before the old Version
546            // can become writable again.
547            self.remove_region(edit_result.region_id).await;
548            edit_result.waiters.reply_with(|| {
549                Err(error.clone()).context(EditRegionSnafu {
550                    region_id: edit_result.region_id,
551                })
552            });
553            return;
554        }
555        let need_compaction = if edit_result.is_staging {
556            // A close may have abandoned this edit and left staging already.
557            if edit_result.update_region_state
558                && region.state() == RegionRoleState::Leader(RegionLeaderState::Editing)
559            {
560                // For staging regions, edits are not applied immediately,
561                // as they remain invisible until the region exits the staging state.
562                region.switch_state_to_staging(RegionLeaderState::Editing);
563            }
564
565            false
566        } else {
567            let need_compaction = self.config.schedule_compaction_after_edit
568                && edit_result.result.is_ok()
569                && edit_result
570                    .edit
571                    .files_to_add
572                    .iter()
573                    .any(|file| file.level < file::LEVEL_SHARED);
574
575            // Only apply the edit if the result is ok and region is not in staging state.
576            if edit_result.result.is_ok() {
577                // Applies the edit to the region.
578                region.version_control.apply_edit(
579                    Some(edit_result.edit),
580                    &[],
581                    region.file_purger.clone(),
582                );
583            }
584            if edit_result.update_region_state {
585                region.switch_state_to_writable(RegionLeaderState::Editing);
586            }
587
588            need_compaction
589        };
590
591        edit_result
592            .waiters
593            .reply_with(|| match &edit_result.result {
594                Ok(()) => Ok(()),
595                Err(e) => Err(e.clone()).context(EditRegionSnafu {
596                    region_id: edit_result.region_id,
597                }),
598            });
599
600        if edit_result.update_region_state {
601            // Writes stalled specifically by this edit are handled before the next queued edit.
602            // Otherwise the next edit could reserve a committed sequence before those writes.
603            self.handle_region_stalled_requests(&edit_result.region_id, false)
604                .await;
605        }
606
607        // Edit completion owns dispatch: stalled writes precede queued edits, and queued
608        // controls precede compaction. Early returns either retire waiters or ignore an old instance.
609        self.dispatch_pending_region_requests(edit_result.region_id)
610            .await;
611
612        if need_compaction {
613            self.schedule_compaction(&region).await;
614        }
615    }

617    /// Retains a control request without blocking the worker that receives its completion.
618    pub(crate) fn enqueue_region_ddl(&mut self, ddl: SenderDdlRequest) {
619        let region_id = ddl.region_id;
620        if let Err(ddl) = self.pending_ddls.enqueue(ddl) {
621            ddl.sender.send(RegionBusySnafu { region_id }.fail());
622        }
623    }

625    /// Whether a scheduler holds lifecycle work that must finish before edits run.
626    ///
627    /// Running ordinary and repartition flushes retain concurrency with edits through the
628    /// existing manifest synchronization. Ordinary and repartition flushes alone do not block
629    /// edits.
630    fn scheduler_blocks_region_edits(&self, region_id: RegionId) -> bool {
631        self.flush_scheduler.has_pending_ddls(region_id)
632            || self.compaction_scheduler.blocks_region_edits(region_id)
633    }

635    /// Defers only recoverable local contention; the original handler validates other states.
636    pub(crate) fn should_defer_region_ddl(&self, ddl: &SenderDdlRequest) -> bool {
637        let Some(region) = self.regions.get_region(ddl.region_id) else {
638            return false;
639        };
640        self.can_defer_ddl_request(ddl, &region)
641            && self.has_work_ahead_of_ddl(ddl.region_id, &region, &ddl.request)
642    }
643
644    /// Whether this request may wait for ongoing work in the current region state.
645    /// Ineligible requests go directly to their handlers for normal validation.
646    fn can_defer_ddl_request(&self, ddl: &SenderDdlRequest, region: &MitoRegionRef) -> bool {
647        if !matches!(
648            region.state(),
649            RegionRoleState::Leader(
650                RegionLeaderState::Writable
651                    | RegionLeaderState::Staging
652                    | RegionLeaderState::Editing
653                    | RegionLeaderState::Altering
654                    | RegionLeaderState::EnteringStaging
655            )
656        ) {
657            return false;
658        }
659        let version = region.version();
660        match &ddl.request {
661            DdlRequest::Alter(_) | DdlRequest::Flush(_) | DdlRequest::ApplyStagingManifest(_) => {
662                true
663            }
664            DdlRequest::EnterStaging(request) => {
665                if version.has_pending_tails() {
666                    return false;
667                }
668                // A different directive is not temporary contention with the active staging.
669                if let Some(info) = region.manifest_ctx.staging_partition_info()
670                    && info.partition_directive != request.partition_directive
671                {
672                    return false;
673                }
674                true
675            }
676            DdlRequest::Close(request) if !region.in_staging_session() => {
677                // Another Close can take queued closes and reply without executing their
678                // handlers, so reject invalid flush-on-close requests before that handoff.
679                !self.flush_scheduler.has_close_time_flush(ddl.region_id)
680                    && !(request.flush_on_close && version.has_pending_tails())
681                    // A later close cannot bypass an earlier close's flush requirement.
682                    && (handle_close::close_needs_flush(region, request)
683                        || self.pending_ddls.has_close(ddl.region_id))
684            }
685            _ => false,
686        }
687    }

689    /// Whether active work or queue ordering requires this DDL to wait.
690    /// EnterStaging also waits for previously accepted edits to drain.
691    fn has_work_ahead_of_ddl(
692        &self,
693        region_id: RegionId,
694        region: &MitoRegionRef,
695        request: &DdlRequest,
696    ) -> bool {
697        self.pending_ddls.contains_region(region_id)
698            || self.scheduler_blocks_region_edits(region_id)
699            || matches!(
700                region.state(),
701                RegionRoleState::Leader(
702                    RegionLeaderState::Editing
703                        | RegionLeaderState::Altering
704                        | RegionLeaderState::EnteringStaging
705                )
706            )
707            || (matches!(request, DdlRequest::EnterStaging(_))
708                && self
709                    .region_edit_queues
710                    .get(&region_id)
711                    .is_some_and(|queue| !queue.is_empty()))
712    }

714    /// Controls retain priority, except that staging first drains its accepted edit prefix.
715    pub(crate) async fn dispatch_pending_region_requests(&mut self, region_id: RegionId) {
716        loop {
717            if self.scheduler_blocks_region_edits(region_id) {
718                return;
719            }
720            if self.regions.get_region(region_id).is_some_and(|region| {
721                matches!(
722                    region.state(),
723                    RegionRoleState::Leader(
724                        RegionLeaderState::Editing
725                            | RegionLeaderState::Altering
726                            | RegionLeaderState::EnteringStaging
727                    )
728                )
729            }) {
730                return;
731            }
732            let drain_edits = self.pending_ddls.front_is_enter_staging(region_id);
733            if !drain_edits
734                || self
735                    .region_edit_queues
736                    .get(&region_id)
737                    .is_none_or(|queue| queue.is_empty())
738            {
739                let next = self.pending_ddls.pop_front(region_id);
740                if let Some(ddl) = next {
741                    self.execute_ddl_request(ddl).await;
742                    continue;
743                }
744            }
745            let next = self
746                .region_edit_queues
747                .get_mut(&region_id)
748                .and_then(RegionEditQueue::dequeue);
749            if self
750                .region_edit_queues
751                .get(&region_id)
752                .is_some_and(RegionEditQueue::is_empty)
753            {
754                self.region_edit_queues.remove(&region_id);
755            }
756            let Some(request) = next else {
757                return;
758            };
759            // Rejected edits produce no completion notification. Keep draining them so neither
760            // the next edit nor the staging boundary is stranded. Execution bypasses admission.
761            match self.handle_region_edit(request) {
762                RegionEditDispatch::Started | RegionEditDispatch::Queued => return,
763                RegionEditDispatch::Rejected => continue,
764            }
765        }
766    }

768    /// Writes truncate action to the manifest and then applies it to the region in background.
769    pub(crate) fn handle_manifest_truncate_action(
770        &self,
771        region: MitoRegionRef,
772        truncate: RegionTruncate,
773        sender: OptionOutputTx,
774    ) {
775        // Marks the region as truncating.
776        // This prevents the region from being accessed by other write requests.
777        if let Err(e) = region.set_truncating() {
778            sender.send(Err(e));
779            return;
780        }
781        // Now the region is in truncating state.
782
783        let request_sender = self.sender.clone();
784        let manifest_ctx = region.manifest_ctx.clone();
785        let is_staging = region.is_staging();
786        let durability_barrier = self
787            .wal
788            .durability_barrier(region.region_id, &region.provider);
789
790        // Updates manifest in background.
791        common_runtime::spawn_global(async move {
792            // The truncated entry id becomes the replay frontier, so the WAL
793            // must be durable through it before the manifest names it.
794            let durable = match &truncate.kind {
795                TruncateKind::All {
796                    truncated_entry_id, ..
797                } => durability_barrier.wait(*truncated_entry_id).await,
798                _ => Ok(()),
799            };
800            // Write region truncated to manifest.
801            let action_list =
802                RegionMetaActionList::with_action(RegionMetaAction::Truncate(truncate.clone()));
803
804            let result = match durable {
805                Ok(()) => manifest_ctx
806                    .update_manifest(RegionLeaderState::Truncating, action_list, is_staging)
807                    .await
808                    .map(|_| ()),
809                Err(e) => Err(e),
810            };
811
812            // Sends the result back to the request sender.
813            let truncate_result = TruncateResult {
814                region_id: truncate.region_id,
815                sender,
816                result,
817                kind: truncate.kind,
818            };
819            let _ = request_sender
820                .send(WorkerRequestWithTime::new(WorkerRequest::Background {
821                    region_id: truncate.region_id,
822                    notify: BackgroundNotify::Truncate(truncate_result),
823                }))
824                .await
825                .inspect_err(|_| warn!("failed to send truncate result"));
826        });
827    }

829    /// Advances the durable replay frontier before discarding a region's memtables.
830    pub(crate) fn handle_manifest_discard_unflushed_action(
831        &self,
832        region: MitoRegionRef,
833        discarded_entry_id: EntryId,
834        discarded_sequence: SequenceNumber,
835        discarded_rows: u64,
836        discarded_bytes: u64,
837        sender: OptionOutputTx,
838    ) {
839        if let Err(e) = region.set_truncating() {
840            sender.send(Err(e));
841            return;
842        }
843
844        let region_id = region.region_id;
845        let request_sender = self.sender.clone();
846        let manifest_ctx = region.manifest_ctx.clone();
847        let durability_barrier = self.wal.durability_barrier(region_id, &region.provider);
848
849        common_runtime::spawn_global(async move {
850            // The frontier moves to the last written entry and sequence, so replaying the
851            // WAL after a restart skips everything the memtables held. The WAL must be
852            // durable through that entry before the manifest names it.
853            let edit = RegionEdit {
854                files_to_add: Vec::new(),
855                files_to_remove: Vec::new(),
856                timestamp_ms: None,
857                compaction_time_window: None,
858                flushed_entry_id: Some(discarded_entry_id),
859                flushed_sequence: Some(discarded_sequence),
860                committed_sequence: None,
861                wal_recovery: Default::default(),
862            };
863            let action_list = RegionMetaActionList::with_action(RegionMetaAction::Edit(edit));
864            let result = match durability_barrier.wait(discarded_entry_id).await {
865                Ok(()) => manifest_ctx
866                    .update_manifest(RegionLeaderState::Truncating, action_list, false)
867                    .await
868                    .map(|_| ()),
869                Err(e) => Err(e),
870            };
871
872            let result = DiscardUnflushedResult {
873                region_id,
874                sender,
875                result,
876                discarded_entry_id,
877                discarded_sequence,
878                discarded_rows,
879                discarded_bytes,
880            };
881            let _ = request_sender
882                .send(WorkerRequestWithTime::new(WorkerRequest::Background {
883                    region_id,
884                    notify: BackgroundNotify::DiscardUnflushed(result),
885                }))
886                .await
887                .inspect_err(|_| warn!("failed to send discard unflushed result"));
888        });
889    }

891    /// Writes region change action to the manifest and then applies it to the region in background.
892    pub(crate) fn handle_manifest_region_change(
893        &self,
894        region: MitoRegionRef,
895        change: RegionChange,
896        need_index: bool,
897        new_options: Option<RegionOptions>,
898        sender: OptionOutputTx,
899    ) {
900        // Marks the region as altering.
901        if let Err(e) = region.set_altering() {
902            sender.send(Err(e));
903            return;
904        }
905        let listener = self.listener.clone();
906        let request_sender = self.sender.clone();
907        let is_staging = region.is_staging();
908        // Now the region is in altering state.
909        common_runtime::spawn_global(async move {
910            let new_meta = change.metadata.clone();
911            let action_list = RegionMetaActionList::with_action(RegionMetaAction::Change(change));
912
913            let result = region
914                .manifest_ctx
915                .update_manifest(RegionLeaderState::Altering, action_list, is_staging)
916                .await
917                .map(|_| ());
918            let notify = WorkerRequest::Background {
919                region_id: region.region_id,
920                notify: BackgroundNotify::RegionChange(RegionChangeResult {
921                    region_id: region.region_id,
922                    sender,
923                    result,
924                    new_meta,
925                    need_index,
926                    new_options,
927                }),
928            };
929            listener
930                .on_notify_region_change_result_begin(region.region_id)
931                .await;
932
933            if let Err(res) = request_sender
934                .send(WorkerRequestWithTime::new(notify))
935                .await
936            {
937                warn!(
938                    "Failed to send region change result back to the worker, region_id: {}, res: {:?}",
939                    region.region_id, res
940                );
941            }
942        });
943    }

945    fn update_region_version(
946        version_control: &VersionControlRef,
947        new_meta: RegionMetadataRef,
948        new_options: Option<RegionOptions>,
949        memtable_builder_provider: &MemtableBuilderProvider,
950    ) {
951        let options_changed = new_options.is_some();
952        let region_id = new_meta.region_id;
953        if let Some(new_options) = new_options {
954            // Needs to update the region with new format and memtables.
955            // Creates a new memtable builder for the new options as it may change the memtable type.
956            let new_memtable_builder = memtable_builder_provider.builder_for_options(&new_options);
957            version_control.alter_schema_and_format(new_meta, new_options, new_memtable_builder);
958        } else {
959            // Only changes the schema.
960            version_control.alter_schema(new_meta);
961        }
962
963        let version_data = version_control.current();
964        let version = version_data.version;
965        info!(
966            "Region {} is altered, metadata is {:?}, options: {:?}, options_changed: {}",
967            region_id, version.metadata, version.options, options_changed,
968        );
969    }

972fn prepare_region_edit(region: &MitoRegionRef, edit: &mut RegionEdit) -> Result<()> {
973    let region_id = region.region_id;
974    edit.validate_shared_reclassification(region_id)?;
975    let current = region.version_control.current();
976    let shared = &current.version.ssts.levels()[file::SHARED_LEVEL_SLOT].files;
977    // Shared adoption is L0-only. The manifest is keyed by FileId and would
978    // replace an existing reference, but installing a shared handle detaches
979    // only the L0 handle, so an existing L1 reference would leave two live
980    // handles for one file and duplicate its rows until reopen.
981    let l1_files = &current.version.ssts.levels()[1].files;
982    snafu::ensure!(
983        edit.files_to_add
984            .iter()
985            .all(|file| file.level != file::LEVEL_SHARED || !l1_files.contains_key(&file.file_id)),
986        crate::error::InvalidRequestSnafu {
987            region_id,
988            reason: "Cannot adopt an existing L1 SST as shared; adoption is L0-only",
989        }
990    );
991    snafu::ensure!(
992        edit.files_to_remove
993            .iter()
994            .all(|file| file.level < file::LEVEL_SHARED && !shared.contains_key(&file.file_id))
995            && edit
996                .files_to_add
997                .iter()
998                .all(|file| file::is_supported_level(file.level)
999                    && (file.level != file::LEVEL_SHARED
1000                        || file.region_id.table_id() == region_id.table_id())),
1001        crate::error::InvalidRequestSnafu {
1002            region_id,
1003            reason: "Invalid shared SST import or removal"
1004        }
1005    );
1006    for file in &edit.files_to_add {
1007        if let Some(existing) = shared.get(&file.file_id) {
1008            let mut imported = file.clone();
1009            imported.sequence = existing.meta_ref().sequence;
1010            imported.preserve_row_sequence = false;
1011            snafu::ensure!(
1012                &imported == existing.meta_ref(),
1013                crate::error::InvalidRequestSnafu {
1014                    region_id,
1015                    reason: "Shared SST import conflicts with a visible reference",
1016                }
1017            );
1018        }
1019    }
1020    let existing_sequence = |file: &file::FileMeta| {
1021        shared
1022            .get(&file.file_id)
1023            .and_then(|handle| handle.meta_ref().sequence)
1024    };
1025    // A repeat of a still-visible shared import is not a new admission. Retired
1026    // imports are not tracked: compaction retries must never re-import old inputs.
1027    let same_import = !edit.files_to_add.is_empty()
1028        && edit.files_to_remove.is_empty()
1029        && edit
1030            .files_to_add
1031            .iter()
1032            .all(|file| existing_sequence(file).is_some());
1033    let sequence =
1034        if same_import {
1035            None
1036        } else {
1037            let sequence = current.last_sequence.checked_add(1).context(
1038                crate::error::InvalidRequestSnafu {
1039                    region_id,
1040                    reason: "Region sequence exhausted",
1041                },
1042            )?;
1043            edit.committed_sequence = Some(sequence);
1044            NonZeroU64::new(sequence)
1045        };
1046    for file in &mut edit.files_to_add {
1047        file.sequence = existing_sequence(file).or(sequence);
1048        file.preserve_row_sequence = false;
1049    }
1050    Ok(())
1051}

1053/// Checks the edit, writes and applies it.
1054async fn edit_region(
1055    region: &MitoRegionRef,
1056    edit: RegionEdit,
1057    cache_manager: CacheManagerRef,
1058    listener: WorkerListener,
1059    is_staging: bool,
1060    preload_sst_cache: bool,
1061) -> Result<()> {
1062    let region_id = region.region_id;
1063    if let Some(write_cache) = cache_manager.write_cache()
1064        && preload_sst_cache
1065    {
1066        for file_meta in &edit.files_to_add {
1067            let write_cache = write_cache.clone();
1068            let layer = region.access_layer.clone();
1069            let listener = listener.clone();
1070
1071            let index_key = IndexKey::new(region_id, file_meta.file_id, FileType::Parquet);
1072            let remote_path =
1073                location::sst_file_path(layer.table_dir(), file_meta.file_id(), layer.path_type());
1074
1075            let has_downloadable_index = file_meta.has_downloadable_index();
1076            let index_file_size = file_meta.index_file_size();
1077
1078            let index_file_index_key = IndexKey::new(
1079                region_id,
1080                file_meta.index_id().file_id.file_id(),
1081                FileType::Puffin(file_meta.index_version),
1082            );
1083            let index_remote_path = location::index_file_path(
1084                layer.table_dir(),
1085                file_meta.index_id(),
1086                layer.path_type(),
1087            );
1088
1089            let file_size = file_meta.file_size;
1090            common_runtime::spawn_global(async move {
1091                WRITE_CACHE_INFLIGHT_DOWNLOAD.add(1);
1092
1093                let parquet_cached = write_cache
1094                    .download_if_absent(index_key, &remote_path, layer.object_store(), file_size)
1095                    .await;
1096
1097                if parquet_cached.is_ok() {
1098                    // Triggers the filling of the parquet metadata cache.
1099                    // The parquet file is already downloaded.
1100                    let mut cache_metrics = Default::default();
1101                    let _ = write_cache
1102                        .file_cache()
1103                        .get_parquet_meta_data(
1104                            index_key,
1105                            &mut cache_metrics,
1106                            PageIndexPolicy::Optional,
1107                        )
1108                        .await;
1109                }
1110                if has_downloadable_index {
1111                    // also download puffin file
1112                    if let Err(err) = write_cache
1113                        .download(
1114                            index_file_index_key,
1115                            &index_remote_path,
1116                            layer.object_store(),
1117                            index_file_size,
1118                        )
1119                        .await
1120                    {
1121                        common_telemetry::error!(
1122                            err; "Failed to download puffin file, region_id: {}, index_file_index_key: {:?}, index_remote_path: {}", region_id, index_file_index_key, index_remote_path
1123                        );
1124                    }
1125                }
1126
1127                WRITE_CACHE_INFLIGHT_DOWNLOAD.sub(1);
1128                if matches!(parquet_cached, Ok(true)) {
1129                    listener.on_file_cache_filled(index_key.file_id);
1130                }
1131            });
1132        }
1133    }
1134
1135    info!(
1136        "Applying {edit:?} to region {}, is_staging: {}",
1137        region_id, is_staging
1138    );
1139
1140    let action_list = RegionMetaActionList::with_action(RegionMetaAction::Edit(edit));
1141    let result = region
1142        .manifest_ctx
1143        .update_manifest(RegionLeaderState::Editing, action_list, is_staging)
1144        .await;
1145    if let Err(error) = &result
1146        && error.may_have_persisted_manifest_update()
1147    {
1148        // Recover the file list before accepting another edit after an uncertain write.
1149        region
1150            .manifest_ctx
1151            .manifest_manager
1152            .write()
1153            .await
1154            .stop()
1155            .await;
1156    }
1157    result.map(|_| ())
1158}
