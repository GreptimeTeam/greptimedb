44impl<S: LogStore> RegionWorkerLoop<S> {
45    /// Takes and handles all write requests.
46    pub(crate) async fn handle_write_requests(
47        &mut self,
48        write_requests: &mut Vec<SenderWriteRequest>,
49        bulk_requests: &mut Vec<SenderBulkRequest>,
50        allow_stall: bool,
51    ) {
52        if write_requests.is_empty() && bulk_requests.is_empty() {
53            return;
54        }
55
56        let write_region_ids = write_region_ids(write_requests, bulk_requests);
57
58        // Check region pressure before writes to match the global write buffer behavior.
59        self.maybe_flush_worker();
60        let pressure = self.maybe_flush_write_regions(write_region_ids);
61
62        if self.should_reject_write() {
63            // The memory pressure is still too high, reject write requests.
64            reject_write_requests(write_requests, bulk_requests);
65            // Also reject all stalled requests.
66            self.reject_stalled_requests();
67            return;
68        }
69
70        if !pressure.rejected_region_ids.is_empty() {
71            reject_region_write_requests(
72                &pressure.rejected_region_ids,
73                write_requests,
74                bulk_requests,
75            );
76            for region_id in &pressure.rejected_region_ids {
77                self.reject_region_stalled_requests(region_id);
78            }
79            if write_requests.is_empty() && bulk_requests.is_empty() {
80                return;
81            }
82        }
83
84        if self.write_buffer_manager.should_stall() && allow_stall {
85            let stalled_count = (write_requests.len() + bulk_requests.len()) as i64;
86            self.stalling_count.add(stalled_count);
87            WRITE_STALL_TOTAL.inc_by(stalled_count as u64);
88            self.stalled_requests.append(write_requests, bulk_requests);
89            self.listener.on_write_stall();
90            return;
91        }
92
93        if allow_stall {
94            self.stall_region_write_requests(
95                &pressure.stalled_region_ids,
96                write_requests,
97                bulk_requests,
98            );
99            if write_requests.is_empty() && bulk_requests.is_empty() {
100                return;
101            }
102        }
103
104        // Prepare write context.
105        let mut region_ctxs = {
106            let _timer = WRITE_STAGE_ELAPSED
107                .with_label_values(&["prepare_ctx"])
108                .start_timer();
109            self.prepare_region_write_ctx(write_requests, bulk_requests)
110        };
111
112        // Write to WAL.
113        {
114            let _timer = WRITE_STAGE_ELAPSED
115                .with_label_values(&["write_wal"])
116                .start_timer();
117            if !write_wal(&self.wal, &mut region_ctxs).await {
118                // Failed to write to the WAL, all waiters are notified with the error.
119                return;
120            }
121        }
122        self.listener.on_write_wal_end().await;
123        self.fail_writes_after_lease_expiry(&mut region_ctxs);
124
125        let (mut put_rows, mut delete_rows) = (0, 0);
126        // Write to memtables.
127        {
128            let _timer = WRITE_STAGE_ELAPSED
129                .with_label_values(&["write_memtable"])
130                .start_timer();
131            if region_ctxs.len() == 1 {
132                // fast path for single region.
133                let mut region_ctx = region_ctxs.into_values().next().unwrap();
134                region_ctx.write_memtables().await;
135                put_rows += region_ctx.put_num;
136                delete_rows += region_ctx.delete_num;
137            } else {
138                let region_write_task = region_ctxs
139                    .into_values()
140                    .map(|mut region_ctx| {
141                        // use tokio runtime to schedule tasks.
142                        common_runtime::spawn_global(async move {
143                            region_ctx.write_memtables().await;
144                            (region_ctx.put_num, region_ctx.delete_num)
145                        })
146                    })
147                    .collect::<Vec<_>>();
148
149                for result in futures::future::join_all(region_write_task).await {
150                    match result {
151                        Ok((put, delete)) => {
152                            put_rows += put;
153                            delete_rows += delete;
154                        }
155                        Err(e) => {
156                            error!(e; "unexpected error when joining region write tasks");
157                        }
158                    }
159                }
160            }
161        }
162        WRITE_ROWS_TOTAL
163            .with_label_values(&["put"])
164            .inc_by(put_rows as u64);
165        WRITE_ROWS_TOTAL
166            .with_label_values(&["delete"])
167            .inc_by(delete_rows as u64);
168    }

170    /// Handles stalled write requests whose regions no longer need to stall.
171    pub(crate) async fn handle_stalled_requests(&mut self) {
172        let region_ids = self
173            .stalled_requests
174            .requests
175            .keys()
176            .copied()
177            .collect::<HashSet<_>>();
178        let pressure = self.maybe_flush_write_regions(region_ids);
179        for region_id in &pressure.rejected_region_ids {
180            self.reject_region_stalled_requests(region_id);
181        }
182        let ready_region_ids = self
183            .stalled_requests
184            .requests
185            .keys()
186            .filter(|region_id| !pressure.stalled_region_ids.contains(region_id))
187            .copied()
188            .collect::<Vec<_>>();
189
190        // These requests have already been stalled. Retry ready regions without stalling the
191        // same requests again. Regions that still exceed their limit remain in the queue until
192        // their own flush releases the pressure.
193        for region_id in ready_region_ids {
194            self.handle_region_stalled_requests(&region_id, false).await;
195        }
196    }
197
198    /// Rejects all stalled requests.
199    pub(crate) fn reject_stalled_requests(&mut self) {
200        let stalled = std::mem::take(&mut self.stalled_requests);
201        self.stalling_count.sub(stalled.stalled_count() as i64);
202        for (_, (_, mut requests, mut bulk)) in stalled.requests {
203            reject_write_requests(&mut requests, &mut bulk);
204        }
205    }
206
207    /// Rejects a specific region's stalled requests.
208    pub(crate) fn reject_region_stalled_requests(&mut self, region_id: &RegionId) {
209        debug!("Rejects stalled requests for region {}", region_id);
210        let (mut requests, mut bulk) = self.stalled_requests.remove(region_id);
211        self.stalling_count
212            .sub((requests.len() + bulk.len()) as i64);
213        reject_write_requests(&mut requests, &mut bulk);
214    }
215
216    /// Fails a specific region's stalled requests if the region no longer exists.
217    pub(crate) fn fail_region_stalled_requests_as_not_found(&mut self, region_id: &RegionId) {
218        debug!(
219            "Fails stalled requests for region {} as region not found",
220            region_id
221        );
222        let (requests, bulk) = self.stalled_requests.remove(region_id);
223        self.stalling_count
224            .sub((requests.len() + bulk.len()) as i64);
225
226        for req in requests {
227            req.sender.send(
228                RegionNotFoundSnafu {
229                    region_id: req.request.region_id,
230                }
231                .fail(),
232            );
233        }
234        for req in bulk {
235            req.sender.send(
236                RegionNotFoundSnafu {
237                    region_id: req.region_id,
238                }
239                .fail(),
240            );
241        }
242    }

244    /// Handles a specific region's stalled requests.
245    ///
246    /// `allow_stall` should be false for backpressure retry paths to avoid stalling the same
247    /// requests again. It should remain true for non-backpressure retries, such as requests stalled
248    /// by alter, staging, and region editing. Global reject backpressure still applies before the
249    /// stall check.
250    pub(crate) async fn handle_region_stalled_requests(
251        &mut self,
252        region_id: &RegionId,
253        allow_stall: bool,
254    ) {
255        debug!("Handles stalled requests for region {}", region_id);
256        let (mut requests, mut bulk) = self.stalled_requests.remove(region_id);
257        self.stalling_count
258            .sub((requests.len() + bulk.len()) as i64);
259        self.handle_write_requests(&mut requests, &mut bulk, allow_stall)
260            .await;
261    }
262
263    /// Processes same-batch writes for a region before handling its edit-completion notification.
264    ///
265    /// The worker dispatch loop handles background notifications before the current batch's write
266    /// buffer. Without this step, writes that arrived during edit N could be classified only after
267    /// edit N+1 is started, placing them behind that next edit.
268    pub(crate) async fn handle_buffered_region_write_requests(
269        &mut self,
270        region_id: &RegionId,
271        write_requests: &mut Vec<SenderWriteRequest>,
272        bulk_requests: &mut Vec<SenderBulkRequest>,
273    ) {
274        let mut current_region_write_requests = write_requests
275            .extract_if(.., |r| r.request.region_id == *region_id)
276            .collect::<Vec<_>>();
277
278        let mut current_region_bulk_requests = bulk_requests
279            .extract_if(.., |r| r.region_id == *region_id)
280            .collect::<Vec<_>>();
281
282        self.handle_write_requests(
283            &mut current_region_write_requests,
284            &mut current_region_bulk_requests,
285            true,
286        )
287        .await;
288    }

291impl<S> RegionWorkerLoop<S> {
292    /// Validates and groups requests by region.
293    fn prepare_region_write_ctx(
294        &mut self,
295        write_requests: &mut Vec<SenderWriteRequest>,
296        bulk_requests: &mut Vec<SenderBulkRequest>,
297    ) -> HashMap<RegionId, RegionWriteCtx> {
298        // Initialize region write context map.
299        let mut region_ctxs = HashMap::new();
300        // One reading serves the batch, as preparing it does not wait.
301        let lease_now = self.node_lease.is_enabled().then(|| self.node_lease.now());
302        self.process_write_requests(&mut region_ctxs, write_requests, lease_now);
303        self.process_bulk_requests(&mut region_ctxs, bulk_requests, lease_now);
304        region_ctxs
305    }
306
307    fn process_write_requests(
308        &mut self,
309        region_ctxs: &mut HashMap<RegionId, RegionWriteCtx>,
310        write_requests: &mut Vec<SenderWriteRequest>,
311        lease_now: Option<u64>,
312    ) {
313        for mut sender_req in write_requests.drain(..) {
314            let region_id = sender_req.request.region_id;
315
316            // If region is waiting for alteration, add requests to pending writes.
317            if self.flush_scheduler.has_pending_ddls(region_id) {
318                // TODO(yingwen): consider adding some metrics for this.
319                // Safety: The region has pending ddls.
320                self.flush_scheduler
321                    .add_write_request_to_pending(sender_req);
322                continue;
323            }
324
325            // Checks whether the region exists and is it stalling.
326            if let hash_map::Entry::Vacant(e) = region_ctxs.entry(region_id) {
327                let Some(region) = self
328                    .regions
329                    .get_region_or(region_id, &mut sender_req.sender)
330                else {
331                    // No such region.
332                    continue;
333                };
334                #[cfg(test)]
335                debug!(
336                    "Handling write request for region {}, state: {:?}",
337                    region_id,
338                    region.state()
339                );
340                match region.state() {
341                    RegionRoleState::Leader(RegionLeaderState::Writable)
342                    | RegionRoleState::Leader(RegionLeaderState::Staging) => {
343                        let leader_deadline = region.manifest_ctx.leader_deadline.get();
344                        if lease_now.is_some_and(|now| now >= leader_deadline) {
345                            sender_req
346                                .sender
347                                .send(RegionLeaseExpiredSnafu { region_id }.fail());
348                            continue;
349                        }
350                        if region.reject_all_writes_in_staging() {
351                            sender_req
352                                .sender
353                                .send(RejectWriteInStagingSnafu { region_id }.fail());
354                            continue;
355                        }
356
357                        let mut region_ctx = RegionWriteCtx::new(
358                            region.region_id,
359                            &region.version_control,
360                            region.provider.clone(),
361                            Some(region.region_stats.written_bytes.clone()),
362                            WriteSource::Request,
363                        );
364                        region_ctx.lease = lease_now.map(|_| (region, leader_deadline));
365
366                        e.insert(region_ctx);
367                    }
368                    RegionRoleState::Leader(RegionLeaderState::Altering)
369                    | RegionRoleState::Leader(RegionLeaderState::Editing) => {
370                        // Editing is transient: queue the write so edit completion can drain it
371                        // before starting the next queued edit.
372                        debug!(
373                            "Region {} is {:?}, add request to pending writes",
374                            region.region_id,
375                            region.state()
376                        );
377                        self.stalling_count.add(1);
378                        WRITE_STALL_TOTAL.inc();
379                        self.stalled_requests.push(sender_req);
380                        continue;
381                    }
382                    RegionRoleState::Leader(RegionLeaderState::EnteringStaging) => {
383                        debug!(
384                            "Region {} is entering staging, add request to pending writes",
385                            region.region_id
386                        );
387                        self.stalling_count.add(1);
388                        WRITE_STALL_TOTAL.inc();
389                        self.stalled_requests.push(sender_req);
390                        continue;
391                    }
392                    state => {
393                        // The region is not writable.
394                        sender_req.sender.send(
395                            RegionStateSnafu {
396                                region_id,
397                                state,
398                                expect: RegionRoleState::Leader(RegionLeaderState::Writable),
399                            }
400                            .fail(),
401                        );
402                        continue;
403                    }
404                }
405            }
406
407            // Safety: Now we ensure the region exists.
408            let region_ctx = region_ctxs.get_mut(&region_id).unwrap();
409            let Some(region) = self
410                .regions
411                .get_region_or(region_id, &mut sender_req.sender)
412            else {
413                continue;
414            };
415            if region.reject_all_writes_in_staging() {
416                sender_req
417                    .sender
418                    .send(RejectWriteInStagingSnafu { region_id }.fail());
419                continue;
420            }
421            let expected_version = region.expected_partition_expr_version();
422            if let Err(e) = check_partition_expr_version(
423                region_id,
424                expected_version,
425                sender_req.request.partition_expr_version,
426            ) {
427                sender_req.sender.send(Err(e));
428                continue;
429            }
430
431            if let Err(e) = check_op_type(
432                region_ctx.version().options.append_mode,
433                &sender_req.request,
434            ) {
435                // Do not allow non-put op under append mode.
436                sender_req.sender.send(Err(e));
437
438                continue;
439            }
440
441            // Double check the request schema
442            let need_fill_missing_columns =
443                if let Some(ref region_metadata) = sender_req.request.region_metadata {
444                    region_ctx.version().metadata.schema_version != region_metadata.schema_version
445                } else {
446                    true
447                };
448            // Only fill missing columns if primary key is dense encoded.
449            if need_fill_missing_columns
450                && sender_req.request.primary_key_encoding() == PrimaryKeyEncoding::Dense
451                && let Err(e) = sender_req
452                    .request
453                    .maybe_fill_missing_columns(&region_ctx.version().metadata)
454            {
455                sender_req.sender.send(Err(e));
456
457                continue;
458            }
459
460            // Collect requests by region.
461            region_ctx.push_mutation(
462                sender_req.request.op_type as i32,
463                Some(sender_req.request.rows),
464                sender_req.request.hint,
465                sender_req.sender,
466                None,
467                sender_req.request.skip_wal,
468            );
469        }
470    }

472    /// Processes bulk insert requests.
473    fn process_bulk_requests(
474        &mut self,
475        region_ctxs: &mut HashMap<RegionId, RegionWriteCtx>,
476        requests: &mut Vec<SenderBulkRequest>,
477        lease_now: Option<u64>,
478    ) {
479        let _timer = metrics::REGION_WORKER_HANDLE_WRITE_ELAPSED
480            .with_label_values(&["prepare_bulk_request"])
481            .start_timer();
482        for mut bulk_req in requests.drain(..) {
483            let region_id = bulk_req.region_id;
484            // If region is waiting for alteration, add requests to pending writes.
485            if self.flush_scheduler.has_pending_ddls(region_id) {
486                // Safety: The region has pending ddls.
487                self.flush_scheduler.add_bulk_request_to_pending(bulk_req);
488                continue;
489            }
490
491            // Checks whether the region exists and is it stalling.
492            if let hash_map::Entry::Vacant(e) = region_ctxs.entry(region_id) {
493                let Some(region) = self.regions.get_region_or(region_id, &mut bulk_req.sender)
494                else {
495                    continue;
496                };
497                match region.state() {
498                    RegionRoleState::Leader(RegionLeaderState::Writable)
499                    | RegionRoleState::Leader(RegionLeaderState::Staging) => {
500                        let leader_deadline = region.manifest_ctx.leader_deadline.get();
501                        if lease_now.is_some_and(|now| now >= leader_deadline) {
502                            bulk_req
503                                .sender
504                                .send(RegionLeaseExpiredSnafu { region_id }.fail());
505                            continue;
506                        }
507                        if region.reject_all_writes_in_staging() {
508                            bulk_req
509                                .sender
510                                .send(RejectWriteInStagingSnafu { region_id }.fail());
511                            continue;
512                        }
513                        let mut region_ctx = RegionWriteCtx::new(
514                            region.region_id,
515                            &region.version_control,
516                            region.provider.clone(),
517                            Some(region.region_stats.written_bytes.clone()),
518                            WriteSource::Request,
519                        );
520                        region_ctx.lease = lease_now.map(|_| (region, leader_deadline));
521
522                        e.insert(region_ctx);
523                    }
524                    RegionRoleState::Leader(RegionLeaderState::Altering)
525                    | RegionRoleState::Leader(RegionLeaderState::Editing) => {
526                        // Editing is transient: queue the bulk write so edit completion can drain
527                        // it before starting the next queued edit.
528                        debug!(
529                            "Region {} is {:?}, add request to pending writes",
530                            region.region_id,
531                            region.state()
532                        );
533                        self.stalling_count.add(1);
534                        WRITE_STALL_TOTAL.inc();
535                        self.stalled_requests.push_bulk(bulk_req);
536                        continue;
537                    }
538                    state => {
539                        // The region is not writable.
540                        bulk_req.sender.send(
541                            RegionStateSnafu {
542                                region_id,
543                                state,
544                                expect: RegionRoleState::Leader(RegionLeaderState::Writable),
545                            }
546                            .fail(),
547                        );
548                        continue;
549                    }
550                }
551            }
552
553            // Safety: Now we ensure the region exists.
554            let region_ctx = region_ctxs.get_mut(&region_id).unwrap();
555            let Some(region) = self.regions.get_region_or(region_id, &mut bulk_req.sender) else {
556                continue;
557            };
558            if region.reject_all_writes_in_staging() {
559                bulk_req
560                    .sender
561                    .send(RejectWriteInStagingSnafu { region_id }.fail());
562                continue;
563            }
564            let expected_version = region.expected_partition_expr_version();
565            if let Err(e) = check_partition_expr_version(
566                region_id,
567                expected_version,
568                bulk_req.partition_expr_version,
569            ) {
570                bulk_req.sender.send(Err(e));
571                continue;
572            }
573
574            // Double-check the request schema
575            let need_fill_missing_columns =
576                !bulk_req.region_metadata.is_some_and(|aligned_schema| {
577                    aligned_schema.schema_version == region_ctx.version().metadata.schema_version
578                });
579
580            // Fill missing columns if needed. The batch is checked again as the region
581            // schema may have changed since the request was checked.
582            if need_fill_missing_columns {
583                let metadata = &region_ctx.version().metadata;
584                if let Err(e) = check_bulk_insert_batch(metadata, &bulk_req.request.batch)
585                    .and_then(|_| bulk_req.request.fill_missing_columns(metadata))
586                {
587                    bulk_req.sender.send(Err(e));
588                    continue;
589                }
590            }
591
592            // Collect requests by region.
593            if !region_ctx.push_bulk(bulk_req.sender, bulk_req.request, None, bulk_req.skip_wal) {
594                continue;
595            }
596        }
597    }

599    /// Fails the waiters of writes whose WAL append completed at or after their region's Leader
600    /// deadline. The writes still go to the memtables, as their entries are in the WAL.
601    fn fail_writes_after_lease_expiry(&self, region_ctxs: &mut HashMap<RegionId, RegionWriteCtx>) {
602        if !self.node_lease.is_enabled() {
603            return;
604        }
605        let now = self.node_lease.now();
606        for (region_id, region_ctx) in region_ctxs.iter_mut() {
607            // A Leader grant may have extended the deadline during the append.
608            let expired = region_ctx.lease.as_ref().is_some_and(|(region, deadline)| {
609                now >= *deadline && now >= region.manifest_ctx.leader_deadline.get()
610            });
611            if !expired || region_ctx.is_failed() {
612                continue;
613            }
614            region_ctx.fail_waiters(Arc::new(
615                UnacknowledgedWriteSnafu {
616                    region_id: *region_id,
617                }
618                .build(),
619            ));
620        }
621    }

623    /// Returns true if the engine needs to reject some write requests.
624    pub(crate) fn should_reject_write(&self) -> bool {
625        // If memory usage reaches high threshold (we should also consider stalled requests) returns true.
626        self.write_buffer_manager.memory_usage() + self.stalled_requests.estimated_size
627            >= self.config.global_write_buffer_reject_size.as_bytes() as usize
628    }
629
630    fn stall_region_write_requests(
631        &mut self,
632        stalled_region_ids: &HashSet<RegionId>,
633        write_requests: &mut Vec<SenderWriteRequest>,
634        bulk_requests: &mut Vec<SenderBulkRequest>,
635    ) {
636        let mut stalled_count = 0;
637        let mut stalled_write_requests = write_requests
638            .extract_if(.., |req| {
639                stalled_region_ids.contains(&req.request.region_id)
640            })
641            .collect::<Vec<_>>();
642        let mut stalled_bulk_requests = bulk_requests
643            .extract_if(.., |req| stalled_region_ids.contains(&req.region_id))
644            .collect::<Vec<_>>();
645
646        stalled_count += stalled_write_requests.len() + stalled_bulk_requests.len();
647        self.stalled_requests
648            .append(&mut stalled_write_requests, &mut stalled_bulk_requests);
649
650        if stalled_count > 0 {
651            let stalled_count = stalled_count as i64;
652            self.stalling_count.add(stalled_count);
653            WRITE_STALL_TOTAL.inc_by(stalled_count as u64);
654            self.listener.on_write_stall();
655        }
656    }

659/// Writes WAL entries of all region contexts to the WAL in one batch and updates
660/// the next entry id of each region on success.
661///
662/// Returns `false` if the batch fails to be written to the WAL. In this case all
663/// contexts are consumed and their waiters are notified with the error, so the
664/// caller should skip the memtable phase.
665async fn write_wal<S: LogStore>(
666    wal: &Wal<S>,
667    region_ctxs: &mut HashMap<RegionId, RegionWriteCtx>,
668) -> bool {
669    let mut wal_writer = wal.writer();
670    let mut has_wal_entries = false;
671    for region_ctx in region_ctxs.values_mut() {
672        if region_ctx.skip_wal() {
673            continue;
674        }
675        if let Err(e) = region_ctx.add_wal_entry(&mut wal_writer).map_err(Arc::new) {
676            region_ctx.set_error(e);
677        } else {
678            has_wal_entries = true;
679        }
680    }
681    // All-skipped batches should not touch the log store, even with an empty append.
682    if !has_wal_entries {
683        return true;
684    }
685    match wal_writer.write_to_wal().await.map_err(Arc::new) {
686        Ok(response) => {
687            for (region_id, region_ctx) in region_ctxs.iter_mut() {
688                if region_ctx.skip_wal() {
689                    continue;
690                }
691                // The entry of a failed region (e.g. failed to build its WAL entry) is
692                // not in the batch so the response has no last entry id for it. Its
693                // waiters are already notified with the error.
694                if region_ctx.is_failed() {
695                    continue;
696                }
697
698                // Safety: the log store implementation ensures that either the `write_to_wal` fails and no
699                // response is returned or the last entry ids for each region in the batch do exist.
700                let last_entry_id = response.last_entry_ids.get(region_id).unwrap();
701                region_ctx.set_next_entry_id(last_entry_id + 1);
702            }
703            true
704        }
705        Err(e) => {
706            // Failed to write wal.
707            for (_, mut region_ctx) in region_ctxs.drain() {
708                region_ctx.set_error(e.clone());
709            }
710            false
711        }
712    }
713}

715/// Send rejected error to all `write_requests`.
716fn reject_write_requests(
717    write_requests: &mut Vec<SenderWriteRequest>,
718    bulk_requests: &mut Vec<SenderBulkRequest>,
719) {
720    WRITE_REJECT_TOTAL.inc_by(write_requests.len() as u64);
721
722    for req in write_requests.drain(..) {
723        req.sender.send(
724            RejectWriteSnafu {
725                region_id: req.request.region_id,
726            }
727            .fail(),
728        );
729    }
730    for req in bulk_requests.drain(..) {
731        let region_id = req.region_id;
732        req.sender.send(RejectWriteSnafu { region_id }.fail());
733    }
734}
735
736fn reject_region_write_requests(
737    rejected_region_ids: &HashSet<RegionId>,
738    write_requests: &mut Vec<SenderWriteRequest>,
739    bulk_requests: &mut Vec<SenderBulkRequest>,
740) {
741    let mut rejected_write_requests = write_requests
742        .extract_if(.., |req| {
743            rejected_region_ids.contains(&req.request.region_id)
744        })
745        .collect::<Vec<_>>();
746    let mut rejected_bulk_requests = bulk_requests
747        .extract_if(.., |req| rejected_region_ids.contains(&req.region_id))
748        .collect::<Vec<_>>();
749    reject_write_requests(&mut rejected_write_requests, &mut rejected_bulk_requests);
750}
751
752fn write_region_ids(
753    write_requests: &[SenderWriteRequest],
754    bulk_requests: &[SenderBulkRequest],
755) -> HashSet<RegionId> {
756    write_requests
757        .iter()
758        .map(|req| req.request.region_id)
759        .chain(bulk_requests.iter().map(|req| req.region_id))
760        .collect()
761}
