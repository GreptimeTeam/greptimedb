34impl<S: LogStore> RegionWorkerLoop<S> {
35    /// Starts replaying the pending tails of the region, unless a replay of it still runs.
36    pub(crate) fn maybe_replay_tails(&mut self, region_id: RegionId) {
37        let Some(region) = self.regions.get_region(region_id) else {
38            return;
39        };
40        if self.tail_replays.contains_key(&region_id) || !should_replay_tails(&region) {
41            return;
42        }
43        info!("Replaying the pending WAL tails of region {region_id}");
44        let state = CancellableTaskState::new();
45        let task = TailReplayTask {
46            flush_task: self.new_flush_task(
47                &region,
48                FlushReason::TailReplay,
49                None,
50                self.config.clone(),
51            ),
52            region,
53            node_id: self.file_ref_manager.node_id(),
54            wal: self.wal.clone(),
55            state: state.clone(),
56        };
57        self.tail_replays.insert(region_id, state);
58        common_runtime::spawn_global(task.run());
59    }
60
61    /// Cancels the tail replay of the region. The replay still reports its end.
62    pub(crate) fn cancel_tail_replay(&self, region_id: RegionId) {
63        if let Some(state) = self.tail_replays.get(&region_id) {
64            let _ = state.request_cancel();
65        }
66    }
67
68    /// Applies a published tail replay unless its instance was replaced, or its version no
69    /// longer records `tails`, the tails it replayed, because a sync rebuilt it from a later
70    /// manifest.
71    pub(crate) async fn handle_tail_replay_committed(
72        &mut self,
73        region_id: RegionId,
74        region: MitoRegionRef,
75        edit: RegionEdit,
76        tails: Vec<PendingTail>,
77    ) {
78        let registered = self.regions.get_region(region_id);
79        if !registered.is_some_and(|registered| Arc::ptr_eq(&registered, &region)) {
80            info!("Region {region_id} changed after its tails were published");
81            return;
82        }
83        if region.version().pending_tails != tails {
84            info!("Region {region_id} no longer records the replayed tails, skipping their edit");
85            return;
86        }
87        // A truncate uses the last entry id, which must not be below the flushed entry id.
88        if let Some(flushed_entry_id) = edit.flushed_entry_id
89            && flushed_entry_id > region.version_control.current().last_entry_id
90        {
91            region.version_control.set_entry_id(flushed_entry_id);
92        }
93        let file_metas = edit.files_to_add.clone();
94        region
95            .version_control
96            .apply_edit(Some(edit), &[], region.file_purger.clone());
97        if self.config.index.build_mode == IndexBuildMode::Async && !file_metas.is_empty() {
98            self.handle_rebuild_index(
99                BuildIndexRequest {
100                    region_id,
101                    build_type: IndexBuildType::Flush,
102                    file_metas,
103                    result_senders: Vec::new(),
104                },
105                OptionOutputTx::new(None),
106            )
107            .await;
108        }
109        self.schedule_compaction(&region).await;
110    }
111
112    /// Forgets the ended tail replay of the region and starts another if it still has tails.
113    pub(crate) fn handle_tail_replay_ended(&mut self, region_id: RegionId) {
114        self.tail_replays.remove(&region_id);
115        self.listener.on_tail_replay_ended(region_id);
116        self.maybe_replay_tails(region_id);
117    }
118}

120/// Asks the worker to replay the pending tails of a region that became writable, if it has
121/// any, without waiting.
122pub(crate) fn notify_writable(sender: &Sender<WorkerRequestWithTime>, region: &MitoRegion) {
123    if !region.version().has_pending_tails() {
124        return;
125    }
126    let sender = sender.clone();
127    let region_id = region.region_id;
128    common_runtime::spawn_global(async move {
129        notify_worker(&sender, region_id, BackgroundNotify::RegionBecameWritable).await;
130    });
