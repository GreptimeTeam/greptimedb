43    /// Starts replaying the pending tails of the region, unless a replay of it still runs.
44    pub(crate) fn maybe_replay_tails(&mut self, region_id: RegionId) {
45        let Some(region) = self.regions.get_region(region_id) else {
46            return;
47        };
48        if self.tail_replays.contains_key(&region_id) || !should_replay_tails(&region) {
49            return;
50        }
51        info!("Replaying the pending WAL tails of region {region_id}");
52        let state = CancellableTaskState::new();
53        let task = TailReplayTask {
54            sst_writer: self.tail_replay_limits.new_sst_writer(
55                &region,
56                self.config.clone(),
57                self.cache_manager.clone(),
58            ),
59            region,
60            node_id: self.file_ref_manager.node_id(),
61            wal: self.wal.clone(),
62            listener: self.listener.clone(),
63            request_sender: self.sender.clone(),
64            limits: self.tail_replay_limits.clone(),
65            state: state.clone(),
66        };
67        self.tail_replays.insert(region_id, state);
68        common_runtime::spawn_global(task.run());
69    }

71    /// Cancels the tail replay of the region. The replay still reports its end.
72    pub(crate) fn cancel_tail_replay(&self, region_id: RegionId) {
73        if let Some(state) = self.tail_replays.get(&region_id) {
74            let _ = state.request_cancel();
75        }
76    }

78    /// Discards the pending tails of the region that node `node_id` wrote, without replaying
79    /// them, and sends the number of discarded tails to `sender`. A region without such tails
80    /// sends 0 and changes nothing.
81    ///
82    /// The edit removes the tails, keeps the others in order and raises the flushed entry id
83    /// to the largest `upto` of the removed tails, for a listed tail too, whose `upto` can lie
84    /// above every entry this node wrote. Replaying the WAL at open then skips this node's
85    /// entries up to the new flushed entry id, except those covered by the node's remaining
86    /// tails, which a background replay still reads.
87    ///
88    /// Where the log store checks history and does not accept missing history, opening the
89    /// region also requires this node's WAL to hold history through the new flushed entry
90    /// id. The discard checks neither condition: whoever adds a tail ensures both (see
91    /// [RegionManifest::wal_replay_start](crate::manifest::action::RegionManifest::wal_replay_start)).
92    ///
93    /// The edit is committed outside the worker against the tails the version records now, so
94    /// a replay or a sync that changed them first makes it fail as busy, and a running replay
95    /// of the region is left to find the tails changed. Without conditional publication it also
96    /// fails as busy while a replay's update has an uncertain result, see
97    /// [ManifestContext::update_manifest_for_tails](crate::region::ManifestContext::update_manifest_for_tails).
98    pub(crate) fn handle_discard_pending_tails(
99        &mut self,
100        region_id: RegionId,
101        node_id: u64,
102        sender: OptionOutputTx,
103    ) {
104        let region = match self.regions.writable_non_staging_region(region_id) {
105            Ok(region) => region,
106            Err(e) => {
107                sender.send(Err(e));
108                return;
109            }
110        };
111        let (tails, flushed_entry_id) = {
112            let version = region.version();
113            (version.pending_tails.clone(), version.flushed_entry_id)
114        };
115        let (discarded, remaining): (Vec<_>, Vec<_>) = tails
116            .iter()
117            .cloned()
118            .partition(|tail| tail.node_id == node_id);
119        let Some(upto) = discarded.iter().map(|tail| tail.upto).max() else {
120            info!("Region {region_id} has no pending tails from node {node_id}");
121            sender.send(Ok(0));
122            return;
123        };
124        warn!(
125            "Discarding the pending WAL tails of region {region_id} from node {node_id}: {discarded:?}, flushed entry id: {flushed_entry_id}"
126        );
127        let edit = RegionEdit {
128            timestamp_ms: Some(chrono::Utc::now().timestamp_millis()),
129            flushed_entry_id: (upto > flushed_entry_id).then_some(upto),
130            wal_recovery: WalRecoveryEdit {
131                pending_tails: Some(remaining),
132                handoff_floor: None,
133            },
134            ..Default::default()
135        };
136        let wal = self.wal.clone();
137        let local_node_id = self.file_ref_manager.node_id();
138        let request_sender = self.sender.clone();
139        let listener = self.listener.clone();
140        common_runtime::spawn_global(async move {
141            let action_list =
142                RegionMetaActionList::with_action(RegionMetaAction::Edit(edit.clone()));
143            match region
144                .manifest_ctx
145                .update_manifest_for_tails(&tails, action_list, TailsEditor::Discard)
146                .await
147            {
148                Ok(TailPublication::Committed(manifest_version)) => {
149                    info!(
150                        "Discarded the pending WAL tails of region {region_id} from node {node_id} in manifest version {manifest_version}"
151                    );
152                    // Notifies first: the release is storage I/O, and a replay that ends
153                    // meanwhile restarts from the old tails until the worker applies the edit.
154                    let notify = BackgroundNotify::PendingTailsDiscarded {
155                        region: region.clone(),
156                        edit,
157                        tails,
158                        discarded: discarded.len(),
159                        sender,
160                    };
161                    notify_worker(&request_sender, region_id, notify).await;
162                    release_wal_after_tails_edit(&region, &wal, local_node_id).await;
163                    listener.on_discarded_tails_wal_released(region_id);
164                }
165                Ok(TailPublication::Busy) => {
166                    info!(
167                        "Region {region_id} is altering or editing, or a tail replay's update has an uncertain result, not discarding the tails from node {node_id}"
168                    );
169                    sender.send(RegionBusySnafu { region_id }.fail());
170                }
171                Ok(TailPublication::Stale) => {
172                    let state = region.state();
173                    if state == RegionRoleState::Leader(RegionLeaderState::Writable) {
174                        info!(
175                            "Region {region_id} changed its pending tails, not discarding the tails from node {node_id}"
176                        );
177                        sender.send(RegionBusySnafu { region_id }.fail());
178                    } else {
179                        sender.send(
180                            RegionStateSnafu {
181                                region_id,
182                                state,
183                                expect: RegionRoleState::Leader(RegionLeaderState::Writable),
184                            }
185                            .fail(),
186                        );
187                    }
188                }
189                Err(e) => {
190                    error!(e; "Failed to discard the pending WAL tails of region {region_id} from node {node_id}");
191                    notify_if_fenced(&request_sender, &region).await;
192                    sender.send(Err(e));
193                }
194            }
195        });
196    }

198    /// Applies a published edit that discards pending tails, see [Self::apply_tails_edit], and
199    /// sends the number of `discarded` tails, even if the region changed since.
200    pub(crate) async fn handle_pending_tails_discarded(
201        &mut self,
202        region_id: RegionId,
203        region: MitoRegionRef,
204        edit: RegionEdit,
205        tails: Vec<PendingTail>,
206        discarded: usize,
207        sender: OptionOutputTx,
208    ) {
209        self.apply_tails_edit(region_id, &region, edit, &tails)
210            .await;
211        sender.send(Ok(discarded));
212        self.maybe_replay_tails(region_id);
213    }

215    /// Applies `edit`, which a tail replay or a discard published for the pending tails
216    /// `tails`, unless the instance `region` was replaced, or its version no longer records
217    /// `tails` because a sync rebuilt it from a later manifest.
218    pub(crate) async fn apply_tails_edit(
219        &mut self,
220        region_id: RegionId,
221        region: &MitoRegionRef,
222        edit: RegionEdit,
223        tails: &[PendingTail],
224    ) {
225        let registered = self.regions.get_region(region_id);
226        if !registered.is_some_and(|registered| Arc::ptr_eq(&registered, region)) {
227            info!("Region {region_id} changed after its tails were published");
228            return;
229        }
230        if region.version().pending_tails != tails {
231            info!("Region {region_id} no longer records the published tails, skipping their edit");
232            return;
233        }
234        // A truncate uses the last entry id, which must not be below the flushed entry id.
235        if let Some(flushed_entry_id) = edit.flushed_entry_id
236            && flushed_entry_id > region.version_control.current().last_entry_id
237        {
238            region.version_control.set_entry_id(flushed_entry_id);
239        }
240        let file_metas = edit.files_to_add.clone();
241        region
242            .version_control
243            .apply_edit(Some(edit), &[], region.file_purger.clone());
244        if self.config.index.build_mode == IndexBuildMode::Async && !file_metas.is_empty() {
245            self.handle_rebuild_index(
246                BuildIndexRequest {
247                    region_id,
248                    build_type: IndexBuildType::Flush,
249                    file_metas,
250                    result_senders: Vec::new(),
251                },
252                OptionOutputTx::new(None),
253            )
254            .await;
255        }
256        self.schedule_compaction(region).await;
257    }
258
259    /// Cancels the tail replay of the region unless `region`, the instance a role change made
260    /// unable to replay, was replaced since.
261    pub(crate) fn handle_region_cannot_replay(&self, region_id: RegionId, region: MitoRegionRef) {
262        let registered = self.regions.get_region(region_id);
263        if registered.is_some_and(|registered| Arc::ptr_eq(&registered, &region)) {
264            self.cancel_tail_replay(region_id);
265        }
266    }
267
268    /// Forgets the ended tail replay of the region and starts another if it still has tails.
269    pub(crate) fn handle_tail_replay_ended(&mut self, region_id: RegionId) {
270        self.tail_replays.remove(&region_id);
271        self.listener.on_tail_replay_ended(region_id);
272        self.maybe_replay_tails(region_id);
273    }

276/// Asks the worker to replay the pending tails of a region that became writable, if it has
277/// any, without waiting.
278pub(crate) fn notify_writable(sender: &Sender<WorkerRequestWithTime>, region: &MitoRegionRef) {
279    if !region.version().has_pending_tails() {
280        return;
281    }
282    let sender = sender.clone();
283    let region_id = region.region_id;
284    common_runtime::spawn_global(async move {
285        notify_worker(&sender, region_id, BackgroundNotify::RegionBecameWritable).await;
286    });
287}

289/// Asks the worker to cancel the tail replay of a region that a role change outside the worker
290/// made unable to replay, if it has pending tails, without waiting. A replay the cancel reaches
291/// after the region became writable again ends and the worker starts another.
292pub(crate) fn notify_cannot_replay(sender: &Sender<WorkerRequestWithTime>, region: &MitoRegionRef) {
293    if !region.version().has_pending_tails() {
294        return;
295    }
296    let sender = sender.clone();
297    let region_id = region.region_id;
298    let notify = BackgroundNotify::RegionCannotReplay {
299        region: region.clone(),
300    };
301    common_runtime::spawn_global(async move {
302        notify_worker(&sender, region_id, notify).await;
303    });
