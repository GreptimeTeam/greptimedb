121/// A fixed size group of [RegionWorkers](RegionWorker).
122///
123/// A worker group binds each region to a specific [RegionWorker] and sends
124/// requests to region's dedicated worker.
125///
126/// ```mermaid
127/// graph LR
128///
129/// RegionRequest -- Route by region id --> Worker0 & Worker1
130///
131/// subgraph MitoEngine
132///     subgraph WorkerGroup
133///         Worker0["RegionWorker 0"]
134///         Worker1["RegionWorker 1"]
135///     end
136/// end
137///
138/// Chan0[" Request channel 0"]
139/// Chan1[" Request channel 1"]
140/// WorkerThread1["RegionWorkerLoop 1"]
141///
142/// subgraph WorkerThread0["RegionWorkerLoop 0"]
143///     subgraph RegionMap["RegionMap (regions bound to worker 0)"]
144///         Region0["Region 0"]
145///         Region2["Region 2"]
146///     end
147///     Buffer0["RequestBuffer"]
148///
149///     Buffer0 -- modify regions --> RegionMap
150/// end
151///
152/// Worker0 --> Chan0
153/// Worker1 --> Chan1
154/// Chan0 --> Buffer0
155/// Chan1 --> WorkerThread1
156/// ```
157pub(crate) struct WorkerGroup {
158    /// Workers of the group.
159    workers: Vec<RegionWorker>,
160    /// Flush background job pool.
161    flush_job_pool: SchedulerRef,
162    /// Compaction background job pool.
163    compact_job_pool: Arc<CompactionCoordinator>,
164    /// Scheduler for index build jobs.
165    index_build_job_pool: SchedulerRef,
166    /// Scheduler for file purgers.
167    purge_scheduler: SchedulerRef,
168    /// Cache.
169    cache_manager: CacheManagerRef,
170    /// File reference manager.
171    file_ref_manager: FileReferenceManagerRef,
172    /// Gc limiter to limit concurrent gc jobs.
173    gc_limiter: GcLimiterRef,
174    node_lease: Arc<NodeLease>,
175    granted_leases: Arc<GrantedLeases>,
176    /// Object store manager.
177    object_store_manager: ObjectStoreManagerRef,
178    /// Puffin manager factory.
179    puffin_manager_factory: PuffinManagerFactory,
180    /// Intermediate manager.
181    intermediate_manager: IntermediateManager,
182    /// Schema metadata manager.
183    schema_metadata_manager: SchemaMetadataManagerRef,
184}

309                    partition_expr_fetcher: partition_expr_fetcher.clone(),
310                    flush_semaphore: flush_semaphore.clone(),
311                    tail_replay_limits: tail_replay_limits.clone(),
312                    #[cfg(feature = "hdfs-object-store")]

322        Ok(WorkerGroup {

328            cache_manager,

332            granted_leases,
333            object_store_manager,
334            puffin_manager_factory,
335            intermediate_manager,
336            schema_metadata_manager,
337        })
338    }

340    /// Stops the worker group.
341    pub(crate) async fn stop(&self) -> Result<()> {
342        info!("Stop region worker group");
343
344        // TODO(yingwen): Do we need to stop gracefully?
345        // Stops the scheduler gracefully.
346        self.compact_job_pool.stop().await?;
347        // Stops the scheduler gracefully.
348        self.flush_job_pool.stop(true).await?;
349        // Stops the purge scheduler gracefully.
350        self.purge_scheduler.stop(true).await?;
351        // Stops the index build job pool gracefully.
352        self.index_build_job_pool.stop(true).await?;
353
354        try_join_all(self.workers.iter().map(|worker| worker.stop())).await?;
355
356        Ok(())
357    }

368    /// Submits the writes of one request, those of each worker in one message, so that the
369    /// worker handles them in one batch. The messages are sent concurrently, so a full worker
370    /// queue holds back no other worker. A write whose worker has stopped receives the error a
371    /// write submitted on its own gets.
372    pub(crate) async fn submit_writes(&self, writes: Vec<SenderWriteRequest>) {
373        let mut writes_by_worker: HashMap<usize, Vec<SenderWriteRequest>> = HashMap::new();
374        for write in writes {
375            let index = region_id_to_index(write.request.region_id, self.workers.len());
376            writes_by_worker.entry(index).or_default().push(write);
377        }
378        join_all(
379            writes_by_worker
380                .into_iter()
381                .map(|(index, writes)| async move {
382                    let worker = &self.workers[index];
383                    let Err(request) = worker
384                        .try_submit_request(WorkerRequest::BatchWrite(writes))
385                        .await
386                    else {
387                        return;
388                    };
389                    let WorkerRequest::BatchWrite(writes) = request else {
390                        unreachable!("the worker returns the request it was given");
391                    };
392                    for write in writes {
393                        write
394                            .sender
395                            .send(WorkerStoppedSnafu { id: worker.id }.fail());
396                    }
397                }),
398        )
399        .await;
400    }

676/// Marks the rebuildable local index layout, and resets the store when it changes.
677///
678/// This branch is experimental and has no deployments, so it pins the token at `1`: a
679/// cache written by any other build is discarded on startup and rebuilt from the SSTs.
680/// Artifacts are additionally rejected on open when they lack the mandatory
681/// `identity_encoding` discriminator, so a stale artifact is never served even if the
682/// reset did not happen (a read that precedes it, or an artifact outside this cache).
683const SERIES_INDEX_CACHE_VERSION: &str = "1";
684
685/// Resets incompatible caches before any worker or reader can use their catalogs.
686async fn initialize_series_index_cache(store: &ObjectStore) -> Result<()> {
687    let version = match store.read("version").await {
688        Ok(bytes) => Some(bytes.to_bytes()),
689        Err(error) if error.kind() == object_store::ErrorKind::NotFound => None,
690        Err(error) => return Err(error).context(crate::error::OpenDalSnafu),
691    };
692    if version.as_deref() == Some(SERIES_INDEX_CACHE_VERSION.as_bytes()) {
693        return Ok(());
694    }
695    common_telemetry::info!("Resetting series-index cache: missing or incompatible layout version");
696    // Preserve the cache root, which may be a mount point.
697    let entries = store.list("/").await.context(crate::error::OpenDalSnafu)?;
698    for entry in entries {
699        // Listings can include the directory itself.
700        if entry.path() == "/" {
701            continue;
702        }
703        store
704            .delete_with(entry.path())
705            .recursive(true)
706            .await
707            .context(crate::error::OpenDalSnafu)?;
708    }
709    store
710        .write("version", SERIES_INDEX_CACHE_VERSION)
711        .await
712        .context(crate::error::OpenDalSnafu)?;
713    common_telemetry::info!("Series-index cache reset complete");
714    Ok(())
715}

962impl RegionWorker {
963    /// Submits request to background worker thread.
964    async fn submit_request(&self, request: WorkerRequest) -> Result<()> {
965        self.try_submit_request(request)
966            .await
967            .map_err(|_| WorkerStoppedSnafu { id: self.id }.build())
968    }
969
970    /// Submits request to background worker thread, or returns it if the worker is stopped.
971    async fn try_submit_request(
972        &self,
973        request: WorkerRequest,
974    ) -> std::result::Result<(), WorkerRequest> {
975        if !self.is_running() {
976            return Err(request);
977        }
978        let request_with_time = WorkerRequestWithTime::new(request);
979        if let Err(mpsc::error::SendError(request_with_time)) =
980            self.sender.send(request_with_time).await
981        {
982            warn!(
983                "Worker {} is already exited but the running flag is still true",
984                self.id
985            );
986            // Manually set the running flag to false to avoid printing more warning logs.
987            self.set_running(false);
988            if let Some(state) = &self.series_index_task_state {
989                state.stop();
990            }
991            return Err(request_with_time.request);
992        }
993
994        Ok(())
995    }
996
997    /// Stop the worker.
998    ///
999    /// This method waits until the worker thread exists.
1000    async fn stop(&self) -> Result<()> {
1001        let handle = self.handle.lock().await.take();
1002        self.set_running(false);
1003        if let Some(state) = &self.series_index_task_state {
1004            state.stop();
1005        }
1006
1007        let mut worker_result = Ok(());
1008        if let Some(handle) = handle {
1009            info!("Stop region worker {}", self.id);
1010
1011            if self
1012                .sender
1013                .send(WorkerRequestWithTime::new(WorkerRequest::Stop))
1014                .await
1015                .is_err()
1016            {
1017                warn!("Worker {} is already exited before stop", self.id);
1018            }
1019
1020            worker_result = handle.await.context(JoinSnafu);
1021        }
1022        let series_index_result = if let Some(handle) = self.series_index_handle.lock().await.take()
1023        {
1024            handle.await.context(JoinSnafu)
1025        } else {
1026            Ok(())
1027        };
1028        worker_result?;
1029        series_index_result?;
1030
1031        Ok(())
1032    }

1036        self.running.load(Ordering::Relaxed)
1037    }

1096/// Buffer for stalled write requests.
1097///
1098/// Maintains stalled write requests, their estimated size, and the regions whose writes wait for
1099/// the end of staging to drain them.
1100#[derive(Default)]
1101pub(crate) struct StalledRequests {
1102    /// Stalled requests.
1103    /// Remember to use `StalledRequests::stalled_count()` to get the total number of stalled requests
1104    /// instead of `StalledRequests::requests.len()`.
1105    ///
1106    /// Key: RegionId
1107    /// Value: (estimated size, stalled requests)
1108    pub(crate) requests:
1109        HashMap<RegionId, (usize, Vec<SenderWriteRequest>, Vec<SenderBulkRequest>)>,
1110    /// Estimated size of all stalled requests.
1111    pub(crate) estimated_size: usize,
1112    /// Regions with writes stalled while they were ending staging. Later writes of these regions
1113    /// stall behind them until [BackgroundNotify::StagingEnded] handles them, so they keep their
1114    /// order.
1115    staging_drains: HashSet<RegionId>,
1116}
1117
1118impl StalledRequests {
1119    /// Returns the estimated size of stalled requests for a region.
1120    pub(crate) fn estimated_size(&self, region_id: &RegionId) -> usize {
1121        self.requests
1122            .get(region_id)
1123            .map(|(size, _, _)| *size)
1124            .unwrap_or_default()
1125    }
1126
1127    /// Appends stalled requests.
1128    pub(crate) fn append(
1129        &mut self,
1130        requests: &mut Vec<SenderWriteRequest>,
1131        bulk_requests: &mut Vec<SenderBulkRequest>,
1132    ) {
1133        for req in requests.drain(..) {
1134            self.push(req);
1135        }
1136        for req in bulk_requests.drain(..) {
1137            self.push_bulk(req);
1138        }
1139    }
1140
1141    /// Pushes a stalled request to the buffer.
1142    pub(crate) fn push(&mut self, req: SenderWriteRequest) {
1143        let (size, requests, _) = self.requests.entry(req.request.region_id).or_default();
1144        let req_size = req.request.estimated_size();
1145        *size += req_size;
1146        self.estimated_size += req_size;
1147        requests.push(req);
1148    }
1149
1150    pub(crate) fn push_bulk(&mut self, req: SenderBulkRequest) {
1151        let region_id = req.region_id;
1152        let (size, _, requests) = self.requests.entry(region_id).or_default();
1153        let req_size = req.request.estimated_size();
1154        *size += req_size;
1155        self.estimated_size += req_size;
1156        requests.push(req);
1157    }
1158
1159    /// Removes stalled requests of specific region.
1160    pub(crate) fn remove(
1161        &mut self,
1162        region_id: &RegionId,
1163    ) -> (Vec<SenderWriteRequest>, Vec<SenderBulkRequest>) {
1164        if let Some((size, write_reqs, bulk_reqs)) = self.requests.remove(region_id) {
1165            self.estimated_size -= size;
1166            (write_reqs, bulk_reqs)
1167        } else {
1168            (vec![], vec![])
1169        }
1170    }
1171
1172    /// Returns the total number of all stalled requests.
1173    pub(crate) fn stalled_count(&self) -> usize {
1174        self.requests
1175            .values()
1176            .map(|(_, reqs, bulk_reqs)| reqs.len() + bulk_reqs.len())
1177            .sum()
1178    }
1179
1180    /// Records that `region_id` stalled a write while ending staging, so its later writes stall
1181    /// behind it until [Self::remove_staging_drain].
1182    pub(crate) fn add_staging_drain(&mut self, region_id: RegionId) {
1183        self.staging_drains.insert(region_id);

1210/// Maximum number of pending DDL requests per region.
1211pub(crate) const MAX_PENDING_DDLS_PER_REGION: usize = 128;
1212
1213/// Lifecycle requests waiting for a region manifest operation to finish.
1214#[derive(Default)]
1215struct PendingDdls(HashMap<RegionId, VecDeque<SenderDdlRequest>>);
1216
1217impl PendingDdls {
1218    fn enqueue(&mut self, ddl: SenderDdlRequest) -> std::result::Result<(), SenderDdlRequest> {
1219        let queue = self.0.entry(ddl.region_id).or_default();
1220        if queue.len() >= MAX_PENDING_DDLS_PER_REGION {
1221            return Err(ddl);
1222        }
1223        debug!(
1224            region_id = %ddl.region_id,
1225            request = ?ddl.request,
1226            queue_len = queue.len() + 1,
1227            "Deferring DDL behind region manifest work"
1228        );
1229        queue.push_back(ddl);
1230        Ok(())
1231    }
1232
1233    fn pop_front(&mut self, region_id: RegionId) -> Option<SenderDdlRequest> {
1234        let queue = self.0.get_mut(&region_id)?;
1235        let ddl = queue.pop_front();
1236        if queue.is_empty() {
1237            self.0.remove(&region_id);
1238        }
1239        ddl
1240    }
1241
1242    fn remove(&mut self, region_id: RegionId) -> Option<VecDeque<SenderDdlRequest>> {
1243        self.0.remove(&region_id)
1244    }
1245
1246    fn contains_region(&self, region_id: RegionId) -> bool {
1247        self.0.contains_key(&region_id)
1248    }
1249
1250    fn has_close(&self, region_id: RegionId) -> bool {
1251        self.0.get(&region_id).is_some_and(|queue| {
1252            queue
1253                .iter()
1254                .any(|ddl| matches!(ddl.request, DdlRequest::Close(_)))
1255        })
1256    }
1257
1258    fn has_enter_staging(&self, region_id: RegionId) -> bool {
1259        self.0.get(&region_id).is_some_and(|queue| {
1260            queue
1261                .iter()
1262                .any(|ddl| matches!(ddl.request, DdlRequest::EnterStaging(_)))
1263        })
1264    }
1265
1266    fn front_is_enter_staging(&self, region_id: RegionId) -> bool {
1267        self.0
1268            .get(&region_id)
1269            .and_then(VecDeque::front)
1270            .is_some_and(|ddl| matches!(ddl.request, DdlRequest::EnterStaging(_)))
1271    }
1272
1273    fn take_closes(&mut self, region_id: RegionId) -> Vec<SenderDdlRequest> {
1274        let Some(mut queue) = self.0.remove(&region_id) else {
1275            return Vec::new();
1276        };
1277        let mut matching = Vec::new();
1278        let mut remaining = VecDeque::new();
1279        while let Some(ddl) = queue.pop_front() {
1280            if matches!(ddl.request, DdlRequest::Close(_)) {
1281                matching.push(ddl);
1282            } else {
1283                remaining.push_back(ddl);
1284            }
1285        }
1286        if !remaining.is_empty() {
1287            self.0.insert(region_id, remaining);
1288        }
1289        matching
1290    }
1291}

1408    /// Publishes a loaded region together with its resident compaction state.
1409    fn register_region(&mut self, region: MitoRegionRef) {
1410        if let Some(old) = self.regions.get_region(region.region_id) {
1411            old.deactivate_region_manifest_size_metric();
1412            self.cancel_tail_replay(region.region_id);
1413        }
1414
1415        self.compaction_scheduler.register_region(
1416            &region.version_control,
1417            &region.access_layer,
1418            &region.compaction_state,
1419        );
1420        self.regions.insert_region(region.clone());
1421        region.activate_region_manifest_size_metric();
1422        // A create can open an existing region as a writable leader.
1423        self.maybe_replay_tails(region.region_id);
1424    }

1426    /// Starts the worker loop.
1427    async fn run(&mut self) {
1428        let init_check_delay = worker_init_check_delay();
1429        info!(
1430            "Start region worker thread {}, init_check_delay: {:?}",
1431            self.id, init_check_delay
1432        );
1433        self.last_periodical_check_millis += init_check_delay.as_millis() as i64;
1434
1435        // Buffer to retrieve requests from receiver.
1436        let mut write_req_buffer: Vec<SenderWriteRequest> =
1437            Vec::with_capacity(self.config.worker_request_batch_size);
1438        let mut bulk_req_buffer: Vec<SenderBulkRequest> =
1439            Vec::with_capacity(self.config.worker_request_batch_size);
1440        let mut ddl_req_buffer: Vec<SenderDdlRequest> =
1441            Vec::with_capacity(self.config.worker_request_batch_size);
1442        let mut general_req_buffer: Vec<WorkerRequest> =
1443            RequestBuffer::with_capacity(self.config.worker_request_batch_size);
1444
1445        while self.running.load(Ordering::Relaxed) {
1446            // Clear the buffer before handling next batch of requests.
1447            write_req_buffer.clear();
1448            ddl_req_buffer.clear();
1449            general_req_buffer.clear();
1450            let mut bulk_insert_req_num = 0;
1451
1452            let max_wait_time = self.time_provider.wait_duration(CHECK_REGION_INTERVAL);
1453            let sleep = tokio::time::sleep(max_wait_time);
1454            tokio::pin!(sleep);
1455
1456            tokio::select! {
1457                request_opt = self.receiver.recv() => {
1458                    match request_opt {
1459                        Some(request_with_time) => {
1460                            // Observe the wait time
1461                            let wait_time = request_with_time.created_at.elapsed();
1462                            self.request_wait_time.observe(wait_time.as_secs_f64());
1463
1464                            match request_with_time.request {
1465                                WorkerRequest::Write(sender_req) => write_req_buffer.push(sender_req),
1466                                WorkerRequest::BatchWrite(sender_reqs) => write_req_buffer.extend(sender_reqs),
1467                                WorkerRequest::Ddl(sender_req) => ddl_req_buffer.push(sender_req),
1468                                WorkerRequest::BulkInserts(bulk_insert) => {
1469                                    bulk_insert_req_num += 1;
1470                                    self.buffer_bulk_insert_request(bulk_insert, &mut bulk_req_buffer)
1471                                        .await;
1472                                }
1473                                req => general_req_buffer.push(req),
1474                            }
1475                        },
1476                        // The channel is disconnected.
1477                        None => break,
1478                    }
1479                }
1480                recv_res = self.flush_receiver.changed() => {
1481                    if recv_res.is_err() {
1482                        // The channel is disconnected.
1483                        break;
1484                    } else {
1485                        // Also flush this worker if other workers trigger flush as this worker may have
1486                        // a large memtable to flush. We may not have chance to flush that memtable if we
1487                        // never write to this worker. So only flushing other workers may not release enough
1488                        // memory.
1489                        self.maybe_flush_worker();
1490                        // A flush job is finished, handles stalled requests.
1491                        self.handle_stalled_requests().await;
1492                        continue;
1493                    }
1494                }
1495                result = self.wal_flush_receiver.changed() => {
1496                    if result.is_err() {
1497                        break;
1498                    }
1499                    let candidates = self.wal_flush_receiver.borrow_and_update().clone();
1500                    if let Err(e) = self.flush_for_wal_pressure(&candidates) {
1501                        error!(e; "Failed to flush regions under WAL pressure");
1502                    }
1503                    #[cfg(any(test, feature = "test"))]
1504                    if let Some(listener) = &self.listener.listener {
1505                        for &region_id in candidates.iter() {
1506                            if self.regions.get_region(region_id).is_some() {
1507                                listener.on_wal_pressure_checked(
1508                                    region_id,
1509                                    self.flush_scheduler.is_flush_requested(region_id),
1510                                );
1511                            }
1512                        }
1513                    }
1514                    // Notifications reset the loop's sleep, so they must also
1515                    // check periodic work even when the candidate list is empty.
1516                    self.handle_periodical_tasks();
1517                    continue;
1518                }
1519                _ = &mut sleep => {
1520                    // Timeout. Checks periodical tasks.
1521                    self.handle_periodical_tasks();
1522                    continue;
1523                }
1524            }
1525
1526            if self.flush_receiver.has_changed().unwrap_or(false) {
1527                // Always checks whether we could process stalled requests to avoid a request
1528                // hangs too long.
1529                // If the channel is closed, do nothing.
1530                self.handle_stalled_requests().await;
1531            }
1532
1533            // Try to recv more requests from the channel.
1534            for _ in 1..self.config.worker_request_batch_size {
1535                // We have received one request so we start from 1.
1536                match self.receiver.try_recv() {
1537                    Ok(request_with_time) => {
1538                        // Observe the wait time
1539                        let wait_time = request_with_time.created_at.elapsed();
1540                        self.request_wait_time.observe(wait_time.as_secs_f64());
1541
1542                        match request_with_time.request {
1543                            WorkerRequest::Write(sender_req) => write_req_buffer.push(sender_req),
1544                            WorkerRequest::BatchWrite(sender_reqs) => {
1545                                write_req_buffer.extend(sender_reqs)
1546                            }
1547                            WorkerRequest::Ddl(sender_req) => ddl_req_buffer.push(sender_req),
1548                            WorkerRequest::BulkInserts(bulk_insert) => {
1549                                bulk_insert_req_num += 1;
1550                                self.buffer_bulk_insert_request(bulk_insert, &mut bulk_req_buffer)
1551                                    .await
1552                            }
1553                            req => general_req_buffer.push(req),
1554                        }
1555                    }
1556                    // We still need to handle remaining requests.
1557                    Err(_) => break,
1558                }
1559            }
1560
1561            self.listener.on_recv_requests(
1562                write_req_buffer.len()
1563                    + ddl_req_buffer.len()
1564                    + general_req_buffer.len()
1565                    + bulk_insert_req_num,
1566            );
1567
1568            self.handle_requests(
1569                &mut write_req_buffer,
1570                &mut ddl_req_buffer,
1571                &mut general_req_buffer,
1572                &mut bulk_req_buffer,
1573            )
1574            .await;
1575            self.sweep_deferred_lease_work().await;
1576
1577            self.handle_periodical_tasks();
1578        }
1579
1580        self.clean().await;
1581
1582        info!("Exit region worker thread {}", self.id);
1583    }

1610    /// Dispatches and processes requests.
1611    ///
1612    /// `general_requests` should not contain categorized write, ddl, or bulk insert requests.
1613    async fn handle_requests(
1614        &mut self,
1615        write_requests: &mut Vec<SenderWriteRequest>,
1616        ddl_requests: &mut Vec<SenderDdlRequest>,
1617        general_requests: &mut Vec<WorkerRequest>,
1618        bulk_requests: &mut Vec<SenderBulkRequest>,
1619    ) {
1620        for worker_req in general_requests.drain(..) {
1621            match worker_req {
1622                WorkerRequest::Write(_) | WorkerRequest::BatchWrite(_) | WorkerRequest::Ddl(_) => {
1623                    // These requests are categorized before dispatching general requests.
1624                    continue;
1625                }
1626                WorkerRequest::BulkInserts(_) => unreachable!("bulk inserts are buffered"),
1627                WorkerRequest::Background { region_id, notify } => {
1628                    if matches!(
1629                        &notify,
1630                        BackgroundNotify::RegionEdit(edit_result)
1631                            if edit_result.update_region_state
1632                    ) {
1633                        // Region state must be Editing when reach here.
1634                        // This call only moves write/bulk write request into stall queue. When region edit result
1635                        // is processed inside handle_background_notify and region state is switched back to Writable,
1636                        // stalled request will be processed before the next region edit is dequeued from
1637                        // RegionEditQueue immediately in handle_region_edit_result. It not only ensured pending writes
1638                        // are processed in time, but also prevents them from starvation.
1639                        // TODO(hl): maybe we need to merge those queues for pending requests like pending_ddl,
1640                        // region edits and stalled request, so we can simplify the coordination between these queues.
1641                        self.handle_buffered_region_write_requests(
1642                            &region_id,
1643                            write_requests,
1644                            bulk_requests,
1645                        )
1646                        .await;
1647                    }
1648                    // For background notify, we handle it directly.
1649                    self.handle_background_notify(region_id, notify).await;
1650                }
1651                WorkerRequest::SetRegionRoleStateGracefully {
1652                    region_id,
1653                    region_role_state,
1654                    sender,
1655                } => {
1656                    self.set_role_state_gracefully(region_id, region_role_state, sender)
1657                        .await;
1658                }
1659                WorkerRequest::EditRegion(request) => {
1660                    self.handle_incoming_region_edit(request);
1661                }
1662                WorkerRequest::PauseCompaction { region_id, sender } => {
1663                    self.compaction_scheduler
1664                        .pause_compaction(region_id, sender);
1665                }
1666                WorkerRequest::CancelCompaction {
1667                    region_id,
1668                    task_id,
1669                    sender,
1670                } => {
1671                    let result = match self.compaction_scheduler.cancel_task(region_id, &task_id) {
1672                        CancelCompactionResult::CancelIssued
1673                        | CancelCompactionResult::AlreadyCancelling => Ok(1),
1674                        CancelCompactionResult::NotFound => Ok(0),
1675                        CancelCompactionResult::TooLateToCancel => error::InvalidRequestSnafu {
1676                            region_id,
1677                            reason: format!(
1678                                "Compaction task {task_id} has crossed its commit boundary"
1679                            ),
1680                        }
1681                        .fail(),
1682                        CancelCompactionResult::Unsupported => error::UnsupportedOperationSnafu {
1683                            err_msg: format!(
1684                                "Cancellation of remote compaction task {task_id} is unsupported"
1685                            ),
1686                        }
1687                        .fail(),
1688                    };
1689                    sender.send(result);
1690                }
1691                WorkerRequest::DiscardPendingTails {
1692                    region_id,
1693                    node_id,
1694                    sender,
1695                } => {
1696                    self.handle_discard_pending_tails(region_id, node_id, sender);
1697                }
1698                WorkerRequest::Stop => {
1699                    debug_assert!(!self.running.load(Ordering::Relaxed));
1700                }
1701                WorkerRequest::SyncRegion(req) => {
1702                    self.handle_region_sync(req).await;
1703                }
1704                WorkerRequest::RemapManifests(req) => {
1705                    self.handle_remap_manifests_request(req);
1706                }
1707                WorkerRequest::CopyRegionFrom(req) => {
1708                    self.handle_copy_region_from_request(req);
1709                }
1710            }
1711        }
1712
1713        // Handles all write requests first. So we can alter regions without
1714        // considering existing write requests.
1715        self.handle_write_requests(write_requests, bulk_requests, true)
1716            .await;
1717
1718        for ddl in ddl_requests.drain(..) {
1719            let region_id = ddl.region_id;
1720            if self.should_defer_region_ddl(&ddl) {
1721                self.enqueue_region_ddl(ddl);
1722            } else {
1723                self.execute_ddl_request(ddl).await;
1724            }
1725            self.dispatch_pending_region_requests(region_id).await;
1726        }
1727    }

1731        for ddl in ddl_requests.drain(..) {

1736    /// Executes an admitted request, including requests handed back by a scheduler.
1737    async fn execute_ddl_request(&mut self, ddl: SenderDdlRequest) {
1738        let res = match ddl.request {
1739            DdlRequest::Create(req) => self.handle_create_request(ddl.region_id, req).await,
1740            DdlRequest::Drop(req) => {
1741                self.handle_drop_request(ddl.region_id, req, ddl.sender)
1742                    .await;
1743                return;
1744            }
1745            DdlRequest::Open((req, wal_entry_receiver)) => {
1746                self.handle_open_request(ddl.region_id, req, wal_entry_receiver, ddl.sender)
1747                    .await;
1748                return;
1749            }
1750            DdlRequest::OfflineCleanup(req) => {
1751                self.handle_offline_cleanup_request(ddl.region_id, req)
1752                    .await
1753            }
1754            DdlRequest::Close(req) => {
1755                self.handle_close_request(ddl.region_id, req, ddl.sender)
1756                    .await;
1757                return;
1758            }
1759            DdlRequest::Alter(req) => {
1760                self.execute_alter_request(ddl.region_id, req, ddl.sender)
1761                    .await;
1762                return;
1763            }
1764            DdlRequest::Flush(req) => {
1765                self.handle_flush_request(ddl.region_id, req, ddl.sender);
1766                return;
1767            }
1768            DdlRequest::Compact(req) => {

1811            .time_provider

1821            .refresh_flush_candidates(self.last_periodical_check_millis, CHECK_REGION_INTERVAL);

1831    /// Handles region background request
1832    async fn handle_background_notify(&mut self, region_id: RegionId, notify: BackgroundNotify) {
1833        // A reload that fails for good would otherwise restart itself forever.
1834        let is_reload_result = matches!(notify, BackgroundNotify::RegionReloaded { .. });
1835        let should_dispatch_pending = notify.should_dispatch_pending();
1836        match notify {
1837            BackgroundNotify::RegionOpened { region, registered } => {
1838                self.register_region(region);
1839                self.region_count.inc();
1840
1841                // Finish this opening batch before the worker can process a close/reopen,
1842                // without waiting for the background opener to resume after the ACK.
1843                let senders = self.opening_regions.remove_sender(region_id);
1844                if let Some(state) = &self.series_index_task_state {
1845                    state.wake();
1846                }
1847                let _ = registered.send(());
1848                for sender in senders {
1849                    sender.send(Ok(0));
1850                }
1851            }
1852            BackgroundNotify::CompactionUnit(req) => {
1853                self.handle_compaction_unit(region_id, req).await;
1854            }
1855            BackgroundNotify::CompactionDdlComplete {
1856                generation,
1857                request_index,
1858            } => {
1859                self.compaction_scheduler
1860                    .on_ddl_complete(region_id, generation, request_index);
1861            }
1862            BackgroundNotify::CompactionPickFinished(req) => {
1863                self.handle_compaction_pick_finished(region_id, req).await
1864            }
1865            BackgroundNotify::FlushFinished(req) => {
1866                self.handle_flush_finished(region_id, req).await
1867            }
1868            BackgroundNotify::FlushFailed(req) => self.handle_flush_failed(region_id, req).await,
1869            BackgroundNotify::IndexBuildFinished(req) => {
1870                self.handle_index_build_finished(region_id, req).await
1871            }
1872            BackgroundNotify::IndexBuildStopped(req) => {
1873                self.handle_index_build_stopped(region_id, req).await
1874            }
1875            BackgroundNotify::IndexBuildRetry(req) => {
1876                self.handle_rebuild_index_inner(req, OptionOutputTx::new(None), true)
1877                    .await;
1878            }
1879            BackgroundNotify::CompactionFinished(req) => {
1880                self.handle_compaction_finished(region_id, req).await
1881            }
1882            BackgroundNotify::CompactionFailed(req) => self.handle_compaction_failure(req).await,
1883            BackgroundNotify::Truncate(req) => self.handle_truncate_result(req).await,
1884            BackgroundNotify::DiscardUnflushed(req) => {
1885                self.handle_discard_unflushed_result(req).await
1886            }
1887            BackgroundNotify::RegionChange(req) => {
1888                self.handle_manifest_region_change_result(req).await
1889            }
1890            BackgroundNotify::EnterStaging(req) => self.handle_enter_staging_result(req).await,
1891            BackgroundNotify::StagingEnded(drained) => {
1892                self.stalled_requests.remove_staging_drain(&region_id);
1893                self.handle_region_stalled_requests(&region_id, true).await;
1894                let _ = drained.send(());
1895            }
1896            BackgroundNotify::RegionEdit(req) => self.handle_region_edit_result(req).await,
1897            BackgroundNotify::CopyRegionFromFinished(req) => {
1898                self.handle_copy_region_from_finished(req)
1899            }
1900            BackgroundNotify::RegionFenced => {}
1901            BackgroundNotify::RegionReloaded { old, result } => {
1902                self.handle_region_reloaded(region_id, old, result).await
1903            }
1904            BackgroundNotify::CaughtUpAgain(caught_up) => {
1905                self.handle_caught_up_again(region_id, caught_up).await
1906            }
1907            BackgroundNotify::HandedOver(handed_over) => {
1908                self.handle_handed_over(region_id, handed_over)
1909            }
1910            // The sweep after the batch runs the region's deferred work.
1911            BackgroundNotify::RegionBecameWritable => {
1912                self.maybe_replay_tails(region_id);
1913                self.listener.on_region_became_writable(region_id);
1914            }
1915            BackgroundNotify::LeaderLeaseWaitEnded => {
1916                self.handle_leader_lease_wait_ended(region_id).await
1917            }
1918            BackgroundNotify::BackfillDeferred { region } => self.handle_backfill_deferred(region),
1919            BackgroundNotify::RegionCannotReplay { region } => {
1920                self.handle_region_cannot_replay(region_id, region)
1921            }
1922            BackgroundNotify::TailReplayCommitted {
1923                region,
1924                edit,
1925                tails,
1926            } => {
1927                self.apply_tails_edit(region_id, &region, edit, &tails)
1928                    .await
1929            }
1930            BackgroundNotify::TailReplayEnded => self.handle_tail_replay_ended(region_id),
1931            BackgroundNotify::PendingTailsDiscarded {
1932                region,
1933                edit,
1934                tails,
1935                discarded,
1936                sender,
1937            } => {
1938                self.handle_pending_tails_discarded(
1939                    region_id, region, edit, tails, discarded, sender,
1940                )
1941                .await
1942            }
1943        }
1944        // Completion handlers may return early after releasing their scheduler. Dispatch
1945        // here covers those paths as well as their normal completion.
1946        if should_dispatch_pending {
1947            self.dispatch_pending_region_requests(region_id).await;
1948        }
1949        // A commit or a tail replay that found a newer manifest fenced the region before its job
1950        // reported here.
1951        if !is_reload_result {
1952            self.maybe_reload_region(region_id);
1953        }
1954    }

1956    /// Handles `set_region_role_gracefully`.
1957    async fn set_role_state_gracefully(
1958        &mut self,
1959        region_id: RegionId,
1960        region_role_state: SettableRegionRoleState,
1961        sender: oneshot::Sender<SetRegionRoleStateResponse>,
1962    ) {
1963        if let Some(region) = self.regions.get_region(region_id) {
1964            if matches!(
1965                region_role_state,
1966                SettableRegionRoleState::Follower | SettableRegionRoleState::DowngradingLeader
1967            ) {
1968                // Fence writes here: writes the worker handles later see
1969                // Downgrading, and earlier ones have finished. This only
1970                // changes a writable or staging leader, but always fences
1971                // promotions issued before.
1972                region.manifest_ctx.begin_downgrade(region_id);
1973                self.cancel_tail_replay(region_id);
1974            }
1975            #[cfg(feature = "hdfs-object-store")]
1976            let sweep_older_than = self.config.gc.unknown_file_lingering_time;
1977            #[cfg(feature = "hdfs-object-store")]
1978            let sweep_limiter = self.sweep_limiter.clone();
1979            let request_sender = self.sender.clone();
1980            // We need to do this in background as we need the manifest lock.
1981            common_runtime::spawn_global(async move {
1982                let drain = drain_ended_staging(&request_sender, region_id);
1983                match region
1984                    .set_role_state_gracefully(region_role_state, drain)
1985                    .await
1986                {
1987                    Ok(()) => {
1988                        let last_entry_id = region.version_control.current().last_entry_id;
1989                        let _ = sender.send(SetRegionRoleStateResponse::success(
1990                            SetRegionRoleStateSuccess::mito(last_entry_id),
1991                        ));
1992                        // Only a writable leader reclaims what a writer that died
1993                        // before it renamed its temporary file left behind.
1994                        #[cfg(feature = "hdfs-object-store")]
1995                        if region_role_state == SettableRegionRoleState::Leader {
1996                            temp_file_sweeper::spawn_region_sweep(
1997                                region.access_layer(),
1998                                region_id,
1999                                sweep_older_than,
2000                                sweep_limiter,
2001                            );
2002                        }
2003                    }
2004                    Err(e) => {
2005                        error!(e; "Failed to set region {} role state to {:?}", region_id, region_role_state);
2006                        let _ = sender.send(SetRegionRoleStateResponse::invalid_transition(
2007                            BoxedError::new(e),
2008                        ));
2009                    }
2010                }
2011                // The call commits outside the worker when it ends staging or backfills metadata.
2012                handle_reload::notify_if_fenced(&request_sender, &region).await;
2013            });
2014        } else {
2015            let _ = sender.send(SetRegionRoleStateResponse::NotFound);
2016        }
2017    }

2021    /// Cleans up the worker.
2022    async fn clean(&mut self) {
2023        for state in self.tail_replays.values() {
2024            let _ = state.request_cancel();
2025        }
2026        // Closes remaining regions.
2027        let regions = self.regions.list_regions();
2028        for region in &regions {
2029            region.deactivate_region_manifest_size_metric();
2030            region.stop().await;
2031        }
2032        // Another engine may open these regions from the same storage after the stop. Regions
2033        // closed earlier are not waited for; a datanode or standalone shutdown closes every region
2034        // first, so there this only covers regions whose close failed.
2035        join_all(
2036            regions
2037                .iter()
2038                .map(|region| region.manifest_ctx.wait_for_checkpoint_after_stop()),
2039        )
2040        .await;
2041
2042        self.regions.clear();
2043        for region in regions {
2044            self.granted_leases.remove(region.region_id);
2045            self.deferred_lease_work.remove(&region.region_id);
2046        }
2047    }

2050    /// workers can handle stalled requests.
2051    fn notify_group(&mut self) {
2052        // Notifies all receivers.
2053        let _ = self.flush_sender.send(());
2054        // Marks the receiver in current worker as seen so the loop won't be waked up immediately.
2055        self.flush_receiver.borrow_and_update();
2056    }
