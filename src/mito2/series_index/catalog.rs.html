40/// Summary of SSTs sharing a compaction-window-aligned start.
41///
42/// New data changes must have sequences greater than those already indexed. A new
43/// start adds a map entry; new data at an existing start raises its maximum sequence.
44/// The maximum end and sequence may come from different files, so this summary
45/// does not establish uniform sequence coverage throughout the interval.
46#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
47pub(crate) struct WindowSequence {
48    /// Inclusive start in epoch seconds, equal to the key in `window_sequences`.
49    pub(crate) start: i64,
50    /// Exclusive interval end in epoch seconds.
51    pub(crate) end: i64,
52    /// Maximum sequence among the SSTs sharing this aligned start.
53    pub(crate) max_sequence: u64,
54}
55
56/// Self-describing coverage stored in a series-index Parquet footer.
57#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
58pub(crate) struct SeriesIndexEntry {
59    pub(crate) index_uuid: FileId,
60    /// Inclusive bucket start.
61    pub(crate) bucket_start: Timestamp,
62    /// Exclusive bucket end.
63    pub(crate) bucket_end: Timestamp,
64    /// Source SST IDs retained for debugging only. Compaction can replace these
65    /// files without changing indexed data, so IDs must not determine index reuse.
66    pub(crate) source_file_ids: Vec<FileId>,
67    pub(crate) min_file_sequence: u64,
68    pub(crate) max_file_sequence: u64,
69    /// Width used to align the half-open compaction windows, in seconds.
70    pub(crate) compaction_window_secs: i64,
71    /// Source SST summaries keyed by aligned start; intervals may overlap.
72    /// Each file contributes one summary regardless of its span. Equal starts merge
73    /// by taking the maximum end and sequence. See [`WindowSequence`] for the
74    /// sequence assumption used to detect new data. Compaction changing summary
75    /// boundaries may conservatively trigger a rebuild.
76    pub(crate) window_sequences: BTreeMap<i64, WindowSequence>,
77}

101pub(crate) fn series_metadata(entry: &SeriesIndexEntry) -> Result<Vec<KeyValue>> {

108pub(crate) async fn load_catalog<T>(store: &ObjectStore, path: &str) -> Option<T>
109where
110    T: DeserializeOwned,
111{
112    let bytes = match store.read(path).await {
113        Ok(bytes) => bytes.to_bytes(),
114        Err(error) if error.kind() == ErrorKind::NotFound => return None,
115        Err(error) => {
116            warn!(error; "Failed to load series-index catalog, path: {path}");
117            return None;
118        }
119    };
120    match serde_json::from_slice(&bytes) {
121        Ok(catalog) => Some(catalog),
122        Err(error) => {
123            warn!(error; "Invalid series-index catalog, path: {path}, phase: load");
124            None
125        }
126    }
127}
128
129pub(crate) async fn store_catalog<T>(store: &ObjectStore, path: &str, catalog: &T) -> Result<()>
130where
131    T: Serialize,
132{
133    let bytes = serde_json::to_vec_pretty(catalog).context(SerdeJsonSnafu)?;
134    store
135        .write(path, bytes)
136        .await
137        .map(|_| ())
138        .context(OpenDalSnafu)
139}

141/// Best-effort removal of both catalogs when dropping a region.
142pub(crate) async fn delete_catalogs(store: &ObjectStore, region_id: RegionId) {
143    for path in [
144        series_catalog_path(region_id),
145        range_catalog_path(region_id),
146    ] {
147        if let Err(error) = store.delete(&path).await
148            && error.kind() != ErrorKind::NotFound
149        {
150            warn!(error; "Failed to delete index catalog, path: {path}");
151        }
152    }
153}

155/// Restores the in-memory snapshot once when opening a region.
156pub(crate) async fn load_version_control(
157    store: &ObjectStore,
158    region_id: RegionId,
159    purger: &IndexFilePurger,
160) -> SeriesIndexVersionControl {
161    let range = load_catalog::<RangeIndexCatalog>(store, &range_catalog_path(region_id))
162        .await
163        .unwrap_or_default();
164    let series = load_catalog::<SeriesIndexCatalog>(store, &series_catalog_path(region_id))
165        .await
166        .unwrap_or_default();
167    // TODO: Handle catalog entries whose index files are missing from storage.
168    let version = SeriesIndexVersion::new(
169        range.indexes.into_iter().collect(),
170        series
171            .indexes
172            .into_iter()
173            .map(|entry| {
174                (
175                    entry.index_uuid,
176                    SeriesIndexFileHandle::new(region_id, entry, purger.clone()),
177                )
178            })
179            .collect(),
180    );
181    let control = SeriesIndexVersionControl::default();
182    control.publish(std::sync::Arc::new(version));
183    control
184}

228    #[tokio::test]
229    async fn test_load_catalog_defaults_on_missing_invalid_or_unreadable_catalog() {
230        let store = ObjectStore::new(Memory::default()).unwrap();
231        let region_id = RegionId::new(1, 1);
232        let (purger, _receiver) = series_index_channel(store.clone());
233        assert!(
234            load_catalog::<SeriesIndexCatalog>(&store, &series_catalog_path(region_id))
235                .await
236                .is_none()
237        );
238        let control = load_version_control(&store, region_id, &purger).await;
239        assert!(control.current().range_indexes.is_empty());
240        assert!(control.current().series_indexes.is_empty());
241        let file_id = FileId::random();
242        store
243            .write(
244                &range_catalog_path(region_id),
245                serde_json::to_vec(&RangeIndexCatalog {
246                    indexes: vec![file_id],
247                })
248                .unwrap(),
249            )
250            .await
251            .unwrap();
252        store
253            .write(&series_catalog_path(region_id), "invalid")
254            .await
255            .unwrap();
256        let control = load_version_control(&store, region_id, &purger).await;
257        assert!(control.current().range_indexes.contains(&file_id));
258        assert!(control.current().series_indexes.is_empty());
259        let layer = MockLayerBuilder::default()
260            .reader_factory(Arc::new(|_, _, _| Box::new(FailingCatalogReader)))
261            .build()
262            .unwrap();
263        assert!(
264            load_catalog::<SeriesIndexCatalog>(&store, &series_catalog_path(region_id))
265                .await
266                .is_none()
267        );
268        let store = store.layer(layer);
269        assert!(
270            load_catalog::<RangeIndexCatalog>(&store, &range_catalog_path(region_id))
271                .await
272                .is_none()
273        );
274        let control = load_version_control(&store, region_id, &purger).await;
275        assert!(control.current().range_indexes.is_empty());
276        assert!(control.current().series_indexes.is_empty());
277    }

279    #[tokio::test]
280    async fn test_catalog_roundtrip() {
281        let store = ObjectStore::new(Memory::default()).unwrap();
282        let region_id = RegionId::new(1, 1);
283        let entry = SeriesIndexEntry {
284            index_uuid: FileId::random(),
285            bucket_start: Timestamp::new_second(0),
286            bucket_end: Timestamp::new_second(100),
287            source_file_ids: vec![FileId::random()],
288            min_file_sequence: 1,
289            max_file_sequence: 2,
290            compaction_window_secs: 10,
291            window_sequences: BTreeMap::from([
292                (
293                    0,
294                    WindowSequence {
295                        start: 0,
296                        end: 20,
297                        max_sequence: 1,
298                    },
299                ),
300                (
301                    20,
302                    WindowSequence {
303                        start: 20,
304                        end: 100,
305                        max_sequence: 2,
306                    },
307                ),
308            ]),
309        };
310        store_catalog(
311            &store,
312            &series_catalog_path(region_id),
313            &SeriesIndexCatalog {
314                indexes: vec![entry.clone()],
315            },
316        )
317        .await
318        .unwrap();
319        let (purger, _receiver) = series_index_channel(store.clone());
320        let current = load_version_control(&store, region_id, &purger)
321            .await
322            .current();
323        assert!(current.range_indexes.is_empty());
324        assert_eq!(&entry, current.series_indexes[&entry.index_uuid].entry());
325        assert_eq!(current.index_buckets.len(), 1);
326        let bucket = &current.index_buckets[&entry.bucket_start];
327        assert_eq!(bucket.start, entry.bucket_start);
328        assert_eq!(bucket.end, entry.bucket_end);
329        assert_eq!(bucket.index_ids.as_slice(), &[entry.index_uuid]);
330        assert_eq!(bucket.compaction_window_secs, entry.compaction_window_secs);
331        assert_eq!(bucket.window_sequences, entry.window_sequences);
332
333        let metadata = series_metadata(&entry).unwrap();
334        let decoded: SeriesIndexEntry =
335            serde_json::from_str(metadata[0].value.as_ref().unwrap()).unwrap();
336        assert_eq!(entry, decoded);
337    }
