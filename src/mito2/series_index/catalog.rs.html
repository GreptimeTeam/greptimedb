41/// Summary of SSTs sharing a compaction-window-aligned start.
42///
43/// New data changes must have sequences greater than those already indexed. A new
44/// start adds a map entry; new data at an existing start raises its maximum sequence.
45/// The maximum end and sequence may come from different files, so this summary
46/// does not establish uniform sequence coverage throughout the interval.
47#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
48pub(crate) struct WindowSequence {
49    /// Inclusive start in epoch seconds, equal to the key in `window_sequences`.
50    pub(crate) start: i64,
51    /// Exclusive interval end in epoch seconds.
52    pub(crate) end: i64,
53    /// Maximum sequence among the SSTs sharing this aligned start.
54    pub(crate) max_sequence: u64,
55}
56
57/// Self-describing coverage stored in a series-index Parquet footer.
58///
59/// A published entry must include every series from every SST in the region
60/// contained by its bucket and inclusive file-sequence interval. Query planning
61/// relies on this complete-coverage contract. Manual builds additionally use exact
62/// `source_file_ids` membership for SSTs without a file sequence.
63#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
64pub(crate) struct SeriesIndexEntry {
65    #[serde(default, skip_serializing_if = "Option::is_none")]
66    pub(crate) artifact: Option<crate::series_index::artifact::ArtifactMetadata>,
67    /// Completed size in the catalog; zero in the footer written before completion.
68    pub(crate) file_size: u64,
69    pub(crate) index_uuid: FileId,
70    /// Inclusive bucket start.
71    pub(crate) bucket_start: Timestamp,
72    /// Exclusive bucket end.
73    pub(crate) bucket_end: Timestamp,
74    /// Source SST IDs retained for diagnostics and exact coverage of unknown-sequence files. Compaction can replace these
75    /// files without changing indexed data, so known-sequence reuse does not depend on IDs.
76    pub(crate) source_file_ids: Vec<FileId>,
77    pub(crate) min_file_sequence: u64,
78    pub(crate) max_file_sequence: u64,
79    /// Width used to align the half-open compaction windows, in seconds.
80    pub(crate) compaction_window_secs: i64,
81    /// Source SST summaries keyed by aligned start; intervals may overlap.
82    /// Each file contributes one summary regardless of its span. Equal starts merge
83    /// by taking the maximum end and sequence. See [`WindowSequence`] for the
84    /// sequence assumption used to detect new data. Compaction changing summary
85    /// boundaries may conservatively trigger a rebuild.
86    pub(crate) window_sequences: BTreeMap<i64, WindowSequence>,
87}

89impl SeriesIndexEntry {
90    /// Entries written before pluggable formats describe the original Parquet layout.
91    pub(crate) fn format(&self) -> crate::series_index::artifact::ArtifactFormat {
92        self.artifact.as_ref().map_or(
93            crate::series_index::artifact::ArtifactFormat::Parquet,
94            |artifact| artifact.format,
95        )
96    }
97
98    /// Whether this index completely covers an SST in the query's sequence domain.
99    pub(crate) fn covers_file(
100        &self,
101        file: &crate::sst::file::FileMeta,
102        region_id: RegionId,
103    ) -> bool {
104        file.region_id == region_id
105            && file.time_range.0 <= file.time_range.1
106            && self.bucket_start <= file.time_range.0
107            && file.time_range.1 < self.bucket_end
108            && match file.sequence {
109                Some(sequence) => {
110                    self.min_file_sequence <= sequence.get()
111                        && sequence.get() <= self.max_file_sequence
112                }
113                // Manual builds can cover immutable files without sequence metadata.
114                // This is exact membership, never a claim about replacement SSTs.
115                None => self.min_file_sequence == 0 && self.source_file_ids.contains(&file.file_id),
116            }
117    }
118}

168pub(crate) async fn load_catalog<T>(store: &ObjectStore, path: &str) -> Option<T>
169where
170    T: DeserializeOwned,
171{
172    let bytes = match store.read(path).await {
173        Ok(bytes) => bytes.to_bytes(),
174        Err(error) if error.kind() == ErrorKind::NotFound => return None,
175        Err(error) => {
176            warn!(error; "Failed to load series-index catalog, path: {path}");
177            return None;
178        }
179    };
180    match serde_json::from_slice(&bytes) {
181        Ok(catalog) => Some(catalog),
182        Err(error) => {
183            warn!(error; "Invalid series-index catalog, path: {path}, phase: load");
184            None
185        }
186    }
187}
188
189pub(crate) async fn store_catalog<T>(store: &ObjectStore, path: &str, catalog: &T) -> Result<()>
190where
191    T: Serialize,
192{
193    let bytes = serde_json::to_vec_pretty(catalog).context(SerdeJsonSnafu)?;
194    store
195        .write(path, bytes)
196        .await
197        .map(|_| ())
198        .context(OpenDalSnafu)
199}

201/// Best-effort removal of both catalogs when dropping a region.
202pub(crate) async fn delete_catalogs(store: &ObjectStore, region_id: RegionId) {
203    for path in [
204        series_catalog_path(region_id),
205        samples_catalog_path(region_id),
206        range_catalog_path(region_id),
207    ] {
208        if let Err(error) = store.delete(&path).await
209            && error.kind() != ErrorKind::NotFound
210        {
211            warn!(error; "Failed to delete index catalog, path: {path}");
212        }
213    }
214}

216/// Restores the in-memory snapshot once when opening a region.
217pub(crate) async fn load_version_control(
218    store: &ObjectStore,
219    region_id: RegionId,
220    purger: &IndexFilePurger,
221) -> SeriesIndexVersionControl {
222    let range = load_catalog::<RangeIndexCatalog>(store, &range_catalog_path(region_id))
223        .await
224        .unwrap_or_default();
225    let series = load_catalog::<SeriesIndexCatalog>(store, &series_catalog_path(region_id))
226        .await
227        .unwrap_or_default();
228    let samples = load_catalog::<SamplesCatalog>(store, &samples_catalog_path(region_id)).await;
229    let (ranges, indexes, sample_entries) = match samples {
230        Some(catalog) => (catalog.ranges, catalog.series, catalog.entries),
231        None => (range.indexes, series.indexes, Vec::new()),
232    };
233    let mut version = SeriesIndexVersion::new(
234        ranges
235            .into_iter()
236            .map(|entry| (entry.file_id, entry))
237            .collect(),
238        indexes
239            .into_iter()
240            .map(|entry| {
241                (
242                    entry.index_uuid,
243                    SeriesIndexFileHandle::new(region_id, entry, purger.clone()),
244                )
245            })
246            .collect(),
247    );
248    for entry in sample_entries {
249        let id = entry.file_id;
250        version.samples.insert(
251            id,
252            crate::series_index::artifact::SamplesFileHandle::new(
253                crate::sst::file::RegionFileId::new(region_id, id),
254                entry,
255                purger.clone(),
256            ),
257        );
258    }
259    let control = SeriesIndexVersionControl::default();
260    control.publish(std::sync::Arc::new(version));
261    control
262}

387        assert!(control.current().range_indexes.is_empty());
388        assert!(control.current().series_indexes.is_empty());

395                        file_id,
396                        file_size: 1,
397                    }],
398                })
399                .unwrap(),
400            )
401            .await
402            .unwrap();
403        store
404            .write(&series_catalog_path(region_id), "invalid")
405            .await
406            .unwrap();
407        let control = load_version_control(&store, region_id, &purger).await;
408        assert_eq!(1, control.current().range_indexes.len());
409        assert_eq!(1, control.current().range_indexes[&file_id].file_size);
410        assert!(control.current().series_indexes.is_empty());

420        let store = store.layer(layer);
421        assert!(

427        assert!(control.current().range_indexes.is_empty());
428        assert!(control.current().series_indexes.is_empty());
429    }

431    #[tokio::test]
432    async fn test_catalog_roundtrip() {
433        let store = ObjectStore::new(Memory::default()).unwrap();
434        let region_id = RegionId::new(1, 1);
435        let entry = SeriesIndexEntry {
436            artifact: None,
437            file_size: 0,
438            index_uuid: FileId::random(),
439            bucket_start: Timestamp::new_second(0),
440            bucket_end: Timestamp::new_second(100),
441            source_file_ids: vec![FileId::random()],
442            min_file_sequence: 1,
443            max_file_sequence: 2,
444            compaction_window_secs: 10,
445            window_sequences: BTreeMap::from([
446                (
447                    0,
448                    WindowSequence {
449                        start: 0,
450                        end: 20,
451                        max_sequence: 1,
452                    },
453                ),
454                (
455                    20,
456                    WindowSequence {
457                        start: 20,
458                        end: 100,
459                        max_sequence: 2,
460                    },
461                ),
462            ]),
463        };
464        store_catalog(
465            &store,
466            &series_catalog_path(region_id),
467            &SeriesIndexCatalog {
468                indexes: vec![entry.clone()],
469            },
470        )
471        .await
472        .unwrap();
473        let (purger, _receiver) = series_index_channel(store.clone());
474        let current = load_version_control(&store, region_id, &purger)
475            .await
476            .current();
477        assert!(current.range_indexes.is_empty());
478        assert_eq!(&entry, current.series_indexes[&entry.index_uuid].entry());
479        assert_eq!(current.index_buckets.len(), 1);
480        let bucket = &current.index_buckets[&entry.bucket_start];
481        assert_eq!(bucket.start, entry.bucket_start);
482        assert_eq!(bucket.end, entry.bucket_end);
483        assert_eq!(bucket.index_ids.as_slice(), &[entry.index_uuid]);
484        assert_eq!(bucket.compaction_window_secs, entry.compaction_window_secs);
485        assert_eq!(bucket.window_sequences, entry.window_sequences);
486
487        let metadata = series_metadata(&entry).unwrap();
488        let decoded: SeriesIndexEntry =
489            serde_json::from_str(metadata[0].value.as_ref().unwrap()).unwrap();
490        assert_eq!(entry, decoded);
491    }
