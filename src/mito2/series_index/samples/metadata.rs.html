23pub(super) const MAGIC: &[u8; 8] = b"GTSAMPLE";
24pub(super) const HEADER_SIZE: u64 = 16;
25pub(super) const TRAILER_SIZE: u64 = 120;
26pub(super) const PAGE_LIMIT: usize = 65536;
27pub(super) const ROOT_CAPACITY: usize = (PAGE_LIMIT - 12) / 72;
28/// Largest decoded block payload, and therefore the largest frame a decoder has to decompress.
29const CHUNK_LIMIT: usize = 1024 * 1024;
30/// Fixed part of a block descriptor, before its 44-byte column descriptors.
31///
32/// The exact series key is not part of it. A descriptor keeps a fixed-size [`KeyRef`] into
33/// the part's key area instead, so descriptor size is independent of key length and a key
34/// longer than a metadata page is representable.
35pub(super) const BLOCK_FIXED_BYTES: usize = 56;
36/// Smallest possible block descriptor: the fixed part plus its three mandatory columns.
37const MIN_BLOCK_BYTES: usize = BLOCK_FIXED_BYTES + 3 * 44;
38
39/// Descriptor bytes for a block with `columns` column descriptors.
40///
41/// Independent of the key's length: the descriptor references the key, it does not hold it.
42/// This is the one place the descriptor size formula lives, for the writer's page planning
43/// and for readers that inspect decoded pages.
44pub(super) fn descriptor_bytes(columns: usize) -> usize {
45    BLOCK_FIXED_BYTES + 44 * columns
46}

48#[derive(Debug, Clone, Copy)]
49pub(super) struct PayloadReference {
50    pub offset: u64,
51    pub stored: u32,
52    pub decoded: u32,
53    pub checksum: u64,
54    pub encoding: u16,
55    pub compression: u16,
56}
57impl PayloadReference {
58    pub fn end(&self) -> Result<u64> {
59        self.offset
60            .checked_add(u64::from(self.stored))
61            .ok_or_else(|| invalid("sample payload offset overflow"))
62    }
63    pub fn validate(&self, start: u64, end: u64, page: bool) -> Result<()> {
64        let limit = if page { PAGE_LIMIT } else { CHUNK_LIMIT };
65        require(
66            self.stored > 0
67                && self.decoded > 0
68                && self.decoded as usize <= limit
69                && self.stored as usize <= limit + 65536,
70            "sample payload exceeds limits",
71        )?;
72        require(
73            self.offset >= start && self.end()? <= end,
74            "sample payload outside area",
75        )?;
76        if self.compression == 0 {
77            require(self.stored == self.decoded, "raw samples length mismatch")?;
78        }
79        Ok(())
80    }
81    fn encode(&self, out: &mut Vec<u8>) {
82        put_u64(out, self.offset);
83        put_u32(out, self.stored);
84        put_u32(out, self.decoded);
85        put_u64(out, self.checksum);
86        out.extend(self.encoding.to_le_bytes());
87        out.extend(self.compression.to_le_bytes());
88        put_u32(out, 0);
89    }
90    fn decode(c: &mut Cursor<'_>) -> Result<Self> {
91        let result = Self {
92            offset: c.u64()?,
93            stored: c.u32()?,
94            decoded: c.u32()?,
95            checksum: c.u64()?,
96            encoding: c.u16()?,
97            compression: c.u16()?,
98        };
99        require(c.u32()? == 0, "nonzero sample payload reserved field")?;
100        Ok(result)
101    }
102    fn validate_payload(&self, bytes: &[u8]) -> Result<()> {
103        require(
104            bytes.len() == self.stored as usize && codec::checksum(bytes) == self.checksum,
105            "sample checksum mismatch",
106        )
107    }
108    pub fn unpack(&self, bytes: &[u8]) -> Result<Vec<u8>> {
109        self.validate_payload(bytes)?;
110        codec::decompress(bytes, self.compression, self.decoded as usize)
111    }
112    pub fn unpack_with_decoder<'a>(
113        &self,
114        bytes: &'a [u8],
115        decoder: &'a mut codec::PayloadDecoder,
116    ) -> Result<&'a [u8]> {
117        self.validate_payload(bytes)?;
118        decoder.decompress(bytes, self.compression, self.decoded as usize)
119    }
120}
121pub(super) fn pack(
122    bytes: &[u8],
123    offset: u64,
124    encoding: u16,
125    page: bool,
126) -> Result<(PayloadReference, Vec<u8>)> {
127    require(
128        !bytes.is_empty() && bytes.len() <= if page { PAGE_LIMIT } else { CHUNK_LIMIT },
129        "sample encoded payload exceeds limit",
130    )?;
131    let compressed =
132        zstd::bulk::compress(bytes, 3).map_err(|e| invalid(format!("sample compression: {e}")))?;
133    let (compression, stored) = if compressed.len() < bytes.len() {
134        (1, compressed)
135    } else {
136        (0, bytes.to_vec())
137    };
138    let reference = PayloadReference {
139        offset,
140        stored: stored.len() as u32,
141        decoded: bytes.len() as u32,
142        checksum: codec::checksum(&stored),
143        encoding,
144        compression,
145    };
146    reference.validate(offset, reference.end()?, page)?;
147    Ok((reference, stored))
148}

150#[derive(Debug, Clone, Copy, PartialEq, Eq)]
151pub(super) struct Bounds {
152    pub first: MetricSeriesId,
153    pub last: MetricSeriesId,
154    pub min: i64,
155    pub max: i64,
156}
157impl Bounds {
158    pub fn validate(&self) -> Result<()> {
159        require(
160            self.first <= self.last && self.min <= self.max,
161            "reversed sample bounds",
162        )
163    }
164    pub fn extend(&mut self, next: Self) {
165        self.last = next.last;
166        self.min = self.min.min(next.min);
167        self.max = self.max.max(next.max);
168    }
169    fn encode(&self, out: &mut Vec<u8>) {
170        for id in [self.first, self.last] {
171            put_u32(out, id.table_id);
172            put_u64(out, id.tsid);
173        }
174        put_u64(out, self.min as u64);
175        put_u64(out, self.max as u64);
176    }
177    fn decode(c: &mut Cursor<'_>) -> Result<Self> {
178        let result = Self {
179            first: MetricSeriesId {
180                table_id: c.u32()?,
181                tsid: c.u64()?,
182            },
183            last: MetricSeriesId {
184                table_id: c.u32()?,
185                tsid: c.u64()?,
186            },
187            min: c.u64()? as i64,
188            max: c.u64()? as i64,
189        };
190        result.validate()?;
191        Ok(result)
192    }
193}
194#[derive(Debug, Clone)]
195pub(super) struct PageReference {
196    pub bounds: Bounds,
197    pub payload: PayloadReference,
198}
199impl PageReference {
200    fn encode(&self, out: &mut Vec<u8>) {
201        self.bounds.encode(out);
202        self.payload.encode(out);
203    }
204    fn decode(c: &mut Cursor<'_>) -> Result<Self> {
205        let reference = Self {
206            bounds: Bounds::decode(c)?,
207            payload: PayloadReference::decode(c)?,
208        };
209        if reference.payload.encoding != 1 || reference.payload.compression > 1 {
210            return Err(unsupported("samples metadata codec"));
211        }
212        Ok(reference)
213    }

215#[derive(Debug, Clone)]
216pub(super) struct Column {
217    pub role: u32,
218    pub id: u32,
219    pub payload: PayloadReference,
220    pub nulls: u32,
221}
222/// The location of one exact series key inside a part's key area.
223///
224/// This reference is the descriptor's whole identity. The key bytes themselves are stored
225/// once per series in the part's key area, outside every bounded metadata page, so a key may
226/// be longer than a page and the descriptor stays 12 bytes whatever its length is.
227#[derive(Debug, Clone, Copy, PartialEq, Eq)]
228pub(super) struct KeyRef {
229    /// Absolute offset of the first key byte from the start of the part.
230    pub offset: u64,
231    /// Key length in bytes; nonzero.
232    pub length: u32,
233}
234
235impl KeyRef {
236    pub fn end(&self) -> Result<u64> {
237        self.offset
238            .checked_add(u64::from(self.length))
239            .ok_or_else(|| invalid("sample key offset overflow"))
240    }
241
242    /// End of this key's stored entry: its bytes followed by its checksum.
243    ///
244    /// `length` covers the key only, so the stored checksum is exactly [`KEY_CHECKSUM_BYTES`]
245    /// past [`Self::end`] and never part of what the reference itself names.
246    pub fn checksum_end(&self) -> Result<u64> {
247        self.end()?
248            .checked_add(KEY_CHECKSUM_BYTES)
249            .ok_or_else(|| invalid("sample key offset overflow"))
250    }
251
252    fn encode(&self, out: &mut Vec<u8>) {
253        put_u64(out, self.offset);
254        put_u32(out, self.length);
255    }
256
257    fn decode(c: &mut Cursor<'_>) -> Result<Self> {
258        Ok(Self {
259            offset: c.u64()?,
260            length: c.u32()?,
261        })
262    }
263
264    /// Requires this reference to name a nonempty key entry inside `area`.
265    ///
266    /// The entry includes the stored checksum, so a reference whose key ends inside the area but
267    /// whose checksum would not is rejected like any other out-of-area reference.
268    pub fn validate(&self, area: KeyArea) -> Result<()> {
269        require(
270            self.length > 0 && self.offset >= area.start && self.checksum_end()? <= area.end,
271            "sample key reference outside key area",
272        )
273    }
274}

276/// Bytes of the XXH64 stored after each key, seed zero.
277///
278/// A key copy is `[key bytes][checksum]`; the descriptor's reference names only the key, so a
279/// fetched key can be verified before it is used to attribute rows to a series. Without it a
280/// flipped label byte would silently move rows to another series, which the old inline-key layout
281/// could not suffer: its keys sat in a page whose stored bytes were already checksummed.
282pub(super) const KEY_CHECKSUM_BYTES: u64 = std::mem::size_of::<u64>() as u64;
283
284/// A part's key area: exact series keys stored back to back, outside every metadata page.
285///
286/// Keys are appended in block order and are never page-bounded, so their length is bounded by
287/// the part, not by `PAGE_LIMIT`. Every copy is followed by its checksum. Because blocks are
288/// ordered by exact key, every block of one series is adjacent and shares a single copy of the
289/// key, which makes an equal [`KeyRef`] mean "the same exact series".
290#[derive(Debug, Clone, Copy)]
291pub(super) struct KeyArea {
292    pub start: u64,
293    pub end: u64,
294}
295
296/// The three regions of one part, in file order: column payloads, exact series keys, pages.
297#[derive(Debug, Clone, Copy)]
298pub(super) struct Areas {
299    /// End of the column payload area and start of the key area.
300    pub payload_end: u64,
301    /// The key area every block descriptor references.
302    pub keys: KeyArea,
303    /// Start of the metadata area, which is the end of the key area.
304    pub metadata_start: u64,
305}

307#[derive(Debug, Clone)]
308pub(super) struct Block {
309    /// Coarse `(table_id, tsid)` prefix decoded from the block's exact key.
310    ///
311    /// It prunes candidates but never names the owner: `(table_id, tsid)` is a hash, so two
312    /// distinct series can share it. The owner is the key [`Self::key`] references.
313    pub id: MetricSeriesId,
314    /// Fixed-size reference to the block's exact key in the part's key area.
315    ///
316    /// Equal references mean the same exact series; a different series that collides on the
317    /// identifier always carries a different reference.
318    pub key: KeyRef,
319    pub min: i64,
320    pub max: i64,
321    pub ordinal: u64,
322    pub rows: u32,
323    pub columns: Vec<Column>,
324}
325/// The ordering boundary retained between pages and parts, without column descriptors.
326///
327/// A [`KeyRef`] is an offset inside one part's key area, so two references are comparable only
328/// when they belong to the same part; `part` carries that scope in the boundary itself so no
329/// caller can compare references across a part boundary by accident. Two parts hold unrelated
330/// key areas, which is why a series that spans parts is recognised by [`BlockOrder::continues`]
331/// on the exact key bytes instead.
332#[derive(Debug, Clone, Copy, PartialEq, Eq)]
333pub(crate) struct BlockOrder {
334    part: u32,
335    id: MetricSeriesId,
336    key: KeyRef,
337    min: i64,
338    max: i64,
339    ordinal: u64,
340}
341
342impl BlockOrder {
343    /// The part this block belongs to.
344    pub(crate) fn part(&self) -> u32 {
345        self.part
346    }
347
348    /// The coarse `(table_id, tsid)` prefix, which is ordered across parts as well.
349    pub(crate) fn id(&self) -> MetricSeriesId {
350        self.id
351    }
352
353    /// The block's key reference, a location inside [`Self::part`]'s key area.
354    pub(crate) fn key(&self) -> KeyRef {
355        self.key
356    }
357
358    /// Checks the next block of the *same* part against this boundary.
359    ///
360    /// Ownership and ordering of a part are decided on the exact `(table_id, tsid)` and on the
361    /// key reference only: two blocks of one part that share one reference share one key copy,
362    /// so they are one exact series, while a different series always has its own copy, placed
363    /// after the previous one and restarting the ordinal. Only descriptor fields are read, so a
364    /// block this request prunes is never fetched and its key content is not validated.
365    pub(crate) fn precedes(&self, next: &Self, adjacent: bool) -> Result<()> {
366        require(
367            self.part == next.part,
368            "sample key references compared across parts",
369        )?;
370        require(self.id <= next.id, "unordered sample series keys")?;
371        if next.key == self.key {
372            self.continues(next, adjacent)?;
373        } else {
374            // A different series must place its one key copy after the previous series' copy.
375            // Overlap, or a second copy of a key that is still current, is a corrupt reference
376            // rather than a silent second read of the same bytes.
377            require(
378                self.key.checksum_end()? <= next.key.offset,
379                "overlapping sample key references",
380            )?;
381            if adjacent {
382                require(next.ordinal == 0, "first sample block has nonzero ordinal")?;
383            }
384        }
385        Ok(())
386    }
387
388    /// Requires `next` to continue one exact series in time and ordinal.
389    ///
390    /// Equal exact key bytes are what make two blocks the same series across a part boundary,
391    /// where their references belong to different key areas. A repeated key that restarts its
392    /// ordinal or goes back in time is a corrupt part, not a second series.
393    pub(crate) fn continues(&self, next: &Self, adjacent: bool) -> Result<()> {
394        require(
395            self.max <= next.min && self.ordinal < next.ordinal,
396            "unordered sample blocks",
397        )?;
398        if adjacent {
399            require(
400                self.ordinal.checked_add(1) == Some(next.ordinal),
401                "nonconsecutive sample ordinal",
402            )?;
403        }
404        Ok(())
405    }
406}

408impl Block {
409    /// The ordering boundary this block contributes, scoped to its part.
410    pub(crate) fn order(&self, part: u32) -> BlockOrder {
411        BlockOrder {
412            part,
413            id: self.id,
414            key: self.key,
415            min: self.min,
416            max: self.max,
417            ordinal: self.ordinal,
418        }
419    }
420
421    /// Checks this block against the previous block of the same page.
422    ///
423    /// A page never spans parts, so both references name one key area and are comparable; a block
424    /// in another part is compared through [`BlockOrder::continues`] and its exact key instead.
425    fn follows_in_page(&self, previous: &Self, adjacent: bool) -> Result<()> {
426        // One scope for both sides: a page belongs to exactly one part, and no reference built
427        // here outlives the page.
428        const PAGE_SCOPE: u32 = 0;
429        previous
430            .order(PAGE_SCOPE)
431            .precedes(&self.order(PAGE_SCOPE), adjacent)
432    }
433
434    /// The coarse bounds used for pruning. Exact ownership lives in `key`.
435    pub fn bounds(&self) -> Bounds {
436        Bounds {
437            first: self.id,
438            last: self.id,
439            min: self.min,
440            max: self.max,
441        }
442    }
443    pub fn encode(&self, out: &mut Vec<u8>) {
444        put_u32(out, self.id.table_id);
445        put_u64(out, self.id.tsid);
446        self.key.encode(out);
447        put_u64(out, self.min as u64);
448        put_u64(out, self.max as u64);
449        put_u64(out, self.ordinal);
450        put_u32(out, self.rows);
451        put_u32(out, self.columns.len() as u32);
452        for column in &self.columns {
453            put_u32(out, column.role);
454            put_u32(out, column.id);
455            column.payload.encode(out);
456            put_u32(out, column.nulls);
457        }
458    }
459    fn decode(
460        c: &mut Cursor<'_>,
461        manifest: &Manifest,
462        areas: Areas,
463        previous_end: &mut u64,
464    ) -> Result<Self> {
465        let id = MetricSeriesId {
466            table_id: c.u32()?,
467            tsid: c.u64()?,
468        };
469        // Only the key's location is decoded here. The bytes stay in the key area, so a
470        // descriptor never grows with the key and a request that prunes this block never
471        // reads it; the reader validates the reference and its `(table_id, tsid)` prefix
472        // when it fetches the key for a block it actually selects.
473        let key = KeyRef::decode(c)?;
474        key.validate(areas.keys)?;
475        let min = c.u64()? as i64;
476        let max = c.u64()? as i64;
477        let ordinal = c.u64()?;
478        let rows = c.u32()?;
479        let count = c.u32()? as usize;
480        require(
481            min <= max && valid_rows(rows as usize, manifest.fields.len()),
482            "invalid sample block bounds/count",
483        )?;
484        require(
485            count >= 3 && count <= manifest.fields.len() + 3 && count <= c.remaining() / 44,
486            "invalid sample column count",
487        )?;
488        let mut columns = Vec::with_capacity(count);
489        let mut next_field = 0;
490        for index in 0..count {
491            let role = c.u32()?;
492            let column_id = c.u32()?;
493            let payload = PayloadReference::decode(c)?;
494            let nulls = c.u32()?;
495            if index < 3 {
496                require(
497                    role == index as u32 + 1
498                        && column_id
499                            == if index == 0 {
500                                manifest.timestamp.column_id
501                            } else {
502                                0
503                            }
504                        && nulls == 0,
505                    "invalid mandatory sample column",
506                )?;
507            } else {
508                require(
509                    role == 4 && nulls < rows,
510                    "invalid sample field role/null count",
511                )?;
512                let position = manifest.fields[next_field..]
513                    .iter()
514                    .position(|f| f.column_id == column_id)
515                    .ok_or_else(|| invalid("unknown or unordered sample field"))?
516                    + next_field;
517                next_field = position + 1;
518            }
519            if payload.encoding <= 7 {
520                let valid = match role {
521                    1 => matches!(payload.encoding, 1 | 2),
522                    2 => matches!(payload.encoding, 3 | 4),
523                    3 => matches!(payload.encoding, 6 | 7),
524                    4 => payload.encoding == 5,
525                    _ => false,
526                };
527                require(valid, "sample codec role mismatch")?;
528            }
529            payload.validate(*previous_end, areas.payload_end, false)?;
530            *previous_end = payload.end()?;
531            columns.push(Column {
532                role,
533                id: column_id,
534                payload,
535                nulls,
536            });
537        }
538        Ok(Self {
539            id,
540            key,
541            min,
542            max,
543            ordinal,
544            rows,
545            columns,
546        })
547    }
548}
549#[derive(Debug)]
550pub(super) enum Page {
551    Leaf(Vec<Block>),
552    Internal(Vec<PageReference>),
553}
554impl Page {
555    pub(crate) fn retained_size(&self) -> usize {
556        std::mem::size_of::<Self>()
557            + match self {
558                Self::Internal(children) => {
559                    children.capacity() * std::mem::size_of::<PageReference>()
560                }
561                Self::Leaf(blocks) => {
562                    blocks.capacity() * std::mem::size_of::<Block>()
563                        // Each block owns its column descriptors on the heap. Its exact
564                        // series key is not retained: it stays in the part's key area.
565                        + blocks
566                            .iter()
567                            .map(|block| block.columns.capacity() * std::mem::size_of::<Column>())
568                            .sum::<usize>()
569                }
570            }
571    }
572
573    pub fn bounds(&self) -> Result<Bounds> {
574        let bounds: Vec<_> = match self {
575            Self::Leaf(v) => v.iter().map(Block::bounds).collect(),
576            Self::Internal(v) => v.iter().map(|c| c.bounds).collect(),
577        };
578        let mut result = *bounds.first().ok_or_else(|| invalid("empty sample page"))?;
579        for bound in &bounds[1..] {
580            result.extend(*bound);
581        }
582        Ok(result)
583    }
584    pub fn encode(&self) -> Vec<u8> {
585        let mut out = Vec::new();
586        match self {
587            Self::Leaf(v) => {
588                put_u32(&mut out, 0);
589                put_u64(&mut out, v.len() as u64);
590                for block in v {
591                    block.encode(&mut out);
592                }
593            }
594            Self::Internal(v) => {
595                put_u32(&mut out, 1);
596                put_u64(&mut out, v.len() as u64);
597                for child in v {
598                    child.encode(&mut out);
599                }
600            }
601        }
602        out
603    }
604    /// Decodes one metadata page.
605    ///
606    /// `areas` bounds what the page may reference: column payloads end where the key area
607    /// starts, keys live in their own area, and the page area starts at `areas.metadata_start`
608    /// so every child page must lie after it and before its parent.
609    pub fn decode(
610        bytes: &[u8],
611        reference: &PageReference,
612        depth: u32,
613        areas: Areas,
614        manifest: &Manifest,
615    ) -> Result<Self> {
616        require(bytes.len() <= PAGE_LIMIT, "sample page exceeds limit")?;
617        let mut c = Cursor::new(bytes);
618        let kind = c.u32()?;
619        let count = c.length()?;
620        require(count > 0, "empty sample page")?;
621        let page = match (kind, depth) {
622            (0, 1) => {
623                require(
624                    count <= c.remaining() / MIN_BLOCK_BYTES,
625                    "sample block count exceeds page",
626                )?;
627                let mut blocks: Vec<Block> = Vec::with_capacity(count);
628                let mut end = HEADER_SIZE;
629                for _ in 0..count {
630                    let block = Block::decode(&mut c, manifest, areas, &mut end)?;
631                    if let Some(previous) = blocks.last() {
632                        block.follows_in_page(previous, true)?;
633                    }
634                    blocks.push(block);
635                }
636                Self::Leaf(blocks)
637            }
638            (1, 2) => {
639                require(
640                    count <= ROOT_CAPACITY && count <= c.remaining() / 72,
641                    "sample root exceeds capacity",
642                )?;
643                let mut children: Vec<PageReference> = Vec::with_capacity(count);
644                let mut end = areas.metadata_start;
645                for _ in 0..count {
646                    let child = PageReference::decode(&mut c)?;
647                    child
648                        .payload
649                        .validate(end, reference.payload.offset, true)?;
650                    end = child.payload.end()?;
651                    if let Some(previous) = children.last() {
652                        require(
653                            previous.bounds.last <= child.bounds.first,
654                            "unordered sample child ranges",
655                        )?;
656                    }
657                    children.push(child);
658                }
659                Self::Internal(children)
660            }
661            _ => return Err(unsupported("samples metadata kind/depth")),
662        };
663        c.end()?;
664        require(
665            page.bounds()? == reference.bounds,
666            "samples page bounds disagree with parent",
667        )?;
668        Ok(page)
669    }
670}
671pub(super) fn header() -> Vec<u8> {
672    let mut out = MAGIC.to_vec();
673    put_u32(&mut out, VERSION);
674    put_u32(&mut out, 0);
675    out
676}
677pub(super) fn validate_header(bytes: &[u8]) -> Result<()> {
678    let mut c = Cursor::new(bytes);
679    require(c.take(8)? == MAGIC, "sample header magic")?;
680    if c.u32()? != VERSION {
681        return Err(unsupported("samples file version"));
682    }
683    require(c.u32()? == 0, "nonzero sample header reserved field")?;
684    c.end()
685}
686pub(super) struct Trailer {
687    pub root: PageReference,
688    pub depth: u32,
689    pub file_length: u64,
690    /// End of the payload area, which is where the key area starts.
691    pub payload_end: u64,
692    /// End of the key area, which is where the metadata pages start.
693    pub key_end: u64,
694}
695impl Trailer {
696    pub fn encode(&self) -> Vec<u8> {
697        let mut out = Vec::new();
698        self.root.encode(&mut out);
699        put_u32(&mut out, self.depth);
700        put_u32(&mut out, VERSION);
701        put_u64(&mut out, self.file_length);
702        put_u64(&mut out, self.payload_end);
703        put_u64(&mut out, self.key_end);
704        out.extend(MAGIC);
705        let sum = codec::checksum(&out);
706        put_u64(&mut out, sum);
707        out
708    }
709    pub fn decode(bytes: &[u8], file_length: u64) -> Result<Self> {
710        require(
711            bytes.len() == TRAILER_SIZE as usize,
712            "truncated sample trailer",
713        )?;
714        let mut c = Cursor::new(bytes);
715        let root = PageReference::decode(&mut c)?;
716        let depth = c.u32()?;
717        let version = c.u32()?;
718        if !(1..=2).contains(&depth) || version != VERSION {
719            return Err(unsupported("sample trailer version/depth"));
720        }
721        let length = c.u64()?;
722        let payload_end = c.u64()?;
723        let key_end = c.u64()?;
724        require(
725            // The checksum covers everything before it, magic included: the trailer body plus
726            // the 8-byte magic, which is `TRAILER_SIZE - 8` and not a frozen constant.
727            c.take(8)? == MAGIC
728                && c.u64()?
729                    == codec::checksum(
730                        &bytes[..TRAILER_SIZE as usize - std::mem::size_of::<u64>()],
731                    ),
732            "sample trailer checksum/magic",
733        )?;
734        require(
735            length == file_length
736                && payload_end >= HEADER_SIZE
737                && key_end >= payload_end
738                && file_length >= HEADER_SIZE + TRAILER_SIZE,
739            "invalid sample framing",
740        )?;
741        root.payload
742            .validate(key_end, file_length - TRAILER_SIZE, true)?;
743        require(
744            root.payload.end()? == file_length - TRAILER_SIZE,
745            "sample root must end at trailer",
746        )?;
747        Ok(Self {
748            root,
749            depth,
750            file_length,
751            payload_end,
752            key_end,
753        })
754    }
755
756    /// The key area every block descriptor in this part references.
757    pub fn keys(&self) -> KeyArea {
758        KeyArea {
759            start: self.payload_end,
760            end: self.key_end,
761        }
762    }
763
764    /// The part's three regions, as page decoding needs them.
765    pub fn areas(&self) -> Areas {
766        Areas {
767            payload_end: self.payload_end,
768            keys: self.keys(),
769            metadata_start: self.key_end,
770        }
771    }
772}
