60    pub(crate) async fn open_cached(
61        store: ObjectStore,
62        handle: SamplesFileHandle,
63        cache: Option<MetadataCache>,
64    ) -> Result<Self> {
65        let identity = format!("{:?}/{:?}", handle.file_id(), handle.entry());
66        let descriptor = metadata_cache::load(
67            cache.as_ref(),
68            identity,
69            |(manifest, schema): &(Arc<Manifest>, SchemaRef)| {
70                manifest.retained_size()
71                    + 8 * std::mem::size_of::<usize>()
72                    + std::mem::size_of::<datatypes::arrow::datatypes::Schema>()
73                    + schema
74                        .fields()
75                        .iter()
76                        .map(|field| field.size() + std::mem::size_of::<usize>())
77                        .sum::<usize>()
78            },
79            async {
80                let reader = Self::open(store.clone(), handle.clone()).await?;
81                Ok((reader.manifest, reader.schema))
82            },
83        )
84        .await?;
85        Ok(Self {
86            store,
87            handle,
88            parts: Arc::new(
89                (0..descriptor.0.files.len())
90                    .map(|_| OnceCell::new())
91                    .collect(),
92            ),
93            manifest: descriptor.0.clone(),
94            schema: descriptor.1.clone(),
95            cache,
96            payload_cache: None,
97        })
98    }

105    pub async fn open(store: ObjectStore, handle: SamplesFileHandle) -> Result<Self> {
106        let entry = handle.entry();
107        let artifact = &entry.artifact;
108        require(
109            artifact.format == ArtifactFormat::ColumnarSamples,
110            "samples artifact format mismatch",
111        )?;
112        let region = handle.file_id().region_id().as_u64();
113        require(
114            handle.file_id().file_id() == entry.file_id,
115            "samples source handle mismatch",
116        )?;
117        let path = format!(
118            "{}manifest.json",
119            crate::series_index::samples::samples_prefix(region, artifact.generation)
120        );
121        let mut objects = HashMap::new();
122        for object in &artifact.objects {
123            require(
124                objects
125                    .insert(object.path.as_str(), object.file_size)
126                    .is_none(),
127                "duplicate sample artifact object",
128            )?;
129        }
130        let length = *objects
131            .get(path.as_str())
132            .ok_or_else(|| invalid("missing samples manifest"))?;
133        require(
134            length > 0 && length <= MANIFEST_LIMIT as u64,
135            "samples manifest exceeds limit",
136        )?;
137        let bytes = read_range(&store, &path, 0, length).await?;
138        // A manifest that the current format cannot read predates exact identity or is
139        // corrupt; either way it must never be reused.
140        let manifest: Manifest = serde_json::from_slice(&bytes).map_err(|error| {
141            invalid(format!(
142                "samples manifest is not readable; rebuild the artifact: {error}"
143            ))
144        })?;
145        if manifest.version != VERSION {
146            return Err(unsupported("samples manifest version"));
147        }
148        // Old artifacts can be structurally valid and perfectly ordered, so identity
149        // encoding is the only thing that keeps their lost labels out of reads.
150        if manifest.identity_encoding != IdentityEncoding::CURRENT_SAMPLES {
151            return Err(unsupported(
152                "samples manifest identity encoding; rebuild the samples artifact",
153            ));
154        }
155        require(
156            manifest.format == artifact.format
157                && manifest.region_id == region
158                && manifest.source_file_id == entry.file_id
159                && manifest.generation == artifact.generation,
160            "samples manifest identity mismatch",
161        )?;
162        manifest.validate_schema()?;
163        let mut paths = HashSet::from([path]);
164        let mut total = length;
165        let mut previous = None;
166        for (index, part) in manifest.files.iter().enumerate() {
167            require(
168                index as u64 == u64::from(part.part_id)
169                    && part.num_rows > 0
170                    && part.num_blocks > 0
171                    && part.num_blocks <= part.num_rows
172                    && part.first <= part.last
173                    && part.min_timestamp <= part.max_timestamp
174                    && part.file_size >= metadata::HEADER_SIZE + metadata::TRAILER_SIZE,
175                "invalid sample part descriptor",
176            )?;
177            if let Some(last) = previous {
178                require(last <= part.first, "unordered sample parts")?;
179            }
180            previous = Some(part.last);
181            let part_path = manifest.part_path(part.part_id);
182            require(
183                objects.get(part_path.as_str()) == Some(&part.file_size),
184                "sample part size mismatch",
185            )?;
186            paths.insert(part_path);
187            total = total
188                .checked_add(part.file_size)
189                .ok_or_else(|| invalid("sample artifact size overflow"))?;
190        }
191        require(
192            total == artifact.file_size
193                && paths.len() == artifact.objects.len()
194                && artifact.objects.iter().all(|o| paths.contains(&o.path)),
195            "sample artifact objects/size mismatch",
196        )?;
197        Ok(Self {
198            store,
199            parts: Arc::new((0..manifest.files.len()).map(|_| OnceCell::new()).collect()),
200            schema: manifest.schema(),
201            cache: None,
202            payload_cache: None,
203            manifest: Arc::new(manifest),
204            handle,
205        })
206    }

208impl SamplesFormatReader for SamplesReader {
209    fn schema(&self) -> SchemaRef {
210        self.schema.clone()
211    }
212    fn read(
213        &self,
214        request: SamplesReadRequest,
215    ) -> Result<BoxStream<'static, Result<SamplesBatch>>> {
216        let schema = self.schema();
217        let mut projection = Vec::new();
218        let mut seen = HashSet::new();
219        for name in &request.projection {
220            require(seen.insert(name), "duplicate sample projection")?;
221            projection.push(
222                schema
223                    .index_of(name)
224                    .map_err(|_| invalid(format!("unknown sample projection {name}")))?,
225            );
226        }
227        require(
228            !projection.contains(&self.manifest.primary_key_index()),
229            "the exact series key is returned once per batch, not projected as a column",
230        )?;
231        require(
232            request
233                .start
234                .zip(request.end)
235                .is_none_or(|(start, end)| start <= end),
236            "reversed sample time range",
237        )?;
238        let empty = request.keys.as_ref().is_some_and(|keys| keys.is_empty())
239            || request.start.zip(request.end).is_some_and(|(a, b)| a == b);
240        let complete = request.keys.is_none()
241            && request.table_id.is_none()
242            && request.start.is_none()
243            && request.end.is_none();
244        // Coarse pruning bounds come from the exact selection itself, so a reader can never
245        // prune away a part or page that owns one of the selected series.
246        let ids: Option<BTreeSet<MetricSeriesId>> = request
247            .keys
248            .as_ref()
249            .map(|keys| keys.iter().map(ExactSeriesKey::id).collect());
250        let projected = Arc::new(
251            schema
252                .project(&projection)
253                .map_err(|e| invalid(format!("sample projection: {e}")))?,
254        );
255        let field_ids: HashSet<_> = projection
256            .iter()
257            .filter(|&&index| index >= 5)
258            // The exact series key column is appended after the data fields and has no
259            // column id of its own.
260            .filter_map(|&index| self.manifest.fields.get(index - 5))
261            .map(|field| field.column_id)
262            .collect();
263        let cache = self.cache.clone();
264        let manifest = self.manifest.clone();
265        let store = self.store.clone();
266        let handle = self.handle.clone();
267        let parts = self.parts.clone();
268        let payload_cache = self.payload_cache.clone();
269        Ok(Box::pin(async_stream::try_stream! {
270            let _handle = &handle;
271            let mut metrics = ColumnarReadMetrics::default();
272            if empty { return; }
273            // One decoder and one scratch per stream, reused by every block. The decoder holds
274            // the only zstd context this stream builds, and the scratch keeps its containers
275            // instead of rebuilding them per block. Neither is shared with another stream.
276            //
277            // What the stream keeps between blocks is one decompression context (bounded by a
278            // frame's window, which the writer caps at the block's chunk limit) plus the output
279            // scratch its release keeps. It is deliberately not charged to the pool: the native
280            // prepared-samples path supplies an unbounded pool today (TODO: forward
281            // `QueryScanContext`), so a charge here would report rather than enforce.
282            let mut decoder = codec::PayloadDecoder::default();
283            let mut scratch = ReadScratch::for_stream();
284            // Only the consumer identity is hoisted: a reservation still belongs to one block's
285            // batch, so it is registered per block and released with that batch.
286            let buffers_consumer = request.memory_pool.as_ref()
287                .map(|_| MemoryConsumer::new("ColumnarSamplesBuffers"));
288            // Traversal state: the last block this stream visited, whatever it selected, and
289            // whether it is the block immediately before the next one in the artifact.
290            let mut previous: Option<BlockOrder> = None;
291            let mut adjacent = true;
292            // Content state: the last block whose exact key this stream loaded. Selection can
293            // skip a block inside a page, a whole page, or a whole part, so this state is not
294            // advanced by a block whose key was pruned: the two loaded keys on either side of a
295            // gap are still compared with each other.
296            let mut loaded_key: Option<BlockOrder> = None;
297            // The last loaded key of the previous part, with the pool bytes it is charged. A
298            // reference is a location inside one part, so the exact bytes are what decides
299            // whether a series continues into another; this copy outlives every per-part
300            // reservation, so it has a consumer of its own.
301            let boundary_reservation = request.memory_pool.as_ref()
302                .map(|pool| MemoryConsumer::new("ColumnarSamplesBoundaryKey").register(pool));
303            let mut spanning_key: Option<ExactSeriesKey> = None;
304            let mut spanning_bytes = 0usize;
305            for part in &manifest.files {
306                let bounds = Bounds { first: part.first, last: part.last, min: part.min_timestamp, max: part.max_timestamp };
307                if !matches(bounds, &request, ids.as_ref()) { adjacent = false; continue; }
308                let path = manifest.part_path(part.part_id);
309                // One lazy reader per accessed part, shared across this query's streams.
310                let reader = parts[part.part_id as usize].get_or_try_init(|| async {
311                    store.reader_with(&path).concurrent(8).gap(0).await.context(OpenDalSnafu)
312                }).await?;
313                let size = part.file_size;
314                let metadata_started = Instant::now();
315                let trailer = metadata_cache::load(cache.as_ref(), format!("{path}/{size}/framing"), |_| std::mem::size_of::<Trailer>(), async {
316                let header = read_part_range(reader, 0, metadata::HEADER_SIZE).await?;
317                metadata::validate_header(&header)?;
318                let bytes = read_part_range(reader, part.file_size - metadata::TRAILER_SIZE, part.file_size).await?;
319                let trailer = Trailer::decode(&bytes, part.file_size)?;
320                    Ok(trailer)
321                }).await?;
322                let payload_reader = PayloadReader {
323                    reader,
324                    cache: payload_cache.as_ref().map(|cache| cache.part(handle.entry().artifact.generation, part.part_id, size)),
325                    end: trailer.payload_end,
326                    key_end: trailer.key_end,
327                };
328                require(trailer.root.bounds == bounds, "sample root/part bounds mismatch")?;
329                let root = load_metadata_page(cache.as_ref(), reader, (&path, size), &trailer.root, trailer.depth, trailer.areas(), &manifest).await?;
330                metrics.add_elapsed("columnar_metadata_wall", metadata_started.elapsed());
331                let (mut active, children) = match root.as_ref() { Page::Leaf(_) => (Some(root.clone()), Vec::new()), Page::Internal(children) => (None, children.clone()) };
332                let mut children = children.into_iter(); let mut row_count = 0u64; let mut block_count = 0u64;
333                let mut payload_position = metadata::HEADER_SIZE;
334                // Exact keys live in the part's key area, outside every metadata page. Every
335                // block of one series references the same copy, so one load serves them all.
336                let keys_reservation = request.memory_pool.as_ref()
337                    .map(|pool| MemoryConsumer::new("ColumnarSamplesKeys").register(pool));
338                // `KeyRef` is the cache key; its fields are readable, so the map needs no
339                // ordering or hashing of the reference type itself.
340                let mut part_keys = HashMap::<(u64, u32), ExactSeriesKey>::new();
341                let mut key_offsets = mito_codec::row_converter::SparseOffsetsCache::new();
342                loop {
343                    let page = if let Some(page) = active.take() { page } else {
344                        let Some(child) = children.next() else { break; };
345                        if !matches(child.bounds, &request, ids.as_ref()) { adjacent = false; continue; }
346                        let started = Instant::now();
347                        let page = load_metadata_page(cache.as_ref(), reader, (&path, size), &child, 1, trailer.areas(), &manifest).await?;
348                        metrics.add_elapsed("columnar_metadata_wall", started.elapsed());
349                        page
350                    };
351                    let blocks = match page.as_ref() { Page::Leaf(blocks) => blocks, Page::Internal(_) => Err(unsupported("internal child in sample depth-two tree"))? };
352                    // One merged read for the keys of this page's selected blocks: a page can
353                    // hold hundreds of short series, and one read per series would multiply
354                    // this scan's I/O by the number of series it touches.
355                    load_page_keys(
356                        &payload_reader,
357                        trailer.areas(),
358                        blocks,
359                        &request,
360                        ids.as_ref(),
361                        &mut part_keys,
362                        keys_reservation.as_ref(),
363                        &mut key_offsets,
364                        &mut metrics,
365                    )
366                    .await?;
367                    let mut prefetched = PrefetchedBlocks::default();
368                    for (block_index, block) in blocks.iter().enumerate() {
369                        row_count = row_count.checked_add(u64::from(block.rows)).ok_or_else(|| invalid("samples row count overflow"))?;
370                        block_count += 1;
371                        let order = block.order(part.part_id);
372                        if let Some(prior) = &previous {
373                            if prior.part() == order.part() {
374                                // Reference positions are part-relative: only a block of the same
375                                // part may be compared with the previous block's reference.
376                                prior.precedes(&order, adjacent)?;
377                            } else {
378                                // Across a part boundary the references name unrelated key areas,
379                                // so only the ordered identifier applies here. Whether the same
380                                // series continues is decided on the exact keys below.
381                                require(prior.id() <= order.id(), "unordered sample series keys")?;
382                            }
383                        } else if complete { require(block.ordinal == 0, "initial sample ordinal is not zero")?; }
384                        for column in &block.columns {
385                            column.payload.validate(payload_position, trailer.payload_end, false)?;
386                            payload_position = column.payload.end()?;
387                        }
388                        // Ownership is exact, so it needs the block's key rather than its
389                        // coarse identifier. Pruned blocks never load one, and every key this
390                        // page selects was already fetched in one merged read.
391                        let key_id = (block.key.offset, block.key.length);
392                        let mut owner = false;
393                        if matches(block.bounds(), &request, ids.as_ref()) {
394                            if let std::collections::hash_map::Entry::Vacant(slot) =
395                                part_keys.entry(key_id)
396                            {
397                                // The page-level prefetch was refused by the memory pool, or
398                                // this series has no copy cached for the part yet. This path
399                                // reserves its own read before issuing it.
400                                let key = load_block_key(
401                                    &payload_reader,
402                                    trailer.areas(),
403                                    block,
404                                    keys_reservation.as_ref(),
405                                    &mut key_offsets,
406                                    &mut metrics,
407                                )
408                                .await?;
409                                slot.insert(key);
410                            }
411                            let key = part_keys
412                                .get(&key_id)
413                                .ok_or_else(|| invalid("missing sample key"))?;
414                            // A key that parses but claims another identifier is a malformed
415                            // reference: ownership would silently point at the wrong series.
416                            require(
417                                key.id() == block.id,
418                                "sample block series key disagrees with its identifier",
419                            )?;
420                            // Only the keys this stream loaded can be checked here: a reference
421                            // is a location, and selection prunes blocks without reading theirs,
422                            // so unvisited ranges are validated structurally only. The check is
423                            // chained across such gaps - it compares the last *loaded* key, not
424                            // the last visited block - so skipping a block cannot hide the keys
425                            // on either side of it. Loaded keys must be nondecreasing, and a
426                            // repeated key must continue its series instead of restarting it.
427                            // That is also the only way a series that spans a part boundary is
428                            // recognised: its two references name unrelated key areas.
429                            if let Some(prior) = &loaded_key {
430                                let prior_key = if prior.part() == order.part() {
431                                    let reference = prior.key();
432                                    part_keys.get(&(reference.offset, reference.length))
433                                } else {
434                                    spanning_key.as_ref()
435                                };
436                                if let Some(prior_key) = prior_key {
437                                    // The exact-next-block rules only hold when nothing was
438                                    // traversed in between; a gap still requires the order and
439                                    // the ordinal to keep increasing.
440                                    let consecutive =
441                                        adjacent && previous.as_ref() == Some(prior);
442                                    require(key >= prior_key, "unordered sample exact series keys")?;
443                                    if key == prior_key {
444                                        prior.continues(&order, consecutive)?;
445                                    } else if consecutive {
446                                        // A new series starts at ordinal zero, which the
447                                        // structural check above only covers inside one part.
448                                        require(block.ordinal == 0, "first sample block has nonzero ordinal")?;
449                                    }
450                                }
451                            }
452                            loaded_key = Some(order);
453                            owner = owns(key.bytes(), &request);
454                        }
455                        let batch = if owner {
456                            if block_index >= prefetched.end {
457                                // Release the previous window before reserving the next one.
458                                drop(std::mem::take(&mut prefetched));
459                                prefetched = prefetch_blocks(&payload_reader, blocks, block_index, &field_ids, &request, ids.as_ref(), &projection, trailer.payload_end, &mut metrics).await?;
460                            }
461                            *metrics.counts.entry("blocks").or_default() += 1;
462                            *metrics.counts.entry("decoded_rows").or_default() += u64::from(block.rows);
463                            read_block(&payload_reader, block, &manifest, &request, buffers_consumer.as_ref(), &projection, &field_ids, projected.clone(), &mut prefetched, &mut scratch, &mut decoder, &mut metrics).await?
464                        } else { None };
465                        // The block's payload slices are shared with the read-ahead window and
466                        // the part cache, so they are dropped before the yield: the block's
467                        // reservation is then the only charge alive with the batch.
468                        scratch.release();
469                        if let Some((batch, _reservation)) = batch {
470                            // The key is copied only now: `read_block` has just reserved the
471                            // bytes the batch retains, so the clone is paid for before it
472                            // happens instead of being allocated first and charged after.
473                            let key = part_keys
474                                .get(&key_id)
475                                .ok_or_else(|| invalid("missing sample key"))?
476                                .clone();
477                            *metrics.counts.entry("batches").or_default() += 1;
478                            *metrics.counts.entry("output_rows").or_default() += batch.num_rows() as u64;
479                            // The block's key is emitted once, not once per row.
480                            yield SamplesBatch { batch, key };
481                        }
482                        previous = Some(order); adjacent = true;
483                    }
484                }
485                // The next part's references name its own key area, so the last loaded block's
486                // exact key is the only thing that can prove a series continues there. It is
487                // kept, with its own pool charge, only while another part remains to be read. A
488                // part that loaded nothing leaves the carried key alone: it is still the last
489                // key this stream read.
490                if let Some(loaded) = loaded_key.filter(|loaded| loaded.part() == part.part_id) {
491                    let retain = ((part.part_id as usize) + 1 < manifest.files.len())
492                        .then(|| {
493                            let reference = loaded.key();
494                            part_keys.get(&(reference.offset, reference.length))
495                        })
496                        .flatten();
497                    retain_boundary_key(
498                        boundary_reservation.as_ref(),
499                        &mut spanning_bytes,
500                        &mut spanning_key,
501                        retain,
502                    )?;
503                }
504                if complete { require(row_count == part.num_rows && block_count == part.num_blocks, "sample part totals mismatch")?; }
505            }
506        }))
507    }

509/// Every block records the exact series key that owns its rows. A coarse identifier
510/// match is not ownership: a colliding partner shares the identifier.
511fn owns(key: &[u8], request: &SamplesReadRequest) -> bool {
512    request.keys.as_ref().is_none_or(|keys| keys.contains(key))
513}
514
515/// Retained bytes per cached key beyond the key bytes themselves: the boxed key and the
516/// allocator's rounding of the key's own allocation. The map entry is not charged, matching the
517/// per-block identity allowance, so a page of short keys still fits a small scan pool.
518const KEY_ENTRY_BYTES: usize = std::mem::size_of::<ExactSeriesKey>() + BUFFER_ALIGNMENT;
519
520/// Bytes of the checksum stored after each key in the key area.
521const KEY_CHECKSUM_BYTES: usize = metadata::KEY_CHECKSUM_BYTES as usize;
522
523/// Retained bytes of one cached key: its bytes, the boxed key, and the allocator's rounding.
524fn retained_key_bytes(length: u32) -> usize {
525    length as usize + KEY_ENTRY_BYTES
526}
527
528/// Replaces the exact key that carries one series across a part boundary.
529///
530/// Unlike the keys of a part's blocks, this copy outlives every per-part reservation, so it has a
531/// consumer of its own: a replacement is charged before it is made, and the previous charge goes
532/// back as soon as it is replaced or no longer needed.
533fn retain_boundary_key(
534    reservation: Option<&MemoryReservation>,
535    charged: &mut usize,
536    slot: &mut Option<ExactSeriesKey>,
537    key: Option<&ExactSeriesKey>,
538) -> Result<()> {
539    *slot = None;
540    if let Some(reservation) = reservation {
541        let _ = reservation.try_shrink(*charged);
542    }
543    *charged = 0;
544    if let Some(key) = key {
545        let bytes = retained_key_bytes(key.bytes().len() as u32);
546        reserve_keys(reservation, bytes)?;
547        *slot = Some(key.clone());
548        *charged = bytes;
549    }
550    Ok(())
551}

553/// Reserves `bytes` for a key fetch, or reports scan-memory exhaustion.
554///
555/// The reservation is taken before the read, so a pool that cannot hold the fetched key entry and
556/// the owned key it becomes fails without issuing I/O.
557fn reserve_keys(reservation: Option<&MemoryReservation>, bytes: usize) -> Result<()> {
558    if let Some(reservation) = reservation {
559        reservation
560            .try_grow(bytes)
561            .context(crate::error::ScanMemoryExhaustedSnafu)?;
562    }
563    Ok(())
564}

566/// Reserves `bytes` for a key fetch if the pool can pay for it.
567///
568/// A prefetch is optional: refusing it leaves the keys to [`load_block_key`], which is where an
569/// unpayable key reports exhaustion.
570fn try_reserve_keys(reservation: Option<&MemoryReservation>, bytes: usize) -> bool {
571    reservation.is_none_or(|reservation| reservation.try_grow(bytes).is_ok())
572}

574/// Releases the transient part of a key-fetch reservation.
575///
576/// The fetched bytes and the owned keys decoded from them coexist, so the pool is charged the
577/// peak up front. Once the buffers are dropped only the owned keys remain, and the difference
578/// goes back to the pool.
579struct KeyPeak<'a> {
580    reservation: Option<&'a MemoryReservation>,
581    transient: usize,
582}
583impl Drop for KeyPeak<'_> {
584    fn drop(&mut self) {
585        if let Some(reservation) = self.reservation
586            && self.transient > 0
587        {
588            let _ = reservation.try_shrink(self.transient);
589        }
590    }
591}

593/// Splits one stored key entry into its key bytes after verifying the stored checksum.
594///
595/// The reference's length covers the key only, so the entry is exactly one key followed by
596/// [`KEY_CHECKSUM_BYTES`]. A flipped label byte keeps the key's framing and identifier valid, so
597/// this is the only place that notices it before the bytes attribute rows to a series.
598fn verified_key(entry: &[u8], length: u32) -> Result<&[u8]> {
599    let length = length as usize;
600    require(
601        entry.len() == length + KEY_CHECKSUM_BYTES,
602        "sample key entry length mismatch",
603    )?;
604    let (key, stored) = entry.split_at(length);
605    let stored: [u8; KEY_CHECKSUM_BYTES] = stored
606        .try_into()
607        .map_err(|_| invalid("sample key checksum framing"))?;
608    require(
609        codec::checksum(key) == u64::from_le_bytes(stored),
610        "sample key checksum mismatch",
611    )?;
612    Ok(key)
613}

615/// Reads one block's exact key from the part's key area.
616///
617/// The page-level prefetch serves a whole page in one read, but it is charged up front. When the
618/// scan pool cannot take that, this path reads one key at a time, so a block that fits the pool
619/// stays readable. Either way the pool is charged the same coexisting peak before any I/O: the
620/// fetched entry, whatever alignment or assembly that fetch costs, and the owned key built from
621/// it. Reading first and refusing the key afterwards would leave the bytes already fetched.
622async fn load_block_key(
623    reader: &PayloadReader<'_>,
624    areas: Areas,
625    block: &Block,
626    reservation: Option<&MemoryReservation>,
627    offsets: &mut mito_codec::row_converter::SparseOffsetsCache,
628    metrics: &mut ColumnarReadMetrics,
629) -> Result<ExactSeriesKey> {
630    block.key.validate(areas.keys)?;
631    let retained = retained_key_bytes(block.key.length);
632    let range = block.key.offset..block.key.checksum_end()?;
633    let ranges = std::slice::from_ref(&range);
634    // The aligned cache turns one key into a whole aligned block plus whatever assembling it
635    // needs, so a warm part is asked for that peak first. A pool that cannot cover it gets an
636    // explicitly cache-less reader instead, whose peak is the key entry alone; only then does the
637    // read run.
638    let mut aligned = reader.cache.is_some();
639    let mut peak = fetch_peak(reader, ranges, areas.keys.end, aligned) + retained;
640    if !try_reserve_keys(reservation, peak) {
641        // Alignment is optional. Retrying on the smaller budget also reports exhaustion when even
642        // one key entry does not fit.
643        aligned = false;
644        peak = fetch_peak(reader, ranges, areas.keys.end, false) + retained;
645        reserve_keys(reservation, peak)?;
646    }
647    let _peak = KeyPeak {
648        reservation,
649        transient: peak - retained,
650    };
651    let uncached = PayloadReader {
652        reader: reader.reader,
653        cache: None,
654        end: reader.end,
655        key_end: reader.key_end,
656    };
657    let target = if aligned { reader } else { &uncached };
658    let mut fetched = fetch_ranges(target, areas.keys.end, ranges, metrics).await?;
659    let entry = fetched
660        .pop()
661        .ok_or_else(|| invalid("missing sample key response"))?;
662    let key = ExactSeriesKey::from_sparse_key(verified_key(&entry, block.key.length)?, offsets)?;
663    // A key that parses but claims another identifier is a malformed reference: ownership would
664    // silently point at the wrong series.
665    require(
666        key.id() == block.id,
667        "sample block series key disagrees with its identifier",
668    )?;
669    Ok(key)
670}

672/// Peak bytes [`fetch_ranges`] can hold at once while serving `ranges`.
673///
674/// The merge rule bounds one read, not a whole fetch: every merged range comes back as its own
675/// buffer and all of them are live together. With the aligned cache, the aligned copies and any
676/// range assembled from several of them coexist as well.
677fn fetch_peak(
678    reader: &PayloadReader<'_>,
679    ranges: &[std::ops::Range<u64>],
680    area_end: u64,
681    aligned: bool,
682) -> usize {
683    if !aligned {
684        return merge_read_ranges(ranges)
685            .iter()
686            .map(|range| (range.end - range.start) as usize)
687            .sum();
688    }
689    reader.reservation_bytes(ranges, area_end)
690        + ranges
691            .iter()
692            .map(|range| (range.end - range.start) as usize)
693            .sum::<usize>()
694}

696/// Fetches the exact keys of the blocks of one metadata page that survive coarse selection.
697///
698/// A descriptor only references its key, so identity never costs a copy per row or per block
699/// descriptor. Keys are read only for selected blocks and at most once per series, and the
700/// selected blocks of a page are merged into runs bounded by [`MAX_READ_BYTES`], so a page of
701/// many short series costs one read rather than one per series.
702///
703/// The pool is charged the coexisting peak before the read - the fetched or aligned and assembled
704/// bytes plus the owned keys decoded from them - and shrunk back to the retained keys afterwards.
705/// A pool that cannot pay the aligned peak is asked for the unbuffered one, then for a narrower
706/// window; the keys left out are read one at a time, which is where a key that does not fit at all
707/// reports exhaustion. Reading first and refusing the key afterwards would leave the bytes already
708/// fetched.
709#[allow(clippy::too_many_arguments)]
710async fn load_page_keys(
711    reader: &PayloadReader<'_>,
712    areas: Areas,
713    blocks: &[Block],
714    request: &SamplesReadRequest,
715    ids: Option<&BTreeSet<MetricSeriesId>>,
716    keys: &mut HashMap<(u64, u32), ExactSeriesKey>,
717    reservation: Option<&MemoryReservation>,
718    offsets: &mut mito_codec::row_converter::SparseOffsetsCache,
719    metrics: &mut ColumnarReadMetrics,
720) -> Result<()> {
721    // Selected blocks whose key is not cached yet, one per reference: blocks of one series
722    // share a copy, and blocks are ordered by series, so the references increase.
723    let mut wanted: Vec<&Block> = Vec::new();
724    for block in blocks {
725        if !matches(block.bounds(), request, ids)
726            || keys.contains_key(&(block.key.offset, block.key.length))
727        {
728            continue;
729        }
730        block.key.validate(areas.keys)?;
731        if wanted.last().is_some_and(|last| last.key == block.key) {
732            continue;
733        }
734        wanted.push(block);
735    }
736    if wanted.is_empty() {
737        return Ok(());
738    }
739    // The same range fetch the column chunks use: aligned blocks of the part are shared with the
740    // payload cache, so a part that is already cached costs no key read at all. Alignment is the
741    // first choice and the first victim: it costs a whole aligned block per key area touched.
742    let uncached = PayloadReader {
743        reader: reader.reader,
744        cache: None,
745        end: reader.end,
746        key_end: reader.key_end,
747    };
748    let mut window = wanted.len();
749    let mut attempt = None;
750    while window > 0 {
751        let selected = &wanted[..window];
752        let ranges = selected
753            .iter()
754            .map(|block| Ok(block.key.offset..block.key.checksum_end()?))
755            .collect::<Result<Vec<_>>>()?;
756        let retained = selected
757            .iter()
758            .map(|block| retained_key_bytes(block.key.length))
759            .sum::<usize>();
760        let mut candidates = Vec::with_capacity(2);
761        if reader.cache.is_some() {
762            candidates.push(true);
763        }
764        candidates.push(false);
765        for aligned in candidates {
766            // The fetched entries and the owned keys they become are live together; the fetch
767            // peak already covers the entries themselves.
768            let peak = fetch_peak(reader, &ranges, areas.keys.end, aligned) + retained;
769            if try_reserve_keys(reservation, peak) {
770                attempt = Some((aligned, peak, retained, ranges));
771                break;
772            }
773        }
774        if attempt.is_some() {
775            break;
776        }
777        // Halve the window rather than give up: the keys left out keep their own, smaller,
778        // on-demand reservation.
779        window /= 2;
780    }
781    let Some((aligned, peak, retained, ranges)) = attempt else {
782        return Ok(());
783    };
784    let _peak = KeyPeak {
785        reservation,
786        transient: peak - retained,
787    };
788    let target = if aligned { reader } else { &uncached };
789    let fetched = fetch_ranges(target, areas.keys.end, &ranges, metrics).await?;
790    for (block, entry) in wanted.iter().take(window).zip(fetched) {
791        let key =
792            ExactSeriesKey::from_sparse_key(verified_key(&entry, block.key.length)?, offsets)?;
793        // A key that parses but claims another identifier is a malformed reference: ownership
794        // would silently point at the wrong series.
795        require(
796            key.id() == block.id,
797            "sample block series key disagrees with its identifier",
798        )?;
799        keys.insert((block.key.offset, block.key.length), key);
800    }
801    Ok(())
802}

804/// Bounds-only pruning. Selection itself is exact and decided by [`owns`].
805fn matches(
806    bounds: Bounds,
807    request: &SamplesReadRequest,
808    ids: Option<&BTreeSet<MetricSeriesId>>,
809) -> bool {
810    if request.start.is_some_and(|start| bounds.max < start)
811        || request.end.is_some_and(|end| bounds.min >= end)
812    {
813        return false;
814    }
815    let mut first = bounds.first;
816    let mut last = bounds.last;
817    if let Some(table_id) = request.table_id {
818        first = first.max(MetricSeriesId { table_id, tsid: 0 });
819        last = last.min(MetricSeriesId {
820            table_id,
821            tsid: u64::MAX,
822        });
823    }
824    first <= last && ids.is_none_or(|ids| ids.range(first..=last).next().is_some())
825}
826async fn read_range(store: &ObjectStore, path: &str, start: u64, end: u64) -> Result<bytes::Bytes> {
827    require(start <= end, "reversed sample byte range")?;
828    let mut buffers = fetch_byte_ranges(path, store.clone(), std::slice::from_ref(&(start..end)))
829        .await
830        .context(OpenDalSnafu)?;
831    let bytes = buffers
832        .pop()
833        .ok_or_else(|| invalid("missing sample range response"))?;
834    require(
835        buffers.is_empty() && bytes.len() as u64 == end - start,
836        "sample range length mismatch",
837    )?;
838    Ok(bytes)
839}

872/// Keep column loads separate from metadata for future Mito cache integration.
873fn load_column_chunk<'a>(
874    buffers: &'a HashMap<u64, bytes::Bytes>,
875    column: &Column,
876    decoder: &'a mut codec::PayloadDecoder,
877) -> Result<&'a [u8]> {
878    let allowed = match column.role {
879        1 => matches!(column.payload.encoding, 1 | 2),
880        2 => matches!(column.payload.encoding, 3 | 4),
881        3 => matches!(column.payload.encoding, 6 | 7),
882        4 => column.payload.encoding == 5,
883        _ => false,
884    };
885    if !allowed {
886        return Err(unsupported("samples column encoding"));
887    }
888    let bytes = buffers
889        .get(&column.payload.offset)
890        .ok_or_else(|| invalid("missing prefetched sample column"))?;
891    column.payload.unpack_with_decoder(bytes, decoder)
892}
893/// Fixed per-column charge `arrow` reports beyond the column's own data.
894///
895/// `Array::get_array_memory_size` is `size_of::<Self>()` plus the buffer capacities, and
896/// `RecordBatch::get_array_memory_size` sums those with no other term. `BinaryArray` is the
897/// widest array type this reader can build, so its struct size bounds every projected column.
898const ARRAY_HEADER_BYTES: usize = std::mem::size_of::<BinaryArray>();
899/// Granularity `arrow_buffer` rounds every buffer allocation up to, so each column can retain up
900/// to this much more than the bytes it holds.
901const BUFFER_ALIGNMENT: usize = 64;
902
903#[cfg(test)]
904std::thread_local! {
905    /// Scratch buffers this thread has built. A stream must build one and empty it between
906    /// blocks, which nothing outside that stream reveals, so tests count the allocations
907    /// directly. Thread-local for the same reason as the codec's context counter: a
908    /// current-thread runtime polls a stream on one thread.
909    static SCRATCHES_BUILT: std::cell::Cell<usize> = const { std::cell::Cell::new(0) };
910}
911
912/// Test-only: scratch buffers this thread has built so far.
913#[cfg(test)]
914pub(crate) fn scratches_built() -> usize {
915    SCRATCHES_BUILT.with(std::cell::Cell::get)
916}

918/// Per-stream scratch for decoding one block.
919///
920/// A block needs a handful of small containers. They keep their allocations across blocks, so a
921/// stream builds them once instead of once per block; [`ReadScratch::release`] empties them
922/// without giving the allocations back.
923#[derive(Default)]
924struct ReadScratch {
925    /// Stored ranges of the columns this block decodes.
926    ranges: Vec<std::ops::Range<u64>>,
927    /// [`Self::ranges`] after bounded gap coalescing.
928    merged: Vec<std::ops::Range<u64>>,
929    /// This block's payload slices, keyed by their stored offset.
930    buffers: HashMap<u64, bytes::Bytes>,
931    /// Row indices this block keeps.
932    selected: Vec<usize>,
933    /// Projection indices in physical order.
934    sorted: Vec<usize>,
935    /// Projected arrays waiting for the batch's column order.
936    arrays: BTreeMap<usize, ArrayRef>,
937}
938
939impl ReadScratch {
940    /// Builds the one scratch a stream empties and reuses for every block it decodes. Reuse
941    /// leaves no other trace outside the stream, so a test counts the constructions.
942    fn for_stream() -> Self {
943        #[cfg(test)]
944        SCRATCHES_BUILT.with(|built| built.set(built.get() + 1));
945        Self::default()
946    }
947
948    /// Empties every container and keeps the allocations for the next block.
949    ///
950    /// The payload slices are handed out before the batch is yielded, so they are dropped here
951    /// rather than carried across the yield.
952    fn release(&mut self) {
953        self.ranges.clear();
954        self.merged.clear();
955        self.buffers.clear();
956        self.selected.clear();
957        self.sorted.clear();
958        self.arrays.clear();
959    }
960}

966        TimestampUnit::Microsecond => Arc::new(TimestampMicrosecondArray::from_iter_values(values)),
967        TimestampUnit::Nanosecond => Arc::new(TimestampNanosecondArray::from_iter_values(values)),

970#[allow(clippy::too_many_arguments)]
971async fn read_block(
972    reader: &PayloadReader<'_>,
973    block: &Block,
974    manifest: &Manifest,
975    request: &SamplesReadRequest,
976    buffers_consumer: Option<&MemoryConsumer>,
977    projection: &[usize],
978    field_ids: &HashSet<u32>,
979    schema: SchemaRef,
980    prefetched: &mut PrefetchedBlocks,
981    scratch: &mut ReadScratch,
982    decoder: &mut codec::PayloadDecoder,
983    metrics: &mut ColumnarReadMetrics,
984) -> Result<Option<(RecordBatch, Option<MemoryReservation>)>> {
985    let boundary = request.start.is_some_and(|start| start > block.min)
986        || request.end.is_some_and(|end| end <= block.max);
987    let needs_time = boundary || projection.contains(&2);
988    let columns: Vec<_> = block
989        .columns
990        .iter()
991        .filter(|column| match column.role {
992            1 => needs_time,
993            2 => projection.contains(&3),
994            3 => projection.contains(&4),
995            4 => field_ids.contains(&column.id),
996            _ => false,
997        })
998        .collect();
999    let reservation = buffers_consumer
1000        .zip(request.memory_pool.as_ref())
1001        .map(|(consumer, pool)| consumer.clone_with_new_id().register(pool));
1002    // Stored buffers, decompression, typed values, selection, and Arrow output coexist.
1003    // Reserve before any payload I/O/allocation. The reservation survives the yield.
1004    scratch.ranges.clear();
1005    for column in &columns {
1006        scratch
1007            .ranges
1008            .push(column.payload.offset..column.payload.end()?);
1009    }
1010    merge_read_ranges_into(&scratch.ranges, &mut scratch.merged);
1011    let gap_bytes = scratch.merged.iter().map(|r| r.end - r.start).sum::<u64>()
1012        - scratch.ranges.iter().map(|r| r.end - r.start).sum::<u64>();
1013    // The block's exact key is returned with the batch, once per block rather than once per
1014    // row, so it costs one key regardless of how many rows the block holds. Time pruning
1015    // cannot add output rows, so no per-row term is needed for identity at all.
1016    let key_bytes =
1017        std::mem::size_of::<ExactSeriesKey>() + block.key.length as usize + BUFFER_ALIGNMENT;
1018    let base_bytes = 2 * gap_bytes as usize
1019        + columns
1020            .iter()
1021            .map(|column| column.payload.stored as usize + column.payload.decoded as usize)
1022            .sum::<usize>()
1023        + block.rows as usize * (32 * (projection.len() + 1) + std::mem::size_of::<usize>())
1024        + key_bytes
1025        // Each projected column also retains one array header and one allocation rounded up to
1026        // the allocator's granularity, which the per-row bytes above do not cover. A block with
1027        // one row and an empty projection selects no payload column at all, so this is the only
1028        // term that keeps its array accounted for.
1029        + projection.len() * (ARRAY_HEADER_BYTES + BUFFER_ALIGNMENT)
1030        + columns.len() * 128;
1031    let bytes = base_bytes + reader.reservation_bytes(&scratch.ranges, reader.end);
1032    let mut bypass_cache = false;
1033    if let Some(reservation) = &reservation {
1034        let mut result = reservation.try_grow(bytes);
1035        if result.is_err() && !prefetched.buffers.is_empty() {
1036            // Read-ahead must not prevent a block that fits on its own from being read.
1037            // Drop every shared slice and its reservation before retrying, and avoid
1038            // attempting the same window again on the next block.
1039            *prefetched = PrefetchedBlocks {
1040                end: prefetched.end,
1041                ..Default::default()
1042            };
1043            result = reservation.try_grow(bytes);
1044        }
1045        if result.is_err() && reader.cache.is_some() {
1046            // Alignment is optional. A block that fits without it must remain readable,
1047            // including on cache hits, whose shared backing buffers still need accounting.
1048            result = reservation.try_grow(base_bytes);
1049            bypass_cache = result.is_ok();
1050        }
1051        result.context(crate::error::ScanMemoryExhaustedSnafu)?;
1052    }
1053    let uncached = PayloadReader {
1054        reader: reader.reader,
1055        cache: None,
1056        end: reader.end,
1057        key_end: reader.key_end,
1058    };
1059    let reader = if bypass_cache { &uncached } else { reader };
1060    // The map keeps its allocation; the previous block's slices were already released.
1061    scratch.buffers.clear();
1062    for column in &columns {
1063        if let Some(bytes) = prefetched.buffers.get(&column.payload.offset) {
1064            scratch.buffers.insert(column.payload.offset, bytes.clone());
1065        }
1066    }
1067    if boundary {
1068        fetch_columns(reader, &[&block.columns[0]], &mut scratch.buffers, metrics).await?;
1069    } else {
1070        fetch_columns(reader, &columns, &mut scratch.buffers, metrics).await?;
1071    }
1072    let decode_started = Instant::now();
1073    scratch.selected.clear();
1074    scratch.selected.reserve(block.rows as usize);
1075    scratch.selected.extend(0..block.rows as usize);
1076    let mut times = None;
1077    if needs_time {
1078        let column = &block.columns[0];
1079        let values = codec::decode_numbers(
1080            load_column_chunk(&scratch.buffers, column, decoder)?,
1081            column.payload.encoding,
1082            block.rows as usize,
1083        )?;
1084        require(
1085            values.first().map(|v| *v as i64) == Some(block.min)
1086                && values.last().map(|v| *v as i64) == Some(block.max),
1087            "sample timestamp bounds mismatch",
1088        )?;
1089        scratch.selected.retain(|&i| {
1090            request
1091                .start
1092                .is_none_or(|start| (values[i] as i64) >= start)
1093                && request.end.is_none_or(|end| (values[i] as i64) < end)
1094        });
1095        if scratch.selected.is_empty() {
1096            metrics.add_elapsed("columnar_decode", decode_started.elapsed());
1097            return Ok(None);
1098        }
1099        times = Some(values);
1100    }
1101    metrics.add_elapsed("columnar_decode", decode_started.elapsed());
1102    if boundary {
1103        // Release the decoder's context and oversized scratch before waiting for the remaining
1104        // payloads, and keep the decoder itself: the stream owns one, it is not rebuilt.
1105        decoder.release();
1106        let remaining = columns
1107            .into_iter()
1108            .filter(|column| column.role != 1)
1109            .collect::<Vec<_>>();
1110        fetch_columns(reader, &remaining, &mut scratch.buffers, metrics).await?;
1111    }
1112    let decode_started = Instant::now();
1113    scratch.arrays.clear();
1114    // Visit columns in physical order, regardless of requested output order.
1115    scratch.sorted.clear();
1116    scratch.sorted.extend_from_slice(projection);
1117    scratch.sorted.sort_unstable();
1118    // The exact key column is never projected: it is returned with the batch instead,
1119    // so a block of any row count carries one key in total.
1120    for &index in scratch.sorted.iter() {
1121        let selected = &scratch.selected;
1122        let array: ArrayRef = if index == 0 {
1123            Arc::new(UInt32Array::from(vec![block.id.table_id; selected.len()]))
1124        } else if index == 1 {
1125            Arc::new(UInt64Array::from(vec![block.id.tsid; selected.len()]))
1126        } else {
1127            match index {
1128                2 => {
1129                    let values = times
1130                        .as_ref()
1131                        .ok_or_else(|| invalid("missing decoded sample time"))?;
1132                    timestamp_array(
1133                        selected.iter().map(|&i| values[i] as i64),
1134                        manifest.timestamp.unit,
1135                    )
1136                }
1137                3 => {
1138                    let column = &block.columns[1];
1139                    let values = codec::decode_numbers(
1140                        load_column_chunk(&scratch.buffers, column, decoder)?,
1141                        column.payload.encoding,
1142                        block.rows as usize,
1143                    )?;
1144                    if let Some(times) = &times {
1145                        require(
1146                            (1..values.len())
1147                                .all(|i| times[i - 1] != times[i] || values[i - 1] >= values[i]),
1148                            "unordered sample sequences",
1149                        )?;
1150                    }
1151                    Arc::new(UInt64Array::from_iter_values(
1152                        selected.iter().map(|&i| values[i]),
1153                    ))
1154                }
1155                4 => {
1156                    let column = &block.columns[2];
1157                    let values = codec::decode_operations(
1158                        load_column_chunk(&scratch.buffers, column, decoder)?,
1159                        column.payload.encoding,
1160                        block.rows as usize,
1161                    )?;
1162                    Arc::new(UInt8Array::from_iter_values(
1163                        selected.iter().map(|&i| values[i]),
1164                    ))
1165                }
1166                field => {
1167                    let field = manifest
1168                        .fields
1169                        .get(field - 5)
1170                        .ok_or_else(|| invalid("unknown sample field"))?;
1171                    let id = field.column_id;
1172                    if let Some(column) = block.columns.iter().find(|c| c.role == 4 && c.id == id) {
1173                        let values = codec::decode_float(
1174                            load_column_chunk(&scratch.buffers, column, decoder)?,
1175                            block.rows as usize,
1176                            column.nulls,
1177                        )?;
1178                        Arc::new(Float64Array::from_iter(
1179                            selected.iter().map(|&i| values[i].map(f64::from_bits)),
1180                        ))
1181                    } else {
1182                        Arc::new(Float64Array::from(vec![None::<f64>; selected.len()]))
1183                    }
1184                }
1185            }
1186        };
1187        scratch.arrays.insert(index, array);
1188    }
1189    let output = projection
1190        .iter()
1191        .map(|i| {
1192            scratch
1193                .arrays
1194                .remove(i)
1195                .ok_or_else(|| invalid("missing sample projected column"))
1196        })
1197        .collect::<Result<Vec<_>>>()?;
1198    let batch = RecordBatch::try_new_with_options(
1199        schema,
1200        output,
1201        &RecordBatchOptions::new().with_row_count(Some(scratch.selected.len())),
1202    )
1203    .context(NewRecordBatchSnafu)?;
1204    metrics.add_elapsed("columnar_decode", decode_started.elapsed());
1205    Ok(Some((batch, reservation)))
1206}

1222/// Fetch selected ranges with bounded gap coalescing and shared-buffer slices.
1223/// The read window and scan reservation bound retained buffers;
1224/// OpenDAL bounds concurrency to 8.
1225async fn fetch_columns(
1226    reader: &PayloadReader<'_>,
1227    columns: &[&Column],
1228    buffers: &mut HashMap<u64, bytes::Bytes>,
1229    metrics: &mut ColumnarReadMetrics,
1230) -> Result<()> {
1231    let columns: Vec<_> = columns
1232        .iter()
1233        .copied()
1234        .filter(|c| !buffers.contains_key(&c.payload.offset))
1235        .collect();
1236    if columns.is_empty() {
1237        return Ok(());
1238    }
1239    let ranges = columns
1240        .iter()
1241        .map(|column| Ok(column.payload.offset..column.payload.end()?))
1242        .collect::<Result<Vec<_>>>()?;
1243    let fetched = fetch_ranges(reader, reader.end, &ranges, metrics).await?;
1244    for (column, bytes) in columns.iter().zip(fetched) {
1245        buffers.insert(column.payload.offset, bytes);
1246    }
1247    Ok(())
1248}

1250/// Fetch arbitrary byte ranges of one part, reusing its aligned block cache when it has one.
1251///
1252/// `end` is where the ranges' area ends (the payload area or the key area), so an aligned block
1253/// never extends past the area it belongs to. Keys use the same path as column chunks: a part
1254/// whose blocks are already cached issues no read at all, which is why a second series in the
1255/// same part can be served from the over-read that the first one paid for.
1256async fn fetch_ranges(
1257    reader: &PayloadReader<'_>,
1258    end: u64,
1259    ranges: &[std::ops::Range<u64>],
1260    metrics: &mut ColumnarReadMetrics,
1261) -> Result<Vec<bytes::Bytes>> {
1262    *metrics.counts.entry("column_ranges").or_default() += ranges.len() as u64;
1263    *metrics.bytes.entry("requested").or_default() +=
1264        ranges.iter().map(|r| r.end - r.start).sum::<u64>();
1265    if ranges.is_empty() {
1266        return Ok(Vec::new());
1267    }
1268    let Some(cache) = &reader.cache else {
1269        // Merging is bounded by [`MAX_READ_BYTES`], and a requested range never exceeds that, so
1270        // every range lies inside exactly one merged read.
1271        let merged = merge_read_ranges(ranges);
1272        *metrics.counts.entry("payload_reads").or_default() += merged.len() as u64;
1273        let started = Instant::now();
1274        let fetched = reader
1275            .reader
1276            .fetch(merged.clone())
1277            .await
1278            .context(OpenDalSnafu)?;
1279        metrics.add_elapsed("columnar_payload_fetch_wall", started.elapsed());
1280        require(
1281            fetched.len() == merged.len(),
1282            "sample fetch response count mismatch",
1283        )?;
1284        let mut result = Vec::with_capacity(ranges.len());
1285        let mut index = 0;
1286        for (range, buffer) in merged.iter().zip(fetched) {
1287            let bytes = buffer.to_bytes();
1288            require(
1289                bytes.len() as u64 == range.end - range.start,
1290                "sample range length mismatch",
1291            )?;
1292            *metrics.bytes.entry("fetched").or_default() += bytes.len() as u64;
1293            while let Some(requested) = ranges.get(index) {
1294                if requested.start >= range.end {
1295                    break;
1296                }
1297                require(
1298                    requested.start >= range.start && requested.end <= range.end,
1299                    "sample range outside fetched range",
1300                )?;
1301                result.push(bytes.slice(
1302                    (requested.start - range.start) as usize
1303                        ..(requested.end - range.start) as usize,
1304                ));
1305                index += 1;
1306            }
1307        }
1308        require(index == ranges.len(), "missing sample range response")?;
1309        return Ok(result);
1310    };
1311    let aligned = aligned_ranges(ranges, end);
1312    let mut blocks: HashMap<u64, bytes::Bytes> = HashMap::new();
1313    let mut missing = Vec::new();
1314    for range in &aligned {
1315        if let Some(bytes) = cache.get(range) {
1316            *metrics.bytes.entry("cache_hit").or_default() += bytes.len() as u64;
1317            blocks.insert(range.start, bytes);
1318        } else {
1319            missing.push(range.clone());
1320        }
1321    }
1322    let merged = merge_read_ranges(&missing);
1323    if !merged.is_empty() {
1324        *metrics.counts.entry("payload_reads").or_default() += merged.len() as u64;
1325        let started = Instant::now();
1326        let fetched = reader
1327            .reader
1328            .fetch(merged.clone())
1329            .await
1330            .context(OpenDalSnafu)?;
1331        metrics.add_elapsed("columnar_payload_fetch_wall", started.elapsed());
1332        require(
1333            fetched.len() == merged.len(),
1334            "sample fetch response count mismatch",
1335        )?;
1336        let mut missing = missing.iter().peekable();
1337        for (range, buffer) in merged.iter().zip(fetched) {
1338            let bytes = buffer.to_bytes();
1339            require(
1340                bytes.len() as u64 == range.end - range.start,
1341                "sample range length mismatch",
1342            )?;
1343            *metrics.bytes.entry("fetched").or_default() += bytes.len() as u64;
1344            while let Some(block) = missing.peek() {
1345                if block.start >= range.end {
1346                    break;
1347                }
1348                require(
1349                    block.start >= range.start && block.end <= range.end,
1350                    "aligned block outside fetched range",
1351                )?;
1352                let data = bytes.slice(
1353                    (block.start - range.start) as usize..(block.end - range.start) as usize,
1354                );
1355                cache.insert(block, &data);
1356                blocks.insert(block.start, data);
1357                missing.next();
1358            }
1359        }
1360        require(missing.next().is_none(), "missing aligned block response")?;
1361    }
1362    let mut result = Vec::with_capacity(ranges.len());
1363    for range in ranges {
1364        let start = range.start / PAYLOAD_BLOCK_BYTES * PAYLOAD_BLOCK_BYTES;
1365        let first = blocks
1366            .get(&start)
1367            .ok_or_else(|| invalid("missing cached payload block"))?;
1368        let offset = (range.start - start) as usize;
1369        let len = (range.end - range.start) as usize;
1370        let bytes = if offset + len <= first.len() {
1371            first.slice(offset..offset + len)
1372        } else {
1373            let mut assembled = bytes::BytesMut::with_capacity(len);
1374            let mut position = range.start;
1375            while position < range.end {
1376                let start = position / PAYLOAD_BLOCK_BYTES * PAYLOAD_BLOCK_BYTES;
1377                let block = blocks
1378                    .get(&start)
1379                    .ok_or_else(|| invalid("missing cached payload block"))?;
1380                let offset = (position - start) as usize;
1381                let take = ((range.end - position) as usize).min(block.len() - offset);
1382                require(take > 0, "empty cached payload block")?;
1383                assembled.extend_from_slice(&block[offset..offset + take]);
1384                position += take as u64;
1385            }
1386            assembled.freeze()
1387        };
1388        result.push(bytes);
1389    }
1390    Ok(result)
1391}

1393/// Stable cache units; adjacent missing units still use bounded merged reads.
1394const PAYLOAD_BLOCK_BYTES: u64 = 16 * 1024;
1395
1396struct PayloadReader<'a> {
1397    reader: &'a Reader,
1398    cache: Option<PayloadPartCache>,
1399    /// End of the column payload area; aligned payload blocks are clipped here.
1400    end: u64,
1401    /// End of the key area; aligned key blocks are clipped here.
1402    key_end: u64,
1403}
1404
1405impl PayloadReader<'_> {
1406    /// Aligned and assembled bytes that serving `ranges` inside `area_end` can hold at once.
1407    fn reservation_bytes(&self, ranges: &[std::ops::Range<u64>], area_end: u64) -> usize {
1408        if self.cache.is_none() {
1409            return 0;
1410        }
1411        // Aligned read allocations and assembly can coexist. Cache-owned copies
1412        // are separately charged to the shared page-cache budget.
1413        2 * aligned_ranges(ranges, area_end)
1414            .iter()
1415            .map(|r| (r.end - r.start) as usize)
1416            .sum::<usize>()
1417    }
1418}
1419
1420/// Aligned cache units covering `ranges`, clipped at `end`.
1421fn aligned_ranges(ranges: &[std::ops::Range<u64>], end: u64) -> Vec<std::ops::Range<u64>> {
1422    let mut offsets = std::collections::BTreeSet::new();
1423    for range in ranges {
1424        if range.start >= range.end {
1425            continue;
1426        }
1427        for block in range.start / PAYLOAD_BLOCK_BYTES..=(range.end - 1) / PAYLOAD_BLOCK_BYTES {
1428            offsets.insert(block * PAYLOAD_BLOCK_BYTES);
1429        }
1430    }
1431    offsets
1432        .into_iter()
1433        .map(|start| start..start.saturating_add(PAYLOAD_BLOCK_BYTES).min(end))
1434        .collect()
1435}

1437const MAX_READ_GAP: u64 = 4 * 1024;
1438const MAX_READ_BYTES: u64 = 256 * 1024;
1439
1440/// Allows bounded over-read to amortize filesystem and scheduling costs.
1441fn can_merge_ranges(previous: &std::ops::Range<u64>, next: &std::ops::Range<u64>) -> bool {
1442    next.start
1443        .checked_sub(previous.end)
1444        .is_some_and(|gap| gap <= MAX_READ_GAP)
1445        && next
1446            .end
1447            .checked_sub(previous.start)
1448            .is_some_and(|bytes| bytes <= MAX_READ_BYTES)
1449}
1450
1451fn merge_read_ranges(ranges: &[std::ops::Range<u64>]) -> Vec<std::ops::Range<u64>> {
1452    let mut merged = Vec::new();
1453    merge_read_ranges_into(ranges, &mut merged);
1454    merged
1455}
1456
1457/// [`merge_read_ranges`] into a caller-owned buffer, so a stream coalesces into one allocation.
1458fn merge_read_ranges_into(ranges: &[std::ops::Range<u64>], merged: &mut Vec<std::ops::Range<u64>>) {
1459    merged.clear();
1460    for range in ranges {
1461        if let Some(previous) = merged.last_mut()
1462            && can_merge_ranges(previous, range)
1463        {
1464            previous.end = range.end;
1465        } else {
1466            merged.push(range.clone());
1467        }
1468    }
1469}

1471/// A bounded read-ahead window. Byte slices retain the shared fetched allocation;
1472/// the reservation therefore lives until the whole window is replaced or cancelled.
1473#[derive(Default)]
1474struct PrefetchedBlocks {
1475    end: usize,
1476    buffers: HashMap<u64, bytes::Bytes>,
1477    _reservation: Option<MemoryReservation>,
1478}
1479
1480#[allow(clippy::too_many_arguments)]
1481async fn prefetch_blocks(
1482    reader: &PayloadReader<'_>,
1483    blocks: &[Block],
1484    start: usize,
1485    field_ids: &HashSet<u32>,
1486    request: &SamplesReadRequest,
1487    ids: Option<&BTreeSet<MetricSeriesId>>,
1488    projection: &[usize],
1489    payload_end: u64,
1490    metrics: &mut ColumnarReadMetrics,
1491) -> Result<PrefetchedBlocks> {
1492    const MAX_BLOCKS: usize = 128;
1493    let mut columns = Vec::new();
1494    let mut bytes = 0usize;
1495    let mut last_range: Option<std::ops::Range<u64>> = None;
1496    let mut payload_ranges = Vec::new();
1497    let mut end = start;
1498    for block in blocks.iter().skip(start).take(MAX_BLOCKS) {
1499        if !matches(block.bounds(), request, ids) {
1500            end += 1;
1501            continue;
1502        }
1503        let boundary = request.start.is_some_and(|t| t > block.min)
1504            || request.end.is_some_and(|t| t <= block.max);
1505        // Boundary blocks request timestamps first. Coalescing may include value
1506        // bytes in gaps, but values are only decoded after timestamp pruning.
1507        let selected: Vec<_> = block
1508            .columns
1509            .iter()
1510            .filter(|column| {
1511                if boundary {
1512                    return column.role == 1;
1513                }
1514                match column.role {
1515                    1 => projection.contains(&2),
1516                    2 => projection.contains(&3),
1517                    3 => projection.contains(&4),
1518                    4 => field_ids.contains(&column.id),
1519                    _ => false,
1520                }
1521            })
1522            .collect();
1523        let mut next_bytes = bytes;
1524        let mut next_range = last_range.clone();
1525        for column in &selected {
1526            column
1527                .payload
1528                .validate(metadata::HEADER_SIZE, payload_end, false)?;
1529            let range = column.payload.offset..column.payload.end()?;
1530            if let Some(previous) = next_range.as_mut()
1531                && can_merge_ranges(previous, &range)
1532            {
1533                next_bytes += (range.end - previous.end) as usize;
1534                previous.end = range.end;
1535            } else {
1536                next_bytes += column.payload.stored as usize;
1537                next_range = Some(range);
1538            }
1539        }
1540        let previous_len = payload_ranges.len();
1541        for column in &selected {
1542            payload_ranges.push(column.payload.offset..column.payload.end()?);
1543        }
1544        let fetch_bytes = if reader.cache.is_some() {
1545            aligned_ranges(&payload_ranges, reader.end)
1546                .iter()
1547                .map(|range| (range.end - range.start) as usize)
1548                .sum()
1549        } else {
1550            next_bytes
1551        };
1552        if fetch_bytes > MAX_READ_BYTES as usize {
1553            payload_ranges.truncate(previous_len);
1554            break;
1555        }
1556        bytes = next_bytes;
1557        last_range = next_range;
1558        columns.extend(selected);
1559        end += 1;
1560    }
1561    // A single oversized block uses the existing per-block reservation and read path.
1562    let end = end.max(start + 1);
1563    let reservation = request
1564        .memory_pool
1565        .as_ref()
1566        .map(|pool| MemoryConsumer::new("ColumnarSamplesPrefetch").register(pool));
1567    // Include aligned payload buffers in the optional read-ahead reservation.
1568    if let Some(reservation) = &reservation
1569        && reservation
1570            .try_grow(
1571                2 * bytes
1572                    + columns.len() * 256
1573                    + reader.reservation_bytes(
1574                        &columns
1575                            .iter()
1576                            .map(|c| Ok(c.payload.offset..c.payload.end()?))
1577                            .collect::<Result<Vec<_>>>()?,
1578                        reader.end,
1579                    ),
1580            )
1581            .is_err()
1582    {
1583        return Ok(PrefetchedBlocks {
1584            end,
1585            ..Default::default()
1586        });
1587    }
1588    let mut buffers = HashMap::new();
1589    fetch_columns(reader, &columns, &mut buffers, metrics).await?;
1590    Ok(PrefetchedBlocks {
1591        end,
1592        buffers,
1593        _reservation: reservation,
1594    })
1595}
