52/// Coalesces encoded columns across blocks before handing them to OpenDAL.
53#[derive(Default)]
54struct SampleBuffer {
55    bytes: Vec<u8>,
56}
57
58impl SampleBuffer {
59    async fn write(&mut self, outputs: &mut Outputs, mut bytes: &[u8]) -> Result<()> {
60        let capacity = DEFAULT_WRITE_BUFFER_SIZE.as_bytes() as usize;
61        while !bytes.is_empty() {
62            if self.bytes.capacity() == 0 {
63                self.bytes.reserve_exact(capacity);
64            }
65            let count = bytes.len().min(capacity - self.bytes.len());
66            self.bytes.extend_from_slice(&bytes[..count]);
67            bytes = &bytes[count..];
68            if self.bytes.len() == capacity {
69                self.flush(outputs).await?;
70            }
71        }
72        Ok(())
73    }
74
75    async fn flush(&mut self, outputs: &mut Outputs) -> Result<()> {
76        if !self.bytes.is_empty() {
77            outputs.write(std::mem::take(&mut self.bytes)).await?;
78        }
79        Ok(())
80    }
81}

89/// One field's values for the rows of the pending block, in row order.
90enum PendingField {
91    Float(Vec<Option<u64>>),
92    /// Runs of rows borrowed from input batches, concatenated once when the block is encoded.
93    Histogram {
94        runs: Vec<(ArrayRef, usize, usize)>,
95        bound: ChunkBound,
96    },
97}
98impl PendingField {
99    fn new(field: &ValueColumn) -> Self {
100        if field.is_histogram() {
101            Self::Histogram {
102                runs: Vec::new(),
103                bound: ChunkBound::default(),
104            }
105        } else {
106            Self::Float(Vec::new())
107        }
108    }
109}
110/// One input batch's column for a field.
111enum FieldInput<'a> {
112    Float(&'a Float64Array),
113    Histogram(&'a ArrayRef, &'a StructArray),
114}
115/// A block descriptor whose exact key has not been placed in the part's key area yet.
116///
117/// The key bytes are held here until the part ends, when they are written to their own area
118/// and replaced by a fixed-size [`KeyRef`]. That is why no descriptor — and therefore no
119/// metadata page — ever has to fit a key.
120struct PendingBlock {
121    key: ExactSeriesKey,
122    min: i64,
123    max: i64,
124    ordinal: u64,
125    rows: u32,
126    columns: Vec<Column>,
127}
128
129impl PendingBlock {
130    /// Descriptor bytes once the key reference replaces the key bytes.
131    fn size(&self) -> usize {
132        metadata::descriptor_bytes(self.columns.len())
133    }
134}
135
136/// Places one copy of each series key in the part's key area and turns pending blocks into
137/// descriptors that reference it.
138///
139/// Blocks are ordered by exact key, so every block of one series is adjacent and shares one
140/// copy of the key. Consecutive equal keys therefore reuse the previous reference, while every
141/// new series appends its bytes after the previous copy: references are ordered, disjoint, and
142/// unique per series, which is what lets a reader treat an equal reference as the same series
143/// without reading the bytes. Each copy is followed by its XXH64, which the reference does not
144/// count: a reader verifies the bytes before they can name a series.
145fn place_keys(leaves: Vec<Vec<PendingBlock>>, start: u64) -> Result<(Vec<Vec<Block>>, Vec<u8>)> {
146    let mut area: Vec<u8> = Vec::new();
147    let mut previous: Option<(Vec<u8>, KeyRef)> = None;
148    let mut placed = Vec::with_capacity(leaves.len());
149    for pending in leaves {
150        let mut blocks = Vec::with_capacity(pending.len());
151        for block in pending {
152            let reuse = previous
153                .as_ref()
154                .filter(|(key, _)| key.as_slice() == block.key.bytes())
155                .map(|(_, reference)| *reference);
156            let reference = match reuse {
157                Some(reference) => reference,
158                None => {
159                    let length = u32::try_from(block.key.bytes().len())
160                        .map_err(|_| invalid("sample series key exceeds 4 GiB"))?;
161                    let offset = start
162                        .checked_add(
163                            u64::try_from(area.len())
164                                .map_err(|_| invalid("sample key area overflow"))?,
165                        )
166                        .ok_or_else(|| invalid("sample key offset overflow"))?;
167                    area.extend_from_slice(block.key.bytes());
168                    area.extend_from_slice(&codec::checksum(block.key.bytes()).to_le_bytes());
169                    let reference = KeyRef { offset, length };
170                    previous = Some((block.key.bytes().to_vec(), reference));
171                    reference
172                }
173            };
174            blocks.push(Block {
175                id: block.key.id(),
176                key: reference,
177                min: block.min,
178                max: block.max,
179                ordinal: block.ordinal,
180                rows: block.rows,
181                columns: block.columns,
182            });
183        }
184        placed.push(blocks);
185    }
186    Ok((placed, area))
187}

217        options: ColumnarSamplesFormat,
218        metadata: RegionMetadataRef,

232        )?;

234        let DataType::Timestamp(unit, None) = timestamp.column_schema.data_type.as_arrow_type()
235        else {
236            return Err(invalid("samples require a timestamp time index"));
237        };

266            manifest_bytes <= MANIFEST_LIMIT,

297    async fn write_batch(&mut self, batch: &RecordBatch) -> Result<()> {
298        require(batch.num_columns() >= 4, "samples require flat batches")?;
299        let column = |name: &str| {
300            batch
301                .column_by_name(name)
302                .ok_or_else(|| invalid(format!("missing sample input column {name}")))
303        };
304        let time = column(&self.manifest.timestamp.name)?;
305        require(
306            time.data_type() == &DataType::Timestamp(self.manifest.timestamp.unit.arrow(), None)
307                && time.null_count() == 0,
308            "samples timestamp type/null mismatch",
309        )?;
310        let (times, _) = timestamp_array_to_primitive(time)
311            .ok_or_else(|| invalid("invalid sample timestamp array"))?;
312        let sequence = column(consts::SEQUENCE_COLUMN_NAME)?
313            .as_any()
314            .downcast_ref::<UInt64Array>()
315            .ok_or_else(|| invalid("invalid sample sequence type"))?;
316        let operations = column(consts::OP_TYPE_COLUMN_NAME)?
317            .as_any()
318            .downcast_ref::<UInt8Array>()
319            .ok_or_else(|| invalid("invalid sample operation type"))?;
320        require(
321            sequence.null_count() == 0 && operations.null_count() == 0,
322            "null sample sequence/operation",
323        )?;
324        let fields = self
325            .manifest
326            .fields
327            .iter()
328            .map(|field| {
329                let array = column(&field.name)?;
330                let input = if field.is_histogram() {
331                    array
332                        .as_struct_opt()
333                        .map(|s| FieldInput::Histogram(array, s))
334                } else {
335                    array.as_any().downcast_ref().map(FieldInput::Float)
336                };
337                input.ok_or_else(|| invalid("invalid sample field array"))
338            })
339            .collect::<Result<Vec<_>>>()?;
340        let keys = column(consts::PRIMARY_KEY_COLUMN_NAME)?;
341        require(keys.null_count() == 0, "null sample primary key")?;
342        let binary = keys.as_any().downcast_ref::<BinaryArray>();
343        let dictionary = keys.as_any().downcast_ref::<DictionaryArray<UInt32Type>>();
344        let values = dictionary.and_then(|d| d.values().as_any().downcast_ref::<BinaryArray>());
345        require(
346            binary.is_some() || values.is_some(),
347            "samples require binary or UInt32 dictionary primary keys",
348        )?;
349        for i in 0..batch.num_rows() {
350            let key = if let Some(binary) = binary {
351                binary.value(i)
352            } else {
353                let dictionary = dictionary.ok_or_else(|| invalid("missing sample dictionary"))?;
354                let values = values.ok_or_else(|| invalid("invalid sample dictionary values"))?;
355                let index = dictionary.keys().value(i) as usize;
356                require(
357                    index < values.len() && !values.is_null(index),
358                    "invalid sample dictionary key",
359                )?;
360                values.value(index)
361            };
362            let time = times.value(i);
363            let seq = sequence.value(i);
364            let op = operations.value(i);
365            require(op <= 1, "unknown sample operation")?;
366            // Identity is the exact sparse key, not `(table_id, tsid)`: two distinct
367            // series may share a `tsid`, and every block records which exact key owns it.
368            let new_series = self
369                .series_key
370                .as_ref()
371                .is_none_or(|current| current.bytes() != key);
372            if new_series {
373                // Validate the whole key once per series: identity is the exact key, and a
374                // malformed one must not reach a block descriptor.
375                let exact = ExactSeriesKey::from_sparse_key(key, &mut self.offsets)?;
376                if self.last.is_some() {
377                    // Rows arrive in canonical sparse-key order.
378                    let previous = self
379                        .series_key
380                        .as_ref()
381                        .ok_or_else(|| invalid("missing sample series key"))?;
382                    require(
383                        exact > *previous,
384                        "sample input is not sorted by primary key",
385                    )?;
386                    // A different key that shares `(table_id, tsid)` is a different
387                    // series: it opens its own block, which records its exact key, so
388                    // timestamps and ordinals may restart.
389                    self.flush_block().await?;
390                    self.ordinal = 0;
391                }
392                self.series_key = Some(exact);
393            }
394            let id = match self.last {
395                Some((id, last_time, last_seq)) if !new_series => {
396                    // The same exact series continues: the existing monotonic rules hold.
397                    require(
398                        time > last_time || (time == last_time && seq <= last_seq),
399                        "sample input is not sorted",
400                    )?;
401                    id
402                }
403                _ => decode_series_id(key, &mut self.offsets)?,
404            };
405            let mut bounds = self.histogram_bounds(&fields, i);
406            if self.rows.len() == self.options.block_rows
407                || !valid_rows(self.rows.len() + 1, fields.len())
408                || bounds.is_none()
409            {
410                self.flush_block().await?;
411                bounds = self.histogram_bounds(&fields, i);
412            }
413            let mut bounds = bounds
414                .ok_or_else(|| invalid("native histogram sample exceeds the chunk limit"))?
415                .into_iter();
416            self.rows.push(Row {
417                id,
418                time,
419                sequence: seq,
420                operation: op,
421            });
422            for (pending, input) in self.fields.iter_mut().zip(&fields) {
423                match (pending, input) {
424                    (PendingField::Float(values), FieldInput::Float(array)) => {
425                        values.push(array.is_valid(i).then(|| array.value(i).to_bits()));
426                    }
427                    (PendingField::Histogram { runs, bound }, FieldInput::Histogram(array, _)) => {
428                        *bound = bounds
429                            .next()
430                            .ok_or_else(|| invalid("missing native histogram bound"))?;
431                        match runs.last_mut() {
432                            Some((run, start, len))
433                                if Arc::ptr_eq(run, array) && *start + *len == i =>
434                            {
435                                *len += 1
436                            }
437                            _ => runs.push(((*array).clone(), i, 1)),
438                        }
439                    }
440                    _ => return Err(invalid("sample field kind changed between rows")),
441                }
442            }
443            self.last = Some((id, time, seq));
444        }
445        Ok(())
446    }
447    /// Each histogram field's chunk bound with row `i` appended to the pending block, in field
448    /// order, or `None` when one of those chunks would no longer fit.
449    fn histogram_bounds(&self, fields: &[FieldInput<'_>], i: usize) -> Option<Vec<ChunkBound>> {
450        self.fields
451            .iter()
452            .zip(fields)
453            .filter_map(|(pending, input)| match (pending, input) {
454                (PendingField::Histogram { bound, .. }, FieldInput::Histogram(_, array)) => {
455                    Some(bound.with_row(array, i))
456                }
457                _ => None,
458            })
459            .collect()
460    }
461    async fn flush_block(&mut self) -> Result<()> {
462        if self.rows.is_empty() {
463            return Ok(());
464        }
465        let rows = std::mem::take(&mut self.rows);
466        let fields = std::mem::replace(
467            &mut self.fields,
468            self.manifest.fields.iter().map(PendingField::new).collect(),
469        );
470        let key = self
471            .series_key
472            .as_ref()
473            .ok_or_else(|| invalid("sample block has no series key"))?;
474        let (mut block, chunks) = encode_block(&rows, fields, key, &self.manifest, self.ordinal)?;
475        // Bytes one copy of this series' key costs in a part's key area: the key and its stored
476        // checksum.
477        let key_len = key.bytes().len() as u64 + metadata::KEY_CHECKSUM_BYTES;
478        self.ordinal = self
479            .ordinal
480            .checked_add(1)
481            .ok_or_else(|| invalid("sample ordinal overflow"))?;
482        let size = block.size();
483        // The descriptor carries a fixed-size key reference, not the key, so this bound is a
484        // function of the field count only: a key longer than a whole metadata page is no
485        // longer a build failure, and only a narrower configured page can reject a descriptor.
486        require(
487            size + 12 <= self.options.page_bytes,
488            "sample descriptor exceeds leaf limit",
489        )?;
490        if !self.leaves.is_empty()
491            && self.leaf_bytes + size > self.options.page_bytes
492            && self.leaves.len() == self.options.root_capacity
493        {
494            self.finish_part().await?;
495        }
496        if self.outputs.active.is_none() {
497            let id = u32::try_from(self.manifest.files.len())
498                .map_err(|_| invalid("too many sample parts"))?;
499            self.outputs.open(self.manifest.part_path(id)).await?;
500            self.output_buffer
501                .write(&mut self.outputs, &metadata::header())
502                .await?;
503            self.position = metadata::HEADER_SIZE;
504        }
505        for (column, chunk) in block.columns.iter_mut().zip(chunks) {
506            column.payload.offset = self.position;
507            self.position = column.payload.end()?;
508            self.output_buffer.write(&mut self.outputs, &chunk).await?;
509        }
510        if self.leaves.is_empty() || self.leaf_bytes + size > self.options.page_bytes {
511            self.leaves.push(Vec::new());
512            self.leaf_bytes = 12;
513        }
514        self.leaf_bytes += size;
515        self.descriptor_bytes += size;
516        self.part_rows = self
517            .part_rows
518            .checked_add(u64::from(block.rows))
519            .ok_or_else(|| invalid("sample row count overflow"))?;
520        self.part_blocks += 1;
521        self.leaves
522            .last_mut()
523            .ok_or_else(|| invalid("missing sample leaf"))?
524            .push(block);
525        // Keys live in their own area but are still bytes of this part: count one copy per
526        // series so the part target covers them too, and a key-heavy part still rolls over.
527        // This is decided after the rollover above, which cleared the part's accounting: a series
528        // that starts a new part still needs its copy counted in that part.
529        if self.part_last_key.as_ref() != self.series_key.as_ref() {
530            self.part_key_bytes = self
531                .part_key_bytes
532                .checked_add(key_len)
533                .ok_or_else(|| invalid("sample key area overflow"))?;
534            self.part_last_key = self.series_key.clone();
535        }
536        if self.position.saturating_add(self.part_key_bytes) >= self.options.part_bytes
537            || self.descriptor_bytes >= self.options.metadata_bytes
538        {
539            self.finish_part().await?;
540        }
541        Ok(())
542    }
543    async fn write_page(&mut self, page: Page) -> Result<PageReference> {
544        let bounds = page.bounds()?;
545        let (payload, bytes) = pack(&page.encode(), self.position, 1, true)?;
546        self.position = payload.end()?;
547        self.output_buffer.write(&mut self.outputs, &bytes).await?;
548        Ok(PageReference { bounds, payload })
549    }
550    async fn finish_part(&mut self) -> Result<()> {
551        if self.leaves.is_empty() {
552            return Ok(());
553        }
554        let payload_end = self.position;
555        // Keys go to their own area after the payload. Every block of one series shares one
556        // copy, so the area holds one key per distinct series of this part.
557        let (leaves, key_area) = place_keys(std::mem::take(&mut self.leaves), payload_end)?;
558        if !key_area.is_empty() {
559            self.position = self
560                .position
561                .checked_add(
562                    u64::try_from(key_area.len())
563                        .map_err(|_| invalid("sample key area overflow"))?,
564                )
565                .ok_or_else(|| invalid("sample file length overflow"))?;
566            self.output_buffer
567                .write(&mut self.outputs, &key_area)
568                .await?;
569        }
570        let key_end = self.position;
571        let mut children = Vec::new();
572        for leaf in leaves {
573            children.push(self.write_page(Page::Leaf(leaf)).await?);
574        }
575        let depth = if children.len() == 1 { 1 } else { 2 };
576        let root = if depth == 1 {
577            children.remove(0)
578        } else {
579            self.write_page(Page::Internal(children)).await?
580        };
581        let bounds = root.bounds;
582        let file_length = self
583            .position
584            .checked_add(metadata::TRAILER_SIZE)
585            .ok_or_else(|| invalid("sample file length overflow"))?;
586        self.output_buffer
587            .write(
588                &mut self.outputs,
589                &Trailer {
590                    root,
591                    depth,
592                    file_length,
593                    payload_end,
594                    key_end,
595                }
596                .encode(),
597            )
598            .await?;
599        self.output_buffer.flush(&mut self.outputs).await?;
600        self.outputs.close().await?;
601        let part = Part {
602            part_id: u32::try_from(self.manifest.files.len())
603                .map_err(|_| invalid("too many sample parts"))?,
604            file_size: file_length,
605            num_rows: self.part_rows,
606            num_blocks: self.part_blocks,
607            first: bounds.first,
608            last: bounds.last,
609            min_timestamp: bounds.min,
610            max_timestamp: bounds.max,
611        };
612        self.manifest_bytes += serde_json::to_vec(&part).context(SerdeJsonSnafu)?.len()
613            + usize::from(!self.manifest.files.is_empty());
614        require(
615            self.manifest_bytes <= MANIFEST_LIMIT,
616            "samples manifest exceeds size limit",
617        )?;
618        self.manifest.files.push(part);
619        self.position = 0;
620        self.part_rows = 0;
621        self.part_blocks = 0;
622        self.part_key_bytes = 0;
623        self.part_last_key = None;
624        self.descriptor_bytes = 0;
625        self.leaf_bytes = 12;
626        Ok(())
627    }
628    async fn complete(&mut self) -> Result<ArtifactMetadata> {
629        require(!self.failed, "sample writer failed previously")?;
630        self.flush_block().await?;
631        self.finish_part().await?;
632        let bytes = serde_json::to_vec(&self.manifest).context(SerdeJsonSnafu)?;
633        require(
634            bytes.len() <= MANIFEST_LIMIT,
635            "samples manifest exceeds limit",
636        )?;
637        let mut file_size = bytes.len() as u64;
638        let mut objects = vec![ArtifactObject {
639            path: self.manifest.path(),
640            file_size,
641        }];
642        for part in &self.manifest.files {
643            objects.push(ArtifactObject {
644                path: self.manifest.part_path(part.part_id),
645                file_size: part.file_size,
646            });
647            file_size = file_size
648                .checked_add(part.file_size)
649                .ok_or_else(|| invalid("sample artifact size overflow"))?;
650        }
651        self.outputs.open(self.manifest.path()).await?;
652        self.outputs.write(bytes).await?;
653        self.outputs.close().await?;
654        self.outputs.committed = true;
655        Ok(ArtifactMetadata {
656            format: ArtifactFormat::ColumnarSamples,
657            generation: self.manifest.generation,
658            objects,
659            file_size,
660        })
661    }
662}

680#[async_trait]
681impl FormatWriter for SamplesWriter {
682    async fn write(&mut self, batch: &RecordBatch) -> Result<()> {
683        require(!self.failed, "sample writer failed previously")?;
684        self.failed = true;
685        let result = self.write_batch(batch).await;
686        if result.is_ok() {
687            self.failed = false;
688        }
689        result
690    }
691    async fn finish(mut self: Box<Self>) -> Result<ArtifactMetadata> {
692        let result = self.complete().await;
693        if result.is_err()
694            && let Err(error) = self.outputs.cleanup().await
695        {
696            common_telemetry::warn!(error; "Cleaning failed samples build");
697        }
698        result
699    }
700    async fn abort(mut self: Box<Self>) -> Result<()> {
701        self.outputs.cleanup().await
702    }

704fn encode_block(
705    rows: &[Row],
706    fields: Vec<PendingField>,
707    key: &ExactSeriesKey,
708    manifest: &Manifest,
709    ordinal: u64,
710) -> Result<(PendingBlock, Vec<Vec<u8>>)> {
711    let first = rows.first().ok_or_else(|| invalid("empty sample block"))?;
712    let last = rows.last().ok_or_else(|| invalid("empty sample block"))?;
713    require(
714        rows.iter().all(|row| row.id == key.id()),
715        "sample block mixes series",
716    )?;
717    let mut block = PendingBlock {
718        key: key.clone(),
719        min: first.time,
720        max: last.time,
721        ordinal,
722        rows: rows.len() as u32,
723        columns: Vec::new(),
724    };
725    let mut chunks = Vec::new();
726    let mut push = |role, id, encoding, bytes: Vec<u8>, nulls| -> Result<()> {
727        let (payload, bytes) = pack(&bytes, 0, encoding, false)?;
728        block.columns.push(Column {
729            role,
730            id,
731            payload,
732            nulls,
733        });
734        chunks.push(bytes);
735        Ok(())
736    };
737    let (encoding, bytes) = codec::encode_time(&rows.iter().map(|r| r.time).collect::<Vec<_>>());
738    push(1, manifest.timestamp.column_id, encoding, bytes, 0)?;
739    let (encoding, bytes) =
740        codec::encode_sequence(&rows.iter().map(|r| r.sequence).collect::<Vec<_>>());
741    push(2, 0, encoding, bytes, 0)?;
742    let (encoding, bytes) =
743        codec::encode_operations(&rows.iter().map(|r| r.operation).collect::<Vec<_>>());
744    push(3, 0, encoding, bytes, 0)?;
745    for (field, pending) in manifest.fields.iter().zip(fields) {
746        match pending {
747            PendingField::Float(values) => {
748                require(
749                    values.len() == rows.len(),
750                    "sample field row count mismatch",
751                )?;
752                let nulls = values.iter().filter(|v| v.is_none()).count();
753                if nulls != rows.len() {
754                    push(
755                        4,
756                        field.column_id,
757                        FLOAT64_ENCODING,
758                        codec::encode_float(&values),
759                        nulls as u32,
760                    )?;
761                }
762            }
763            PendingField::Histogram { runs, .. } => {
764                let slices = runs
765                    .iter()
766                    .map(|(array, start, len)| array.slice(*start, *len))
767                    .collect::<Vec<_>>();
768                let array = concat(&slices.iter().map(|a| a.as_ref()).collect::<Vec<_>>())
769                    .context(ComputeArrowSnafu)?;
770                let array = array
771                    .as_struct_opt()
772                    .ok_or_else(|| invalid("invalid native histogram sample array"))?;
773                require(array.len() == rows.len(), "sample field row count mismatch")?;
774                // An all-null field has no chunk, as for Float64 fields.
775                if array.null_count() != rows.len() {
776                    push(
777                        4,
778                        field.column_id,
779                        histogram::ENCODING,
780                        histogram::encode(array)?,
781                        array.null_count() as u32,
782                    )?;
783                }
784            }
785        }
786    }
787    Ok((block, chunks))
788}

803    #[tokio::test]
804    async fn buffered_output_boundaries_and_parts() {
805        let capacity = DEFAULT_WRITE_BUFFER_SIZE.as_bytes() as usize;
806        let mut outputs = outputs();
807        let mut buffer = SampleBuffer::default();
808        for size in [0, capacity - 1, capacity, capacity + 1, capacity * 2 + 13] {
809            let path = format!("part-{size}");
810            outputs.open(path.clone()).await.unwrap();
811            let bytes: Vec<_> = (0..size).map(|i| (i % 251) as u8).collect();
812            // Exercise both tiny fragments and an input larger than the staging buffer.
813            let split = size.min(1000);
814            for chunk in bytes[..split].chunks(7) {
815                buffer.write(&mut outputs, chunk).await.unwrap();
816            }
817            buffer.write(&mut outputs, &bytes[split..]).await.unwrap();
818            assert_eq!(buffer.bytes.len(), size % capacity);
819            buffer.flush(&mut outputs).await.unwrap();
820            outputs.close().await.unwrap();
821            assert!(buffer.bytes.is_empty());
822            assert_eq!(outputs.store.read(&path).await.unwrap().to_vec(), bytes);
823        }
824        outputs.cleanup().await.unwrap();
825    }
