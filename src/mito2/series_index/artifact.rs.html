26/// Persisted artifact identifiers. Concrete codecs are added with their implementations.
27#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
28pub(crate) enum ArtifactFormat {
29    #[serde(rename = "parquet")]
30    Parquet,
31    #[serde(rename = "columnar-series")]
32    ColumnarSeries,
33    #[serde(rename = "columnar-samples")]
34    ColumnarSamples,
35    #[cfg(test)]
36    #[serde(rename = "fake-samples")]
37    FakeSamples,
38    #[cfg(test)]
39    #[serde(rename = "fake-index")]
40    FakeIndex,
41}

43/// One immutable object produced by a backend build.

45pub(crate) struct ArtifactObject {
46    pub(crate) path: String,
47    pub(crate) file_size: u64,

50/// Identifies the exact bytes produced by one backend build.
51#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
52pub(crate) struct ArtifactMetadata {
53    pub(crate) format: ArtifactFormat,
54    pub(crate) generation: FileId,
55    /// Every object owned by this build, including metadata and auxiliary files.
56    pub(crate) objects: Vec<ArtifactObject>,
57    pub(crate) file_size: u64,
58}

60/// One samples artifact derived from immutable Parquet SSTs.
61#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
62pub(crate) struct SamplesEntry {
63    pub(crate) file_id: FileId,
64    pub(crate) artifact: ArtifactMetadata,
65    /// Empty for legacy per-SST artifacts. Otherwise the complete source set for a UTC day.
66    #[serde(default, skip_serializing_if = "Vec::is_empty")]
67    pub(crate) source_file_ids: Vec<FileId>,
68    /// UTC day number since the Unix epoch; absent for legacy per-SST artifacts.
69    #[serde(default, skip_serializing_if = "Option::is_none")]
70    pub(crate) window: Option<i64>,
71    /// Complete nonempty UTC-day coverage, stored on one anchor entry per source group.
72    #[serde(default, skip_serializing_if = "std::collections::HashMap::is_empty")]
73    pub(crate) source_windows: std::collections::HashMap<FileId, Vec<i64>>,
74}
75
76impl SamplesEntry {
77    pub(crate) fn covers(&self, id: FileId) -> bool {
78        if self.window.is_some() {
79            self.source_file_ids.contains(&id)
80        } else {
81            self.file_id == id
82        }
83    }
84
85    pub(crate) fn sources_visible(&self, visible: &std::collections::HashSet<FileId>) -> bool {
86        if self.window.is_some() {
87            !self.source_file_ids.is_empty()
88                && self.source_file_ids.iter().all(|id| visible.contains(id))
89        } else {
90            visible.contains(&self.file_id)
91        }
92    }
93}
94
95/// Requires either a legacy whole-SST artifact or complete daily coverage.
96pub(crate) fn samples_cover_file(
97    samples: &std::collections::HashMap<FileId, SamplesFileHandle>,
98    file: &crate::sst::file::FileHandle,
99) -> bool {
100    let id = file.file_id().file_id();
101    if samples
102        .values()
103        .any(|h| h.entry().window.is_none() && h.entry().covers(id))
104    {
105        return true;
106    }
107    samples.values().any(|handle| {
108        let Some(days) = handle.entry().source_windows.get(&id) else {
109            // Earlier daily catalogs require rebuilding to establish actual row coverage.
110            return false;
111        };
112        !days.is_empty()
113            && days.iter().all(|day| {
114                samples.values().any(|h| {
115                    h.entry().window == Some(*day)
116                        && h.entry().covers(id)
117                        && h.entry().source_file_ids == handle.entry().source_file_ids
118                })
119            })
120    })
121}

123#[derive(Debug, Clone)]
124pub(crate) struct SamplesFileHandle(Arc<SamplesFileInner>);
125
126#[derive(Debug)]
127struct SamplesFileInner {
128    file_id: RegionFileId,
129    entry: SamplesEntry,
130    retired: AtomicBool,
131    lifetime: SamplesLifetime,
132}
133
134#[derive(Debug)]
135enum SamplesLifetime {
136    Sidecar(IndexFilePurger),
137    Narthex(crate::sst::file::FileHandle),
138}
139
140impl SamplesFileHandle {
141    pub(crate) fn new(file_id: RegionFileId, entry: SamplesEntry, purger: IndexFilePurger) -> Self {
142        Self(Arc::new(SamplesFileInner {
143            file_id,
144            entry,
145            retired: AtomicBool::new(false),
146            lifetime: SamplesLifetime::Sidecar(purger),
147        }))
148    }
149
150    /// Pins an authoritative SST instead of giving the sidecar purger ownership.
151    pub(crate) fn from_narthex(
152        file: crate::sst::file::FileHandle,
153        region_dir: &str,
154    ) -> crate::error::Result<Self> {
155        let crate::sst::file::FileFormat::NarthexSamples(_) = &file.meta_ref().format else {
156            return crate::error::UnsupportedOperationSnafu {
157                err_msg: "Expected narthex samples file",
158            }
159            .fail();
160        };
161        let objects = file
162            .meta_ref()
163            .narthex_objects(region_dir)
164            .expect("matched Narthex samples");
165        let file_size = file.meta_ref().file_size;
166        let file_id = file.file_id();
167        Ok(Self(Arc::new(SamplesFileInner {
168            file_id,
169            entry: SamplesEntry {
170                file_id: file_id.file_id(),
171                artifact: ArtifactMetadata {
172                    format: ArtifactFormat::ColumnarSamples,
173                    generation: file_id.file_id(),
174                    objects,
175                    file_size,
176                },
177                source_file_ids: Vec::new(),
178                window: None,
179                source_windows: Default::default(),
180            },
181            retired: AtomicBool::new(false),
182            lifetime: SamplesLifetime::Narthex(file),
183        })))
184    }
185
186    pub(crate) fn file_id(&self) -> RegionFileId {
187        match &self.0.lifetime {
188            SamplesLifetime::Sidecar(_) => self.0.file_id,
189            SamplesLifetime::Narthex(file) => file.file_id(),
190        }
191    }
192
193    pub(crate) fn manifest_path(&self) -> String {
194        if matches!(self.0.lifetime, SamplesLifetime::Narthex(_)) {
195            return self
196                .0
197                .entry
198                .artifact
199                .objects
200                .last()
201                .expect("Narthex manifest")
202                .path
203                .clone();
204        }
205        format!(
206            "{}manifest.json",
207            crate::series_index::samples::samples_prefix(
208                self.file_id().region_id().as_u64(),
209                self.entry().artifact.generation
210            )
211        )
212    }
213
214    pub(crate) fn entry(&self) -> &SamplesEntry {
215        &self.0.entry
216    }
217
218    pub(crate) fn mark_deleted(&self) {
219        self.0.retired.store(true, Ordering::Release);
220    }
221}
222
223impl Drop for SamplesFileInner {
224    fn drop(&mut self) {
225        if self.retired.load(Ordering::Acquire)
226            && let SamplesLifetime::Sidecar(purger) = &self.lifetime
227        {
228            purger.purge_objects(
229                self.file_id,
230                self.entry
231                    .artifact
232                    .objects
233                    .iter()
234                    .map(|object| object.path.clone())
235                    .collect(),
236            );
237        }
238    }
239}

241#[cfg(test)]
242mod tests {
243    use super::*;
244    use std::collections::HashSet;
245
246    #[test]
247    fn frozen_sample_catalog_coverage() {
248        let legacy: SamplesEntry = serde_json::from_str(include_str!(
249            "../../../../tests/compatibility/wire/mito_columnar_samples/legacy-catalog-entry.json"
250        ))
251        .unwrap();
252        assert!(legacy.window.is_none());
253        assert!(legacy.source_file_ids.is_empty());
254        assert!(legacy.covers(legacy.file_id));
255        let daily: SamplesEntry = serde_json::from_str(include_str!(
256            "../../../../tests/compatibility/wire/mito_columnar_samples/daily-catalog-entry.json"
257        ))
258        .unwrap();
259        assert!(daily.source_windows.is_empty());
260        let sparse: SamplesEntry = serde_json::from_str(include_str!(
261            "../../../../tests/compatibility/wire/mito_columnar_samples/sparse-daily-catalog-entry.json"
262        )).unwrap();
263        assert_eq!(sparse.source_windows[&legacy.file_id], vec![0, 3]);
264        assert_eq!(
265            sparse,
266            serde_json::from_str(&serde_json::to_string(&sparse).unwrap()).unwrap()
267        );
268        let mut visible: HashSet<_> = daily.source_file_ids.iter().copied().collect();
269        assert!(daily.sources_visible(&visible));
270        assert!(daily.covers(legacy.file_id));
271        assert!(!daily.covers(daily.file_id));
272        visible.remove(&legacy.file_id);
273        assert!(!daily.sources_visible(&visible));
274        assert_eq!(
275            daily,
276            serde_json::from_str(&serde_json::to_string(&daily).unwrap()).unwrap()
277        );
278        assert!(
279            !serde_json::to_string(&legacy)
280                .unwrap()
281                .contains("source_file_ids")
282        );
283    }
