39/// Searches a series-index file for metric series matching query predicates.
40pub struct SeriesIndexSearcher {
41    /// The file to search, or `None` when an empty time range rules out
42    /// every series without opening the file.
43    reader: Option<ParquetIndexReader>,
44    /// Row-group pruning predicate built from all of the file's filters.
45    pruning_predicate: Predicate,
46    filters: Vec<SimpleFilterEvaluator>,
47}
48
49impl SeriesIndexSearcher {
50    /// Creates a searcher for the series-index file of `metadata` at `path`.
51    /// Predicates are built from the unit recorded in the file's schema, so a
52    /// file written before a time index unit widening keeps being interpreted
53    /// in its own unit.
54    pub async fn try_new(
55        metadata: RegionMetadataRef,
56        object_store: ObjectStore,
57        path: &str,
58        predicate: Option<&Predicate>,
59        time_range: Option<TimestampRange>,
60    ) -> Result<Self> {
61        // Keep search-time metadata validation identical to the writer.
62        series_index_schema(&metadata)?;
63        if time_range.as_ref().is_some_and(TimestampRange::is_empty) {
64            return Ok(Self {
65                reader: None,
66                pruning_predicate: Predicate::new(Vec::new()),
67                filters: Vec::new(),
68            });
69        }
70
71        let reader = ParquetIndexReader::open(object_store, path).await?;
72        let unit = validate_index_schema(reader.schema())?;
73
74        let mut filters = simple_tag_filters(&metadata, None, predicate);
75        for expr in time_range_exprs(unit, time_range.as_ref()) {
76            let filter = SimpleFilterEvaluator::try_new(&expr).context(UnexpectedSnafu {
77                reason: "failed to build an internal series-index time filter",
78            })?;
79            filters.push((expr, filter));
80        }
81        // An older index file may not contain tags added by schema evolution.
82        // Ignore filters on those tags to preserve a conservative candidate set.
83        let (pruning_predicate, filters) = filters_for_schema(reader.schema(), &filters);
84
85        Ok(Self {
86            reader: Some(reader),
87            pruning_predicate,
88            filters,
89        })
90    }
91
92    /// Searches the index file and returns sorted batches of matching
93    /// metric-series IDs.
94    pub fn search(&self) -> Result<MetricSeriesIdStream> {
95        let Some(reader) = self.reader.as_ref() else {
96            return Ok(Box::pin(futures::stream::empty()));
97        };
98        let mut projection_columns = Vec::with_capacity(self.filters.len() + 2);
99        projection_columns.extend([TABLE_ID_COLUMN, TSID_COLUMN]);
100        projection_columns.extend(self.filters.iter().map(SimpleFilterEvaluator::column_name));
101        let mut batches = reader.read(&self.pruning_predicate, &projection_columns)?;
102        let filters = self.filters.clone();
103
104        Ok(Box::pin(try_stream! {
105            let mut last_series = None;
106            let mut output = Vec::with_capacity(METRIC_SERIES_ID_BATCH_SIZE);
107            while let Some(batch) = batches.try_next().await? {
108                let mut mask = BooleanBuffer::new_set(batch.num_rows());
109                for filter in &filters {
110                    let column = column(&batch, filter.column_name())?;
111                    let evaluated = filter.evaluate_array(column).context(RecordBatchSnafu)?;
112                    mask = &mask & &evaluated;
113                }
114
115                let table_ids = column(&batch, TABLE_ID_COLUMN)?
116                    .as_any()
117                    .downcast_ref::<UInt32Array>()
118                    .context(InvalidRecordBatchSnafu {
119                        reason: "series index __table_id is not UInt32",
120                    })?;
121                let tsids = column(&batch, TSID_COLUMN)?
122                    .as_any()
123                    .downcast_ref::<UInt64Array>()
124                    .context(InvalidRecordBatchSnafu {
125                        reason: "series index __tsid is not UInt64",
126                    })?;
127
128                for (row, matched) in mask.iter().enumerate() {
129                    if !matched {
130                        continue;
131                    }
132                    let series = MetricSeriesId {
133                        table_id: table_ids.value(row),
134                        tsid: tsids.value(row),
135                    };
136                    if last_series == Some(series) {
137                        continue;
138                    }
139                    last_series = Some(series);
140                    output.push(series);
141                    if output.len() == METRIC_SERIES_ID_BATCH_SIZE {
142                        yield std::mem::replace(
143                            &mut output,
144                            Vec::with_capacity(METRIC_SERIES_ID_BATCH_SIZE),
145                        );
146                    }
147                }
148            }
149            if !output.is_empty() {
150                yield output;
151            }
152        }))
153    }
154}

156fn filters_for_schema(
157    schema: &SchemaRef,
158    filters: &[(Expr, SimpleFilterEvaluator)],
159) -> (Predicate, Vec<SimpleFilterEvaluator>) {
160    let (exprs, filters): (Vec<_>, Vec<_>) = filters
161        .iter()
162        .filter(|(_, filter)| schema.field_with_name(filter.column_name()).is_ok())
163        .cloned()
164        .unzip();
165    (Predicate::new(exprs), filters)
166}

168// Builds `__series_min_ts`/`__series_max_ts` predicates in `unit`, the unit
169// recorded in the file being searched: its raw i64 bounds were written in that
170// unit even if the region's time index has since been widened. The searcher
171// handles empty ranges itself, so `time_range`, if present, is non-empty.
172fn time_range_exprs(unit: TimeUnit, time_range: Option<&TimestampRange>) -> Vec<Expr> {
173    let Some(time_range) = time_range else {
174        return Vec::new();
175    };
176    let mut exprs = Vec::with_capacity(2);
177    // A series overlaps [start, end) only if its maximum is at least start.
178    // Round start up so a series ending before an unaligned start is pruned.
179    if let Some(start) = time_range
180        .start()
181        .and_then(|start| start.convert_to_ceil(unit))
182    {
183        exprs.push(
184            col(MAX_TS_COLUMN).gt_eq(lit(timestamp_to_scalar_value(unit, Some(start.value())))),
185        );
186    }
187    // A series overlaps [start, end) only if its minimum is less than end.
188    // Round the exclusive end up to avoid pruning the containing unit interval.
189    if let Some(end) = time_range.end().and_then(|end| end.convert_to_ceil(unit)) {
190        exprs.push(col(MIN_TS_COLUMN).lt(lit(timestamp_to_scalar_value(unit, Some(end.value())))));
191    }
192    exprs
193}

195/// Validates an index file's schema and returns the time unit of its
196/// `__series_min_ts`/`__series_max_ts` columns. They are native
197/// `Timestamp(unit)` columns, so the unit is part of the datatype and each
198/// file is interpreted in the unit it was written with.
199fn validate_index_schema(schema: &SchemaRef) -> Result<TimeUnit> {
200    for (name, data_type) in [
201        (ROW_COUNT_COLUMN, DataType::UInt64),
202        (TABLE_ID_COLUMN, DataType::UInt32),
203        (TSID_COLUMN, DataType::UInt64),
204    ] {
205        let field = schema
206            .field_with_name(name)
207            .ok()
208            .with_context(|| InvalidRecordBatchSnafu {
209                reason: format!("series index is missing internal column {name}"),
210            })?;
211        ensure!(
212            field.data_type() == &data_type && !field.is_nullable(),
213            InvalidRecordBatchSnafu {
214                reason: format!(
215                    "series index internal column {name} must be non-nullable {data_type:?}, got {:?}",
216                    field.data_type()
217                ),
218            }
219        );
220    }
221    let unit = |name: &str| {
222        let field = schema
223            .field_with_name(name)
224            .ok()
225            .context(InvalidRecordBatchSnafu {
226                reason: format!("series index is missing internal column {name}"),
227            })?;
228        ensure!(
229            !field.is_nullable(),
230            InvalidRecordBatchSnafu {
231                reason: format!("series index column {name} must be non-nullable"),
232            }
233        );
234        match field.data_type() {
235            DataType::Timestamp(unit, _) => Ok(unit.into()),
236            data_type => InvalidRecordBatchSnafu {
237                reason: format!(
238                    "series index column {name} must be a Timestamp, got {data_type:?}"
239                ),
240            }
241            .fail(),
242        }
243    };
244    let min_unit = unit(MIN_TS_COLUMN)?;
245    let max_unit = unit(MAX_TS_COLUMN)?;
246    ensure!(
247        min_unit == max_unit,
248        InvalidRecordBatchSnafu {
249            reason: format!(
250                "series index columns {MIN_TS_COLUMN} and {MAX_TS_COLUMN} have different time units"
251            ),
252        }
253    );
254    Ok(min_unit)
255}

257fn column<'a>(batch: &'a RecordBatch, name: &str) -> Result<&'a ArrayRef> {
258    let index = batch
259        .schema()
260        .index_of(name)
261        .ok()
262        .with_context(|| InvalidRecordBatchSnafu {
263            reason: format!("series index batch is missing column {name}"),
264        })?;
265    Ok(batch.column(index))
266}

320        RecordBatch::try_new(
321            schema,
322            vec![
323                ts_column,
324                Arc::new(BinaryArray::from_iter_values(
325                    primary_keys.iter().map(Vec::as_slice),

625    async fn search_reads_each_file_in_its_recorded_unit() {
626        // The first file is written while the region's time index is
627        // milliseconds; its series sit at 10ms and 20ms.
628        let metadata = Arc::new(sst_region_metadata_with_encoding(
629            PrimaryKeyEncoding::Sparse,
630        ));
631        let object_store = object_store();
632        write_index(
633            metadata.clone(),
634            object_store.clone(),
635            "old_ms.parquet",
636            &[(1, 10, "a", "x", 10), (1, 20, "b", "x", 20)],
637            2,
638        )
639        .await;
640
641        // The region's time index is then widened to microseconds and a
642        // second file is written; its series sit at 15_000µs and 15µs.
643        let mut widened = (*metadata).clone();
644        for column in &mut widened.column_metadatas {
645            if column.column_schema.name == "ts" {
646                column.column_schema.data_type = ConcreteDataType::timestamp_microsecond_datatype();
647            }
648        }
649        let widened = Arc::new(widened);
650        write_index_with_time_unit(
651            widened.clone(),
652            object_store.clone(),
653            "new_us.parquet",
654            &[(1, 30, "c", "x", 15_000), (1, 40, "d", "x", 15)],
655            2,
656            ArrowTimeUnit::Microsecond,
657        )
658        .await;
659
660        // Each file is searched by its own searcher, which interprets it in
661        // the unit recorded in its schema. For [10.5ms, 25ms):
662        // - the millisecond file (ceil: [11ms, 25ms)) keeps only the 20ms
663        //   series; reading it in microseconds would match nothing;
664        // - the microsecond file keeps only the 15_000µs series; reading it
665        //   in milliseconds would also keep the 15µs series.
666        let time_range = TimestampRange::new(
667            common_time::Timestamp::new_microsecond(10_500),
668            common_time::Timestamp::new_microsecond(25_000),
669        )
670        .unwrap();
671        let searcher = SeriesIndexSearcher::try_new(
672            widened.clone(),
673            object_store.clone(),
674            "old_ms.parquet",
675            None,
676            Some(time_range),
677        )
678        .await
679        .unwrap();
680        let ids = collect_ids(searcher.search().unwrap()).await;
681        assert_eq!(
682            ids,
683            vec![MetricSeriesId {
684                table_id: 1,
685                tsid: 20
686            }]
687        );
688        let searcher = SeriesIndexSearcher::try_new(
689            widened,
690            object_store,
691            "new_us.parquet",
692            None,
693            Some(time_range),
694        )
695        .await
696        .unwrap();
697        let ids = collect_ids(searcher.search().unwrap()).await;
698        assert_eq!(
699            ids,
700            vec![MetricSeriesId {
701                table_id: 1,
702                tsid: 30
703            }]
704        );
705    }

800    #[tokio::test]
801    async fn search_prunes_row_groups_and_empty_ranges() {
802        let metadata = Arc::new(sst_region_metadata_with_encoding(
803            PrimaryKeyEncoding::Sparse,
804        ));
805        let object_store = object_store();
806        let path = "pruning.parquet";
807        write_index(
808            metadata.clone(),
809            object_store.clone(),
810            path,
811            &[
812                (1, 0, "a", "x", 0),
813                (1, 1, "b", "x", 1),
814                (1, 2, "m", "x", 2),
815                (1, 3, "m", "x", 3),
816                (1, 4, "y", "x", 4),
817                (1, 5, "z", "x", 5),
818            ],
819            2,
820        )
821        .await;
822
823        let reader = ParquetIndexReader::open(object_store.clone(), path)
824            .await
825            .unwrap();
826        let unit = validate_index_schema(reader.schema()).unwrap();
827
828        // Tag predicates prune row groups through the tag-column statistics.
829        let predicate = Predicate::new(vec![col("tag_0").eq(lit("m"))]);
830        let mut filters = simple_tag_filters(&metadata, None, Some(&predicate));
831        let (pruning_predicate, _) = filters_for_schema(reader.schema(), &filters);
832        assert_eq!(reader.row_groups_to_read(&pruning_predicate), vec![1]);
833
834        // Time-range predicates prune row groups in the file's own unit:
835        // [2ms, 4ms) keeps only the row group holding the 2ms and 3ms series.
836        let time_range = TimestampRange::new(
837            common_time::Timestamp::new_millisecond(2),
838            common_time::Timestamp::new_millisecond(4),
839        )
840        .unwrap();
841        for expr in time_range_exprs(unit, Some(&time_range)) {
842            let filter = SimpleFilterEvaluator::try_new(&expr)
843                .context(UnexpectedSnafu {
844                    reason: "failed to build an internal series-index time filter",
845                })
846                .unwrap();
847            filters.push((expr, filter));
848        }
849        let (pruning_predicate, _) = filters_for_schema(reader.schema(), &filters);
850        assert_eq!(reader.row_groups_to_read(&pruning_predicate), vec![1]);
851
852        // An empty time range yields no series without opening the file.
853        let empty = SeriesIndexSearcher::try_new(
854            metadata,
855            object_store,
856            "does-not-need-to-exist.parquet",
857            None,
858            Some(TimestampRange::empty()),
859        )
860        .await
861        .unwrap();
862        assert!(
863            empty
864                .search()
865                .unwrap()
866                .try_collect::<Vec<_>>()
867                .await
868                .unwrap()
869                .is_empty()
870        );
871    }
