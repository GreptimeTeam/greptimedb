15//! The exact series identity that SSTs already store.
16//!
17//! `tsid` is a hash, so two distinct series can share one [`MetricSeriesId`]. The
18//! sparse primary key stored by the SST is exact, so every place that decides
19//! whether two rows belong to the same series decides it on [`ExactSeriesKey`].

33/// The identity semantics *and storage layout* an artifact was written with.
34///
35/// This is the one mandatory discriminator every artifact manifest carries. Artifacts
36/// written under hash-only identity are structurally valid and perfectly ordered, so
37/// nothing else in them distinguishes them from exact ones: an already collapsed series
38/// index looks fine and its lost labels cannot be recovered. A manifest that does not
39/// name this encoding is therefore not reusable, whatever its coverage says.
40///
41/// The serialized value names the layout as well as the identity, because two artifacts can
42/// store the same exact key in incompatible places: the samples format kept one copy per block
43/// inside its pages, then moved to one copy per series in a dedicated key area with a stored
44/// checksum. An artifact of the older layout is structurally valid and ordered too, so without
45/// the layout in this value maintenance would mark it reusable and only a part read would fail.
46#[derive(Debug, Clone, Copy, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
47pub(crate) enum IdentityEncoding {
48    /// Identity is [`ExactSeriesKey`], stored inside the artifact's own pages: the
49    /// series-index layout, and the retired samples layout that repeated the key in every
50    /// block descriptor.
51    #[serde(rename = "exact-sparse-key")]
52    ExactSparseKey,
53    /// Identity is [`ExactSeriesKey`], stored once per series in a part-level key area after
54    /// the column payloads, each copy followed by its checksum.
55    #[serde(rename = "exact-sparse-key-area")]
56    ExactSparseKeyArea,
57}
58
59impl IdentityEncoding {
60    /// The encoding new samples artifacts are written with; their readers require exactly this.
61    ///
62    /// Distinct from the series-index encoding on purpose: it names the key-area layout, so an
63    /// artifact written before the key area cannot be taken for a current one.
64    pub(crate) const CURRENT_SAMPLES: Self = Self::ExactSparseKeyArea;
65
66    /// The encoding new series-index artifacts are written with; their readers require this.
67    pub(crate) const CURRENT_SERIES_INDEX: Self = Self::ExactSparseKey;
68}

70/// The exact identity of one series: its canonical sparse primary-key bytes.
71///
72/// The key is the SST's own encoding of
73/// `(table_id, tsid, label column-id/value pairs sorted by name, non-null only)`,
74/// as produced by [`mito_codec::row_converter::sparse::SparsePrimaryKeyCodec`]. No
75/// second label encoding exists: callers hand in the bytes they read from the SST.
76///
77/// # Ordering
78///
79/// Identity and order are both defined on the encoded bytes only:
80///
81/// * `PartialEq`/`Hash` use the bytes, so two keys are the same series iff their
82///   encodings are equal.
83/// * `Ord` is byte-lexicographic. The reserved `table_id`/`tsid` prefix is fixed
84///   width and order preserving, so this order groups every key that shares one
85///   [`MetricSeriesId`] together and sorts those groups by `MetricSeriesId`.
86///   Inside one group it orders by the encoded `(column_id, value)` label sequence.
87///
88/// This is also the order SSTs store rows in, so artifact writers receive input in
89/// key order and can reject regressions instead of guessing.
90///
91/// `id` is always derived from `key` and never participates in comparison: every
92/// comparison below reads `key` only.
93///
94/// # Assumptions
95///
96/// This is *storage* identity, not Prometheus label-set identity. It relies on the
97/// existing sparse format's invariants, none of which it redefines:
98///
99/// * `tsid` is a stable hash for a label set within one logical table; a rebuild of
100///   the same rows writes the same `tsid`.
101/// * Column IDs are stable for a region, so the encoded `(column_id, value)` pairs
102///   identify the same labels over time. Re-creating a region with different IDs
103///   changes the key bytes.
104/// * Labels are encoded in the codec's canonical order (sorted by column name), and
105///   null labels are omitted while empty strings are encoded. This type never
106///   normalizes either case.
107///
108/// Equal keys therefore mean "the same series as far as this database stores it".
109/// They are not claimed to be equal to any Prometheus series identity.
110#[derive(Clone)]
111pub(crate) struct ExactSeriesKey {
112    /// Order-preserving `(table_id, tsid)` prefix decoded from `key`.
113    id: MetricSeriesId,
114    /// Canonical sparse primary-key encoding of the whole series.
115    key: Box<[u8]>,
116}
117
118impl ExactSeriesKey {
119    /// Wraps one encoded sparse primary key after validating its framing.
120    ///
121    /// # Length
122    ///
123    /// This entry point deliberately has no length cap, and neither does any other:
124    /// key length is bounded by the containers that store a key (one key per block,
125    /// plus the descriptor, page, and column limits), and a cap enforced on only one
126    /// entry point would let a writer publish a key another one refuses. Long labels
127    /// are real data — a single 8 KiB label already encodes to a larger sparse key —
128    /// so framing validation is the only requirement here.
129    pub(crate) fn from_sparse_key(key: &[u8], offsets: &mut SparseOffsetsCache) -> Result<Self> {
130        let mut view = SparsePrimaryKeyView::new(key, offsets).context(DecodeSnafu)?;
131        // `new` validates the reserved prefix only; walk the labels so a truncated or
132        // trailing-garbage key cannot become an identity that happens to share a prefix.
133        view.visit_label_ids(|_| Ok(())).context(DecodeSnafu)?;
134        Ok(Self::from_view(&view, key))
135    }
136
137    /// The reserved `(table_id, tsid)` prefix. It is a coarse bucket, not identity.
138    pub(crate) fn id(&self) -> MetricSeriesId {
139        self.id
140    }
141
142    /// The canonical bytes; the only thing that decides identity.
143    pub(crate) fn bytes(&self) -> &[u8] {
144        &self.key
145    }
146
147    /// Builds the key from a view already validated against `key`.
148    fn from_view(view: &SparsePrimaryKeyView<'_, '_>, key: &[u8]) -> Self {
149        Self {
150            id: MetricSeriesId {
151                table_id: view.table_id(),
152                tsid: view.tsid(),
153            },
154            key: key.into(),
155        }
156    }
157}
158
159// Only `key` participates in identity. `id` is a prefix function of `key`, so a
160// future field cannot accidentally join (or reorder) the identity.
161
162impl PartialEq for ExactSeriesKey {
163    fn eq(&self, other: &Self) -> bool {
164        self.key == other.key
165    }
166}
167
168impl Eq for ExactSeriesKey {}
169
170impl PartialOrd for ExactSeriesKey {
171    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
172        Some(self.cmp(other))
173    }
174}
175
176impl Ord for ExactSeriesKey {
177    fn cmp(&self, other: &Self) -> Ordering {
178        self.key.cmp(&other.key)
179    }
180}
181
182impl fmt::Debug for ExactSeriesKey {
183    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
184        f.debug_struct("ExactSeriesKey")
185            .field("id", &self.id)
186            .field("key", &Hex(&self.key))
187            .finish()
188    }
189}

191/// Lets a map keyed by the exact key be probed with the raw canonical bytes it was
192/// built from, so hot loops do not allocate a key per row. `Ord`/`Eq` are byte
193/// comparisons, which is exactly `[u8]`'s order, so the borrow is consistent.
194impl std::borrow::Borrow<[u8]> for ExactSeriesKey {
195    fn borrow(&self) -> &[u8] {
196        &self.key
197    }
198}
199
200impl Hash for ExactSeriesKey {
201    fn hash<H: Hasher>(&self, state: &mut H) {
202        self.key.hash(state);
203    }
204}

206/// Prints at most [`HEX_LIMIT`] bytes of a key, so logs stay bounded.
207struct Hex<'a>(&'a [u8]);
208
209const HEX_LIMIT: usize = 64;
210
211impl fmt::Debug for Hex<'_> {
212    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
213        for byte in self.0.iter().take(HEX_LIMIT) {
214            write!(f, "{byte:02x}")?;
215        }
216        if self.0.len() > HEX_LIMIT {
217            write!(f, "...({} bytes)", self.0.len())?;
218        }
219        Ok(())
220    }
221}

223/// Decodes one sparse primary key into its exact identity and `tag_columns` labels.
224///
225/// This is the only decoder artifact writers use, so the identity they aggregate and
226/// the labels they persist always come from the same key.
227pub(crate) fn decode_series_identity(
228    primary_key: &[u8],
229    tag_columns: &[(ColumnId, String)],
230    offsets: &mut SparseOffsetsCache,
231    buf: &mut Vec<u8>,
232) -> Result<(ExactSeriesKey, Vec<Option<String>>)> {
233    let mut view = SparsePrimaryKeyView::new(primary_key, offsets).context(DecodeSnafu)?;
234    // Validate every label once and cache its offset; the tag lookups below reuse it,
235    // and a truncated key never reaches a persisted row.
236    view.visit_label_ids(|_| Ok(())).context(DecodeSnafu)?;
237    let key = ExactSeriesKey::from_view(&view, primary_key);
238    let mut tags = Vec::with_capacity(tag_columns.len());
239    for (column_id, _) in tag_columns {
240        tags.push(
241            view.label(*column_id, buf)
242                .context(DecodeSnafu)?
243                .map(str::to_owned),
244        );
245    }
246    Ok((key, tags))
247}
248
249/// Decodes only the reserved `(table_id, tsid)` prefix, without allocating the key.
250///
251/// Callers that already compared the raw bytes must not use this to decide identity;
252/// it only exists where the coarse bucket is the output.
253pub(crate) fn decode_series_id(
254    primary_key: &[u8],
255    offsets: &mut SparseOffsetsCache,
256) -> Result<MetricSeriesId> {
257    let view = SparsePrimaryKeyView::new(primary_key, offsets).context(DecodeSnafu)?;
258    Ok(MetricSeriesId {
259        table_id: view.table_id(),
260        tsid: view.tsid(),
261    })
262}

281    #[test]
282    fn identity_is_the_exact_key_not_the_tsid() {
283        let mut offsets = SparseOffsetsCache::new();
284        let first =
285            ExactSeriesKey::from_sparse_key(&key(&["host-020078", "z"], 1, 7), &mut offsets)
286                .unwrap();
287        let partner =
288            ExactSeriesKey::from_sparse_key(&key(&["host-029838", "z"], 1, 7), &mut offsets)
289                .unwrap();
290        assert_eq!(first.id(), partner.id());
291        assert_ne!(first, partner);
292        assert_ne!(first.bytes(), partner.bytes());
293        let same = ExactSeriesKey::from_sparse_key(&key(&["host-020078", "z"], 1, 7), &mut offsets)
294            .unwrap();
295        assert_eq!(first, same);
296    }

298    #[test]
299    fn order_groups_equal_identifiers_and_sorts_groups_by_identifier() {
300        let mut offsets = SparseOffsetsCache::new();
301        let mut keys = Vec::new();
302        for (table_id, tsid, label) in [
303            (1, 2, "b"),
304            (1, 2, "a"),
305            (1, 3, "a"),
306            (2, 1, "a"),
307            (1, 2, "c"),
308        ] {
309            keys.push(
310                ExactSeriesKey::from_sparse_key(&key(&[label, "z"], table_id, tsid), &mut offsets)
311                    .unwrap(),
312            );
313        }
314        let mut sorted = keys.clone();
315        sorted.sort();
316        // Byte order sorts by (table_id, tsid) first, then by the encoded labels, and
317        // every key sharing an identifier stays adjacent.
318        let ids: Vec<_> = sorted.iter().map(|key| key.id()).collect();
319        assert_eq!(
320            ids,
321            vec![
322                MetricSeriesId {
323                    table_id: 1,
324                    tsid: 2
325                },
326                MetricSeriesId {
327                    table_id: 1,
328                    tsid: 2
329                },
330                MetricSeriesId {
331                    table_id: 1,
332                    tsid: 2
333                },
334                MetricSeriesId {
335                    table_id: 1,
336                    tsid: 3
337                },
338                MetricSeriesId {
339                    table_id: 2,
340                    tsid: 1
341                },
342            ]
343        );
344        // Ordering is decided on the encoded bytes alone.
345        for pair in keys.windows(2) {
346            assert_eq!(pair[0].cmp(&pair[1]), pair[0].bytes().cmp(pair[1].bytes()));
347        }
348        assert!(sorted.is_sorted());
349    }

351    #[test]
352    fn long_keys_are_identity_not_an_error() {
353        let mut offsets = SparseOffsetsCache::new();
354        // One 8 KiB label already encodes to a sparse key past 8 KiB. Both entry points
355        // must accept it: the containers that store a key bound its length, and a cap on
356        // one entry point alone would publish keys another refuses.
357        let label = "x".repeat(8 * 1024);
358        let encoded = key(&[&label, "z"], 1, 7);
359        assert!(encoded.len() > 8 * 1024, "{}", encoded.len());
360        let exact = ExactSeriesKey::from_sparse_key(&encoded, &mut offsets).unwrap();
361        assert_eq!(encoded, exact.bytes());
362        let tag_columns = [(0, "tag_0".to_string()), (1, "tag_1".to_string())];
363        let (decoded, tags) =
364            decode_series_identity(&encoded, &tag_columns, &mut offsets, &mut Vec::new()).unwrap();
365        assert_eq!(exact, decoded);
366        assert_eq!(Some(label), tags[0]);
367    }

369    #[test]
370    fn malformed_keys_are_rejected() {
371        let mut offsets = SparseOffsetsCache::new();
372        assert!(ExactSeriesKey::from_sparse_key(&[], &mut offsets).is_err());
373        assert!(ExactSeriesKey::from_sparse_key(&[0; 13], &mut offsets).is_err());
374        let good = key(&["a", "b"], 1, 1);
375        assert!(ExactSeriesKey::from_sparse_key(&good[..good.len() - 1], &mut offsets).is_err());
376        assert_eq!(
377            decode_series_id(&good, &mut offsets).unwrap(),
378            MetricSeriesId {
379                table_id: 1,
380                tsid: 1
381            }
382        );
383    }
