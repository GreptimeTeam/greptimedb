53    Reconcile {
54        region: MitoRegionRef,
55        sender: oneshot::Sender<Result<ReconcileStats>>,
56    },

60/// Shared lifecycle state and command sender for a worker's series-index task.
61#[derive(Debug)]
62pub(crate) struct SeriesIndexTaskState {
63    worker_id: u32,
64    running: AtomicBool,
65    sender: Sender<SeriesIndexCommand>,
66    pub(crate) worker: Option<Sender<WorkerRequestWithTime>>,
67}
68
69impl SeriesIndexTaskState {
70    pub(crate) fn new(worker_id: u32, channel_size: usize) -> (Self, Receiver<SeriesIndexCommand>) {
71        let (sender, receiver) = mpsc::channel(channel_size);
72        (
73            Self {
74                worker_id,
75                running: AtomicBool::new(true),
76                sender,
77                worker: None,
78            },
79            receiver,
80        )
81    }
82
83    pub(crate) fn is_running(&self) -> bool {
84        self.running.load(Ordering::Acquire)
85    }
86
87    pub(crate) fn wake(&self) {
88        // Queued work already wakes the task. Periodic maintenance still runs even
89        // if the queued command only reconciles one region.
90        if self.is_running() && self.sender.capacity() == self.sender.max_capacity() {
91            let _ = self.sender.try_send(SeriesIndexCommand::Wake);
92        }
93    }
94
95    /// Returns a receiver for build completion, or RegionBusy if the queue is full.
96    pub(crate) fn try_reconcile(
97        &self,
98        region: MitoRegionRef,
99    ) -> Result<oneshot::Receiver<Result<ReconcileStats>>> {
100        ensure!(self.is_running(), WorkerStoppedSnafu { id: self.worker_id });
101        let region_id = region.region_id;
102        let (sender, receiver) = oneshot::channel();
103        self.sender
104            .try_send(SeriesIndexCommand::Reconcile { region, sender })
105            .map_err(|error| match error {
106                mpsc::error::TrySendError::Full(_) => RegionBusySnafu { region_id }.build(),
107                mpsc::error::TrySendError::Closed(_) => {
108                    WorkerStoppedSnafu { id: self.worker_id }.build()
109                }
110            })?;
111        Ok(receiver)
112    }
113
114    pub(crate) fn stop(&self) {
115        if self.running.swap(false, Ordering::AcqRel) {
116            // A full queue already wakes the task, which checks the running flag.
117            let _ = self.sender.try_send(SeriesIndexCommand::Stop);
118        }
119    }
120}

192const MERGE_FAN_IN: usize = 4;
193
194/// Default selection order across sequence-adjacent groups.
195#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
196struct MergeRank {
197    total_bytes: u128,
198    window_start: i64,
199    file_ids: [FileId; MERGE_FAN_IN],
200}
201
202struct IndexMerge {
203    rank: MergeRank,
204    inputs: [FileHandle; MERGE_FAN_IN],
205    window: WindowSequence,
206    expected_watermark: u64,
207}
208
209/// Select the cheapest balanced run across all persisted windows.
210fn pick_index_merge<'a>(
211    files: impl Iterator<Item = &'a FileHandle>,
212    after: Option<MergeRank>,
213) -> Option<IndexMerge> {
214    // A permanently unreadable group must not monopolize every pass. Visit
215    // later eligible groups first, then wrap around so repaired inputs retry.
216    let priority = |rank: MergeRank| (after.is_some_and(|last| rank <= last), rank);
217    let mut windows = HashMap::new();
218    for file in files {
219        if let Some(window) = file.meta_ref().series_window() {
220            windows
221                .entry((window.start, window.end))
222                .or_insert_with(Vec::new)
223                .push((file, window.max_sequence));
224        }
225    }
226    let mut best: Option<IndexMerge> = None;
227    for ((start, end), mut files) in windows {
228        let expected_watermark = files.iter().map(|(_, seq)| *seq).max().unwrap_or_default();
229        files.sort_unstable_by_key(|(f, seq)| (*seq, f.meta_ref().file_id));
230        for group in files.windows(MERGE_FAN_IN) {
231            let total = group
232                .iter()
233                .map(|(f, _)| u128::from(f.meta_ref().file_size))
234                .sum();
235            let largest = group
236                .iter()
237                .map(|(f, _)| f.meta_ref().file_size)
238                .max()
239                .unwrap_or_default();
240            if !crate::compaction::bytes_balanced(largest.into(), total) {
241                continue;
242            }
243            let rank = MergeRank {
244                total_bytes: total,
245                window_start: start,
246                file_ids: std::array::from_fn(|i| group[i].0.meta_ref().file_id),
247            };
248            if best
249                .as_ref()
250                .is_none_or(|previous| priority(rank) < priority(previous.rank))
251            {
252                best = Some(IndexMerge {
253                    rank,
254                    inputs: std::array::from_fn(|i| group[i].0.clone()),
255                    window: WindowSequence {
256                        start,
257                        end,
258                        max_sequence: group[MERGE_FAN_IN - 1].1,
259                    },
260                    expected_watermark,
261                });
262            }
263        }
264    }
265    best
266}

268impl SeriesIndexTask {
269    /// Runs periodic maintenance until the worker stops.
270    async fn run(mut self) {
271        let worker_id = self.worker_id;
272        info!("Start series-index background task, worker: {worker_id}");
273        let interval = self.time_provider.wait_duration(self.interval);
274        let mut deadline = Instant::now() + interval;
275        while self.state.is_running() {
276            // Check before receiving so a busy channel cannot starve periodic maintenance.
277            if Instant::now() >= deadline {
278                self.maintain().await;
279                deadline = Instant::now() + interval;
280                continue;
281            }
282            match tokio::time::timeout_at(deadline, self.receiver.recv()).await {
283                Err(_) => continue,
284                Ok(None | Some(SeriesIndexCommand::Stop)) => break,
285                Ok(Some(SeriesIndexCommand::Wake)) => {
286                    if self.state.is_running() {
287                        self.maintain().await;
288                        deadline = Instant::now() + interval;
289                    }
290                }
291                Ok(Some(SeriesIndexCommand::Reconcile { region, sender })) => {
292                    let result = self.reconcile_manual(region).await;
293                    let _ = sender.send(result);
294                }
295            }
296        }
297        self.state.stop();
298        self.receiver.close();
299        while let Some(command) = self.receiver.recv().await {
300            if let SeriesIndexCommand::Reconcile { sender, .. } = command {
301                let _ = sender.send(WorkerStoppedSnafu { id: worker_id }.fail());
302            }
303        }
304        info!("Stop series-index background task, worker: {worker_id}");
305    }

344    async fn reconcile_region_with_mode(
345        &mut self,
346        region: MitoRegionRef,
347        manual: bool,
348    ) -> Result<ReconcileStats> {
349        if region.version().options.sst_format == Some(crate::sst::FormatType::Narthex) {
350            if region.state() != RegionRoleState::Leader(RegionLeaderState::Writable) {
351                return Ok(ReconcileStats::default());
352            }
353            return self.reconcile_native(region, manual).await;
354        }
355        // Full capacity defers builds, but cleanup must still reclaim published usage.
356        let allow_builds = self.disk_usage.load(Ordering::Relaxed) < self.max_size;
357        if !allow_builds {
358            SERIES_INDEX_CAPACITY_DEFERRED.inc();
359        }
360        let result = reconcile_series_indexes_with_mode(
361            self.worker_id,
362            self.store.clone(),
363            region,
364            self.bucket_width,
365            self.time_provider.current_time_millis(),
366            self.purger.clone(),
367            self.enable_range_index,
368            allow_builds,
369            manual,
370        )
371        .await;
372        if result.is_err() {
373            SERIES_INDEX_RECONCILE_TOTAL
374                .with_label_values(&["failure"])
375                .inc();
376        }
377        result
378    }

380    async fn reconcile_native(
381        &mut self,
382        region: MitoRegionRef,
383        manual: bool,
384    ) -> Result<ReconcileStats> {
385        let mut stats = self.build_and_expire_native(&region, manual).await?;
386        let version = region.version();
387        if !version.pending_tails.is_empty() || !self.state.is_running() {
388            return Ok(stats);
389        }
390        let after = self.failed_merge.remove(&region.region_id);
391        if let Some(IndexMerge {
392            rank,
393            inputs,
394            window,
395            expected_watermark,
396        }) = pick_index_merge(version.ssts.series_indexes().values(), after)
397        {
398            let truncated = version.truncated_entry_id;
399            let metadata = version.metadata.clone();
400            drop(version);
401            let output =
402                crate::sst::narthex::write_series(&region.access_layer, metadata, &inputs, window)
403                    .await
404                    .inspect_err(|_| {
405                        self.failed_merge.insert(region.region_id, rank);
406                    })?;
407            let handle = FileHandle::new(output.clone(), region.file_purger.clone());
408            let edit = crate::manifest::action::RegionEdit {
409                series_indexes_to_add: vec![output],
410                series_indexes_to_remove: inputs.iter().map(FileHandle::file_id).collect(),
411                ..Default::default()
412            };
413            self.publish_native(
414                &region,
415                edit,
416                &[handle],
417                truncated,
418                &HashMap::from([(window.start, expected_watermark)]),
419            )
420            .await?;
421            stats.built_series += 1;
422            stats.removed_series += inputs.len();
423        }
424        Ok(stats)
425    }

427    async fn build_and_expire_native(
428        &self,
429        region: &MitoRegionRef,
430        manual: bool,
431    ) -> Result<ReconcileStats> {
432        let mut stats = ReconcileStats::default();
433        let (ssts, metadata, frontier, truncated_entry_id, options, series_window_seconds) = {
434            let current = region.version_control.current();
435            let version = current.version;
436            if !version.pending_tails.is_empty() {
437                return Ok(stats);
438            }
439            let frontier = if version.memtables.is_empty()
440                && current.last_sequence == current.committed_sequence
441            {
442                current.committed_sequence
443            } else {
444                version.flushed_sequence
445            };
446            (
447                version.ssts.clone(),
448                version.metadata.clone(),
449                frontier,
450                version.truncated_entry_id,
451                version.options.clone(),
452                version.series_window_seconds,
453            )
454        };
455        let ttl = match options.ttl {
456            Some(ttl) => ttl,
457            None => crate::compaction::find_dynamic_options(
458                region.region_id,
459                &options,
460                &self.schema_manager,
461            )
462            .await
463            .map(|(_, ttl)| ttl)
464            .unwrap_or_else(|error| {
465                warn!(error; "Failed to resolve database TTL for region {}", region.region_id);
466                common_time::TimeToLive::Forever
467            }),
468        };
469        let now = common_time::Timestamp::new_millisecond(self.time_provider.current_time_millis());
470        let expired = |f: &FileHandle| ttl.is_expired(&f.time_range().1, &now).unwrap_or(false);
471        let width = series_window_seconds.unwrap_or_else(|| {
472            i64::try_from(self.bucket_width.as_secs())
473                .unwrap_or(i64::MAX)
474                .max(1)
475        });
476        ensure!(
477            width > 0,
478            InvalidRequestSnafu {
479                region_id: region.region_id,
480                reason: "invalid series window width"
481            }
482        );
483        let files = || ssts.levels().iter().flat_map(|l| l.files.values());
484        let live_ranges: Vec<_> = files()
485            .filter(|f| !expired(f))
486            .map(FileHandle::time_range)
487            .collect();
488        let cutoff = match ttl {
489            common_time::TimeToLive::Duration(duration) => now.sub_duration(duration).ok(),
490            _ => None,
491        };
492        // Include expired samples in a partly live window: a later watermark covers
493        // the whole window, including samples still present after a TTL change.
494        let first_live_start = cutoff
495            .and_then(|cut| WindowSequence::starts((cut, cut), width).next())
496            .map(common_time::Timestamp::new_second);
497        let mut edit = crate::manifest::action::RegionEdit::default();
498        let mut watermarks = HashMap::<i64, u64>::new();
499        for file in ssts.series_indexes().values() {
500            if let Some(w) = file.meta_ref().series_window()
501                && !ttl
502                    .is_expired(&common_time::Timestamp::new_second(w.end), &now)
503                    .unwrap_or(false)
504                && live_ranges.iter().any(|range| w.intersects(*range))
505            {
506                let seq = watermarks.entry(w.start).or_default();
507                *seq = (*seq).max(w.max_sequence);
508            } else {
509                edit.series_indexes_to_remove.push(file.file_id());
510            }
511        }
512        let mut handles: Vec<FileHandle> = Vec::new();
513        // Merge the ordered window starts lazily. One wide SST can span many windows,
514        // and a periodic pass only needs the first buildable one.
515        let can_build = !ttl.is_instant() && live_ranges.len() >= self.min_files;
516        let mut windows = files()
517            .filter(|f| can_build && f.meta_ref().sequence.is_some_and(|s| s.get() <= frontier))
518            .map(|f| {
519                let (mut start, end) = f.time_range();
520                if let Some(first) = first_live_start {
521                    start = start.max(first);
522                }
523                let file = f.clone();
524                WindowSequence::starts((start, end), width).map(move |start| (start, file.clone()))
525            })
526            .kmerge_by(|a, b| a.0 < b.0)
527            .peekable();
528        while let Some(&(start, _)) = windows.peek() {
529            let previous = watermarks.get(&start).copied().unwrap_or_default();
530            let mut sources = Vec::new();
531            let mut live_sources = 0;
532            while let Some((_, file)) = windows.next_if(|(next, _)| *next == start) {
533                if file
534                    .meta_ref()
535                    .sequence
536                    .is_some_and(|seq| seq.get() > previous)
537                {
538                    live_sources += usize::from(!expired(&file));
539                    sources.push(file);
540                }
541            }
542            if live_sources < self.min_files {
543                continue;
544            }
545            let window = WindowSequence {
546                start,
547                end: start + width,
548                max_sequence: frontier,
549            };
550            let output = match crate::sst::narthex::write_series(
551                &region.access_layer,
552                metadata.clone(),
553                &sources,
554                window,
555            )
556            .await
557            {
558                Ok(output) => output,
559                Err(error) => {
560                    for handle in handles {
561                        handle.mark_deleted();
562                    }
563                    return Err(error);
564                }
565            };
566            handles.push(FileHandle::new(output.clone(), region.file_purger.clone()));
567            edit.series_indexes_to_add.push(output);
568            stats.source_files += sources.len();
569            if !manual {
570                break;
571            }
572        }
573        stats.built_series = handles.len();
574        stats.removed_series = edit.series_indexes_to_remove.len();
575        if stats.built_series + stats.removed_series > 0 {
576            self.publish_native(region, edit, &handles, truncated_entry_id, &watermarks)
577                .await?;
578        }
579        Ok(stats)
580    }

582    async fn publish_native(
583        &self,
584        region: &MitoRegionRef,
585        edit: crate::manifest::action::RegionEdit,
586        handles: &[FileHandle],
587        truncated_entry_id: Option<crate::wal::EntryId>,
588        watermarks: &HashMap<i64, u64>,
589    ) -> Result<()> {
590        let Some(worker) = self
591            .state
592            .worker
593            .as_ref()
594            .filter(|_| self.state.is_running())
595        else {
596            for handle in handles {
597                handle.mark_deleted();
598            }
599            return WorkerStoppedSnafu { id: self.worker_id }.fail();
600        };
601        let (tx, rx) = oneshot::channel();
602        let result = region
603            .manifest_ctx
604            .publish_series_indexes(&edit, truncated_entry_id, watermarks)
605            .await;
606        if let Err(error) = &result
607            && (matches!(error, crate::error::Error::RegionBusy { .. })
608                || !error.may_have_persisted_manifest_update())
609        {
610            for handle in handles {
611                handle.mark_deleted();
612            }
613        }
614        let notification = WorkerRequest::Background {
615            region_id: region.region_id,
616            notify: BackgroundNotify::RegionEdit(RegionEditResult {
617                region_id: region.region_id,
618                origin: region.clone(),
619                edit,
620                result: result
621                    .map(|()| RegionEditOutcome::Written)
622                    .map_err(Arc::new),
623                waiters: Waiters::one(tx),
624                is_staging: false,
625                update_region_state: false,
626            }),
627        };
628        worker
629            .send(WorkerRequestWithTime::new(notification))
630            .await
631            .map_err(|_| WorkerStoppedSnafu { id: self.worker_id }.build())?;
632        rx.await
633            .map_err(|_| WorkerStoppedSnafu { id: self.worker_id }.build())??;
634        Ok(())
635    }

637    /// Each worker contributes only its open regions, refreshed at maintenance boundaries.
638    fn refresh_usage(&mut self) {
639        let usage = self
640            .regions
641            .list_regions()
642            .iter()
643            .map(|region| region.series_index_version().disk_usage())
644            .sum();
645        self.report_usage(usage);

648    fn report_usage(&mut self, usage: u64) {
649        if usage >= self.reported_usage {
650            let delta = usage - self.reported_usage;
651            self.disk_usage.fetch_add(delta, Ordering::Relaxed);
652            SERIES_INDEX_DISK_BYTES.add(delta as i64);
653        } else {
654            let delta = self.reported_usage - usage;
655            self.disk_usage.fetch_sub(delta, Ordering::Relaxed);
656            SERIES_INDEX_DISK_BYTES.sub(delta as i64);
657        }
658        self.reported_usage = usage;
659    }

661    /// Runs periodic maintenance independently of incoming deletion requests.
662    async fn maintain(&mut self) {
663        self.failed_merge
664            .retain(|id, _| self.regions.get_region(*id).is_some());
665        self.refresh_usage();
666        for region in self.regions.list_regions() {
667            if !self.state.is_running() {
668                break;
669            }
670            // Best effort: the region can still change state during reconciliation.
671            // Local indexes can be built on followers as well as writable leaders.
672            if !matches!(
673                region.state(),
674                RegionRoleState::Follower | RegionRoleState::Leader(RegionLeaderState::Writable)
675            ) {
676                continue;
677            }
678            if let Err(error) = self.reconcile_region(region.clone()).await {
679                warn!(error; "Failed to reconcile series indexes, worker: {}, region: {}", self.worker_id, region.region_id);
680            }
681            self.refresh_usage();
682        }
683    }
