34/// Shared lifecycle state for a worker's series-index task.
35#[derive(Debug)]
36pub(crate) struct SeriesIndexTaskState {
37    running: AtomicBool,
38    notify: Notify,
39}
40
41impl SeriesIndexTaskState {
42    pub(crate) fn new() -> Self {
43        Self {
44            running: AtomicBool::new(true),
45            notify: Notify::new(),
46        }
47    }
48
49    pub(crate) fn is_running(&self) -> bool {
50        self.running.load(Ordering::Acquire)
51    }
52
53    pub(crate) fn wake(&self) {
54        self.notify.notify_one();
55    }
56
57    pub(crate) fn stop(&self) {
58        self.running.store(false, Ordering::Release);
59        // Retain a permit if maintenance has not started waiting yet.
60        self.notify.notify_one();
61    }
62
63    pub(crate) async fn notified(&self) {
64        self.notify.notified().await;
65    }
66}

68/// Starts both tasks on the compaction runtime, detaching purge and returning the maintenance handle.
69#[allow(clippy::too_many_arguments)]
70pub(crate) fn spawn_series_index_tasks(
71    worker_id: u32,
72    store: ObjectStore,
73    regions: RegionMapRef,
74    state: Arc<SeriesIndexTaskState>,
75    bucket_width: Duration,
76    purger: IndexFilePurger,
77    purge_receiver: UnboundedReceiver<PurgeRequest>,
78    interval: Duration,
79    time_provider: TimeProviderRef,
80) -> JoinHandle<()> {
81    // Snapshots may retain senders after the worker stops; purge until all senders drop.
82    common_runtime::spawn_compact(run_index_purge_task(
83        worker_id,
84        store.clone(),
85        purge_receiver,
86    ));
87    common_runtime::spawn_compact(async move {
88        SeriesIndexTask {
89            worker_id,
90            store,
91            regions,
92            bucket_width,
93            purger,
94            state,
95            interval,
96            time_provider,
97        }
98        .run()
99        .await;
100    })
101}

103/// Periodic series-index maintenance for one region worker.
104struct SeriesIndexTask {
105    store: ObjectStore,
106    regions: RegionMapRef,
107    bucket_width: Duration,
108    purger: IndexFilePurger,
109    worker_id: u32,
110    state: Arc<SeriesIndexTaskState>,
111    interval: Duration,
112    time_provider: TimeProviderRef,
113}
114
115impl SeriesIndexTask {
116    /// Runs periodic maintenance until the worker stops.
117    async fn run(mut self) {
118        let worker_id = self.worker_id;
119        info!("Start series-index background task, worker: {worker_id}");
120        let interval = self.time_provider.wait_duration(self.interval);
121        let mut timer = tokio::time::interval_at(Instant::now() + interval, interval);
122        // Schedule future ticks from a late tick rather than the original cadence.
123        timer.set_missed_tick_behavior(MissedTickBehavior::Delay);
124        while self.state.is_running() {
125            tokio::select! {
126                _ = self.state.notified() => {}
127                _ = timer.tick() => {}
128            }
129            if self.state.is_running() {
130                self.maintain().await;
131            }
132        }
133        info!("Stop series-index background task, worker: {worker_id}");
134    }
135
136    /// Runs periodic maintenance independently of incoming deletion requests.
137    async fn maintain(&mut self) {
138        for region in self.regions.list_regions() {
139            if !self.state.is_running() {
140                break;
141            }
142            // Best effort: the region can still change state during reconciliation.
143            // Local indexes can be built on followers as well as writable leaders.
144            if !matches!(
145                region.state(),
146                RegionRoleState::Follower | RegionRoleState::Leader(RegionLeaderState::Writable)
147            ) {
148                continue;
149            }
150            if let Err(error) = reconcile_series_indexes(
151                self.worker_id,
152                self.store.clone(),
153                region.clone(),
154                self.bucket_width,
155                self.time_provider.current_time_millis(),
156                self.purger.clone(),
157            )
158            .await
159            {
160                SERIES_INDEX_RECONCILE_TOTAL
161                    .with_label_values(&["failure"])
162                    .inc();
163                warn!(error; "Failed to reconcile series indexes, worker: {}, region: {}", self.worker_id, region.region_id);
164            }
165        }
166    }
167}

169#[cfg(test)]
170mod tests {
171    use object_store::services::Memory;
172    use store_api::region_engine::RegionEngine;
173
174    use super::*;
175    use crate::region::RegionMap;
176    use crate::series_index::catalog::{range_catalog_path, series_catalog_path};
177    use crate::series_index::purger::series_index_channel;
178    use crate::series_index::tests::prepare_region;
179    use crate::test_util::TestEnv;
180
181    #[rstest::rstest]
182    #[case::follower(RegionRoleState::Follower, true)]
183    #[case::writable(RegionRoleState::Leader(RegionLeaderState::Writable), true)]
184    #[case::staging(RegionRoleState::Leader(RegionLeaderState::Staging), false)]
185    #[case::entering_staging(RegionRoleState::Leader(RegionLeaderState::EnteringStaging), false)]
186    #[case::altering(RegionRoleState::Leader(RegionLeaderState::Altering), false)]
187    #[case::dropping(RegionRoleState::Leader(RegionLeaderState::Dropping), false)]
188    #[case::truncating(RegionRoleState::Leader(RegionLeaderState::Truncating), false)]
189    #[case::editing(RegionRoleState::Leader(RegionLeaderState::Editing), false)]
190    #[case::downgrading(RegionRoleState::Leader(RegionLeaderState::Downgrading), false)]
191    #[tokio::test]
192    async fn test_maintenance_region_states(#[case] role: RegionRoleState, #[case] builds: bool) {
193        let mut env = TestEnv::with_prefix("series-maintenance-state").await;
194        let (engine, region) = prepare_region(&mut env).await;
195        // Install the desired state without triggering the corresponding DDL.
196        region.switch_state_to_staging(RegionLeaderState::Writable);
197        region
198            .manifest_ctx
199            .exit_staging(region.region_id, role)
200            .unwrap();
201        let store = ObjectStore::new(Memory::default()).unwrap();
202        let (purger, _receiver) = series_index_channel(store.clone());
203        let regions = Arc::new(RegionMap::default());
204        regions.insert_region(region.clone());
205        let mut task = SeriesIndexTask {
206            store: store.clone(),
207            regions,
208            bucket_width: Duration::from_secs(100),
209            purger,
210            worker_id: 0,
211            state: Arc::new(SeriesIndexTaskState::new()),
212            interval: Duration::from_secs(3600),
213            time_provider: Arc::new(crate::time_provider::StdTimeProvider),
214        };
215        task.maintain().await;
216        assert_eq!(
217            builds,
218            !region.series_index_version().series_indexes.is_empty()
219        );
220        for path in [
221            range_catalog_path(region.region_id),
222            series_catalog_path(region.region_id),
223        ] {
224            assert_eq!(builds, store.exists(&path).await.unwrap());
225        }
226        engine.stop().await.unwrap();
227    }
