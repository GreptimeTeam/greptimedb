15//! Owned outputs shared by immutable columnar artifacts.
16
17use bytes::Bytes;
18use object_store::{ObjectStore, Writer};
19use snafu::{IntoError, ResultExt};
20
21use crate::error::{JoinSnafu, OpenDalSnafu, Result};
22use crate::series_index::columnar::{invalid, require};
23use crate::sst::{DEFAULT_WRITE_BUFFER_SIZE, DEFAULT_WRITE_CONCURRENCY};
24
25/// Drop owns the live OpenDAL writer, including its private atomic-write temporary file.
26pub(super) struct LiveFile {
27    writer: Option<Writer>,
28    store: ObjectStore,
29    path: String,
30}
31impl Drop for LiveFile {
32    fn drop(&mut self) {
33        if let Some(mut writer) = self.writer.take() {
34            let store = self.store.clone();
35            let path = self.path.clone();
36            common_runtime::spawn_compact(async move {
37                if let Err(error) = writer.abort().await {
38                    common_telemetry::warn!(error; "Aborting artifact output {path}");
39                }
40                if let Err(error) = store.delete(&path).await {
41                    common_telemetry::warn!(error; "Deleting artifact output {path}");
42                }
43            });
44        }
45    }
46}
47
48pub(super) struct Outputs {
49    pub(super) store: ObjectStore,
50    pub(super) active: Option<LiveFile>,
51    pub(super) paths: Vec<String>,
52    pub(super) committed: bool,
53}
54impl Outputs {
55    pub(super) async fn open(&mut self, path: String) -> Result<()> {
56        require(self.active.is_none(), "artifact output already open")?;
57        self.paths.push(path.clone());
58        let store = self.store.clone();
59        // The task owns opening: cancellation drops the returned LiveFile and aborts it.
60        self.active = Some(
61            common_runtime::spawn_compact(async move {
62                let writer = store
63                    .writer_with(&path)
64                    .chunk(DEFAULT_WRITE_BUFFER_SIZE.as_bytes() as usize)
65                    .concurrent(DEFAULT_WRITE_CONCURRENCY)
66                    .await
67                    .context(OpenDalSnafu)?;
68                Ok::<_, crate::error::Error>(LiveFile {
69                    writer: Some(writer),
70                    store,
71                    path,
72                })
73            })
74            .await
75            .context(JoinSnafu)??,
76        );
77        Ok(())
78    }
79    pub(super) async fn write(&mut self, bytes: Vec<u8>) -> Result<()> {
80        self.active
81            .as_mut()
82            .and_then(|f| f.writer.as_mut())
83            .ok_or_else(|| invalid("artifact output is not open"))?
84            .write(Bytes::from(bytes))
85            .await
86            .context(OpenDalSnafu)?;
87        Ok(())
88    }
89    pub(super) async fn close(&mut self) -> Result<()> {
90        let file = self
91            .active
92            .as_mut()
93            .ok_or_else(|| invalid("artifact output is not open"))?;
94        file.writer
95            .as_mut()
96            .ok_or_else(|| invalid("artifact output is closed"))?
97            .close()
98            .await
99            .context(OpenDalSnafu)?;
100        file.writer = None;
101        self.active = None;
102        Ok(())
103    }
104    pub(super) async fn cleanup(&mut self) -> Result<()> {
105        let mut first_error = None;
106        if let Some(file) = self.active.as_mut()
107            && let Some(writer) = file.writer.as_mut()
108        {
109            match writer.abort().await {
110                Ok(()) => file.writer = None,
111                Err(error) => first_error = Some(OpenDalSnafu.into_error(error)),
112            }
113        }
114        self.active = None;
115        for path in &self.paths {
116            if let Err(error) = self.store.delete(path).await {
117                first_error.get_or_insert_with(|| OpenDalSnafu.into_error(error));
118            }
119        }
120        if let Some(error) = first_error {
121            Err(error)
122        } else {
123            self.committed = true;
124            Ok(())
125        }
126    }
127}
128impl Drop for Outputs {
129    fn drop(&mut self) {
130        if !self.committed {
131            let mut owned = Self {
132                store: self.store.clone(),
133                active: self.active.take(),
134                paths: std::mem::take(&mut self.paths),
135                committed: true,
136            };
137            common_runtime::spawn_compact(async move {
138                if let Err(error) = owned.cleanup().await {
139                    common_telemetry::warn!(error; "Cleaning unpublished format artifact");
140                }
141            });
142        }
143    }
144}
