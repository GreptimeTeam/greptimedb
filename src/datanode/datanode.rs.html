95impl Datanode {
96    pub async fn start(&mut self) -> Result<()> {
97        info!("Starting datanode instance...");
98
99        self.start_heartbeat().await?;
100        self.wait_coordinated().await;
101
102        self.start_telemetry();
103
104        self.services.start_all().await.context(StartServerSnafu)
105    }
106
107    pub fn server_handlers(&self) -> &ServerHandlers {
108        &self.services
109    }
110
111    pub fn start_telemetry(&self) {
112        if let Err(e) = self.greptimedb_telemetry_task.start() {
113            warn!(e; "Failed to start telemetry task!");
114        }
115    }
116
117    pub async fn start_heartbeat(&mut self) -> Result<()> {
118        if let Some(task) = &self.heartbeat_task {
119            // Safety: The event_receiver must exist.
120            let receiver = self.region_event_receiver.take().unwrap();
121
122            task.start(receiver, self.leases_notifier.clone()).await?;
123        }
124        Ok(())
125    }
126
127    /// If `leases_notifier` exists, it waits until leases have been obtained in all regions.
128    pub async fn wait_coordinated(&mut self) {
129        if let Some(notifier) = self.leases_notifier.take() {
130            notifier.notified().await;
131        }
132    }
133
134    pub fn setup_services(&mut self, services: ServerHandlers) {
135        self.services = services;
136    }
137
138    pub async fn shutdown(&mut self) -> Result<()> {
139        self.services
140            .shutdown_all()
141            .await
142            .context(ShutdownServerSnafu)?;
143
144        let _ = self.greptimedb_telemetry_task.stop().await;
145        if let Some(heartbeat_task) = &self.heartbeat_task {
146            heartbeat_task
147                .close()
148                .map_err(BoxedError::new)
149                .context(ShutdownInstanceSnafu)?;
150        }
151        self.region_server.stop().await?;
152        Ok(())
153    }
154
155    pub fn region_server(&self) -> RegionServer {
156        self.region_server.clone()
157    }
158
159    pub fn plugins(&self) -> Plugins {
160        self.plugins.clone()
161    }

258    pub async fn build(mut self) -> Result<Datanode> {
259        validate_region_engine_config(&self.opts.region_engine)?;
260        let node_id = self.opts.node_id.context(MissingNodeIdSnafu)?;
261        set_default_prefix(self.opts.default_column_prefix.as_deref())
262            .map_err(BoxedError::new)
263            .context(BuildDatanodeSnafu)?;
264
265        let meta_client = self.meta_client.take();
266
267        // If metasrv client is provided, we will use it to control the region server.
268        // Otherwise the region server is self-controlled, meaning no heartbeat and immediately
269        // writable upon open.
270        let controlled_by_metasrv = meta_client.is_some();
271        if let DatanodeWalConfig::ObjectStore(config) = &self.opts.wal {
272            validate_object_store_wal_config(config)?;
273        }
274
275        // build and initialize region server
276        let (region_event_listener, region_event_receiver) = if controlled_by_metasrv {
277            let (tx, rx) = new_region_server_event_channel();
278            (Box::new(tx) as _, Some(rx))
279        } else {
280            (Box::new(NoopRegionServerEventListener) as _, None)
281        };
282
283        let cache_registry = self.cache_registry.take().context(MissingCacheSnafu)?;
284        let schema_cache: SchemaCacheRef = cache_registry.get().context(MissingCacheSnafu)?;
285        let table_id_schema_cache: TableSchemaCacheRef =
286            cache_registry.get().context(MissingCacheSnafu)?;
287
288        let schema_metadata_manager = Arc::new(SchemaMetadataManager::new(
289            table_id_schema_cache,
290            schema_cache,
291        ));
292
293        let gc_enabled = self.opts.region_engine.iter().any(|engine| {
294            if let RegionEngineConfig::Mito(config) = engine {
295                config.gc.enable
296            } else {
297                false
298            }
299        });
300
301        let file_ref_manager = Arc::new(FileReferenceManager::with_gc_enabled(
302            Some(node_id),
303            gc_enabled,
304        ));
305        let region_server = self
306            .new_region_server(
307                schema_metadata_manager,
308                region_event_listener,
309                file_ref_manager,
310            )
311            .await?;
312
313        // TODO(weny): Considering introducing a readonly kv_backend trait.
314        let runtime_switch_manager = RuntimeSwitchManager::new(self.kv_backend.clone());
315        let is_recovery_mode = runtime_switch_manager
316            .recovery_mode()
317            .await
318            .context(GetMetadataSnafu)?;
319
320        let region_open_requests =
321            build_region_open_requests(node_id, self.kv_backend.clone()).await?;
322        let open_with_writable = self
323            .open_regions_writable_override
324            .unwrap_or(!controlled_by_metasrv);
325        let open_all_regions = open_all_regions(
326            region_server.clone(),
327            region_open_requests,
328            open_with_writable,
329            self.opts.init_regions_parallelism,
330            // Ignore nonexistent regions in recovery mode.
331            is_recovery_mode,
332        );
333
334        if self.opts.init_regions_in_background {
335            // Opens regions in background.
336            common_runtime::spawn_global(async move {
337                if let Err(err) = open_all_regions.await {
338                    error!(err; "Failed to open regions during the startup.");
339                }
340            });
341        } else {
342            open_all_regions.await?;
343        }
344
345        let heartbeat_task = if let Some(meta_client) = meta_client {
346            let task = self
347                .create_heartbeat_task(&region_server, meta_client, cache_registry)
348                .await?;
349            Some(task)
350        } else {
351            None
352        };
353
354        let is_standalone = heartbeat_task.is_none();
355        let greptimedb_telemetry_task = get_greptimedb_telemetry_task(
356            Some(self.opts.storage.data_home.clone()),
357            is_standalone && self.opts.enable_telemetry,
358        )
359        .await;
360
361        let leases_notifier = if self.opts.require_lease_before_startup && !is_standalone {
362            Some(Arc::new(Notify::new()))
363        } else {
364            None
365        };
366
367        Ok(Datanode {
368            services: ServerHandlers::default(),
369            heartbeat_task,
370            region_server,
371            greptimedb_telemetry_task,
372            region_event_receiver,
373            leases_notifier,
374            plugins: self.plugins.clone(),
375        })
376    }

378    async fn create_heartbeat_task(
379        &self,
380        region_server: &RegionServer,
381        meta_client: MetaClientRef,
382        cache_invalidator: CacheInvalidatorRef,
383    ) -> Result<HeartbeatTask> {
384        let stat = {
385            let mut stat = ResourceStatImpl::default();
386            stat.start_collect_cpu_usage();
387            Arc::new(stat)
388        };
389
390        HeartbeatTask::try_new(
391            &self.opts,
392            region_server.clone(),
393            meta_client,
394            self.kv_backend.clone(),
395            cache_invalidator,
396            self.plugins.clone(),
397            stat,
398        )
399        .await
400    }

402    /// Builds [ObjectStoreManager] from [StorageConfig].
403    pub async fn build_object_store_manager(cfg: &StorageConfig) -> Result<ObjectStoreManagerRef> {
404        let object_store = store::new_object_store(cfg.store.clone(), &cfg.data_home).await?;
405        let default_name = cfg.store.config_name();
406        let mut object_store_manager = ObjectStoreManager::new(default_name, object_store);
407        for store in &cfg.providers {
408            object_store_manager.add(
409                store.config_name(),
410                store::new_object_store(store.clone(), &cfg.data_home).await?,
411            );
412        }
413        Ok(Arc::new(object_store_manager))
414    }

444    async fn new_region_server(
445        &mut self,
446        schema_metadata_manager: SchemaMetadataManagerRef,
447        event_listener: RegionServerEventListenerRef,
448        file_ref_manager: FileReferenceManagerRef,
449    ) -> Result<RegionServer> {
450        let opts: &DatanodeOptions = &self.opts;
451
452        let query_engine_factory = QueryEngineFactory::try_new_with_plugins(
453            // query engine in datanode only executes plan with resolved table source.
454            DummyCatalogManager::arc(),
455            None,
456            None,
457            None,
458            None,
459            None,
460            false,
461            self.plugins.clone(),
462            opts.query.clone(),
463        )
464        .context(DataFusionSnafu)?;
465        let query_engine = query_engine_factory.query_engine();
466
467        let table_provider_factory = self
468            .table_provider_factory
469            .clone()
470            .unwrap_or_else(|| Arc::new(DummyTableProviderFactory));
471        let mut region_server = RegionServer::with_table_provider(
472            query_engine,
473            common_runtime::global_runtime(),
474            event_listener,
475            table_provider_factory,
476            opts.max_concurrent_queries,
477            opts.concurrent_query_limiter_timeout,
478            opts.grpc.flight_compression,
479        );
480        region_server.install_remote_dyn_filter_receiver_injector(&self.plugins);
481
482        let object_store_manager = Self::build_object_store_manager(&opts.storage).await?;
483        let engines = self
484            .build_store_engines(
485                object_store_manager,
486                schema_metadata_manager,
487                file_ref_manager,
488                self.plugins.clone(),
489            )
490            .await?;
491        for engine in engines {
492            region_server.register_engine(engine);
493        }
494        if let Some(topic_stats_reporter) = self.topic_stats_reporter.take() {
495            region_server.set_topic_stats_reporter(topic_stats_reporter);
496        }
497
498        Ok(region_server)
499    }

503    /// Builds [RegionEngineRef] from `store_engine` section in `opts`
504    async fn build_store_engines(
505        &mut self,
506        object_store_manager: ObjectStoreManagerRef,
507        schema_metadata_manager: SchemaMetadataManagerRef,
508        file_ref_manager: FileReferenceManagerRef,
509        plugins: Plugins,
510    ) -> Result<Vec<RegionEngineRef>> {
511        let mut metric_engine_config = metric_engine::config::EngineConfig::default();
512        let mut mito_engine_config = MitoConfig::default();
513        let mut file_engine_config = file_engine::config::EngineConfig::default();
514
515        for engine in &self.opts.region_engine {
516            match engine {
517                RegionEngineConfig::Mito(config) => {
518                    mito_engine_config = config.clone();
519                }
520                RegionEngineConfig::File(config) => {
521                    file_engine_config = config.clone();
522                }
523                RegionEngineConfig::Metric(metric_config) => {
524                    metric_engine_config = metric_config.clone();
525                }
526            }
527        }
528
529        // Build a fetcher to backfill partition_expr on open.
530        let fetcher = Arc::new(MetaPartitionExprFetcher::new(self.kv_backend.clone()));
531        let mito_engine = self
532            .build_mito_engine(
533                object_store_manager.clone(),
534                mito_engine_config,
535                schema_metadata_manager.clone(),
536                file_ref_manager.clone(),
537                fetcher.clone(),
538                plugins.clone(),
539            )
540            .await?;
541
542        let metric_engine = MetricEngine::try_new(mito_engine.clone(), metric_engine_config)
543            .context(BuildMetricEngineSnafu)?;
544
545        let file_engine = FileRegionEngine::new(
546            file_engine_config,
547            object_store_manager.default_object_store().clone(), // TODO: implement custom storage for file engine
548            self.local_file_access.clone(),
549        );
550
551        Ok(vec![
552            Arc::new(mito_engine) as _,
553            Arc::new(metric_engine) as _,
554            Arc::new(file_engine) as _,
555        ])
556    }

558    /// Builds [MitoEngine] according to options.
559    async fn build_mito_engine(
560        &mut self,
561        object_store_manager: ObjectStoreManagerRef,
562        mut config: MitoConfig,
563        schema_metadata_manager: SchemaMetadataManagerRef,
564        file_ref_manager: FileReferenceManagerRef,
565        partition_expr_fetcher: PartitionExprFetcherRef,
566        plugins: Plugins,
567    ) -> Result<MitoEngine> {
568        let opts = &self.opts;
569        if opts.storage.is_object_storage() {
570            // Enable the write cache when setting object storage
571            config.enable_write_cache = true;
572            info!("Configured 'enable_write_cache=true' for mito engine.");
573        }
574
575        let mito_engine = match &opts.wal {
576            DatanodeWalConfig::RaftEngine(raft_engine_config) => {
577                let log_store =
578                    Self::build_raft_engine_log_store(&opts.storage.data_home, raft_engine_config)
579                        .await?;
580
581                let builder = MitoEngineBuilder::new(
582                    &opts.storage.data_home,
583                    config,
584                    log_store,
585                    object_store_manager,
586                    schema_metadata_manager,
587                    file_ref_manager,
588                    partition_expr_fetcher.clone(),
589                    plugins,
590                );
591
592                #[cfg(feature = "enterprise")]
593                let builder = builder.with_extension_range_provider_factory(
594                    self.extension_range_provider_factory.take(),
595                );
596
597                builder.try_build().await.context(BuildMitoEngineSnafu)?
598            }
599            DatanodeWalConfig::Kafka(kafka_config) => {
600                if kafka_config.create_index && opts.node_id.is_none() {
601                    warn!("The WAL index creation only available in distributed mode.")
602                }
603                let global_index_collector = if kafka_config.create_index
604                    && let Some(node_id) = opts.node_id
605                {
606                    let operator = new_object_store_without_cache(
607                        &opts.storage.store,
608                        &opts.storage.data_home,
609                    )
610                    .await?;
611                    let path = default_index_file(node_id);
612                    Some(Self::build_global_index_collector(
613                        kafka_config.dump_index_interval,
614                        operator,
615                        path,
616                    ))
617                } else {
618                    None
619                };
620
621                let log_store =
622                    Self::build_kafka_log_store(kafka_config, global_index_collector).await?;
623                self.topic_stats_reporter = Some(log_store.topic_stats_reporter());
624                let builder = MitoEngineBuilder::new(
625                    &opts.storage.data_home,
626                    config,
627                    log_store,
628                    object_store_manager,
629                    schema_metadata_manager,
630                    file_ref_manager,
631                    partition_expr_fetcher,
632                    plugins,
633                );
634
635                #[cfg(feature = "enterprise")]
636                let builder = builder.with_extension_range_provider_factory(
637                    self.extension_range_provider_factory.take(),
638                );
639
640                builder.try_build().await.context(BuildMitoEngineSnafu)?
641            }
642            DatanodeWalConfig::Noop => {
643                let log_store = Arc::new(NoopLogStore);
644
645                let builder = MitoEngineBuilder::new(
646                    &opts.storage.data_home,
647                    config,
648                    log_store,
649                    object_store_manager,
650                    schema_metadata_manager,
651                    file_ref_manager,
652                    partition_expr_fetcher.clone(),
653                    plugins,
654                );
655
656                #[cfg(feature = "enterprise")]
657                let builder = builder.with_extension_range_provider_factory(
658                    self.extension_range_provider_factory.take(),
659                );
660
661                builder.try_build().await.context(BuildMitoEngineSnafu)?
662            }
663            DatanodeWalConfig::ObjectStore(_) => return ObjectStoreWalNotSupportedSnafu.fail(),
664        };
665        Ok(mito_engine)
666    }

717/// Rejects repeated engine types before their configs can silently overwrite each other.
718fn validate_region_engine_config(configs: &[RegionEngineConfig]) -> Result<()> {
719    let mut engine_indices = HashMap::new();
720    for (index, config) in configs.iter().enumerate() {
721        let engine = match config {
722            RegionEngineConfig::Mito(_) => MITO_ENGINE,
723            RegionEngineConfig::File(_) => FILE_ENGINE,
724            RegionEngineConfig::Metric(_) => METRIC_ENGINE,
725        };
726        if let Some(first_index) = engine_indices.insert(engine, index) {
727            return DuplicateRegionEngineConfigSnafu {
728                engine,
729                first_index,
730                duplicate_index: index,
731            }
732            .fail();
733        }
734    }
735    Ok(())
736}

738/// Rejects an object store WAL config the log store cannot run on.
739fn validate_object_store_wal_config(config: &ObjectStoreWalConfig) -> Result<()> {
740    let prefix = config.prefix.trim();
741    ensure!(
742        !prefix.is_empty(),
743        InvalidObjectStoreWalConfigSnafu {
744            field: "prefix",
745            value: &config.prefix,
746            reason: "must not be empty",
747        }
748    );
749    ensure!(
750        !prefix.starts_with('/'),
751        InvalidObjectStoreWalConfigSnafu {
752            field: "prefix",
753            value: &config.prefix,
754            reason: "must be a relative path",
755        }
756    );
757    ensure!(
758        !prefix
759            .split('/')
760            .any(|component| component.is_empty() || component == ".."),
761        InvalidObjectStoreWalConfigSnafu {
762            field: "prefix",
763            value: &config.prefix,
764            reason: "must not contain empty or `..` components",
765        }
766    );
767    ensure!(
768        config.flush_interval >= Duration::from_millis(10),
769        InvalidObjectStoreWalConfigSnafu {
770            field: "flush_interval",
771            value: format!("{:?}", config.flush_interval),
772            reason: "must be at least 10ms",
773        }
774    );
775    ensure!(
776        config.max_batch_bytes.as_bytes() > 0,
777        InvalidObjectStoreWalConfigSnafu {
778            field: "max_batch_bytes",
779            value: config.max_batch_bytes.to_string(),
780            reason: "must be greater than 0",
781        }
782    );
783    Ok(())
784}

786/// Open all regions belong to this datanode.
787async fn open_all_regions(
788    region_server: RegionServer,
789    region_open_requests: RegionOpenRequests,
790    open_with_writable: bool,
791    init_regions_parallelism: usize,
792    ignore_nonexistent_region: bool,
793) -> Result<()> {
794    let RegionOpenRequests {
795        leader_regions,
796        #[cfg(feature = "enterprise")]
797        follower_regions,
798    } = region_open_requests;
799
800    let leader_region_num = leader_regions.len();
801    info!("going to open {} region(s)", leader_region_num);
802    let now = Instant::now();
803    let open_regions = region_server
804        .handle_batch_open_requests(
805            init_regions_parallelism,
806            leader_regions,
807            ignore_nonexistent_region,
808        )
809        .await?;
810    info!(
811        "Opened {} regions in {:?}",
812        open_regions.len(),
813        now.elapsed()
814    );
815    if !ignore_nonexistent_region {
816        ensure!(
817            open_regions.len() == leader_region_num,
818            error::UnexpectedSnafu {
819                violated: format!(
820                    "Expected to open {} of regions, only {} of regions has opened",
821                    leader_region_num,
822                    open_regions.len()
823                )
824            }
825        );
826    } else if open_regions.len() != leader_region_num {
827        warn!(
828            "ignore nonexistent region, expected to open {} of regions, only {} of regions has opened",
829            leader_region_num,
830            open_regions.len()
831        );
832    }
833
834    for region_id in open_regions {
835        if open_with_writable {
836            let res = region_server.set_region_role(region_id, RegionRole::Leader);
837            match res {
838                Ok(_) => {
839                    // Finalize leadership: persist backfilled metadata.
840                    if let SetRegionRoleStateResponse::InvalidTransition(err) = region_server
841                        .set_region_role_state_gracefully(
842                            region_id,
843                            SettableRegionRoleState::Leader,
844                        )
845                        .await?
846                    {
847                        error!(err; "failed to convert region {region_id} to leader");
848                    }
849                }
850                Err(e) => {
851                    error!(e; "failed to convert region {region_id} to leader");
852                }
853            }
854        }
855    }
856
857    #[cfg(feature = "enterprise")]
858    if !follower_regions.is_empty() {
859        use tokio::time::Instant;
860
861        let follower_region_num = follower_regions.len();
862        info!("going to open {} follower region(s)", follower_region_num);
863
864        let now = Instant::now();
865        let open_regions = region_server
866            .handle_batch_open_requests(
867                init_regions_parallelism,
868                follower_regions,
869                ignore_nonexistent_region,
870            )
871            .await?;
872        info!(
873            "Opened {} follower regions in {:?}",
874            open_regions.len(),
875            now.elapsed()
876        );
877
878        if !ignore_nonexistent_region {
879            ensure!(
880                open_regions.len() == follower_region_num,
881                error::UnexpectedSnafu {
882                    violated: format!(
883                        "Expected to open {} of follower regions, only {} of regions has opened",
884                        follower_region_num,
885                        open_regions.len()
886                    )
887                }
888            );
889        } else if open_regions.len() != follower_region_num {
890            warn!(
891                "ignore nonexistent region, expected to open {} of follower regions, only {} of regions has opened",
892                follower_region_num,
893                open_regions.len()
894            );
895        }
896    }
897
898    info!("all regions are opened");
899
900    Ok(())
901}
