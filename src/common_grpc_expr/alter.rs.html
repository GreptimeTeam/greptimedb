49/// Classifies a SET/UNSET key batch: `Ok(Some(family))` when every key belongs
50/// to the same annotation family, `Ok(None)` when none does, and an error on a
51/// mixed batch — annotation alters skip region dispatch, so they cannot share
52/// a statement with options that regions must see.
53fn annotation_family_of_keys<'a>(
54    keys: impl Iterator<Item = &'a str>,
55) -> Result<Option<AnnotationFamily>> {
56    table::requests::validate_annotation_keys(keys).map_err(|err| {
57        error::InvalidTableOptionRequestSnafu {
58            err_msg: err.to_string(),
59        }
60        .build()
61    })
62}

64fn json_settings_from_proto(settings: api::v1::JsonSettings) -> Result<JsonSettings> {
65    let type_hints = settings
66        .type_hints
67        .into_iter()
68        .map(|hint| {
69            let data_type = ConcreteDataType::from(
70                ColumnDataTypeWrapper::try_new(hint.data_type, hint.datatype_extension)
71                    .context(error::ColumnDataTypeSnafu)?,
72            );
73
74            Ok(JsonTypeHint {
75                path: hint.path,
76                data_type,
77                // Index configuration is not supported yet, so this is temporarily
78                // hardcoded to false.
79                inverted_index: false,
80            })
81        })
82        .collect::<Result<Vec<_>>>()?;
83
84    JsonSettings::try_new(type_hints, settings.max_auto_expanded_paths).map_err(|err| {
85        InvalidJsonSettingsSnafu {
86            err: err.to_string(),
87        }
88        .build()
89    })
90}

92/// Returns the annotation family when `kind` is a SET/UNSET whose keys all
93/// belong to one family — the alters that only rewrite table metadata and skip
94/// region dispatch. A mixed batch is an error; never interpret it as "not an
95/// annotation alter", or the batch falls through to a path that reports a
96/// misleading error (or dispatches to regions).
97pub fn annotation_alter_family(kind: &Kind) -> Result<Option<AnnotationFamily>> {
98    match kind {
99        Kind::SetTableOptions(api::v1::SetTableOptions { table_options }) => {
100            annotation_family_of_keys(table_options.iter().map(|option| option.key.as_str()))
101        }
102        Kind::UnsetTableOptions(api::v1::UnsetTableOptions { keys }) => {
103            annotation_family_of_keys(keys.iter().map(|key| key.as_str()))
104        }
105        _ => Ok(None),
106    }
107}

109fn set_index_option_from_proto(set_index: api::v1::SetIndex) -> Result<SetIndexOption> {
110    let options = set_index.options.context(MissingAlterIndexOptionSnafu)?;
111    Ok(match options {
112        api::v1::set_index::Options::Fulltext(f) => SetIndexOption::Fulltext {
113            column_name: f.column_name.clone(),
114            options: FulltextOptions::new(
115                f.enable,
116                as_fulltext_option_analyzer(
117                    Analyzer::try_from(f.analyzer).context(InvalidSetFulltextOptionRequestSnafu)?,
118                ),
119                f.case_sensitive,
120                as_fulltext_option_backend(
121                    PbFulltextBackend::try_from(f.backend)
122                        .context(InvalidSetFulltextOptionRequestSnafu)?,
123                ),
124                f.granularity as u32,
125                f.false_positive_rate,
126            )
127            .context(InvalidIndexOptionSnafu)?,
128        },
129        api::v1::set_index::Options::Inverted(i) => SetIndexOption::Inverted {
130            column_name: i.column_name,
131        },
132        api::v1::set_index::Options::Skipping(s) => SetIndexOption::Skipping {
133            column_name: s.column_name,
134            options: SkippingIndexOptions::new(
135                s.granularity as u32,
136                s.false_positive_rate,
137                as_skipping_index_type(
138                    PbSkippingIndexType::try_from(s.skipping_index_type)
139                        .context(InvalidSetSkippingIndexOptionRequestSnafu)?,
140                ),
141            )
142            .context(InvalidIndexOptionSnafu)?,
143        },
144    })
145}

162/// Convert an [`AlterTableExpr`] to an [`AlterTableRequest`]
163///
164/// note: `table_meta` must not be None if [`AlterTableExpr`] is `SetDefault`
165pub fn alter_expr_to_request(
166    table_id: TableId,
167    expr: AlterTableExpr,
168    table_meta: Option<&TableMeta>,
169) -> Result<AlterTableRequest> {
170    let catalog_name = expr.catalog_name;
171    let schema_name = expr.schema_name;
172    let kind = expr.kind.context(MissingFieldSnafu { field: "kind" })?;
173    let alter_kind = match kind {
174        Kind::AddColumns(add_columns) => {
175            let add_column_requests = add_columns
176                .add_columns
177                .into_iter()
178                .map(|ac| {
179                    let column_def = ac.column_def.context(MissingFieldSnafu {
180                        field: "column_def",
181                    })?;
182
183                    let schema = column_def::try_as_column_schema(&column_def).context(
184                        InvalidColumnDefSnafu {
185                            column: &column_def.name,
186                        },
187                    )?;
188                    Ok(AddColumnRequest {
189                        column_schema: schema,
190                        is_key: column_def.semantic_type == SemanticType::Tag as i32,
191                        location: parse_location(ac.location)?,
192                        add_if_not_exists: ac.add_if_not_exists,
193                    })
194                })
195                .collect::<Result<Vec<_>>>()?;
196
197            AlterKind::AddColumns {
198                columns: add_column_requests,
199            }
200        }
201        Kind::ModifyColumnTypes(ModifyColumnTypes {
202            modify_column_types,
203        }) => {
204            let modify_column_type_requests = modify_column_types
205                .into_iter()
206                .map(|cct| {
207                    let target_type =
208                        ColumnDataTypeWrapper::new(cct.target_type(), cct.target_type_extension)
209                            .into();
210
211                    Ok(ModifyColumnTypeRequest {
212                        column_name: cct.column_name,
213                        target_type,
214                    })
215                })
216                .collect::<Result<Vec<_>>>()?;
217
218            AlterKind::ModifyColumnTypes {
219                columns: modify_column_type_requests,
220            }
221        }
222        Kind::SetJsonSettings(set_json_settings) => {
223            let settings = set_json_settings
224                .settings
225                .context(MissingFieldSnafu { field: "settings" })?;
226            AlterKind::SetJsonSettings {
227                request: SetJsonSettingsRequest {
228                    column_name: set_json_settings.column_name,
229                    settings: json_settings_from_proto(settings)?,
230                },
231            }
232        }
233        Kind::DropColumns(DropColumns { drop_columns }) => AlterKind::DropColumns {
234            names: drop_columns.into_iter().map(|c| c.name).collect(),
235        },
236        Kind::RenameTable(RenameTable { new_table_name }) => {
237            AlterKind::RenameTable { new_table_name }
238        }
239        Kind::SetTableOptions(api::v1::SetTableOptions { table_options }) => {
240            if let Some(family) = annotation_family_of_keys(
241                table_options.iter().map(|option| option.key.as_str()),
242            )? {
243                AlterKind::SetAnnotations {
244                    family,
245                    options: table_options
246                        .into_iter()
247                        .map(|option| (option.key, option.value))
248                        .collect(),
249                }
250            } else {
251                AlterKind::SetTableOptions {
252                    options: table_options
253                        .iter()
254                        .map(SetRegionOption::try_from)
255                        .collect::<std::result::Result<Vec<_>, _>>()
256                        .context(InvalidSetTableOptionRequestSnafu)?,
257                }
258            }
259        }
260        Kind::UnsetTableOptions(api::v1::UnsetTableOptions { keys }) => {
261            if let Some(family) = annotation_family_of_keys(keys.iter().map(|key| key.as_str()))? {
262                AlterKind::UnsetAnnotations { family, keys }
263            } else {
264                AlterKind::UnsetTableOptions {
265                    keys: keys
266                        .iter()
267                        .map(|key| UnsetRegionOption::try_from(key.as_str()))
268                        .collect::<std::result::Result<Vec<_>, _>>()
269                        .context(InvalidUnsetTableOptionRequestSnafu)?,
270                }
271            }
272        }
273        Kind::SetIndex(o) => {
274            let option = set_index_option_from_proto(o)?;
275            AlterKind::SetIndexes {
276                options: vec![option],
277            }
278        }
279        Kind::UnsetIndex(o) => {
280            let option = unset_index_option_from_proto(o)?;
281            AlterKind::UnsetIndexes {
282                options: vec![option],
283            }
284        }
285        Kind::SetIndexes(o) => {
286            let options = o
287                .set_indexes
288                .into_iter()
289                .map(set_index_option_from_proto)
290                .collect::<Result<Vec<_>>>()?;
291            AlterKind::SetIndexes { options }
292        }
293        Kind::UnsetIndexes(o) => {
294            let options = o
295                .unset_indexes
296                .into_iter()
297                .map(unset_index_option_from_proto)
298                .collect::<Result<Vec<_>>>()?;
299            AlterKind::UnsetIndexes { options }
300        }
301        Kind::DropDefaults(o) => {
302            let names = o
303                .drop_defaults
304                .into_iter()
305                .map(|col| {
306                    ensure!(
307                        !col.column_name.is_empty(),
308                        MissingFieldSnafu {
309                            field: "column_name"
310                        }
311                    );
312                    Ok(col.column_name)
313                })
314                .collect::<Result<Vec<_>>>()?;
315            AlterKind::DropDefaults { names }
316        }
317        Kind::SetDefaults(o) => {
318            let table_meta = table_meta.context(MissingTableMetaSnafu { table_id })?;
319            let defaults = o
320                .set_defaults
321                .into_iter()
322                .map(|col| {
323                    let column_scheme = table_meta
324                        .schema
325                        .column_schema_by_name(&col.column_name)
326                        .context(ColumnNotFoundSnafu {
327                        column_name: &col.column_name,
328                    })?;
329                    let default_constraint = common_sql::convert::deserialize_default_constraint(
330                        col.default_constraint.as_slice(),
331                        &col.column_name,
332                        &column_scheme.data_type,
333                    )
334                    .context(crate::error::SqlCommonSnafu)?;
335                    Ok(SetDefaultRequest {
336                        column_name: col.column_name,
337                        default_constraint,
338                    })
339                })
340                .collect::<Result<Vec<_>>>()?;
341            AlterKind::SetDefaults { defaults }
342        }
343        Kind::Repartition(_) => error::UnexpectedSnafu {
344            err_msg: "Repartition operation should be handled through DdlManager and not converted to AlterTableRequest",
345        }
346        .fail()?,
347    };
348
349    let request = AlterTableRequest {
350        catalog_name,
351        schema_name,
352        table_name: expr.table_name,
353        table_id,
354        alter_kind,
355        table_version: None,
356    };
357    Ok(request)
358}

360pub fn create_table_schema(expr: &CreateTableExpr, require_time_index: bool) -> Result<Schema> {
361    let column_schemas = expr
362        .column_defs
363        .iter()
364        .map(|x| {
365            column_def::try_as_column_schema(x).context(InvalidColumnDefSnafu { column: &x.name })
366        })
367        .collect::<Result<Vec<ColumnSchema>>>()?;
368
369    // allow external table schema without the time index
370    if require_time_index {
371        ensure!(
372            column_schemas
373                .iter()
374                .any(|column| column.name == expr.time_index),
375            MissingTimestampColumnSnafu {
376                msg: format!("CreateExpr: {expr:?}")
377            }
378        );
379    }
380
381    let column_schemas = column_schemas
382        .into_iter()
383        .map(|column_schema| {
384            if column_schema.name == expr.time_index {
385                column_schema.with_time_index(true)
386            } else {
387                column_schema
388            }
389        })
390        .collect::<Vec<_>>();
391
392    Ok(Schema::new(column_schemas))
393}

395fn parse_location(location: Option<Location>) -> Result<Option<AddColumnLocation>> {
396    match location {
397        Some(Location {
398            location_type: LOCATION_TYPE_FIRST,
399            ..
400        }) => Ok(Some(AddColumnLocation::First)),
401        Some(Location {
402            location_type: LOCATION_TYPE_AFTER,
403            after_column_name,
404        }) => Ok(Some(AddColumnLocation::After {
405            column_name: after_column_name,
406        })),
407        Some(Location { location_type, .. }) => UnknownLocationTypeSnafu { location_type }.fail(),
408        None => Ok(None),
409    }
410}

608    #[test]
609    fn test_repartition_hints_together() {
610        let keys = [
611            REPARTITION_COLUMN_HINT_KEY,
612            table::requests::REPARTITION_PARTITION_NUM_HINT_KEY,
613        ];
614        let options = vec![
615            (keys[0].to_string(), "host".to_string()),
616            (keys[1].to_string(), "10".to_string()),
617        ];
618        let set = Kind::SetTableOptions(SetTableOptions {
619            table_options: options
620                .iter()
621                .map(|(key, value)| PbOption {
622                    key: key.clone(),
623                    value: value.clone(),
624                })
625                .collect(),
626        });
627        let unset = Kind::UnsetTableOptions(UnsetTableOptions {
628            keys: keys.map(str::to_string).to_vec(),
629        });
630        for kind in [set, unset] {
631            assert_eq!(
632                annotation_alter_family(&kind).unwrap(),
633                Some(AnnotationFamily::RepartitionHint)
634            );
635            let request = alter_expr_to_request(
636                1,
637                AlterTableExpr {
638                    kind: Some(kind),
639                    ..Default::default()
640                },
641                None,
642            )
643            .unwrap();
644            match request.alter_kind {
645                AlterKind::SetAnnotations {
646                    family,
647                    options: actual,
648                } => {
649                    assert_eq!(family, AnnotationFamily::RepartitionHint);
650                    assert_eq!(actual, options);
651                }
652                AlterKind::UnsetAnnotations {
653                    family,
654                    keys: actual,
655                } => {
656                    assert_eq!(family, AnnotationFamily::RepartitionHint);
657                    assert_eq!(actual, keys);
658                }
659                other => panic!("unexpected alter kind: {other:?}"),
660            }
661        }
662    }

687                None,

695                AlterKind::UnsetAnnotations { family, keys } => {
696                    assert_eq!(family, AnnotationFamily::RepartitionHint);
697                    assert_eq!(keys, vec![key.to_string()]);

719                    alter_expr_to_request(
720                        1,
721                        AlterTableExpr {
722                            kind: Some(kind),
723                            ..Default::default()
724                        },
