58impl DistributedInformationExtension {
59    pub fn new(meta_client: MetaClientRef, datanode_manager: DatanodeManagerRef) -> Self {
60        Self {
61            meta_client,
62            datanode_manager,
63        }
64    }
65
66    async fn inspect_datanode_stream(
67        meta_client: MetaClientRef,
68        datanode_manager: DatanodeManagerRef,
69        request: DatanodeInspectRequest,
70    ) -> crate::error::Result<SendableRecordBatchStream> {
71        let nodes = meta_client
72            .list_nodes(Some(Role::Datanode))
73            .await
74            .map_err(BoxedError::new)
75            .context(crate::error::ListNodesSnafu)?;
76        Self::inspect_nodes(datanode_manager, request, nodes).await
77    }
78
79    /// Executes inspection against discovered nodes after applying safe node pruning.
80    async fn inspect_nodes(
81        datanode_manager: DatanodeManagerRef,
82        request: DatanodeInspectRequest,
83        nodes: Vec<NodeInfo>,
84    ) -> crate::error::Result<SendableRecordBatchStream> {
85        let nodes = prune_compaction_nodes(&request, nodes);
86        let limit = request.scan.limit;
87        let plan = request
88            .build_plan()
89            .context(crate::error::DatafusionSnafu)?;
90        let stream = Self::inspect_query_nodes(datanode_manager, plan, nodes).await?;
91        match limit {
92            Some(limit) => Ok(Box::pin(LimitedRecordBatchStream::new(stream, limit))),
93            None => Ok(stream),
94        }
95    }
96
97    async fn inspect_query_stream(
98        meta_client: MetaClientRef,
99        datanode_manager: DatanodeManagerRef,
100        plan: LogicalPlan,
101    ) -> crate::error::Result<SendableRecordBatchStream> {
102        let nodes = meta_client
103            .list_nodes(Some(Role::Datanode))
104            .await
105            .map_err(BoxedError::new)
106            .context(crate::error::ListNodesSnafu)?;
107        Self::inspect_query_nodes(datanode_manager, plan, nodes).await
108    }
109
110    /// Shares query dispatch and empty-stream schema handling for scans and query plans.
111    async fn inspect_query_nodes(
112        datanode_manager: DatanodeManagerRef,
113        plan: LogicalPlan,
114        nodes: Vec<NodeInfo>,
115    ) -> crate::error::Result<SendableRecordBatchStream> {
116        let schema = inspection_query_schema(&plan)?;
117        let streams = try_join_all(nodes.into_iter().map(|node| {
118            let datanode_manager = datanode_manager.clone();
119            let plan = plan.clone();
120            async move {
121                let client = datanode_manager.datanode(&node.peer).await;
122                client
123                    .handle_query(QueryRequest {
124                        plan,
125                        region_id: RegionId::default(),
126                        header: None,
127                    })
128                    .await
129                    .context(crate::error::HandleQuerySnafu {
130                        addr: node.peer.addr,
131                    })
132            }
133        }))
134        .await?;
135
136        chain_inspection_streams(streams, schema)
137    }
138}
139
140fn inspection_query_schema(plan: &LogicalPlan) -> crate::error::Result<SchemaRef> {
141    datatypes::schema::Schema::try_from(plan.schema().clone())
142        .map(Arc::new)
143        .map_err(|error| datafusion::common::DataFusionError::External(Box::new(error)))
144        .context(crate::error::DatafusionSnafu)
145}
146
147fn chain_inspection_streams(
148    streams: Vec<SendableRecordBatchStream>,
149    schema: SchemaRef,
150) -> crate::error::Result<SendableRecordBatchStream> {
151    if streams.is_empty() {
152        return Ok(Box::pin(EmptyRecordBatchStream::new(schema)));
153    }
154    Ok(Box::pin(
155        ChainedRecordBatchStream::new(streams).context(crate::error::CreateRecordBatchSnafu)?,
156    ))
157}

159#[derive(Debug, Clone)]
160enum InspectPlan {
161    Scan(DatanodeInspectRequest),
162    Query(LogicalPlan),
163}
164
165/// Prunes reporting nodes only for live compaction inspection. Region routes cannot
166/// locate tasks left on a former owner, so only known node IDs can exclude nodes.
167fn prune_compaction_nodes(request: &DatanodeInspectRequest, nodes: Vec<NodeInfo>) -> Vec<NodeInfo> {
168    if request.kind != DatanodeInspectKind::CompactionTasks {
169        return nodes;
170    }
171    let filter = MetadataFilter::new(&request.scan.filters);
172    nodes
173        .into_iter()
174        .filter(|node| {
175            filter.matches(|name| {
176                (name == compaction_task::NODE_ID)
177                    .then_some(ScalarValue::UInt64(Some(node.peer.id)))
178            })
179        })
180        .collect()
181}

183struct DistributedInspectExec {
184    plan: InspectPlan,
185    meta_client: MetaClientRef,
186    datanode_manager: DatanodeManagerRef,
187    schema: SchemaRef,
188    arrow_schema: ArrowSchemaRef,
189    properties: Arc<PlanProperties>,
190}

192impl std::fmt::Debug for DistributedInspectExec {
193    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
194        f.debug_struct("DistributedInspectExec")
195            .field("plan", &self.plan)
196            .field("schema", &self.schema)
197            .finish_non_exhaustive()
198    }
199}

201impl DistributedInspectExec {
202    fn new(
203        request: DatanodeInspectRequest,
204        schema: SchemaRef,
205        meta_client: MetaClientRef,
206        datanode_manager: DatanodeManagerRef,
207    ) -> Self {
208        Self::with_plan(
209            InspectPlan::Scan(request),
210            schema,
211            meta_client,
212            datanode_manager,
213        )
214    }
215
216    fn with_plan(
217        plan: InspectPlan,
218        schema: SchemaRef,
219        meta_client: MetaClientRef,
220        datanode_manager: DatanodeManagerRef,
221    ) -> Self {
222        let arrow_schema = schema.arrow_schema().clone();
223        let properties = Arc::new(PlanProperties::new(
224            EquivalenceProperties::new(arrow_schema.clone()),
225            Partitioning::UnknownPartitioning(1),
226            EmissionType::Incremental,
227            Boundedness::Bounded,
228        ));
229        Self {
230            plan,
231            meta_client,
232            datanode_manager,
233            schema,
234            arrow_schema,
235            properties,
236        }
237    }
238}

240impl DisplayAs for DistributedInspectExec {

251                plan.display_indent(),
252                self.arrow_schema
253            ),
254        }
255    }

258impl ExecutionPlan for DistributedInspectExec {
259    fn name(&self) -> &str {
260        "DistributedInspectExec"
261    }
262
263    fn schema(&self) -> ArrowSchemaRef {
264        self.arrow_schema.clone()
265    }
266
267    fn properties(&self) -> &Arc<PlanProperties> {
268        &self.properties
269    }
270
271    fn children(&self) -> Vec<&Arc<dyn ExecutionPlan>> {
272        vec![]
273    }
274
275    fn apply_expressions(
276        &self,
277        _f: &mut dyn FnMut(&Arc<dyn PhysicalExpr>) -> Result<TreeNodeRecursion>,
278    ) -> Result<TreeNodeRecursion> {
279        Ok(TreeNodeRecursion::Continue)
280    }
281
282    fn with_new_children(
283        self: Arc<Self>,
284        children: Vec<Arc<dyn ExecutionPlan>>,
285    ) -> datafusion::error::Result<Arc<dyn ExecutionPlan>> {
286        if children.is_empty() {
287            Ok(self)
288        } else {
289            Err(datafusion::error::DataFusionError::Internal(
290                "DistributedInspectExec is a leaf execution plan and cannot accept children"
291                    .to_string(),
292            ))
293        }
294    }
295
296    fn execute(
297        &self,
298        partition: usize,
299        _context: Arc<TaskContext>,
300    ) -> datafusion::error::Result<DfSendableRecordBatchStream> {
301        if partition != 0 {
302            return Err(datafusion::error::DataFusionError::Execution(format!(
303                "DistributedInspectExec invalid partition. Expected 0, got {partition}"
304            )));
305        }
306
307        let meta_client = self.meta_client.clone();
308        let datanode_manager = self.datanode_manager.clone();
309        let plan = self.plan.clone();
310        let schema = self.schema.clone();
311        let future = async move {
312            match plan {
313                InspectPlan::Scan(request) => {
314                    DistributedInformationExtension::inspect_datanode_stream(
315                        meta_client,
316                        datanode_manager,
317                        request,
318                    )
319                    .await
320                }
321                InspectPlan::Query(plan) => {
322                    DistributedInformationExtension::inspect_query_stream(
323                        meta_client,
324                        datanode_manager,
325                        plan,
326                    )
327                    .await
328                }
329            }
330            .map_err(BoxedError::new)
331            .context(common_recordbatch::error::ExternalSnafu)
332        };
333        let stream = AsyncRecordBatchStreamAdapter::new(schema, Box::pin(future));
334        Ok(Box::pin(DfRecordBatchStreamAdapter::new(Box::pin(stream))))
335    }
336}

342    async fn nodes(&self) -> std::result::Result<Vec<NodeInfo>, Self::Error> {

345            .await
346            .map_err(BoxedError::new)
347            .context(error::ListNodesSnafu)
348    }

350    async fn procedures(&self) -> std::result::Result<Vec<(String, ProcedureInfo)>, Self::Error> {
351        let procedures = self
352            .meta_client
353            .list_procedures(&ExecutorContext::default())
354            .await
355            .map_err(BoxedError::new)
356            .context(error::ListProceduresSnafu)?
357            .procedures;
358        let mut result = Vec::with_capacity(procedures.len());
359        for procedure in procedures {
360            let pid = match procedure.id {
361                Some(pid) => pid,
362                None => return error::ProcedureIdNotFoundSnafu {}.fail(),
363            };
364            let pid = procedure::pb_pid_to_pid(&pid)
365                .map_err(BoxedError::new)
366                .context(error::ConvertProtoDataSnafu)?;
367            let status = ProcedureStatus::try_from(procedure.status)
368                .map(|v| v.as_str_name())
369                .unwrap_or("Unknown")
370                .to_string();
371            let procedure_info = ProcedureInfo {
372                id: pid,
373                type_name: procedure.type_name,
374                start_time_ms: procedure.start_time_ms,
375                end_time_ms: procedure.end_time_ms,
376                state: ProcedureState::Running,
377                lock_keys: procedure.lock_keys,
378            };
379            result.push((status, procedure_info));
380        }
381
382        Ok(result)
383    }

492    /// Uses the default optimizer to produce the predicates seen by inspection scans.
493    async fn compaction_sql_filters(predicate: Option<&str>) -> Vec<Expr> {
494        let ctx = SessionContext::new();
495        ctx.register_table(
496            "compaction_tasks",
497            Arc::new(EmptyTable::new(
498                CompactionTaskEntry::schema().arrow_schema().clone(),
499            )),
500        )
501        .unwrap();
502        let mut sql = "SELECT * FROM compaction_tasks".to_string();
503        if let Some(predicate) = predicate {
504            sql.push_str(" WHERE ");
505            sql.push_str(predicate);
506        }
507        let plan = ctx.sql(&sql).await.unwrap().into_optimized_plan().unwrap();
508        let mut filters = Vec::new();
509        plan.apply(|plan| {
510            match plan {
511                LogicalPlan::Filter(filter) => filters.push(filter.predicate.clone()),
512                LogicalPlan::TableScan(scan) => filters.extend(scan.filters.iter().cloned()),
513                _ => {}
514            }
515            Ok(TreeNodeRecursion::Continue)
516        })
517        .unwrap();
518        filters
519    }

522    async fn test_compaction_node_pruning() {
523        for (predicate, expected, expect_cast) in [
524            (Some("node_id = 2"), vec![2], false),
525            (Some("node_id != 2"), vec![1, 3], false),
526            (Some("node_id > 2"), vec![3], false),
527            (Some("node_id >= 2"), vec![2, 3], false),
528            (Some("node_id < 2"), vec![1], false),
529            (Some("node_id <= 2"), vec![1, 2], false),
530            (Some("2 = node_id"), vec![2], false),
531            (Some("2 < node_id"), vec![3], false),
532            (None, vec![1, 2, 3], false),
533            (Some("state = 'running'"), vec![1, 2, 3], false),
534            (Some("node_id = 2 OR table_id = 42"), vec![1, 2, 3], false),
535            (Some("node_id = 2 AND state = 'running'"), vec![2], false),
536        ] {
537            let filters = compaction_sql_filters(predicate).await;
538            let mut has_cast = false;
539            for filter in &filters {
540                filter
541                    .apply(|expr| {
542                        has_cast |= matches!(expr, Expr::Cast(cast) if cast.field.data_type() == &DataType::Decimal128(20, 0)
543                            && matches!(cast.expr.as_ref(), Expr::Column(_)));
544                        Ok(TreeNodeRecursion::Continue)
545                    })
546                    .unwrap();
547            }
548            assert_eq!(has_cast, expect_cast, "optimized predicate: {predicate:?}");
549            let request = DatanodeInspectRequest {
550                kind: DatanodeInspectKind::CompactionTasks,
551                scan: ScanRequest {
552                    filters,
553                    ..Default::default()
554                },
555            };
556            let actual: Vec<_> = prune_compaction_nodes(&request, inspection_nodes())
557                .into_iter()
558                .map(|node| node.peer.id)
559                .collect();
560            assert_eq!(
561                actual, expected,
562                "{predicate:?}: {:?}",
563                request.scan.filters
564            );
565        }
566        let filters = compaction_sql_filters(Some("node_id = 99")).await;
567        for kind in [
568            DatanodeInspectKind::SstManifest,
569            DatanodeInspectKind::SstStorage,
570            DatanodeInspectKind::SstIndexMeta,
571            DatanodeInspectKind::RegionInfo,
572        ] {
573            let request = DatanodeInspectRequest {
574                kind,
575                scan: ScanRequest {
576                    filters: filters.clone(),
577                    ..Default::default()
578                },
579            };
580            let actual: Vec<_> = prune_compaction_nodes(&request, inspection_nodes())
581                .into_iter()
582                .map(|node| node.peer.id)

695        let ctx = datafusion::prelude::SessionContext::new();
696        let remote = ctx
697            .sql("SELECT count(*), sum(value) FROM (VALUES (1), (2)) AS t(value)")
698            .await
699            .unwrap()
700            .into_unoptimized_plan();
701        let extension = DistributedInformationExtension::new(
702            Arc::new(MetaClientBuilder::new(0, api::v1::meta::Role::Frontend).build()),
703            Arc::new(UnusedDatanodeManager),
704        );
705        assert!(extension.supports_inspect_query_plan());
706        let exec = extension
707            .inspect_datanode_query_plan(remote.clone())
708            .unwrap()
709            .unwrap();
710        assert_eq!(exec.schema(), remote.schema().inner().clone());
711        let inspect = exec.downcast_ref::<DistributedInspectExec>().unwrap();
712        let InspectPlan::Query(plan) = &inspect.plan else {
713            panic!("expected inspection query")
714        };
715        assert_eq!(plan, &remote);
716        assert!(exec.children().is_empty());
717        // Creating or dropping an unpolled execution must not contact metadata or datanodes.
718        for _ in 0..2 {
719            let stream = exec.execute(0, ctx.task_ctx()).unwrap();
720            assert_eq!(stream.schema(), exec.schema());
721            drop(stream);
722        }
723        assert!(exec.execute(1, ctx.task_ctx()).is_err());
724    }

742    #[tokio::test]
743    async fn test_sql_manifest_sort_stays_on_frontend() {
744        let ctx = datafusion::prelude::SessionContext::new();
745        ctx.register_table("ssts_manifest", Arc::new(ManifestScanProvider))
746            .unwrap();
747        for sql in [
748            "SELECT region_id, file_id, min_ts, max_ts, file_size FROM ssts_manifest WHERE region_id = 42 ORDER BY min_ts LIMIT 100",
749            "SELECT file_id FROM ssts_manifest WHERE region_id = 42 ORDER BY max_ts DESC LIMIT 10 OFFSET 3",
750            "SELECT min_ts AS ts, file_id FROM ssts_manifest WHERE region_id = 42 ORDER BY ts DESC NULLS FIRST LIMIT 10",
751            "SELECT file_id FROM ssts_manifest WHERE region_id = 42 ORDER BY min_ts, max_ts DESC LIMIT 10",
752        ] {
753            let plan = ctx
754                .sql(sql)
755                .await
756                .unwrap()
757                .create_physical_plan()
758                .await
759                .unwrap();
760            let mut found_scan = false;
761            let mut found_sort = false;
762            plan.apply(|plan| {
763                found_sort |= plan.is::<SortExec>();
764                if let Some(scan) = plan.downcast_ref::<DistributedInspectExec>() {
765                    found_scan = true;
766                    // All matching rows must reach the frontend global sort.
767                    let InspectPlan::Scan(request) = &scan.plan else {
768                        panic!("expected scan plan")
769                    };
770                    assert_eq!(request.scan.limit, None);
771                    let remote = request.clone().build_plan().unwrap();
772                    remote.apply(|node| {
773                        assert!(!matches!(
774                            node,
775                            LogicalPlan::Sort(_) | LogicalPlan::Limit(_)
776                        ));
777                        Ok(TreeNodeRecursion::Continue)
778                    })?;
779                }
780                Ok(TreeNodeRecursion::Continue)
781            })
782            .unwrap();
783            assert!(
784                found_scan && found_sort,
785                "{}",
786                datafusion::physical_plan::displayable(plan.as_ref()).indent(true)
787            );
788        }
789    }
