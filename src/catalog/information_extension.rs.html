49pub struct DistributedInformationExtension {
50    meta_client: MetaClientRef,
51    datanode_manager: DatanodeManagerRef,
52}
53
54impl DistributedInformationExtension {
55    pub fn new(meta_client: MetaClientRef, datanode_manager: DatanodeManagerRef) -> Self {
56        Self {
57            meta_client,
58            datanode_manager,
59        }
60    }
61
62    async fn inspect_datanode_stream(
63        meta_client: MetaClientRef,
64        datanode_manager: DatanodeManagerRef,
65        request: DatanodeInspectRequest,
66    ) -> std::result::Result<SendableRecordBatchStream, crate::error::Error> {
67        let limit = request.scan.limit;
68        let plan = request
69            .build_plan()
70            .context(crate::error::DatafusionSnafu)?;
71        let stream = Self::inspect_query_stream(meta_client, datanode_manager, plan).await?;
72        match limit {
73            Some(limit) => Ok(Box::pin(LimitedRecordBatchStream::new(stream, limit))),
74            None => Ok(stream),
75        }
76    }
77
78    async fn inspect_query_stream(
79        meta_client: MetaClientRef,
80        datanode_manager: DatanodeManagerRef,
81        plan: LogicalPlan,
82    ) -> crate::error::Result<SendableRecordBatchStream> {
83        let schema = inspection_query_schema(&plan)?;
84        let nodes = meta_client
85            .list_nodes(Some(Role::Datanode))
86            .await
87            .map_err(BoxedError::new)
88            .context(crate::error::ListNodesSnafu)?;
89
90        let streams = try_join_all(nodes.into_iter().map(|node| {
91            let datanode_manager = datanode_manager.clone();
92            let plan = plan.clone();
93            async move {
94                let client = datanode_manager.datanode(&node.peer).await;
95                client
96                    .handle_query(QueryRequest {
97                        plan,
98                        region_id: RegionId::default(),
99                        header: None,
100                    })
101                    .await
102                    .context(crate::error::HandleQuerySnafu {
103                        addr: node.peer.addr,
104                    })
105            }
106        }))
107        .await?;
108
109        chain_inspection_streams(streams, schema)
110    }
111}
112
113fn inspection_query_schema(plan: &LogicalPlan) -> crate::error::Result<SchemaRef> {
114    datatypes::schema::Schema::try_from(plan.schema().clone())
115        .map(Arc::new)
116        .map_err(|error| datafusion::common::DataFusionError::External(Box::new(error)))
117        .context(crate::error::DatafusionSnafu)
118}
119
120fn chain_inspection_streams(
121    streams: Vec<SendableRecordBatchStream>,
122    schema: SchemaRef,
123) -> crate::error::Result<SendableRecordBatchStream> {
124    if streams.is_empty() {
125        return Ok(Box::pin(EmptyRecordBatchStream::new(schema)));
126    }
127    Ok(Box::pin(
128        ChainedRecordBatchStream::new(streams).context(crate::error::CreateRecordBatchSnafu)?,
129    ))
130}

132#[derive(Debug, Clone)]
133enum InspectPlan {
134    Scan(DatanodeInspectRequest),
135    Query(LogicalPlan),
136}
137
138struct DistributedInspectExec {
139    plan: InspectPlan,
140    meta_client: MetaClientRef,
141    datanode_manager: DatanodeManagerRef,
142    schema: SchemaRef,
143    arrow_schema: ArrowSchemaRef,
144    properties: Arc<PlanProperties>,
145}
146
147impl std::fmt::Debug for DistributedInspectExec {
148    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
149        f.debug_struct("DistributedInspectExec")
150            .field("plan", &self.plan)
151            .field("schema", &self.schema)
152            .finish_non_exhaustive()
153    }
154}
155
156impl DistributedInspectExec {
157    fn new(
158        request: DatanodeInspectRequest,
159        schema: SchemaRef,
160        meta_client: MetaClientRef,
161        datanode_manager: DatanodeManagerRef,
162    ) -> Self {
163        Self::with_plan(
164            InspectPlan::Scan(request),
165            schema,
166            meta_client,
167            datanode_manager,
168        )
169    }
170
171    fn with_plan(
172        plan: InspectPlan,
173        schema: SchemaRef,
174        meta_client: MetaClientRef,
175        datanode_manager: DatanodeManagerRef,
176    ) -> Self {
177        let arrow_schema = schema.arrow_schema().clone();
178        let properties = Arc::new(PlanProperties::new(
179            EquivalenceProperties::new(arrow_schema.clone()),
180            Partitioning::UnknownPartitioning(1),
181            EmissionType::Incremental,
182            Boundedness::Bounded,
183        ));
184        Self {
185            plan,
186            meta_client,
187            datanode_manager,
188            schema,
189            arrow_schema,
190            properties,
191        }
192    }
193}
194
195impl DisplayAs for DistributedInspectExec {
196    fn fmt_as(&self, _t: DisplayFormatType, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
197        match &self.plan {
198            InspectPlan::Scan(request) => write!(
199                f,
200                "DistributedInspectExec: kind={:?}, scan={}, schema={:?}",
201                request.kind, request.scan, self.arrow_schema
202            ),
203            InspectPlan::Query(plan) => write!(
204                f,
205                "DistributedInspectExec: query={}, schema={:?}",
206                plan.display_indent(),
207                self.arrow_schema
208            ),
209        }
210    }
211}

213impl ExecutionPlan for DistributedInspectExec {
214    fn name(&self) -> &str {
215        "DistributedInspectExec"
216    }
217
218    fn schema(&self) -> ArrowSchemaRef {
219        self.arrow_schema.clone()
220    }
221
222    fn properties(&self) -> &Arc<PlanProperties> {
223        &self.properties
224    }
225
226    fn children(&self) -> Vec<&Arc<dyn ExecutionPlan>> {
227        vec![]
228    }
229
230    fn apply_expressions(
231        &self,
232        _f: &mut dyn FnMut(&Arc<dyn PhysicalExpr>) -> Result<TreeNodeRecursion>,
233    ) -> Result<TreeNodeRecursion> {
234        Ok(TreeNodeRecursion::Continue)
235    }
236
237    fn with_new_children(
238        self: Arc<Self>,
239        children: Vec<Arc<dyn ExecutionPlan>>,
240    ) -> datafusion::error::Result<Arc<dyn ExecutionPlan>> {
241        if children.is_empty() {
242            Ok(self)
243        } else {
244            Err(datafusion::error::DataFusionError::Internal(
245                "DistributedInspectExec is a leaf execution plan and cannot accept children"
246                    .to_string(),
247            ))
248        }
249    }
250
251    fn execute(
252        &self,
253        partition: usize,
254        _context: Arc<TaskContext>,
255    ) -> datafusion::error::Result<DfSendableRecordBatchStream> {
256        if partition != 0 {
257            return Err(datafusion::error::DataFusionError::Execution(format!(
258                "DistributedInspectExec invalid partition. Expected 0, got {partition}"
259            )));
260        }
261
262        let meta_client = self.meta_client.clone();
263        let datanode_manager = self.datanode_manager.clone();
264        let plan = self.plan.clone();
265        let schema = self.schema.clone();
266        let future = async move {
267            match plan {
268                InspectPlan::Scan(request) => {
269                    DistributedInformationExtension::inspect_datanode_stream(
270                        meta_client,
271                        datanode_manager,
272                        request,
273                    )
274                    .await
275                }
276                InspectPlan::Query(plan) => {
277                    DistributedInformationExtension::inspect_query_stream(
278                        meta_client,
279                        datanode_manager,
280                        plan,
281                    )
282                    .await
283                }
284            }
285            .map_err(BoxedError::new)
286            .context(common_recordbatch::error::ExternalSnafu)
287        };
288        let stream = AsyncRecordBatchStreamAdapter::new(schema, Box::pin(future));
289        Ok(Box::pin(DfRecordBatchStreamAdapter::new(Box::pin(stream))))
290    }
291}

293#[async_trait::async_trait]
294impl InformationExtension for DistributedInformationExtension {
295    type Error = crate::error::Error;
296
297    async fn nodes(&self) -> std::result::Result<Vec<NodeInfo>, Self::Error> {
298        self.meta_client
299            .list_nodes(None)
300            .await
301            .map_err(BoxedError::new)
302            .context(error::ListNodesSnafu)
303    }
304
305    async fn procedures(&self) -> std::result::Result<Vec<(String, ProcedureInfo)>, Self::Error> {
306        let procedures = self
307            .meta_client
308            .list_procedures(&ExecutorContext::default())
309            .await
310            .map_err(BoxedError::new)
311            .context(error::ListProceduresSnafu)?
312            .procedures;
313        let mut result = Vec::with_capacity(procedures.len());
314        for procedure in procedures {
315            let pid = match procedure.id {
316                Some(pid) => pid,
317                None => return error::ProcedureIdNotFoundSnafu {}.fail(),
318            };
319            let pid = procedure::pb_pid_to_pid(&pid)
320                .map_err(BoxedError::new)
321                .context(error::ConvertProtoDataSnafu)?;
322            let status = ProcedureStatus::try_from(procedure.status)
323                .map(|v| v.as_str_name())
324                .unwrap_or("Unknown")
325                .to_string();
326            let procedure_info = ProcedureInfo {
327                id: pid,
328                type_name: procedure.type_name,
329                start_time_ms: procedure.start_time_ms,
330                end_time_ms: procedure.end_time_ms,
331                state: ProcedureState::Running,
332                lock_keys: procedure.lock_keys,
333            };
334            result.push((status, procedure_info));
335        }
336
337        Ok(result)
338    }
339
340    async fn region_stats(&self) -> std::result::Result<Vec<RegionStat>, Self::Error> {
341        self.meta_client
342            .list_region_stats()
343            .await
344            .map_err(BoxedError::new)
345            .context(error::ListRegionStatsSnafu)
346    }
347
348    async fn flow_stats(&self) -> std::result::Result<Option<FlowStat>, Self::Error> {
349        self.meta_client
350            .list_flow_stats()
351            .await
352            .map_err(BoxedError::new)
353            .context(crate::error::ListFlowStatsSnafu)
354    }
355
356    async fn inspect_datanode(
357        &self,
358        request: DatanodeInspectRequest,
359    ) -> std::result::Result<SendableRecordBatchStream, Self::Error> {
360        Self::inspect_datanode_stream(
361            self.meta_client.clone(),
362            self.datanode_manager.clone(),
363            request,
364        )
365        .await
366    }
367
368    fn supports_inspect_query_plan(&self) -> bool {
369        true
370    }
371
372    fn inspect_datanode_query_plan(
373        &self,
374        plan: LogicalPlan,
375    ) -> crate::error::Result<Option<Arc<dyn ExecutionPlan>>> {
376        let schema = inspection_query_schema(&plan)?;
377        Ok(Some(Arc::new(DistributedInspectExec::with_plan(
378            InspectPlan::Query(plan),
379            schema,
380            self.meta_client.clone(),
381            self.datanode_manager.clone(),
382        ))))
383    }
384
385    fn inspect_datanode_plan(
386        &self,
387        request: DatanodeInspectRequest,
388        schema: SchemaRef,
389    ) -> std::result::Result<Option<Arc<dyn ExecutionPlan>>, Self::Error> {
390        Ok(Some(Arc::new(DistributedInspectExec::new(
391            request,
392            schema,
393            self.meta_client.clone(),
394            self.datanode_manager.clone(),
395        ))))
396    }

471    #[tokio::test]
472    async fn test_inspection_query_plan_is_lazy_and_preserves_schema() {
473        let ctx = datafusion::prelude::SessionContext::new();
474        let remote = ctx
475            .sql("SELECT count(*), sum(value) FROM (VALUES (1), (2)) AS t(value)")
476            .await
477            .unwrap()
478            .into_unoptimized_plan();
479        let extension = DistributedInformationExtension::new(
480            Arc::new(MetaClientBuilder::new(0, api::v1::meta::Role::Frontend).build()),
481            Arc::new(UnusedDatanodeManager),
482        );
483        assert!(extension.supports_inspect_query_plan());
484        let exec = extension
485            .inspect_datanode_query_plan(remote.clone())
486            .unwrap()
487            .unwrap();
488        assert_eq!(exec.schema(), remote.schema().inner().clone());
489        let inspect = exec.downcast_ref::<DistributedInspectExec>().unwrap();
490        let InspectPlan::Query(plan) = &inspect.plan else {
491            panic!("expected inspection query")
492        };
493        assert_eq!(plan, &remote);
494        assert!(exec.children().is_empty());
495        // Creating or dropping an unpolled execution must not contact metadata or datanodes.
496        for _ in 0..2 {
497            let stream = exec.execute(0, ctx.task_ctx()).unwrap();
498            assert_eq!(stream.schema(), exec.schema());
499            drop(stream);
500        }
501        assert!(exec.execute(1, ctx.task_ctx()).is_err());
502    }

520    #[tokio::test]
521    async fn test_sql_manifest_sort_stays_on_frontend() {
522        let ctx = datafusion::prelude::SessionContext::new();
523        ctx.register_table("ssts_manifest", Arc::new(ManifestScanProvider))
524            .unwrap();
525        for sql in [
526            "SELECT region_id, file_id, min_ts, max_ts, file_size FROM ssts_manifest WHERE region_id = 42 ORDER BY min_ts LIMIT 100",
527            "SELECT file_id FROM ssts_manifest WHERE region_id = 42 ORDER BY max_ts DESC LIMIT 10 OFFSET 3",
528            "SELECT min_ts AS ts, file_id FROM ssts_manifest WHERE region_id = 42 ORDER BY ts DESC NULLS FIRST LIMIT 10",
529            "SELECT file_id FROM ssts_manifest WHERE region_id = 42 ORDER BY min_ts, max_ts DESC LIMIT 10",
530        ] {
531            let plan = ctx
532                .sql(sql)
533                .await
534                .unwrap()
535                .create_physical_plan()
536                .await
537                .unwrap();
538            let mut found_scan = false;
539            let mut found_sort = false;
540            plan.apply(|plan| {
541                found_sort |= plan.is::<SortExec>();
542                if let Some(scan) = plan.downcast_ref::<DistributedInspectExec>() {
543                    found_scan = true;
544                    // All matching rows must reach the frontend global sort.
545                    let InspectPlan::Scan(request) = &scan.plan else {
546                        panic!("expected scan plan")
547                    };
548                    assert_eq!(request.scan.limit, None);
549                    let remote = request.clone().build_plan().unwrap();
550                    remote.apply(|node| {
551                        assert!(!matches!(
552                            node,
553                            LogicalPlan::Sort(_) | LogicalPlan::Limit(_)
554                        ));
555                        Ok(TreeNodeRecursion::Continue)
556                    })?;
557                }
558                Ok(TreeNodeRecursion::Continue)
559            })
560            .unwrap();
561            assert!(
562                found_scan && found_sort,
563                "{}",
564                datafusion::physical_plan::displayable(plan.as_ref()).indent(true)
565            );
566        }
567    }
