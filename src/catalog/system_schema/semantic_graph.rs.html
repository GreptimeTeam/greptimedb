15//! The computed entity-graph tables `greptime_private.semantic_entities` and
16//! `greptime_private.semantic_relationships`.
17//!
18//! They live in `greptime_private`, not `information_schema`: scanning them
19//! triggers read-time derivation over telemetry tables (trace self-joins, ...),
20//! which breaks the "cheap, metadata-only" expectation users have of
21//! `information_schema`. `greptime_private` also hosts the physical
22//! declared-edge table (`semantic_relationships_declared`), whose rows are
23//! unioned into the computed `semantic_relationships`.
24//!
25//! These are thin forwarders: their rows are derived at read time by the injected
26//! [`EntityGraphProvider`], which enumerates the `table_semantics` declarations,
27//! builds typed DataFusion derivation plans, and executes them via the query
28//! engine. When no provider is injected (e.g. before the engine is up, or on a
29//! non-frontend node) they stream empty. The fixed schemas here must match the
30//! columns the provider's plans project; JSON columns are `json` (JSONB), whose
31//! Arrow storage type is `Binary` — the derived batches are rebuilt against the
32//! declared Arrow schema (which carries the `json` extension metadata) in
33//! [`SystemTable::to_stream`].

67/// Produces the rows of the computed entity-graph tables at read time.
68///
69/// Implemented above the query engine (in the frontend) and injected into the
70/// catalog manager *after* construction — the provider needs the engine, which
71/// needs the catalog manager — so this late binding breaks the
72/// `catalog -> query` dependency cycle.
73///
74/// `query_ctx` is the outer query's context, captured when the computed table
75/// is resolved: the derivation must read only sources that context may read
76/// and execute under it, inheriting the caller's permissions, cancellation and
77/// deadline (the RFC's derivation contract). `None` (context-less internal
78/// resolution) keeps the provider's default behaviour.
79#[async_trait::async_trait]
80pub trait EntityGraphProvider: Send + Sync {
81    /// Produces the entity registry (`semantic_entities`) rows for `catalog`.
82    /// `None` means no source table declared an entity.
83    async fn scan_entities(
84        &self,
85        catalog: &str,
86        request: ScanRequest,
87        query_ctx: Option<QueryContextRef>,
88    ) -> std::result::Result<Option<SendableRecordBatchStream>, BoxedError>;
89
90    /// Produces the relationship set (`semantic_relationships`) rows for `catalog`.
91    async fn scan_relationships(
92        &self,
93        catalog: &str,
94        request: ScanRequest,
95        query_ctx: Option<QueryContextRef>,
96    ) -> std::result::Result<Option<SendableRecordBatchStream>, BoxedError>;
97}

99/// Serves the computed graph tables under `greptime_private`, overlaid on the
100/// schema's physical tables the same way the `numbers` table overlays `public`
101/// (the system catalog is consulted before physical table resolution).
102pub(crate) struct SemanticGraphTableProvider {
103    catalog_name: String,
104    catalog_manager: Weak<dyn CatalogManager>,
105    /// The resolving query's context; captured per resolution (the provider is
106    /// built on demand, never cached), so it cannot leak across sessions.
107    query_ctx: Option<QueryContextRef>,
108}

110impl SemanticGraphTableProvider {
111    pub(crate) fn new(
112        catalog_name: String,
113        catalog_manager: Weak<dyn CatalogManager>,
114        query_ctx: Option<QueryContextRef>,
115    ) -> Self {
116        Self {
117            catalog_name,
118            catalog_manager,
119            query_ctx,
120        }
121    }
122
123    pub(crate) fn table_names() -> Vec<String> {
124        vec![
125            SEMANTIC_ENTITIES.to_string(),
126            SEMANTIC_RELATIONSHIPS.to_string(),
127        ]
128    }
129
130    pub(crate) fn table_exists(name: &str) -> bool {
131        name == SEMANTIC_ENTITIES || name == SEMANTIC_RELATIONSHIPS
132    }
133
134    pub(crate) fn table(&self, name: &str) -> Option<TableRef> {
135        self.build_table(name)
136    }

175/// Schema of `semantic_entities` — the node set of the graph, one row per entity
176/// observed in a time window. Must match the registry derivation projection.
177///
178/// Columns:
179/// - `observed_at`   — TIME INDEX; the 60s time bucket the entity was observed in.
180/// - `window_start`  — start of that observation window.
181/// - `window_end`    — end of the window (`window_start` + 60s).
182/// - `fresh_until`   — time up to which the entity is considered present; equals
183///   `window_end` for derived rows (the graph is a sliding window, not a
184///   current-state table: an entity exists in a query window only if it has
185///   observed evidence there).
186/// - `entity_type`   — the entity's type, e.g. `service`, `host`, `k8s.pod`,
187///   `process`, `service.instance` (the OTel-style, possibly dotted, type).
188/// - `entity_id`     — canonical identifier: the value verbatim for a
189///   single-attribute identity, or a sorted `k=v,k=v` rendering for a composite.
190/// - `entity_id_attrs` — JSON object of the identifying attributes (the
191///   escaping-safe source of truth for composite ids); NULL for single-attribute ids.
192/// - `scope`         — namespace/environment the id is scoped to; empty when none.
193/// - `descriptive`   — JSON snapshot of the entity's descriptive (non-identifying)
194///   attributes; NULL when no descriptive columns were declared.
195/// - `source_tables` — JSON array of the telemetry tables that contributed this entity.
196static ENTITIES_SCHEMA: LazyLock<SchemaRef> = LazyLock::new(|| {
197    Arc::new(Schema::new(vec![
198        ColumnSchema::new(OBSERVED_AT_COLUMN, ts(), false).with_time_index(true),
199        ColumnSchema::new(WINDOW_START_COLUMN, ts(), true),
200        ColumnSchema::new(WINDOW_END_COLUMN, ts(), true),
201        ColumnSchema::new(FRESH_UNTIL_COLUMN, ts(), true),
202        ColumnSchema::new(ENTITY_TYPE_COLUMN, string(), false),
203        ColumnSchema::new(ENTITY_ID_COLUMN, string(), false),
204        ColumnSchema::new(ENTITY_ID_ATTRS_COLUMN, json(), true),
205        ColumnSchema::new(ENTITY_SCOPE_COLUMN, string(), true),
206        ColumnSchema::new(ENTITY_DESCRIPTIVE_COLUMN, json(), true),
207        ColumnSchema::new(SOURCE_TABLES_COLUMN, json(), true),
208    ]))
209});
210
211/// Schema of `semantic_relationships` — the edge set of the graph, one row per
212/// edge observed in a time window. This is the 16-column contract every derived
213/// branch and the declared-edge table must project for the top-level `UNION ALL`.
214///
215/// Columns:
216/// - `observed_at`   — TIME INDEX; the 60s time bucket the edge was observed in.
217/// - `window_start` / `window_end` — the observation window (`window_start` + 60s).
218/// - `fresh_until`   — time up to which the edge is considered live; equals
219///   `window_end` for derived edges (from `valid_until` for declared edges).
220/// - `src_type` / `src_id` — type and canonical id of the source endpoint.
221/// - `dst_type` / `dst_id` — type and canonical id of the destination endpoint.
222/// - `rel_type`      — relationship kind, e.g. `calls`, `runs_on`, `contains`,
223///   `part_of`, `depends_on` (direction is src → dst; the inverse is a query concern).
224/// - `provenance`    — how the edge was obtained: `trace` (derived from spans),
225///   `attribute` (shared-identity join), `declared` (hand-inserted), or `agent`
226///   (agent-inferred). Part of the edge identity, so edges of different provenance
227///   for the same pair coexist.
228/// - `confidence`    — derivation certainty in `[0, 1]`: `1.0` for paired or
229///   declared edges, lower for virtual-node or agent-inferred edges. It does
230///   not correct for trace sampling.
231/// - `request_count` — RED: number of requests over the window (`calls` edges).
232/// - `error_count`   — RED: number of errored requests over the window.
233/// - `duration_sum`  — RED: sum of request durations, in seconds, over the window.
234/// - `duration_count`— RED: number of durations summed (pair with `duration_sum`
235///   to get an average).
236/// - `attributes`    — JSON of edge attributes, e.g. `connection_type`,
237///   `db.system`, `peer.service`.
238static RELATIONSHIPS_SCHEMA: LazyLock<SchemaRef> = LazyLock::new(|| {
239    Arc::new(Schema::new(vec![
240        ColumnSchema::new(OBSERVED_AT_COLUMN, ts(), false).with_time_index(true),
241        ColumnSchema::new(WINDOW_START_COLUMN, ts(), true),
242        ColumnSchema::new(WINDOW_END_COLUMN, ts(), true),
243        ColumnSchema::new(FRESH_UNTIL_COLUMN, ts(), true),
244        ColumnSchema::new(SRC_TYPE_COLUMN, string(), false),
245        ColumnSchema::new(SRC_ID_COLUMN, string(), false),
246        ColumnSchema::new(DST_TYPE_COLUMN, string(), false),
247        ColumnSchema::new(DST_ID_COLUMN, string(), false),
248        ColumnSchema::new(REL_TYPE_COLUMN, string(), false),
249        ColumnSchema::new(PROVENANCE_COLUMN, string(), false),
250        ColumnSchema::new(
251            CONFIDENCE_COLUMN,
252            ConcreteDataType::float64_datatype(),
253            true,
254        ),
255        ColumnSchema::new(
256            REQUEST_COUNT_COLUMN,
257            ConcreteDataType::int64_datatype(),
258            true,
259        ),
260        ColumnSchema::new(ERROR_COUNT_COLUMN, ConcreteDataType::int64_datatype(), true),
261        ColumnSchema::new(
262            DURATION_SUM_COLUMN,
263            ConcreteDataType::float64_datatype(),
264            true,
265        ),
266        ColumnSchema::new(
267            DURATION_COUNT_COLUMN,
268            ConcreteDataType::int64_datatype(),
269            true,
270        ),
271        ColumnSchema::new(EDGE_ATTRIBUTES_COLUMN, json(), true),
272    ]))
273});

275/// Which computed table this shell represents, so the two share one forwarder.
276#[derive(Clone, Copy)]
277enum GraphTableKind {
278    Entities,
279    Relationships,
280}
281
282/// Forwarder for a computed entity-graph table.
283struct SemanticGraphTable {
284    kind: GraphTableKind,
285    schema: SchemaRef,
286    catalog_name: String,
287    catalog_manager: Weak<dyn CatalogManager>,
288    query_ctx: Option<QueryContextRef>,
289}
290
291impl SemanticGraphTable {
292    fn new(
293        kind: GraphTableKind,
294        catalog_name: String,
295        catalog_manager: Weak<dyn CatalogManager>,
296        query_ctx: Option<QueryContextRef>,
297    ) -> Self {
298        let schema = match kind {
299            GraphTableKind::Entities => ENTITIES_SCHEMA.clone(),
300            GraphTableKind::Relationships => RELATIONSHIPS_SCHEMA.clone(),
301        };
302        Self {
303            kind,
304            schema,
305            catalog_name,
306            catalog_manager,
307            query_ctx,
308        }
309    }
310
311    async fn derive(
312        kind: GraphTableKind,
313        catalog: String,
314        catalog_manager: Weak<dyn CatalogManager>,
315        request: ScanRequest,
316        query_ctx: Option<QueryContextRef>,
317    ) -> Result<Option<SendableRecordBatchStream>> {
318        let provider = utils::entity_graph_provider(&catalog_manager)?;
319        // No provider (engine not up / non-frontend node): stream empty.
320        let Some(provider) = provider else {
321            return Ok(None);
322        };
323        match kind {
324            GraphTableKind::Entities => provider.scan_entities(&catalog, request, query_ctx).await,
325            GraphTableKind::Relationships => {
326                provider
327                    .scan_relationships(&catalog, request, query_ctx)
328                    .await
329            }
330        }
331        .context(InternalSnafu)
332    }
333
334    fn align_schema(
335        stream: SendableRecordBatchStream,
336        schema: SchemaRef,
337    ) -> SendableRecordBatchStream {
338        let batch_schema = schema.clone();
339        let arrow_schema = schema.arrow_schema().clone();
340        let batches = stream.map(move |batch| {
341            let batch = batch?;
342            // The derivation output is structurally identical, but its JSON
343            // fields are plain Binary without the declared extension metadata.
344            let batch = DfRecordBatch::try_new(
345                arrow_schema.clone(),
346                batch.into_df_record_batch().columns().to_vec(),
347            )
348            .context(common_recordbatch::error::NewDfRecordBatchSnafu)?;
349            Ok(RecordBatch::from_df_record_batch(
350                batch_schema.clone(),
351                batch,
352            ))
353        });
354        Box::pin(RecordBatchStreamWrapper::new(schema, Box::pin(batches)))
355    }
356}
357
358impl SystemTable for SemanticGraphTable {
359    fn table_id(&self) -> TableId {
360        match self.kind {
361            GraphTableKind::Entities => SEMANTIC_ENTITIES_TABLE_ID,
362            GraphTableKind::Relationships => SEMANTIC_RELATIONSHIPS_TABLE_ID,
363        }
364    }
365
366    fn table_name(&self) -> &'static str {
367        match self.kind {
368            GraphTableKind::Entities => SEMANTIC_ENTITIES,
369            GraphTableKind::Relationships => SEMANTIC_RELATIONSHIPS,
370        }
371    }
372
373    fn schema(&self) -> SchemaRef {
374        self.schema.clone()
375    }
376
377    fn to_stream(&self, request: ScanRequest) -> Result<SendableRecordBatchStream> {
378        let schema = self.schema.clone();
379        let kind = self.kind;
380        let catalog = self.catalog_name.clone();
381        let catalog_manager = self.catalog_manager.clone();
382        let query_ctx = self.query_ctx.clone();
383
384        let stream_schema = schema.clone();
385        let stream = async move {
386            let stream = Self::derive(kind, catalog, catalog_manager, request, query_ctx)
387                .await
388                .map_err(BoxedError::new)
389                .context(common_recordbatch::error::ExternalSnafu)?;
390            Ok(match stream {
391                Some(stream) => Self::align_schema(stream, stream_schema.clone()),
392                None => Box::pin(EmptyRecordBatchStream::new(stream_schema.clone())),
393            })
394        };
395
396        Ok(Box::pin(AsyncRecordBatchStreamAdapter::new(
397            schema,
398            Box::pin(stream),
399        )))
400    }
401}
