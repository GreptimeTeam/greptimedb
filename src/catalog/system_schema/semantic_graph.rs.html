15//! The computed entity-graph tables `greptime_private.semantic_entities` and
16//! `greptime_private.semantic_relationships`.
17//!
18//! They live in `greptime_private`, not `information_schema`: scanning them
19//! triggers read-time derivation over telemetry tables (trace self-joins, ...),
20//! which breaks the "cheap, metadata-only" expectation users have of
21//! `information_schema`. `greptime_private` already signals "system-managed,
22//! computed data objects", and it is also where the physical declared-edge
23//! table will live (a follow-up), so derived and declared edges will share one
24//! schema.
25//!
26//! These are thin forwarders: their rows are derived at read time by the injected
27//! [`EntityGraphProvider`], which enumerates the `table_semantics` declarations,
28//! builds typed DataFusion derivation plans, and executes them via the query
29//! engine. When no provider is injected (e.g. before the engine is up, or on a
30//! non-frontend node) they stream empty. The fixed schemas here must match the
31//! columns the provider's plans project; JSON columns are `json` (JSONB), whose
32//! Arrow storage type is `Binary` — the derived batches are rebuilt against the
33//! declared Arrow schema (which carries the `json` extension metadata) in
34//! [`SystemTable::to_stream`].

62/// Produces the rows of the computed entity-graph tables
63/// (`semantic_entities` / `semantic_relationships`) at read time.
64///
65/// The computed tables are thin forwarders to this provider, which is
66/// implemented above the query engine (in the frontend): it enumerates the entity
67/// declarations from `table_semantics`, builds the typed derivation plans, and
68/// executes them. It is injected into the catalog manager *after* construction —
69/// the provider needs the engine, which needs the catalog manager — so this late
70/// binding breaks the `catalog -> query` dependency cycle. Keeping derivation out
71/// of `catalog` also respects the crate layering (the plan builders live in
72/// `operator`). See `docs/rfcs/2026-06-25-entity-relationships-and-graph-query.md`.
73#[async_trait::async_trait]
74pub trait EntityGraphProvider: Send + Sync {
75    /// Produces the entity registry (`semantic_entities`) rows for `catalog`.
76    /// `None` means no source table declared an entity.
77    async fn scan_entities(
78        &self,
79        catalog: &str,
80        request: ScanRequest,
81    ) -> std::result::Result<Option<SendableRecordBatchStream>, BoxedError>;
82
83    /// Produces the relationship set (`semantic_relationships`) rows for `catalog`.
84    async fn scan_relationships(
85        &self,
86        catalog: &str,
87        request: ScanRequest,
88    ) -> std::result::Result<Option<SendableRecordBatchStream>, BoxedError>;
89}

158/// Schema of `semantic_entities` — the node set of the graph, one row per entity
159/// observed in a time window. Must match the registry derivation projection.
160///
161/// Columns:
162/// - `observed_at`   — TIME INDEX; the 60s time bucket the entity was observed in.
163/// - `window_start`  — start of that observation window.
164/// - `window_end`    — end of the window (`window_start` + 60s).
165/// - `fresh_until`   — time up to which the entity is considered present; equals
166///   `window_end` for derived rows (the graph is a sliding window, not a
167///   current-state table: an entity exists in a query window only if it has
168///   observed evidence there).
169/// - `entity_type`   — the entity's type, e.g. `service`, `host`, `k8s.pod`,
170///   `process`, `service.instance` (the OTel-style, possibly dotted, type).
171/// - `entity_id`     — canonical identifier: the value verbatim for a
172///   single-attribute identity, or a sorted `k=v,k=v` rendering for a composite.
173/// - `entity_id_attrs` — JSON object of the identifying attributes (the
174///   escaping-safe source of truth for composite ids); NULL for single-attribute ids.
175/// - `scope`         — namespace/environment the id is scoped to; empty when none.
176/// - `descriptive`   — JSON snapshot of the entity's descriptive (non-identifying)
177///   attributes; NULL when no descriptive columns were declared.
178/// - `source_tables` — JSON array of the telemetry tables that contributed this entity.
179static ENTITIES_SCHEMA: LazyLock<SchemaRef> = LazyLock::new(|| {
180    Arc::new(Schema::new(vec![
181        ColumnSchema::new("observed_at", ts(), false).with_time_index(true),
182        ColumnSchema::new("window_start", ts(), true),
183        ColumnSchema::new("window_end", ts(), true),
184        ColumnSchema::new("fresh_until", ts(), true),
185        ColumnSchema::new("entity_type", string(), false),
186        ColumnSchema::new("entity_id", string(), false),
187        ColumnSchema::new("entity_id_attrs", json(), true),
188        ColumnSchema::new("scope", string(), true),
189        ColumnSchema::new("descriptive", json(), true),
190        ColumnSchema::new("source_tables", json(), true),
191    ]))
192});
193
194/// Schema of `semantic_relationships` — the edge set of the graph, one row per
195/// edge observed in a time window. This is the 16-column contract every derived
196/// branch and the declared-edge table must project for the top-level `UNION ALL`.
197///
198/// Columns:
199/// - `observed_at`   — TIME INDEX; the 60s time bucket the edge was observed in.
200/// - `window_start` / `window_end` — the observation window (`window_start` + 60s).
201/// - `fresh_until`   — time up to which the edge is considered live; equals
202///   `window_end` for derived edges (from `valid_until` for declared edges).
203/// - `src_type` / `src_id` — type and canonical id of the source endpoint.
204/// - `dst_type` / `dst_id` — type and canonical id of the destination endpoint.
205/// - `rel_type`      — relationship kind, e.g. `calls`, `runs_on`, `contains`,
206///   `part_of`, `depends_on` (direction is src → dst; the inverse is a query concern).
207/// - `provenance`    — how the edge was obtained: `trace` (derived from spans),
208///   `attribute` (shared-identity join), `declared` (hand-inserted), or `agent`
209///   (agent-inferred). Part of the edge identity, so edges of different provenance
210///   for the same pair coexist.
211/// - `confidence`    — derivation certainty in `[0, 1]`: `1.0` for paired or
212///   declared edges, lower for virtual-node or agent-inferred edges. It does
213///   not correct for trace sampling.
214/// - `request_count` — RED: number of requests over the window (`calls` edges).
215/// - `error_count`   — RED: number of errored requests over the window.
216/// - `duration_sum`  — RED: sum of request durations, in seconds, over the window.
217/// - `duration_count`— RED: number of durations summed (pair with `duration_sum`
218///   to get an average).
219/// - `attributes`    — JSON of edge attributes, e.g. `connection_type`,
220///   `db.system`, `peer.service`.
221static RELATIONSHIPS_SCHEMA: LazyLock<SchemaRef> = LazyLock::new(|| {
222    Arc::new(Schema::new(vec![
223        ColumnSchema::new("observed_at", ts(), false).with_time_index(true),
224        ColumnSchema::new("window_start", ts(), true),
225        ColumnSchema::new("window_end", ts(), true),
226        ColumnSchema::new("fresh_until", ts(), true),
227        ColumnSchema::new("src_type", string(), false),
228        ColumnSchema::new("src_id", string(), false),
229        ColumnSchema::new("dst_type", string(), false),
230        ColumnSchema::new("dst_id", string(), false),
231        ColumnSchema::new("rel_type", string(), false),
232        ColumnSchema::new("provenance", string(), false),
233        ColumnSchema::new("confidence", ConcreteDataType::float64_datatype(), true),
234        ColumnSchema::new("request_count", ConcreteDataType::int64_datatype(), true),
235        ColumnSchema::new("error_count", ConcreteDataType::int64_datatype(), true),
236        ColumnSchema::new("duration_sum", ConcreteDataType::float64_datatype(), true),
237        ColumnSchema::new("duration_count", ConcreteDataType::int64_datatype(), true),
238        ColumnSchema::new("attributes", json(), true),
239    ]))
240});

242/// Which computed table this shell represents, so the two share one forwarder.
243#[derive(Clone, Copy)]
244enum GraphTableKind {
245    Entities,
246    Relationships,
247}
248
249/// Forwarder for a computed entity-graph table.
250struct SemanticGraphTable {
251    kind: GraphTableKind,
252    schema: SchemaRef,
253    catalog_name: String,
254    catalog_manager: Weak<dyn CatalogManager>,
255}
256
257impl SemanticGraphTable {
258    fn new(
259        kind: GraphTableKind,
260        catalog_name: String,
261        catalog_manager: Weak<dyn CatalogManager>,
262    ) -> Self {
263        let schema = match kind {
264            GraphTableKind::Entities => ENTITIES_SCHEMA.clone(),
265            GraphTableKind::Relationships => RELATIONSHIPS_SCHEMA.clone(),
266        };
267        Self {
268            kind,
269            schema,
270            catalog_name,
271            catalog_manager,
272        }
273    }
274
275    async fn derive(
276        kind: GraphTableKind,
277        catalog: String,
278        catalog_manager: Weak<dyn CatalogManager>,
279        request: ScanRequest,
280    ) -> Result<Option<SendableRecordBatchStream>> {
281        let provider = utils::entity_graph_provider(&catalog_manager)?;
282        // No provider (engine not up / non-frontend node): stream empty.
283        let Some(provider) = provider else {
284            return Ok(None);
285        };
286        match kind {
287            GraphTableKind::Entities => provider.scan_entities(&catalog, request).await,
288            GraphTableKind::Relationships => provider.scan_relationships(&catalog, request).await,
289        }
290        .context(InternalSnafu)
291    }
292
293    fn align_schema(
294        stream: SendableRecordBatchStream,
295        schema: SchemaRef,
296    ) -> SendableRecordBatchStream {
297        let batch_schema = schema.clone();
298        let arrow_schema = schema.arrow_schema().clone();
299        let batches = stream.map(move |batch| {
300            let batch = batch?;
301            // The derivation output is structurally identical, but its JSON
302            // fields are plain Binary without the declared extension metadata.
303            let batch = DfRecordBatch::try_new(
304                arrow_schema.clone(),
305                batch.into_df_record_batch().columns().to_vec(),
306            )
307            .context(common_recordbatch::error::NewDfRecordBatchSnafu)?;
308            Ok(RecordBatch::from_df_record_batch(
309                batch_schema.clone(),
310                batch,
311            ))
312        });
313        Box::pin(RecordBatchStreamWrapper::new(schema, Box::pin(batches)))
314    }
315}
316
317impl SystemTable for SemanticGraphTable {
318    fn table_id(&self) -> TableId {
319        match self.kind {
320            GraphTableKind::Entities => SEMANTIC_ENTITIES_TABLE_ID,
321            GraphTableKind::Relationships => SEMANTIC_RELATIONSHIPS_TABLE_ID,
322        }
323    }
324
325    fn table_name(&self) -> &'static str {
326        match self.kind {
327            GraphTableKind::Entities => SEMANTIC_ENTITIES,
328            GraphTableKind::Relationships => SEMANTIC_RELATIONSHIPS,
329        }
330    }
331
332    fn schema(&self) -> SchemaRef {
333        self.schema.clone()
334    }
335
336    fn to_stream(&self, request: ScanRequest) -> Result<SendableRecordBatchStream> {
337        let schema = self.schema.clone();
338        let kind = self.kind;
339        let catalog = self.catalog_name.clone();
340        let catalog_manager = self.catalog_manager.clone();
341
342        let stream_schema = schema.clone();
343        let stream = async move {
344            let stream = Self::derive(kind, catalog, catalog_manager, request)
345                .await
346                .map_err(BoxedError::new)
347                .context(common_recordbatch::error::ExternalSnafu)?;
348            Ok(match stream {
349                Some(stream) => Self::align_schema(stream, stream_schema.clone()),
350                None => Box::pin(EmptyRecordBatchStream::new(stream_schema.clone())),
351            })
352        };
353
354        Ok(Box::pin(AsyncRecordBatchStreamAdapter::new(
355            schema,
356            Box::pin(stream),
357        )))
358    }
359}
