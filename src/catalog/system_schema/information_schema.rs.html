100const DENSE_PRIMARY_KEY_ENCODING: &str = "dense";
101const SPARSE_PRIMARY_KEY_ENCODING: &str = "sparse";

103pub(crate) fn primary_key_encoding_index_type(options: &HashMap<String, String>) -> &'static str {
104    options
105        .get(PRIMARY_KEY_ENCODING)
106        .or_else(|| options.get(MEMTABLE_PARTITION_TREE_PRIMARY_KEY_ENCODING))
107        .map(|value| {
108            if value.eq_ignore_ascii_case(SPARSE_PRIMARY_KEY_ENCODING) {
109                SPARSE_PRIMARY_KEY_ENCODING
110            } else {
111                DENSE_PRIMARY_KEY_ENCODING
112            }
113        })
114        .unwrap_or(DENSE_PRIMARY_KEY_ENCODING)
115}

167/// The visibility guaranteed by an information-schema table provider.
168#[derive(Debug, Clone, Copy, PartialEq, Eq)]
169pub enum InformationSchemaTableScope {
170    /// Every query is filtered to the catalog supplied when constructing the provider.
171    Catalog,
172    /// Catalog-independent data that is safe for authenticated non-admin users to read.
173    PublicGlobal,
174    /// Shared data without a non-admin visibility guarantee, including compatibility stubs.
175    Global,
176}
177
178impl InformationSchemaTableScope {
179    pub const fn as_str(self) -> &'static str {
180        match self {
181            Self::Catalog => "catalog",
182            Self::PublicGlobal => "public_global",
183            Self::Global => "global",
184        }
185    }
186}

346impl InformationSchemaProvider {
347    pub fn new(
348        catalog_name: String,
349        catalog_manager: Weak<dyn CatalogManager>,
350        flow_metadata_manager: Arc<FlowMetadataManager>,
351        process_manager: Option<ProcessManagerRef>,
352        kv_backend: KvBackendRef,
353    ) -> Self {
354        let mut provider = Self {
355            catalog_name,
356            catalog_manager,
357            flow_metadata_manager,
358            process_manager,
359            tables: HashMap::new(),
360            kv_backend,
361            extra_table_factories: HashMap::new(),
362        };
363
364        provider.build_tables();
365
366        provider
367    }
368
369    pub(crate) fn with_extra_table_factories(
370        mut self,
371        factories: HashMap<String, InformationSchemaTableFactoryRef>,
372    ) -> Self {
373        self.extra_table_factories = factories;
374        self.build_tables();
375        self
376    }
377
378    fn build_tables(&mut self) {
379        let mut table_names = HashSet::new();
380
381        // SECURITY NOTE:
382        // Carefully consider the tables that may expose sensitive cluster configurations,
383        // authentication details, and other critical information.
384        // Only put these tables under `greptime` catalog to prevent info leak.
385        if self.catalog_name == DEFAULT_CATALOG_NAME {
386            table_names.extend([
387                REGION_PEERS,
388                CLUSTER_INFO,
389                PROCEDURE_INFO,
390                REGION_STATISTICS,
391                COMPACTION_TASKS,
392                REGION_INFO,
393                SSTS_MANIFEST,
394                SSTS_STORAGE,
395                SSTS_INDEX_META,
396            ]);
397        }
398
399        table_names.extend([
400            BUILD_INFO,
401            TABLES,
402            VIEWS,
403            SCHEMATA,
404            COLUMNS,
405            KEY_COLUMN_USAGE,
406            TABLE_CONSTRAINTS,
407            STATISTICS,
408            FLOWS,
409            FLOW_STATISTICS,
410            #[cfg(feature = "enterprise")]
411            RECYCLE_BIN,
412            TABLE_SEMANTICS,
413            PARTITIONS,
414        ]);
415        if self.process_manager.is_some() {
416            table_names.insert(PROCESS_LIST);
417        }
418        table_names.extend(MEMORY_TABLES.iter().copied());
419        table_names.extend(self.extra_table_factories.keys().map(String::as_str));
420        self.tables = table_names
421            .into_iter()
422            .map(|name| {
423                let table = self.information_table(name).expect(name);
424                let options = TableOptions {
425                    extra_options: HashMap::from([(
426                        SEMANTIC_TABLE_SCOPE.to_string(),
427                        table.scope().as_str().to_string(),
428                    )]),
429                    ..Default::default()
430                };
431                let table = build_system_table(
432                    self.catalog_name.clone(),
433                    INFORMATION_SCHEMA_NAME,
434                    table,
435                    options,
436                );
437                (name.to_string(), table)
438            })
439            .collect();
440    }
441}

453/// The `InformationExtension` trait provides the extension methods for the `information_schema` tables.
454#[async_trait::async_trait]
455pub trait InformationExtension {
456    type Error: ErrorExt;
457
458    /// Gets the nodes information.
459    async fn nodes(&self) -> std::result::Result<Vec<NodeInfo>, Self::Error>;
460
461    /// Gets the procedures information.
462    async fn procedures(&self) -> std::result::Result<Vec<(String, ProcedureInfo)>, Self::Error>;
463
464    /// Gets the region statistics.
465    async fn region_stats(&self) -> std::result::Result<Vec<RegionStat>, Self::Error>;
466
467    /// Get the flow statistics. If no flownode is available, return `None`.
468    async fn flow_stats(&self) -> std::result::Result<Option<FlowStat>, Self::Error>;
469
470    /// Inspects the datanode.
471    async fn inspect_datanode(
472        &self,
473        request: DatanodeInspectRequest,
474    ) -> std::result::Result<SendableRecordBatchStream, Self::Error>;
475
476    /// Whether this extension can execute an inspection query on every datanode.
477    fn supports_inspect_query_plan(&self) -> bool {
478        false
479    }
480
481    /// Builds a plan that executes an inspection query independently on each datanode.
482    /// The caller supplies a plan over reserved inspection tables and retains any
483    /// required global aggregation. The returned schema is the query output schema.
484    fn inspect_datanode_query_plan(
485        &self,
486        _plan: LogicalPlan,
487    ) -> std::result::Result<Option<Arc<dyn ExecutionPlan>>, Self::Error> {
488        Ok(None)
489    }
490
491    /// Builds a physical plan for datanode inspect if the extension can expose
492    /// the distributed fan-in semantics to DataFusion.
493    fn inspect_datanode_plan(
494        &self,
495        _request: DatanodeInspectRequest,
496        _schema: SchemaRef,
497    ) -> std::result::Result<Option<Arc<dyn ExecutionPlan>>, Self::Error> {
498        Ok(None)
499    }
500}

502/// The request to inspect the datanode.
503#[derive(Debug, Clone, PartialEq)]
504pub struct DatanodeInspectRequest {
505    /// Kind to fetch from datanode.
506    pub kind: DatanodeInspectKind,
507
508    /// Pushdown scan configuration (projection/predicate/limit) for the returned stream.
509    /// This allows server-side filtering to reduce I/O and network costs.
510    pub scan: ScanRequest,
511}
512
513/// The kind of the datanode inspect request.
514#[derive(Debug, Clone, Copy, PartialEq, Eq)]
515pub enum DatanodeInspectKind {
516    /// Live compaction cycles.
517    CompactionTasks,
518    /// List SST entries recorded in manifest
519    SstManifest,
520    /// List SST entries discovered in storage layer
521    SstStorage,
522    /// List index metadata collected from manifest
523    SstIndexMeta,
524    /// List region runtime and manifest info
525    RegionInfo,
526}
527
528impl DatanodeInspectRequest {
529    /// Builds a logical plan for the datanode inspect request.
530    pub fn build_plan(self) -> std::result::Result<LogicalPlan, DataFusionError> {
531        match self.kind {
532            DatanodeInspectKind::CompactionTasks => CompactionTaskEntry::build_plan(self.scan),
533            DatanodeInspectKind::SstManifest => ManifestSstEntry::build_plan(self.scan),
534            DatanodeInspectKind::SstStorage => StorageSstEntry::build_plan(self.scan),
535            DatanodeInspectKind::SstIndexMeta => PuffinIndexMetaEntry::build_plan(self.scan),
536            DatanodeInspectKind::RegionInfo => RegionInfoEntry::build_plan(self.scan),
537        }
538    }

662            STATISTICS,
663            FLOWS,
664            FLOW_STATISTICS,
665            TABLE_SEMANTICS,
666            PARTITIONS,
667            REGION_PEERS,
668            #[cfg(feature = "enterprise")]
669            RECYCLE_BIN,
670            "future_catalog_table",
671        ];
672        let public_global_tables = [
673            ENGINES,
674            CHARACTER_SETS,
675            COLLATIONS,
676            COLLATION_CHARACTER_SET_APPLICABILITY,
677            "future_public_table",
678        ];
679
680        for (catalog, with_process_manager) in [
681            (DEFAULT_CATALOG_NAME, true),
682            (DEFAULT_CATALOG_NAME, false),
683            ("tenant", true),
684            ("tenant", false),
685        ] {
686            let provider = InformationSchemaProvider::new(
687                catalog.to_string(),
688                Arc::downgrade(&manager),
689                Arc::new(FlowMetadataManager::new(backend.clone())),
690                with_process_manager.then(|| Arc::new(ProcessManager::new(String::new(), None))),
691                backend.clone(),
692            );
693            for name in [
694                BUILD_INFO,
695                ENGINES,
696                CHARACTER_SETS,
697                COLLATIONS,
698                COLLATION_CHARACTER_SET_APPLICABILITY,
699            ] {
700                assert_eq!(
701                    provider
702                        .table(name)
703                        .unwrap()
704                        .table_info()
705                        .meta
706                        .options
707                        .extra_options
708                        .get(SEMANTIC_TABLE_SCOPE)
709                        .map(String::as_str),
710                    Some(PublicGlobal.as_str()),
711                    "{catalog}.{name}"
712                );
713            }
714            // An extension replacing a built-in supplies the replacement table's scope.
715            let provider = provider.with_extra_table_factories(factories.clone());
716            let expected_count =
717                45 + usize::from(with_process_manager) + usize::from(cfg!(feature = "enterprise"))
718                    - if catalog == DEFAULT_CATALOG_NAME {
719                        0
720                    } else {
721                        9
722                    };
723            assert_eq!(provider.tables().len(), expected_count);
724            for (name, table) in provider.tables() {
725                let scope = if catalog_tables.contains(&name.as_str()) {
726                    Catalog
727                } else if public_global_tables.contains(&name.as_str()) {
728                    PublicGlobal
729                } else {
730                    Global
731                };
732                assert_eq!(
733                    table
734                        .table_info()
735                        .meta
736                        .options
737                        .extra_options
738                        .get(SEMANTIC_TABLE_SCOPE)
739                        .map(String::as_str),
740                    Some(scope.as_str()),
741                    "{catalog}.{name}"
742                );
743            }
744            assert_eq!(
745                provider.table(REGION_PEERS).is_some(),
746                catalog == DEFAULT_CATALOG_NAME
747            );
748            assert_eq!(provider.table(PROCESS_LIST).is_some(), with_process_manager);
749            if let Some(table) = provider.table(PROCESS_LIST) {
750                assert!(table.schema().column_schema_by_name("catalog").is_some());
751            }
752        }
753    }
