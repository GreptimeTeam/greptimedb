100pub(crate) fn primary_key_encoding_index_type(options: &HashMap<String, String>) -> &'static str {
101    options
102        .get(PRIMARY_KEY_ENCODING)
103        .or_else(|| options.get(MEMTABLE_PARTITION_TREE_PRIMARY_KEY_ENCODING))
104        .map(|value| {
105            if value.eq_ignore_ascii_case(SPARSE_PRIMARY_KEY_ENCODING) {
106                SPARSE_PRIMARY_KEY_ENCODING
107            } else {
108                DENSE_PRIMARY_KEY_ENCODING
109            }
110        })
111        .unwrap_or(DENSE_PRIMARY_KEY_ENCODING)
112}

165pub enum InformationSchemaTableScope {
166    /// Every query is filtered to the catalog supplied when constructing the provider.

336impl InformationSchemaProvider {
337    pub fn new(
338        catalog_name: String,
339        catalog_manager: Weak<dyn CatalogManager>,
340        flow_metadata_manager: Arc<FlowMetadataManager>,
341        process_manager: Option<ProcessManagerRef>,
342        kv_backend: KvBackendRef,
343    ) -> Self {
344        let mut provider = Self {
345            catalog_name,
346            catalog_manager,
347            flow_metadata_manager,
348            process_manager,
349            tables: HashMap::new(),
350            kv_backend,
351            extra_table_factories: HashMap::new(),
352        };
353
354        provider.build_tables();
355
356        provider
357    }
358
359    pub(crate) fn with_extra_table_factories(
360        mut self,
361        factories: HashMap<String, InformationSchemaTableFactoryRef>,
362    ) -> Self {
363        self.extra_table_factories = factories;
364        self.build_tables();
365        self
366    }
367
368    fn build_tables(&mut self) {
369        let mut table_names = HashSet::new();
370
371        // SECURITY NOTE:
372        // Carefully consider the tables that may expose sensitive cluster configurations,
373        // authentication details, and other critical information.
374        // Only put these tables under `greptime` catalog to prevent info leak.
375        if self.catalog_name == DEFAULT_CATALOG_NAME {
376            table_names.extend([
377                BUILD_INFO,
378                REGION_PEERS,
379                CLUSTER_INFO,
380                PROCEDURE_INFO,
381                REGION_STATISTICS,
382                REGION_INFO,
383                SSTS_MANIFEST,
384                SSTS_STORAGE,
385                SSTS_INDEX_META,
386            ]);
387        }
388
389        table_names.extend([
390            TABLES,
391            VIEWS,
392            SCHEMATA,
393            COLUMNS,
394            KEY_COLUMN_USAGE,
395            TABLE_CONSTRAINTS,
396            STATISTICS,
397            FLOWS,
398            FLOW_STATISTICS,
399            #[cfg(feature = "enterprise")]
400            RECYCLE_BIN,
401            TABLE_SEMANTICS,
402            PARTITIONS,
403        ]);
404        if self.process_manager.is_some() {
405            table_names.insert(PROCESS_LIST);
406        }
407        table_names.extend(MEMORY_TABLES.iter().copied());
408        table_names.extend(self.extra_table_factories.keys().map(String::as_str));
409        self.tables = table_names
410            .into_iter()
411            .map(|name| {
412                let table = self.information_table(name).expect(name);
413                let options = TableOptions {
414                    extra_options: HashMap::from([(
415                        SEMANTIC_TABLE_SCOPE.to_string(),
416                        table.scope().as_str().to_string(),
417                    )]),
418                    ..Default::default()
419                };
420                let table = build_system_table(
421                    self.catalog_name.clone(),
422                    INFORMATION_SCHEMA_NAME,
423                    table,
424                    options,
425                );
426                (name.to_string(), table)
427            })
428            .collect();
429    }
430}

473    }

476/// The request to inspect the datanode.
477#[derive(Debug, Clone, PartialEq)]
478pub struct DatanodeInspectRequest {
479    /// Kind to fetch from datanode.
480    pub kind: DatanodeInspectKind,
481
482    /// Pushdown scan configuration (projection/predicate/limit) for the returned stream.
483    /// This allows server-side filtering to reduce I/O and network costs.
484    pub scan: ScanRequest,
485}
486
487/// The kind of the datanode inspect request.
488#[derive(Debug, Clone, Copy, PartialEq, Eq)]
489pub enum DatanodeInspectKind {
490    /// List SST entries recorded in manifest
491    SstManifest,
492    /// List SST entries discovered in storage layer
493    SstStorage,
494    /// List index metadata collected from manifest
495    SstIndexMeta,
496    /// List region runtime and manifest info
497    RegionInfo,
498}
499
500impl DatanodeInspectRequest {
501    /// Builds a logical plan for the datanode inspect request.
502    pub fn build_plan(self) -> std::result::Result<LogicalPlan, DataFusionError> {
503        match self.kind {
504            DatanodeInspectKind::SstManifest => ManifestSstEntry::build_plan(self.scan),
505            DatanodeInspectKind::SstStorage => StorageSstEntry::build_plan(self.scan),
506            DatanodeInspectKind::SstIndexMeta => PuffinIndexMetaEntry::build_plan(self.scan),
507            DatanodeInspectKind::RegionInfo => RegionInfoEntry::build_plan(self.scan),
508        }
509    }
510}

673                    - if catalog == DEFAULT_CATALOG_NAME {
674                        0
675                    } else {
676                        9
677                    };
678            assert_eq!(provider.tables().len(), expected_count);
679            for (name, table) in provider.tables() {
680                let scope = if catalog_tables.contains(&name.as_str()) {
681                    Catalog
682                } else {
683                    Global
684                };
685                assert_eq!(
686                    table
687                        .table_info()
688                        .meta
689                        .options
690                        .extra_options
691                        .get(SEMANTIC_TABLE_SCOPE)
692                        .map(String::as_str),
693                    Some(scope.as_str()),
694                    "{catalog}.{name}"
695                );
696            }
697            assert_eq!(
698                provider.table(REGION_PEERS).is_some(),
699                catalog == DEFAULT_CATALOG_NAME
700            );
701            assert_eq!(provider.table(PROCESS_LIST).is_some(), with_process_manager);
702            if let Some(table) = provider.table(PROCESS_LIST) {
703                assert!(table.schema().column_schema_by_name("catalog").is_some());
704            }
705        }
706    }
