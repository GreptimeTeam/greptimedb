36/// Converts prepared rows in target-schema order without Prom-specific renaming.
37///
38/// Missing columns evaluate their defaults once per conversion. Explicit nulls
39/// remain nulls. Semantic types are checked against the table time index and
40/// primary-key indices.
41///
42/// # Panics
43///
44/// Panics if `rows.rows` is empty. Callers must skip empty writes before conversion.
45pub fn rows_to_record_batch(rows: &Rows, table_info: &TableInfo) -> Result<RecordBatch> {
46    assert!(!rows.rows.is_empty(), "prepared rows must not be empty");
47    let schema = &table_info.meta.schema;
48    let mut source_columns = HashMap::with_capacity(rows.schema.len());
49    for (index, source) in rows.schema.iter().enumerate() {
50        ensure!(
51            source_columns
52                .insert(source.column_name.as_str(), index)
53                .is_none(),
54            error::InvalidInsertRequestSnafu {
55                reason: format!("Duplicate input column {}", source.column_name),
56            }
57        );
58        let target = schema
59            .column_schema_by_name(&source.column_name)
60            .with_context(|| error::InvalidInsertRequestSnafu {
61                reason: format!("Unknown input column {}", source.column_name),
62            })?;
63        let data_type =
64            ColumnDataTypeWrapper::try_new(source.datatype, source.datatype_extension.clone())
65                .map_err(BoxedError::new)
66                .context(error::ExternalSnafu)?;
67        ensure!(
68            ConcreteDataType::from(data_type) == target.data_type,
69            error::InvalidInsertRequestSnafu {
70                reason: format!("Input datatype differs for column {}", source.column_name),
71            }
72        );
73        let semantic = SemanticType::try_from(source.semantic_type)
74            .ok()
75            .with_context(|| error::InvalidInsertRequestSnafu {
76                reason: format!("Invalid semantic type for column {}", source.column_name),
77            })?;
78        let is_tag = table_info.meta.primary_key_indices.iter().any(|&index| {
79            schema
80                .column_schemas()
81                .get(index)
82                .is_some_and(|column| column.name == source.column_name)
83        });
84        let expected_semantic = if target.is_time_index() {
85            SemanticType::Timestamp
86        } else if is_tag {
87            SemanticType::Tag
88        } else {
89            SemanticType::Field
90        };
91        ensure!(
92            semantic == expected_semantic,
93            error::InvalidInsertRequestSnafu {
94                reason: format!("Input semantics differ for column {}", source.column_name),
95            }
96        );
97    }
98    for row in &rows.rows {
99        ensure!(
100            row.values.len() == rows.schema.len(),
101            error::InvalidInsertRequestSnafu {
102                reason: format!(
103                    "Expected {} values, got {}",
104                    rows.schema.len(),
105                    row.values.len()
106                ),
107            }
108        );
109    }
110
111    let mut arrays = Vec::with_capacity(schema.num_columns());
112    for column in schema.column_schemas() {
113        let vector = if let Some(&index) = source_columns.get(column.name.as_str()) {
114            let mut builder = column.create_mutable_vector(rows.rows.len());
115            for row in &rows.rows {
116                let value = &row.values[index];
117                ensure!(
118                    value.value_data.is_some() || column.is_nullable(),
119                    error::InvalidInsertRequestSnafu {
120                        reason: format!("Null supplied for non-nullable column {}", column.name),
121                    }
122                );
123                ensure!(
124                    value_matches_type(
125                        value,
126                        rows.schema[index].datatype,
127                        rows.schema[index].datatype_extension.as_ref(),
128                    ),
129                    error::InvalidInsertRequestSnafu {
130                        reason: format!("Value datatype differs for column {}", column.name),
131                    }
132                );
133                builder
134                    .try_push_value_ref(&pb_value_to_value_ref(
135                        value,
136                        rows.schema[index].datatype_extension.as_ref(),
137                    ))
138                    .map_err(BoxedError::new)
139                    .context(error::ExternalSnafu)?;
140            }
141            builder.to_vector()
142        } else {
143            column
144                .create_default_vector(rows.rows.len())
145                .map_err(BoxedError::new)
146                .context(error::ExternalSnafu)?
147                .with_context(|| error::InvalidInsertRequestSnafu {
148                    reason: format!("Missing required column {}", column.name),
149                })?
150        };
151        arrays.push(vector.to_arrow_array());
152    }
153    let arrow_schema = align_schema_with_json_array(schema.arrow_schema().clone(), &arrays);
154    RecordBatch::try_new(arrow_schema, arrays).context(error::ComputeArrowSnafu)
155}

157// Validate nested values before the infallible protobuf conversion reads type extensions.
158pub(crate) fn value_matches_type(
159    value: &Value,
160    datatype: i32,
161    extension: Option<&ColumnDataTypeExtension>,
162) -> bool {
163    let Some(value_type) = proto_value_type(value) else {
164        return true;
165    };
166    let Ok(column_type) = ColumnDataType::try_from(datatype) else {
167        return false;
168    };
169    if !proto_value_type_match(column_type, value_type) {
170        return false;
171    }
172    match value.value_data.as_ref() {
173        Some(ValueData::ListValue(list)) => {
174            let Some(TypeExt::ListType(item)) = extension.and_then(|ext| ext.type_ext.as_ref())
175            else {
176                return false;
177            };
178            list.items.iter().all(|value| {
179                value_matches_type(value, item.datatype, item.datatype_extension.as_deref())
180            })
181        }
182        Some(ValueData::StructValue(value)) => {
183            let Some(TypeExt::StructType(schema)) = extension.and_then(|ext| ext.type_ext.as_ref())
184            else {
185                return false;
186            };
187            value.items.len() == schema.fields.len()
188                && value
189                    .items
190                    .iter()
191                    .zip(&schema.fields)
192                    .all(|(value, field)| {
193                        value_matches_type(value, field.datatype, field.datatype_extension.as_ref())
194                    })
195        }
196        _ => true,
197    }
198}

281    fn json_rows(column: &ColumnSchema, values: impl IntoIterator<Item = DtValue>) -> Rows {
282        let (kind, extension) = ColumnDataTypeWrapper::try_from(column.data_type.clone())
283            .unwrap()
284            .to_parts();

286        input.datatype_extension = extension;
287        Rows {
288            schema: vec![input],
289            rows: values
290                .into_iter()
291                .map(|value| Row {
292                    values: vec![to_grpc_value(value)],
293                })
294                .collect(),
295        }
296    }

637    #[test]
638    fn test_dynamic_default_is_evaluated_per_conversion() {
639        use std::time::{Duration, Instant};
640
641        use arrow::array::TimestampMillisecondArray;
642
643        let (mut rows, base_schema) = fixture();
644        let mut columns = base_schema.column_schemas().to_vec();
645        columns.push(
646            ColumnSchema::new(
647                "created",
648                ConcreteDataType::timestamp_millisecond_datatype(),
649                true,
650            )
651            .with_default_constraint(Some(ColumnDefaultConstraint::Function(
652                "current_timestamp()".to_string(),
653            )))
654            .unwrap(),
655        );
656        let schema = Schema::new(columns);
657        let timestamp = |batch: &RecordBatch| {
658            batch
659                .column(3)
660                .as_any()
661                .downcast_ref::<TimestampMillisecondArray>()
662                .unwrap()
663                .value(0)
664        };
665        let first = timestamp(&rows_to_record_batch(&rows, &table_info(schema.clone())).unwrap());
666        // The default uses wall-clock milliseconds, not Tokio's controllable
667        // clock. Poll its observable result with a bound instead of sleeping.
668        let deadline = Instant::now() + Duration::from_secs(1);
669        loop {
670            let next =
671                timestamp(&rows_to_record_batch(&rows, &table_info(schema.clone())).unwrap());
672            if next != first {
673                break;
674            }
675            assert!(
676                Instant::now() < deadline,
677                "dynamic default was not refreshed"
678            );
679            yield_now();
680        }
681        rows.schema
682            .push(source("created", ColumnDataType::TimestampMillisecond));
683        rows.rows[0].values.push(Value::default());
684        assert!(
685            rows_to_record_batch(&rows, &table_info(schema.clone()))
686                .unwrap()
687                .column(3)
688                .is_null(0)
689        );
690    }

754    #[test]
755    fn test_nested_value_validation() {
756        let int = Value {
757            value_data: Some(ValueData::I32Value(1)),
758        };
759        let list = Value {
760            value_data: Some(ValueData::ListValue(api::v1::ListValue {
761                items: vec![int.clone()],
762            })),
763        };
764        let structure = Value {
765            value_data: Some(ValueData::StructValue(api::v1::StructValue {
766                items: vec![int.clone()],
767            })),
768        };
769        let types = [
770            (
771                ColumnDataTypeWrapper::list_datatype(ColumnDataTypeWrapper::int32_datatype()),
772                list.clone(),
773            ),
774            (
775                ColumnDataTypeWrapper::struct_datatype(vec![(
776                    "count".to_string(),
777                    ColumnDataTypeWrapper::int32_datatype(),
778                )]),
779                structure.clone(),
780            ),
781        ];
782        for (datatype, valid) in types {
783            let (kind, extension) = datatype.to_parts();
784            let schema = Schema::new(vec![ColumnSchema::new("nested", datatype.into(), true)]);
785            let mut column = source("nested", kind);
786            column.datatype_extension = extension;
787            let mut rows = Rows {
788                schema: vec![column],
789                rows: vec![Row {
790                    values: vec![valid],
791                }],
792            };
793            let table = table_info(schema);
794            assert_eq!(rows_to_record_batch(&rows, &table).unwrap().num_rows(), 1);
795            for invalid in [list.clone(), structure.clone()] {
796                match rows.rows[0].values[0].value_data.as_mut().unwrap() {
797                    ValueData::ListValue(v) => v.items = vec![invalid],
798                    ValueData::StructValue(v) => v.items = vec![invalid],
799                    _ => unreachable!(),
800                }
801                assert!(matches!(
802                    rows_to_record_batch(&rows, &table),
803                    Err(error::Error::InvalidInsertRequest { .. })
804                ));
805            }
806        }
807        let datatype = ColumnDataTypeWrapper::list_datatype(ColumnDataTypeWrapper::list_datatype(
808            ColumnDataTypeWrapper::int32_datatype(),
809        ));
810        let (kind, extension) = datatype.to_parts();
811        let value = Value {
812            value_data: Some(ValueData::ListValue(api::v1::ListValue {
813                items: vec![list],
814            })),
815        };
816        assert!(value_matches_type(&value, kind as i32, extension.as_ref()));
817        assert!(!value_matches_type(&value, kind as i32, None));
818    }
