56// The buffer size should be greater than 5MB (minimum multipart upload size).
57/// Buffer size to flush data to object stores.
58const WRITE_BUFFER_THRESHOLD: ReadableSize = ReadableSize::mb(8);
59
60/// Default number of concurrent write, it only works on object store backend(e.g., S3).
61const WRITE_CONCURRENCY: usize = 8;

63impl StatementExecutor {
64    async fn stream_to_file(
65        stream: SendableRecordBatchStream,
66        format: &Format,
67        object_store: ObjectStore,
68        path: &str,
69    ) -> Result<usize> {
70        let threshold = WRITE_BUFFER_THRESHOLD.as_bytes() as usize;
71
72        let stream = Box::pin(SendableRecordBatchMapper::new(
73            stream,
74            map_json_type_to_string,
75            map_json_type_to_string_schema,
76        ));
77        match format {
78            Format::Csv(format) => stream_to_csv(
79                Box::pin(DfRecordBatchStreamAdapter::new(stream)),
80                object_store,
81                path,
82                threshold,
83                WRITE_CONCURRENCY,
84                format,
85            )
86            .await
87            .context(error::WriteStreamToFileSnafu { path }),
88            Format::Json(format) => stream_to_json(
89                Box::pin(DfRecordBatchStreamAdapter::new(stream)),
90                object_store,
91                path,
92                threshold,
93                WRITE_CONCURRENCY,
94                format,
95            )
96            .await
97            .context(error::WriteStreamToFileSnafu { path }),
98            Format::Parquet(_) => stream_to_parquet(
99                Box::pin(DfRecordBatchStreamAdapter::new(stream)),
100                object_store,
101                path,
102                WRITE_CONCURRENCY,
103            )
104            .await
105            .context(error::WriteStreamToFileSnafu { path }),
106            _ => error::UnsupportedFormatSnafu {
107                format: format.clone(),
108            }
109            .fail(),
110        }
111    }

134    pub(crate) async fn copy_captured_table_to_managed(
135        &self,
136        table: TableRef,
137        req: CopyTableRequest,
138        query_ctx: QueryContextRef,
139        managed: Option<(&ExportWriteBudget, &CancellationToken)>,
140        packed: Option<PackedTableWriter>,
141    ) -> Result<usize> {
142        let info = table.table_info();
143        let table_ref = TableReference::full(&info.catalog_name, &info.schema_name, &info.name);
144        let table_id = info.table_id();
145        let format = Format::try_from(&req.with).context(error::ParseFileFormatSnafu)?;
146
147        let df_table_ref = DfTableReference::from(table_ref);
148
149        let filters = table
150            .schema()
151            .timestamp_column()
152            .and_then(|c| {
153                common_query::logical_plan::build_filter_from_timestamp(
154                    &c.name,
155                    req.timestamp_range.as_ref(),
156                )
157            })
158            .into_iter()
159            .collect::<Vec<_>>();
160
161        let table_provider = Arc::new(DfTableProviderAdapter::new(table));
162        let table_source = Arc::new(DefaultTableSource::new(table_provider));
163
164        let mut builder = LogicalPlanBuilder::scan_with_filters(
165            df_table_ref,
166            table_source,
167            None,
168            filters.clone(),
169        )
170        .context(BuildDfLogicalPlanSnafu)?;
171        for f in filters {
172            builder = builder.filter(f).context(BuildDfLogicalPlanSnafu)?;
173        }
174        let plan = builder.build().context(BuildDfLogicalPlanSnafu)?;
175
176        let output = self
177            .query_engine
178            .execute(plan, query_ctx)
179            .await
180            .context(ExecLogicalPlanSnafu)?;
181
182        let CopyTableRequest {
183            location,
184            connection,
185            ..
186        } = &req;
187
188        debug!("Copy table: {table_id} to location: {location}");
189        Self::copy_to_file_managed(
190            &self.local_file_access,
191            &format,
192            output,
193            location,
194            connection,
195            managed,
196            packed,
197        )
198        .await
199    }

220    async fn copy_to_file_managed(
221        local_file_access: &LocalFileAccess,
222        format: &Format,
223        output: Output,
224        location: &str,
225        connection: &HashMap<String, String>,
226        managed: Option<(&ExportWriteBudget, &CancellationToken)>,
227        packed: Option<PackedTableWriter>,
228    ) -> Result<usize> {
229        let output = if managed.is_none() {
230            output
231                .map_dictionary_to_values()
232                .context(error::BuildRecordBatchSnafu)?
233        } else {
234            output
235        };
236        let stream = match output.data {
237            OutputData::Stream(stream) => stream,
238            OutputData::RecordBatches(record_batches) => record_batches.as_stream(),
239            _ => unreachable!(),
240        };
241
242        if let (Some((budget, token)), Some(packed)) = (managed, packed) {
243            return stream_to_managed_parquet_with_packed(
244                stream,
245                ManagedParquetDestination::Packed(packed),
246                budget,
247                token,
248            )
249            .await;
250        }
251
252        let backend = build_backend_for_write_with_path(location, connection, local_file_access)
253            .await
254            .context(error::BuildBackendSnafu)?;
255        let filename = backend
256            .object_path
257            .with_context(|| error::UnexpectedSnafu {
258                violated: format!("Expected filename, path: {location}"),
259            })?;
260        if let Some((budget, token)) = managed {
261            stream_to_managed_parquet_with_packed(
262                stream,
263                ManagedParquetDestination::Object {
264                    store: backend.object_store,
265                    path: filename,
266                },
267                budget,
268                token,
269            )
270            .await
271        } else {
272            Self::stream_to_file(stream, format, backend.object_store, &filename).await
273        }
274    }

303async fn stream_to_managed_parquet_with_packed(
304    mut stream: SendableRecordBatchStream,
305    destination: ManagedParquetDestination,
306    budget: &ExportWriteBudget,
307    token: &CancellationToken,
308) -> Result<usize> {
309    use common_recordbatch::{RecordBatch, map_dictionary_to_values_schema};
310    let original = stream.schema();
311    let (expanded_schema, expand) = map_dictionary_to_values_schema(original.clone());
312    let json_columns = expanded_schema
313        .column_schemas()
314        .iter()
315        .enumerate()
316        .filter_map(|(index, column)| {
317            (column.data_type.is_json() && !column.data_type.is_json2()).then_some(index)
318        })
319        .collect::<Vec<_>>();
320    let (mapped_schema, json) = map_json_type_to_string_schema(expanded_schema.clone());
321    let output_schema = if json {
322        mapped_schema
323    } else {
324        expanded_schema.clone()
325    };
326    let packed_destination = matches!(destination, ManagedParquetDestination::Packed(_));
327    let schema = output_schema.arrow_schema().clone();
328    let (writer, path) = match destination {
329        ManagedParquetDestination::Packed(packed) => {
330            let path = packed.path().to_owned();
331            let limits =
332                crate::statement::export_logical_tables::LogicalTableExportLimits::default();
333            (
334                ParquetFileWriter::open_packed(schema, Some(limits.writer), packed),
335                path,
336            )
337        }
338        ManagedParquetDestination::Object { store, path } => (
339            ParquetFileWriter::open(schema, store, &path, 1, None).await,
340            path,
341        ),
342    };
343    let path = path.as_str();
344    let mut writer = writer.context(error::WriteStreamToFileSnafu { path })?;
345    let mut started = false;
346    let result = async {
347        let mut rows = 0;
348        loop {
349            let batch = tokio::select! {
350                biased;
351                _ = token.cancelled() => return error::LogicalTableExportCancelledSnafu.fail(),
352                batch = stream.next() => batch,
353            };
354            let Some(batch) = batch else { break };
355            let batch = batch
356                .context(error::BuildRecordBatchSnafu)?
357                .into_df_record_batch();
358            let mut offset = 0;
359            while offset < batch.num_rows() {
360                // The scan batch remains query-owned. Bound the downstream slice and
361                // detach its buffers below if it would retain more than its reservation.
362                let (conversion, retained) =
363                    ExportWriteBudget::conversion_budget(0, batch.num_columns(), usize::MAX)?;
364                let input = batch.clone();
365                let json_columns = json_columns.clone();
366                let (len, estimated) = common_runtime::spawn_blocking_global(move || {
367                    rows_within_budget(&input, offset, input.num_rows(), conversion, &json_columns)
368                })
369                .await
370                .context(error::JoinTaskSnafu)??;
371                let reservation = retained.saturating_add(estimated.saturating_mul(4));
372                let permit = budget.reserve(reservation, token).await?;
373                let batch = batch.clone();
374                let (original, expanded_schema, output_schema) = (
375                    original.clone(),
376                    expanded_schema.clone(),
377                    output_schema.clone(),
378                );
379                let (batch, len) = common_runtime::spawn_blocking_global(move || {
380                    let mut batch = RecordBatch::from_df_record_batch(
381                        original.clone(),
382                        batch.slice(offset, len),
383                    );
384                    if expand {
385                        batch = RecordBatch::from_df_record_batch(
386                            expanded_schema.clone(),
387                            expand_export_batch(
388                                &batch.into_df_record_batch(),
389                                expanded_schema.arrow_schema().clone(),
390                            )?,
391                        );
392                    }
393                    if json {
394                        batch = map_json_type_to_string(batch, &expanded_schema, &output_schema)
395                            .context(error::BuildRecordBatchSnafu)?;
396                    }
397                    let mut batch = batch.into_df_record_batch();
398                    if batch.get_array_memory_size() > reservation {
399                        let indices = UInt64Array::from_iter_values(0..batch.num_rows() as u64);
400                        batch = arrow::compute::take_record_batch(&batch, &indices)
401                            .context(error::ComputeArrowSnafu)?;
402                        let arrays = batch
403                            .columns()
404                            .iter()
405                            .map(compact_view_buffers)
406                            .collect::<Result<Vec<_>>>()?;
407                        batch = arrow::record_batch::RecordBatch::try_new(batch.schema(), arrays)
408                            .context(error::ComputeArrowSnafu)?;
409                    }
410                    ensure!(
411                        batch.get_array_memory_size() <= reservation,
412                        error::LogicalTableExportResourceSnafu {
413                            reason: "converted backing buffers exceed reservation"
414                        }
415                    );
416                    Ok::<_, error::Error>((batch, len))
417                })
418                .await
419                .context(error::JoinTaskSnafu)??;
420                started = true;
421                let write = writer.write(batch, Some(token)).await;
422                drop(permit);
423                write.map_err(|error| map_writer_error(error, path))?;
424                rows += len;
425                offset += len;
426            }
427        }
428        started = true;
429        writer
430            .finish(Some(token))
431            .await
432            .map_err(|error| map_writer_error(error, path))?;
433        Ok(rows)
434    }
435    .await;
436    if result.is_err() {
437        token.cancel();
438        if (started || packed_destination)
439            && let Err(error) = writer.abort().await
440        {
441            common_telemetry::warn!(error; "Failed to abort ordinary export file");
442        }
443    }
444    result
445}

447// Arrow take copies ordinary buffers but shares view data buffers, including
448// views nested inside lists or structs. Reclaim those unselected values too.
449fn compact_view_buffers(array: &ArrayRef) -> Result<ArrayRef> {
450    match array.data_type() {
451        DataType::Utf8View => Ok(Arc::new(array.as_string_view().gc())),
452        DataType::BinaryView => Ok(Arc::new(array.as_binary_view().gc())),
453        _ => {
454            let data = array.to_data();
455            if data.child_data().is_empty() {
456                return Ok(array.clone());
457            }
458            let children = data
459                .child_data()
460                .iter()
461                .map(|child| {
462                    compact_view_buffers(&make_array(child.clone())).map(|array| array.to_data())
463                })
464                .collect::<Result<Vec<_>>>()?;
465            Ok(make_array(
466                data.into_builder()
467                    .child_data(children)
468                    .build()
469                    .context(error::ComputeArrowSnafu)?,
470            ))
471        }
472    }

682                for batch in reader {
683                    let batch = batch.unwrap();
684                    let strings = arrow::compute::cast(batch.column(0), &DataType::Utf8).unwrap();
685                    assert!(
686                        strings
687                            .as_string::<i32>()
688                            .iter()
689                            .all(|item| item == Some(value.as_str()))
690                    );
691                    actual_rows += batch.num_rows();
692                }
693                assert_eq!(actual_rows, rows);
694                assert_eq!(budget.available(), (1, 64 * 1024 * 1024));
695            }
696        }

728            Arc::new(dictionary) as ArrayRef,
729            Arc::new(BinaryArray::from(vec![
730                Some(json.as_slice()),
731                None,
732                None,
733                Some(json.as_slice()),
734            ])),
735            Arc::new(StringViewArray::from(vec![
736                Some("short"),
737                None,
738                Some("long view"),
739                Some("long view"),
740            ])),
741            Arc::new(BinaryViewArray::from(vec![
742                Some(&b"short"[..]),
743                None,
744                Some(&b"long view"[..]),
745                Some(&b"long view"[..]),
746            ])),
747        ];

749            arrow::record_batch::RecordBatch::try_new(schema.arrow_schema().clone(), arrays)

754                schema.clone(),

759            let budget = ExportWriteBudget::new(1);

799                    .unwrap()

803            }
