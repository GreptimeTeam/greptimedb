98async fn list_copy_from_paths(
99    req: &CopyTableRequest,
100    local_file_access: &LocalFileAccess,
101    remote_storage_access: &RemoteStorageAccess,
102) -> Result<(ObjectStore, Vec<String>)> {
103    let backend = build_backend_with_path(
104        &req.location,
105        &req.connection,
106        local_file_access,
107        remote_storage_access,
108    )
109    .await
110    .context(error::BuildBackendSnafu)?;
111    let regex = req
112        .pattern
113        .as_ref()
114        .map(|x| Regex::new(x))
115        .transpose()
116        .context(error::BuildRegexSnafu)?;
117
118    // Listing a known file's parent for every table makes COPY DATABASE do
119    // quadratic directory work.
120    if let Some(filename) = &backend.object_path {
121        let is_file = backend
122            .is_file(filename)
123            .await
124            .with_context(|_| common_datasource::error::ListObjectsSnafu {
125                path: req.location.clone(),
126            })
127            .context(error::ListObjectsSnafu)?;
128        let paths = if is_file {
129            vec![filename.clone()]
130        } else {
131            vec![]
132        };
133        return Ok((backend.object_store, paths));
134    }
135
136    let lister = Lister::new(
137        backend.object_store.clone(),
138        Source::Dir,
139        req.location.clone(),
140        regex,
141    );
142
143    let entries = lister.list().await.context(error::ListObjectsSnafu)?;
144    debug!(
145        "Copy from location: {:?}, entries: {entries:?}",
146        req.location
147    );
148    let paths = entries
149        .into_iter()
150        .filter(|entry| entry.metadata().mode() == EntryMode::FILE)
151        .map(|entry| entry.path().to_string())
152        .collect();
153    Ok((backend.object_store, paths))
154}

157    async fn collect_metadata(
158        &self,
159        object_store: &ObjectStore,
160        format: Format,
161        path: String,
162    ) -> Result<FileMetadata> {
163        match format {
164            Format::Csv(format) => Ok(FileMetadata::Csv {
165                schema: Arc::new(
166                    format
167                        .infer_schema(object_store, &path)
168                        .await
169                        .context(error::InferSchemaSnafu { path: &path })?,
170                ),
171                format,
172                path,
173            }),
174            Format::Json(format) => Ok(FileMetadata::Json {
175                schema: Arc::new(
176                    format
177                        .infer_schema(object_store, &path)
178                        .await
179                        .context(error::InferSchemaSnafu { path: &path })?,
180                ),
181                format,
182                path,
183            }),
184            Format::Parquet(_) => {
185                let meta = object_store
186                    .stat(&path)
187                    .await
188                    .context(error::ReadObjectSnafu { path: &path })?;
189                let mut reader = object_store
190                    .reader(&path)
191                    .await
192                    .context(error::ReadObjectSnafu { path: &path })?
193                    .into_futures_async_read(0..meta.content_length())
194                    .await
195                    .context(error::ReadObjectSnafu { path: &path })?
196                    .compat();
197                let metadata = ArrowReaderMetadata::load_async(&mut reader, Default::default())
198                    .await
199                    .context(error::ReadParquetMetadataSnafu)?;
200
201                Ok(FileMetadata::Parquet {
202                    schema: metadata.schema().clone(),
203                    metadata,
204                    path,
205                })
206            }
207            Format::Orc(_) => {
208                let meta = object_store
209                    .stat(&path)
210                    .await
211                    .context(error::ReadObjectSnafu { path: &path })?;
212
213                let reader = object_store
214                    .reader(&path)
215                    .await
216                    .context(error::ReadObjectSnafu { path: &path })?;
217
218                let schema = infer_orc_schema(ReaderAdapter::new(reader, meta.content_length()))
219                    .await
220                    .context(error::ReadOrcSnafu)?;
221
222                Ok(FileMetadata::Orc {
223                    schema: Arc::new(schema),
224                    path,
225                })
226            }
227        }
228    }

230    async fn build_read_stream(
231        &self,
232        compat_schema: SchemaRef,
233        object_store: &ObjectStore,
234        file_metadata: &FileMetadata,
235        projection: Vec<usize>,
236        filters: Vec<Expr>,
237    ) -> Result<DfSendableRecordBatchStream> {
238        match file_metadata {
239            FileMetadata::Csv {
240                format,
241                path,
242                schema,
243            } => {
244                let output_schema = Arc::new(
245                    compat_schema
246                        .project(&projection)
247                        .context(error::ProjectSchemaSnafu)?,
248                );
249
250                let options = CsvOptions::default()
251                    .with_has_header(format.has_header)
252                    .with_delimiter(format.delimiter);
253                let reader_schema = exact_numeric_reader_schema(schema, &compat_schema, true);
254                let csv_source = CsvSource::new(Arc::new(reader_schema))
255                    .with_csv_options(options)
256                    .with_batch_size(DEFAULT_BATCH_SIZE);
257                let stream = if format.skip_bad_records {
258                    let reader_schema =
259                        csv_reader_schema_for_skip_bad_records(schema, &compat_schema);
260                    // This reader takes its types from the table, not from the inferred
261                    // schema, so the table types also say which columns to parse.
262                    let numbers = projection
263                        .iter()
264                        .map(|idx| table_text_number(compat_schema.fields().get(*idx)))
265                        .collect();
266                    let stream = tolerant_csv_stream(
267                        object_store,
268                        path,
269                        Arc::new(reader_schema),
270                        projection.clone(),
271                        format,
272                    )
273                    .await
274                    .context(error::BuildFileStreamSnafu)?;
275                    // The stream has one record in a batch, so a number that fails to parse
276                    // is skipped with its record.
277                    parse_number_columns(stream, numbers, true)
278                } else {
279                    let numbers = text_number_columns(schema, &compat_schema, &projection, true);
280                    let stream = file_to_stream(
281                        object_store,
282                        path,
283                        csv_source,
284                        Some(projection),
285                        format.compression_type,
286                    )
287                    .await
288                    .context(error::BuildFileStreamSnafu)?;
289                    // An empty csv field is NULL.
290                    parse_number_columns(stream, numbers, true)
291                };
292
293                let stream = Box::pin(
294                    // The projection is already applied in the CSV reader when we created the stream,
295                    // so we pass None here to avoid double projection which would cause schema mismatch errors.
296                    RecordBatchStreamTypeAdapter::new(output_schema, stream, None)
297                        .with_filter(filters)
298                        .context(error::PhysicalExprSnafu)?,
299                );
300                if format.skip_bad_records {
301                    Ok(Box::pin(SkipBadRecordsStream::new(stream, path)))
302                } else {
303                    Ok(stream)
304                }
305            }
306            FileMetadata::Json {
307                path,
308                format,
309                schema,
310            } => {
311                let output_schema = Arc::new(
312                    compat_schema
313                        .project(&projection)
314                        .context(error::ProjectSchemaSnafu)?,
315                );
316
317                let reader_schema = exact_numeric_reader_schema(schema, &compat_schema, false);
318                let numbers = text_number_columns(schema, &compat_schema, &projection, false);
319                let stream = if numbers.iter().any(Option::is_some) {
320                    // Only this stream reads a number as its text.
321                    let projected_schema = Arc::new(
322                        reader_schema
323                            .project(&projection)
324                            .context(error::ProjectSchemaSnafu)?,
325                    );
326                    json_text_stream(
327                        object_store,
328                        path,
329                        projected_schema,
330                        format,
331                        DEFAULT_BATCH_SIZE,
332                    )
333                    .await
334                    .context(error::BuildFileStreamSnafu)?
335                } else {
336                    let json_source = JsonSource::new(Arc::new(reader_schema))
337                        .with_batch_size(DEFAULT_BATCH_SIZE);
338                    file_to_stream(
339                        object_store,
340                        path,
341                        json_source,
342                        Some(projection),
343                        format.compression_type,
344                    )
345                    .await
346                    .context(error::BuildFileStreamSnafu)?
347                };
348                // An empty json string is not a number, so it is an error like other text.
349                let stream = parse_number_columns(stream, numbers, false);
350
351                Ok(Box::pin(
352                    // The projection is already applied in the JSON reader when we created the stream,
353                    // so we pass None here to avoid double projection which would cause schema mismatch errors.
354                    RecordBatchStreamTypeAdapter::new(output_schema, stream, None)
355                        .with_filter(filters)
356                        .context(error::PhysicalExprSnafu)?,
357                ))
358            }
359            FileMetadata::Parquet { metadata, path, .. } => {
360                let meta = object_store
361                    .stat(path)
362                    .await
363                    .context(error::ReadObjectSnafu { path })?;
364                let reader = object_store
365                    .reader_with(path)
366                    .chunk(DEFAULT_READ_BUFFER)
367                    .await
368                    .context(error::ReadObjectSnafu { path })?
369                    .into_futures_async_read(0..meta.content_length())
370                    .await
371                    .context(error::ReadObjectSnafu { path })?
372                    .compat();
373                let builder =
374                    ParquetRecordBatchStreamBuilder::new_with_metadata(reader, metadata.clone());
375                let stream = builder
376                    .build()
377                    .context(error::BuildParquetRecordBatchStreamSnafu)?;
378
379                let output_schema = Arc::new(
380                    compat_schema
381                        .project(&projection)
382                        .context(error::ProjectSchemaSnafu)?,
383                );
384                Ok(Box::pin(
385                    RecordBatchStreamTypeAdapter::new(output_schema, stream, Some(projection))
386                        .with_filter(filters)
387                        .context(error::PhysicalExprSnafu)?,
388                ))
389            }
390            FileMetadata::Orc { path, .. } => {
391                let meta = object_store
392                    .stat(path)
393                    .await
394                    .context(error::ReadObjectSnafu { path })?;
395
396                let reader = object_store
397                    .reader_with(path)
398                    .chunk(DEFAULT_READ_BUFFER)
399                    .await
400                    .context(error::ReadObjectSnafu { path })?;
401                let stream =
402                    new_orc_stream_reader(ReaderAdapter::new(reader, meta.content_length()))
403                        .await
404                        .context(error::ReadOrcSnafu)?;
405
406                let output_schema = Arc::new(
407                    compat_schema
408                        .project(&projection)
409                        .context(error::ProjectSchemaSnafu)?,
410                );
411
412                Ok(Box::pin(
413                    RecordBatchStreamTypeAdapter::new(output_schema, stream, Some(projection))
414                        .with_filter(filters)
415                        .context(error::PhysicalExprSnafu)?,
416                ))
417            }
418        }
419    }

421    /// Imports one indexed stream with the ordinary COPY schema mapping and inserter.
422    pub(crate) async fn copy_indexed_parquet<
423        R: datafusion::parquet::arrow::async_reader::AsyncFileReader + Send + Unpin + 'static,
424    >(
425        &self,
426        reader: R,
427        table: table::TableRef,
428        expected_rows: u64,
429        pending: &mut crate::statement::import_packed::PendingPackedInserts,
430        cancellation: &tokio_util::sync::CancellationToken,
431        query_ctx: QueryContextRef,
432    ) -> Result<()> {
433        let builder = tokio::select! {
434            _ = cancellation.cancelled() => return error::PackedImportCancelledSnafu.fail(),
435            result = ParquetRecordBatchStreamBuilder::new(reader) => result.context(error::ReadParquetMetadataSnafu)?,
436        };
437        let table_schema = table.schema().arrow_schema().clone();
438        let (file_projection, table_projection, _) =
439            generated_schema_projection_and_compatible_file_schema(builder.schema(), &table_schema);
440        let file_schema = Arc::new(
441            builder
442                .schema()
443                .project(&file_projection)
444                .context(error::ProjectSchemaSnafu)?,
445        );
446        let target_schema = Arc::new(
447            table_schema
448                .project(&table_projection)
449                .context(error::ProjectSchemaSnafu)?,
450        );
451        ensure_schema_compatible(&file_schema, &target_schema)?;
452        let stream = builder
453            .with_batch_size(DEFAULT_BATCH_SIZE)
454            .build()
455            .context(error::BuildParquetRecordBatchStreamSnafu)?;
456        let mut stream = Box::pin(RecordBatchStreamTypeAdapter::new(
457            target_schema.clone(),
458            stream,
459            Some(file_projection),
460        ));
461        let info = table.table_info();
462        let mut rows = 0u64;
463        loop {
464            pending.before_decode().await?;
465            if cancellation.is_cancelled() {
466                return error::PackedImportCancelledSnafu.fail();
467            }
468            let batch = tokio::select! {
469                _ = cancellation.cancelled() => return error::PackedImportCancelledSnafu.fail(),
470                batch = stream.next() => batch,
471            };
472            let Some(batch) = batch else {
473                break;
474            };
475            let batch = batch.context(error::ReadDfRecordBatchSnafu)?;
476            rows += batch.num_rows() as u64;
477            if rows > expected_rows {
478                return error::InvalidCopyParameterSnafu {
479                    key: "row_count",
480                    value: expected_rows.to_string(),
481                }
482                .fail();
483            }
484            let vectors = Helper::try_into_vectors(batch.columns()).context(IntoVectorsSnafu)?;
485            let retained_bytes = vectors.iter().map(|v| v.memory_size()).sum::<usize>();
486            let columns_values = target_schema
487                .fields()
488                .iter()
489                .map(|f| f.name().clone())
490                .zip(vectors)
491                .collect();
492            let bytes = retained_bytes.max(batch.get_array_memory_size());
493            let inserter = self.inserter.clone();
494            let request = InsertRequest {
495                catalog_name: info.catalog_name.clone(),
496                schema_name: info.schema_name.clone(),
497                table_name: info.name.clone(),
498                columns_values,
499                skip_wal: query_ctx.skip_wal(),
500            };
501            let ctx = query_ctx.clone();
502            pending
503                .admit(
504                    bytes,
505                    async move { inserter.handle_table_insert(request, ctx).await },
506                    cancellation,
507                )
508                .await?;
509        }
510        if rows != expected_rows {
511            return error::InvalidCopyParameterSnafu {
512                key: "row_count",
513                value: expected_rows.to_string(),
514            }
515            .fail();
516        }
517        Ok(())
518    }

520    #[tracing::instrument(skip_all)]
521    pub async fn copy_table_from(
522        &self,
523        req: CopyTableRequest,
524        query_ctx: QueryContextRef,
525    ) -> Result<Output> {
526        let table_ref = TableReference {
527            catalog: &req.catalog_name,
528            schema: &req.schema_name,
529            table: &req.table_name,
530        };
531        let table = self.get_table(&table_ref).await?;
532        let format = Format::try_from(&req.with).context(error::ParseFileFormatSnafu)?;
533        let (object_store, paths) =
534            list_copy_from_paths(&req, &self.local_file_access, &self.remote_storage_access)
535                .await?;
536        let mut files = Vec::with_capacity(paths.len());
537        let table_schema = table.schema().arrow_schema().clone();
538        let filters = table
539            .schema()
540            .timestamp_column()
541            .and_then(|c| {
542                common_query::logical_plan::build_same_type_ts_filter(c, req.timestamp_range)
543            })
544            .into_iter()
545            .collect::<Vec<_>>();
546
547        for path in paths {
548            let file_metadata = self
549                .collect_metadata(&object_store, format.clone(), path)
550                .await?;
551
552            validate_csv_headers_if_required(&file_metadata, &table_schema)?;
553            let schema_mapping = copy_from_schema_mapping(&file_metadata, &table_schema);
554            let projected_file_schema = Arc::new(
555                file_metadata
556                    .schema()
557                    .project(&schema_mapping.file_projection)
558                    .context(error::ProjectSchemaSnafu)?,
559            );
560            let projected_table_schema = Arc::new(
561                table_schema
562                    .project(&schema_mapping.table_projection)
563                    .context(error::ProjectSchemaSnafu)?,
564            );
565            ensure_schema_compatible(&projected_file_schema, &projected_table_schema)?;
566
567            files.push((
568                Arc::new(schema_mapping.compat_file_schema),
569                schema_mapping.file_projection,
570                projected_table_schema,
571                file_metadata,
572            ))
573        }
574
575        let mut rows_inserted = 0;
576        let mut insert_cost = 0;
577        let max_insert_rows = req
578            .limit
579            .map(|n| {
580                usize::try_from(n).map_err(|_| {
581                    error::InvalidCopyParameterSnafu {
582                        key: "limit".to_string(),
583                        value: n.to_string(),
584                    }
585                    .build()
586                })
587            })
588            .transpose()?;
589        if max_insert_rows == Some(0) {
590            return Ok(gen_insert_output(rows_inserted, insert_cost));
591        }
592
593        let mut accepted_rows = 0;
594        for (compat_schema, file_schema_projection, projected_table_schema, file_metadata) in files
595        {
596            let mut stream = self
597                .build_read_stream(
598                    compat_schema,
599                    &object_store,
600                    &file_metadata,
601                    file_schema_projection,
602                    filters.clone(),
603                )
604                .await?;
605
606            let fields = projected_table_schema
607                .fields()
608                .iter()
609                .map(|f| f.name().clone())
610                .collect::<Vec<_>>();
611
612            // TODO(hl): make this configurable through options.
613            let pending_mem_threshold = ReadableSize::mb(32).as_bytes();
614            let mut pending_mem_size = 0;
615            let mut pending = vec![];
616
617            while let Some(r) = stream.next().await {
618                let record_batch = r.context(error::ReadDfRecordBatchSnafu)?;
619                let record_batch = if let Some(max_insert_rows) = max_insert_rows {
620                    let remaining_rows = max_insert_rows - accepted_rows;
621                    if record_batch.num_rows() > remaining_rows {
622                        record_batch.slice(0, remaining_rows)
623                    } else {
624                        record_batch
625                    }
626                } else {
627                    record_batch
628                };
629                let record_batch_rows = record_batch.num_rows();
630                let vectors =
631                    Helper::try_into_vectors(record_batch.columns()).context(IntoVectorsSnafu)?;
632
633                pending_mem_size += vectors.iter().map(|v| v.memory_size()).sum::<usize>();
634
635                let columns_values = fields
636                    .iter()
637                    .cloned()
638                    .zip(vectors)
639                    .collect::<HashMap<_, _>>();
640
641                pending.push(self.inserter.handle_table_insert(
642                    InsertRequest {
643                        catalog_name: req.catalog_name.clone(),
644                        schema_name: req.schema_name.clone(),
645                        table_name: req.table_name.clone(),
646                        columns_values,
647                        skip_wal: query_ctx.skip_wal(),
648                    },
649                    query_ctx.clone(),
650                ));
651                accepted_rows += record_batch_rows;
652
653                if pending_mem_size as u64 >= pending_mem_threshold {
654                    let (rows, cost) = batch_insert(&mut pending, &mut pending_mem_size).await?;
655                    rows_inserted += rows;
656                    insert_cost += cost;
657                }
658
659                if let Some(max_insert_rows) = max_insert_rows
660                    && accepted_rows == max_insert_rows
661                {
662                    if !pending.is_empty() {
663                        let (rows, cost) =
664                            batch_insert(&mut pending, &mut pending_mem_size).await?;
665                        rows_inserted += rows;
666                        insert_cost += cost;
667                    }
668                    return Ok(gen_insert_output(rows_inserted, insert_cost));
669                }
670            }
671
672            if !pending.is_empty() {
673                let (rows, cost) = batch_insert(&mut pending, &mut pending_mem_size).await?;
674                rows_inserted += rows;
675                insert_cost += cost;
676            }
677        }
678
679        Ok(gen_insert_output(rows_inserted, insert_cost))
680    }
681}

714        let this = self.get_mut();
715        loop {
716            match this.inner.as_mut().poll_next(cx) {
717                Poll::Ready(Some(Err(error))) if is_skippable_record_error(&error) => {
718                    common_telemetry::warn!(
719                        "Skipping bad record while copying from {}: {}",

721                        error
722                    );
723                    continue;
724                }
725                other => return other,
726            }
727        }
728    }

731fn is_skippable_record_error(error: &DataFusionError) -> bool {
732    match error {
733        DataFusionError::ArrowError(error, _) => is_skippable_arrow_error(error),
734        DataFusionError::External(error) => {
735            error
736                .downcast_ref::<ArrowError>()
737                .is_some_and(is_skippable_arrow_error)
738                // The error of a stream that wraps another one.
739                || error
740                    .downcast_ref::<DataFusionError>()
741                    .is_some_and(is_skippable_record_error)
742        }
743        DataFusionError::Context(_, error) => is_skippable_record_error(error),
744        _ => false,
745    }

748/// Executes all pending inserts all at once, drain pending requests and reset pending bytes.
749async fn batch_insert(
750    pending: &mut Vec<impl Future<Output = Result<Output>>>,
751    pending_bytes: &mut usize,
752) -> Result<(OutputRows, OutputCost)> {
753    let batch = pending.drain(..);
754    let result = futures::future::try_join_all(batch)
755        .await?
756        .iter()
757        .map(|o| o.extract_rows_and_cost())
758        .reduce(|(a, b), (c, d)| (a + c, b + d))
759        .unwrap_or((0, 0));
760    *pending_bytes = 0;
761    Ok(result)
762}

777/// Returns the schema to read the file with.
778///
779/// Schema inference reads a number as `Int64` or `Float64`. Casting such a column to a wider
780/// table type afterwards rounds every value a `Float64` can't hold, and a value outside the
781/// `Int64` range fails to parse. So a column that loads into a `Decimal128` or `UInt64` column
782/// is read as a string and parsed by [`parse_number_columns`], see [`text_number_columns`].
783fn exact_numeric_reader_schema(file: &SchemaRef, compat: &SchemaRef, csv: bool) -> Schema {
784    let fields = file
785        .fields()
786        .iter()
787        .enumerate()
788        .map(
789            |(idx, file_field)| match text_number(file_field, compat.fields().get(idx), csv) {
790                Some(_) => text_field(file_field),
791                None => file_field.clone(),
792            },
793        )
794        .collect::<Vec<_>>();
795
796    Schema::new_with_metadata(fields, file.metadata().clone())
797}

817/// Returns the type to parse the text of a file column into, if it is read as a string for
818/// that.
819///
820/// A csv file has no types of its own. When its numbers don't all look like an `Int64` or a
821/// `Float64`, as with a value above the `Int64` range next to one in scientific notation, the
822/// column is inferred as a string. So with `csv` such a column is parsed in the same way. A
823/// json string is left to the cast.
824fn text_number(
825    file_field: &FieldRef,
826    compat_field: Option<&FieldRef>,
827    csv: bool,
828) -> Option<TextNumber> {
829    let number_text = match file_field.data_type() {
830        ArrowDataType::Int64 | ArrowDataType::Float64 => true,
831        ArrowDataType::Utf8 => csv,
832        _ => false,
833    };
834    if !number_text {
835        return None;
836    }
837    table_text_number(compat_field)
838}

840/// Returns the type to parse text into for a table column that is a `Decimal128` or a
841/// `UInt64`.
842fn table_text_number(compat_field: Option<&FieldRef>) -> Option<TextNumber> {
843    match compat_field?.data_type() {
844        ArrowDataType::Decimal128(precision, scale) => Some(TextNumber::Decimal128 {
845            precision: *precision,
846            scale: *scale,
847        }),
848        ArrowDataType::UInt64 => Some(TextNumber::UInt64),
849        _ => None,
850    }
851}

853fn csv_reader_schema_for_skip_bad_records(file: &SchemaRef, compat: &SchemaRef) -> Schema {
854    let fields = file
855        .fields()
856        .iter()
857        .enumerate()
858        .map(|(idx, file_field)| match compat.fields().get(idx) {
859            // Arrow's csv reader drops the digits of a decimal beyond its scale, is wrong for
860            // some numbers in scientific notation, and only reads an unsigned integer from
861            // plain digits.
862            // Whatever type the column was inferred as: this reader uses the table type for
863            // every other column too.
864            compat_field if table_text_number(compat_field).is_some() => text_field(file_field),
865            Some(compat_field) if can_csv_reader_parse_type(compat_field.data_type()) => {
866                compat_field.clone()
867            }
868            _ => file_field.clone(),
869        })
870        .collect::<Vec<_>>();
871
872    Schema::new_with_metadata(fields, file.metadata().clone())
873}

926fn validate_csv_headers_if_required(file_metadata: &FileMetadata, table: &SchemaRef) -> Result<()> {
927    let FileMetadata::Csv {
928        schema,
929        format,
930        path,
931    } = file_metadata
932    else {
933        return Ok(());
934    };
935
936    if !format.strict_headers {
937        return Ok(());
938    }
939
940    let mut seen_file_columns = HashSet::with_capacity(schema.fields().len());
941    let duplicate_columns = schema
942        .fields()
943        .iter()
944        .filter_map(|field| {
945            if seen_file_columns.insert(field.name().clone()) {
946                None
947            } else {
948                Some(field.name().clone())
949            }
950        })
951        .collect::<BTreeSet<_>>()
952        .into_iter()
953        .collect::<Vec<_>>();
954    let file_columns = seen_file_columns.into_iter().collect::<BTreeSet<_>>();
955    let table_columns = table
956        .fields()
957        .iter()
958        .map(|field| field.name().clone())
959        .collect::<BTreeSet<_>>();
960    let unknown_columns = file_columns
961        .difference(&table_columns)
962        .cloned()
963        .collect::<Vec<_>>();
964    let missing_columns = table_columns
965        .difference(&file_columns)
966        .cloned()
967        .collect::<Vec<_>>();
968
969    ensure!(
970        unknown_columns.is_empty() && missing_columns.is_empty() && duplicate_columns.is_empty(),
971        error::CsvHeaderMismatchSnafu {
972            path,
973            unknown_columns,
974            missing_columns,
975            duplicate_columns,
976        }
977    );
978
979    Ok(())
980}

982/// Generates a maybe compatible schema of the file schema.
983///
984/// If there is a field is found in table schema,
985/// copy the field data type to maybe compatible schema(`compatible_fields`).
986fn generated_schema_projection_and_compatible_file_schema(
987    file: &SchemaRef,
988    table: &SchemaRef,
989) -> (Vec<usize>, Vec<usize>, Schema) {
990    let mut file_projection = Vec::with_capacity(file.fields.len());
991    let mut table_projection = Vec::with_capacity(file.fields.len());
992    let mut compatible_fields = file.fields.iter().cloned().collect::<Vec<_>>();
993    for (file_idx, file_field) in file.fields.iter().enumerate() {
994        if let Some((table_idx, table_field)) = table.fields.find(file_field.name()) {
995            file_projection.push(file_idx);
996            table_projection.push(table_idx);
997
998            // Safety: the compatible_fields has same length as file schema
999            compatible_fields[file_idx] = table_field.clone();
1000        }
1001    }
1002
1003    (
1004        file_projection,
1005        table_projection,
1006        Schema::new(compatible_fields),
1007    )
1008}

1032                table_projection,
1033                compat_file_schema,
1034            }
1035        }
1036    }

1039fn generated_positional_schema_projection_and_compatible_file_schema(
1040    file: &SchemaRef,
1041    table: &SchemaRef,
1042) -> CopyFromSchemaMapping {
1043    let len = file.fields.len().min(table.fields.len());
1044    let file_projection = (0..len).collect::<Vec<_>>();
1045    let table_projection = (0..len).collect::<Vec<_>>();

1047        .fields
1048        .iter()
1049        .enumerate()
1050        .map(|(idx, file_field)| {
1051            if idx < len {
1052                table.fields[idx].clone()
1053            } else {
1054                file_field.clone()
1055            }
1056        })
1057        .collect::<Vec<_>>();
1058
1059    CopyFromSchemaMapping {
1060        file_projection,
1061        table_projection,
1062        compat_file_schema: Schema::new(compatible_fields),
1063    }
