118/// Builds the `semantic_relationships` plan: the service-calls, agent-calls,
119/// co-declared, and declared-edge branches unioned and re-projected to the
120/// 16-column contract. Returns `None` when no source can contribute edges, so
121/// the computed table streams empty.
122pub fn build_relationships_plan(
123    sources: RelationshipSources,
124    window: &GraphQueryWindow,
125) -> DfResult<Option<LogicalPlan>> {
126    let mut union_df = calls_branch(&sources.traces, window)?;
127    if let Some(agent_calls) = agent_calls_branch(&sources.traces, window)? {
128        union_df = union_all(union_df, agent_calls)?;
129    }
130    if let Some(co_declared) = co_declared_branch(sources.co_declared, window)? {
131        union_df = union_all(union_df, co_declared)?;
132    }
133    if let Some(declared) = sources.declared {
134        union_df = union_all(union_df, declared_edges(declared.scan, window)?)?;
135    }
136    union_df
137        .map(|df| {
138            Ok(df
139                .select_columns(&RELATIONSHIP_COLUMNS)?
140                .into_unoptimized_plan())
141        })
142        .transpose()
143}

147/// The same-row co-declared branch: for each source table, every vocabulary
148/// pair whose two entity types the table declares yields an edge projection,
149/// expanded from one scan via `unnest_rows` (the registry idiom). Attribute
150/// edges carry `provenance = 'attribute'`; agent edges are span-structure
151/// observations and carry `provenance = 'trace'`.
152fn co_declared_branch(
153    sources: Vec<CoDeclaredSource>,
154    window: &GraphQueryWindow,
155) -> DfResult<Option<DataFrame>> {
156    let vocabulary = builtin()?;
157    let mut union_df: Option<DataFrame> = None;
158    for source in sources {
159        let find = |ty: &str| source.declarations.iter().find(|d| d.entity_type == ty);
160        let mut pairs = Vec::new();
161        for rule in &vocabulary.co_declared_edges {
162            if let (Some(src), Some(dst)) = (find(&rule.src), find(&rule.dst)) {
163                pairs.push((src, dst, rule.rel.as_str(), PROVENANCE_ATTRIBUTE));
164            }
165        }
166        if source.is_trace {
167            // Agent edges are tied to span structure by the RFC: a non-trace
168            // table co-declaring these types must not fabricate invocations.
169            for rule in &vocabulary.trace_co_declared_edges {
170                if let (Some(src), Some(dst)) = (find(&rule.src), find(&rule.dst)) {
171                    pairs.push((src, dst, rule.rel.as_str(), PROVENANCE_TRACE));
172                }
173            }
174        }
175        let Some((first, _, _, _)) = pairs.first() else {
176            continue;
177        };
178
179        let ts = ident(&first.time_index);
180        let bin = bin_ms(ts.clone());
181        let window_predicate = ts
182            .clone()
183            .gt_eq(window.source_start())
184            .and(ts.lt(window.source_end()));
185        let null_i64 = || lit(ScalarValue::Int64(None));
186        let rows = pairs
187            .iter()
188            .map(|(src, dst, rel_type, provenance)| {
189                // Both endpoints must identify something on the row for the
190                // row to witness the edge.
191                let valid = src
192                    .id_columns
193                    .iter()
194                    .chain(&dst.id_columns)
195                    .fold(lit(true), |predicate, id| predicate.and(identifies(id)));
196                vec![
197                    valid,
198                    bin.clone(),
199                    bin.clone(),
200                    bin.clone() + bin_interval(),
201                    bin.clone() + bin_interval(),
202                    lit(src.entity_type.as_str()),
203                    entity_id_expr(&src.id_columns, &|c| ident(c)),
204                    lit(dst.entity_type.as_str()),
205                    entity_id_expr(&dst.id_columns, &|c| ident(c)),
206                    lit(*rel_type),
207                    lit(*provenance),
208                    lit(1.0_f64),
209                    null_i64(),
210                    null_i64(),
211                    lit(ScalarValue::Float64(None)),
212                    null_i64(),
213                    null_json(),
214                ]
215            })
216            .collect();
217
218        let branch = unnest_rows(
219            source.scan,
220            window_predicate,
221            CO_DECLARED_VALID_COLUMN,
222            &RELATIONSHIP_COLUMNS,
223            rows,
224        )?;
225        union_df = union_all(union_df, branch)?;
226    }
227    // The per-source DISTINCT inside `unnest_rows` cannot see other sources:
228    // two tables witnessing the same edge in the same window must still fold
229    // into one row per (window, edge).
230    union_df.map(DataFrame::distinct).transpose()
231}

233/// Ranking column used to pick the latest declared-edge revision per edge key.
234const DECLARED_REVISION_COLUMN: &str = "__declared_revision";
235
236/// The declared-edge branch: the latest revision per edge key *as of the
237/// queried window* (mito dedups on primary key + `observed_at`, so re-asserting
238/// an edge stores a new revision), filtered by business-validity overlap.
239///
240/// `valid_from` defaults to the declaration time; a NULL `valid_until` means
241/// the edge holds while its row exists (TTL retires it), so `window_end` /
242/// `fresh_until` take the window's upper bound. The output `observed_at` is
243/// synthesized inside the queried range: the scan's filters are re-applied
244/// above the computed table, and the physical revision time would fail them.
245fn declared_edges(scan: DataFrame, window: &GraphQueryWindow) -> DfResult<DataFrame> {
246    let eff_valid_from =
247        core_fns::coalesce().call(vec![ident(VALID_FROM_COLUMN), ident(OBSERVED_AT_COLUMN)]);
248    let eff_valid_until =
249        core_fns::coalesce().call(vec![ident(VALID_UNTIL_COLUMN), window.observed_end()]);
250
251    // As-of the queried window: a revision recorded after its end, or whose
252    // validity starts after it, was not the edge's state inside the window and
253    // must not outrank (and thereby hide) the revision that was.
254    let as_of = ident(OBSERVED_AT_COLUMN)
255        .lt(window.observed_end())
256        .and(eff_valid_from.clone().lt(window.observed_end()));
257
258    let revision = row_number()
259        .partition_by(
260            DECLARED_EDGE_IDENTITY_COLUMNS
261                .iter()
262                .map(|c| ident(*c))
263                .collect(),
264        )
265        // generation_id/scope break observed_at ties deterministically (rows
266        // differing only in them are not merged by the storage dedup).
267        .order_by(vec![
268            ident(OBSERVED_AT_COLUMN).sort(false, false),
269            ident(GENERATION_ID_COLUMN).sort(false, false),
270            ident(ENTITY_SCOPE_COLUMN).sort(false, false),
271        ])
272        .build()?
273        .alias(DECLARED_REVISION_COLUMN);
274
275    // The as-of filter already bounds eff_valid_from below the window's end;
276    // only the retirement side of the overlap remains to check.
277    let still_valid = ident(VALID_UNTIL_COLUMN)
278        .is_null()
279        .or(ident(VALID_UNTIL_COLUMN).gt(window.observed_start()));
280    // Tags come out of the storage engine dictionary-encoded; cast to plain
281    // strings so the union with the derived branches type-aligns.
282    let tag_utf8 = |name: &str| cast(ident(name), DataType::Utf8).alias(name);
283
284    scan.filter(as_of)?
285        .window(vec![revision])?
286        .filter(ident(DECLARED_REVISION_COLUMN).eq(lit(1_u64)))?
287        .filter(still_valid)?
288        .select(vec![
289            core_fns::greatest()
290                .call(vec![eff_valid_from.clone(), window.observed_start()])
291                .alias(OBSERVED_AT_COLUMN),
292            eff_valid_from.alias(WINDOW_START_COLUMN),
293            eff_valid_until.clone().alias(WINDOW_END_COLUMN),
294            eff_valid_until.alias(FRESH_UNTIL_COLUMN),
295            tag_utf8(SRC_TYPE_COLUMN),
296            tag_utf8(SRC_ID_COLUMN),
297            tag_utf8(DST_TYPE_COLUMN),
298            tag_utf8(DST_ID_COLUMN),
299            tag_utf8(REL_TYPE_COLUMN),
300            tag_utf8(PROVENANCE_COLUMN),
301            ident(CONFIDENCE_COLUMN),
302            ident(REQUEST_COUNT_COLUMN),
303            ident(ERROR_COUNT_COLUMN),
304            ident(DURATION_SUM_COLUMN),
305            ident(DURATION_COUNT_COLUMN),
306            ident(EDGE_ATTRIBUTES_COLUMN),
307        ])
308}

310/// Confidence of a virtual-node edge: the peer was named by a client-side
311/// attribute, not witnessed by a server span (the RFC requires `< 1.0`).
312const VIRTUAL_NODE_CONFIDENCE: f64 = 0.5;
313
314/// The `calls` derivation: union the client spans and the server spans of all
315/// trace tables (each projected to a normalized shape with its own table's
316/// `service` identity), left-join clients to their child server spans on
317/// `trace_id` + `parent_span_id`, and aggregate RED metrics per 60s window —
318/// the plan form of the Tempo servicegraph connector. Unioning spans before
319/// the join pairs a client with a server stored in a *different* trace table
320/// (`x-greptime-trace-table-name` routing); with a single table it degenerates
321/// to the previous self-join. Returns `None` when no trace table has a
322/// usable `service` declaration.
323///
324/// A client with no matching server span is an edge to a **virtual node**
325/// named by the conventions' virtual-destination candidates, with the
326/// client's own status/duration
327/// and `confidence < 1.0`. Real pairs win: when a window's edge key holds any
328/// pair, the edge reports only the pair population (the RFC pins RED metrics
329/// to observed span pairs) and the unmatched clients are suppressed, so one
330/// `(window, edge)` never yields two rows.
331fn calls_branch(traces: &[CallsSource], window: &GraphQueryWindow) -> DfResult<Option<DataFrame>> {
332    let mut clients: Option<DataFrame> = None;
333    let mut servers: Option<DataFrame> = None;
334    for trace in traces {
335        let Some(service) = &trace.service else {
336            continue;
337        };
338        clients = union_all(clients, client_spans(service, &trace.scan, window)?)?;
339        servers = union_all(servers, server_spans(service, trace.scan.clone(), window)?)?;
340    }
341    let (Some(clients), Some(servers)) = (clients, servers) else {
342        return Ok(None);
343    };
344
345    let client = clients.alias("client")?;
346    let server = servers.alias("server")?;
347    let join_conditions = vec![
348        qcol("client", TRACE_ID_COLUMN).eq(qcol("server", TRACE_ID_COLUMN)),
349        qcol("server", PARENT_SPAN_ID_COLUMN).eq(qcol("client", SPAN_ID_COLUMN)),
350        qcol("server", TRACE_TIMESTAMP_COLUMN)
351            .gt_eq(qcol("client", TRACE_TIMESTAMP_COLUMN) - interval(CHILD_SPAN_EARLY_NANOS)),
352        qcol("server", TRACE_TIMESTAMP_COLUMN)
353            .lt_eq(qcol("client", TRACE_TIMESTAMP_COLUMN) + interval(CHILD_SPAN_LATE_NANOS)),
354    ];
355
356    let observations = client
357        .join_on(server, JoinType::Left, join_conditions)?
358        .select(vec![
359            bin_ms(qcol("client", TRACE_TIMESTAMP_COLUMN)).alias(OBSERVED_AT_COLUMN),
360            qcol("client", SRC_ID_COLUMN).alias(SRC_ID_COLUMN),
361            core_fns::coalesce()
362                .call(vec![
363                    qcol("server", DST_ID_COLUMN),
364                    qcol("client", "virtual_dst"),
365                ])
366                .alias(DST_ID_COLUMN),
367            qcol("server", TRACE_ID_COLUMN)
368                .is_not_null()
369                .alias("paired"),
370            qcol("server", SPAN_STATUS_CODE_COLUMN).alias("server_status"),
371            qcol("server", DURATION_NANO_COLUMN).alias("server_duration_nano"),
372            qcol("client", SPAN_STATUS_CODE_COLUMN).alias("client_status"),
373            qcol("client", DURATION_NANO_COLUMN).alias("client_duration_nano"),
374            qcol("client", "virtual_conn").alias("virtual_conn"),
375        ])?
376        // No destination means no edge; a self-call is not an edge between
377        // two distinct entities.
378        .filter(
379            ident(DST_ID_COLUMN)
380                .is_not_null()
381                .and(ident(SRC_ID_COLUMN).not_eq(ident(DST_ID_COLUMN))),
382        )?;
383
384    let paired = ident("paired");
385    let df = observations
386        .aggregate(
387            vec![
388                ident(OBSERVED_AT_COLUMN),
389                ident(SRC_ID_COLUMN),
390                ident(DST_ID_COLUMN),
391            ],
392            vec![
393                count(lit(1))
394                    .filter(paired.clone())
395                    .build()?
396                    .alias("pair_count"),
397                count(lit(1))
398                    .filter(
399                        paired
400                            .clone()
401                            .and(ident("server_status").eq(lit(SPAN_STATUS_ERROR))),
402                    )
403                    .build()?
404                    .alias("pair_errors"),
405                sum(ident("server_duration_nano"))
406                    .filter(paired.clone())
407                    .build()?
408                    .alias("pair_duration_nano"),
409                count(lit(1))
410                    .filter(!paired.clone())
411                    .build()?
412                    .alias("unmatched_count"),
413                count(lit(1))
414                    .filter(
415                        (!paired.clone()).and(ident("client_status").eq(lit(SPAN_STATUS_ERROR))),
416                    )
417                    .build()?
418                    .alias("unmatched_errors"),
419                sum(ident("client_duration_nano"))
420                    .filter(!paired.clone())
421                    .build()?
422                    .alias("unmatched_duration_nano"),
423                bool_or(paired.clone()).alias("has_pair"),
424                // `min` makes a mixed-provenance virtual edge deterministic
425                // (`database` sorts before `virtual_node`).
426                min(ident("virtual_conn"))
427                    .filter(!paired)
428                    .build()?
429                    .alias("virtual_conn"),
430            ],
431        )?
432        .select(vec![
433            ident(OBSERVED_AT_COLUMN),
434            ident(OBSERVED_AT_COLUMN).alias(WINDOW_START_COLUMN),
435            (ident(OBSERVED_AT_COLUMN) + bin_interval()).alias(WINDOW_END_COLUMN),
436            (ident(OBSERVED_AT_COLUMN) + bin_interval()).alias(FRESH_UNTIL_COLUMN),
437            lit(ENTITY_TYPE_SERVICE).alias(SRC_TYPE_COLUMN),
438            ident(SRC_ID_COLUMN),
439            lit(ENTITY_TYPE_SERVICE).alias(DST_TYPE_COLUMN),
440            ident(DST_ID_COLUMN),
441            lit(REL_TYPE_CALLS).alias(REL_TYPE_COLUMN),
442            lit(PROVENANCE_TRACE).alias(PROVENANCE_COLUMN),
443            real_wins(lit(1.0_f64), lit(VIRTUAL_NODE_CONFIDENCE))?.alias(CONFIDENCE_COLUMN),
444            real_wins(ident("pair_count"), ident("unmatched_count"))?.alias(REQUEST_COUNT_COLUMN),
445            real_wins(ident("pair_errors"), ident("unmatched_errors"))?.alias(ERROR_COUNT_COLUMN),
446            // duration sums in nanoseconds; the contract column is seconds.
447            (cast(
448                real_wins(
449                    ident("pair_duration_nano"),
450                    ident("unmatched_duration_nano"),
451                )?,
452                DataType::Float64,
453            ) / lit(1e9_f64))
454            .alias(DURATION_SUM_COLUMN),
455            real_wins(ident("pair_count"), ident("unmatched_count"))?.alias(DURATION_COUNT_COLUMN),
456            real_wins(null_json(), virtual_attrs_expr()?)?.alias(EDGE_ATTRIBUTES_COLUMN),
457        ])?;
458    Ok(Some(df))
459}

461/// `CASE WHEN has_pair THEN real ELSE virtual END` — the real-wins projection
462/// over the mixed aggregate.
463fn real_wins(real: Expr, r#virtual: Expr) -> DfResult<Expr> {
464    when(ident("has_pair"), real).otherwise(r#virtual)
465}

467/// The `attributes` JSON of a virtual edge, from the aggregated

481/// The window + span-kind + identity predicate shared by both join sides.
482/// `strict` bounds the scan to the source window; the non-strict side widens
483/// by the join's time-proximity allowance (the join bounds reference the other
484/// side's timestamp and cannot prune this side's scan on their own).
485fn span_predicate(service: &EntityDeclaration, window: &GraphQueryWindow, strict: bool) -> Expr {
486    let ts = ident(TRACE_TIMESTAMP_COLUMN);
487    let mut predicate = if strict {
488        ts.clone()
489            .gt_eq(window.source_start())
490            .and(ts.lt(window.source_end()))
491    } else {
492        ts.clone()
493            .gt_eq(window.source_start() - interval(CHILD_SPAN_EARLY_NANOS))
494            .and(ts.lt(window.source_end() + interval(CHILD_SPAN_LATE_NANOS)))
495    };
496    // An absent identity component identifies nothing, on either endpoint.
497    for id in &service.id_columns {
498        predicate = predicate.and(identifies(id));
499    }
500    predicate
501}

503/// One trace table's client spans, normalized to
504/// `(timestamp, trace_id, span_id, src_id, status_code, duration_nano,
505/// virtual_dst, virtual_conn)`. `src_id` is built from the table's `service`
506/// declaration, so edges land on exactly the entity ids the registry emits (a
507/// composite identity renders the same sorted `k=v` form); the per-table
508/// projection is what lets tables with different declarations union.
509fn client_spans(
510    service: &EntityDeclaration,
511    scan: &DataFrame,
512    window: &GraphQueryWindow,
513) -> DfResult<DataFrame> {
514    // Attribute columns are dynamic in `greptime_trace_v1` (created on first
515    // use), so only candidate columns present in the table's schema
516    // participate. NULLIF('') lets an empty value fall through to the next
517    // candidate.
518    let present: Vec<(Expr, &str)> = {
519        let schema = scan.schema();
520        builtin()?
521            .virtual_dst_candidates
522            .iter()
523            .filter(|candidate| schema.has_column_with_unqualified_name(&candidate.column))
524            .map(|candidate| {
525                let value = core_fns::nullif().call(vec![
526                    cast(ident(&candidate.column), DataType::Utf8),
527                    lit(""),
528                ]);
529                (value, candidate.connection_type.as_str())
530            })
531            .collect()
532    };
533    let null_utf8 = || lit(ScalarValue::Utf8(None));
534    let virtual_dst = if present.is_empty() {
535        null_utf8()
536    } else {

587/// The `parent_agent calls agent` derivation over trace tables declaring an
588/// `agent` entity: pair each span with its child span (no span-kind filter —
589/// agent spans are typically INTERNAL) across all agent-declaring tables, keep
590/// pairs whose agent identities differ, and aggregate RED metrics per 60s
591/// window. Anchored like the service derivation on the caller: the parent side
592/// is bounded to the source window and stamps `observed_at`, the child side
593/// widens by the time-proximity allowance and supplies status/duration.
594fn agent_calls_branch(
595    traces: &[CallsSource],
596    window: &GraphQueryWindow,
597) -> DfResult<Option<DataFrame>> {
598    let mut parents: Option<DataFrame> = None;
599    let mut children: Option<DataFrame> = None;
600    for trace in traces {
601        let Some(agent) = &trace.agent else {
602            continue;
603        };
604        let parent = trace
605            .scan
606            .clone()
607            .filter(span_predicate(agent, window, true))?
608            .select(vec![
609                ident(TRACE_TIMESTAMP_COLUMN),
610                ident(TRACE_ID_COLUMN),
611                ident(SPAN_ID_COLUMN),
612                entity_id_expr(&agent.id_columns, &|c| ident(c)).alias(SRC_ID_COLUMN),
613            ])?;
614        let child = trace
615            .scan
616            .clone()
617            .filter(span_predicate(agent, window, false))?
618            .select(vec![
619                ident(TRACE_TIMESTAMP_COLUMN),
620                ident(TRACE_ID_COLUMN),
621                ident(PARENT_SPAN_ID_COLUMN),
622                entity_id_expr(&agent.id_columns, &|c| ident(c)).alias(DST_ID_COLUMN),
623                ident(SPAN_STATUS_CODE_COLUMN),
624                ident(DURATION_NANO_COLUMN),
625            ])?;
626        parents = union_all(parents, parent)?;
627        children = union_all(children, child)?;
628    }
629    let (Some(parents), Some(children)) = (parents, children) else {
630        return Ok(None);
631    };
632
633    let parent = parents.alias("parent")?;
634    let child = children.alias("child")?;
635    let join_conditions = vec![
636        qcol("parent", TRACE_ID_COLUMN).eq(qcol("child", TRACE_ID_COLUMN)),
637        qcol("child", PARENT_SPAN_ID_COLUMN).eq(qcol("parent", SPAN_ID_COLUMN)),
638        qcol("child", TRACE_TIMESTAMP_COLUMN)
639            .gt_eq(qcol("parent", TRACE_TIMESTAMP_COLUMN) - interval(CHILD_SPAN_EARLY_NANOS)),
640        qcol("child", TRACE_TIMESTAMP_COLUMN)
641            .lt_eq(qcol("parent", TRACE_TIMESTAMP_COLUMN) + interval(CHILD_SPAN_LATE_NANOS)),
642    ];
643
644    let df = parent
645        .join_on(child, JoinType::Inner, join_conditions)?
646        .select(vec![
647            bin_ms(qcol("parent", TRACE_TIMESTAMP_COLUMN)).alias(OBSERVED_AT_COLUMN),
648            qcol("parent", SRC_ID_COLUMN).alias(SRC_ID_COLUMN),
649            qcol("child", DST_ID_COLUMN).alias(DST_ID_COLUMN),
650            qcol("child", SPAN_STATUS_CODE_COLUMN).alias("status_code"),
651            qcol("child", DURATION_NANO_COLUMN).alias(DURATION_NANO_COLUMN),
652        ])?
653        // A sub-span of the same agent is internal structure, not a call
654        // between two agents.
655        .filter(ident(SRC_ID_COLUMN).not_eq(ident(DST_ID_COLUMN)))?
656        .aggregate(
657            vec![
658                ident(OBSERVED_AT_COLUMN),
659                ident(SRC_ID_COLUMN),
660                ident(DST_ID_COLUMN),
661            ],
662            vec![
663                count(lit(1)).alias(REQUEST_COUNT_COLUMN),
664                count(lit(1))
665                    .filter(ident("status_code").eq(lit(SPAN_STATUS_ERROR)))
666                    .build()?
667                    .alias(ERROR_COUNT_COLUMN),
668                sum(ident(DURATION_NANO_COLUMN)).alias("duration_nano_sum"),
669            ],
670        )?
671        .select(vec![
672            ident(OBSERVED_AT_COLUMN),
673            ident(OBSERVED_AT_COLUMN).alias(WINDOW_START_COLUMN),
674            (ident(OBSERVED_AT_COLUMN) + bin_interval()).alias(WINDOW_END_COLUMN),
675            (ident(OBSERVED_AT_COLUMN) + bin_interval()).alias(FRESH_UNTIL_COLUMN),
676            lit(ENTITY_TYPE_GEN_AI_AGENT).alias(SRC_TYPE_COLUMN),
677            ident(SRC_ID_COLUMN),
678            lit(ENTITY_TYPE_GEN_AI_AGENT).alias(DST_TYPE_COLUMN),
679            ident(DST_ID_COLUMN),
680            lit(REL_TYPE_CALLS).alias(REL_TYPE_COLUMN),
681            lit(PROVENANCE_TRACE).alias(PROVENANCE_COLUMN),
682            lit(1.0_f64).alias(CONFIDENCE_COLUMN),
683            ident(REQUEST_COUNT_COLUMN),
684            ident(ERROR_COUNT_COLUMN),
685            (cast(ident("duration_nano_sum"), DataType::Float64) / lit(1e9_f64))
686                .alias(DURATION_SUM_COLUMN),
687            ident(REQUEST_COUNT_COLUMN).alias(DURATION_COUNT_COLUMN),
688            null_json().alias(EDGE_ATTRIBUTES_COLUMN),
689        ])?;
690    Ok(Some(df))
691}
