50/// Dispatches already partitioned writes, retaining only failed payloads for retries.
51pub struct WriteDispatcher {
52    node_manager: NodeManagerRef,
53    partition_manager: PartitionRuleManagerRef,
54    retry: Option<WriteRetry>,
55}
56
57/// Shared reference to a [`WriteDispatcher`].
58pub type WriteDispatcherRef = Arc<WriteDispatcher>;

60/// The result of one write RPC attempt.
61pub(crate) enum WriteAttempt {
62    Success(RegionResponse),
63    Failed {
64        error: Error,
65        /// Retained when retries are enabled and the original client error allows retry.
66        /// The remaining retry budget is checked separately.
67        request: Option<RoutedWrite>,
68    },
69}

81    /// Enables shared remote sends and retries using frontend configuration and caches.
82    pub fn with_retry(
83        mut self,
84        options: WriteRetryOptions,
85        catalog_manager: CatalogManagerRef,
86        cache_invalidator: CacheInvalidatorRef,
87    ) -> Self {
88        if options.max_retries > 0 {
89            self.retry = Some(WriteRetry {
90                node_manager: self.node_manager.clone(),
91                partition_manager: self.partition_manager.clone(),
92                options,
93                catalog_manager,
94                cache_invalidator,
95            });
96        } else {
97            self.retry = None;
98        }
99        self
100    }

116    /// Dispatches shared batches with row ownership retained across retries.
117    pub async fn dispatch_attributed(
118        &self,
119        requests: Vec<(RegionRequest, WriteAttribution)>,
120        histogram: Option<Histogram>,
121    ) -> Result<RegionResponse> {
122        let regions = requests
123            .iter()
124            .map(|(request, _)| write_region_id(request))
125            .collect::<Result<Vec<_>>>()?;
126        let peers = self
127            .resolve_peers(&regions)
128            .await
129            .map_err(BoxedError::new)
130            .context(ExternalSnafu)?;
131        let writes = requests
132            .into_iter()
133            .zip(regions)
134            .map(|((request, attribution), region)| {
135                Ok((get_peer(&peers, region)?, request, attribution))
136            })
137            .collect::<Result<Vec<_>>>()?;
138        let attempts = future::join_all(writes.into_iter().map(|(peer, request, attribution)| {
139            self.send_attributed(peer, request, attribution, histogram.as_ref())
140        }))
141        .await;
142        self.finish(attempts, histogram.as_ref(), None).await
143    }

145    /// Resolves routes before writes are detached from the caller.
146    pub(crate) async fn prepare(
147        &self,
148        requests: Vec<RegionRequest>,
149    ) -> error::Result<Vec<(PeerRef, RegionRequest)>> {
150        let requests = requests
151            .into_iter()
152            .map(|mut request| {
153                let header = request.header.take();
154                (header, split_request(request))
155            })
156            .collect::<Vec<_>>();
157        let regions = requests
158            .iter()
159            .flat_map(|(_, requests)| requests)
160            .map(write_region_id)
161            .collect::<Result<Vec<_>>>()
162            .context(RequestRegionSnafu)?;
163        let peers = self
164            .resolve_peers(&regions)
165            .await
166            .context(FindRegionLeaderSnafu)?;
167        let mut writes = Vec::new();
168        // Preserve each original envelope and peer batching when retries are disabled.
169        for (header, requests) in requests {
170            let mut by_peer: HashMap<PeerRef, RegionRequest> = HashMap::new();
171            for request in requests {
172                let peer = get_peer(
173                    &peers,
174                    write_region_id(&request).context(RequestRegionSnafu)?,
175                )
176                .context(RequestRegionSnafu)?;
177                match by_peer.entry(peer) {
178                    Entry::Vacant(entry) => {
179                        entry.insert(request);
180                    }
181                    Entry::Occupied(mut entry) => match (&mut entry.get_mut().body, request.body) {
182                        (Some(Body::Inserts(existing)), Some(Body::Inserts(next))) => {
183                            existing.requests.extend(next.requests)
184                        }
185                        (Some(Body::Deletes(existing)), Some(Body::Deletes(next))) => {
186                            existing.requests.extend(next.requests)
187                        }
188                        _ => {
189                            return UnexpectedSnafu {
190                                err_msg: "Cannot combine different write kinds",
191                            }
192                            .fail()
193                            .context(RequestRegionSnafu);
194                        }
195                    },
196                }
197            }
198            for (peer, mut request) in by_peer {
199                request.header = header.clone();
200                writes.push((peer, request));
201            }
202        }
203        Ok(writes)
204    }

206    /// Sends requests using routes already resolved by the caller.
207    /// Cancellation stops future retries while allowing in-flight requests to complete.
208    pub(crate) async fn execute(
209        &self,
210        writes: Vec<(PeerRef, RegionRequest)>,
211        histogram: Option<Histogram>,
212        cancellation: Option<&CancellationToken>,
213    ) -> Result<RegionResponse> {
214        self.execute_with_progress(writes, histogram, cancellation, None)
215            .await
216    }

218    pub(crate) async fn execute_with_progress(
219        &self,
220        writes: Vec<(PeerRef, RegionRequest)>,
221        histogram: Option<Histogram>,
222        cancellation: Option<&CancellationToken>,
223        progress: Option<Arc<WriteProgress>>,
224    ) -> Result<RegionResponse> {
225        let histogram = histogram.as_ref();
226        let progress = progress.as_ref();
227        let attempts = future::join_all(writes.into_iter().map(|(peer, request)| async move {
228            if self.retry.is_some() || progress.is_some() {
229                self.send_batch(peer, request, histogram, progress).await
230            } else {
231                vec![self.send(peer, request, histogram).await]
232            }
233        }))
234        .await
235        .into_iter()
236        .flatten()
237        .collect();
238        self.finish(attempts, histogram, cancellation).await
239    }

241    /// Sends the first attempt immediately so bulk encoding can overlap other sends.
242    pub(crate) async fn send(
243        &self,
244        peer: PeerRef,
245        request: RegionRequest,
246        histogram: Option<&Histogram>,
247    ) -> WriteAttempt {
248        self.send_attributed(peer, request, WriteAttribution::default(), histogram)
249            .await
250    }
251
252    async fn send_attributed(
253        &self,
254        peer: PeerRef,
255        mut request: RegionRequest,
256        attribution: WriteAttribution,
257        histogram: Option<&Histogram>,
258    ) -> WriteAttempt {
259        if let Some(retry) = &self.retry {
260            if matches!(&request.body, Some(Body::Inserts(_) | Body::Deletes(_))) {
261                api::region::set_accept_partial_result(request.header.get_or_insert_default());
262            }
263            return retry
264                .send(
265                    RoutedWrite {
266                        peer,
267                        request: Arc::new(request),
268                        attribution,
269                    },
270                    histogram,
271                )
272                .await;
273        }
274        let _timer = histogram.map(Histogram::start_timer);
275        let datanode = self.node_manager.datanode(&peer).await;
276        match datanode.handle(request).await {
277            Ok(response) => acknowledged(response, &attribution),
278            Err(error) => WriteAttempt::Failed {
279                error,
280                request: None,
281            },
282        }
283    }

285    /// Collects first attempts and retries their failures together, refreshing routes once per round.
286    pub(crate) async fn finish(
287        &self,
288        attempts: Vec<WriteAttempt>,
289        histogram: Option<&Histogram>,
290        cancellation: Option<&CancellationToken>,
291    ) -> Result<RegionResponse> {
292        if let Some(retry) = &self.retry {
293            if retry.should_retry(&attempts, 0) {
294                return retry.retry(attempts, histogram, cancellation).await;
295            }
296            let mut response = RegionResponse::new(0);
297            let failures = merge_attempts(attempts, &mut response)?;
298            return match failures.into_iter().next() {
299                Some((_, error)) => Err(error),
300                None => Ok(response),
301            };
302        }
303        let mut response = RegionResponse::new(0);
304        for attempt in attempts {
305            match attempt {
306                WriteAttempt::Success(result) => merge_response(&mut response, result),
307                WriteAttempt::Failed { error, .. } => return Err(error),
308            }
309        }
310        Ok(response)
311    }

321    async fn send_batch(
322        &self,
323        peer: PeerRef,
324        mut request: RegionRequest,
325        histogram: Option<&Histogram>,
326        progress: Option<&Arc<WriteProgress>>,
327    ) -> Vec<WriteAttempt> {
328        if !matches!(&request.body, Some(Body::Inserts(_) | Body::Deletes(_))) {
329            return vec![self.send(peer, request, histogram).await];
330        }
331        // Upgrade datanodes first and roll back frontends first: old datanodes
332        // cannot report which entries succeeded, so retrying a batch may replay writes.
333        api::region::set_accept_partial_result(request.header.get_or_insert_default());
334        if self.retry.is_none() {
335            // Standalone consumes owned requests; retaining an Arc prevents local dispatch.
336            let requester = self.node_manager.datanode(&peer).await;
337            let _timer = histogram.map(Histogram::start_timer);
338            return match requester.handle(request).await {
339                Ok(response) => {
340                    if let Some(progress) = progress {
341                        progress.add_written(response.affected_rows);
342                    }
343                    vec![WriteAttempt::Success(response)]
344                }
345                Err(Error::PartialRegion { response }) => {
346                    if let Some(progress) = progress {
347                        progress.add_written(response.affected_rows);
348                    }
349                    let mut attempts = vec![WriteAttempt::Success(RegionResponse {
350                        affected_rows: response.affected_rows,
351                        extensions: response.extensions,
352                        metadata: Vec::new(),
353                    })];
354                    attempts.extend(response.failures.into_iter().map(|(_, error)| {
355                        WriteAttempt::Failed {
356                            error,
357                            request: None,
358                        }
359                    }));
360                    attempts
361                }
362                Err(error) => vec![WriteAttempt::Failed {
363                    error,
364                    request: None,
365                }],
366            };
367        }
368        let request = Arc::new(request);
369        let requester = self.node_manager.datanode(&peer).await;
370        let result = {
371            let _timer = histogram.map(Histogram::start_timer);
372            requester.handle(Arc::clone(&request)).await
373        };
374        let error = match result {
375            Ok(response) => {
376                if let Some(progress) = progress {
377                    progress.add_written(response.affected_rows);
378                }
379                return vec![WriteAttempt::Success(response)];
380            }
381            Err(error) => error,
382        };
383        let request = match Arc::try_unwrap(request) {
384            Ok(request) => request,
385            Err(_) => {
386                return vec![WriteAttempt::Failed {
387                    error: UnexpectedSnafu {
388                        err_msg: "Region request is still shared after its RPC completed",
389                    }
390                    .build(),
391                    request: None,
392                }];
393            }
394        };
395        let requests = split_request(request);
396        match error {
397            Error::PartialRegion { response } => {
398                if let Some(progress) = progress {
399                    progress.add_written(response.affected_rows);
400                }
401                let mut failures: HashMap<_, _> = response.failures.into_iter().collect();
402                let mut attempts = vec![WriteAttempt::Success(RegionResponse {
403                    affected_rows: response.affected_rows,
404                    extensions: response.extensions,
405                    metadata: Vec::new(),
406                })];
407                for (index, request) in requests.into_iter().enumerate() {
408                    if let Some(error) = failures.remove(&index) {
409                        let request = error.retry_hint().is_retryable().then(|| RoutedWrite {
410                            peer: Arc::clone(&peer),
411                            attribution: direct_attribution(&request, progress.cloned()),
412                            request: Arc::new(request),
413                        });
414                        attempts.push(WriteAttempt::Failed { error, request });
415                    }
416                }
417                attempts
418            }
419            error => {
420                // A whole-RPC failure cannot confirm success for any entry.
421                let error = Arc::new(error);
422                requests
423                    .into_iter()
424                    .map(|request| {
425                        let request = error.retry_hint().is_retryable().then(|| RoutedWrite {
426                            peer: Arc::clone(&peer),
427                            attribution: direct_attribution(&request, progress.cloned()),
428                            request: Arc::new(request),
429                        });
430                        WriteAttempt::Failed {
431                            error: BatchRequestSnafu.into_error(Arc::clone(&error)),
432                            request,
433                        }
434                    })
435                    .collect()
436            }
437        }
438    }

454/// Sends shared requests and coordinates retries with refreshed routes.
455struct WriteRetry {
456    node_manager: NodeManagerRef,
457    partition_manager: PartitionRuleManagerRef,
458    options: WriteRetryOptions,
459    catalog_manager: CatalogManagerRef,
460    cache_invalidator: CacheInvalidatorRef,
461}

463/// A single-region write request paired with its destination peer.
464pub(crate) struct RoutedWrite {
465    peer: PeerRef,
466    request: Arc<RegionRequest>,
467    attribution: WriteAttribution,
468}
469
470impl WriteRetry {
471    async fn send(&self, write: RoutedWrite, histogram: Option<&Histogram>) -> WriteAttempt {
472        let _timer = histogram.map(Histogram::start_timer);
473        let requester = self.node_manager.datanode(&write.peer).await;
474        match requester.handle(Arc::clone(&write.request)).await {
475            Ok(response) => acknowledged(response, &write.attribution),
476            Err(error) => {
477                let request = error.retry_hint().is_retryable().then_some(write);
478                WriteAttempt::Failed { error, request }
479            }
480        }
481    }
482
483    fn should_retry(&self, attempts: &[WriteAttempt], retries: usize) -> bool {
484        if retries >= self.options.max_retries {
485            return false;
486        }
487        let mut retryable = false;
488        for attempt in attempts {
489            match attempt {
490                WriteAttempt::Success(_) => {}
491                WriteAttempt::Failed {
492                    request: Some(_), ..
493                } => retryable = true,
494                WriteAttempt::Failed { request: None, .. } => return false,
495            }
496        }
497        retryable
498    }
499
500    async fn retry(
501        &self,
502        mut attempts: Vec<WriteAttempt>,
503        histogram: Option<&Histogram>,
504        cancellation: Option<&CancellationToken>,
505    ) -> Result<RegionResponse> {
506        let mut response = RegionResponse::new(0);
507        let mut backoff = ExponentialBuilder::default()
508            .with_min_delay(self.options.initial_backoff)
509            .with_max_delay(self.options.max_backoff)
510            .with_factor(self.options.backoff_multiplier)
511            .with_max_times(self.options.max_retries)
512            .with_jitter()
513            .build();
514        let mut retries = 0;
515        while self.should_retry(&attempts, retries) {
516            let Some(delay) = backoff.next() else {
517                break;
518            };
519            if let Some(cancellation) = cancellation {
520                tokio::select! {
521                    biased;
522                    _ = cancellation.cancelled() => break,
523                    _ = tokio::time::sleep(delay) => {}
524                }
525            } else {
526                tokio::time::sleep(delay).await;
527            }
528            if cancellation.is_some_and(CancellationToken::is_cancelled) {
529                break;
530            }
531            let failures = merge_attempts(attempts, &mut response)?;
532            let mut first_error = None;
533            let mut refresh = Vec::new();
534            let mut pending = Vec::new();
535            for (write, error) in failures {
536                if error.retry_hint() == RetryHint::RetryAfterRouteRefresh {
537                    refresh.push(write);
538                } else {
539                    pending.push(write);
540                }
541                first_error.get_or_insert(error);
542            }
543            if !refresh.is_empty() {
544                pending.extend(self.reroute_failed_writes(refresh).await?);
545            }
546            if let Some(error) =
547                first_error.filter(|_| cancellation.is_some_and(CancellationToken::is_cancelled))
548            {
549                return Err(error);
550            }
551            attempts =
552                future::join_all(pending.into_iter().map(|write| self.send(write, histogram)))
553                    .await;
554            retries += 1;
555        }
556        match merge_attempts(attempts, &mut response)?.into_iter().next() {
557            Some((_, error)) => Err(error),
558            None => Ok(response),
559        }
560    }

562    async fn reroute_failed_writes(&self, writes: Vec<RoutedWrite>) -> Result<Vec<RoutedWrite>> {
563        let regions = writes
564            .iter()
565            .map(|write| write_region_id(&write.request))
566            .collect::<Result<Vec<_>>>()?;
567        self.invalidate_routes(&regions).await?;
568        let mut table_names = HashMap::new();
569        for table_id in regions
570            .iter()
571            .map(|region| region.table_id())
572            .collect::<HashSet<_>>()
573        {
574            // Raw metadata is only used to recover the name of the failed request's table.
575            let table_info = self
576                .catalog_manager
577                .table_info_by_id(table_id)
578                .await
579                .map_err(BoxedError::new)
580                .context(ExternalSnafu)?
581                .with_context(|| TableInfoNotFoundSnafu {
582                    table: table_id.to_string(),
583                })?;
584            table_names.insert(
585                table_id,
586                TableName::new(
587                    &table_info.catalog_name,
588                    &table_info.schema_name,
589                    &table_info.name,
590                ),
591            );
592        }
593        // TableId invalidation refreshes metadata and routes, but TableCache is keyed by name.
594        // Use the refreshed names in case a table was renamed while the write was pending.
595        let caches = table_names
596            .values()
597            .cloned()
598            .map(CacheIdent::TableName)
599            .collect::<Vec<_>>();
600        self.cache_invalidator
601            .invalidate(&Context::default(), &caches)
602            .await?;
603        let mut partitioners = HashMap::new();
604        for (table_id, name) in table_names {
605            let table = self
606                .catalog_manager
607                .table(
608                    &name.catalog_name,
609                    &name.schema_name,
610                    &name.table_name,
611                    None,
612                )
613                .await
614                .map_err(BoxedError::new)
615                .context(ExternalSnafu)?
616                .with_context(|| TableInfoNotFoundSnafu {
617                    table: name.to_string(),
618                })?;
619            let table_info = table.table_info();
620            ensure!(
621                table_info.table_id() == table_id,
622                UnexpectedSnafu {
623                    err_msg: format!(
624                        "Table {name} changed ID from {table_id} to {} while retrying writes",
625                        table_info.table_id()
626                    ),
627                }
628            );
629            let partitioner = TablePartitioner::try_new(&self.partition_manager, &table_info)
630                .await
631                .map_err(BoxedError::new)
632                .context(ExternalSnafu)?;
633            partitioners.insert(table_id, (table_info, partitioner));
634        }
635        let mut rebuilt_requests = Vec::new();
636        for write in writes {
637            let request = Arc::try_unwrap(write.request).map_err(|_| {
638                UnexpectedSnafu {
639                    err_msg: "Write request is still shared after its RPC completed",
640                }
641                .build()
642            })?;
643            let table_id = write_region_id(&request)?.table_id();
644            let (table_info, partitioner) =
645                partitioners.get(&table_id).context(UnexpectedSnafu {
646                    err_msg: "Refreshed write has no partition snapshot",
647                })?;
648            let requests = partitioner
649                .rebuild_attributed_write(request, table_info, write.attribution)
650                .map_err(BoxedError::new)
651                .context(ExternalSnafu)?;
652            rebuilt_requests.extend(requests);
653        }
654        self.prepare(rebuilt_requests).await
655    }

657    async fn prepare(
658        &self,
659        requests: Vec<(RegionRequest, WriteAttribution)>,
660    ) -> Result<Vec<RoutedWrite>> {
661        let regions = requests
662            .iter()
663            .map(|(request, _)| write_region_id(request))
664            .collect::<Result<Vec<_>>>()?;
665        let peers = resolve_peers(&self.partition_manager, &regions)
666            .await
667            .map_err(BoxedError::new)
668            .context(ExternalSnafu)?;
669        requests
670            .into_iter()
671            .zip(regions)
672            .map(|((request, attribution), region)| {
673                Ok(RoutedWrite {
674                    peer: get_peer(&peers, region)?,
675                    request: Arc::new(request),
676                    attribution,
677                })
678            })
679            .collect()
680    }

682    async fn invalidate_routes(&self, regions: &[RegionId]) -> Result<()> {
683        let mut table_ids = regions
684            .iter()
685            .map(|region| region.table_id())
686            .collect::<HashSet<_>>();
687        let mut physical_ids = HashSet::new();
688        for table_id in &table_ids {
689            let (physical_id, _) = self
690                .partition_manager
691                .find_physical_table_route_with_id(*table_id)
692                .await
693                .map_err(BoxedError::new)
694                .context(ExternalSnafu)?;
695            physical_ids.insert(physical_id);
696        }
697        table_ids.extend(physical_ids);
698        let caches = table_ids
699            .into_iter()
700            .map(CacheIdent::TableId)
701            .collect::<Vec<_>>();
702        self.cache_invalidator
703            .invalidate(&Context::default(), &caches)
704            .await
705    }
706}

711        Err(error) => WriteAttempt::Failed {

718pub(crate) fn write_row_count(request: &RegionRequest) -> usize {
719    match &request.body {
720        Some(Body::Inserts(inserts)) => inserts
721            .requests
722            .iter()
723            .filter_map(|request| request.rows.as_ref())
724            .map(|rows| rows.rows.len())
725            .sum(),
726        Some(Body::Deletes(deletes)) => deletes
727            .requests
728            .iter()
729            .filter_map(|request| request.rows.as_ref())
730            .map(|rows| rows.rows.len())
731            .sum(),
732        _ => 0,
733    }
734}

743async fn resolve_peers(
744    partition_manager: &PartitionRuleManager,
745    regions: &[RegionId],
746) -> partition::error::Result<HashMap<RegionId, PeerRef>> {
747    let table_ids = regions
748        .iter()
749        .map(|region| region.table_id())
750        .collect::<HashSet<_>>();
751    let mut physical_tables = HashMap::new();
752    let mut routes = HashMap::new();
753    for table_id in table_ids {
754        let (physical_id, route) = partition_manager
755            .find_physical_table_route_with_id(table_id)
756            .await?;
757        physical_tables.insert(table_id, physical_id);
758        routes.entry(physical_id).or_insert(route);
759    }
760    let mut shared_peers: HashMap<&Peer, PeerRef> = HashMap::new();
761    let mut peers = HashMap::with_capacity(regions.len());
762    for region in regions {
763        if peers.contains_key(region) {
764            continue;
765        }
766        let peer = physical_tables
767            .get(&region.table_id())
768            .and_then(|physical_id| routes.get(physical_id))
769            .and_then(|route| {
770                router::find_region_leader_ref(&route.region_routes, region.region_number())
771            })
772            .context(partition::error::FindLeaderSnafu {
773                region_id: *region,
774                table_id: region.table_id(),
775            })?;
776        let peer = shared_peers
777            .entry(peer)
778            .or_insert_with(|| Arc::new(peer.clone()));
779        peers.insert(*region, Arc::clone(peer));
780    }
781    Ok(peers)
782}

784fn get_peer(peers: &HashMap<RegionId, PeerRef>, region: RegionId) -> Result<PeerRef> {
785    peers.get(&region).cloned().context(UnexpectedSnafu {
786        err_msg: "Planned write has no resolved peer",
787    })
788}

790fn merge_attempts(
791    attempts: Vec<WriteAttempt>,
792    response: &mut RegionResponse,
793) -> Result<Vec<(RoutedWrite, Error)>> {
794    let mut failures = Vec::new();
795    let mut terminal_error = None;
796    for attempt in attempts {
797        match attempt {
798            WriteAttempt::Success(result) => merge_response(response, result),
799            WriteAttempt::Failed {
800                error,
801                request: Some(write),
802            } => failures.push((write, error)),
803            WriteAttempt::Failed {
804                error,
805                request: None,
806            } => {
807                terminal_error.get_or_insert(error);
808            }
809        }
810    }
811    // A terminal error takes precedence over retained retryable failures.
812    match terminal_error {
813        Some(error) => Err(error),
814        None => Ok(failures),
815    }
816}
817
818fn merge_response(response: &mut RegionResponse, result: RegionResponse) {
819    response.affected_rows += result.affected_rows;
820    response.extensions.extend(result.extensions);
821    response.metadata = result.metadata;
822}

824fn split_request(request: RegionRequest) -> Vec<RegionRequest> {
825    let RegionRequest { header, body } = request;
826    let bodies: Vec<Body> = match body {
827        Some(Body::Inserts(inserts)) => inserts
828            .requests
829            .into_iter()
830            .map(|request| {
831                Body::Inserts(InsertRequests {
832                    requests: vec![request],
833                })
834            })
835            .collect(),
836        Some(Body::Deletes(deletes)) => deletes
837            .requests
838            .into_iter()
839            .map(|request| {
840                Body::Deletes(DeleteRequests {
841                    requests: vec![request],
842                })
843            })
844            .collect(),
845        body => return vec![RegionRequest { header, body }],
846    };
847    bodies
848        .into_iter()
849        .map(|body| RegionRequest {
850            header: header.clone(),
851            body: Some(body),
852        })
853        .collect()
854}

856fn write_region_id(request: &RegionRequest) -> Result<RegionId> {
857    let region_id = match &request.body {
858        Some(Body::Inserts(inserts)) => match inserts.requests.as_slice() {
859            [request] => Some(request.region_id),
860            _ => None,
861        },
862        Some(Body::Deletes(deletes)) => match deletes.requests.as_slice() {
863            [request] => Some(request.region_id),
864            _ => None,
865        },
866        Some(Body::BulkInsert(bulk)) => Some(bulk.region_id),
867        _ => None,
868    };
869    region_id.map(RegionId::from_u64).ok_or_else(|| {
870        UnexpectedSnafu {
871            err_msg: "Expected a single-region write request",
872        }
873        .build()
874    })
