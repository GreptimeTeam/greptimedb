15//! Table semantic layer vocabulary.
16//!
17//! A thin layer of semantic metadata attached to a table via `table_options`, so
18//! machine consumers (LLM agents, alert/dashboard builders, MCP servers, ETL) can
19//! align a table with the observability concept it stands for without guessing
20//! from column names. See `docs/rfcs/2026-05-28-table-semantic-layer.md`.
21//!
22//! The vocabulary is intentionally small: a key earns its place only when it
23//! records something a consumer cannot cheaply and reliably recover from the
24//! schema/data itself. Keys whose value is already in the metric name by
25//! convention, is a constant, or duplicates an existing column are deliberately
26//! omitted rather than stamped for completeness.
27//!
28//! All public keys share the [`SEMANTIC_PREFIX`] namespace and are string-valued.
29//! [`is_semantic_option_key`] gates them through
30//! [`crate::requests::validate_table_option`], so they are accepted both on the
31//! ingestion auto-create path and on explicit `CREATE TABLE ... WITH (...)` DDL.
32
33use datatypes::prelude::ConcreteDataType;
34
35/// Reserved prefix for every public semantic table-option key.
36pub const SEMANTIC_PREFIX: &str = "greptime.semantic.";
37
38/// Internal `QueryContext` extension key carrying the per-table semantic index
39/// (a `{table_name -> {semantic_key: value}}` JSON blob) from the ingestion
40/// encode path to the auto-create site. Deliberately OUTSIDE [`SEMANTIC_PREFIX`]
41/// so it is not a valid table option and never leaks into a table's options.
42pub const SEMANTIC_PER_TABLE_INDEX_KEY: &str = "greptime.internal.semantic.per_table_index";

79pub const SEMANTIC_METRIC_ORIGINAL_NAME: &str = "greptime.semantic.metric.original_name";

81// ---- Entity keys (open sub-namespace) ----
82
83/// Reserved prefix for the entity-identity sub-namespace:
84/// `greptime.semantic.entity.<type>.{id|descriptive|scope}`. Unlike the rest of
85/// the vocabulary (a closed whitelist), entity types are open-ended (`service`,
86/// `host`, `k8s.pod`, `process`, `agent`, custom, ...), so keys here are validated
87/// by prefix + shape rather than membership. See
88/// `docs/rfcs/2026-06-25-entity-relationships-and-graph-query.md`.
89pub const SEMANTIC_ENTITY_PREFIX: &str = "greptime.semantic.entity.";
90
91/// The well-known entity-identity key auto-stamped on OTLP trace tables; its value
92/// is the `service_name` tag column, declaring the logical `service` entity.
93pub const SEMANTIC_ENTITY_SERVICE_ID: &str = "greptime.semantic.entity.service.id";
94
95/// The role a set of columns plays for an entity: `id` (identifying
96/// attributes), `descriptive`, or `scope`. Columns may be tags or fields; DDL
97/// validation only requires that they exist and render as stable strings
98/// ([`has_stable_string_form`]).
99#[derive(Debug, Clone, Copy, PartialEq, Eq)]
100pub enum EntityRole {
101    Id,
102    Descriptive,
103    Scope,
104}
105
106impl EntityRole {
107    fn parse(role: &str) -> Option<Self> {
108        match role {
109            "id" => Some(EntityRole::Id),
110            "descriptive" => Some(EntityRole::Descriptive),
111            "scope" => Some(EntityRole::Scope),
112            _ => None,
113        }
114    }
115}

134/// Sentinel for a key that cannot be determined at stamp time.
135pub const SEMANTIC_VALUE_UNKNOWN: &str = "unknown";
136/// Sentinel for a single-valued key that saw conflicting sources.
137pub const SEMANTIC_VALUE_MIXED: &str = "mixed";

139/// Every recognised public semantic table-option key. The set is a closed
140/// whitelist: keys under [`SEMANTIC_PREFIX`] that are not listed here are rejected,
141/// so an unknown key like `greptime.semantic.unknown_key` does not silently land
142/// in a table's options. Adding a key to the vocabulary means adding it here.
143pub const SEMANTIC_OPTION_KEYS: &[&str] = &[
144    SEMANTIC_SIGNAL_TYPE,
145    SEMANTIC_SOURCE,
146    SEMANTIC_SOURCE_VERSION,
147    SEMANTIC_PIPELINE,
148    SEMANTIC_TRACE_CONVENTIONS,
149    SEMANTIC_METRIC_TYPE,
150    SEMANTIC_METRIC_UNIT,
151    SEMANTIC_METRIC_TEMPORALITY,
152    SEMANTIC_METRIC_METADATA_QUALITY,
153    SEMANTIC_METRIC_ORIGINAL_NAME,
154];
155
156/// Returns true if `ty` is a syntactically valid entity type, e.g. `service`,
157/// `host`, `k8s.pod`, `service.instance`. An entity type is one or more
158/// dot-separated segments, each a non-empty `[a-z0-9_]+` token. The dotted form
159/// carries the two-entity-layer convention (`service` vs `service.instance`).
160fn is_valid_entity_type(ty: &str) -> bool {
161    !ty.is_empty()
162        && ty.split('.').all(|seg| {
163            !seg.is_empty()
164                && seg
165                    .bytes()
166                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
167        })
168}
169
170/// Parses a well-formed entity-identity option key of the shape
171/// `greptime.semantic.entity.<type>.{id|descriptive|scope}` into
172/// `(entity_type, role)`. The `<type>` may itself contain dots; the role is the
173/// final dot-separated segment. This is the single parser of the key format —
174/// DDL validation and the read-time derivation both go through it.
175pub fn parse_entity_option_key(key: &str) -> Option<(&str, EntityRole)> {
176    let rest = key.strip_prefix(SEMANTIC_ENTITY_PREFIX)?;
177    let (ty, role) = rest.rsplit_once('.')?;
178    if !is_valid_entity_type(ty) {
179        return None;
180    }
181    Some((ty, EntityRole::parse(role)?))
182}
183
184/// Returns true if `key` is a well-formed entity-identity option key.
185pub fn is_entity_option_key(key: &str) -> bool {
186    parse_entity_option_key(key).is_some()
187}
188
189/// Returns true if a column of `data_type` renders as a stable string — the
190/// requirement for entity id/descriptive/scope columns. The read-time
191/// derivation casts them to strings, so a type without a stable string form
192/// would fail only when the graph is scanned; DDL validation rejects it up
193/// front instead.
194pub fn has_stable_string_form(data_type: &ConcreteDataType) -> bool {
195    !matches!(
196        data_type,
197        ConcreteDataType::Binary(_)
198            | ConcreteDataType::Json(_)
199            | ConcreteDataType::Vector(_)
200            | ConcreteDataType::List(_)
201            | ConcreteDataType::Struct(_)
202            | ConcreteDataType::Dictionary(_)
203            | ConcreteDataType::Null(_)
204    )
205}
206
207/// Tokenizes an entity option's comma-separated column list (trimmed, empty
208/// tokens dropped). [`validate_semantic_option`] rejects empty tokens at DDL
209/// time, so readers only ever drop what validation already refused.
210pub fn parse_entity_columns(value: &str) -> Vec<String> {
211    value
212        .split(',')
213        .map(|c| c.trim().to_string())
214        .filter(|c| !c.is_empty())
215        .collect()
216}
217
218/// Returns true if `key` is a recognised semantic table-option key.
219///
220/// Two acceptance rules: membership in the closed [`SEMANTIC_OPTION_KEYS`]
221/// whitelist, OR the open entity sub-namespace ([`is_entity_option_key`], validated
222/// by prefix + shape). Everything else under [`SEMANTIC_PREFIX`] is rejected, and
223/// the internal [`SEMANTIC_PER_TABLE_INDEX_KEY`] (outside the prefix) never matches.
224pub fn is_semantic_option_key(key: &str) -> bool {
225    SEMANTIC_OPTION_KEYS.contains(&key) || is_entity_option_key(key)
226}
227
228/// Validates a `greptime.semantic.*` option's `value` against its allowed domain.
229///
230/// Open-value keys (unit, original_name, pipeline, conventions) accept any
231/// non-empty string. Closed-domain keys accept a fixed set, plus the `unknown`
232/// sentinel, plus `mixed` for the keys where one long-lived table can
233/// legitimately see multiple values. Entity keys ([`is_entity_option_key`]) take a
234/// comma-separated column-name list (each token non-empty); column existence and
235/// the stable-string-form rule for entity columns are enforced later against
236/// the table schema at DDL time, not here. Keys that are neither whitelisted nor a well-formed entity key
237/// are rejected.
238pub fn validate_semantic_option(key: &str, value: &str) -> bool {
239    if is_entity_option_key(key) {
240        return !value.is_empty() && value.split(',').all(|col| !col.trim().is_empty());
241    }
242    match key {
243        SEMANTIC_PIPELINE
244        | SEMANTIC_SOURCE_VERSION
245        | SEMANTIC_METRIC_UNIT
246        | SEMANTIC_METRIC_ORIGINAL_NAME
247        | SEMANTIC_TRACE_CONVENTIONS => !value.is_empty(),
248
249        SEMANTIC_SIGNAL_TYPE => matches!(value, "trace" | "log" | "metric" | "event" | "unknown"),
250        SEMANTIC_SOURCE => matches!(
251            value,
252            "opentelemetry"
253                | "prometheus"
254                | "influxdb"
255                | "opentsdb"
256                | "elasticsearch"
257                | "loki"
258                | "custom"
259                | "mixed"
260                | "unknown"
261        ),
262        SEMANTIC_METRIC_TYPE => matches!(
263            value,
264            "counter"
265                | "gauge"
266                | "histogram"
267                | "summary"
268                | "updown_counter"
269                | "gauge_histogram"
270                | "info"
271                | "stateset"
272                | "mixed"
273                | "unknown"
274        ),
275        SEMANTIC_METRIC_TEMPORALITY => {
276            matches!(value, "cumulative" | "delta" | "mixed" | "unknown")
277        }
278        SEMANTIC_METRIC_METADATA_QUALITY => matches!(value, "declared" | "inferred" | "unknown"),
279
280        _ => false,
281    }
282}

288    #[test]
289    fn test_is_semantic_option_key() {
290        assert!(is_semantic_option_key(SEMANTIC_SIGNAL_TYPE));
291        assert!(is_semantic_option_key(SEMANTIC_METRIC_TYPE));
292        assert!(is_semantic_option_key(SEMANTIC_PIPELINE));
293
294        // Unknown keys under the prefix are not whitelisted.
295        assert!(!is_semantic_option_key("greptime.semantic.future.key"));
296        assert!(!is_semantic_option_key("greptime.semantic.unknown_key"));
297        // Keys cut from the vocabulary are no longer accepted.
298        assert!(!is_semantic_option_key(
299            "greptime.semantic.metric.monotonic"
300        ));
301        assert!(!is_semantic_option_key(
302            "greptime.semantic.resource.attributes_dropped"
303        ));
304        // Near-misses must not match.
305        assert!(!is_semantic_option_key("greptime.semanticx"));
306        assert!(!is_semantic_option_key("semantic.signal_type"));
307        assert!(!is_semantic_option_key("table_data_model"));
308        // The internal transport key must never be treated as a table option.
309        assert!(!is_semantic_option_key(SEMANTIC_PER_TABLE_INDEX_KEY));
310    }

312    #[test]
313    fn test_validate_semantic_option() {
314        // Enum keys reject out-of-domain values.
315        assert!(validate_semantic_option(SEMANTIC_SIGNAL_TYPE, "metric"));
316        assert!(!validate_semantic_option(SEMANTIC_SIGNAL_TYPE, "spans"));
317        assert!(validate_semantic_option(SEMANTIC_METRIC_TYPE, "counter"));
318        assert!(validate_semantic_option(SEMANTIC_METRIC_TYPE, "mixed"));
319        assert!(!validate_semantic_option(SEMANTIC_METRIC_TYPE, "bogus"));
320
321        // Sentinels and open values.
322        assert!(validate_semantic_option(
323            SEMANTIC_METRIC_TEMPORALITY,
324            "unknown"
325        ));
326        assert!(validate_semantic_option(SEMANTIC_METRIC_UNIT, "By"));
327        assert!(!validate_semantic_option(SEMANTIC_METRIC_UNIT, ""));
328        assert!(validate_semantic_option(
329            SEMANTIC_PIPELINE,
330            "greptime_trace_v1"
331        ));
332
333        // A cut key validates to false regardless of value.
334        assert!(!validate_semantic_option(
335            "greptime.semantic.metric.monotonic",
336            "true"
337        ));
338        // Unknown key is rejected regardless of value.
339        assert!(!validate_semantic_option(
340            "greptime.semantic.future.key",
341            "x"
342        ));
343
344        // Drift guard: every value stamped by the ingestion path must validate.
345        assert!(validate_semantic_option(
346            SEMANTIC_SIGNAL_TYPE,
347            SIGNAL_TYPE_TRACE
348        ));
349        assert!(validate_semantic_option(
350            SEMANTIC_SIGNAL_TYPE,
351            SIGNAL_TYPE_METRIC
352        ));
353        assert!(validate_semantic_option(
354            SEMANTIC_SIGNAL_TYPE,
355            SIGNAL_TYPE_LOG
356        ));
357        assert!(validate_semantic_option(
358            SEMANTIC_SOURCE,
359            SOURCE_OPENTELEMETRY
360        ));
361        assert!(validate_semantic_option(SEMANTIC_SOURCE, SOURCE_PROMETHEUS));
362        assert!(validate_semantic_option(SEMANTIC_SOURCE_VERSION, "2.0"));
363        assert!(validate_semantic_option(
364            SEMANTIC_METRIC_METADATA_QUALITY,
365            METADATA_QUALITY_INFERRED
366        ));
367        assert!(validate_semantic_option(
368            SEMANTIC_METRIC_METADATA_QUALITY,
369            METADATA_QUALITY_DECLARED
370        ));
371        assert!(validate_semantic_option(
372            SEMANTIC_TRACE_CONVENTIONS,
373            SEMANTIC_VALUE_UNKNOWN
374        ));
375        // An empty value never validates, for any whitelisted key.
376        for key in SEMANTIC_OPTION_KEYS {
377            assert!(
378                !validate_semantic_option(key, ""),
379                "empty value should never validate for {key}"
380            );
381        }
382    }

384    #[test]
385    fn test_entity_option_key() {
386        // Well-formed entity keys are accepted by the open sub-namespace, and are
387        // therefore recognised as semantic option keys.
388        for key in [
389            "greptime.semantic.entity.service.id",
390            "greptime.semantic.entity.k8s.pod.id",
391            "greptime.semantic.entity.service.instance.descriptive",
392            "greptime.semantic.entity.host.scope",
393            "greptime.semantic.entity.agent.id",
394        ] {
395            assert!(is_entity_option_key(key), "should accept {key}");
396            assert!(is_semantic_option_key(key), "should recognise {key}");
397        }
398
399        // Malformed: empty type segment, bogus role, missing role, bare prefix,
400        // invalid (uppercase) charset in the type.
401        for key in [
402            "greptime.semantic.entity..id",
403            "greptime.semantic.entity.service.bogusrole",
404            "greptime.semantic.entity.service",
405            "greptime.semantic.entity.",
406            "greptime.semantic.entity.Service.id",
407        ] {
408            assert!(!is_entity_option_key(key), "should reject {key}");
409            assert!(!is_semantic_option_key(key), "should not recognise {key}");
410        }
411
412        // A non-entity semantic key is not an entity key.
413        assert!(!is_entity_option_key(SEMANTIC_SIGNAL_TYPE));
414
415        // Drift guard: the auto-stamped constant is a well-formed entity key.
416        assert!(is_entity_option_key(SEMANTIC_ENTITY_SERVICE_ID));
417        assert!(is_semantic_option_key(SEMANTIC_ENTITY_SERVICE_ID));
418    }

420    #[test]
421    fn test_validate_entity_option() {
422        // Single- and composite-id column lists validate.
423        assert!(validate_semantic_option(
424            "greptime.semantic.entity.service.id",
425            "service_name"
426        ));
427        assert!(validate_semantic_option(
428            "greptime.semantic.entity.process.id",
429            "pid,start_time"
430        ));
431        // Empty value and blank/empty tokens do not.
432        assert!(!validate_semantic_option(
433            "greptime.semantic.entity.service.id",
434            ""
435        ));
436        assert!(!validate_semantic_option(
437            "greptime.semantic.entity.process.id",
438            "pid,"
439        ));
440        // A malformed entity key validates to false regardless of value.
441        assert!(!validate_semantic_option(
442            "greptime.semantic.entity.service.bogusrole",
443            "service_name"
444        ));
445    }
