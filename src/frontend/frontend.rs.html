44/// Configures frontend protocols, query execution, and distributed write retries.
45#[derive(Clone, Debug, Serialize, Deserialize, PartialEq)]
46#[serde(default)]
47pub struct FrontendOptions {
48    pub node_id: Option<String>,
49    pub default_timezone: Option<String>,
50    pub default_column_prefix: Option<String>,
51    /// Server-side global switch for auto table creation on write.
52    /// Acts as an upper bound: when `false`, missing tables are never auto-created
53    /// even if a request sets the `auto_create_table` hint to `true`. When `true`
54    /// (default), the per-request hint still applies. Default: `true`.
55    pub auto_create_table: bool,
56    /// Maximum total memory for all concurrent write request bodies and messages (HTTP, gRPC, Flight).
57    /// Set to 0 to disable the limit. Default: "0" (unlimited)
58    pub max_in_flight_write_bytes: ReadableSize,
59    /// Policy when write bytes quota is exhausted.
60    /// Options: "wait" (default, 10s), "wait(<duration>)", "fail"
61    pub write_bytes_exhausted_policy: OnExhaustedPolicy,
62    pub http: HttpOptions,
63    pub grpc: GrpcOptions,
64    /// The internal gRPC options for the frontend service.
65    /// It serves the same services as the public one plus the internal handler.
66    /// CORS is always off on it.
67    pub internal_grpc: Option<GrpcOptions>,
68    pub mysql: MysqlOptions,
69    pub postgres: PostgresOptions,
70    pub opentsdb: OpentsdbOptions,
71    pub influxdb: InfluxdbOptions,
72    /// Ordinary-table batching with independent logical-table controls.
73    pub pending_rows_batcher: PendingRowsBatcherOptions,
74    pub prom_store: PromStoreOptions,
75    pub jaeger: JaegerOptions,
76    pub otlp: OtlpOptions,
77    pub meta_client: Option<MetaClientOptions>,
78    pub logging: LoggingOptions,
79    pub datanode: DatanodeClientOptions,
80    /// Retry policy for retryable distributed write failures.
81    pub write_retry: WriteRetryOptions,
82    /// The user provider for authentication.
83    ///
84    /// The inline `static_user_provider:cmd:` form carries plaintext
85    /// passwords, so the value is held as a [`SecretString`] and serialized
86    /// with those passwords redacted.
87    #[serde(serialize_with = "auth::serialize_user_provider_option")]
88    pub user_provider: Option<SecretString>,
89    pub tracing: TracingOptions,
90    pub query: QueryOptions,
91    pub slow_query: SlowQueryOptions,
92    pub memory: MemoryOptions,
93    /// The pipeline options.
94    pub pipeline: PipelineOptions,
95    /// The event recorder options.
96    pub event_recorder: EventRecorderOptions,
97    /// Environment variable keys to read and report in heartbeat messages.
98    pub heartbeat_env_vars: Vec<String>,
99}

137impl FrontendOptions {
138    /// Returns the address this frontend advertises to the rest of the cluster.
139    ///
140    /// It is used as the heartbeat peer address and as the identity of the
141    /// frontend in the process list. When a dedicated internal gRPC server is
142    /// configured, it is that server's address, because the internal server is
143    /// the one that serves the frontend management service and peers connect to
144    /// it. Otherwise the public gRPC address is used: the peer address only
145    /// promises to be reachable by other components.
146    pub fn peer_addr(&self) -> String {
147        let grpc = self.internal_grpc.as_ref().unwrap_or(&self.grpc);
148        servers::addrs::resolve_addr(&grpc.bind_addr, Some(&grpc.server_addr))
149    }
150
151    /// Creates the process manager that tracks this frontend's running queries.
152    ///
153    /// The manager identifies this frontend by [`Self::peer_addr`], the same
154    /// address other frontends discover from metasrv. It must not use the public
155    /// gRPC address: a frontend with a dedicated internal gRPC server would fail
156    /// to recognize itself among the discovered peers, list its own queries
157    /// twice, and send `KILL` requests to an address that does not serve the
158    /// frontend management service.
159    pub fn create_process_manager(&self, meta_client: Option<MetaClientRef>) -> ProcessManagerRef {
160        Arc::new(ProcessManager::new(self.peer_addr(), meta_client))
161    }
162}

176/// The [`Frontend`] struct is the main entry point for the frontend service
177/// which contains server handlers, frontend instance and some background tasks.
178pub struct Frontend {
179    pub instance: Arc<Instance>,
180    pub servers: ServerHandlers,
181    pub heartbeat_task: Option<HeartbeatTask>,
182}
183
184impl Frontend {
185    pub async fn start(&mut self) -> Result<()> {
186        if let Some(t) = &self.heartbeat_task
187            && let Err(error) = t.start().await
188        {
189            t.shutdown().await;
190            return Err(error);
191        }
192
193        if let Err(source) = self.servers.start_all().await {
194            if let Some(t) = &self.heartbeat_task {
195                t.shutdown().await;
196            }
197            return Err(source).context(error::StartServerSnafu);
198        }
199        Ok(())
200    }
201
202    pub async fn shutdown(&mut self) -> Result<()> {
203        if let Some(t) = &self.heartbeat_task {
204            t.shutdown().await;
205        }
206        self.servers
207            .shutdown_all()
208            .await
209            .context(error::ShutdownServerSnafu)
210    }
211
212    pub fn server_handlers(&self) -> &ServerHandlers {
213        &self.servers
214    }
215}

558    #[async_trait]
559    impl heartbeat_server::Heartbeat for SuspendableHeartbeatServer {
560        type HeartbeatStream = GrpcStream<HeartbeatResponse>;
561
562        async fn heartbeat(
563            &self,
564            request: Request<Streaming<HeartbeatRequest>>,
565        ) -> std::result::Result<Response<Self::HeartbeatStream>, Status> {
566            if self.fail_heartbeat {
567                return Err(Status::unavailable("mock initial heartbeat failure"));
568            }
569
570            let (tx, rx) = mpsc::channel(4);
571
572            common_runtime::spawn_global({
573                let mut requests = request.into_inner();
574                let suspend = self.suspend.clone();
575                async move {
576                    // Make the heartbeat interval short in unit tests to reduce the waiting time.
577                    // Only the handshake response needs to populate it (as metasrv does).
578                    let heartbeat_interval_ms = Duration::from_millis(200).as_millis() as u64;
579                    let mut is_handshake = true;
580                    while let Some(request) = requests.next().await {
581                        if let Err(e) = request {
582                            let _ = tx.send(Err(e)).await;
583                            return;
584                        }
585
586                        let mailbox_message =
587                            suspend.load(Ordering::Relaxed).then(|| MailboxMessage {
588                                payload: Some(Payload::Json(
589                                    serde_json::to_string(&Instruction::Suspend).unwrap(),
590                                )),
591                                ..Default::default()
592                            });
593                        let heartbeat_config =
594                            is_handshake.then_some(api::v1::meta::HeartbeatConfig {
595                                heartbeat_interval_ms,
596                                retry_interval_ms: heartbeat_interval_ms,
597                                gc_enabled: false,
598                            });
599                        is_handshake = false;
600                        let response = HeartbeatResponse {
601                            header: Some(ResponseHeader::success()),
602                            mailbox_message,
603                            heartbeat_config,
604                            ..Default::default()
605                        };
606
607                        let _ = tx.send(Ok(response)).await;
608                    }
609                }
610            });
611
612            Ok(Response::new(Box::pin(ReceiverStream::new(rx))))
613        }
614
615        async fn ask_leader(
616            &self,
617            _: Request<AskLeaderRequest>,
618        ) -> std::result::Result<Response<AskLeaderResponse>, Status> {
619            Ok(Response::new(AskLeaderResponse {
620                header: Some(ResponseHeader::success()),
621                leader: Some(Peer {
622                    addr: "localhost:0".to_string(),
623                    ..Default::default()
624                }),
625            }))
626        }
627    }

629    async fn create_meta_client(
630        options: &MetaClientOptions,
631        heartbeat_server: Arc<SuspendableHeartbeatServer>,
632    ) -> MetaClientRef {
633        let (client, server) = tokio::io::duplex(1024);
634
635        // create the heartbeat server:
636        common_runtime::spawn_global(async move {
637            let mut router = tonic::transport::Server::builder();
638            let router = router.add_service(
639                HeartbeatServer::from_arc(heartbeat_server)
640                    .accept_compressed(CompressionEncoding::Zstd)
641                    .send_compressed(CompressionEncoding::Zstd),
642            );
643            router
644                .serve_with_incoming(futures::stream::iter([Ok::<_, std::io::Error>(server)]))
645                .await
646        });
647
648        // Move client to an option so we can _move_ the inner value
649        // on the first attempt to connect. All other attempts will fail.
650        let mut client = Some(client);
651        let connector = tower::service_fn(move |_| {
652            let client = client.take();
653            async move {
654                if let Some(client) = client {
655                    Ok(hyper_util::rt::TokioIo::new(client))
656                } else {
657                    Err(std::io::Error::other("client already taken"))
658                }
659            }
660        });
661        let manager = ChannelManager::new();
662        manager
663            .reset_with_connector("localhost:0", connector)
664            .unwrap();
665
666        // create the heartbeat client:
667        let mut client = MetaClientBuilder::new(0, Role::Frontend)
668            .enable_heartbeat()
669            .heartbeat_channel_manager(manager)
670            .build();
671        client.start(&options.metasrv_addrs).await.unwrap();
672        Arc::new(client)
673    }

796    async fn verify_suspend_state_by_http(
797        frontend: &Frontend,
798        expected: std::result::Result<&str, (StatusCode, &str)>,
799    ) {
800        let addr = frontend.server_handlers().addr(HTTP_SERVER).unwrap();
801        let response = reqwest::get(format!("http://{}/v1/sql?sql=SELECT 1", addr))
802            .await
803            .unwrap();
804
805        let headers = response.headers();
806        let response = if let Some((code, error)) = from_header_to_err_code_msg(headers) {
807            Err((code, error))
808        } else {
809            Ok(response.text().await.unwrap())
810        };
811
812        match (response, expected) {
813            (Ok(response), Ok(expected)) => {
814                let response: GreptimedbV1Response = serde_json::from_str(&response).unwrap();
815                let response = serde_json::to_string(response.output()).unwrap();
816                assert_eq!(&response, expected);
817            }
818            (Err(actual), Err(expected)) => assert_eq!(actual, expected),
819            _ => unreachable!(),
820        }
821    }

849    async fn wait_for_suspend_state(frontend: &Frontend, expected: bool) {
850        let check = || frontend.instance.is_suspended() == expected;
851        if check() {
852            return;
853        }
854
855        tokio::time::timeout(Duration::from_secs(5), async move {
856            while !check() {
857                tokio::time::sleep(Duration::from_millis(20)).await;
858            }
859        })
860        .await
861        .unwrap();
862    }

865    async fn test_suspend_frontend() -> Result<()> {
866        common_telemetry::init_default_ut_logging();
867
868        let meta_client_options = MetaClientOptions {
869            metasrv_addrs: vec!["localhost:0".to_string()],
870            ..Default::default()
871        };
872        let options = FrontendOptions {
873            http: HttpOptions {
874                addr: "127.0.0.1:0".to_string(),
875                ..Default::default()
876            },
877            grpc: GrpcOptions {
878                bind_addr: "127.0.0.1:0".to_string(),
879                flight_compression: FlightCompression::None,
880                ..Default::default()
881            },
882            mysql: MysqlOptions {
883                enable: false,
884                ..Default::default()
885            },
886            postgres: PostgresOptions {
887                enable: false,
888                ..Default::default()
889            },
890            meta_client: Some(meta_client_options.clone()),
891            ..Default::default()
892        };
893
894        let server = Arc::new(SuspendableHeartbeatServer {
895            suspend: Arc::new(AtomicBool::new(false)),
896            fail_heartbeat: false,
897        });
898        let meta_client = create_meta_client(&meta_client_options, server.clone()).await;
899        let frontend = create_frontend(&options, meta_client).await?;
900
901        // initial state: not suspend:
902        assert!(!frontend.instance.is_suspended());
903        verify_suspend_state_by_http(&frontend, Ok(r#"[{"records":{"schema":{"column_schemas":[{"name":"Int64(1)","data_type":"Int64"}]},"rows":[[1]],"total_rows":1}}]"#)).await;
904        verify_suspend_state_by_grpc(
905            &frontend,
906            Ok(r#"
907+----------+
908| Int64(1) |
909+----------+
910| 1        |
911+----------+"#),
912        )
913        .await;
914
915        // make heartbeat server returned "suspend" instruction,
916        server.suspend.store(true, Ordering::Relaxed);
917        wait_for_suspend_state(&frontend, true).await;
918        // ... then the frontend is suspended:
919        assert!(frontend.instance.is_suspended());
920        verify_suspend_state_by_http(
921            &frontend,
922            Err((
923                StatusCode::Suspended,
924                "error: Service suspended, execution_time_ms: 0",
925            )),
926        )
927        .await;
928        verify_suspend_state_by_grpc(&frontend, Err((StatusCode::Suspended, "Service suspended")))
929            .await;
930
931        // make heartbeat server NOT returned "suspend" instruction,
932        server.suspend.store(false, Ordering::Relaxed);
933        wait_for_suspend_state(&frontend, false).await;
934        // ... then frontend's suspend state is cleared:
935        assert!(!frontend.instance.is_suspended());
936        verify_suspend_state_by_http(&frontend, Ok(r#"[{"records":{"schema":{"column_schemas":[{"name":"Int64(1)","data_type":"Int64"}]},"rows":[[1]],"total_rows":1}}]"#)).await;
937        verify_suspend_state_by_grpc(
938            &frontend,
939            Ok(r#"
940+----------+
941| Int64(1) |
942+----------+
943| 1        |
944+----------+"#),
945        )
946        .await;
947        Ok(())
948    }
