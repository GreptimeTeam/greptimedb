43#[async_trait::async_trait]
44pub trait UserProvider: Send + Sync {
45    fn name(&self) -> &str;
46
47    /// Checks whether a user is valid and allowed to access the database.
48    async fn authenticate(&self, id: Identity<'_>, password: Password<'_>) -> Result<UserInfoRef>;
49
50    /// Checks whether a connection request
51    /// from a certain user to a certain catalog/schema is legal.
52    /// This method should be called after [authenticate()](UserProvider::authenticate()).
53    async fn authorize(&self, catalog: &str, schema: &str, user_info: &UserInfoRef) -> Result<()>;
54
55    /// Combination of [authenticate()](UserProvider::authenticate()) and [authorize()](UserProvider::authorize()).
56    /// In most cases it's preferred for both convenience and performance.
57    async fn auth(
58        &self,
59        id: Identity<'_>,
60        password: Password<'_>,
61        catalog: &str,
62        schema: &str,
63    ) -> Result<UserInfoRef> {
64        let user_info = self.authenticate(id, password).await?;
65        self.authorize(catalog, schema, &user_info).await?;
66        Ok(user_info)
67    }
68
69    /// Authenticates and authorizes an opaque bearer token (e.g. a JWT or an
70    /// OAuth2 access token).
71    ///
72    /// Unlike [auth()](Self::auth), the caller has no `Identity`/`Password` —
73    /// the provider validates the token and *derives* the identity from it.
74    /// The token is opaque to the server, so JWT/JWKS/OIDC validation policy
75    /// stays pluggable and out of core.
76    ///
77    /// The default rejects token auth with
78    /// [`Error::UnsupportedAuthMethod`], so password-only providers keep
79    /// today's behavior. Providers that support token auth override this to
80    /// validate the token, resolve it to a user, and
81    /// [`authorize`](Self::authorize) the connection.
82    async fn auth_bearer_token(
83        &self,
84        _token: &str,
85        _catalog: &str,
86        _schema: &str,
87    ) -> Result<UserInfoRef> {
88        UnsupportedAuthMethodSnafu {
89            method: "bearer token",
90        }
91        .fail()
92    }
93
94    async fn postgres_auth_info(&self, _id: Identity<'_>) -> Result<PgAuthInfo> {
95        Ok(PgAuthInfo::Cleartext)
96    }
97
98    /// Returns whether this user provider implementation is backed by an external system.
99    fn external(&self) -> bool {
100        false
101    }
102}

112#[derive(Clone)]
113pub(crate) enum PasswordVerifier {
114    PlainText {
115        password: String,
116        /// SCRAM verifier derived once at load time. Precomputing it keeps the
117        /// Postgres SCRAM `server-first-message` (stable salt, fixed iterations)
118        /// and per-connection cost indistinguishable from stored-hash and
119        /// unknown users, instead of running PBKDF2 with a fresh salt on every
120        /// connection.
121        scram: PgScramSha256Verifier,
122    },
123    Pbkdf2Sha256 {
124        iterations: u32,
125        salt: Vec<u8>,
126        hash: Vec<u8>,
127    },
128    MysqlNativePassword {
129        hash_stage_2: Vec<u8>,
130    },
131    PgScramSha256(PgScramSha256Verifier),
132}
133
134impl PartialEq for PasswordVerifier {
135    fn eq(&self, other: &Self) -> bool {
136        // The precomputed SCRAM verifier is a cache derived from the password, so
137        // two plaintext verifiers are equal iff their passwords match.
138        match (self, other) {
139            (
140                PasswordVerifier::PlainText { password: a, .. },
141                PasswordVerifier::PlainText { password: b, .. },
142            ) => a == b,
143            (
144                PasswordVerifier::Pbkdf2Sha256 {
145                    iterations: i1,
146                    salt: s1,
147                    hash: h1,
148                },
149                PasswordVerifier::Pbkdf2Sha256 {
150                    iterations: i2,
151                    salt: s2,
152                    hash: h2,
153                },
154            ) => i1 == i2 && s1 == s2 && h1 == h2,
155            (
156                PasswordVerifier::MysqlNativePassword { hash_stage_2: a },
157                PasswordVerifier::MysqlNativePassword { hash_stage_2: b },
158            ) => a == b,
159            (PasswordVerifier::PgScramSha256(a), PasswordVerifier::PgScramSha256(b)) => a == b,
160            _ => false,
161        }
162    }
163}

191impl PasswordVerifier {
192    /// Builds a plaintext verifier, precomputing a stable SCRAM verifier so the
193    /// Postgres SCRAM handshake for this user carries a stable salt and runs no
194    /// per-connection PBKDF2. Uses [`DEFAULT_PBKDF2_SHA256_ITERATIONS`] to match
195    /// the mock verifier handed to unknown users.
196    fn plain_text(password: String) -> Option<Self> {
197        let salt = rand::random::<[u8; DEFAULT_PBKDF2_SHA256_SALT_LEN]>();
198        let scram = PgScramSha256Verifier::from_password(
199            password.as_bytes(),
200            &salt,
201            crate::DEFAULT_PBKDF2_SHA256_ITERATIONS,
202        )
203        .ok()?;
204        Some(Self::PlainText { password, scram })
205    }
206
207    fn parse(input: &str) -> Option<Self> {
208        if let Some(password) = input.strip_prefix("plain:") {
209            return Self::plain_text(password.to_string());
210        }
211
212        if let Some(verifier) = input.strip_prefix("pbkdf2_sha256:") {
213            let mut parts = verifier.split(':');
214            let iterations = parts.next()?.parse::<u32>().ok()?;
215            let salt = hex::decode(parts.next()?).ok()?;
216            let hash = hex::decode(parts.next()?).ok()?;
217            if parts.next().is_some()
218                || iterations == 0
219                || iterations > MAX_PBKDF2_SHA256_ITERATIONS
220                || salt.is_empty()
221                || salt.len() > MAX_PBKDF2_SHA256_SALT_LEN
222                || hash.len() != PBKDF2_SHA256_HASH_LEN
223            {
224                return None;
225            }
226
227            return Some(Self::Pbkdf2Sha256 {
228                iterations,
229                salt,
230                hash,
231            });
232        }
233
234        if let Some(verifier) = input.strip_prefix("mysql_native_password:") {
235            let hash_stage_2 = hex::decode(verifier).ok()?;
236            if hash_stage_2.len() != 20 {
237                return None;
238            }
239
240            return Some(Self::MysqlNativePassword { hash_stage_2 });
241        }
242
243        if let Some(verifier) = input.strip_prefix("pg_scram_sha256:") {
244            let mut parts = verifier.split(':');
245            let iterations = parts.next()?.parse::<u32>().ok()?;
246            let salt = hex::decode(parts.next()?).ok()?;
247            let stored_key = hex::decode(parts.next()?).ok()?;
248            let server_key = hex::decode(parts.next()?).ok()?;
249            if parts.next().is_some()
250                || stored_key.len() != PG_SCRAM_SHA256_KEY_LEN
251                || server_key.len() != PG_SCRAM_SHA256_KEY_LEN
252            {
253                return None;
254            }
255
256            return PgScramSha256Verifier::new(iterations, salt, stored_key, server_key)
257                .ok()
258                .map(Self::PgScramSha256);
259        }
260
261        Self::plain_text(input.to_string())
262    }
263
264    fn supports_pg_scram_sha256(&self) -> bool {
265        matches!(
266            self,
267            PasswordVerifier::PlainText { .. } | PasswordVerifier::PgScramSha256(_)
268        )
269    }
270
271    fn to_pg_scram_sha256_verifier(&self) -> Option<PgScramSha256Verifier> {
272        match self {
273            PasswordVerifier::PlainText { scram, .. } => Some(scram.clone()),
274            // Legacy PBKDF2 verifiers were derived without SASLprep, and the
275            // original password is unavailable to normalize them safely.
276            PasswordVerifier::Pbkdf2Sha256 { .. } => None,
277            PasswordVerifier::PgScramSha256(verifier) => Some(verifier.clone()),
278            PasswordVerifier::MysqlNativePassword { .. } => None,
279        }
280    }
281
282    fn verify_plain_text(&self, password: &str) -> bool {
283        match self {
284            PasswordVerifier::PlainText {
285                password: expected, ..
286            } => expected.as_bytes().ct_eq(password.as_bytes()).into(),
287            PasswordVerifier::Pbkdf2Sha256 {
288                iterations,
289                salt,
290                hash,
291            } => {
292                if hash.len() != PBKDF2_SHA256_HASH_LEN {
293                    return false;
294                }
295                let mut actual = [0u8; PBKDF2_SHA256_HASH_LEN];
296                pbkdf2_hmac::<Sha256>(password.as_bytes(), salt, *iterations, &mut actual);
297                hash.as_slice().ct_eq(&actual[..]).into()
298            }
299            PasswordVerifier::MysqlNativePassword { .. } => false,
300            PasswordVerifier::PgScramSha256(verifier) => verifier
301                .verify_plain_password(password.as_bytes())
302                .unwrap_or(false),
303        }
304    }
305
306    fn verify_mysql_native_password(
307        &self,
308        auth_data: &[u8],
309        salt: &[u8],
310        username: &str,
311    ) -> Result<()> {
312        match self {
313            PasswordVerifier::PlainText { password, .. } => {
314                auth_mysql(auth_data, salt, username, password.as_bytes())
315            }
316            PasswordVerifier::MysqlNativePassword { hash_stage_2 } => {
317                auth_mysql_with_hash_stage_2(auth_data, salt, username, hash_stage_2)
318            }
319            PasswordVerifier::Pbkdf2Sha256 { .. } => UnsupportedPasswordTypeSnafu {
320                password_type: "mysql_native_password_with_pbkdf2_sha256_verifier",
321            }
322            .fail(),
323            PasswordVerifier::PgScramSha256(_) => UnsupportedPasswordTypeSnafu {
324                password_type: "mysql_native_password_with_pg_scram_sha256_verifier",
325            }
326            .fail(),
327        }
328    }
329}

335fn load_credential_from_file(filepath: &str) -> Result<UserInfoMap> {
336    // check valid path
337    let path = Path::new(filepath);
338    if !path.exists() {
339        return InvalidConfigSnafu {
340            value: filepath.to_string(),
341            msg: "UserProvider file must exist",
342        }
343        .fail();
344    }
345
346    ensure!(
347        path.is_file(),
348        InvalidConfigSnafu {
349            value: filepath,
350            msg: "UserProvider file must be a file",
351        }
352    );
353    let file = File::open(path).context(IoSnafu)?;
354    let credential = io::BufReader::new(file)
355        .lines()
356        .enumerate()
357        .map_while(|(idx, line)| match line {
358            Ok(line) => Some((idx, line)),
359            Err(err) => {
360                // A read error (I/O failure or invalid UTF-8) ends the iterator,
361                // so every remaining credential is dropped. Warn instead of
362                // vanishing silently, matching the malformed-line handling below.
363                warn!(
364                    "Failed to read line {} of user provider file {}: {}; \
365                     all remaining credentials are ignored",
366                    idx + 1,
367                    filepath,
368                    err
369                );
370                None
371            }
372        })
373        .filter_map(|(idx, line)| {
374            // The line format is:
375            // - `username=password` - Basic user with default permissions
376            // - `username:permission_mode=password` - User with specific permission mode
377            // - Lines starting with '#' are treated as comments and ignored
378            // - Empty lines are ignored
379            let line = line.trim();
380            if line.is_empty() || line.starts_with('#') {
381                return None;
382            }
383
384            let parsed = parse_credential_line(line);
385            if parsed.is_none() {
386                // Don't log the line: it carries the password/verifier. A common
387                // cause is a plaintext password containing `=`, which splits the
388                // line into more than two parts.
389                warn!(
390                    "Ignoring malformed credential at line {} of user provider file {}: \
391                     expected `username[:permission]=verifier` with exactly one `=` \
392                     (passwords containing `=` are not supported)",
393                    idx + 1,
394                    filepath
395                );
396            }
397            parsed
398        })
399        .collect::<HashMap<String, _>>();
400
401    ensure!(
402        !credential.is_empty(),
403        InvalidConfigSnafu {
404            value: filepath,
405            msg: "UserProvider's file must contains at least one valid credential",
406        }
407    );
408
409    warn_if_pg_scram_disabled(&credential);
410
411    Ok(credential)
412}

414/// Returns the users whose verifier cannot back a Postgres SCRAM handshake.
415///
416/// Only [`PasswordVerifier::PlainText`] and [`PasswordVerifier::PgScramSha256`]
417/// support SCRAM. A `mysql_native_password` verifier is a double-SHA1 digest
418/// unrelated to PBKDF2, and a `pbkdf2_sha256` verifier was derived without
419/// SASLprep and cannot be safely reused as a SCRAM secret without the original
420/// password. Either kind forces the whole Postgres endpoint to fall back to
421/// cleartext (see [`postgres_auth_info_with_credential`]).
422fn pg_scram_unsupported_users(users: &UserInfoMap) -> Vec<&str> {
423    users
424        .iter()
425        .filter(|(_, (verifier, _))| !verifier.supports_pg_scram_sha256())
426        .map(|(username, _)| username.as_str())
427        .collect()
428}

430/// Warns once per credential load when the set disables Postgres SCRAM, so
431/// operators don't unknowingly serve cleartext passwords over Postgres while
432/// believing SCRAM is in effect.
433pub(crate) fn warn_if_pg_scram_disabled(users: &UserInfoMap) {
434    let unsupported = pg_scram_unsupported_users(users);
435    if !unsupported.is_empty() {
436        warn!(
437            "Postgres SCRAM authentication is disabled: {} of {} user(s) use a \
438             non-SCRAM password verifier {:?}, so all Postgres password \
439             authentication falls back to cleartext. Ensure TLS is enabled; if you \
440             rely on Postgres SCRAM, generate every user's verifier with the \
441             pg_scram_sha256 format.",
442            unsupported.len(),
443            users.len(),
444            unsupported
445        );
446    }
447}

449/// Parse a line of credential in the format of `username=password` or `username:permission_mode=password`.
450///
451/// The password part accepts legacy plain text and explicit verifier formats:
452/// - `plain:<password>`
453/// - `pbkdf2_sha256:<iterations>:<hex-encoded-salt>:<hex-encoded-hash>`
454/// - `mysql_native_password:<hex-encoded-sha1-sha1-password>`
455/// - `pg_scram_sha256:<iterations>:<hex-encoded-salt>:<hex-encoded-stored-key>:<hex-encoded-server-key>`
456pub(crate) fn parse_credential_line(
457    line: &str,
458) -> Option<(String, (PasswordVerifier, PermissionMode))> {
459    let parts = line.split('=').collect::<Vec<&str>>();
460    if parts.len() != 2 {
461        return None;
462    }
463
464    let (username_part, password) = (parts[0], parts[1]);
465    let (username, permission_mode) = if let Some((user, perm)) = username_part.split_once(':') {
466        (user, PermissionMode::from_str(perm)?)
467    } else {
468        (username_part, PermissionMode::default())
469    };
470
471    let verifier = PasswordVerifier::parse(password)?;
472
473    Some((username.to_string(), (verifier, permission_mode)))
474}

476pub(crate) fn postgres_auth_info_with_credential(
477    users: &UserInfoMap,
478    input_id: Identity<'_>,
479) -> Result<PgAuthInfo> {
480    match input_id {
481        Identity::UserId(username, _) => {
482            ensure!(
483                !username.is_empty(),
484                IllegalParamSnafu {
485                    msg: "blank username"
486                }
487            );
488
489            if !users
490                .values()
491                .all(|(verifier, _)| verifier.supports_pg_scram_sha256())
492            {
493                // PostgreSQL chooses one auth method during startup. Selecting it
494                // per username would expose user or verifier existence.
495                return Ok(PgAuthInfo::Cleartext);
496            }
497
498            if let Some((verifier, permission_mode)) = users.get(username) {
499                if let Some(verifier) = verifier.to_pg_scram_sha256_verifier() {
500                    return Ok(PgAuthInfo::ScramSha256 {
501                        verifier,
502                        user_info: Some(DefaultUserInfo::with_name_and_permission(
503                            username,
504                            *permission_mode,
505                        )),
506                    });
507                }
508
509                return Ok(PgAuthInfo::Cleartext);
510            }
511
512            // Unknown user: hand back a deterministic mock verifier so the SCRAM
513            // handshake is indistinguishable from a real user, without running
514            // PBKDF2 or leaking existence through an unstable salt.
515            Ok(PgAuthInfo::ScramSha256 {
516                verifier: PgScramSha256Verifier::mock_for_unknown_user(username.as_bytes()),
517                user_info: None,
518            })
519        }
520    }
521}

523fn authenticate_with_credential(
524    users: &UserInfoMap,
525    input_id: Identity<'_>,
526    input_pwd: Password<'_>,
527) -> Result<UserInfoRef> {
528    match input_id {
529        Identity::UserId(username, _) => {
530            ensure!(
531                !username.is_empty(),
532                IllegalParamSnafu {
533                    msg: "blank username"
534                }
535            );
536            let (verifier, permission_mode) = users.get(username).context(UserNotFoundSnafu {
537                username: username.to_string(),
538            })?;
539
540            match input_pwd {
541                Password::PlainText(pwd) => {
542                    ensure!(
543                        !pwd.expose_secret().is_empty(),
544                        IllegalParamSnafu {
545                            msg: "blank password"
546                        }
547                    );
548                    if verifier.verify_plain_text(pwd.expose_secret()) {
549                        Ok(DefaultUserInfo::with_name_and_permission(
550                            username,
551                            *permission_mode,
552                        ))
553                    } else {
554                        UserPasswordMismatchSnafu {
555                            username: username.to_string(),
556                        }
557                        .fail()
558                    }
559                }
560                Password::MysqlNativePassword(auth_data, salt) => verifier
561                    .verify_mysql_native_password(auth_data, salt, username)
562                    .map(|_| DefaultUserInfo::with_name_and_permission(username, *permission_mode)),
563                Password::PgMD5(_, _) => UnsupportedPasswordTypeSnafu {
564                    password_type: "pg_md5",
565                }
566                .fail(),
567            }
568        }
569    }

583    fn sha1_one(data: &[u8]) -> Vec<u8> {
584        let mut hasher = Sha1::new();
585        hasher.update(data);
586        hasher.finalize().to_vec()
587    }
588
589    fn mysql_native_password_auth_data(password: &str, salt: &[u8]) -> Vec<u8> {
590        let hash_stage_1 = sha1_one(password.as_bytes());
591        let hash_stage_2 = mysql_native_password_hash(password.as_bytes());
592        let mut hasher = Sha1::new();
593        hasher.update(salt);
594        hasher.update(hash_stage_2);
595        let scramble = hasher.finalize();
596
597        hash_stage_1
598            .iter()
599            .zip(scramble.iter())
600            .map(|(lhs, rhs)| lhs ^ rhs)
601            .collect()
602    }

662
663        // Username with spaces should be preserved
664        let result = parse_credential_line("user name:WriteOnly=password");
665        assert_eq!(
666            result,
667            Some((
668                "user name".to_string(),
669                (plain("password"), PermissionMode::WriteOnly)
670            ))
671        );
672
673        let result = parse_credential_line("user=plain:password");
674        assert_eq!(
675            result,
676            Some((
677                "user".to_string(),
678                (plain("password"), PermissionMode::default())
679            ))
680        );
681
682        let iterations = 4096;
683        let salt = b"salt";
684        let mut hash = [0u8; 32];
685        pbkdf2_hmac::<Sha256>("password".as_bytes(), salt, iterations, &mut hash);
686        let result = parse_credential_line(&format!(
687            "user=pbkdf2_sha256:{iterations}:{}:{}",
688            hex::encode(salt),
689            hex::encode(hash)
690        ));
691        assert_eq!(
692            result,
693            Some((
694                "user".to_string(),
695                (
696                    PasswordVerifier::Pbkdf2Sha256 {
697                        iterations,
698                        salt: salt.to_vec(),
699                        hash: hash.to_vec(),
700                    },
701                    PermissionMode::default()
702                )
703            ))
704        );
705
706        let result = parse_credential_line("user=pbkdf2_sha256:4096:not-hex:abcd");
707        assert_eq!(result, None);
708
709        // A well-formed but truncated hash must be rejected: a short hash would let
710        // many wrong passwords pass by matching only a few derived bytes.
711        let result = parse_credential_line(&format!(
712            "user=pbkdf2_sha256:4096:{}:abcd",
713            hex::encode(salt)
714        ));
715        assert_eq!(result, None);
716
717        let result = parse_credential_line(&format!(
718            "user=pbkdf2_sha256:{}:{}:{}",
719            MAX_PBKDF2_SHA256_ITERATIONS + 1,
720            hex::encode(salt),
721            hex::encode(hash)
722        ));
723        assert_eq!(result, None);
724
725        let hash_stage_2 = mysql_native_password_hash("password".as_bytes());
726        let result = parse_credential_line(&format!(
727            "user=mysql_native_password:{}",
728            hex::encode(&hash_stage_2)
729        ));
730        assert_eq!(
731            result,
732            Some((
733                "user".to_string(),
734                (
735                    PasswordVerifier::MysqlNativePassword { hash_stage_2 },
736                    PermissionMode::default()
737                )
738            ))
739        );
740
741        let result = parse_credential_line("user=mysql_native_password:abcd");
742        assert_eq!(result, None);
743
744        let verifier =
745            format_pg_scram_sha256_password_verifier(b"password", b"salt", 4096).unwrap();
746        let result = parse_credential_line(&format!("user={verifier}"));
747        assert!(matches!(
748            result,
749            Some((
750                _,
751                (
752                    PasswordVerifier::PgScramSha256(PgScramSha256Verifier { .. }),
753                    PermissionMode::ReadWrite
754                )
755            ))
756        ));
757
758        let result = parse_credential_line("user=pg_scram_sha256:4096:73616c74:abcd:abcd");
759        assert_eq!(result, None);
760
761        // Invalid format - no equals sign
762        let result = parse_credential_line("invalid_line");
763        assert_eq!(result, None);
764
765        // Invalid format - multiple equals signs
766        let result = parse_credential_line("user=pass=word");
767        assert_eq!(result, None);
768
769        for line in [
770            "user:readonyl=password",
771            "user:=password",
772            "user:arbitrary=password",
773        ] {
774            assert_eq!(parse_credential_line(line), None);
775        }
776
777        // Empty password
778        let result = parse_credential_line("user=");
779        assert_eq!(
780            result,
781            Some(("user".to_string(), (plain(""), PermissionMode::default())))
782        );
783
784        // Empty username
785        let result = parse_credential_line("=password");
786        assert_eq!(
787            result,
788            Some((
789                "".to_string(),
790                (plain("password"), PermissionMode::default())
791            ))
792        );
793    }

892    #[test]
893    fn test_plain_text_scram_verifier_is_stable() {
894        let verifier = PasswordVerifier::plain_text("password".to_string()).unwrap();
895        let first = verifier.to_pg_scram_sha256_verifier().unwrap();
896        let second = verifier.to_pg_scram_sha256_verifier().unwrap();
897
898        // A plaintext-backed user must present a stable salt and fixed iterations
899        // across connections without re-running PBKDF2, otherwise the SCRAM
900        // server-first message (and its timing) leaks that the user exists.
901        assert_eq!(first.salt(), second.salt());
902        assert_eq!(first.salt().len(), DEFAULT_PBKDF2_SHA256_SALT_LEN);
903        assert_eq!(first.iterations(), crate::DEFAULT_PBKDF2_SHA256_ITERATIONS);
904
905        // The derived verifier must still accept the real password.
906        assert!(first.verify_plain_password(b"password").unwrap());
907        assert!(!first.verify_plain_password(b"wrong").unwrap());
908    }
