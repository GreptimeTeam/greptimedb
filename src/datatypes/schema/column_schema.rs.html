89impl fmt::Debug for ColumnSchema {
90    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
91        write!(
92            f,
93            "{} {} {}",
94            self.name,
95            self.data_type,
96            if self.is_nullable { "null" } else { "not null" },
97        )?;
98
99        if self.is_time_index {
100            write!(f, " time_index")?;
101        }
102
103        // Add default constraint if present
104        if let Some(default_constraint) = &self.default_constraint {
105            write!(f, " default={:?}", default_constraint)?;
106        }
107
108        // Add metadata if present
109        if !self.metadata.is_empty() {
110            write!(f, " metadata={:?}", self.metadata)?;
111        }
112
113        Ok(())
114    }
115}

305    /// Creates a vector with default value for this column.
306    ///
307    /// If the column is `NOT NULL` but doesn't has `DEFAULT` value supplied, returns `Ok(None)`.
308    pub fn create_default_vector(&self, num_rows: usize) -> Result<Option<VectorRef>> {
309        match &self.default_constraint {
310            Some(c) => c
311                .create_default_vector(&self.data_type, self.is_nullable, num_rows)
312                .map(Some),
313            None => {
314                if self.is_nullable {
315                    // No default constraint, use null as default value.
316                    // TODO(yingwen): Use NullVector once it supports setting logical type.
317                    ColumnDefaultConstraint::null_value()
318                        .create_default_vector(&self.data_type, self.is_nullable, num_rows)
319                        .map(Some)
320                } else {
321                    Ok(None)
322                }
323            }
324        }
325    }

327    /// Creates a vector for padding.
328    ///
329    /// This method always returns a vector since it uses [DataType::default_value]
330    /// to fill the vector. Callers should only use the created vector for padding
331    /// and never read its content.
332    pub fn create_default_vector_for_padding(&self, num_rows: usize) -> VectorRef {
333        let padding_value = if self.is_nullable {
334            Value::Null
335        } else {
336            // If the column is not null, use the data type's default value as it is
337            // more efficient to acquire.
338            self.data_type.default_value()
339        };
340        let value_ref = padding_value.as_value_ref();
341        let mut mutable_vector = self.data_type.create_mutable_vector(num_rows);
342        for _ in 0..num_rows {
343            mutable_vector.push_value_ref(&value_ref);
344        }
345        mutable_vector.to_vector()
346    }

456    /// Sets the SST compression options. Nested types are rejected: their values are
457    /// stored in child columns that the options would not reach.

460        self.metadata.insert(

467    fn validate_compression_options(&self, options: &CompressionOptions) -> Result<()> {
468        options.validate()?;
469        ensure!(
470            !self.data_type.as_arrow_type().is_nested(),
471            error::InvalidCompressionOptionSnafu {
472                msg: format!(
473                    "compression options are not supported on column {} of type {}",
474                    self.name, self.data_type
475                ),
476            }
477        );
478        if options.encoding == Some(ColumnEncoding::DeltaBinaryPacked) {
479            ensure!(
480                self.data_type.is_timestamp() || self.data_type.as_arrow_type().is_integer(),
481                error::InvalidCompressionOptionSnafu {
482                    msg: format!(
483                        "encoding delta_binary_packed requires an integer or timestamp column, but column {} is {}",
484                        self.name, self.data_type
485                    ),
486                }
487            );
488        }
489        Ok(())
490    }

492    pub fn extension_type<E>(&self) -> Result<Option<E>>
493    where
494        E: ExtensionType,
495    {
496        let extension_type_name = self.metadata.get(EXTENSION_TYPE_NAME_KEY);
497
498        if extension_type_name.map(|s| s.as_str()) == Some(E::NAME) {
499            let extension_metadata = self.metadata.get(EXTENSION_TYPE_METADATA_KEY);
500            let extension_metadata =
501                E::deserialize_metadata(extension_metadata.map(|s| s.as_str()))
502                    .context(ArrowMetadataSnafu)?;
503
504            let extension = E::try_new(&self.data_type.as_arrow_type(), extension_metadata)
505                .context(ArrowMetadataSnafu)?;
506            Ok(Some(extension))
507        } else {
508            Ok(None)
509        }
510    }

560    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
561        match self {
562            ColumnExtType::Json => write!(f, "Json"),
563            ColumnExtType::Vector(dim) => write!(f, "Vector({})", dim),

568impl FromStr for ColumnExtType {
569    type Err = String;
570
571    fn from_str(s: &str) -> std::result::Result<Self, Self::Err> {
572        match s {
573            "Json" => Ok(ColumnExtType::Json),
574            _ if s.starts_with("Vector(") && s.ends_with(')') => s[7..s.len() - 1]
575                .parse::<u32>()
576                .map(ColumnExtType::Vector)
577                .map_err(|_| "Invalid dimension for Vector".to_string()),
578            _ => Err("Unknown variant".to_string()),
579        }
580    }
581}

583impl TryFrom<&Field> for ColumnSchema {
584    type Error = Error;
585
586    fn try_from(field: &Field) -> Result<ColumnSchema> {
587        let mut data_type = ConcreteDataType::try_from(field.data_type())?;
588        // Override the data type if it is specified in the metadata.
589        if let Some(s) = field.metadata().get(TYPE_KEY) {
590            let extype = ColumnExtType::from_str(s)
591                .map_err(|_| ParseExtendedTypeSnafu { value: s }.build())?;
592            match extype {
593                ColumnExtType::Json => {
594                    data_type = ConcreteDataType::json_datatype();
595                }
596                ColumnExtType::Vector(dim) => {
597                    data_type = ConcreteDataType::vector_datatype(dim);
598                }
599            }
600        }
601        let mut metadata = field.metadata().clone();
602        let default_constraint = match metadata.remove(DEFAULT_CONSTRAINT_KEY) {
603            Some(json) => {
604                Some(serde_json::from_str(&json).context(error::DeserializeSnafu { json })?)
605            }
606            None => None,
607        };
608        let mut is_time_index = metadata.contains_key(TIME_INDEX_KEY);
609        if is_time_index && !data_type.is_timestamp() {
610            // If the column is time index but the data type is not timestamp, it is invalid.
611            // We set the time index to false and remove the metadata.
612            // This is possible if we cast the time index column to another type. DataFusion will
613            // keep the metadata:
614            // https://github.com/apache/datafusion/pull/12951
615            is_time_index = false;
616            metadata.remove(TIME_INDEX_KEY);
617            common_telemetry::debug!(
618                "Column {} is not timestamp ({:?}) but has time index metadata",
619                data_type,
620                field.name(),
621            );
622        }
623
624        Ok(ColumnSchema {
625            name: field.name().clone(),
626            data_type,
627            is_nullable: field.is_nullable(),
628            is_time_index,
629            default_constraint,
630            metadata,
631        })
632    }
633}

635impl TryFrom<&ColumnSchema> for Field {
636    type Error = Error;
637
638    fn try_from(column_schema: &ColumnSchema) -> Result<Field> {
639        let mut metadata = column_schema.metadata.clone();
640        if let Some(value) = &column_schema.default_constraint {
641            // Adds an additional metadata to store the default constraint.
642            let old = metadata.insert(
643                DEFAULT_CONSTRAINT_KEY.to_string(),
644                serde_json::to_string(&value).context(error::SerializeSnafu)?,
645            );
646
647            ensure!(
648                old.is_none(),
649                error::DuplicateMetaSnafu {
650                    key: DEFAULT_CONSTRAINT_KEY,
651                }
652            );
653        }
654
655        Ok(Field::new(
656            &column_schema.name,
657            column_schema.data_type.as_arrow_type(),
658            column_schema.is_nullable(),
659        )
660        .with_metadata(metadata))
661    }
662}
