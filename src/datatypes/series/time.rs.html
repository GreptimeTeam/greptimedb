28/// Value description of a non-empty, fixed-step evaluation grid.
29///
30/// The serde form is the wire form a series frame writes under `greptime.series.grid`: field names
31/// in declaration order, no optional members. Two nodes compare that text byte for byte, so the
32/// only place that spells it is a series frame's own grid metadata entry.
33#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
34#[serde(deny_unknown_fields)]
35pub struct GridSpec {
36    /// First evaluation timestamp in milliseconds.
37    pub start_ms: i64,
38    /// Positive time step in milliseconds.
39    pub step_ms: i64,
40    /// Number of evaluation points.
41    pub len: u64,
42}
43
44impl GridSpec {
45    /// Creates a validated grid specification.
46    pub fn try_new(start_ms: i64, step_ms: i64, len: u64) -> Result<Self> {
47        if step_ms <= 0 || len == 0 {
48            return invalid("grid step must be positive and grid length nonzero");
49        }
50        let Some(last) = ((len - 1) as i128)
51            .checked_mul(step_ms as i128)
52            .and_then(|offset| (start_ms as i128).checked_add(offset))
53        else {
54            return invalid("grid timestamp overflows i128");
55        };
56        let Ok(platform_len) = usize::try_from(len) else {
57            return invalid("grid timestamp or length overflows supported range");
58        };
59        if i64::try_from(last).is_err()
60            || platform_len
61                .checked_mul(std::mem::size_of::<i64>())
62                .is_none()
63        {
64            return invalid("grid timestamp or length overflows supported range");
65        }
66        Ok(Self {
67            start_ms,
68            step_ms,
69            len,
70        })
71    }
72
73    /// Creates a grid for an inclusive query range, retaining the last point not after end.
74    pub fn try_from_range(start_ms: i64, end_ms: i64, step_ms: i64) -> Result<Self> {
75        if step_ms <= 0 || end_ms < start_ms {
76            return invalid("invalid inclusive grid range or step");
77        }
78        let len = ((end_ms as i128 - start_ms as i128) / step_ms as i128) + 1;
79        Self::try_new(
80            start_ms,
81            step_ms,
82            u64::try_from(len).map_err(|_| invalid_series("grid length overflow"))?,
83        )
84    }
85}

87/// A lazily materialized, shareable timestamp grid.
88#[derive(Debug)]
89pub struct EvalGrid {
90    spec: GridSpec,
91    timestamps: OnceCell<ScalarBuffer<i64>>,
92}
93impl serde::Serialize for EvalGrid {
94    fn serialize<S: serde::Serializer>(
95        &self,
96        serializer: S,
97    ) -> std::result::Result<S::Ok, S::Error> {
98        self.spec.serialize(serializer)
99    }
100}

102    fn deserialize<D: serde::Deserializer<'de>>(
103        deserializer: D,
104    ) -> std::result::Result<Self, D::Error> {
105        let spec = GridSpec::deserialize(deserializer)?;
106        Self::try_deferred(spec).map_err(serde::de::Error::custom)
107    }
108}

111    fn eq(&self, other: &Self) -> bool {
112        self.spec == other.spec
113    }

115impl Eq for EvalGrid {}
116impl std::hash::Hash for EvalGrid {
117    fn hash<H: std::hash::Hasher>(&self, state: &mut H) {
118        self.spec.start_ms.hash(state);
119        self.spec.step_ms.hash(state);
120        self.spec.len.hash(state)
121    }

123
124impl EvalGrid {
125    /// Allocates the grid timestamps once for sharing across result series.
126    pub fn try_new(spec: GridSpec) -> Result<Self> {
127        let grid = Self::try_deferred(spec)?;
128        grid.timestamps()?;
129        Ok(grid)
130    }
131
132    /// Creates a validated grid without allocating its timestamp array.
133    pub fn try_deferred(spec: GridSpec) -> Result<Self> {
134        let spec = GridSpec::try_new(spec.start_ms, spec.step_ms, spec.len)?;
135        Ok(Self {
136            spec,
137            timestamps: OnceCell::new(),
138        })
139    }
140
141    /// Returns the value specification.
142    pub fn spec(&self) -> GridSpec {
143        self.spec
144    }
145
146    /// Returns the shared timestamp buffer, materializing it on first access.
147    pub fn timestamps(&self) -> Result<&ScalarBuffer<i64>> {
148        self.timestamps.get_or_try_init(|| {
149            let len = usize::try_from(self.spec.len)
150                .map_err(|_| invalid_series("grid length overflows usize"))?;
151            let mut values = Vec::new();
152            values
153                .try_reserve_exact(len)
154                .map_err(|_| invalid_series("grid timestamp allocation failed"))?;
155            for index in 0..len {
156                let timestamp =
157                    (self.spec.start_ms as i128) + (index as i128) * (self.spec.step_ms as i128);
158                values.push(timestamp as i64);
159            }
160            Ok(values.into())
161        })
162    }
163
164    fn timestamp_at(&self, position: usize) -> Option<i64> {
165        if position as u64 >= self.spec.len {
166            return None;
167        }
168        let timestamp =
169            (self.spec.start_ms as i128) + (position as i128) * (self.spec.step_ms as i128);
170        // GridSpec validation bounds every position, and i128 holds usize * i64.
171        Some(timestamp as i64)
172    }
173}

175/// A contiguous range referencing a shared evaluation grid.
176#[derive(Clone, Debug)]
177pub struct GridRange {
178    grid: Arc<EvalGrid>,
179    start: usize,
180    len: usize,
181}
182
183impl GridRange {
184    /// Creates a checked subrange; an empty range may begin at the grid end.
185    pub fn try_new(grid: Arc<EvalGrid>, start: usize, len: usize) -> Result<Self> {
186        let end = start.checked_add(len).context(InvalidSeriesSnafu {
187            reason: "grid range overflow".to_string(),
188        })?;
189        if end
190            > usize::try_from(grid.spec.len)
191                .map_err(|_| invalid_series("grid length overflows usize"))?
192        {
193            return invalid("grid range exceeds its grid");
194        }
195        Ok(Self { grid, start, len })
196    }
197
198    /// Returns the shared grid.
199    pub fn grid(&self) -> &Arc<EvalGrid> {
200        &self.grid
201    }
202    /// Returns the start index in the grid.
203    pub fn start(&self) -> usize {
204        self.start
205    }
206    /// Returns the range length.
207    pub fn len(&self) -> usize {
208        self.len
209    }
210    /// Returns whether this range is empty.
211    pub fn is_empty(&self) -> bool {
212        self.len == 0
213    }
214    /// Returns this range as a zero-copy Arrow buffer slice.
215    pub fn timestamps(&self) -> Result<ScalarBuffer<i64>> {
216        Ok(self.grid.timestamps()?.slice(self.start, self.len))
217    }
218}

220impl PartialEq for GridRange {
221    fn eq(&self, other: &Self) -> bool {
222        self.grid.spec == other.grid.spec && self.start == other.start && self.len == other.len
223    }
224}

227/// One contiguous run of positions on a shared evaluation grid.
228///
229/// The span is half-open: it names the absolute grid indices `start..start + len`. A span is
230/// never empty, so `start + len - 1` is always its last position.
231#[derive(Clone, Copy, Debug, PartialEq, Eq)]
232pub struct GridSpan {
233    start: usize,
234    len: usize,
235}
236
237// A span is never empty (the checked constructors reject a zero length), so there is no
238// `is_empty` to pair with `len`.
239#[allow(clippy::len_without_is_empty)]
240impl GridSpan {
241    /// Creates a span of `len` positions starting at absolute grid index `start`.
242    ///
243    /// [`TimeAxis::from_grid_spans`] validates it against its grid; this constructor only records
244    /// it.
245    pub fn new(start: usize, len: usize) -> Self {
246        Self { start, len }
247    }
248
249    /// Absolute grid index of the first position.
250    pub fn start(&self) -> usize {
251        self.start
252    }
253    /// Number of positions in the span.
254    pub fn len(&self) -> usize {
255        self.len
256    }
257}

259/// Several ordered, non-overlapping runs of one shared evaluation grid.
260///
261/// The runs are the rows an axis carries: everything outside them holds no value, so an axis over
262/// a few scattered positions costs a few spans instead of a row per empty position. The spans are
263/// strictly increasing and never overlap (`prev.start + prev.len <= next.start`); they may sit
264/// next to each other, and nothing is stored between them.
265///
266/// An axis with no rows is an empty span list that still carries its grid, so a consumer can tell
267/// "nothing here" from "not on a grid".
268#[derive(Clone, Debug)]
269pub struct GridRanges {
270    grid: Arc<EvalGrid>,
271    /// The runs, in absolute grid order; no span is empty.
272    spans: Arc<[GridSpan]>,
273    /// Compact row of each span's first row, parallel to `spans`: prefix sums of the lengths,
274    /// kept so that compact-row lookups need a binary search and not a scan.
275    rows: Arc<[usize]>,
276    /// Number of carried rows, cached so that [`Self::len`] never walks the spans.
277    len: usize,
278}
279
280// `len` is the number of carried rows; "carries no row" is answered by `TimeAxis::is_empty`.
281#[allow(clippy::len_without_is_empty)]
282impl GridRanges {
283    /// Creates a validated list of runs over `grid`.
284    ///
285    /// Rejects an empty span, a span past the end of the grid, spans that are out of order or
286    /// overlap, and any length or index that overflows `usize`.
287    ///
288    /// Not public: a list of one span is not normalized into the single run a [`TimeAxis`] fast
289    /// path expects, so callers outside this crate build axes through
290    /// [`TimeAxis::from_grid_spans`], which keeps [`TimeAxis::Grid`] for a single span.
291    pub(crate) fn try_new(grid: Arc<EvalGrid>, spans: Vec<GridSpan>) -> Result<Self> {
292        let grid_len = usize::try_from(grid.spec.len)
293            .map_err(|_| invalid_series("grid length overflows usize"))?;
294        let mut rows = Vec::new();
295        rows.try_reserve_exact(spans.len())
296            .map_err(|_| invalid_series("grid span allocation failed"))?;
297        let mut len = 0usize;
298        let mut previous_end: Option<usize> = None;
299        for span in &spans {
300            if span.len == 0 {
301                return invalid("grid spans must not be empty");
302            }
303            let end = span
304                .start
305                .checked_add(span.len)
306                .context(InvalidSeriesSnafu {
307                    reason: "grid span overflow".to_string(),
308                })?;
309            if end > grid_len {
310                return invalid("grid span exceeds its grid");
311            }
312            if previous_end.is_some_and(|previous_end| span.start < previous_end) {
313                return invalid("grid spans must be ordered and must not overlap");
314            }
315            rows.push(len);
316            len = len.checked_add(span.len).context(InvalidSeriesSnafu {
317                reason: "grid span length overflow".to_string(),
318            })?;
319            previous_end = Some(end);
320        }
321        Ok(Self {
322            grid,
323            spans: spans.into(),
324            rows: rows.into(),
325            len,
326        })
327    }
328
329    /// Creates an axis over `grid` that carries no row.
330    pub fn empty(grid: Arc<EvalGrid>) -> Self {
331        Self {
332            grid,
333            spans: Arc::from([]),
334            rows: Arc::from([]),
335            len: 0,
336        }
337    }
338
339    /// Returns the shared grid the spans refer to.
340    pub fn grid(&self) -> &Arc<EvalGrid> {
341        &self.grid
342    }
343
344    /// Returns the number of carried rows, not the distance from the first to the last position.
345    pub fn len(&self) -> usize {
346        self.len
347    }
348
349    /// Maps an absolute grid position to its compact row, or `None` when it holds no row.
350    ///
351    /// Binary search over the span starts: a point lookup never scans.
352    pub fn row_of(&self, position: usize) -> Option<usize> {
353        let index = self
354            .spans
355            .partition_point(|span| span.start <= position)
356            .checked_sub(1)?;
357        let span = self.spans[index];
358        let offset = position - span.start;
359        (offset < span.len).then(|| self.rows[index] + offset)
360    }
361
362    /// Returns the span holding compact row `index`, with the offset of the row inside it.
363    fn span_of(&self, index: usize) -> Option<(GridSpan, usize)> {
364        let span_index = self
365            .rows
366            .partition_point(|&row| row <= index)
367            .checked_sub(1)?;
368        let span = self.spans[span_index];
369        let offset = index - self.rows[span_index];
370        (offset < span.len).then_some((span, offset))
371    }
372
373    /// Returns the span that starts at or after compact row `index`, as an index into `spans`.
374    fn span_index_at(&self, index: usize) -> usize {
375        self.rows
376            .partition_point(|&row| row <= index)
377            .saturating_sub(1)
378    }
379
380    /// The spans covering compact rows `offset..offset + len`, renumbered from zero.
381    fn spans_between(&self, offset: usize, len: usize) -> Vec<GridSpan> {
382        let end = offset + len;
383        let mut sliced = Vec::new();
384        for index in self.span_index_at(offset)..self.spans.len() {
385            let row = self.rows[index];
386            if row >= end {
387                break;
388            }
389            let span = self.spans[index];
390            let from = offset.max(row) - row;
391            let to = (end - row).min(span.len);
392            if from < to {
393                sliced.push(GridSpan::new(span.start + from, to - from));
394            }
395        }
396        sliced
397    }
398
399    /// The absolute position of compact row `offset`, or the position after the last row when
400    /// `offset` is one past the end; `None` when the axis carries no row at all.
401    fn position_at(&self, offset: usize) -> Option<usize> {
402        let last = self.spans.last()?;
403        if offset >= self.len {
404            return Some(last.start + last.len);
405        }
406        let (span, inside) = self.span_of(offset)?;
407        Some(span.start + inside)
408    }
409
410    /// Cuts compact rows `offset..offset + len`, keeping the grid.
411    ///
412    /// The result is a [`TimeAxis::Grid`] when it stays within one span, including every empty
413    /// result over a non-empty axis, and [`TimeAxis::GridRanges`] otherwise.
414    fn slice(&self, offset: usize, len: usize) -> Result<TimeAxis> {
415        if len == 0 {
416            return Ok(match self.position_at(offset) {
417                Some(position) => {
418                    TimeAxis::Grid(GridRange::try_new(self.grid.clone(), position, 0)?)
419                }
420                None => TimeAxis::GridRanges(self.clone()),
421            });
422        }
423        let spans = self.spans_between(offset, len);
424        match spans.as_slice() {
425            [span] => Ok(TimeAxis::Grid(GridRange::try_new(
426                self.grid.clone(),
427                span.start,
428                span.len,
429            )?)),
430            _ => Ok(TimeAxis::GridRanges(Self::try_new(
431                self.grid.clone(),
432                spans,
433            )?)),
434        }
435    }
436}

442}

445/// A timestamp axis represented as a grid or explicit shared Arrow buffers.
446///
447/// Checked constructors validate timestamp ordering. Public payloads are validated again by the
448/// downstream series boundary.
449///
450/// Two coordinate conventions meet here, and every method says which one it speaks:
451///
452/// * a **compact row** is an index into the axis and into the fields aligned with it, and it
453///   counts only the rows the axis carries: [`TimeAxis::len`], [`TimeAxis::timestamp`],
454///   [`TimeAxis::slice`] and the row half of [`TimeAxis::positions`] speak compact rows;
455/// * an **absolute grid position** is an index into the shared [`EvalGrid`], holes included;
456///   [`TimeAxis::positions`], [`TimeAxis::row_of`], [`TimeAxis::first_position`] and
457///   [`TimeAxis::last_position`] speak absolute grid positions.
458#[derive(Clone, Debug)]
459pub enum TimeAxis {
460    /// A slice of a shared regular evaluation grid.
461    Grid(GridRange),
462    /// Several ordered runs of one shared evaluation grid.
463    GridRanges(GridRanges),
464    /// An explicit timestamp sequence.
465    Explicit(ScalarBuffer<i64>),
466    /// Explicit timestamps held in ordered chunks without flattening.
467    ExplicitChunks(Arc<[ScalarBuffer<i64>]>),
468}
469
470/// Sequential traversal of an axis, as returned by [`TimeAxis::positions`].
471///
472/// Yields `(absolute grid position, compact row)` pairs in increasing position order, and walks
473/// the runs of a [`TimeAxis::GridRanges`] with one cursor, so a row never costs a scan over the
474/// spans before it.
475pub struct Positions<'a> {
476    source: PositionSource<'a>,
477}
478
479enum PositionSource<'a> {
480    /// A dense grid range: position and row advance together.
481    Grid {
482        start: usize,
483        row: usize,
484        len: usize,
485    },
486    /// Several runs: the cursor is the span index plus the offset inside it.
487    GridRanges {
488        spans: &'a [GridSpan],
489        span: usize,
490        offset: usize,
491        row: usize,
492        len: usize,
493    },
494    /// No grid: a row sits at its own position.
495    Rows { row: usize, len: usize },
496}
497
498impl Iterator for Positions<'_> {
499    type Item = (usize, usize);
500
501    fn next(&mut self) -> Option<(usize, usize)> {
502        match &mut self.source {
503            PositionSource::Grid { start, row, len } => {
504                if *row >= *len {
505                    return None;
506                }
507                let item = (*start + *row, *row);
508                *row += 1;
509                Some(item)
510            }
511            PositionSource::GridRanges {
512                spans,
513                span,
514                offset,
515                row,
516                len,
517            } => {
518                if *row >= *len {
519                    return None;
520                }
521                loop {
522                    let current = spans.get(*span)?;
523                    if *offset < current.len {
524                        let item = (current.start + *offset, *row);
525                        *offset += 1;
526                        *row += 1;
527                        return Some(item);
528                    }
529                    *span += 1;
530                    *offset = 0;
531                }
532            }
533            PositionSource::Rows { row, len } => {
534                if *row >= *len {
535                    return None;
536                }
537                let item = (*row, *row);
538                *row += 1;
539                Some(item)
540            }
541        }
542    }
543
544    fn size_hint(&self) -> (usize, Option<usize>) {
545        let (row, len) = match &self.source {
546            PositionSource::Grid { row, len, .. } => (*row, *len),
547            PositionSource::GridRanges { row, len, .. } => (*row, *len),
548            PositionSource::Rows { row, len } => (*row, *len),
549        };
550        let remaining = len - row;
551        (remaining, Some(remaining))
552    }
553}

557impl TimeAxis {
558    /// Validates an explicit, strictly increasing timestamp sequence.
559    pub fn try_new(timestamps: ScalarBuffer<i64>) -> Result<Self> {
560        let axis = Self::Explicit(timestamps);
561        axis.try_len()?;
562        Ok(axis)
563    }
564
565    /// Validates and stores timestamp chunks without concatenating their buffers.
566    pub fn try_from_chunks(chunks: Vec<ScalarBuffer<i64>>) -> Result<Self> {
567        let axis = Self::ExplicitChunks(
568            chunks
569                .into_iter()
570                .filter(|chunk| !chunk.is_empty())
571                .collect(),
572        );
573        axis.try_len()?;
574        Ok(axis)
575    }
576
577    /// Builds an axis over `grid` from the runs `spans`, keeping the single-run fast path.
578    ///
579    /// This is what a producer holding only the rows it decided should use: one span becomes
580    /// [`Self::Grid`], several become [`Self::GridRanges`], and no span becomes an empty
581    /// [`Self::GridRanges`] that still carries the grid. In particular it never fills the
582    /// positions between two spans, which is what keeps a sparse result from costing a row per
583    /// empty position.
584    pub fn from_grid_spans(grid: Arc<EvalGrid>, spans: Vec<GridSpan>) -> Result<Self> {
585        match spans.as_slice() {
586            [] => Ok(Self::GridRanges(GridRanges::empty(grid))),
587            [span] => Ok(Self::Grid(GridRange::try_new(grid, span.start, span.len)?)),
588            _ => Ok(Self::GridRanges(GridRanges::try_new(grid, spans)?)),
589        }
590    }
591
592    /// Returns the validated timestamp count.
593    pub(crate) fn try_len(&self) -> Result<usize> {
594        match self {
595            Self::Grid(range) => Ok(range.len()),
596            // `GridRanges` is only reachable through its checked constructor.
597            Self::GridRanges(ranges) => Ok(ranges.len()),
598            Self::Explicit(values) => {
599                if values.windows(2).any(|pair| pair[0] >= pair[1]) {
600                    return invalid("timestamps must be strictly increasing");
601                }
602                Ok(values.len())
603            }
604            Self::ExplicitChunks(chunks) => {
605                let mut total = 0usize;
606                let mut previous = None;
607                for chunk in chunks.iter() {
608                    total = total
609                        .checked_add(chunk.len())
610                        .ok_or_else(|| invalid_series("timestamp length overflow"))?;
611                    if chunk.windows(2).any(|pair| pair[0] >= pair[1])
612                        || chunk
613                            .first()
614                            .is_some_and(|first| previous.is_some_and(|last| last >= *first))
615                    {
616                        return invalid("timestamps must be strictly increasing across chunks");
617                    }
618                    if let Some(last) = chunk.last() {
619                        previous = Some(*last);
620                    }
621                }
622                Ok(total)
623            }
624        }
625    }
626
627    /// Returns the compact row count, not the distance from the first to the last position.
628    ///
629    /// A [`Self::GridRanges`] answers from its cached total and does not walk its spans.
630    pub fn len(&self) -> usize {
631        match self {
632            Self::Grid(range) => range.len,
633            Self::GridRanges(ranges) => ranges.len(),
634            Self::Explicit(values) => values.len(),
635            Self::ExplicitChunks(chunks) => chunks.iter().map(|chunk| chunk.len()).sum(),
636        }
637    }
638    /// Returns whether there are no timestamps.
639    pub fn is_empty(&self) -> bool {
640        self.len() == 0
641    }
642
643    /// Returns a timestamp by compact row index.
644    pub fn timestamp(&self, index: usize) -> Option<i64> {
645        match self {
646            Self::Grid(range) => (index < range.len)
647                .then(|| range.grid.timestamp_at(range.start.checked_add(index)?))
648                .flatten(),
649            Self::GridRanges(ranges) => {
650                let (span, offset) = ranges.span_of(index)?;
651                ranges.grid.timestamp_at(span.start.checked_add(offset)?)
652            }
653            Self::Explicit(values) => values.get(index).copied(),
654            Self::ExplicitChunks(chunks) => {
655                let mut offset = index;
656                for chunk in chunks.iter() {
657                    if offset < chunk.len() {
658                        return chunk.get(offset).copied();
659                    }
660                    offset -= chunk.len();
661                }
662                None
663            }
664        }
665    }
666
667    /// Returns a zero-copy slice, retaining Arrow owners; chunked axes remain chunked.
668    ///
669    /// `offset` and `len` count compact rows, so the result stays aligned with the fields of the
670    /// series the axis belongs to. A slice that stays within one run of a [`Self::GridRanges`]
671    /// comes back as [`Self::Grid`]; every empty slice keeps the grid it was cut from.
672    pub fn slice(&self, offset: usize, len: usize) -> Result<Self> {
673        let end = offset.checked_add(len).context(InvalidSeriesSnafu {
674            reason: "timestamp slice overflow".to_string(),
675        })?;
676        if end > self.len() {
677            return invalid("timestamp slice exceeds axis");
678        }
679        match self {
680            Self::Grid(range) => Ok(Self::Grid(GridRange::try_new(
681                range.grid.clone(),
682                range.start + offset,
683                len,
684            )?)),
685            Self::GridRanges(ranges) => ranges.slice(offset, len),
686            Self::Explicit(values) => Ok(Self::Explicit(values.slice(offset, len))),
687            Self::ExplicitChunks(chunks) => {
688                let mut remain = len;
689                let mut skip = offset;
690                let mut sliced = Vec::new();
691                for chunk in chunks.iter() {
692                    if skip >= chunk.len() {
693                        skip -= chunk.len();
694                        continue;
695                    }
696                    let take = remain.min(chunk.len() - skip);
697                    if take > 0 {
698                        sliced.push(chunk.slice(skip, take));
699                        remain -= take;
700                    }
701                    skip = 0;
702                    if remain == 0 {
703                        break;
704                    }
705                }
706                Ok(Self::ExplicitChunks(sliced.into()))
707            }
708        }
709    }

711    /// Returns the single contiguous range this axis covers, if it covers exactly one.
712    ///
713    /// [`Self::Grid`] always answers `Some`, including for an empty range that still names the
714    /// position it sits at. A [`Self::GridRanges`] holding several runs, and every explicit axis,
715    /// answer `None`: their rows are not one range, and a caller that assumed one would read the
716    /// holes between them as rows.
717    pub fn as_grid_range(&self) -> Option<&GridRange> {
718        match self {
719            Self::Grid(range) => Some(range),
720            Self::GridRanges(_) | Self::Explicit(_) | Self::ExplicitChunks(_) => None,
721        }
722    }

724    /// Returns the evaluation grid the axis is addressed on, if it has one.
725    ///
726    /// Both [`Self::Grid`] and [`Self::GridRanges`] answer `Some`; the explicit axes answer `None`.
727    pub fn grid(&self) -> Option<&Arc<EvalGrid>> {
728        match self {
729            Self::Grid(range) => Some(range.grid()),
730            Self::GridRanges(ranges) => Some(ranges.grid()),
731            Self::Explicit(_) | Self::ExplicitChunks(_) => None,
732        }
733    }

735    /// Maps an absolute grid position to its compact row, or `None` when the axis carries no row
736    /// there.
737    ///
738    /// A point lookup: a [`Self::GridRanges`] binary-searches its spans, it does not scan them.
739    /// On an explicit axis a row sits at its own position, so only the length bounds it.
740    pub fn row_of(&self, position: usize) -> Option<usize> {
741        match self {
742            Self::Grid(range) => (position >= range.start
743                && position
744                    .checked_sub(range.start)
745                    .is_some_and(|row| row < range.len))
746            .then(|| position - range.start),
747            Self::GridRanges(ranges) => ranges.row_of(position),
748            Self::Explicit(_) | Self::ExplicitChunks(_) => {
749                (position < self.len()).then_some(position)
750            }
751        }
752    }
753
754    /// Walks the axis once, yielding `(absolute grid position, compact row)` in position order.
755    ///
756    /// This is the traversal every consumer of a grid axis should use: a helper walks the runs of
757    /// a [`Self::GridRanges`] with a single cursor, so a row never rescans the spans before it.
758    /// On an explicit axis there is no grid, so a row sits at its own position.
759    pub fn positions(&self) -> Positions<'_> {
760        let len = self.len();
761        let source = match self {
762            Self::Grid(range) => PositionSource::Grid {
763                start: range.start,
764                row: 0,
765                len,
766            },
767            Self::GridRanges(ranges) => PositionSource::GridRanges {
768                spans: &ranges.spans,
769                span: 0,
770                offset: 0,
771                row: 0,
772                len,
773            },
774            Self::Explicit(_) | Self::ExplicitChunks(_) => PositionSource::Rows { row: 0, len },
775        };
776        Positions { source }
777    }

779    /// Returns the absolute grid position of the first carried row.
780    pub fn first_position(&self) -> Option<usize> {
781        match self {
782            Self::Grid(range) => (!range.is_empty()).then_some(range.start),
783            Self::GridRanges(ranges) => ranges.spans.first().map(GridSpan::start),
784            Self::Explicit(_) | Self::ExplicitChunks(_) => (!self.is_empty()).then_some(0),
785        }
786    }
787
788    /// Returns the absolute grid position of the last carried row.
789    pub fn last_position(&self) -> Option<usize> {
790        match self {
791            Self::Grid(range) => (!range.is_empty()).then(|| range.start + range.len - 1),
792            Self::GridRanges(ranges) => ranges.spans.last().map(|span| span.start + span.len - 1),
793            Self::Explicit(_) | Self::ExplicitChunks(_) => self.len().checked_sub(1),
794        }
795    }
796
797    /// Returns whether the carried rows form one run, which is what a fast path can rely on.
798    ///
799    /// A [`Self::GridRanges`] answers `true` only for a single span, so an empty span list reports
800    /// `false`: it names no run, and therefore no position a fast path could start at.
801    pub fn is_contiguous(&self) -> bool {
802        match self {
803            Self::Grid(_) => true,
804            Self::GridRanges(ranges) => ranges.spans.len() == 1,
805            Self::Explicit(_) => true,
806            Self::ExplicitChunks(chunks) => chunks.len() <= 1,
807        }
808    }

810    /// Returns the timestamps of the carried rows, in compact-row order.
811    ///
812    /// A single-run grid axis lends its buffer without copying; an axis of several runs, and a
813    /// chunked explicit axis, allocate one array of the carried rows.
814    pub fn timestamps(&self) -> Result<Cow<'_, [i64]>> {
815        match self {
816            Self::Grid(range) => Ok(Cow::Borrowed(
817                &range.grid.timestamps()?[range.start()..range.start() + range.len()],
818            )),
819            Self::GridRanges(ranges) => {
820                let mut timestamps = Vec::with_capacity(ranges.len());
821                for (position, _) in self.positions() {
822                    timestamps.push(
823                        ranges
824                            .grid
825                            .timestamp_at(position)
826                            .ok_or_else(|| invalid_series("grid span timestamp is out of range"))?,
827                    );
828                }
829                Ok(Cow::Owned(timestamps))
830            }
831            Self::Explicit(values) => Ok(Cow::Borrowed(values)),
832            Self::ExplicitChunks(_) => Ok(Cow::Owned(
833                (0..self.len())
834                    .filter_map(|row| self.timestamp(row))
835                    .collect(),
836            )),
837        }
838    }

840    /// Returns one run per stored span of carried rows, as [`GridRun`]s.
841    ///
842    /// One run for a contiguous axis, one per stored span for [`Self::GridRanges`], and none when
843    /// no row is carried. Adjacent stored spans stay two runs, so a run is a maximal run only when
844    /// the axis was built from maximal spans (as the series builder does). A run is where bulk
845    /// work pays off: the rows inside it are consecutive in both coordinates, so its samples can
846    /// be sliced and zipped instead of read one by one.
847    pub fn runs(&self) -> GridRuns<'_> {
848        let (one, spans) = match self {
849            Self::Grid(range) => (
850                Some(GridRun {
851                    position: range.start,
852                    row: 0,
853                    len: range.len,
854                }),
855                &[][..],
856            ),
857            Self::GridRanges(ranges) => (None, &ranges.spans[..]),
858            Self::Explicit(_) | Self::ExplicitChunks(_) => (
859                Some(GridRun {
860                    position: 0,
861                    row: 0,
862                    len: self.len(),
863                }),
864                &[][..],
865            ),
866        };
867        GridRuns {
868            one: one.filter(|run| run.len > 0),
869            spans,
870            span: 0,
871            row: 0,
872        }
873    }

875    /// Intersects the carried rows of two axes, yielding one shared run per pair of overlapping
876    /// stored runs as `(first absolute position, row in `self`, row in `other`, length)`.
877    ///
878    /// A single merge pass over the two run lists, so the cost is the number of runs and not their
879    /// product. This is how a pairwise operator aligns two axes: it never builds the dense
880    /// envelope `first_position()..last_position()`, which would visit -- and densify -- the holes,
881    /// and it never re-derives the row of a position by subtracting starts, which is wrong as soon
882    /// as either side skips a row.
883    ///
884    /// Both axes must be addressed on the same [`GridSpec`]: the positions it yields are indices
885    /// into one shared [`EvalGrid`], so two different grids line up only by coincidence. An
886    /// explicit axis carries no grid and must not be passed at all -- its rows are not absolute
887    /// positions on any shared grid, so the pairs would be silent nonsense. Debug builds assert
888    /// the precondition and therefore panic on a violation; release builds trust the caller.
889    pub fn intersect_runs<'a>(&'a self, other: &'a Self) -> impl Iterator<Item = GridRunPair> + 'a {
890        debug_assert!(
891            matches!(
892                (self.grid(), other.grid()),
893                (Some(lhs), Some(rhs)) if lhs.spec() == rhs.spec()
894            ),
895            "intersect_runs requires two axes on one grid, \
896             but an explicit axis carries none and two grids only meet by coincidence",
897        );
898        let mut lhs = self.runs().peekable();
899        let mut rhs = other.runs().peekable();
900        std::iter::from_fn(move || {
901            loop {
902                let left = *lhs.peek()?;
903                let right = *rhs.peek()?;
904                let start = left.position.max(right.position);
905                let end = (left.position + left.len).min(right.position + right.len);
906                if start >= end {
907                    // No shared position in this pair of runs: drop the one that ends first.
908                    if left.position + left.len <= right.position + right.len {
909                        lhs.next();
910                    } else {
911                        rhs.next();
912                    }
913                    continue;
914                }
915                let len = end - start;
916                let pair = GridRunPair {
917                    position: start,
918                    row: left.row + (start - left.position),
919                    other_row: right.row + (start - right.position),
920                    len,
921                };
922                // Consume the shared piece from both sides, keeping the remainder for the next
923                // round.
924                if left.position + left.len <= end {
925                    lhs.next();
926                }
927                if right.position + right.len <= end {
928                    rhs.next();
929                }
930                return Some(pair);
931            }
932        })
933    }
934}

936/// A run of carried rows: `len` rows starting at absolute position `position` and compact
937/// `row`, both of which advance by one per row.
938#[derive(Clone, Copy, Debug, PartialEq, Eq)]
939pub struct GridRun {
940    /// Absolute grid position of the run's first row.
941    pub position: usize,
942    /// Compact row of the run's first row.
943    pub row: usize,
944    /// Number of rows in the run; never zero.
945    pub len: usize,
946}

948/// A run shared by two axes, with the compact row of each.
949#[derive(Clone, Copy, Debug, PartialEq, Eq)]
950pub struct GridRunPair {
951    /// Absolute grid position of the shared run's first row.
952    pub position: usize,
953    /// Compact row of the first shared position in the axis `intersect_runs` was called on.
954    pub row: usize,
955    /// Compact row of the first shared position in the other axis.
956    pub other_row: usize,
957    /// Number of rows in the shared run; never zero.
958    pub len: usize,

961/// Lazily produced runs of one axis; see [`TimeAxis::runs`].
962///
963/// A contiguous axis holds its one run inline; a sparse axis walks its spans with a cursor.
964pub struct GridRuns<'a> {
965    one: Option<GridRun>,
966    spans: &'a [GridSpan],
967    span: usize,
968    row: usize,
969}
970
971impl Iterator for GridRuns<'_> {
972    type Item = GridRun;
973
974    fn next(&mut self) -> Option<GridRun> {
975        if let Some(run) = self.one.take() {
976            return Some(run);
977        }
978        let span = *self.spans.get(self.span)?;
979        self.span += 1;
980        let run = GridRun {
981            position: span.start,
982            row: self.row,
983            len: span.len,
984        };
985        self.row += span.len;
986        Some(run)
987    }
988}

990/// Walks every position at least one of `axes` carries, once, in increasing order.
991///
992/// A merge over one cursor per axis: each step costs one comparison per axis, and no position is
993/// visited twice. This is how an operator covering several axes (a histogram's bucket series, an
994/// aggregate over an input set) visits the positions that hold something, instead of scanning the
995/// envelope from the earliest to the latest position and densifying every hole in between.
996///
997/// Every axis must be addressed on the same [`GridSpec`]: a position names a point of one shared
998/// [`EvalGrid`], so merging axes of two different grids would interleave unrelated positions. An
999/// explicit axis carries no grid and must not be passed at all -- its rows are not absolute
1000/// positions on any shared grid. Debug builds assert the precondition and therefore panic on a
1001/// violation; release builds trust the caller.
1002pub fn union_positions<'a>(axes: impl IntoIterator<Item = &'a TimeAxis>) -> UnionPositions<'a> {
1003    // Materialized once so that the debug assert above can see every axis' grid before the first
1004    // cursor is built; the cursors below borrow these references, not the list.
1005    let axes: Vec<&'a TimeAxis> = axes.into_iter().collect();
1006    debug_assert!(
1007        {
1008            let mut specs = axes.iter().map(|axis| axis.grid().map(|grid| grid.spec()));
1009            match specs.next() {
1010                // No axis at all: the merge yields nothing and reads no grid.
1011                None => true,
1012                Some(None) => false,
1013                Some(Some(first)) => specs.all(|spec| spec == Some(first)),
1014            }
1015        },
1016        "union_positions requires every axis on one grid, \
1017         but an explicit axis carries none and two grids only meet by coincidence",
1018    );
1019    let cursors: Vec<Positions<'a>> = axes.into_iter().map(TimeAxis::positions).collect();
1020    let pending = vec![None; cursors.len()];
1021    UnionPositions { cursors, pending }
1022}
1023
1024/// The positions of several axes, merged; see [`union_positions`].
1025pub struct UnionPositions<'a> {
1026    cursors: Vec<Positions<'a>>,
1027    /// The next `(position, row)` of each cursor, buffered so that it can be peeked without a
1028    /// second cursor type.
1029    pending: Vec<Option<(usize, usize)>>,
1030}
1031
1032impl Iterator for UnionPositions<'_> {
1033    type Item = usize;
1034
1035    fn next(&mut self) -> Option<usize> {
1036        let mut lowest: Option<usize> = None;
1037        for (cursor, pending) in self.cursors.iter_mut().zip(&mut self.pending) {
1038            if pending.is_none() {
1039                *pending = cursor.next();
1040            }
1041            if let Some((position, _)) = *pending {
1042                lowest = Some(lowest.map_or(position, |lowest| lowest.min(position)));
1043            }
1044        }
1045        let lowest = lowest?;
1046        // Every axis standing at this position advances; the others keep their buffered position.
1047        for pending in &mut self.pending {
1048            if matches!(*pending, Some((position, _)) if position == lowest) {
1049                *pending = None;
1050            }
1051        }
1052        Some(lowest)
1053    }
1054}
