26/// Rewrites string literals that feed timestamp columns at an INSERT boundary.
27/// Constants are folded at the assignment to avoid changing source types;
28/// `VALUES` and `UNION` inputs are rewritten per row or branch. Explicit casts
29/// stay on DataFusion's existing path.
30pub(super) fn rewrite_insert_assignments(
31    plan: LogicalPlan,
32    query_ctx: QueryContextRef,
33) -> Result<LogicalPlan> {
34    let LogicalPlan::Projection(assignment) = plan else {
35        return Ok(plan);
36    };
37
38    let converter = InsertAssignmentConverter { query_ctx };
39    let mut exprs = assignment.expr.clone();
40    let mut input = assignment.input.as_ref().clone();
41    let mut changed = false;
42    for (output_idx, expr) in assignment.expr.iter().enumerate() {
43        let target_type = assignment.schema.field(output_idx).data_type();
44        if !matches!(target_type, DataType::Timestamp(_, _)) {
45            continue;
46        }
47
48        let Some(column) = assignment_input_column(expr) else {
49            continue;
50        };
51        let Some(input_idx) = input.schema().maybe_index_of_column(column) else {
52            continue;
53        };
54
55        let literal = lineage_literal(&input, input_idx).cloned();
56        if let Some(literal) = literal
57            && let Some(folded) = converter.convert_literal(&literal, target_type)
58        {
59            let (qualifier, field) = assignment.schema.qualified_field(output_idx);
60            exprs[output_idx] = folded.alias_qualified(qualifier.cloned(), field.name());
61            changed = true;
62            continue;
63        }
64
65        // A hand-built DML plan (e.g. from flow) may share one source column
66        // between targets; rewriting it in place would retype every reader.
67        if assignment
68            .expr
69            .iter()
70            .enumerate()
71            .any(|(idx, other)| idx != output_idx && other.column_refs().contains(column))
72        {
73            continue;
74        }
75        if let Some(rewritten) = converter.rewrite_output_column(&input, input_idx, target_type)? {
76            input = rewritten;
77            changed = true;
78        }
79    }
80
81    if !changed {
82        return Ok(LogicalPlan::Projection(assignment));
83    }
84    Projection::try_new(exprs, Arc::new(input)).map(LogicalPlan::Projection)
85}

87/// Resolves a literal when every row carries the same value at `output_idx`.
88fn lineage_literal(plan: &LogicalPlan, output_idx: usize) -> Option<&ScalarValue> {
89    if output_idx >= plan.schema().fields().len() {
90        return None;
91    }
92
93    match plan {
94        LogicalPlan::Projection(projection) => match unalias(&projection.expr[output_idx]) {
95            Expr::Literal(value, _) => Some(value),
96            Expr::Column(column) => {
97                let input_idx = projection.input.schema().maybe_index_of_column(column)?;
98                lineage_literal(projection.input.as_ref(), input_idx)
99            }
100            _ => None,
101        },
102        // These nodes drop, reorder or deduplicate rows without touching the
103        // value a surviving row carries, and their schema stays positional.
104        LogicalPlan::Filter(_)
105        | LogicalPlan::Sort(_)
106        | LogicalPlan::Limit(_)
107        | LogicalPlan::SubqueryAlias(_)
108        | LogicalPlan::Distinct(Distinct::All(_)) => {
109            let inputs = plan.inputs();
110            let [input] = inputs.as_slice() else {
111                return None;
112            };
113            lineage_literal(input, output_idx)
114        }
115        _ => None,
116    }
117}
118
119struct InsertAssignmentConverter {
120    query_ctx: QueryContextRef,
121}
122
123impl InsertAssignmentConverter {
124    fn rewrite_output_column(
125        &self,
126        plan: &LogicalPlan,
127        output_idx: usize,
128        target_type: &DataType,
129    ) -> Result<Option<LogicalPlan>> {
130        if output_idx >= plan.schema().fields().len() {
131            return Ok(None);
132        }
133
134        match plan {
135            // DataFusion pushes INSERT assignment casts into Values, so inspect
136            // them even though the Values schema already has the target type.
137            LogicalPlan::Values(values) => self.rewrite_values(values, output_idx, target_type),
138            // Once the source query has produced the target type, its casts and
139            // coercions belong to the source query rather than INSERT assignment.
140            _ if plan.schema().field(output_idx).data_type() == target_type => Ok(None),
141            LogicalPlan::Projection(projection) => {
142                self.rewrite_projection(projection, output_idx, target_type)
143            }
144            LogicalPlan::Union(union) => self.rewrite_union(union, output_idx, target_type),
145            // Filter and Sort are excluded here: their predicates and keys read
146            // the retyped column, which would change the source query. Constants
147            // still reach the assignment through `lineage_literal`. Distinct::All
148            // holds no expressions; allowing it shifts dedup keys from raw
149            // strings to parsed instants.
150            LogicalPlan::Limit(_)
151            | LogicalPlan::SubqueryAlias(_)
152            | LogicalPlan::Distinct(Distinct::All(_)) => {
153                self.rewrite_passthrough(plan, output_idx, target_type)
154            }
155            _ => Ok(None),
156        }
157    }
158
159    fn rewrite_projection(
160        &self,
161        projection: &Projection,
162        output_idx: usize,
163        target_type: &DataType,
164    ) -> Result<Option<LogicalPlan>> {
165        match unalias(&projection.expr[output_idx]) {
166            Expr::Literal(value, _) => {
167                let Some(converted) = self.convert_literal(value, target_type) else {
168                    return Ok(None);
169                };
170                let (qualifier, field) = projection.schema.qualified_field(output_idx);
171                let mut exprs = projection.expr.clone();
172                exprs[output_idx] = converted.alias_qualified(qualifier.cloned(), field.name());
173
174                Projection::try_new(exprs, projection.input.clone())
175                    .map(LogicalPlan::Projection)
176                    .map(Some)
177            }
178            Expr::Column(column) => {
179                // Retyping the input column affects every output column reading
180                // it, so only follow lineage with a single consumer.
181                if projection
182                    .expr
183                    .iter()
184                    .enumerate()
185                    .any(|(idx, other)| idx != output_idx && other.column_refs().contains(column))
186                {
187                    return Ok(None);
188                }
189
190                let input = projection.input.as_ref();
191                let Some(input_idx) = input.schema().maybe_index_of_column(column) else {
192                    return Ok(None);
193                };
194                let Some(rewritten) = self.rewrite_output_column(input, input_idx, target_type)?
195                else {
196                    return Ok(None);
197                };
198
199                Projection::try_new(projection.expr.clone(), Arc::new(rewritten))
200                    .map(LogicalPlan::Projection)
201                    .map(Some)
202            }
203            _ => Ok(None),
204        }
205    }
206
207    fn rewrite_values(
208        &self,
209        values: &Values,
210        output_idx: usize,
211        target_type: &DataType,
212    ) -> Result<Option<LogicalPlan>> {
213        let mut rewritten = values.clone();
214        let mut changed = false;
215        for row in &mut rewritten.values {
216            let Some(expr) = row.get_mut(output_idx) else {
217                return Ok(None);
218            };
219
220            if let Expr::Cast(Cast {
221                expr: inner,
222                data_type,
223            }) = expr
224                && data_type == target_type
225                && let Expr::Literal(value, _) = inner.as_ref()
226                && let Some(value) = self.convert_literal(value, target_type)
227            {
228                *expr = value;
229                changed = true;
230                continue;
231            }
232
233            if expr.get_type(values.schema.as_ref()).ok().as_ref() != Some(target_type) {
234                return Ok(None);
235            }
236        }
237
238        Ok(changed.then_some(LogicalPlan::Values(rewritten)))
239    }
240
241    fn rewrite_union(
242        &self,
243        union: &Union,
244        output_idx: usize,
245        target_type: &DataType,
246    ) -> Result<Option<LogicalPlan>> {
247        let mut inputs = Vec::with_capacity(union.inputs.len());
248        // A union has one shared schema, so rewrite it only when the same output
249        // column is an INSERT-assigned literal in every branch.
250        for input in &union.inputs {
251            let Some(rewritten) = self.rewrite_output_column(input, output_idx, target_type)?
252            else {
253                return Ok(None);
254            };
255            inputs.push(Arc::new(rewritten));
256        }
257
258        // Untouched columns may still have mismatched branch types before
259        // TypeCoercion runs, so rebuild loosely like the SQL planner does.
260        Union::try_new_with_loose_types(inputs)
261            .map(LogicalPlan::Union)
262            .map(Some)
263    }
264
265    fn rewrite_passthrough(
266        &self,
267        plan: &LogicalPlan,
268        output_idx: usize,
269        target_type: &DataType,
270    ) -> Result<Option<LogicalPlan>> {
271        let inputs = plan.inputs();
272        let [input] = inputs.as_slice() else {
273            return Ok(None);
274        };
275        let Some(rewritten_input) = self.rewrite_output_column(input, output_idx, target_type)?
276        else {
277            return Ok(None);
278        };
279
280        plan.with_new_exprs(plan.expressions_consider_join(), vec![rewritten_input])
281            .map(Some)
282    }
283
284    fn convert_literal(&self, value: &ScalarValue, target_type: &DataType) -> Option<Expr> {
285        let ScalarValue::Utf8(Some(value)) = value else {
286            return None;
287        };
288        cast_string_to_timestamp(value, target_type, Some(&self.query_ctx.timezone()))
289            .ok()
290            .filter(|value| !value.is_null())
291            .map(|value| Expr::Literal(value, None))
292    }
293}
294
295fn assignment_input_column(expr: &Expr) -> Option<&Column> {
296    let expr = match unalias(expr) {
297        Expr::Cast(Cast { expr, .. }) => expr.as_ref(),
298        expr => expr,
299    };
300    let Expr::Column(column) = expr else {
301        return None;
302    };
303    Some(column)
304}
305
306fn unalias(expr: &Expr) -> &Expr {
307    match expr {
308        Expr::Alias(Alias { expr, .. }) => unalias(expr),
309        expr => expr,
310    }
311}

314mod tests {
315    use datafusion_common::ScalarValue;
316    use datafusion_common::arrow::datatypes::TimeUnit;
317    use session::context::QueryContext;
318
319    use super::*;
320
321    #[test]
322    fn test_convert_literal_falls_back_for_unsupported_literal() {
323        let converter = InsertAssignmentConverter {
324            query_ctx: QueryContext::arc(),
325        };
326        let target_type = DataType::Timestamp(TimeUnit::Nanosecond, None);
327
328        for literal in ["1970-01-01", "-8-01-01 00:00:01.5"] {
329            assert_eq!(
330                converter
331                    .convert_literal(&ScalarValue::Utf8(Some(literal.to_string())), &target_type,),
332                None
333            );
334        }
335    }
