30/// Derives an input bound from an existential HAVING condition and MIN/MAX.
31///
32/// For example, `MIN(x), SUM(CASE WHEN p AND x < u THEN 1 ELSE 0 END)` with
33/// HAVING `SUM(...) > 0` can discard rows outside `x < u`: they contribute zero
34/// to SUM, and every surviving group has a witness below u, preserving its MIN.
35/// MAX permits the symmetric lower bound. The original HAVING must remain.
36///
37/// This deliberately requires exactly these two aggregates and ordinary column
38/// grouping. Extra aggregates, grouping sets, floating-point extrema and arbitrary
39/// expressions need separate preservation proofs. It must run before MergeScan
40/// hides remote inputs, as well as in the non-distributed logical optimizer.
41#[derive(Debug)]
42pub(crate) struct InferAggregateInputBoundsRule;
43
44impl OptimizerRule for InferAggregateInputBoundsRule {
45    fn name(&self) -> &str {
46        "infer_aggregate_input_bounds"
47    }
48
49    fn apply_order(&self) -> Option<ApplyOrder> {
50        Some(ApplyOrder::TopDown)
51    }
52
53    fn rewrite(
54        &self,
55        plan: LogicalPlan,
56        _config: &dyn OptimizerConfig,
57    ) -> Result<Transformed<LogicalPlan>> {
58        let LogicalPlan::Filter(mut having) = plan else {
59            return Ok(Transformed::no(plan));
60        };
61        let LogicalPlan::Aggregate(aggregate) = having.input.as_ref() else {
62            return Ok(Transformed::no(LogicalPlan::Filter(having)));
63        };
64        // The pre-MergeScan pass already handled remote inputs. Their filters
65        // are hidden from this pass, so re-inference can add useless local work.
66        if aggregate.input.exists(|plan| {
67            Ok(matches!(plan, LogicalPlan::Extension(extension)
68                if extension.node.as_any().is::<MergeScanLogicalPlan>()))
69        })? {
70            return Ok(Transformed::no(LogicalPlan::Filter(having)));
71        }
72        let Some(bound) = infer_bound(aggregate, &having.predicate)? else {
73            return Ok(Transformed::no(LogicalPlan::Filter(having)));
74        };
75        if already_filtered(&aggregate.input, &bound) {
76            return Ok(Transformed::no(LogicalPlan::Filter(having)));
77        }
78
79        let mut aggregate = aggregate.clone();
80        aggregate.input = Arc::new(LogicalPlan::Filter(Filter::try_new(
81            bound,
82            aggregate.input,
83        )?));
84        having.input = Arc::new(LogicalPlan::Aggregate(aggregate));
85        Ok(Transformed::yes(LogicalPlan::Filter(having)))
86    }
87}

89fn infer_bound(aggregate: &Aggregate, having: &Expr) -> Result<Option<Expr>> {
90    if aggregate.aggr_expr.len() != 2
91        || aggregate.group_expr.is_empty()
92        || !aggregate
93            .group_expr
94            .iter()
95            .all(|e| matches!(e, Expr::Column(_)))
96        || !safe_predicate(having, aggregate.schema.as_ref())?
97    {
98        return Ok(None);
99    }
100
101    for conjunct in split_conjunction(having) {
102        let Some(sum_index) = positive_sum_index(conjunct, aggregate) else {
103            continue;
104        };
105        let Some(sum) = plain_aggregate(&aggregate.aggr_expr[sum_index]) else {
106            continue;
107        };
108        let Some(extremum) = plain_aggregate(&aggregate.aggr_expr[1 - sum_index]) else {
109            continue;
110        };
111        if let Some(bound) = bound_from_aggregates(sum, extremum, &aggregate.input)? {
112            return Ok(Some(bound));
113        }
114    }
115    Ok(None)
116}

118fn positive_sum_index(predicate: &Expr, aggregate: &Aggregate) -> Option<usize> {
119    let Expr::BinaryExpr(comparison) = predicate else {
120        return None;
121    };
122    let (output, threshold, operator) = match (comparison.left.as_ref(), comparison.right.as_ref())
123    {
124        (Expr::Column(column), literal) => (column, literal, comparison.op),
125        (literal, Expr::Column(column)) => (column, literal, comparison.op.swap()?),
126        _ => return None,
127    };
128    if !((operator == Operator::Gt && integer_literal(threshold) == Some(0))
129        || (operator == Operator::GtEq && integer_literal(threshold) == Some(1)))
130    {
131        return None;
132    }
133    let index = aggregate.schema.index_of_column(output).ok()?;
134    index
135        .checked_sub(aggregate.group_expr.len())
136        .filter(|&i| i < 2)
137}

139fn plain_aggregate(expr: &Expr) -> Option<&AggregateFunction> {
140    let expr = match expr {
141        Expr::Alias(alias) => alias.expr.as_ref(),
142        _ => expr,
143    };
144    let Expr::AggregateFunction(function) = expr else {
145        return None;
146    };
147    let params = &function.params;
148    (params.args.len() == 1
149        && !params.distinct
150        && params.filter.is_none()
151        && params.order_by.is_empty()
152        && params.null_treatment.is_none())
153    .then_some(function)
154}

156fn bound_from_aggregates(
157    sum: &AggregateFunction,
158    extremum: &AggregateFunction,
159    input: &LogicalPlan,
160) -> Result<Option<Expr>> {
161    let schema = input.schema();
162    // Compare function identities, not names: a user-defined "sum" need not
163    // have the built-in aggregate's zero-contribution semantics.
164    if sum.func != sum_udaf() || (extremum.func != min_udaf() && extremum.func != max_udaf()) {
165        return Ok(None);
166    }
167    let argument = &extremum.params.args[0];
168    let argument_type = argument.get_type(schema)?;
169    if !(argument_type.is_integer() || matches!(argument_type, DataType::Timestamp(..)))
170        || !safe_operand(argument, schema)?
171    {
172        return Ok(None);
173    }
174    let Some(condition) = indicator_condition(&sum.params.args[0]) else {
175        return Ok(None);
176    };
177    if !safe_predicate(condition, schema)?
178        || !columns_are_from_scan(argument, input)?
179        || !columns_are_from_scan(condition, input)?
180    {
181        return Ok(None);
182    }
183    Ok(split_conjunction(condition)
184        .into_iter()
185        .find_map(|predicate| extremum_bound(predicate, argument, extremum, &argument_type)))
186}
187
188fn indicator_condition(expr: &Expr) -> Option<&Expr> {
189    let Expr::Case(case) = expr else {
190        return None;
191    };
192    let [(condition, contribution)] = case.when_then_expr.as_slice() else {
193        return None;
194    };
195    (case.expr.is_none()
196        && integer_literal(contribution) == Some(1)
197        && case.else_expr.as_deref().and_then(integer_literal) == Some(0))
198    .then_some(condition)
199}

201fn extremum_bound(
202    predicate: &Expr,
203    argument: &Expr,
204    extremum: &AggregateFunction,
205    argument_type: &DataType,
206) -> Option<Expr> {
207    let Expr::BinaryExpr(comparison) = predicate else {
208        return None;
209    };
210    let (literal, operator) = if comparison.left.as_ref() == argument {
211        (comparison.right.as_ref(), comparison.op)
212    } else if comparison.right.as_ref() == argument {
213        (comparison.left.as_ref(), comparison.op.swap()?)
214    } else {
215        return None;
216    };
217    let Expr::Literal(bound, _) = literal else {
218        return None;
219    };
220    if bound.is_null() || &bound.data_type() != argument_type {
221        return None;
222    }
223    let preserves_extremum = if extremum.func == min_udaf() {
224        matches!(operator, Operator::Lt | Operator::LtEq)
225    } else {
226        matches!(operator, Operator::Gt | Operator::GtEq)
227    };
228    preserves_extremum.then(|| predicate.clone())
229}

231fn integer_literal(expr: &Expr) -> Option<i64> {
232    match expr {
233        Expr::Literal(ScalarValue::Int64(integer), _) => *integer,
234        Expr::Literal(ScalarValue::Int32(integer), _) => integer.map(i64::from),
235        _ => None,
236    }
237}

239/// Rejects explicitly volatile, fallible or subquery expressions. Column
240/// references additionally require input-plan provenance checks.
241fn safe_predicate(expr: &Expr, schema: &DFSchema) -> Result<bool> {
242    match expr {
243        Expr::BinaryExpr(binary) => Ok(matches!(
244            binary.op,
245            Operator::And
246                | Operator::Or
247                | Operator::Eq
248                | Operator::NotEq
249                | Operator::Lt
250                | Operator::LtEq
251                | Operator::Gt
252                | Operator::GtEq
253        ) && safe_predicate(&binary.left, schema)?
254            && safe_predicate(&binary.right, schema)?),
255        Expr::IsNull(expr) | Expr::IsNotNull(expr) => safe_operand(expr, schema),
256        _ => safe_operand(expr, schema),
257    }
258}
259
260fn safe_operand(expr: &Expr, schema: &DFSchema) -> Result<bool> {
261    match expr {
262        Expr::Column(_) | Expr::Literal(..) => Ok(true),
263        // Timestamp-to-Int64 preserves its native unit and cannot fail. Keep
264        // the cast intact rather than assuming timestamps are nanoseconds.
265        Expr::Cast(cast) => Ok(cast.field.data_type() == &DataType::Int64
266            && matches!(cast.expr.as_ref(), Expr::Column(_))
267            && matches!(cast.expr.get_type(schema)?, DataType::Timestamp(..))),
268        _ => Ok(false),
269    }
270}

272/// A projected column can hide division, a fallible cast or a volatile UDF.
273/// PushDownFilter may inline it below the WHERE that made evaluation safe.
274fn columns_are_from_scan(expr: &Expr, input: &LogicalPlan) -> Result<bool> {
275    for column in expr.column_refs() {
276        if !column_is_from_scan(column, input)? {
277            return Ok(false);
278        }
279    }
280    Ok(true)
281}
282
283fn column_is_from_scan(column: &Column, input: &LogicalPlan) -> Result<bool> {
284    let index = input.schema().index_of_column(column)?;
285    match input {
286        LogicalPlan::TableScan(_) => Ok(true),
287        LogicalPlan::Filter(filter) => column_is_from_scan(column, &filter.input),
288        LogicalPlan::SubqueryAlias(alias) => column_at_is_from_scan(index, &alias.input),
289        LogicalPlan::Projection(projection) => {
290            let expr = match &projection.expr[index] {
291                Expr::Alias(alias) => alias.expr.as_ref(),
292                expr => expr,
293            };
294            Ok(safe_operand(expr, projection.input.schema())?
295                && columns_are_from_scan(expr, &projection.input)?)
296        }
297        LogicalPlan::Join(join) => {
298            match (
299                join.left.schema().has_column(column),
300                join.right.schema().has_column(column),
301            ) {
302                (true, false) => column_is_from_scan(column, &join.left),
303                (false, true) => column_is_from_scan(column, &join.right),
304                _ => Ok(false),
305            }
306        }
307        LogicalPlan::Union(union) => {
308            for input in &union.inputs {
309                if !column_at_is_from_scan(index, input)? {
310                    return Ok(false);
311                }
312            }
313            Ok(true)
314        }
315        // Unknown plan nodes may synthesize columns or hide their origin.
316        _ => Ok(false),
317    }
318}
319
320fn column_at_is_from_scan(index: usize, input: &LogicalPlan) -> Result<bool> {
321    let (qualifier, field) = input.schema().qualified_field(index);
322    column_is_from_scan(&Column::new(qualifier.cloned(), field.name()), input)
323}
324
325fn already_filtered(input: &LogicalPlan, bound: &Expr) -> bool {
326    match input {
327        LogicalPlan::Filter(filter) => {
328            split_conjunction(&filter.predicate).contains(&bound)
329                || already_filtered(&filter.input, bound)
330        }
331        LogicalPlan::TableScan(scan) => scan
332            .filters
333            .iter()
334            .any(|predicate| split_conjunction(predicate).contains(&bound)),
335        _ => false,
336    }
337}

432    async fn analyzed(ctx: &SessionContext, sql: &str) -> LogicalPlan {
433        let state = ctx.state();
434        let plan = state.create_logical_plan(sql).await.unwrap();
435        let plan = state
436            .analyzer()
437            .execute_and_check(plan, state.config_options(), |_, _| {})
438            .unwrap();
439        Optimizer::with_rules(vec![Arc::new(SimplifyExpressions::new())])
440            .optimize(plan, &OptimizerContext::new(), |_, _| {})
441            .unwrap()
442    }
443
444    fn optimize(plan: LogicalPlan) -> LogicalPlan {
445        Optimizer::with_rules(vec![Arc::new(InferAggregateInputBoundsRule)])
446            .optimize(plan, &OptimizerContext::new(), |_, _| {})
447            .unwrap()
448    }
449
450    fn aggregate_input_bound(plan: &LogicalPlan) -> Option<&Expr> {
451        if let LogicalPlan::Aggregate(aggregate) = plan
452            && let LogicalPlan::Filter(filter) = aggregate.input.as_ref()
453        {
454            return Some(&filter.predicate);
455        }
456        plan.inputs().into_iter().find_map(aggregate_input_bound)
457    }
