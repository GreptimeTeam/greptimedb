15//! Compiles a PromQL AST into a [`NativeTree`].
16//!
17//! Each expression kind binds to one node, recursively: a selector to `Input` → `Instant`, a
18//! range function to `Window` over raw samples or over an evaluated subquery on its own grid,
19//! functions to `Map` or `Vector`, operators to `Binary` or `Set`, aggregators to `Aggregate` or
20//! `Vector`.
21//! Constant arguments are folded by [`constant_value`]. Any other shape is a
22//! [`DataFusionError::NotImplemented`] that quotes the accepted shapes.
23//!
24//! Aggregates are bound as [`AggregateStage::Complete`]: the caller must feed every contribution to
25//! a group into one execution, or split the aggregation into `Partial` and `Final`.
26//!
27//! Tree legality is checked by [`NativeTree::try_new`], not here. An `Interleaved` scan ordering is
28//! refused at the entry because no node can consume interleaved series.

66#[derive(Clone, Copy, Debug, PartialEq, Eq)]
67pub enum SelectorTiming {
68    Instant { lookback_ms: i64 },
69    Window { range_ms: i64 },
70}
71
72#[derive(Clone, Debug)]
73pub struct SelectorRequirement {
74    pub metric: Option<String>,
75    pub matchers: Matchers,
76    pub offset_ms: i64,
77    pub at_ms: Option<i64>,
78    pub grid_start_ms: i64,
79    pub grid_end_ms: i64,
80    pub timing: SelectorTiming,
81}
82
83impl SelectorRequirement {
84    /// Returns `(start_ms, end_ms, include_start)` for the scan hull covering this selector.
85    ///
86    /// The end is inclusive; the start is inclusive only for a zero-lookback instant selector.
87    /// `grid_end_ms` is the grid's last point, not the statement's end.
88    pub fn raw_scan_bounds(&self) -> Result<(i64, i64, bool)> {
89        let end = self
90            .at_ms
91            .unwrap_or(self.grid_end_ms)
92            .checked_sub(self.offset_ms)
93            .ok_or_else(|| DataFusionError::Plan("selector scan end overflows".into()))?;
94        let lower = self
95            .at_ms
96            .unwrap_or(self.grid_start_ms)
97            .checked_sub(self.offset_ms)
98            .ok_or_else(|| DataFusionError::Plan("selector scan start overflows".into()))?;
99        let (duration, inclusive) = match self.timing {
100            SelectorTiming::Instant { lookback_ms } => (lookback_ms, lookback_ms == 0),
101            SelectorTiming::Window { range_ms } => (range_ms, false),
102        };
103        let start = lower
104            .checked_sub(duration)
105            .ok_or_else(|| DataFusionError::Plan("selector scan start overflows".into()))?;
106        Ok((start, end, inclusive))
107    }
108}

164/// Compiles a single-selector `stmt` into a tree that reads `input`, bound as floats.
165///
166/// `input` must be the schema after the selector's `__field__`, `__schema__` and `__database__`
167/// matchers are applied; binding every field of the table folds fields the query did not select.
168/// `ordering` is what the scan guarantees; `Interleaved` is refused. A statement without exactly one
169/// selector is refused; use [`compile_selectors`] for several.
170pub fn compile(
171    stmt: &EvalStmt,
172    input: Arc<SeriesSchema>,
173    ordering: SeriesOrdering,
174) -> Result<NativeTree> {
175    // The planner passes the table's field kind through `compile_selectors_with_grid`.
176    let compiled = compile_selectors(
177        stmt,
178        &|_| Ok(Arc::clone(&input)),
179        ordering,
180        Some(FieldKind::Float),
181    )?;
182    if compiled.inputs.len() != 1 {
183        return Err(not_implemented(&format!(
184            "`compile` with {} selector inputs; use `compile_selectors` to bind each input",
185            compiled.inputs.len()
186        )));
187    }
188    Ok(compiled.tree)
189}

191fn data_free_value_field(stmt: &EvalStmt) -> Arc<str> {
192    fn visit(expr: &PromExpr, targets: &mut Vec<String>) {
193        match expr {
194            PromExpr::Aggregate(aggregate) => {
195                if aggregate.op.id() == token::T_COUNT_VALUES
196                    && let Some(param) = &aggregate.param
197                    && let PromExpr::StringLiteral(value) = unparen(param)
198                {
199                    targets.push(value.val.clone());
200                }
201                if let Some(modifier) = &aggregate.modifier {
202                    match modifier {
203                        LabelModifier::Include(labels) | LabelModifier::Exclude(labels) => {
204                            targets.extend(labels.labels.iter().cloned());
205                        }
206                    }
207                }
208                // `topk` and friends take an expression parameter, whose groupings count too: an
209                // aggregate refuses to group by the field it folds.
210                if let Some(param) = &aggregate.param {
211                    visit(param, targets);
212                }
213                visit(&aggregate.expr, targets);
214            }
215            PromExpr::Unary(unary) => visit(&unary.expr, targets),
216            PromExpr::Binary(binary) => {
217                if let Some(modifier) = &binary.modifier {
218                    if let Some(matching) = &modifier.matching {
219                        targets.extend(matching.labels().labels.iter().cloned());
220                    }
221                    match &modifier.card {
222                        VectorMatchCardinality::OneToMany(labels)
223                        | VectorMatchCardinality::ManyToOne(labels) => {
224                            targets.extend(labels.labels.iter().cloned());
225                        }
226                        _ => {}
227                    }
228                }
229                visit(&binary.lhs, targets);
230                visit(&binary.rhs, targets);
231            }
232            PromExpr::Paren(paren) => visit(&paren.expr, targets),
233            PromExpr::Subquery(subquery) => visit(&subquery.expr, targets),
234            PromExpr::Call(call) => call.args.args.iter().for_each(|arg| visit(arg, targets)),
235            _ => {}
236        }
237    }
238    let mut targets = Vec::new();
239    visit(&stmt.expr, &mut targets);
240    if !targets.iter().any(|name| name == VECTOR_FIELD) {
241        return Arc::from(VECTOR_FIELD);
242    }
243    (0usize..)
244        .map(|index| format!("native_value_{index}"))
245        .find(|candidate| !targets.iter().any(|name| name == candidate))
246        .map(Arc::from)
247        .expect("the value field candidate sequence is unbounded")
248}

250/// Compiles a statement that reads no data (a constant, `time()`, `vector(time())`, `hour()`) on
251/// the statement's grid. A statement with a selector is refused.
252pub fn compile_statement(stmt: &EvalStmt) -> Result<NativeTree> {
253    let output_field = data_free_value_field(stmt);
254    let compiled = compile_selectors_with_value_field(
255        stmt,
256        &|_| {
257            Err(not_implemented(&format!(
258                "a selector in `{}`, which is compiled as a statement that reads no data",
259                stmt.expr
260            )))
261        },
262        SeriesOrdering::Contiguous,
263        output_field,
264        // No source, so no source kind; the walk reads it only at an input node.
265        None,
266    )?;
267    Ok(compiled.tree)
268}

348fn compile_selectors_with_grid_and_value_field(
349    stmt: &EvalStmt,
350    bindings: &dyn Fn(&SelectorRequirement) -> Result<Arc<SeriesSchema>>,
351    ordering: SeriesOrdering,
352    value_field: Arc<str>,
353    grid: Option<Arc<EvalGrid>>,
354    source_kind: Option<FieldKind>,
355) -> Result<CompiledTree> {
356    if ordering == SeriesOrdering::Interleaved {
357        return Err(not_implemented(
358            "raw samples whose series interleave, which this compiler cannot consume: a native \
359             evaluation reads one series to its last chunk before the next, so chunks of different \
360             series may not interleave",
361        ));
362    }
363    if !matches!(
364        stmt.expr.value_type(),
365        ValueType::Vector | ValueType::Scalar
366    ) {
367        return Err(not_implemented(&format!(
368            "the expression `{}`, which is a {} where a tree reports evaluated samples or a \
369             scalar",
370            stmt.expr,
371            stmt.expr.value_type()
372        )));
373    }
374    let inputs = RefCell::new(Vec::new());
375    let context = Context {
376        stmt,
377        bindings,
378        ordering,
379        inputs: &inputs,
380        grid: match grid {
381            Some(grid) => grid,
382            None => statement_grid(stmt)?,
383        },
384        value_field,
385        source_kind,
386    };
387    let tree = NativeTree::try_new(compile_expr(&context, &stmt.expr)?)?;
388    let inputs = inputs.into_inner();
389    validate_inputs(tree.root(), &inputs)?;
390    Ok(CompiledTree {
391        tree,
392        inputs: inputs.into(),
393    })
394}

396fn validate_inputs(root: &NativeNode, inputs: &[CompiledInput]) -> Result<()> {
397    let bound = root.inputs();
398    if bound.len() == inputs.len()
399        && bound
400            .iter()
401            .zip(inputs)
402            .enumerate()
403            .all(|(index, (bound, input))| {
404                input.source_index == index
405                    && bound.source_index == input.source_index
406                    && bound.ordering == input.ordering
407                    && bound
408                        .output
409                        .samples()
410                        .is_some_and(|schema| Arc::ptr_eq(schema, &input.schema))
411            })
412    {
413        return Ok(());
414    }
415    Err(DataFusionError::Internal(
416        "a compiled tree's recorded inputs do not match its input nodes".into(),
417    ))
418}

420/// Shared statement, per-selector binding callback, occurrence records, ordering and current grid.
421///
422/// A subquery substitutes its own grid while reusing the statement-level bindings and input list.
423struct Context<'a> {
424    stmt: &'a EvalStmt,
425    bindings: &'a dyn Fn(&SelectorRequirement) -> Result<Arc<SeriesSchema>>,
426    ordering: SeriesOrdering,
427    inputs: &'a RefCell<Vec<CompiledInput>>,
428    grid: Arc<EvalGrid>,
429    value_field: Arc<str>,
430    /// The field kind of the statement's sources; `None` for a statement without sources.
431    source_kind: Option<FieldKind>,
432}
433
434fn compile_expr(context: &Context, expr: &PromExpr) -> Result<NativeNode> {
435    // Constants fold first, so `x * (1 + 2)` binds `3` and no shape below handles a constant
436    // operand.
437    if let Some(value) = constant_value(expr) {
438        return Ok(scalar_node(context, value));
439    }
440    match expr {
441        PromExpr::Paren(paren) => compile_expr(context, &paren.expr),
442        PromExpr::Call(call) => compile_call(context, call),
443        PromExpr::VectorSelector(selector) => instant_selection(context, selector),
444        PromExpr::Unary(unary) => negate(context, &unary.expr),
445        PromExpr::Binary(binary) => binary_operation(context, binary),
446        PromExpr::Aggregate(aggregate) => aggregate_expr(context, aggregate),
447        found => Err(not_implemented(&format!("the expression `{found}`"))),
448    }
449}

469/// Compiles a binary operation, preserving written operand order and vector matching modifiers.
470fn binary_operation(context: &Context, binary: &BinaryExpr) -> Result<NativeNode> {
471    if let Some(op) = set_op(&binary.op) {
472        return set_operation(context, binary, op);
473    }
474    let Some(op) = binary_op(&binary.op) else {
475        return Err(not_implemented(&format!(
476            "the binary operator `{}` in `{binary}`",
477            binary.op
478        )));
479    };
480    if binary.lhs.value_type() == ValueType::Vector && binary.rhs.value_type() == ValueType::Vector
481    {
482        let left = compile_expr(context, &binary.lhs)?;
483        let right = compile_expr(context, &binary.rhs)?;
484        return bind_vector_binary(binary, left, right, op);
485    }
486    // One side is a scalar. Prometheus drops an empty `on()`/`ignoring()` here and rejects one with
487    // labels or a fill at parse time; this refuses all of them.
488    if refuses_modifier(binary.modifier.as_ref()) {
489        return Err(not_implemented(&format!(
490            "the vector matching in `{binary}`, which a constant operand has nothing to match: \
491             `on`, `ignoring`, `group_left` and `group_right` name labels of a vector, and a fill \
492             value fills in a match group that could miss"
493        )));
494    }
495    // Prometheus refuses to parse a scalar–scalar comparison without `bool`. Two constants never
496    // reach here: they fold.
497    if binary.lhs.value_type() == ValueType::Scalar
498        && binary.rhs.value_type() == ValueType::Scalar
499        && matches!(op.kind(), BinaryKind::Comparison(_))
500        && !binary.return_bool()
501    {
502        return Err(not_implemented(&format!(
503            "the binary operation `{binary}` between two scalars, which reports no value: a \
504             comparison of two scalars is written with `bool` to say whether it holds"
505        )));
506    }
507    Ok(broadcast(
508        compile_expr(context, &binary.lhs)?,
509        compile_expr(context, &binary.rhs)?,
510        op,
511        binary.return_bool(),
512    ))
513}

515/// Binds the same vector operation over compiled or already evaluated operands.
516fn bind_vector_binary(
517    binary: &BinaryExpr,
518    left: NativeNode,
519    right: NativeNode,
520    op: BinaryOp,
521) -> Result<NativeNode> {
522    let (Some(lhs), Some(rhs)) = (left.output.samples(), right.output.samples()) else {
523        return Err(DataFusionError::Internal(format!(
524            "the binary operation `{binary}` compiled a vector operand to something other than samples"
525        )));
526    };
527    let output = NativeType::Samples(binary_output_schema(lhs, rhs)?);
528    Ok(NativeNode {
529        expr: NativeExpr::Binary {
530            left: Box::new(left),
531            right: Box::new(right),
532            matching: vector_matching(binary, op)?,
533        },
534        output,
535    })
536}

553/// Compiles `and`, `or` or `unless` between two vectors, matched by the labels the query wrote.
554///
555/// The parser already refuses a scalar operand, `bool` and `group_left` / `group_right` on a set
556/// operator. It does not refuse a fill value, which Prometheus does ("filling in missing series not
557/// allowed for set operators"), so that is refused here along with any cardinality other than the
558/// many-to-many the parser assigns.
559fn set_operation(context: &Context, binary: &BinaryExpr, op: SetOp) -> Result<NativeNode> {
560    let modifier = binary.modifier.as_ref();
561    if modifier.is_some_and(|modifier| {
562        modifier.card != VectorMatchCardinality::ManyToMany
563            || modifier.fill_values != VectorMatchFillValues::default()
564    }) {
565        return Err(not_implemented(&format!(
566            "the modifiers in `{binary}`: a set operator matches many-to-many and takes no fill \
567             value"
568        )));
569    }
570    let labels = matching_labels(modifier);
571    let left = compile_expr(context, &binary.lhs)?;
572    let right = compile_expr(context, &binary.rhs)?;
573    let (Some(lhs), Some(rhs)) = (left.output.samples(), right.output.samples()) else {
574        return Err(DataFusionError::Internal(format!(
575            "the set operation `{binary}` compiled an operand to something other than samples: the \
576             parser only accepts a set operator between two vectors"
577        )));
578    };
579    // Prometheus has one value per sample, and the existing planner refuses a set operator over
580    // several fields, so per-field presence has no reference semantics to follow.
581    if lhs.fields.len() != 1 || rhs.fields.len() != 1 {
582        return Err(not_implemented(&format!(
583            "the set operation `{binary}` between vectors of {} and {} fields: a set operator is \
584             compiled over one field on each side",
585            lhs.fields.len(),
586            rhs.fields.len()
587        )));
588    }
589    // Kept samples pass through unchanged, so the output is the left side's layout.
590    let output = left.output.clone();
591    Ok(NativeNode {
592        expr: NativeExpr::Set {
593            left: Box::new(left),
594            right: Box::new(right),
595            op: BoundSetOp { op, labels },
596        },
597        output,
598    })
599}

601/// Binds `op` between two operands, at least one of them a scalar, in written order.
602///
603/// The output is the vector side's layout, or the left scalar's grid when both are scalars (e.g.
604/// `time() - 60`).
605fn broadcast(left: NativeNode, right: NativeNode, op: BinaryOp, return_bool: bool) -> NativeNode {
606    let output = if matches!(left.output, NativeType::Scalar(_)) {
607        right.output.clone()
608    } else {
609        left.output.clone()
610    };
611    NativeNode {
612        expr: NativeExpr::Binary {
613            left: Box::new(left),
614            right: Box::new(right),
615            matching: scalar_matching(op, return_bool),
616        },
617        output,
618    }
619}

621/// The matching of an operation with a scalar side.
622///
623/// A scalar has no labels, so `labels` and `cardinality` are unused and hold the vector–vector
624/// defaults. `return_bool` is the only modifier that applies.
625fn scalar_matching(op: BinaryOp, return_bool: bool) -> BinaryMatching {
626    BinaryMatching {
627        op,
628        return_bool,
629        labels: MatchingLabels::Ignoring(Vec::new()),
630        cardinality: Cardinality::OneToOne,
631        // A scalar exists at every grid point, so there is no missing side to fill.
632        fill: FillValues::default(),
633    }
634}

636/// The matching a vector–vector operation records.
637///
638/// Without a modifier this is PromQL's default: `ignoring()` and one-to-one. Fill values keep the
639/// sides as written; the kernel swaps them with the operands under `group_right`. `ManyToMany`
640/// belongs to set operators, which [`binary_operation`] dispatches first, so it is refused here.
641fn vector_matching(binary: &BinaryExpr, op: BinaryOp) -> Result<BinaryMatching> {
642    let modifier = binary.modifier.as_ref();
643    let labels = matching_labels(modifier);
644    let cardinality = match modifier.map(|modifier| &modifier.card) {
645        None | Some(VectorMatchCardinality::OneToOne) => Cardinality::OneToOne,
646        Some(VectorMatchCardinality::ManyToOne(labels)) => {
647            Cardinality::ManyToOne(label_names(&labels.labels))
648        }
649        Some(VectorMatchCardinality::OneToMany(labels)) => {
650            Cardinality::OneToMany(label_names(&labels.labels))
651        }
652        Some(VectorMatchCardinality::ManyToMany) => {
653            return Err(not_implemented(&format!(
654                "the many-to-many matching in `{binary}`, which is the cardinality of a set \
655                 operator: it combines two vectors' series, and this node is an arithmetic or \
656                 comparison operation over values"
657            )));
658        }
659    };
660    let fill = modifier.map_or_else(FillValues::default, |modifier| FillValues {
661        lhs: modifier.fill_values.lhs,
662        rhs: modifier.fill_values.rhs,
663    });
664    Ok(BinaryMatching {
665        op,
666        return_bool: binary.return_bool(),
667        labels,
668        cardinality,
669        fill,
670    })
671}

720/// Compiles an aggregation as a [`AggregateStage::Complete`] aggregate node.
721///
722/// `topk`, `bottomk`, `limitk`, `limit_ratio` and `count_values` do not fold a group into one value
723/// and bind as a vector operation instead ([`vector_aggregate`]). A `quantile` whose φ is a
724/// non-constant scalar binds as [`NativeExpr::ParameterizedAggregate`] and reads φ at every step,
725/// as Prometheus does.
726fn aggregate_expr(context: &Context, aggregate: &AggregateExpr) -> Result<NativeNode> {
727    let AggregateExpr {
728        op,
729        expr,
730        param,
731        modifier,
732    } = aggregate;
733    // A `quantile` φ read per step, e.g. `quantile(scalar(foo), v)`.
734    let mut per_point = None;
735    let kind = match op.id() {
736        token::T_SUM => AggregateKind::Sum,
737        token::T_AVG => AggregateKind::Avg,
738        token::T_COUNT => AggregateKind::Count,
739        token::T_MIN => AggregateKind::Min,
740        token::T_MAX => AggregateKind::Max,
741        token::T_GROUP => AggregateKind::Group,
742        token::T_STDDEV => AggregateKind::Stddev,
743        token::T_STDVAR => AggregateKind::Stdvar,
744        token::T_QUANTILE => {
745            let param = param.as_ref().ok_or_else(|| {
746                not_implemented(&format!(
747                    "the aggregate `{op}` in `{aggregate}`, which takes its quantile as its first \
748                     argument"
749                ))
750            })?;
751            // A constant φ folds below. A vector φ is left to `constant_argument` to refuse, as
752            // Prometheus refuses `quantile(foo, v)` at type check.
753            if constant_value(param).is_none() && param.value_type() == ValueType::Scalar {
754                per_point = Some(param);
755            }
756            AggregateKind::Quantile {
757                // Unused when φ is read per step.
758                quantile: match per_point {
759                    Some(_) => 0.0,
760                    None => constant_argument(&op.to_string(), param, 0)?,
761                },
762            }
763        }
764        token::T_TOPK
765        | token::T_BOTTOMK
766        | token::T_LIMITK
767        | token::T_LIMIT_RATIO
768        | token::T_COUNT_VALUES => {
769            return vector_aggregate(context, aggregate);
770        }
771        // Unreachable for the parser's `aggregate_op` rule, which lists exactly the names above.
772        _ => {
773            return Err(not_implemented(&format!(
774                "the aggregate `{op}` in `{aggregate}`"
775            )));
776        }
777    };
778    let input = compile_expr(context, expr)?;
779    // The state layout follows the direct input's kind, not the sources':
780    // `sum(count_over_time(h[5m]))` folds floats built from histograms. An unnameable kind over
781    // histogram sources is refused. Over float sources it is a float: a histogram sample can only
782    // come from a histogram source.
783    let field_kind = match input.output_field_kind(context.source_kind) {
784        Some(kind) => kind,
785        None if context.source_kind == Some(FieldKind::Histogram) => {
786            return Err(not_implemented(&format!(
787                "the aggregate `{op}` in `{aggregate}`, whose input's field kind this compile cannot \
788                 name while its sources carry native histograms: the aggregate's state depends on \
789                 whether the input carries floats or histograms"
790            )));
791        }
792        None => FieldKind::Float,
793    };
794    let output = input.output.clone();
795    let op = BoundAggregate {
796        kind,
797        group: grouping(modifier.as_ref()),
798        stage: AggregateStage::Complete,
799        field_kind,
800    };
801    // Input before parameter: selector occurrence numbers follow this order.
802    let expr = match per_point {
803        Some(parameter) => NativeExpr::ParameterizedAggregate {
804            input: Box::new(input),
805            op,
806            parameter: Box::new(compile_expr(context, parameter)?),
807        },
808        None => NativeExpr::Aggregate {
809            input: Box::new(input),
810            op,
811        },
812    };
813    Ok(NativeNode { expr, output })
814}

816/// Binds `topk`, `bottomk`, `limitk`, `limit_ratio` or `count_values` as a vector operation over
817/// the expression it aggregates.
818///
819/// The parser gives each of them its parameter: a number for the first four, folded to the constant
820/// it means like a `quantile`, and a string for `count_values`, the label it writes the value to.
821fn vector_aggregate(context: &Context, aggregate: &AggregateExpr) -> Result<NativeNode> {
822    let AggregateExpr {
823        op,
824        expr,
825        param,
826        modifier,
827    } = aggregate;
828    let name = op.to_string();
829    let param = param.as_ref().ok_or_else(|| {
830        not_implemented(&format!(
831            "the aggregate `{op}` in `{aggregate}`, which takes a parameter as its first argument"
832        ))
833    })?;
834    let group = grouping(modifier.as_ref());
835    if matches!(op.id(), token::T_TOPK | token::T_BOTTOMK) && constant_value(param).is_none() {
836        let input = compile_expr(context, expr)?;
837        let output = input.output.clone();
838        let parameter = compile_expr(context, param)?;
839        return Ok(NativeNode {
840            expr: NativeExpr::ParameterizedTopK {
841                input: Box::new(input),
842                op: BoundVectorOp::TopK {
843                    k: 0.0,
844                    top: op.id() == token::T_TOPK,
845                    group,
846                },
847                parameter: Box::new(parameter),
848            },
849            output,
850        });
851    }
852    let bound = match op.id() {
853        token::T_TOPK | token::T_BOTTOMK => BoundVectorOp::TopK {
854            k: constant_argument(&name, param, 0)?,
855            top: op.id() == token::T_TOPK,
856            group,
857        },
858        token::T_LIMITK => BoundVectorOp::LimitK {
859            k: constant_argument(&name, param, 0)?,
860            group,
861        },
862        // `limit_ratio` keeps whole series by their label hash, so a grouping changes nothing.
863        token::T_LIMIT_RATIO => BoundVectorOp::LimitRatio {
864            ratio: constant_argument(&name, param, 0)?,
865        },
866        _ => {
867            let PromExpr::StringLiteral(StringLiteral { val }) = unparen(param) else {
868                return Err(not_implemented(&format!(
869                    "the aggregate `{op}` in `{aggregate}`, whose label `{param}` is not a string \
870                     literal"
871                )));
872            };
873            BoundVectorOp::CountValues {
874                label: Arc::from(val.as_str()),
875                group,
876            }
877        }
878    };
879    vector_call(context, &name, bound, expr)
880}

882/// How a `by` or `without` modifier identifies a group, in the tree's own terms.
883///
884/// Labels keep their written order.
885fn grouping(modifier: Option<&LabelModifier>) -> Grouping {
886    fn labels(labels: &[String]) -> Arc<[Arc<str>]> {
887        labels
888            .iter()
889            .map(|label| Arc::<str>::from(label.as_str()))
890            .collect()
891    }
892    match modifier {
893        // No modifier means `by()`: one group.
894        None => Grouping::By(Arc::new([])),
895        Some(LabelModifier::Include(by)) => Grouping::By(labels(&by.labels)),
896        Some(LabelModifier::Exclude(without)) => Grouping::Without(labels(&without.labels)),
897    }
898}

900/// Compiles `timestamp`, projecting selected source times for a selector and evaluation times for
901/// an already evaluated vector.
902fn timestamp_call(context: &Context, args: &[Box<PromExpr>]) -> Result<NativeNode> {
903    if args.len() != 1 {
904        return Err(not_implemented("timestamp expects one argument"));
905    }
906    let argument = unparen(&args[0]);
907    if argument.value_type() != ValueType::Vector {
908        return Err(not_implemented("timestamp argument is not a vector"));
909    }
910    if let PromExpr::VectorSelector(selector) = argument {
911        return instant_selection_of(context, selector, InstantValue::Timestamp);
912    }
913    let input = compile_expr(context, argument)?;
914    let NativeType::Samples(schema) = &input.output else {
915        return Err(not_implemented("timestamp argument is not a vector"));
916    };
917    if schema.fields.len() != 1 {
918        return Err(not_implemented(
919            "timestamp() over a multi-value input is not supported natively",
920        ));
921    }
922    let output = input.output.clone();
923    Ok(NativeNode {
924        expr: NativeExpr::Timestamp {
925            input: Box::new(input),
926        },
927        output,
928    })
929}

931/// Compiles `info(v[, {selector}])` as its base `v` and the instant selection of the info series it
932/// reads.
933///
934/// The info series are selected as Prometheus' `fetchInfoSeries` does: by the info metric's name
935/// and the data matchers, at the reference time the selectors of `v` share ([`info_input`]), with
936/// the statement's lookback. The kernel matches base series to them.
937fn info_call(context: &Context, args: &[Box<PromExpr>]) -> Result<NativeNode> {
938    let InfoInput { selector, kernel } = info_input(context.stmt, args)?;
939    let base = compile_expr(context, &args[0])?;
940    if base
941        .output
942        .samples()
943        .is_none_or(|schema| schema.eval_grid.is_none())
944    {
945        return Err(DataFusionError::Internal(format!(
946            "info's first argument `{}` compiled to something other than an evaluated vector: the \
947             parser only accepts a vector there",
948            args[0]
949        )));
950    }
951    let info = instant_selection_of(context, &selector, InstantValue::InfoTimestamp)?;
952    let fields = info
953        .output
954        .samples()
955        .map_or(0, |schema| schema.fields.len());
956    // Prometheus reads one sample per info series and point, and the existing planner has no
957    // `info` to follow for several fields.
958    if fields != 1 {
959        return Err(not_implemented(&format!(
960            "info over the info metric `{selector}` with {fields} fields: an info series is read as \
961             one field"
962        )));
963    }
964    let output = base.output.clone();
965    Ok(NativeNode {
966        expr: NativeExpr::Info {
967            base: Box::new(base),
968            info: Box::new(info),
969            selector: kernel,
970        },
971        output,
972    })
973}

976/// combines by.

979    pub selector: VectorSelector,

983/// Splits the arguments of `info` into the selector of its info series and the kernel's selector.
984///
985/// Shared by the compiler and the planner, which bind the info scan at the same position. The
986/// metric must be `target_info` or one `__name__="..."` matcher, since one info metric is one
987/// table. `__field__`, `__schema__` and `__database__` matchers are refused.
988pub(crate) fn info_input(stmt: &EvalStmt, args: &[Box<PromExpr>]) -> Result<InfoInput> {
989    let (base, labels) = match args {
990        [base] => (base, None),
991        [base, labels] => (base, Some(labels)),
992        _ => {
993            return Err(not_implemented(&format!(
994                "a call to `info` with {} argument(s)",
995                args.len()
996            )));
997        }
998    };
999    let matchers: &[Matcher] = match labels.map(|labels| labels.as_ref()) {
1000        None => &[],
1001        Some(PromExpr::VectorSelector(VectorSelector {
1002            name: None,
1003            matchers,
1004            offset: None,
1005            at: None,
1006        })) if matchers.or_matchers.is_empty() => &matchers.matchers,
1007        Some(other) => {
1008            return Err(not_implemented(&format!(
1009                "`info` whose second argument `{other}` is not one label selector without a name \
1010                 or modifiers"
1011            )));
1012        }
1013    };
1014    let (names, data): (Vec<&Matcher>, Vec<&Matcher>) = matchers
1015        .iter()
1016        .partition(|matcher| matcher.name == "__name__");
1017    let metric = match names.as_slice() {
1018        [] => "target_info".to_string(),
1019        [name] if name.op == ParserMatchOp::Equal => name.value.clone(),
1020        _ => {
1021            return Err(not_implemented(
1022                "`info` over the info metrics a regular expression, a negative matcher or several \
1023                 `__name__` matchers select, which can be several tables",
1024            ));
1025        }
1026    };
1027    if data.iter().any(|matcher| {
1028        matches!(
1029            matcher.name.as_str(),
1030            "__field__" | "__schema__" | "__database__"
1031        )
1032    }) {
1033        return Err(not_implemented(
1034            "`info` with a `__field__`, `__schema__` or `__database__` matcher, which selects \
1035             what is read rather than a data label",
1036        ));
1037    }
1038    let (at, offset_ms) = info_reference(stmt, base)?;
1039    let offset = match offset_ms {
1040        0 => None,
1041        ms if ms > 0 => Some(Offset::Pos(Duration::from_millis(ms.unsigned_abs()))),
1042        ms => Some(Offset::Neg(Duration::from_millis(ms.unsigned_abs()))),
1043    };
1044    let kernel = InfoSelector {
1045        name_matchers: names
1046            .iter()
1047            .map(|matcher| (match_op(&matcher.op), Arc::from(matcher.value.as_str())))
1048            .collect(),
1049        data_labels: data
1050            .iter()
1051            .map(|matcher| Arc::from(matcher.name.as_str()))
1052            .collect(),
1053        keep_unmatched: data.iter().all(|matcher| matcher.is_match("")),
1054    };
1055    Ok(InfoInput {
1056        selector: VectorSelector {
1057            name: Some(metric),
1058            matchers: Matchers::new(data.into_iter().cloned().collect()),
1059            offset,
1060            at,
1061        },
1062        kernel,
1063    })
1064}

1066/// The `@` anchor and offset the info series are selected at: those of the first selector of
1067/// `expr` when every vector-producing path of it reaches a selector with the same effective
1068/// reference time, and none otherwise, as Prometheus `infoSeriesSelectTimestampAndOffset` decides.
1069///
1070/// A selector inside subqueries adds their offsets until one of them anchors it with `@`. A path
1071/// that produces a vector without a selector, such as `vector(1)`, is evaluated at the step itself,
1072/// so it leaves no shared reference.
1073fn info_reference(stmt: &EvalStmt, expr: &PromExpr) -> Result<(Option<AtModifier>, i64)> {
1074    #[derive(Default)]
1075    struct Walk {
1076        /// The first selector's anchor, its resolved time and its offset.
1077        first: Option<(Option<AtModifier>, Option<i64>, i64)>,
1078        uniform: bool,
1079        reference_free: bool,
1080    }
1081
1082    fn inspect<'a>(
1083        stmt: &EvalStmt,
1084        expr: &'a PromExpr,
1085        subqueries: &mut Vec<&'a SubqueryExpr>,
1086        walk: &mut Walk,
1087    ) -> Result<bool> {
1088        if !matches!(expr.value_type(), ValueType::Vector | ValueType::Matrix) {
1089            return Ok(false);
1090        }
1091        let selector = match expr {
1092            PromExpr::VectorSelector(selector) => Some(selector),
1093            PromExpr::MatrixSelector(MatrixSelector { vs, .. }) => Some(vs),
1094            _ => None,
1095        };
1096        if let Some(selector) = selector {
1097            let mut at = selector.at.clone();
1098            let mut offset = selector_offset_ms(&selector.offset)?;
1099            for subquery in subqueries.iter().rev() {
1100                if at.is_some() {
1101                    break;
1102                }
1103                offset = offset
1104                    .checked_add(selector_offset_ms(&subquery.offset)?)
1105                    .ok_or_else(|| DataFusionError::Plan("selector offset overflows".into()))?;
1106                at = subquery.at.clone();
1107            }
1108            let at_ms = selector_anchor_ms(&at, stmt)?;
1109            match &walk.first {
1110                None => walk.first = Some((at, at_ms, offset)),
1111                Some((_, first_at, first_offset)) => {
1112                    let same = match (first_at, at_ms) {
1113                        (None, None) => *first_offset == offset,
1114                        (Some(first_at), Some(at)) => {
1115                            i128::from(*first_at) - i128::from(*first_offset)
1116                                == i128::from(at) - i128::from(offset)
1117                        }
1118                        _ => false,
1119                    };
1120                    walk.uniform &= same;
1121                }
1122            }
1123            return Ok(true);
1124        }
1125        let children: Vec<&PromExpr> = match expr {
1126            PromExpr::Paren(paren) => vec![&paren.expr],
1127            PromExpr::Unary(unary) => vec![&unary.expr],
1128            PromExpr::Binary(binary) => vec![&binary.lhs, &binary.rhs],
1129            PromExpr::Aggregate(aggregate) => aggregate
1130                .param
1131                .iter()
1132                .chain([&aggregate.expr])
1133                .map(|child| child.as_ref())
1134                .collect(),
1135            // The second argument of a nested `info` is selector syntax, not a vector it evaluates.
1136            PromExpr::Call(call) if call.func.name == "info" => call
1137                .args
1138                .args
1139                .iter()
1140                .take(1)
1141                .map(|arg| arg.as_ref())
1142                .collect(),
1143            PromExpr::Call(call) => call.args.args.iter().map(|arg| arg.as_ref()).collect(),
1144            PromExpr::Subquery(subquery) => {
1145                subqueries.push(subquery);
1146                let found = inspect(stmt, &subquery.expr, subqueries, walk);
1147                subqueries.pop();
1148                let found = found?;
1149                walk.reference_free |= !found;
1150                return Ok(found);
1151            }
1152            other => {
1153                return Err(not_implemented(&format!(
1154                    "`info` over `{other}`, whose reference time is not derived here"
1155                )));
1156            }
1157        };
1158        let mut found = false;
1159        for child in children {
1160            found |= inspect(stmt, child, subqueries, walk)?;
1161        }
1162        walk.reference_free |= !found;
1163        Ok(found)
1164    }
1165
1166    let mut walk = Walk {
1167        uniform: true,
1168        ..Default::default()
1169    };
1170    inspect(stmt, expr, &mut Vec::new(), &mut walk)?;
1171    match walk.first {
1172        Some((at, _, offset)) if walk.uniform && !walk.reference_free => Ok((at, offset)),
1173        _ => Ok((None, 0)),
1174    }
1175}

1180        ParserMatchOp::NotEqual => MatchOp::NotEqual,

1186/// Compiles a function call by looking its name up in each function table; an unknown name is
1187/// refused.
1188fn compile_call(context: &Context, call: &Call) -> Result<NativeNode> {
1189    if call.func.name == "timestamp" {
1190        return timestamp_call(context, &call.args.args);
1191    }
1192    if call.func.name == "info" {
1193        return info_call(context, &call.args.args);
1194    }
1195    if call.func.name == ABSENT_OVER_TIME {
1196        return absent_over_time_call(context, &call.args.args);
1197    }
1198    let Call { func, args } = call;
1199    if let Some(node) = scalar_call(context, func.name, &args.args)? {
1200        return Ok(node);
1201    }
1202    if let Some(signature) = range_signature(func.name) {
1203        return window_call(context, func.name, signature, &args.args);
1204    }
1205    if let Some(node) = parameterized_histogram_call(context, func.name, &args.args)? {
1206        return Ok(node);
1207    }
1208    if let Some((op, vector)) = vector_op(func.name, &args.args)? {
1209        return vector_call(context, func.name, op, vector);
1210    }
1211    if let Some(node) = scalar_function_call(context, func.name, &args.args)? {
1212        return Ok(node);
1213    }
1214    if let Some(node) = parameterized_call(context, func.name, &args.args)? {
1215        return Ok(node);
1216    }
1217    if let Some((op, vector)) = math_op(func.name, &args.args)? {
1218        return map_call(context, func.name, MapOp::Math(op), vector);
1219    }
1220    if let Some(function) = histogram_function(func.name) {
1221        let [vector] = args.args.as_slice() else {
1222            return Err(not_implemented(&format!(
1223                "a call to `{}` with {} argument(s), where it takes its vector",
1224                func.name,
1225                args.args.len()
1226            )));
1227        };
1228        return map_call(context, func.name, MapOp::Histogram(function), vector);
1229    }
1230    if let Some((rewrite, vector)) = label_rewrite(func.name, &args.args)? {
1231        return map_call(context, func.name, MapOp::Label(Arc::new(rewrite)), vector);
1232    }
1233    Err(not_implemented(&format!(
1234        "a call to `{}`, which has no native kernel",
1235        func.name
1236    )))
1237}

1239/// Binds `time()`, `vector(s)` and a date function called without an argument, or
1240/// returns `None` when the call is none of them.
1241///
1242/// A date function without an argument is `f(vector(time()))` in Prometheus, but reads the
1243/// evaluation time truncated to whole seconds, so it binds as `EvalTime { date }` rather than over
1244/// `time()`.
1245fn scalar_call(
1246    context: &Context,
1247    name: &str,
1248    args: &[Box<PromExpr>],
1249) -> Result<Option<NativeNode>> {

1251    let eval_time = |date| NativeNode {
1252        expr: NativeExpr::EvalTime { date },
1253        output: NativeType::Scalar(grid()),
1254    };

1256        ("time", []) => eval_time(None),
1257        ("vector", [scalar]) => vector_of(
1258            compile_expr(context, scalar)?,
1259            &grid(),
1260            &context.value_field,
1261        ),
1262        ("scalar", [vector]) => {
1263            if vector.value_type() != ValueType::Vector {
1264                return Err(not_implemented("scalar() requires a vector argument"));

1268                return Err(not_implemented("scalar() requires an evaluated vector"));
1269            };
1270            let Some(input_grid) = schema.eval_grid.clone() else {

1274                expr: NativeExpr::ScalarOf {
1275                    input: Box::new(input),
1276                },
1277                output: NativeType::Scalar(input_grid),

1279        }
1280        (name, []) => match math_signature(name) {
1281            Some(MathSignature::Unary(MathOp::Date(date))) => {
1282                vector_of(eval_time(Some(date)), &grid(), &context.value_field)

1285        },

1287    };

1291/// Binds `max_of(a, b)` or `min_of(a, b)`, whose arguments are both scalars, or returns `None` for
1292/// any other call.
1293///
1294/// Prometheus types both as `(scalar, scalar) -> scalar`. The first argument is the map's input and

1297    context: &Context,
1298    name: &str,
1299    args: &[Box<PromExpr>],

1303        "min_of" => ParameterizedOp::MinOf,

1354/// Binds `round`, `clamp`, `clamp_min` or `clamp_max` whose parameters are not all constants, or
1355/// returns `None` for any other call. Prometheus reads each parameter at every step, so each one is
1356/// bound as the scalar node it compiles to; with constant parameters the call is a [`MathOp`].
1357fn parameterized_call(
1358    context: &Context,
1359    name: &str,
1360    args: &[Box<PromExpr>],
1361) -> Result<Option<NativeNode>> {
1362    let op = match name {
1363        "round" => ParameterizedOp::Round,
1364        "clamp" => ParameterizedOp::Clamp,
1365        "clamp_min" => ParameterizedOp::ClampMin,
1366        "clamp_max" => ParameterizedOp::ClampMax,
1367        _ => return Ok(None),
1368    };
1369    let Some((vector, parameters)) = args.split_first() else {
1370        return Ok(None);
1371    };

1375            .all(|parameter| constant_value(parameter).is_some())

1382             function takes a vector",

1385    }
1386    let input = compile_expr(context, vector)?;
1387    let parameters = parameters
1388        .iter()
1389        .map(|parameter| compile_expr(context, parameter))
1390        .collect::<Result<Vec<_>>>()?;
1391    let output = input.output.clone();
1392    Ok(Some(NativeNode {
1393        expr: NativeExpr::ParameterizedMap {

1599/// The vector operation a sort, `absent` or a classic histogram function binds, and the vector it
1600/// reads, or `None` when `name` is none of them.
1601fn vector_op<'a>(
1602    name: &str,
1603    arguments: &'a [Box<PromExpr>],
1604) -> Result<Option<(BoundVectorOp, &'a PromExpr)>> {
1605    let order = match name {
1606        "sort" => Some(SortOrder::Asc),
1607        "sort_desc" => Some(SortOrder::Desc),
1608        "sort_by_label" | "sort_by_label_desc" => None,
1609        "histogram_quantile" | "histogram_fraction" => {
1610            let Some((vector, constants)) = arguments.split_last() else {
1611                return Err(not_implemented(&format!(
1612                    "a call to `{name}` with no argument"
1613                )));
1614            };
1615            let function = match constants {
1616                [quantile] if name == "histogram_quantile" => {
1617                    HistogramFunction::Quantile(constant_argument(name, quantile, 0)?)
1618                }
1619                [lower, upper] if name == "histogram_fraction" => HistogramFunction::Fraction {
1620                    lower: constant_argument(name, lower, 0)?,
1621                    upper: constant_argument(name, upper, 1)?,
1622                },
1623                _ => {
1624                    return Err(not_implemented(&format!(
1625                        "a call to `{name}` with {} argument(s)",
1626                        arguments.len()
1627                    )));
1628                }
1629            };
1630            return Ok(Some((BoundVectorOp::Histogram(function), vector)));
1631        }
1632        "absent" => {
1633            let [vector] = arguments else {
1634                return Err(not_implemented(&format!(
1635                    "a call to `absent` with {} argument(s)",
1636                    arguments.len()
1637                )));
1638            };
1639            let labels = absent_labels_of(vector)?;
1640            return Ok(Some((BoundVectorOp::Absent { labels }, vector)));
1641        }
1642        _ => return Ok(None),
1643    };
1644    let Some((vector, strings)) = arguments.split_first() else {
1645        return Err(not_implemented(&format!(
1646            "a call to `{name}` with no argument"
1647        )));
1648    };
1649    let order = match order {
1650        Some(order) if strings.is_empty() => order,
1651        Some(_) => {
1652            return Err(not_implemented(&format!(
1653                "a call to `{name}` with {} argument(s)",
1654                arguments.len()
1655            )));
1656        }
1657        None => {
1658            let labels = strings
1659                .iter()
1660                .map(|argument| match unparen(argument) {
1661                    PromExpr::StringLiteral(StringLiteral { val }) => Ok(Arc::from(val.as_str())),
1662                    other => Err(not_implemented(&format!(
1663                        "a call to `{name}` whose argument `{other}` is not a string literal"
1664                    ))),
1665                })
1666                .collect::<Result<Vec<_>>>()?;
1667            if name == "sort_by_label" {
1668                SortOrder::ByLabel(labels)
1669            } else {
1670                SortOrder::ByLabelDesc(labels)
1671            }
1672        }
1673    };
1674    Ok(Some((BoundVectorOp::Sort(order), vector)))
1675}

1677/// The labels `absent(expr)` and `absent_over_time(m[d])` answer with, as Prometheus
1678/// `createLabelsForAbsentFunction` derives them: from a vector or range selector's matchers, and
1679/// none for any other expression. `__field__`, `__schema__` and `__database__` are not labels.
1680fn absent_labels_of(expr: &PromExpr) -> Result<SeriesLabels> {
1681    let (selector, alternatives) = match unparen(expr) {
1682        PromExpr::VectorSelector(selector) => (&selector.matchers, &selector.matchers.or_matchers),
1683        PromExpr::MatrixSelector(MatrixSelector { vs, .. }) => {
1684            (&vs.matchers, &vs.matchers.or_matchers)
1685        }
1686        _ => return absent_labels(&[]),
1687    };
1688    if !alternatives.is_empty() {
1689        return Err(not_implemented(&format!(
1690            "`absent` or `absent_over_time` over `{expr}`, whose matchers are alternatives"
1691        )));
1692    }
1693    let matchers = selector
1694        .matchers
1695        .iter()
1696        .filter(|matcher| {
1697            !matches!(
1698                matcher.name.as_str(),
1699                "__field__" | "__schema__" | "__database__"
1700            )
1701        })
1702        .map(|matcher| LabelMatcher {
1703            name: Arc::from(matcher.name.as_str()),
1704            op: match_op(&matcher.op),
1705            value: Arc::from(matcher.value.as_str()),
1706            matches_empty: matcher.is_match(""),
1707        })
1708        .collect::<Vec<_>>();
1709    absent_labels(&matchers)
1710}

1712/// Binds `function` over the range selector `vs[range]` into a window over the raw input.
1713fn window_selection(
1714    context: &Context,
1715    function: RangeFunction,
1716    vs: &VectorSelector,
1717    range: &Duration,
1718) -> Result<NativeNode> {
1719    let range_ms = range_millis(range.as_millis())?;
1720    let offset_ms = selector_offset_ms(&vs.offset)?;
1721    let anchor_ms = selector_anchor_ms(&vs.at, context.stmt)?;
1722    let output_grid = Arc::clone(&context.grid);
1723
1724    let (grid_start_ms, grid_end_ms) = grid_endpoints(&output_grid)?;
1725    let (source_index, schema) = bind_selector(
1726        context,
1727        SelectorRequirement {
1728            metric: vs.name.clone(),
1729            matchers: vs.matchers.clone(),
1730            offset_ms,
1731            at_ms: anchor_ms,
1732            grid_start_ms,
1733            grid_end_ms,
1734            timing: SelectorTiming::Window { range_ms },
1735        },
1736    )?;
1737    let output = NativeType::Samples(Arc::new(SeriesSchema {
1738        fields: Arc::clone(&schema.fields),
1739        eval_grid: Some(Arc::clone(&output_grid)),
1740    }));
1741    let raw = NativeType::Samples(Arc::clone(&schema));
1742
1743    Ok(NativeNode {
1744        expr: NativeExpr::Window {
1745            input: Box::new(NativeNode {
1746                expr: NativeExpr::Input(BoundInput {
1747                    output: raw.clone(),
1748                    source_index,
1749                    ordering: context.ordering,
1750                }),
1751                output: raw,
1752            }),
1753            op: BoundWindow {
1754                function,
1755                window: RangeWindow {
1756                    range_ms,
1757                    offset_ms,
1758                    anchor_ms,
1759                    modifier: None,
1760                },
1761                input_grid: None,
1762                output_grid,
1763            },
1764        },

1769/// The step a subquery names when it names none, as in `x[5m:]`.

1771/// Prometheus uses the global evaluation interval, one minute by default, not the statement's step.

1775/// The grid a subquery's inner expression is evaluated on, and the window that reads it.
1776///
1777/// The compiler and the planner both derive it here so they cannot disagree on the grid.
1778pub(crate) struct SubqueryGrid {
1779    /// The interval the range function folds, and how far back from the anchor it reads.
1780    pub(crate) window: RangeWindow,
1781    /// The grid `subquery.expr` is evaluated on.
1782    pub(crate) grid: Arc<EvalGrid>,

1785/// Derives the [`SubqueryGrid`] of `subquery`, whose outer statement is evaluated on `outer`.
1786///
1787/// The child grid has one point per `step` over the interval the subquery's window reads. `None`
1788/// when that interval contains no step boundary. Errors on a zero step or an out-of-range duration.
1789pub(crate) fn subquery_child_grid(
1790    stmt: &EvalStmt,
1791    outer: &EvalGrid,
1792    subquery: &SubqueryExpr,
1793) -> Result<Option<SubqueryGrid>> {
1794    let anchor_ms = selector_anchor_ms(&subquery.at, stmt)?;
1795    let range_ms = range_millis(subquery.range.as_millis())?;
1796    let step_ms = match subquery.step {
1797        Some(step) => millis(step.as_millis()).ok_or_else(|| {
1798            DataFusionError::Plan(format!(
1799                "the subquery's step of {}ms is out of range",
1800                step.as_millis()
1801            ))
1802        })?,
1803        None => DEFAULT_SUBQUERY_STEP_MS,
1804    };
1805    if step_ms <= 0 {
1806        return Err(DataFusionError::Plan(format!(
1807            "the subquery's step is {step_ms}ms, and a step has to be longer than zero"
1808        )));
1809    }
1810    let offset_ms = selector_offset_ms(&subquery.offset)?;
1811    let window = RangeWindow {
1812        range_ms,
1813        offset_ms,
1814        anchor_ms,
1815        modifier: None,
1816    };
1817    let Some(spec) = subquery_grid(outer.spec(), window, step_ms)? else {
1818        return Ok(None);
1819    };
1820    // Keep subquery grids deferred: selector binding needs only their spec endpoints, not a
1821    // materialized timestamp vector proportional to the range.
1822    let grid = Arc::new(EvalGrid::try_deferred(spec).map_err(grid_error)?);
1823    Ok(Some(SubqueryGrid { window, grid }))
1824}

1827/// grid. The window's `input_grid` is that child grid; its output grid is the enclosing one.

1831    function: RangeFunction,
1832    subquery: &SubqueryExpr,

1839            expr.value_type()

1841    }

1843        window,

1883        },

1895    context: &Context,
1896    selector: &VectorSelector,
1897    value: InstantValue,
1898) -> Result<NativeNode> {

1912            grid_end_ms,

1922        fields: Arc::clone(&schema.fields),

1932                    ordering: context.ordering,

1963fn grid_endpoints(grid: &EvalGrid) -> Result<(i64, i64)> {
1964    let spec = grid.spec();
1965    let end = i128::from(spec.start_ms) + i128::from(spec.len - 1) * i128::from(spec.step_ms);
1966    let end = i64::try_from(end)
1967        .map_err(|_| DataFusionError::Internal("evaluation grid endpoint overflows".into()))?;
1968    Ok((spec.start_ms, end))

1971/// The grid the statement is evaluated on: `(end - start) / step + 1` points, the last of which
1972/// does not pass `end`.
1973///
1974/// The planner reads it here so the tree and its binding use the same grid.
1975pub(crate) fn statement_grid(stmt: &EvalStmt) -> Result<Arc<EvalGrid>> {
1976    let start_ms = signed_millis_since_epoch(stmt.start)?;
1977    let end_ms = signed_millis_since_epoch(stmt.end)?;
1978    let step_ms = millis(stmt.interval.as_millis()).ok_or_else(|| {
1979        DataFusionError::Plan(format!(
1980            "the statement's evaluation step of {}ms is out of range",
1981            stmt.interval.as_millis()
1982        ))
1983    })?;
1984    let spec = GridSpec::try_from_range(start_ms, end_ms, step_ms).map_err(|error| {

1986            "the statement's grid of {start_ms}ms..{end_ms}ms at {step_ms}ms per step is invalid: \

1990    Ok(Arc::new(EvalGrid::try_deferred(spec).map_err(grid_error)?))

2058/// Where a range function's matrix sits among its arguments and how its constants bind.
2059#[derive(Clone, Copy)]
2060enum RangeSignature {
2061    /// A function that reads only its range selector, as in `rate(x[5m])`.
2062    Selector(RangeFunction),
2063    /// One constant, then the range selector, as in `quantile_over_time(0.9, x[5m])`.
2064    ConstantBeforeSelector(fn(f64) -> RangeFunction),
2065    /// The range selector, then one constant, as in `predict_linear(x[5m], 300)`.
2066    ConstantAfterSelector(fn(f64) -> RangeFunction),
2067    /// The range selector, then two constants, as in
2068    /// `double_exponential_smoothing(x[5m], 0.5, 0.1)`.
2069    TwoConstantsAfterSelector(fn(f64, f64) -> RangeFunction),
2070}
2071
2072impl RangeSignature {
2073    /// The position of the range selector among the call's arguments.
2074    fn selector_index(self) -> usize {
2075        match self {
2076            Self::Selector(_)
2077            | Self::ConstantAfterSelector(_)
2078            | Self::TwoConstantsAfterSelector(_) => 0,
2079            Self::ConstantBeforeSelector(_) => 1,
2080        }
2081    }
2082
2083    /// How many constants the function reads.
2084    fn constants(self) -> usize {
2085        match self {
2086            Self::Selector(_) => 0,
2087            Self::ConstantBeforeSelector(_) | Self::ConstantAfterSelector(_) => 1,
2088            Self::TwoConstantsAfterSelector(_) => 2,
2089        }
2090    }
2091
2092    fn arity(self) -> usize {
2093        self.constants() + 1
2094    }
2095
2096    /// The call's arguments, for the arity rejection.
2097    fn arguments(self) -> &'static str {
2098        match self {
2099            Self::Selector(_) => "its range selector",
2100            Self::ConstantBeforeSelector(_) => "a number and its range selector",
2101            Self::ConstantAfterSelector(_) => "its range selector and a number",
2102            Self::TwoConstantsAfterSelector(_) => "its range selector and two numbers",
2103        }
2104    }
2105
2106    /// Binds the function from arguments that already passed the arity check.
2107    fn bind(self, name: &str, args: &[Box<PromExpr>]) -> Result<RangeFunction> {
2108        Ok(match self {
2109            Self::Selector(function) => function,
2110            Self::ConstantBeforeSelector(bind) => bind(constant_argument(name, &args[0], 0)?),
2111            Self::ConstantAfterSelector(bind) => bind(constant_argument(name, &args[1], 1)?),
2112            Self::TwoConstantsAfterSelector(bind) => bind(
2113                constant_argument(name, &args[1], 1)?,
2114                constant_argument(name, &args[2], 2)?,
2115            ),
2116        })
2117    }
2118}

2235/// The math kernel `name` denotes and the vector it maps, or `None` when `name` is not in
2236/// [`math_signature`]. `arguments` is the whole argument list, vector first; the constants after
2237/// it must fold.
2238fn math_op<'a>(
2239    name: &str,
2240    arguments: &'a [Box<PromExpr>],
2241) -> Result<Option<(MathOp, &'a PromExpr)>> {
2242    let Some(signature) = math_signature(name) else {
2243        return Ok(None);
2244    };
2245    let Some((vector, parameters)) = arguments.split_first() else {
2246        return Err(arity_error(name, signature, 0));
2247    };
2248    let given = parameters.len() + 1;
2249    if given < signature.required() + 1 || given > signature.parameters() + 1 {
2250        return Err(arity_error(name, signature, given));
2251    }
2252    let op = match signature {
2253        MathSignature::Unary(op) => op,
2254        MathSignature::OneConstant(bind) => bind(constant_argument(name, &parameters[0], 1)?),
2255        MathSignature::DefaultedConstant { default, bind } => bind(match parameters.first() {
2256            Some(parameter) => constant_argument(name, parameter, 1)?,
2257            None => default,
2258        }),
2259        MathSignature::TwoConstants(bind) => bind(
2260            constant_argument(name, &parameters[0], 1)?,
2261            constant_argument(name, &parameters[1], 2)?,
2262        ),
2263    };
2264    Ok(Some((op, vector)))
2265}

2352/// Whether a binary operation carries a modifier this compiler refuses to bind.
2353///
2354/// Matching, a non-default cardinality or a fill value has no meaning with a scalar operand. Shared
2355/// by [`constant_value`] and [`binary_operation`] so the fold never drops a modifier the dispatch
2356/// refuses.
2357fn refuses_modifier(modifier: Option<&BinModifier>) -> bool {
2358    modifier.is_some_and(|modifier| {
2359        modifier.matching.is_some()

2365/// The number `expr` folds to, or `None` when it depends on data or evaluation time.
2366///
2367/// Folds number literals, parentheses, negation, `pi()`, and arithmetic or `bool` comparisons of
2368/// two folded operands, using the kernel's [`BinaryOp::kind`] so the fold and the kernel agree.
2369pub(crate) fn constant_value(expr: &PromExpr) -> Option<f64> {
2370    match expr {
2371        PromExpr::NumberLiteral(NumberLiteral { val }) => Some(*val),
2372        PromExpr::Paren(paren) => constant_value(&paren.expr),
2373        PromExpr::Unary(unary) => constant_value(&unary.expr).map(|value| -value),
2374        PromExpr::Binary(binary) => {
2375            // Left unfolded so `binary_operation` refuses the modifier instead of the fold
2376            // dropping it, e.g. the fill in `x * (1 + fill(0) 2)`.
2377            if refuses_modifier(binary.modifier.as_ref()) {
2378                return None;
2379            }
2380            let left = constant_value(&binary.lhs)?;
2381            let right = constant_value(&binary.rhs)?;
2382            match binary_op(&binary.op)?.kind() {
2383                BinaryKind::Arithmetic(op) => Some(op.calculate(left, right)),
2384                // Without `bool`, `1 > 2` is a parse error in Prometheus; left unfolded, the
2385                // dispatch refuses it.
2386                BinaryKind::Comparison(op) => binary
2387                    .return_bool()
2388                    .then(|| f64::from(op.apply(left, right))),
2389            }
2390        }
2391        PromExpr::Call(call) if call.func.name == "pi" && call.args.args.is_empty() => {
2392            Some(std::f64::consts::PI)
2393        }
2394        _ => None,
2395    }
2396}
