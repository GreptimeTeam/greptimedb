60/// Planner for converting merge sort logical plan to physical plan.
61///
62/// `MergeSortExec` always represents the distributed merge stage. It declares
63/// the required input ordering to DataFusion, so `EnforceSorting` inserts a
64/// `SortExec` below it when the input `MergeScanExec` cannot preserve per-region
65/// ordering, for example when one output partition may merge multiple region
66/// streams.
67pub struct MergeSortExtensionPlanner {}
68
69impl MergeSortExtensionPlanner {
70    fn ordering(
71        planner: &dyn PhysicalPlanner,
72        session: &dyn Session,
73        planning_ctx: &PhysicalPlanningContext,
74        merge_sort: &MergeSortLogicalPlan,
75    ) -> Result<LexOrdering> {
76        let ordering = merge_sort
77            .expr
78            .iter()
79            .map(|sort_expr| {
80                let physical_expr = planner.create_physical_expr(
81                    &sort_expr.expr,
82                    merge_sort.input.schema(),
83                    session,
84                    planning_ctx,
85                )?;
86                Ok(PhysicalSortExpr::new(
87                    physical_expr,
88                    SortOptions {
89                        descending: !sort_expr.asc,
90                        nulls_first: sort_expr.nulls_first,
91                    },
92                ))
93            })
94            .collect::<Result<Vec<_>>>()?;
95
96        LexOrdering::new(ordering).ok_or_else(|| {
97            DataFusionError::Internal(
98                "Expect MergeSort to have non-empty sort expressions".to_string(),
99            )
100        })
101    }
102}

104#[async_trait]
105impl ExtensionPlanner for MergeSortExtensionPlanner {
106    async fn plan_extension(
107        &self,
108        planner: &dyn PhysicalPlanner,
109        node: &dyn UserDefinedLogicalNode,
110        _logical_inputs: &[&LogicalPlan],
111        physical_inputs: &[Arc<dyn ExecutionPlan>],
112        session: &dyn Session,
113        planning_ctx: &PhysicalPlanningContext,
114    ) -> Result<Option<Arc<dyn ExecutionPlan>>> {
115        if let Some(merge_sort) = node.as_any().downcast_ref::<MergeSortLogicalPlan>() {
116            if let LogicalPlan::Extension(ext) = &merge_sort.input.as_ref()
117                && ext
118                    .node
119                    .as_any()
120                    .downcast_ref::<MergeScanLogicalPlan>()
121                    .is_some()
122            {
123                let input = physical_inputs.first().cloned().ok_or_else(|| {
124                    DataFusionError::Internal(
125                        "Expect MergeSort to have one physical input".to_string(),
126                    )
127                })?;
128                if input.downcast_ref::<MergeScanExec>().is_none() {
129                    return Err(DataFusionError::Internal(format!(
130                        "Expect MergeSort's input is a MergeScanExec, found {:?}",
131                        physical_inputs
132                    )));
133                }
134
135                let ordering = Self::ordering(planner, session, planning_ctx, merge_sort)?;
136                Ok(Some(Arc::new(MergeSortExec::new(
137                    ordering,
138                    input,
139                    merge_sort.fetch,
140                ))))
141            } else {
142                Ok(None)
143            }
144        } else {
145            Ok(None)
146        }
147    }
148}

173#[async_trait]
174impl ExtensionPlanner for DistExtensionPlanner {
175    async fn plan_extension(
176        &self,
177        planner: &dyn PhysicalPlanner,
178        node: &dyn UserDefinedLogicalNode,
179        _logical_inputs: &[&LogicalPlan],
180        _physical_inputs: &[Arc<dyn ExecutionPlan>],
181        session: &dyn Session,
182        _planning_ctx: &PhysicalPlanningContext,
183    ) -> Result<Option<Arc<dyn ExecutionPlan>>> {
184        let Some(merge_scan) = node.as_any().downcast_ref::<MergeScanLogicalPlan>() else {
185            return Ok(None);
186        };
187
188        let input_plan = merge_scan.input();
189        let fallback = |logical_plan| async move {
190            let optimized_plan = self.optimize_input_logical_plan(session, logical_plan)?;
191            planner
192                .create_physical_plan(&optimized_plan, session)
193                .await
194                .map(Some)
195        };
196
197        if merge_scan.is_placeholder() {
198            // ignore placeholder
199            return fallback(input_plan).await;
200        }
201
202        let optimized_plan = input_plan;
203        let direct_raw = matches!(input_plan, LogicalPlan::Extension(extension)
204            if extension.node.as_any().is::<promql::extension_plan::RawSeriesSource>());
205        let Some(table_name) = Self::extract_full_table_name(input_plan)? else {
206            if direct_raw {
207                return Err(DataFusionError::Plan(
208                    "raw series source has no region route table".into(),
209                ));
210            }
211            // no relation found in input plan, going to execute them locally
212            return fallback(optimized_plan).await;
213        };
214
215        let (regions, series_disjoint) = match self.get_regions(&table_name, input_plan).await {
216            Ok(regions) => regions,
217            Err(error) if direct_raw => {
218                return Err(error);
219            }
220            // Existing non-raw plans may still execute locally when no peers are found.
221            Err(_) => return fallback(optimized_plan).await,
222        };
223
224        // TODO(ruihang): generate different execution plans for different variant merge operation
225        let schema = merge_scan.schema().as_arrow();
226        let session_state = session
227            .as_any()
228            .downcast_ref::<SessionState>()
229            .ok_or_else(|| {
230                DataFusionError::Internal(
231                    "MergeScan requires a SessionState for physical planning".to_string(),
232                )
233            })?;
234        let query_ctx = session_state
235            .config()
236            .get_extension()
237            .unwrap_or_else(QueryContext::arc);
238        let merge_scan_plan = MergeScanExec::new(
239            session_state,
240            table_name,
241            regions,
242            input_plan.clone(),
243            schema,
244            self.region_query_handler.clone(),
245            query_ctx,
246            session.config().target_partitions(),
247            merge_scan.remote_dyn_filter_producer_id(),
248            self.enable_per_region_metrics,
249        )?
250        .with_raw_series_disjoint(series_disjoint)?;
251        Ok(Some(Arc::new(merge_scan_plan) as _))
252    }
253}

255impl DistExtensionPlanner {
256    /// Extract fully resolved table name from logical plan
257    fn extract_full_table_name(plan: &LogicalPlan) -> Result<Option<TableName>> {
258        let mut extractor = TableScanExtractor::default();
259        let _ = plan.visit(&mut extractor)?;
260        Ok(extractor.table_name)
261    }
262
263    /// Selects the regions of the ordinary distributed plan.
264    async fn get_regions(
265        &self,
266        table_name: &TableName,
267        logical_plan: &LogicalPlan,
268    ) -> Result<(Vec<RegionId>, bool)> {
269        let mut extractor = TableScanExtractor::default();
270        let _ = logical_plan.visit(&mut extractor)?;
271        // Resolving by name again could bind an authorized scan to a replacement table.
272        let captured_table = extractor.captured_table;
273        let table = match &captured_table {
274            Some(table) => table.clone(),
275            None => self
276                .catalog_manager
277                .table(
278                    &table_name.catalog_name,
279                    &table_name.schema_name,
280                    &table_name.table_name,
281                    None,
282                )
283                .await
284                .context(CatalogSnafu)?
285                .with_context(|| TableNotFoundSnafu {
286                    table: table_name.to_string(),
287                })?,
288        };
289
290        let table_info = table.table_info();
291        let (physical_table_id, physical_table_route) = self
292            .partition_rule_manager
293            .find_physical_table_route_with_id(table_info.table_id())
294            .await
295            .context(PartitionRuleManagerSnafu)?;
296        let all_regions = physical_table_route
297            .region_routes
298            .iter()
299            .map(|r| RegionId::new(table_info.table_id(), r.region.id.region_number()))
300            .collect::<Vec<_>>();
301        let logical_partition_columns = partition_column_types(&table_info);
302        let partition_columns = logical_partition_columns
303            .iter()
304            .map(|(name, _)| name.clone())
305            .collect::<Vec<_>>();
306        debug!(
307            "DistExtensionPlanner: loaded table partition metadata, table: {}, table_id: {}, partition_key_indices: {:?}, partition_columns: {:?}, all_regions: {:?}",
308            table_name,
309            table_info.table_id(),
310            table_info.meta.partition_key_indices,
311            partition_columns,
312            all_regions,
313        );
314        let partitioned_by_table_id = physical_table_id != table_info.table_id()
315            && table_info.meta.engine == METRIC_ENGINE_NAME
316            && table.partition_rules().is_some_and(|rules| {
317                rules
318                    .extra_phy_cols_not_in_logical_table
319                    .iter()
320                    .any(|name| name == DATA_SCHEMA_TABLE_ID_COLUMN_NAME)
321            });
322        let regions = 'pruning: {
323            let partition_expressions = if partition_columns.is_empty() {
324                Vec::new()
325            } else {
326                PredicateExtractor::extract_partition_expressions(logical_plan, &partition_columns)
327                    .unwrap_or_else(|err| {
328                        debug!(
329                            "Failed to extract partition predicates for table {table_name}: {err}"
330                        );
331                        Vec::new()
332                    })
333            };
334
335            if partition_expressions.is_empty()
336                && (!partitioned_by_table_id || all_regions.len() == 1)
337            {
338                break 'pruning all_regions;
339            }
340            let partitions = match create_partitions_from_region_routes(
341                table_info.table_id(),
342                &physical_table_route.region_routes,
343            ) {
344                Ok(partitions) => partitions,
345                Err(err) => {
346                    debug!("Failed to load partitions for table {table_name}: {err}");
347                    break 'pruning all_regions;
348                }
349            };
350
351            let (mut regions, partitions) = if partitioned_by_table_id {
352                let Some(candidates) =
353                    simplify_metric_partitions(table_info.table_id(), partitions)
354                else {
355                    debug!("Could not simplify partition rules for table {table_name}");
356                    break 'pruning all_regions;
357                };
358                candidates
359            } else {
360                (Vec::new(), partitions)
361            };
362
363            // Rules that simplify to true impose no constraints on the SQL predicates.
364            // Send only the residual rules to the constraint pruner.
365            if !partitions.is_empty()
366                && !partition_expressions.is_empty()
367                && let Some(column_types) = self
368                    .partition_column_types_for_pruning(
369                        table_name,
370                        table_info.as_ref(),
371                        physical_table_id,
372                        &partition_expressions,
373                    )
374                    .await
375            {
376                match ConstraintPruner::prune_regions(
377                    &partition_expressions,
378                    &partitions,
379                    column_types,
380                ) {
381                    Ok(pruned) => regions.extend(pruned),
382                    Err(err) => {
383                        debug!("Failed to prune partitions for table {table_name}: {err}");
384                        regions.extend(partitions.iter().map(|partition| partition.id));
385                    }
386                }
387            } else {
388                regions.extend(partitions.iter().map(|partition| partition.id));
389            }
390
391            debug!(
392                "Region pruning for table {table_name}: {} -> {} regions",
393                all_regions.len(),
394                regions.len()
395            );
396            regions
397        };
398
399        // A selected region is trivially disjoint. For several regions the proof must
400        // use the captured scan and the physical route's table metadata, not a logical
401        // metric table's possibly incomplete partition columns.
402        let series_disjoint = if regions.len() <= 1 {
403            true
404        } else if captured_table.is_none() {
405            false
406        } else {
407            let physical_info = if physical_table_id == table_info.table_id() {
408                Some(table_info.clone())
409            } else {
410                self.catalog_manager
411                    .table_info_by_id(physical_table_id)
412                    .await
413                    .ok()
414                    .flatten()
415            };
416            physical_info.is_some_and(|info| {
417                physical_regions_disjoint(&info, &physical_table_route.region_routes, &regions)
418            })
419        };
420        Ok((regions, series_disjoint))
421    }
422
423    /// Resolves the partition-column types that are safe to use for region pruning.
424    ///
425    /// A logical metric table may not contain every physical partition column, either for backward
426    /// compatibility or because its physical table was repartitioned after the logical table was
427    /// created. Predicate extraction must remain bounded by the logical schema, while pruning
428    /// needs the physical datatypes to evaluate route expressions. Any lookup failure or
429    /// logical/physical datatype mismatch returns `None`, preserving the current candidate regions.
430    async fn partition_column_types_for_pruning(
431        &self,
432        table_name: &TableName,
433        logical_table_info: &TableInfo,
434        physical_table_id: u32,
435        partition_expressions: &[PartitionExpr],
436    ) -> Option<HashMap<String, ConcreteDataType>> {
437        let physical_partition_columns = if physical_table_id == logical_table_info.table_id() {
438            partition_column_types(logical_table_info)
439        } else {
440            match self
441                .catalog_manager
442                .table_info_by_id(physical_table_id)
443                .await
444            {
445                Ok(Some(physical_table_info)) => {
446                    partition_column_types(physical_table_info.as_ref())
447                }
448                Ok(None) => {
449                    debug!(
450                        "DistExtensionPlanner: physical table info not found for table {} (id: {})",
451                        table_name, physical_table_id
452                    );
453                    return None;
454                }
455                Err(err) => {
456                    debug!(
457                        "DistExtensionPlanner: failed to load physical table info for table {} (id: {}): {}",
458                        table_name, physical_table_id, err
459                    );
460                    return None;
461                }
462            }
463        };
464        let physical_column_types = physical_partition_columns
465            .into_iter()
466            .collect::<HashMap<_, _>>();
467        let logical_column_types = partition_column_types(logical_table_info)
468            .into_iter()
469            .collect::<HashMap<_, _>>();
470        let mut predicate_column_names = std::collections::HashSet::new();
471        for expression in partition_expressions {
472            expression.collect_column_names(&mut predicate_column_names);
473        }
474        if predicate_column_names
475            .iter()
476            .any(|name| logical_column_types.get(name) != physical_column_types.get(name))
477        {
478            debug!(
479                "DistExtensionPlanner: logical and physical partition metadata mismatch for table {} (physical id: {})",
480                table_name, physical_table_id
481            );
482            return None;
483        }
484
485        Some(physical_column_types)
486    }

488    /// Input logical plan is analyzed. Thus only call logical optimizer to optimize it.
489    fn optimize_input_logical_plan(
490        &self,
491        session: &dyn Session,
492        plan: &LogicalPlan,
493    ) -> Result<LogicalPlan> {
494        expr_depth::validate_plan(plan)?;
495        let session_state = session
496            .as_any()
497            .downcast_ref::<SessionState>()
498            .ok_or_else(|| {
499                DataFusionError::Internal(
500                    "MergeScan requires a SessionState for logical optimization".to_string(),
501                )
502            })?;
503
504        session_state
505            .optimizer()
506            .optimize(plan.clone(), session_state, |_, _| {})
507    }

510enum SimplifiedPartitionExpr {
511    Constant(bool),
512    Residual(PartitionExpr),
513}
514
515/// Returns unconditional regions and regions with residual partition rules.
516/// Incomplete or inconsistent routing rules leave the original route in use.
517fn simplify_metric_partitions(
518    table_id: TableId,
519    partitions: Vec<PartitionInfo>,
520) -> Option<(Vec<RegionId>, Vec<PartitionInfo>)> {
521    let table_id = Value::UInt32(table_id);
522    let hash = IcebergHash::evaluate_value(&table_id).ok()?;
523    let mut unconditional = Vec::new();
524    let mut residual = Vec::new();
525    for PartitionInfo { id, partition_expr } in partitions {
526        match simplify_metric_partition_expr(partition_expr?, &table_id, &hash)? {
527            SimplifiedPartitionExpr::Constant(false) => {}
528            SimplifiedPartitionExpr::Constant(true) => unconditional.push(id),
529            SimplifiedPartitionExpr::Residual(expr) => residual.push(PartitionInfo {
530                id,
531                partition_expr: Some(expr),
532            }),
533        }
534    }
535    // A complete set of physical rules covers every logical table ID.
536    if unconditional.is_empty() && residual.is_empty() {
537        return None;
538    }
539    Some((unconditional, residual))
540}
541
542fn simplify_metric_partition_expr(
543    expr: PartitionExpr,
544    table_id: &Value,
545    hash: &Value,
546) -> Option<SimplifiedPartitionExpr> {
547    use SimplifiedPartitionExpr::{Constant, Residual};
548
549    // Route deserialization canonicalizes comparisons to dimension op literal.
550    match (expr.op, *expr.lhs, *expr.rhs) {
551        (op @ (RestrictedOp::And | RestrictedOp::Or), Operand::Expr(lhs), Operand::Expr(rhs)) => {
552            let lhs = simplify_metric_partition_expr(lhs, table_id, hash)?;
553            let rhs = simplify_metric_partition_expr(rhs, table_id, hash)?;
554            let absorbing = op == RestrictedOp::Or;
555            match (lhs, rhs) {
556                (Constant(lhs), Constant(rhs)) => {
557                    Some(Constant(if absorbing { lhs || rhs } else { lhs && rhs }))
558                }
559                (Constant(value), residual) | (residual, Constant(value)) => {
560                    Some(if value == absorbing {
561                        Constant(value)
562                    } else {
563                        residual
564                    })
565                }
566                (Residual(lhs), Residual(rhs)) => Some(Residual(PartitionExpr {
567                    lhs: Box::new(Operand::Expr(lhs)),
568                    op,
569                    rhs: Box::new(Operand::Expr(rhs)),
570                })),
571            }
572        }
573        (RestrictedOp::And | RestrictedOp::Or, _, _) => None,
574        (op, dimension @ (Operand::Column(_) | Operand::IcebergHash(_)), Operand::Value(bound)) => {
575            let known = match &dimension {
576                Operand::Column(name) if name == DATA_SCHEMA_TABLE_ID_COLUMN_NAME => table_id,
577                Operand::IcebergHash(function)
578                    if function.column() == DATA_SCHEMA_TABLE_ID_COLUMN_NAME =>
579                {
580                    hash
581                }
582                _ => return Some(Residual(PartitionExpr::new(dimension, op, bound.into()))),
583            };
584
585            // DDL types the bound to the dimension. Partition routing orders NULL
586            // before non-NULL values. A mismatched bound cannot safely be folded.
587            if !matches!(bound, Value::Null) && known.data_type() != bound.data_type() {
588                return None;
589            }
590            let ordering = known.cmp(&bound);
591            let matches = match op {
592                RestrictedOp::Eq => ordering.is_eq(),
593                RestrictedOp::NotEq => !ordering.is_eq(),
594                RestrictedOp::Lt => ordering.is_lt(),
595                RestrictedOp::LtEq => ordering.is_le(),
596                RestrictedOp::Gt => ordering.is_gt(),
597                RestrictedOp::GtEq => ordering.is_ge(),
598                RestrictedOp::And | RestrictedOp::Or => return None,
599            };
600            Some(Constant(matches))
601        }
602        _ => None,
603    }
604}

606/// Prove that every physical routing column belongs to the complete raw primary key.
607/// Unknown indices, missing route expressions, or route columns not represented in physical
608/// partition metadata cannot establish disjointness. Reserved-looking names have no special status.
609fn physical_regions_disjoint(
610    table_info: &TableInfo,
611    routes: &[common_meta::rpc::router::RegionRoute],
612    regions: &[RegionId],
613) -> bool {
614    let columns = table_info.meta.schema.column_schemas();
615    let partition_keys = &table_info.meta.partition_key_indices;
616    // Routing by any nonempty subset of the complete key keeps equal complete keys together.
617    // The reverse inclusion is not required: (host, rack) partitioned by host is disjoint.
618    if partition_keys.is_empty()
619        || table_info.meta.primary_key_indices.is_empty()
620        || partition_keys.iter().any(|index| {
621            columns.get(*index).is_none_or(|column| {
622                column.is_time_index() || !table_info.meta.primary_key_indices.contains(index)
623            })
624        })
625        || table_info
626            .meta
627            .primary_key_indices
628            .iter()
629            .any(|index| columns.get(*index).is_none())
630    {
631        return false;
632    }
633    let partition_names = partition_keys
634        .iter()
635        .map(|index| columns[*index].name.as_str())
636        .collect::<std::collections::HashSet<_>>();
637    for region in regions {
638        let mut matching = routes
639            .iter()
640            .filter(|route| route.region.id.region_number() == region.region_number());
641        let Some(route) = matching.next() else {
642            return false;
643        };
644        if matching.next().is_some() {
645            return false;
646        }
647        let Ok(Some(expression)) = PartitionExpr::from_json_str(&route.region.partition_expr())
648        else {
649            return false;
650        };
651        let mut names = std::collections::HashSet::new();
652        expression.collect_column_names(&mut names);
653        if names.is_empty()
654            || names
655                .iter()
656                .any(|name| !partition_names.contains(name.as_str()))
657        {
658            return false;
659        }
660    }
661    true
662}

664fn partition_column_types(table_info: &TableInfo) -> Vec<(String, ConcreteDataType)> {
665    table_info
666        .meta
667        .partition_columns()

669        .collect()

672/// Extract the scan name and captured table identity from a logical plan.
673#[derive(Default)]
674struct TableScanExtractor {
675    pub table_name: Option<TableName>,
676    captured_table: Option<TableRef>,
677}
678
679impl TreeNodeVisitor<'_> for TableScanExtractor {
680    type Node = LogicalPlan;
681
682    fn f_down(&mut self, node: &Self::Node) -> Result<TreeNodeRecursion> {
683        match node {
684            LogicalPlan::TableScan(scan) => {
685                if let Some(source) = scan.source.downcast_ref::<DefaultTableSource>()
686                    && let Some(provider) = source
687                        .table_provider
688                        .downcast_ref::<DfTableProviderAdapter>()
689                {
690                    if provider.table().table_type() == TableType::Base {
691                        self.captured_table = Some(provider.table());
692                        let info = provider.table().table_info();
693                        self.table_name = Some(TableName::new(
694                            info.catalog_name.clone(),
695                            info.schema_name.clone(),
696                            info.name.clone(),
697                        ));
698                    }
699                    return Ok(TreeNodeRecursion::Stop);
700                }
701                match &scan.table_name {
702                    TableReference::Full {
703                        catalog,
704                        schema,
705                        table,
706                    } => {
707                        self.table_name = Some(TableName::new(
708                            catalog.to_string(),
709                            schema.to_string(),
710                            table.to_string(),
711                        ));
712                        Ok(TreeNodeRecursion::Stop)
713                    }
714                    // TODO(ruihang): Maybe the following two cases should not be valid
715                    TableReference::Partial { schema, table } => {
716                        self.table_name = Some(TableName::new(
717                            DEFAULT_CATALOG_NAME.to_string(),
718                            schema.to_string(),
719                            table.to_string(),
720                        ));
721                        Ok(TreeNodeRecursion::Stop)
722                    }
723                    TableReference::Bare { table } => {
724                        self.table_name = Some(TableName::new(
725                            DEFAULT_CATALOG_NAME.to_string(),
726                            DEFAULT_SCHEMA_NAME.to_string(),
727                            table.to_string(),
728                        ));
729                        Ok(TreeNodeRecursion::Stop)
730                    }
731                }
732            }
733            _ => Ok(TreeNodeRecursion::Continue),
734        }
735    }
736}

876    async fn planner_and_plan(
877        physical_partition_keys: Vec<usize>,
878        expressions: Vec<Option<PartitionExpr>>,
879    ) -> (DistExtensionPlanner, LogicalPlan, TableName) {
880        let logical_info = table_info(LOGICAL_TABLE_ID, "logical", &["host"], vec![0]);
881        let physical_info = table_info(
882            PHYSICAL_TABLE_ID,
883            "physical",
884            &["host", "rack", DATA_SCHEMA_TABLE_ID_COLUMN_NAME],
885            physical_partition_keys,
886        );
887        let hidden_columns: Vec<_> = physical_info
888            .meta
889            .partition_column_names()
890            .filter(|name| {
891                logical_info
892                    .meta
893                    .schema
894                    .column_index_by_name(name)
895                    .is_none()
896            })
897            .cloned()
898            .collect();
899        let partition_rules = (!hidden_columns.is_empty()).then_some(PartitionRules {
900            extra_phy_cols_not_in_logical_table: hidden_columns,
901        });
902        let logical_table =
903            DistTable::table_partitioned(Arc::new(logical_info.clone()), partition_rules);
904        let physical_table = EmptyTable::from_table_info(&physical_info);
905        let catalog_manager = MemoryCatalogManager::with_default_setup();
906        for table in [&logical_table, &physical_table] {
907            let info = table.table_info();
908            catalog_manager
909                .register_table_sync(RegisterTableRequest {
910                    catalog: info.catalog_name.clone(),
911                    schema: info.schema_name.clone(),
912                    table_name: info.name.clone(),
913                    table_id: info.table_id(),
914                    table: table.clone(),
915                })
916                .unwrap();
917        }
918
919        let backend = Arc::new(MemoryKvBackend::default());
920        let metadata_manager = TableMetadataManager::new(backend.clone());
921        let routes = expressions
922            .into_iter()
923            .enumerate()
924            .map(|(index, expression)| region_route(index as u32 + 1, expression))
925            .collect();
926        metadata_manager
927            .create_table_metadata(
928                physical_info,
929                TableRouteValue::physical(routes),
930                HashMap::new(),
931            )
932            .await
933            .unwrap();
934        metadata_manager
935            .create_table_metadata(
936                logical_info,
937                TableRouteValue::logical(PHYSICAL_TABLE_ID),
938                HashMap::new(),
939            )
940            .await
941            .unwrap();
942
943        let table_route_cache = Arc::new(new_table_route_cache(
944            "planner-test-routes".to_string(),
945            CacheBuilder::new(16).build(),
946            backend.clone(),
947        ));
948        let partition_info_cache = Arc::new(new_partition_info_cache(
949            "planner-test-partitions".to_string(),
950            CacheBuilder::new(16).build(),
951            table_route_cache.clone(),
952        ));
953        let partition_rule_manager = Arc::new(PartitionRuleManager::new(
954            backend,
955            table_route_cache,
956            partition_info_cache,
957        ));
958        let (resolved_physical_id, physical_route) = partition_rule_manager
959            .find_physical_table_route_with_id(PHYSICAL_TABLE_ID)
960            .await
961            .unwrap();
962        assert_eq!(PHYSICAL_TABLE_ID, resolved_physical_id);
963        let (resolved_logical_id, logical_route) = partition_rule_manager
964            .find_physical_table_route_with_id(LOGICAL_TABLE_ID)
965            .await
966            .unwrap();
967        assert_eq!(PHYSICAL_TABLE_ID, resolved_logical_id);
968        assert_eq!(physical_route.region_routes, logical_route.region_routes);
969        let catalog_manager: CatalogManagerRef = catalog_manager;
970        let planner = DistExtensionPlanner::new(
971            catalog_manager,
972            partition_rule_manager,
973            Arc::new(UnusedRegionQueryHandler),
974            false,
975        );
976        let table_source = Arc::new(DefaultTableSource::new(Arc::new(
977            DfTableProviderAdapter::new(logical_table),
978        )));
979        let plan = LogicalPlanBuilder::scan_with_filters("logical", table_source, None, vec![])
980            .unwrap()
981            .filter(df_col("host").eq(lit("a")))
982            .unwrap()
983            .build()
984            .unwrap();
985        (
986            planner,
987            plan,
988            TableName::new(DEFAULT_CATALOG_NAME, DEFAULT_SCHEMA_NAME, "logical"),
989        )
990    }
