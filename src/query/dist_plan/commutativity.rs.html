38/// Whether `node` is a node the analyzer runs where the scan it reads is planned.
39///
40/// A region only ever runs a fused node over its own scan, and the analyzer puts one there by
41/// classifying it as commutative - which is also the only placement a node over a bare scan has,
42/// because a bare scan is never shipped across a region boundary. A planner that has to commit to
43/// a plan before the analyzer runs asks this to know whether that placement exists: `false` says
44/// every input of this node has to be a source that names its region.
45pub(crate) fn region_local_node(node: &dyn UserDefinedLogicalNode) -> bool {
46    matches!(
47        Categorizer::check_extension_plan(node, &AliasMapping::default()),
48        Commutativity::Commutative
49    )
50}

52pub struct StepTransformAction {
53    extra_parent_plans: Vec<LogicalPlan>,
54    new_child_plan: Option<LogicalPlan>,
55}

57/// generate the upper aggregation plan that will execute on the frontend.
58/// Basically a logical plan resembling the following:
59/// Projection:
60///     Aggregate:
61///
62/// from Aggregate
63///
64/// The upper Projection exists sole to make sure parent plan can recognize the output
65/// of the upper aggregation plan.
66pub fn step_aggr_to_upper_aggr(
67    aggr_plan: &LogicalPlan,
68) -> datafusion_common::Result<StepTransformAction> {
69    let LogicalPlan::Aggregate(input_aggr) = aggr_plan else {
70        return Err(datafusion_common::DataFusionError::Plan(
71            "step_aggr_to_upper_aggr only accepts Aggregate plan".to_string(),
72        ));
73    };
74    if !is_all_aggr_exprs_steppable(&input_aggr.aggr_expr) {
75        return Err(datafusion_common::DataFusionError::NotImplemented(format!(
76            "Some aggregate expressions are not steppable in [{}]",
77            input_aggr
78                .aggr_expr
79                .iter()
80                .map(|e| e.to_string())
81                .collect::<Vec<_>>()
82                .join(", ")
83        )));
84    }
85
86    let step_aggr_plan = StateMergeHelper::split_aggr_node(input_aggr.clone())?;
87
88    // TODO(discord9): remove duplication
89    let ret = StepTransformAction {
90        extra_parent_plans: vec![step_aggr_plan.upper_merge.clone()],
91        new_child_plan: Some(step_aggr_plan.lower_state.clone()),
92    };
93    Ok(ret)
94}

97pub enum Commutativity {
98    Commutative,
99    PartialCommutative,
100    ConditionalCommutative(Option<Transformer>),
101    TransformedCommutative {
102        /// Return plans from parent to child order
103        transformer: Option<StageTransformer>,
104    },
105    NonCommutative,
106    Unimplemented,
107    /// For unrelated plans like DDL
108    Unsupported,
109}

113impl Categorizer {
114    pub fn check_plan(
115        plan: &LogicalPlan,
116        partition_cols: Option<AliasMapping>,
117    ) -> DfResult<Commutativity> {
118        // Subquery is treated separately in `inspect_plan_with_subquery`. To avoid rewrite the
119        // "maybe rewritten" plan, stop the check here.
120        if has_subquery(plan)? {
121            return Ok(Commutativity::Unimplemented);
122        }
123
124        let partition_cols = partition_cols.unwrap_or_default();
125
126        let comm = match plan {
127            LogicalPlan::Projection(proj) => {
128                for expr in &proj.expr {
129                    let commutativity = Self::check_expr(expr);
130                    if !matches!(commutativity, Commutativity::Commutative) {
131                        return Ok(commutativity);
132                    }
133                }
134                Commutativity::Commutative
135            }
136            LogicalPlan::Filter(filter) => Self::check_expr(&filter.predicate),
137            LogicalPlan::Window(_) => Commutativity::Unimplemented,
138            LogicalPlan::Aggregate(aggr) => {
139                // The pinned Substrait codec drops IGNORE NULLS, so keep these aggregates on the frontend.
140                if aggr.aggr_expr.iter().any(|expr| {
141                    get_aggr_func(expr).is_some_and(|func| {
142                        func.params.null_treatment == Some(NullTreatment::IgnoreNulls)
143                    })
144                }) {
145                    return Ok(Commutativity::NonCommutative);
146                }
147                // The state/merge split maps each group expression to one output column,
148                // which doesn't hold for grouping sets.
149                let has_grouping_set = aggr
150                    .group_expr
151                    .iter()
152                    .any(|expr| matches!(expr, Expr::GroupingSet(_)));
153                let is_all_steppable =
154                    !has_grouping_set && is_all_aggr_exprs_steppable(&aggr.aggr_expr);
155                let matches_partition = Self::check_partition(&aggr.group_expr, &partition_cols);
156                if !matches_partition && is_all_steppable {
157                    debug!("Plan is steppable: {plan}");
158                    return Ok(Commutativity::TransformedCommutative {
159                        transformer: Some(Arc::new(|plan: &LogicalPlan| {
160                            debug!("Before Step optimize: {plan}");
161                            let ret = step_aggr_to_upper_aggr(plan);
162                            ret.inspect_err(|err| {
163                                common_telemetry::error!("Failed to step aggregate plan: {err:?}");
164                            })
165                            .map(|s| TransformerAction {
166                                extra_parent_plans: s.extra_parent_plans,
167                                new_child_plan: s.new_child_plan,
168                            })
169                        })),
170                    });
171                }
172                if !matches_partition {
173                    return Ok(Commutativity::NonCommutative);
174                }
175                for expr in &aggr.aggr_expr {
176                    let commutativity = Self::check_expr(expr);
177                    if !matches!(commutativity, Commutativity::Commutative) {
178                        return Ok(commutativity);
179                    }
180                }
181                // all group by expressions are partition columns can push down, unless
182                // another push down(including `Limit` or `Sort`) is already in progress(which will then prevent next cond commutative node from being push down).
183                // TODO(discord9): This is a temporary solution(that works), a better description of
184                // commutativity is needed under this situation.
185                Commutativity::ConditionalCommutative(None)
186            }
187            LogicalPlan::Sort(_sort) => {
188                if partition_cols.is_empty() {
189                    return Ok(Commutativity::Commutative);
190                }
191
192                // sort plan needs to consider column priority
193                // Change Sort to MergeSort which assumes the input streams are already sorted hence can be more efficient.
194                Commutativity::ConditionalCommutative(Some(Arc::new(merge_sort_transformer)))
195            }
196            LogicalPlan::Join(_) => Commutativity::NonCommutative,
197            LogicalPlan::Repartition(_) => {
198                // unsupported? or non-commutative
199                Commutativity::Unimplemented
200            }
201            LogicalPlan::Union(_) => Commutativity::Unimplemented,
202            LogicalPlan::TableScan(_) => Commutativity::Commutative,
203            LogicalPlan::EmptyRelation(_) => Commutativity::NonCommutative,
204            LogicalPlan::Subquery(_) => Commutativity::Unimplemented,
205            LogicalPlan::SubqueryAlias(_) => Commutativity::Commutative,
206            LogicalPlan::Limit(limit) => {
207                // Only execute `fetch` on remote nodes.
208                // wait for https://github.com/apache/arrow-datafusion/pull/7669
209                if partition_cols.is_empty() && limit.fetch.is_some() {
210                    Commutativity::Commutative
211                } else if limit.skip.is_none() && limit.fetch.is_some() {
212                    Commutativity::PartialCommutative
213                } else {
214                    Commutativity::Unimplemented
215                }
216            }
217            LogicalPlan::Extension(extension) => {
218                Self::check_extension_plan(extension.node.as_ref() as _, &partition_cols)
219            }
220            LogicalPlan::Distinct(_) => {
221                if partition_cols.is_empty() {
222                    Commutativity::Commutative
223                } else {
224                    Commutativity::PartialCommutative
225                }
226            }
227            LogicalPlan::Unnest(_) => Commutativity::Commutative,
228            LogicalPlan::Statement(_) => Commutativity::Unsupported,
229            LogicalPlan::Values(_) => Commutativity::Unsupported,
230            LogicalPlan::Explain(_) => Commutativity::Unsupported,
231            LogicalPlan::Analyze(_) => Commutativity::Unsupported,
232            LogicalPlan::DescribeTable(_) => Commutativity::Unsupported,
233            LogicalPlan::Dml(_) => Commutativity::Unsupported,
234            LogicalPlan::Ddl(_) => Commutativity::Unsupported,
235            LogicalPlan::Copy(_) => Commutativity::Unsupported,
236            LogicalPlan::RecursiveQuery(_) => Commutativity::Unsupported,
237        };
238
239        Ok(comm)
240    }

242    pub fn check_extension_plan(
243        plan: &dyn UserDefinedLogicalNode,
244        partition_cols: &AliasMapping,
245    ) -> Commutativity {
246        match plan.name() {
247            name if name == SeriesDivide::name() => {
248                let series_divide = plan.as_any().downcast_ref::<SeriesDivide>().unwrap();
249                // Metric engine `__tsid` uniquely identifies a time-series. Treat a series divide
250                // that keys by `__tsid` as commutative across regions so it can be pushed down.
251                if series_divide
252                    .tags()
253                    .iter()
254                    .any(|tag| tag == DATA_SCHEMA_TSID_COLUMN_NAME)
255                {
256                    return Commutativity::Commutative;
257                }
258
259                let tags = series_divide.tags().iter().collect::<HashSet<_>>();
260
261                for all_alias in partition_cols.values() {
262                    let all_alias = all_alias.iter().map(|c| &c.name).collect::<HashSet<_>>();
263                    if tags.intersection(&all_alias).count() == 0 {
264                        return Commutativity::NonCommutative;
265                    }
266                }
267
268                Commutativity::Commutative
269            }
270            name if name == FusedSeries::name() => {
271                let fused = plan.as_any().downcast_ref::<FusedSeries>().unwrap();
272                if fused.is_canonical_producer() {
273                    // A canonical producer merges the pieces of one label set that its own input
274                    // partitions hold, and its physical canonical router collects those pieces from
275                    // exactly those partitions. A region holds only part of a label set, so a
276                    // producer there would answer a label set from one region's share whenever the
277                    // pieces are spread. Nothing below the node changes that: a canonical cut wraps
278                    // its inputs in sources independently of how the scan was chosen, so the shape
279                    // is not a placement proof. The per-series rules below are not consulted: they
280                    // describe what a node computes, not where the pieces of a label set are.
281                    return Commutativity::NonCommutative;
282                }
283                // The raw-series reader supports one scan, or the narrow same-table multi-scan
284                // one-to-one case. This classification is for region-local execution only.
285                let children = fused.children();
286                if children.len() == 1 {
287                    if !matches!(children[0], LogicalPlan::TableScan(_)) {
288                        return Commutativity::NonCommutative;
289                    }
290                } else if !(same_table_scans(children)
291                    && independent_scan_pair(fused.tree().root()))
292                {
293                    return Commutativity::NonCommutative;
294                }
295                let binding = fused.binding();
296                if binding.metric_name.is_none() {
297                    return Commutativity::NonCommutative;
298                }
299                let root = fused.tree().root();
300                // A `Partial` root is safe below the region boundary only when its prefix itself
301                // reads per-series work. Do not rely on split admission here: hand-built trees can
302                // put a whole-vector operation below the aggregate. The `Final` half above the
303                // exchange is not classified here because its input is the exchange, not a scan.
304                let partial = matches!(
305                    &root.expr,
306                    NativeExpr::Aggregate { input, op }
307                        if op.stage == AggregateStage::Partial && !reads_across_series(input)
308                );
309                let per_series = pointwise_operand(root) || independent_scan_pair(root);
310                // A gather of one raw scan does not establish region-global ownership. Cross-series
311                // consumers need their existing split or frontend placement.
312                if reads_across_series(root) && !per_series && !partial {
313                    return Commutativity::NonCommutative;
314                }
315
316                // The multi-scan exception describes only in-process, same-table region-local
317                // execution. It is not a claim that matching is globally commutative across regions.
318                Commutativity::Commutative
319            }
320            // A region answers its own series of one table, which is what the node above the merge
321            // scans reads from every region.
322            name if name == RawSeriesSource::name() => {
323                if matches!(plan.inputs().as_slice(), [LogicalPlan::TableScan(_)]) {
324                    Commutativity::Commutative
325                } else {
326                    Commutativity::NonCommutative
327                }
328            }
329            name if name == SeriesNormalize::name()
330                || name == InstantManipulate::name()
331                || name == RangeManipulate::name() =>
332            {
333                // They should always follows Series Divide.
334                // Either all commutative or all non-commutative (which will be blocked by SeriesDivide).
335                Commutativity::Commutative
336            }
337            name if name == EmptyMetric::name()
338                || name == MergeScanLogicalPlan::name()
339                || name == MergeSortLogicalPlan::name() =>
340            {
341                Commutativity::Unimplemented
342            }
343            _ => Commutativity::Unsupported,
344        }
345    }

347    pub fn check_expr(expr: &Expr) -> Commutativity {
348        #[allow(deprecated)]
349        match expr {
350            Expr::Column(_)
351            | Expr::ScalarVariable(_, _)
352            | Expr::Literal(_, _)
353            | Expr::BinaryExpr(_)
354            | Expr::Not(_)
355            | Expr::IsNotNull(_)
356            | Expr::IsNull(_)
357            | Expr::IsTrue(_)
358            | Expr::IsFalse(_)
359            | Expr::IsNotTrue(_)
360            | Expr::IsNotFalse(_)
361            | Expr::Negative(_)
362            | Expr::Between(_)
363            | Expr::Exists(_)
364            | Expr::InList(_)
365            | Expr::Like(_)
366            | Expr::Case(_) => Commutativity::Commutative,
367            // Annotation collection must not affect distribution. Keep scalar UDFs pushdownable;
368            // TODO: Preserve datanode PromQL annotations through plan decoding, return them in
369            // Flight stream metadata, and merge them on the frontend.
370            Expr::ScalarFunction(_) => Commutativity::Commutative,
371            Expr::AggregateFunction(_udaf) => Commutativity::Commutative,
372
373            Expr::SimilarTo(_)
374            | Expr::IsUnknown(_)
375            | Expr::IsNotUnknown(_)
376            | Expr::WindowFunction(_)
377            | Expr::InSubquery(_)
378            | Expr::ScalarSubquery(_)
379            | Expr::HigherOrderFunction(_)
380            | Expr::Lambda(_)
381            | Expr::LambdaVariable(_)
382            | Expr::Wildcard { .. } => Commutativity::Unimplemented,
383
384            Expr::Alias(alias) => Self::check_expr(&alias.expr),
385            Expr::Cast(cast) => Self::check_expr(&cast.expr),
386            Expr::TryCast(try_cast) => Self::check_expr(&try_cast.expr),
387
388            Expr::Unnest(_)
389            | Expr::GroupingSet(_)
390            | Expr::Placeholder(_)
391            | Expr::OuterReferenceColumn(_, _)
392            | Expr::SetComparison(_) => Commutativity::Unimplemented,
393        }
394    }
395
396    /// Return true if the given expr and partition cols satisfied the rule.
397    /// In this case the plan can be treated as fully commutative.
398    ///
399    /// So only if every partition column is itself one of `exprs`, return true.
400    /// Otherwise return false.
401    ///
402    /// An expression that only references a partition column, like `substr(host, 3, 1)`
403    /// or `k % 2`, doesn't count: it can put rows from different partitions into the same
404    /// group.
405    fn check_partition(exprs: &[Expr], partition_cols: &AliasMapping) -> bool {
406        let group_cols = exprs
407            .iter()
408            .filter_map(|expr| {
409                let mut expr = expr;
410                while let Expr::Alias(alias) = expr {
411                    expr = &alias.expr;
412                }
413                match expr {
414                    Expr::Column(column) => Some(column.name.clone()),
415                    _ => None,
416                }
417            })
418            .collect::<HashSet<_>>();
419        for all_alias in partition_cols.values() {
420            let all_alias = all_alias
421                .iter()
422                .map(|c| c.name.clone())
423                .collect::<HashSet<_>>();
424            // check if group columns intersect with all alias of partition columns
425            // is empty, if it's empty, not all partition columns show up in `exprs`
426            if group_cols.intersection(&all_alias).count() == 0 {
427                return false;
428            }
429        }
430
431        true
432    }

530        let mut stmt = native_statement();

532            parser::parse("some_metric{host=\"left\"} + on(host) some_metric{host=\"right\"}")

534        let series_schema = Arc::new(SeriesSchema {

539            &stmt,
540            &|_| Ok(Arc::clone(&series_schema)),
541            SeriesOrdering::Contiguous,
542            Some(FieldKind::Float),
543        )
544        .unwrap()
545        .tree;
546        let binding = FusedSeriesBinding {
547            grid: statement_grid(&stmt).unwrap(),
548            time_index: "ts".to_string(),
549            tags: vec!["host".to_string()],
550            metric_name: Some(Arc::from("some_metric")),
551        };
552        let output_schema: DFSchemaRef = Arc::new(
553            DFSchema::new_with_metadata(
554                vec![(
555                    None,
556                    Arc::new(Field::new("v0", ArrowDataType::Float64, true)),
557                )],
558                Default::default(),
559            )
560            .unwrap(),
561        );
562        LogicalPlan::Extension(Extension {
563            node: Arc::new(
564                FusedSeries::try_new(
565                    Arc::from("some_metric{host=\"left\"} + on(host) some_metric{host=\"right\"}"),
566                    tree,
567                    binding,
568                    vec![left, right],
569                    output_schema,
570                )
571                .unwrap(),
572            ),
573        })
574    }

707    /// An extension node with no `ExtensionSerializer` branch: neither one under its own name nor
708    /// one it shares, since the serializer lists names and not shapes. It stands in for the nodes
709    /// that cannot reach a region as bytes for their own reasons, which the encoding gate must keep
710    /// honouring. The categorizer refuses it, as it refuses every name it does not know.
711    #[derive(Debug, PartialEq, Eq, Hash)]
712    struct UnserializableNode {
713        input: LogicalPlan,
714        schema: DFSchemaRef,
715    }
716
717    impl PartialOrd for UnserializableNode {
718        fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
719            let ordering = self.input.partial_cmp(&other.input);
720            match ordering {
721                Some(std::cmp::Ordering::Equal) if self.schema != other.schema => None,
722                ordering => ordering,
723            }
724        }
725    }
726
727    impl UnserializableNode {
728        fn new(input: LogicalPlan) -> Self {
729            let schema = input.schema().clone();
730            Self { input, schema }
731        }
732    }

734    impl UserDefinedLogicalNodeCore for UnserializableNode {
735        fn name(&self) -> &str {
736            "UnserializableNode"
737        }
738
739        fn inputs(&self) -> Vec<&LogicalPlan> {
740            vec![&self.input]
741        }
742
743        fn schema(&self) -> &DFSchemaRef {
744            &self.schema
745        }
746
747        fn expressions(&self) -> Vec<Expr> {
748            vec![]
749        }
750
751        fn fmt_for_explain(&self, f: &mut std::fmt::Formatter) -> std::fmt::Result {
752            write!(f, "UnserializableNode")
753        }
754
755        fn with_exprs_and_inputs(
756            &self,
757            _exprs: Vec<Expr>,
758            mut inputs: Vec<LogicalPlan>,
759        ) -> DfResult<Self> {
760            Ok(Self::new(inputs.swap_remove(0)))
761        }
762    }

1206    /// The compiler's tree is split into a `Partial` state-exporting node over the scan and a
1207    /// `Final` node above it, matching the two halves a query plans.
1208    fn split_aggregate_plan(input: LogicalPlan) -> LogicalPlan {

1210            expr: parser::parse("sum(some_metric)").unwrap(),

1218            &stmt,
1219            Arc::clone(&series_schema),
1220            SeriesOrdering::KeyThenTime,
1221        )
1222        .unwrap();
1223        let binding = FusedSeriesBinding {
1224            grid: statement_grid(&stmt).unwrap(),
1225            metric_name: Some(Arc::from("some_metric")),
1226            time_index: "ts".to_string(),
1227            tags: Vec::new(),
1228        };
1229        let output_schema: DFSchemaRef = Arc::new(
1230            DFSchema::new_with_metadata(
1231                vec![
1232                    (
1233                        None,
1234                        Arc::new(Field::new(
1235                            "ts",
1236                            ArrowDataType::Timestamp(ArrowTimeUnit::Millisecond, None),
1237                            false,
1238                        )),
1239                    ),
1240                    (
1241                        None,
1242                        Arc::new(Field::new("sum(v0)", ArrowDataType::Float64, true)),
1243                    ),
1244                ],
1245                Default::default(),
1246            )
1247            .unwrap(),
1248        );
1249
1250        let (partial_tree, final_tree) =
1251            FusedSeries::split_aggregate(&tree, "ts").unwrap().unwrap();
1252        let partial = FusedSeries::try_new_state_export(
1253            Arc::from("sum(some_metric)"),
1254            partial_tree,
1255            binding.clone(),
1256            vec![input],
1257        )
1258        .unwrap();
1259        let final_node = FusedSeries::try_new(
1260            Arc::from("sum(some_metric)"),
1261            final_tree,
1262            binding,
1263            vec![LogicalPlan::Extension(Extension {
1264                node: Arc::new(partial),
1265            })],
1266            output_schema,
1267        )
1268        .unwrap();
1269        LogicalPlan::Extension(Extension {
1270            node: Arc::new(final_node),
1271        })
1272    }

1274    fn as_final_split(plan: &LogicalPlan) -> Option<&FusedSeries> {
1275        if let LogicalPlan::Extension(extension) = plan {
1276            extension
1277                .node
1278                .as_any()
1279                .downcast_ref::<FusedSeries>()
1280                .filter(|fused| fused.reads_aggregate_state())
1281        } else {
1282            None
1283        }
1284    }

1286    /// The first fused producer in `plan`'s visible tree.
1287    fn first_fused_series(plan: &LogicalPlan) -> Option<&FusedSeries> {
1288        if let LogicalPlan::Extension(extension) = plan
1289            && let Some(fused) = extension.node.as_any().downcast_ref::<FusedSeries>()
1290        {
1291            return Some(fused);
1292        }
1293        plan.inputs().into_iter().find_map(first_fused_series)
1294    }

1315    /// The final half stays on the frontend and the partial half rides inside the region scan, for
1316    /// every deployment: the split aggregate's two halves are decided by what the plan is, so a
1317    /// stand-alone deployment and a distributed frontend share the structure.
1318    #[test]
1319    fn a_split_aggregate_finalizes_above_the_region() {
1320        let plan = split_aggregate_plan(table_scan_plan());
1321        assert!(
1322            DFLogicalSubstraitConvertor
1323                .encode(&plan, DefaultSerializer)
1324                .is_ok(),
1325            "a split plan encodes like any other plan; where it may run is a placement decision"
1326        );
1327
1328        let analyzed_plan = analyze(DistPlannerAnalyzer::default(), plan);
1329        assert!(
1330            as_final_split(&analyzed_plan).is_some(),
1331            "the final node must stay on the frontend: {analyzed_plan}"
1332        );
1333        let final_node = as_final_split(&analyzed_plan).unwrap();
1334        let [child] = final_node.children() else {
1335            panic!("expected the final node to keep one input: {analyzed_plan}");
1336        };
1337        let merge_scan = first_merge_scan(child).unwrap_or_else(|| {
1338            panic!("expected a merge scan below the final node: {analyzed_plan}")
1339        });
1340        let partial = first_fused_series(merge_scan.input()).unwrap_or_else(|| {
1341            panic!("expected the partial node in the region input: {analyzed_plan}")
1342        });
1343        assert!(!partial.reads_aggregate_state());
1344        assert!(matches!(
1345            &partial.tree().root().expr,
1346            NativeExpr::Aggregate { op, .. } if op.stage == AggregateStage::Partial
1347        ));
1348        assert!(holds_table_scan(merge_scan.input()));
1349        // The analyzer, not the codec, decides placement: a split aggregate's partial producer is
1350        // region-local work and belongs below the boundary.
1351        assert!(
1352            any_remote_input_holds(&analyzed_plan, holds_fused_series),
1353            "the analyzer keeps the partial producer below the region boundary: {analyzed_plan}"
1354        );

1357    /// The final half of `plan`, with its partial input replaced by `producer`.
1358    fn final_over(plan: LogicalPlan, producer: LogicalPlan) -> LogicalPlan {
1359        let LogicalPlan::Extension(extension) = &plan else {
1360            panic!("a split aggregate is an extension node: {plan}");
1361        };
1362        let fused = extension
1363            .node
1364            .as_any()
1365            .downcast_ref::<FusedSeries>()
1366            .expect("a split aggregate's final half is a FusedSeries");
1367        // Qualified: both `UserDefinedLogicalNode` and `UserDefinedLogicalNodeCore` declare this
1368        // method, and only the latter's answer is the node itself.
1369        let rebuilt =
1370            UserDefinedLogicalNodeCore::with_exprs_and_inputs(fused, Vec::new(), vec![producer])
1371                .expect("the final node holds the one input it is given");
1372        LogicalPlan::Extension(Extension {
1373            node: Arc::new(rebuilt),
1374        })
1375    }

1464fn has_subquery(plan: &LogicalPlan) -> DfResult<bool> {
1465    let mut found = false;
1466    plan.apply_expressions(|e| {
1467        e.apply(|x| {
1468            if matches!(
1469                x,
1470                Expr::Exists(_) | Expr::InSubquery(_) | Expr::ScalarSubquery(_)
1471            ) {
1472                found = true;
1473                Ok(TreeNodeRecursion::Stop)
1474            } else {
1475                Ok(TreeNodeRecursion::Continue)
1476            }
1477        })
1478    })?;
1479    Ok(found)
1480}
