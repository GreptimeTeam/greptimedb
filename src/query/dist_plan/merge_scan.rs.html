88/// Returns a deterministic upper bound on rows emitted by the remote plan for
89/// one region.
90///
91/// Only explicit caps and row-non-increasing wrappers are followed. Nodes that
92/// may multiply rows fail open, so the result remains a sound upper bound.
93fn remote_plan_row_bound(plan: &LogicalPlan) -> Option<usize> {
94    match plan {
95        LogicalPlan::Limit(limit) => {
96            let input_bound = remote_plan_row_bound(&limit.input);
97            match limit.get_fetch_type() {
98                Ok(FetchType::Literal(Some(fetch))) => {
99                    Some(input_bound.map_or(fetch, |bound| bound.min(fetch)))
100                }
101                _ => input_bound,
102            }
103        }
104        LogicalPlan::Sort(sort) => {
105            let input_bound = remote_plan_row_bound(&sort.input);
106            sort.fetch
107                .map(|fetch| input_bound.map_or(fetch, |bound| bound.min(fetch)))
108                .or(input_bound)
109        }
110        LogicalPlan::Projection(projection) => remote_plan_row_bound(&projection.input),
111        LogicalPlan::Filter(filter) => remote_plan_row_bound(&filter.input),
112        LogicalPlan::SubqueryAlias(alias) => remote_plan_row_bound(&alias.input),
113        LogicalPlan::Window(window) => remote_plan_row_bound(&window.input),
114        LogicalPlan::Repartition(repartition) => remote_plan_row_bound(&repartition.input),
115        LogicalPlan::Distinct(distinct) => remote_plan_row_bound(distinct.input()),
116        LogicalPlan::Aggregate(aggregate) => {
117            if aggregate
118                .group_expr
119                .iter()
120                .any(|expr| matches!(expr, Expr::GroupingSet(_)))
121            {
122                None
123            } else if aggregate.group_expr.is_empty() {
124                Some(1)
125            } else {
126                remote_plan_row_bound(&aggregate.input)
127            }
128        }
129        _ => None,
130    }
131}

162/// Returns true when the field is a JSON column, identified either by its
163/// Arrow extension type (`greptime.json` / `greptime.json2`) or by its
164/// `greptime:type=Json` marker.
165fn is_json_field(field: &Field) -> bool {
166    is_any_json_extension_type(field)
167        || field
168            .metadata()
169            .get(datatypes::schema::TYPE_KEY)
170            .map(String::as_ref)
171            == Some("Json")
172}
173
174/// Returns true when two fields describe the same column semantics: same
175/// name, same data type, and same nullability.
176///
177/// Arrow `Field` metadata is auxiliary information (indexing, encoding,
178/// compression hints, ...) and does not participate in the comparison. This
179/// applies to every field, JSON columns included: JSON identity keys
180/// (`greptime:type`, `ARROW:extension:metadata`) are never compared here.
181/// The wire/decoded physical-type difference of JSON columns is exempted
182/// separately by [`json_fields_compatible`].
183fn fields_semantically_equal(expected_field: &Field, actual_field: &Field) -> bool {
184    expected_field.name() == actual_field.name()
185        && expected_field.data_type() == actual_field.data_type()
186        && expected_field.is_nullable() == actual_field.is_nullable()
187}
188
189/// Returns true when two fields are semantically equivalent JSON columns.
190///
191/// A JSON column is carried on the wire in its binary-encoded form (e.g.
192/// `Binary` + `ARROW:extension:name=greptime.json` / `greptime:type=Json`) and,
193/// after decoding on the remote side, in a concretized structured form (e.g.
194/// `Struct(...)` / `List(...)` carrying the same extension metadata). Both forms
195/// describe the same column, so the raw arrow data type must not be compared
196/// directly. The JSON extension identity (`greptime:type`) and the JSON2
197/// settings (`ARROW:extension:metadata`, e.g. type hints) are the semantic
198/// parts of the field and must match.
199fn json_fields_compatible(expected_field: &Field, actual_field: &Field) -> bool {
200    is_json_field(expected_field)
201        && is_json_field(actual_field)
202        && expected_field.name() == actual_field.name()
203        && expected_field.is_nullable() == actual_field.is_nullable()
204        // Both must carry the same JSON marker.
205        && expected_field.metadata().get(datatypes::schema::TYPE_KEY)
206            == actual_field.metadata().get(datatypes::schema::TYPE_KEY)
207        // JSON2 settings (type hints etc.) must match.
208        && expected_field
209            .metadata()
210            .get(arrow_schema::extension::EXTENSION_TYPE_METADATA_KEY)
211            == actual_field
212                .metadata()
213                .get(arrow_schema::extension::EXTENSION_TYPE_METADATA_KEY)
214}
215
216/// Validates the remote schema before positional column handling.
217///
218/// Field metadata (indexing/encoding/compression hints) is non-semantic at
219/// this boundary and is ignored: two fields are equal when their name, data
220/// type, and nullability match (see [`fields_semantically_equal`]). A
221/// timestamp timezone difference is the only intentional exception, accepted
222/// when the fields share the same timestamp unit, distinct timezones, and
223/// equal name and nullability. Top-level Arrow schema metadata is
224/// non-semantic at this boundary as well. JSON columns are compared
225/// semantically (see [`json_fields_compatible`]) because their wire and
226/// decoded representations use different physical arrow types.
227fn validate_remote_schema(
228    expected: &ArrowSchema,
229    actual: &ArrowSchema,
230    source: &str,
231) -> Result<()> {
232    if expected.fields().len() != actual.fields().len() {
233        return Err(remote_schema_mismatch(format!(
234            "MergeScan {source} schema field count mismatch: expected {}, actual {}",
235            expected.fields().len(),
236            actual.fields().len()
237        )));
238    }
239
240    for (index, (expected_field, actual_field)) in expected
241        .fields()
242        .iter()
243        .zip(actual.fields().iter())
244        .enumerate()
245    {
246        // Field metadata does not participate in the semantic comparison.
247        if fields_semantically_equal(expected_field, actual_field) {
248            continue;
249        }
250
251        // JSON columns are equivalent in their binary wire form and their
252        // decoded structured form; compare them semantically instead of
253        // comparing the raw arrow data type.
254        if json_fields_compatible(expected_field, actual_field) {
255            continue;
256        }
257
258        // Intentionally mirrors Arrow Field equality properties, except timezone.
259        let timezone_only_difference = matches!(
260            (expected_field.data_type(), actual_field.data_type()),
261            (
262                ArrowDataType::Timestamp(expected_unit, expected_timezone),
263                ArrowDataType::Timestamp(actual_unit, actual_timezone),
264            ) if expected_unit == actual_unit
265                && expected_timezone != actual_timezone
266                && expected_field.name() == actual_field.name()
267                && expected_field.is_nullable() == actual_field.is_nullable()
268        );
269        if !timezone_only_difference {
270            return Err(remote_schema_mismatch(format!(
271                "MergeScan {source} schema field mismatch at position {index}: expected {:?}, actual {:?}",
272                expected_field, actual_field
273            )));
274        }
275    }
276
277    Ok(())
278}

462impl MergeScanExec {
463    #[allow(clippy::too_many_arguments)]
464    pub fn new(
465        session_state: &SessionState,
466        table: TableName,
467        regions: Vec<RegionId>,
468        plan: LogicalPlan,
469        arrow_schema: &ArrowSchema,
470        region_query_handler: RegionQueryHandlerRef,
471        query_ctx: QueryContextRef,
472        target_partition: usize,
473        partition_cols: AliasMapping,
474        remote_dyn_filter_producer_id: Option<RemoteDynFilterProducerId>,
475        enable_per_region_metrics: bool,
476    ) -> Result<Self> {
477        let arrow_schema = maybe_amend_json2_field(arrow_schema);
478        let output_partition_count = Self::output_partition_count(regions.len(), target_partition);
479
480        // States the output ordering of the plan.
481        //
482        // When the input plan is a sort, we can use the sort ordering as the output ordering
483        // if the target partition is greater than the number of regions, which means we won't
484        // break the ordering on merging (of MergeScan).
485        //
486        // Otherwise, we need to use the default ordering.
487        let eq_properties = if let LogicalPlan::Sort(sort) = &plan
488            && output_partition_count >= regions.len()
489        {
490            let lex_ordering = sort
491                .expr
492                .iter()
493                .map(|sort_expr| {
494                    let physical_expr = session_state
495                        .create_physical_expr(sort_expr.expr.clone(), plan.schema())?;
496                    Ok(PhysicalSortExpr::new(
497                        physical_expr,
498                        SortOptions {
499                            descending: !sort_expr.asc,
500                            nulls_first: sort_expr.nulls_first,
501                        },
502                    ))
503                })
504                .collect::<Result<Vec<_>>>()?;
505            EquivalenceProperties::new_with_orderings(arrow_schema.clone(), vec![lex_ordering])
506        } else {
507            EquivalenceProperties::new(arrow_schema.clone())
508        };
509
510        let partition_exprs = partition_cols
511            .iter()
512            .filter_map(|col| {
513                if let Some(first_alias) = col.1.first() {
514                    session_state
515                        .create_physical_expr(
516                            Expr::Column(ColumnExpr::new_unqualified(
517                                first_alias.name().to_string(),
518                            )),
519                            plan.schema(),
520                        )
521                        .ok()
522                } else {
523                    None
524                }
525            })
526            .collect();
527        let partitioning = Partitioning::Hash(partition_exprs, output_partition_count);
528
529        let properties = Arc::new(PlanProperties::new(
530            eq_properties,
531            partitioning,
532            EmissionType::Incremental,
533            Boundedness::Bounded,
534        ));
535        Ok(Self {
536            table,
537            regions,
538            plan,
539            arrow_schema,
540            region_query_handler,
541            metric: ExecutionPlanMetricsSet::new(),
542            sub_stage_metrics: Arc::default(),
543            partition_metrics: Arc::default(),
544            properties,
545            query_ctx,
546            remote_dyn_filter_producer_id,
547            captured_remote_dyn_filters: Arc::default(),
548            target_partition,
549            partition_cols,
550            enable_per_region_metrics,
551        })
552    }
553
554    /// Conservative row-count upper bound for all selected regions.
555    ///
556    /// This is not an expected cardinality: DataFusion receives it as an
557    /// inexact estimate only because `Statistics` has no upper-bound precision.
558    fn estimated_num_rows(&self) -> Precision<usize> {
559        if self.regions.is_empty() {
560            return Precision::Inexact(0);
561        }
562
563        let Some(rows_per_region) = remote_plan_row_bound(&self.plan) else {
564            return Precision::Absent;
565        };
566        rows_per_region
567            .checked_mul(self.regions.len())
568            .map_or(Precision::Absent, Precision::Inexact)
569    }

571    /// Number of partitions populated by the region striping in [`Self::to_stream`].
572    fn output_partition_count(num_regions: usize, target_partition: usize) -> usize {
573        num_regions.max(1).min(target_partition.max(1))
574    }
575
576    pub fn to_stream(
577        &self,
578        context: Arc<TaskContext>,
579        partition: usize,
580    ) -> Result<SendableRecordBatchStream> {
581        // prepare states to move
582        let regions = self.regions.clone();
583        let region_query_handler = self.region_query_handler.clone();
584        let metric = MergeScanMetric::new(&self.metric);
585        let arrow_schema = self.arrow_schema.clone();
586        let query_ctx = self.query_ctx.clone();
587        let sub_stage_metrics_moved = self.sub_stage_metrics.clone();
588        let partition_metrics_moved = self.partition_metrics.clone();
589        let plan = self.plan.clone();
590        let target_partition =
591            Self::output_partition_count(self.regions.len(), self.target_partition);
592        let remote_dyn_filter_enabled = remote_dyn_filter_enabled(&self.query_ctx)?;
593        let captured_remote_dyn_filters = if remote_dyn_filter_enabled {
594            self.captured_remote_dyn_filters()
595        } else {
596            Vec::new()
597        };
598        let dbname = context.task_id().unwrap_or_default();
599        let tracing_context = TracingContext::from_json(context.session_id().as_str());
600        let current_channel = self.query_ctx.channel();
601        let read_preference = self.query_ctx.read_preference();
602        let explain_verbose = self.query_ctx.explain_verbose();
603        let live_analyze_metrics = explain_verbose && self.query_ctx.live_analyze_metrics_enabled();
604        let remote_dyn_filter_registry_lease = acquire_remote_dyn_filter_registry_lease(
605            context.as_ref(),
606            &query_ctx,
607            &captured_remote_dyn_filters,
608        );
609
610        let stream = Box::pin(stream!({
611            let remote_dyn_filter_registry_lease = remote_dyn_filter_registry_lease;
612            // only report metrics once for each MergeScan
613            if partition == 0 {
614                MERGE_SCAN_REGIONS.observe(regions.len() as f64);
615            }
616
617            let _finish_timer = metric.finish_time().timer();
618            let mut ready_timer = metric.ready_time().timer();
619            let mut first_consume_timer = Some(metric.first_consume_time().timer());
620
621            // Per-partition timings, scoped to this partition's stream for `EXPLAIN VERBOSE`.
622            let partition_start = Instant::now();
623            let mut partition_ready_time: Option<Duration> = None;
624            let mut partition_first_consume_time: Option<Duration> = None;
625
626            for region_id in regions
627                .iter()
628                .skip(partition)
629                .step_by(target_partition)
630                .copied()
631            {
632                let region_span = tracing_context.attach(tracing::info_span!(
633                    parent: &Span::current(),
634                    "merge_scan_region",
635                    region_id = %region_id,
636                    partition = partition
637                ));
638                let region_start = Instant::now();
639                register_remote_dyn_filters_for_region(
640                    remote_dyn_filter_registry_lease.as_ref(),
641                    &captured_remote_dyn_filters,
642                );
643                let select_target_start = Instant::now();
644                let target = region_query_handler
645                    .select_target(read_preference, region_id)
646                    .instrument(region_span.clone())
647                    .await
648                    .map_err(|e| {
649                        MERGE_SCAN_ERRORS_TOTAL.inc();
650                        DataFusionError::External(Box::new(e))
651                    })?;
652                let select_target_cost = select_target_start.elapsed();
653                let mut subscriber_rollback =
654                    remote_dyn_filter_registry_lease.as_ref().map(|lease| {
655                        SubscriberRollbackGuard::new(
656                            lease.registry(),
657                            register_dyn_filter_subscribers_for_region(
658                                lease.registry(),
659                                region_id,
660                                target.clone(),
661                                &captured_remote_dyn_filters,
662                            ),
663                        )
664                    });
665                let mut region_query_ctx = query_context_with_refreshed_initial_dyn_filter_regs(
666                    &query_ctx,
667                    region_id,
668                    &captured_remote_dyn_filters,
669                );
670                if live_analyze_metrics {
671                    let remote_query_id = region_query_ctx.remote_query_id().map(str::to_string);
672                    if let Some(remote_query_id) = remote_query_id {
673                        region_query_ctx.set_extension(
674                            SUPPORT_FLIGHT_METRICS_BEFORE_BATCH_EXTENSION_KEY,
675                            remote_query_id,
676                        );
677                    }
678                }
679                let request = QueryRequest {
680                    header: Some(RegionRequestHeader {
681                        tracing_context: tracing_context.to_w3c(),
682                        dbname: dbname.clone(),
683                        query_context: Some((&region_query_ctx).into()),
684                    }),
685                    region_id,
686                    plan: plan.clone(),
687                };
688                if explain_verbose {
689                    common_telemetry::info!(
690                        "Merge scan one region, partition: {}, region_id: {}",
691                        partition,
692                        region_id
693                    );
694                }
695
696                let do_get_start = Instant::now();
697                let do_get_result = region_query_handler
698                    .do_get(&target, request)
699                    .instrument(region_span.clone())
700                    .await;
701                if do_get_result.is_err() {
702                    drop(subscriber_rollback.take());
703                }
704                let mut stream = do_get_result.map_err(|e| {
705                    MERGE_SCAN_ERRORS_TOTAL.inc();
706                    DataFusionError::External(Box::new(e))
707                })?;
708
709                if let Some(subscriber_rollback) = subscriber_rollback.as_mut() {
710                    subscriber_rollback.disarm();
711                }
712                let mut advertised_schema = stream.schema().arrow_schema().clone();
713                validate_remote_schema(
714                    arrow_schema.as_ref(),
715                    advertised_schema.as_ref(),
716                    "advertised remote stream",
717                )
718                .inspect_err(|_| record_merge_scan_schema_error())?;
719                let do_get_cost = select_target_cost + do_get_start.elapsed();
720
721                if let Some(remote_dyn_filter_registry_lease) =
722                    remote_dyn_filter_registry_lease.as_ref()
723                {
724                    remote_dyn_filter_registry_lease
725                        .ensure_fanout_task(region_query_handler.clone());
726                }
727
728                ready_timer.stop();
729                if partition_ready_time.is_none() {
730                    partition_ready_time = Some(partition_start.elapsed());
731                }
732
733                let mut poll_duration = Duration::ZERO;
734                let mut poll_timer = Instant::now();
735                loop {
736                    let batch = if live_analyze_metrics {
737                        match time::timeout(
738                            FLIGHT_METRICS_HEARTBEAT_INTERVAL,
739                            stream.next().instrument(region_span.clone()),
740                        )
741                        .await
742                        {
743                            Ok(batch) => batch,
744                            Err(_) => {
745                                if let Some(metrics) = stream.metrics() {
746                                    let mut sub_stage_metrics =
747                                        sub_stage_metrics_moved.lock().unwrap();
748                                    sub_stage_metrics.insert(region_id, metrics);
749                                }
750                                continue;
751                            }
752                        }
753                    } else {
754                        stream.next().instrument(region_span.clone()).await
755                    };
756                    let Some(batch) = batch else {
757                        break;
758                    };
759                    let poll_elapsed = poll_timer.elapsed();
760                    poll_duration += poll_elapsed;
761
762                    let batch = batch.map_err(|e| DataFusionError::External(Box::new(e)))?;
763                    let df_batch = batch.into_df_record_batch();
764                    if !Arc::ptr_eq(&advertised_schema, df_batch.schema_ref()) {
765                        validate_remote_schema(
766                            arrow_schema.as_ref(),
767                            df_batch.schema_ref().as_ref(),
768                            "remote record batch",
769                        )
770                        .inspect_err(|_| record_merge_scan_schema_error())?;
771                        advertised_schema = df_batch.schema_ref().clone();
772                    }
773                    let batch =
774                        patch_batch_timezone(arrow_schema.clone(), df_batch.columns().to_vec())?;
775                    metric.record_output_batch_rows(batch.num_rows());
776                    if let Some(mut first_consume_timer) = first_consume_timer.take() {
777                        first_consume_timer.stop();
778                        partition_first_consume_time = Some(partition_start.elapsed());
779                    }
780
781                    if let Some(metrics) = stream.metrics() {
782                        let mut sub_stage_metrics = sub_stage_metrics_moved.lock().unwrap();
783                        sub_stage_metrics.insert(region_id, metrics);
784                    }
785
786                    yield Ok(batch);
787                    // reset poll timer
788                    poll_timer = Instant::now();
789                }
790                // Also stop on an exhausted stream that yielded no batch. The `take()`
791                // guard ensures it only records once, on the first such region.
792                if let Some(mut first_consume_timer) = first_consume_timer.take() {
793                    first_consume_timer.stop();
794                    partition_first_consume_time = Some(partition_start.elapsed());
795                }
796                let total_cost = region_start.elapsed();
797
798                // Record region metrics and push to global partition_metrics
799                let region_metrics = RegionMetrics {
800                    region_id,
801                    poll_duration,
802                    do_get_cost,
803                    total_cost,
804                };
805
806                // Push RegionMetrics to global partition_metrics immediately after scanning this region
807                {
808                    let mut partition_metrics_guard = partition_metrics_moved.lock().unwrap();
809                    let partition_metrics = partition_metrics_guard
810                        .entry(partition)
811                        .or_insert_with(|| PartitionMetrics::new(partition, explain_verbose));
812                    partition_metrics.add_region_metrics(region_metrics);
813                }
814
815                if explain_verbose {
816                    common_telemetry::info!(
817                        "Merge scan finish one region, partition: {}, region_id: {}, poll_duration: {:?}, first_consume: {}, do_get_cost: {:?}",
818                        partition,
819                        region_id,
820                        poll_duration,
821                        metric.first_consume_time(),
822                        do_get_cost
823                    );
824                }
825
826                // process metrics after all data is drained.
827                if let Some(metrics) = stream.metrics() {
828                    let load = region_scan_load(&metrics);
829                    let (c, s) = parse_catalog_and_schema_from_db_string(&dbname);
830                    let value = read_meter!(c, s, load, current_channel as u8);
831                    metric.record_greptime_exec_cost(value as usize);
832
833                    // record metrics from sub sgates
834                    let mut sub_stage_metrics = sub_stage_metrics_moved.lock().unwrap();
835                    sub_stage_metrics.insert(region_id, metrics);
836                }
837
838                MERGE_SCAN_POLL_ELAPSED.observe(poll_duration.as_secs_f64());
839            }
840
841            // Stop the global timers for partitions with no region, otherwise they keep
842            // running until drop and inflate the shared metrics. No-op otherwise.
843            ready_timer.stop();
844            if let Some(mut first_consume_timer) = first_consume_timer.take() {
845                first_consume_timer.stop();
846            }
847
848            // Finish partition metrics and log results
849            let partition_finish_time = partition_start.elapsed();
850            {
851                let mut partition_metrics_guard = partition_metrics_moved.lock().unwrap();
852                if let Some(partition_metrics) = partition_metrics_guard.get_mut(&partition) {
853                    partition_metrics.set_timings(
854                        partition_ready_time.unwrap_or_default(),
855                        partition_first_consume_time.unwrap_or_default(),
856                        partition_finish_time,
857                    );
858                    partition_metrics.finish();
859                }
860            }
861        }));
862
863        Ok(Box::pin(RecordBatchStreamAdapter::new(
864            self.arrow_schema.clone(),
865            stream,
866        )))
867    }

869    pub fn try_with_new_distribution(&self, distribution: Distribution) -> Option<Self> {
870        let Distribution::HashPartitioned(hash_exprs) = distribution else {
871            // not applicable
872            return None;
873        };
874
875        if let Partitioning::Hash(curr_dist, _) = &self.properties.partitioning
876            && curr_dist == &hash_exprs
877        {
878            // No need to change the distribution
879            return None;
880        }
881
882        let hash_expr_col_names: HashSet<_> = hash_exprs
883            .iter()
884            .filter_map(|expr| {
885                expr.as_any()
886                    .downcast_ref::<Column>()
887                    .map(|col_expr| col_expr.name())
888            })
889            .collect();
890
891        let covers_all_partition_cols = self.partition_cols.values().all(|aliases| {
892            aliases
893                .iter()
894                .any(|col| hash_expr_col_names.contains(col.name()))
895        });
896        if !covers_all_partition_cols {
897            return None;
898        }
899
900        let all_partition_col_aliases: HashSet<_> = self
901            .partition_cols
902            .values()
903            .flat_map(|aliases| aliases.iter().map(|c| c.name()))
904            .collect();
905        let overlaps: Vec<_> = hash_exprs
906            .iter()
907            .filter(|expr| {
908                expr.as_any()
909                    .downcast_ref::<Column>()
910                    .is_some_and(|col_expr| all_partition_col_aliases.contains(col_expr.name()))
911            })
912            .cloned()
913            .collect();
914
915        if overlaps.is_empty() {
916            return None;
917        }
918
919        Some(Self {
920            table: self.table.clone(),
921            regions: self.regions.clone(),
922            plan: self.plan.clone(),
923            arrow_schema: self.arrow_schema.clone(),
924            region_query_handler: self.region_query_handler.clone(),
925            metric: self.metric.clone(),
926            properties: Arc::new(PlanProperties::new(
927                self.properties.eq_properties.clone(),
928                Partitioning::Hash(overlaps, self.partition_count()),
929                self.properties.emission_type,
930                self.properties.boundedness,
931            )),
932            sub_stage_metrics: self.sub_stage_metrics.clone(),
933            partition_metrics: self.partition_metrics.clone(),
934            query_ctx: self.query_ctx.clone(),
935            remote_dyn_filter_producer_id: self.remote_dyn_filter_producer_id,
936            captured_remote_dyn_filters: self.captured_remote_dyn_filters.clone(),
937            target_partition: self.target_partition,
938            partition_cols: self.partition_cols.clone(),
939            enable_per_region_metrics: self.enable_per_region_metrics,
940        })
941    }

996// If the schema has JSON2 field, AND the field is of empty Struct datatype, amend it with Binary
997// datatype.
998// This is a very hacky way to make it possible to query the whole JSON2 column. Because when
999// querying a whole JSON2 column, like in the SQL `select * from ...`, we can't concretize the JSON2
1000// datatype from the query. Hence, the JSON2 datatype remains what in the column schema, i.e., empty
1001// Struct. An empty Struct is not alignable like any other concretized JSON2 datatypes, so to make
1002// the query work, we amend(rewrite) it to Binary datatype.
1003// Why the Binary datatype? Because underlying the scan and projection stage, the JSON2 data are
1004// variant shape, will be all converted to bytes.
1005// Anyway, this is not clean nor elegant. TODO(LFC) Maybe make it into some plan analyzer rule?
1006fn maybe_amend_json2_field(schema: &ArrowSchema) -> ArrowSchemaRef {
1007    let schema = schema.clone();
1008    let mut new_fields = Vec::with_capacity(schema.fields().len());
1009    for field in schema.fields().iter() {
1010        let new_field = if is_json2_extension_type(field)
1011            && matches!(field.data_type(), DataType::Struct(fields) if fields.is_empty())
1012        {
1013            let is_legacy_json2 = is_legacy_json2_extension_type(field);
1014            let mut new_field = field.as_ref().clone();
1015            new_field.set_data_type(DataType::Binary);
1016            if is_legacy_json2 {
1017                // Pre-type-hint JSON2 is identified partly by its Struct data type. Promote the
1018                // ephemeral field before rewriting it to Binary so later checks retain its JSON2
1019                // identity.
1020                new_field = new_field.with_extension_type(Json2ExtensionType::default());
1021            }
1022            Arc::new(new_field)
1023        } else {
1024            field.clone()
1025        };
1026        new_fields.push(new_field);
1027    }
1028    Arc::new(ArrowSchema::new_with_metadata(
1029        new_fields,
1030        schema.metadata().clone(),
1031    ))
1032}

1336        // controlled by `enable_per_region_metrics`. Region-owned counters for
1337        // heartbeat reporting are updated on datanodes when query metrics resolve.
1338        if !self.enable_per_region_metrics {

1343        for (region_id, metrics) in metrics.iter() {

1345            report_region_query_load(query_load_region_id(*region_id, metrics), &load);
1346        }
1347    }
