158    #[allow(clippy::too_many_arguments)]
159    pub fn try_new(
160        catalog_list: CatalogManagerRef,
161        partition_rule_manager: Option<PartitionRuleManagerRef>,
162        region_query_handler: Option<RegionQueryHandlerRef>,
163        table_mutation_handler: Option<TableMutationHandlerRef>,
164        procedure_service_handler: Option<ProcedureServiceHandlerRef>,
165        flow_service_handler: Option<FlowServiceHandlerRef>,
166        with_dist_planner: bool,
167        plugins: Plugins,
168        options: QueryOptionsNew,
169    ) -> DfResult<Self> {
170        let total_memory = get_total_memory_bytes().max(0) as u64;
171        let memory_pool_size = options.memory_pool_size.resolve(total_memory) as usize;
172        let runtime_provider = plugins.get::<QueryRuntimeProviderRef>();
173        let mut session_config = SessionConfig::new().with_create_default_catalog_and_schema(false);
174        if options.parallelism > 0 {
175            session_config = session_config.with_target_partitions(options.parallelism);
176        }
177        if options.experimental_enable_factorized_join_agg {
178            session_config
179                .options_mut()
180                .extensions
181                .insert(FactorizedJoinAggOptions { enabled: true });
182        }
183        if options.allow_query_fallback {
184            session_config
185                .options_mut()
186                .extensions
187                .insert(DistPlannerOptions {
188                    allow_query_fallback: true,
189                });
190        }
191
192        // todo(hl): This serves as a workaround for https://github.com/GreptimeTeam/greptimedb/issues/5659
193        // and we can add that check back once we upgrade datafusion.
194        session_config
195            .options_mut()
196            .execution
197            .skip_physical_aggregate_schema_check = true;
198
199        let runtime_context = QueryRuntimeContext::new(&options, memory_pool_size);
200        let default_runtime_provider = DefaultQueryRuntimeProvider;
201        default_runtime_provider.configure_session_config(runtime_context, &mut session_config);
202        if let Some(provider) = runtime_provider.as_ref() {
203            provider.configure_session_config(runtime_context, &mut session_config);
204        }
205        let runtime_builder = DefaultQueryRuntimeProvider::runtime_env_builder(runtime_context);
206        let runtime_env = match runtime_provider {
207            Some(provider) => provider.build_runtime_env(runtime_context, runtime_builder)?,
208            None => default_runtime_provider.build_runtime_env(runtime_context, runtime_builder)?,
209        };
210
211        // Apply extension rules
212        let mut extension_rules = Vec::new();
213
214        // The [`TypeConversionRule`] must be at first
215        extension_rules.insert(0, Arc::new(TypeConversionRule) as _);
216        extension_rules.push(Arc::new(CountNestAggrRule) as _);
217
218        // Apply the datafusion rules
219        let mut analyzer = Analyzer::new();
220        analyzer.rules.insert(0, Arc::new(TranscribeAtatRule));
221        analyzer.rules.insert(0, Arc::new(StringNormalizationRule));
222        analyzer
223            .rules
224            .insert(0, Arc::new(CountWildcardToTimeIndexRule));
225        analyzer.rules.push(Arc::new(ConstNormalizationRule));
226
227        // Add ApplyFunctionRewrites rule,
228        // Note we cannot use `analyzer.add_function_rewrite`
229        // because only rules are copied into session_state
230        analyzer.rules.insert(
231            0,
232            Arc::new(ApplyFunctionRewrites::new(
233                FUNCTION_REGISTRY.function_rewrites(),
234            )),
235        );
236        if inspect_query_extension(&catalog_list).is_some() {
237            analyzer.rules.push(Arc::new(SstManifestAggregateRule));
238        }
239        if with_dist_planner {
240            analyzer
241                .rules
242                .push(Arc::new(DistPlannerAnalyzer::default()));
243            analyzer.rules.push(Arc::new(JsonSchemaConcretizeRule));
244        }
245        analyzer.rules.push(Arc::new(FixStateUdafOrderingAnalyzer));
246        analyzer.rules.push(Arc::new(expr_depth::ExprDepthRule));
247
248        // Note: Postgres oid-alias string coercion (`'x'::regclass`,
249        // `'public'::regnamespace`, ...) is handled by the
250        // `PostgresCompatibilityParser`'s built-in `RewriteRegCastToSubquery`
251        // rule at SQL-parse time, so no analyzer rule is registered here. The
252        // implicit `oid_col = 'name'` form is not resolved (it needs schema
253        // awareness the parser lacks); the few client queries that use it are
254        // handled by the parser's blacklist.
255
256        let mut optimizer = Optimizer::new();
257        // Constants must be simplified, but CASE arguments must not yet be
258        // hidden by common-subexpression extraction.
259        let filter_pushdown_index = optimizer
260            .rules
261            .iter()
262            .position(|rule| rule.name() == "push_down_filter")
263            .ok_or_else(|| {
264                datafusion_common::internal_datafusion_err!("Missing PushDownFilter rule")
265            })?;
266        optimizer.rules.insert(
267            filter_pushdown_index,
268            Arc::new(InferAggregateInputBoundsRule),
269        );
270        if with_dist_planner {
271            optimizer.rules.push(Arc::new(MergeScanProjection));
272        }
273        optimizer.rules.push(Arc::new(ScanHintRule));
274        optimizer.rules.push(Arc::new(JsonTypeConcretizeRule));
275        optimizer.rules.push(Arc::new(FactorizedJoinAggRule));
276
277        // add physical optimizer
278        let mut physical_optimizer = PhysicalOptimizer::new();
279        // Change TableScan's partition right before enforcing distribution
280        physical_optimizer
281            .rules
282            .insert(5, Arc::new(ParallelizeScan));
283        // Prefer collecting narrow PromQL build sides over repartitioning wide label streams.
284        physical_optimizer
285            .rules
286            .insert(6, Arc::new(PromqlTsidNarrowJoin));
287        // Re-enforce sorting after custom rules update scan partitioning and distribution.
288        // Keep it immediately before DataFusion's default EnsureRequirements.
289        physical_optimizer.rules.insert(7, Arc::new(EnforceSorting));
290        // Add rule for windowed sort
291        physical_optimizer
292            .rules
293            .push(Arc::new(WindowedSortPhysicalRule));
294        // explicitly not do filter pushdown for windowed sort&part sort
295        // (notice that `PartSortExec` create another new dyn filter that need to be pushdown if want to use dyn filter optimization)
296        // benchmark shows it can cause performance regression due to useless filtering and extra shuffle.
297        // We can add a rule to do filter pushdown for windowed sort in the future if we find a way to avoid the performance regression.
298        physical_optimizer
299            .rules
300            .push(Arc::new(MatchesConstantTermOptimizer));
301        physical_optimizer
302            .rules
303            .push(Arc::new(EnsureGlobalLimitForFetch));
304        physical_optimizer.rules.push(Arc::new(TopKFilterBatchSize));
305        physical_optimizer
306            .rules
307            .push(Arc::new(GlobalTopKOverFilteredScan));
308        // Add rule to remove duplicate nodes generated by other rules. Run this in the last.
309        physical_optimizer.rules.push(Arc::new(RemoveDuplicate));
310        // Place SanityCheckPlan at the end of the list to ensure that it runs after all other rules.
311        Self::remove_physical_optimizer_rule(
312            &mut physical_optimizer.rules,
313            SanityCheckPlan {}.name(),
314        );
315        physical_optimizer.rules.push(Arc::new(SeriesFrameOutput));
316        physical_optimizer.rules.push(Arc::new(SanityCheckPlan {}));
317
318        let session_state = SessionStateBuilder::new()
319            .with_config(session_config)
320            .with_runtime_env(runtime_env)
321            .with_default_features()
322            .with_analyzer_rules(analyzer.rules)
323            .with_serializer_registry(Arc::new(DefaultSerializer))
324            .with_query_planner(Arc::new(DfQueryPlanner::new(
325                catalog_list.clone(),
326                partition_rule_manager.clone(),
327                region_query_handler.clone(),
328                options.enable_per_region_metrics,
329            )))
330            .with_optimizer_rules(optimizer.rules)
331            .with_physical_optimizer_rules(physical_optimizer.rules)
332            .build();
333        let df_context = SessionContext::new_with_state(session_state);
334        register_function_aliases(&df_context);
335        register_pg_catalog_compat(&df_context);
336
337        Ok(Self {
338            df_context,
339            catalog_manager: catalog_list,
340            dyn_filter_registry_manager: Arc::new(DynFilterRegistryManager::default()),
341            function_state: Arc::new(FunctionState {
342                plugins: plugins.clone(),
343                table_mutation_handler,
344                procedure_service_handler,
345                flow_service_handler,
346            }),
347            aggr_functions: Arc::new(RwLock::new(HashMap::new())),
348            table_functions: Arc::new(RwLock::new(HashMap::new())),
349            extension_rules,
350            plugins,
351            experimental_native_promql: options.experimental_native_promql,
352            with_dist_planner,
353            scalar_functions: Arc::new(RwLock::new(HashMap::new())),
354        })
355    }

364    /// Applies validated query options to one per-query DataFusion configuration.
365    pub fn apply_query_options(
366        &self,
367        state: &mut SessionState,
368        query_ctx: QueryContextRef,
369    ) -> crate::error::Result<QueryContextRef> {
370        let query_ctx = Arc::new(query_ctx.query_option_snapshot().map_err(|error| {
371            InvalidQueryContextExtensionSnafu {
372                reason: error.to_string(),
373            }
374            .build()
375        })?);
376        let options = query_ctx.extensions();
377        const MASTER: &str = "datafusion.optimizer.enable_dynamic_filter_pushdown";
378        // The master switch cascades to children, so apply explicit child overrides afterward.
379        for (key, value) in options
380            .get_key_value(MASTER)
381            .into_iter()
382            .chain(options.iter().filter(|(key, _)| key.as_str() != MASTER))
383        {
384            if key.starts_with("datafusion.") {
385                state
386                    .config_mut()
387                    .options_mut()
388                    .set(key, value)
389                    .map_err(|error| {
390                        InvalidQueryContextExtensionSnafu {
391                            reason: error.to_string(),
392                        }
393                        .build()
394                    })?;
395            }
396        }
397        if let Some(partitions) = options.get("query.parallelism") {
398            let partitions = partitions.parse::<usize>().map_err(|error| {
399                InvalidQueryContextExtensionSnafu {
400                    reason: error.to_string(),
401                }
402                .build()
403            })?;
404            *state.config_mut() = state.config().clone().with_target_partitions(partitions);
405        }
406        if let Some(fallback) = options.get("query.allow_query_fallback") {
407            state
408                .config_mut()
409                .options_mut()
410                .extensions
411                .insert(DistPlannerOptions {
412                    allow_query_fallback: fallback == "true",
413                });
414        }
415        // The native PromQL route resolves like the other query options: a request value over a
416        // session `SET` over the engine-wide default. It is recorded here, once, so that planning
417        // and `SHOW query.experimental_native_promql` read the same value instead of each resolving
418        // the sources again.
419        let native_promql = match query_ctx.extension(NATIVE_PROMQL_EXTENSION_KEY) {
420            // The snapshot states the option: the request value, or the session `SET` it inherits.
421            Some(_) => native_promql_requested(&query_ctx),
422            // Neither source states it, so the engine-wide default answers.
423            None => self.experimental_native_promql(),
424        };
425        state
426            .config_mut()
427            .options_mut()
428            .extensions
429            .insert(NativePromqlExtension {
430                requested: native_promql,
431            });
432        if let Some(enabled) = options.get("query.experimental_enable_factorized_join_agg") {
433            state
434                .config_mut()
435                .options_mut()
436                .extensions
437                .insert(FactorizedJoinAggOptions {
438                    enabled: enabled == "true",
439                });
440        }
441        Ok(query_ctx)
442    }

444    /// Optimize the logical plan using registered extension analyzer rules.
445    pub fn optimize_by_extension_rules(
446        &self,
447        plan: DfLogicalPlan,
448        context: &QueryEngineContext,
449    ) -> DfResult<DfLogicalPlan> {
450        expr_depth::validate_plan(&plan)?;
451        self.extension_rules
452            .iter()
453            .try_fold(plan, |acc_plan, rule| {
454                let plan = rule.analyze(acc_plan, context, context.state().config_options())?;
455                expr_depth::validate_plan(&plan)?;
456                Ok(plan)
457            })
458    }

538    /// Register an aggregate function.
539    ///
540    /// # Panics
541    /// Will panic if the function with same name is already registered.
542    ///
543    /// Panicking consideration: currently the aggregated functions are all statically registered,
544    /// user cannot define their own aggregate functions on the fly. So we can panic here. If that
545    /// invariant is broken in the future, we should return an error instead of panicking.

551            .unwrap()
552            .insert(name.clone(), func);
553        assert!(
554            x.is_none(),
555            "Already registered aggregate function '{name}'"
556        );
557    }

664/// MySQL-compatible scalar function aliases: (target_name, alias)
665const SCALAR_FUNCTION_ALIASES: &[(&str, &str)] = &[
666    ("upper", "ucase"),
667    ("lower", "lcase"),
668    ("ceil", "ceiling"),
669    ("substr", "mid"),
670    ("random", "rand"),
671];
672
673/// MySQL-compatible aggregate function aliases: (target_name, alias)
674const AGGREGATE_FUNCTION_ALIASES: &[(&str, &str)] =
675    &[("stddev_pop", "std"), ("var_pop", "variance")];
676
677/// Register function aliases.
678///
679/// This function adds aliases like `ucase` -> `upper`, `lcase` -> `lower`, etc.
680/// to make GreptimeDB more compatible with MySQL syntax.
681fn register_function_aliases(ctx: &SessionContext) {
682    let state = ctx.state();
683
684    for (target, alias) in SCALAR_FUNCTION_ALIASES {
685        if let Some(func) = state.scalar_functions().get(*target) {
686            let aliased = func.as_ref().clone().with_aliases([*alias]);
687            ctx.register_udf(aliased);
688        }
689    }
690
691    for (target, alias) in AGGREGATE_FUNCTION_ALIASES {
692        if let Some(func) = state.aggregate_functions().get(*target) {
693            let aliased = func.as_ref().clone().with_aliases([*alias]);
694            ctx.register_udaf(aliased);
695        }
696    }
697}

699/// Register the session-level (not function-registry) Postgres-compatibility
700/// extension sourced from `datafusion-pg-catalog`: widen Postgres `int4`
701/// bounds to `int8` for `generate_series`/`range` (their stock implementations
702/// require `int8` literals, and DF invokes them at planning time, before any
703/// analyzer rule can run).
704///
705/// The oid-alias type planner is supplied per SQL-planner context by
706/// [`DfContextProviderAdapter::get_type_planner`](crate::datafusion::planner::DfContextProviderAdapter::get_type_planner).
707/// Forward oid-alias name->oid resolution is handled at SQL-parse time by the
708/// `PostgresCompatibilityParser`'s built-in rewrite rule, not here.
709fn register_pg_catalog_compat(ctx: &SessionContext) {
710    datafusion_pg_catalog::pg_catalog::generate_series_arg_coercion::CoerceIntArgsToBigInt::widen(
711        ctx,
712    );
713}

755/// A wrapper around a memory pool that records metrics.
756///
757/// This wrapper intercepts all memory pool operations and updates
758/// Prometheus metrics for monitoring query memory usage and rejections.
759///
760/// The inner pool is wrapped with `TrackConsumersPool` to preserve
761/// top-consumer error context on rejection.
762#[derive(Debug)]
763pub(super) struct MetricsMemoryPool {
764    inner: Arc<dyn MemoryPool>,
765}
766
767impl MetricsMemoryPool {
768    // Number of top memory consumers to report in OOM error messages
769    const TOP_CONSUMERS_TO_REPORT: usize = 5;
770
771    /// Create a new metrics-wrapped memory pool with the given size limit and
772    /// allocation policy.
773    pub(super) fn new(limit: usize, policy: QueryMemoryPoolPolicy) -> Self {
774        let top_n = NonZeroUsize::new(Self::TOP_CONSUMERS_TO_REPORT).unwrap();
775        let inner: Arc<dyn MemoryPool> = match policy {
776            QueryMemoryPoolPolicy::Greedy => {
777                Arc::new(TrackConsumersPool::new(GreedyMemoryPool::new(limit), top_n))
778            }
779            QueryMemoryPoolPolicy::Fair => {
780                Arc::new(TrackConsumersPool::new(FairSpillPool::new(limit), top_n))
781            }
782        };
783        Self { inner }
784    }
785
786    #[inline]
787    fn update_metrics(&self) {
788        QUERY_MEMORY_POOL_USAGE_BYTES.set(self.inner.reserved() as i64);
789    }
790}
791
792impl fmt::Display for MetricsMemoryPool {
793    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
794        write!(f, "{}(inner_pool: {})", self.name(), self.inner)
795    }
796}

801    }

804        self.inner.register(consumer);
805    }

808        self.inner.unregister(consumer);
809    }

811    fn grow(&self, reservation: &MemoryReservation, additional: usize) {
812        self.inner.grow(reservation, additional);
813        self.update_metrics();
814    }
815
816    fn shrink(&self, reservation: &MemoryReservation, shrink: usize) {
817        self.inner.shrink(reservation, shrink);
818        self.update_metrics();
819    }

821    fn try_grow(
822        &self,
823        reservation: &MemoryReservation,
824        additional: usize,
825    ) -> datafusion_common::Result<()> {
826        let result = self.inner.try_grow(reservation, additional);
827        if result.is_err() {
828            QUERY_MEMORY_POOL_REJECTED_TOTAL.inc();
829        }
830        self.update_metrics();
831        result
832    }

835        self.inner.reserved()
836    }

840    }

1343    /// Builds a runtime with custom spill configuration and a bounded memory
1344    /// pool, then runs sort queries that probe spill-to-disk behaviour:
1345    ///
1346    /// - A pool large enough for merge chunks but still much smaller
1347    ///   than the total data.  Executes the physical plan directly so we can
1348    ///   walk the plan tree afterwards and sum `spill_count` / `spilled_rows` /
1349    ///   `spilled_bytes` on `SortExec` nodes.  All three must be > 0.
1350    #[tokio::test]
1351    async fn test_sort_spill_smoke_with_custom_runtime() {
1352        use arrow::array::Int32Array;
1353        use arrow::datatypes::{DataType, Field, Schema};
1354        use arrow::record_batch::RecordBatch;
1355        use datafusion::datasource::MemTable;
1356        use datafusion::physical_plan::collect as df_collect;
1357
1358        let spill_dir =
1359            std::env::temp_dir().join(format!("greptime_spill_smoke_{}", std::process::id()));
1360        let _ = std::fs::remove_dir_all(&spill_dir);
1361
1362        let opts = QueryOptions {
1363            experimental_spill_mode: QuerySpillMode::Custom,
1364            experimental_spill_path: Some(spill_dir.clone()),
1365            experimental_spill_max_temp_directory_size: ReadableSize::gb(1),
1366            experimental_memory_pool_policy: QueryMemoryPoolPolicy::Greedy,
1367            ..Default::default()
1368        };
1369
1370        let session_config = SessionConfig::new()
1371            .with_target_partitions(1)
1372            .with_sort_in_place_threshold_bytes(0)
1373            .with_sort_spill_reservation_bytes(64 * 1024);
1374
1375        let schema = Arc::new(Schema::new(vec![
1376            Field::new("id", DataType::Int32, false),
1377            Field::new("val", DataType::Int32, false),
1378        ]));
1379        let n_rows: i32 = 200_000;
1380        let batch_size: i32 = 5_000;
1381        let partitions = 1usize;
1382
1383        let mut table_partitions: Vec<Vec<RecordBatch>> = Vec::with_capacity(partitions);
1384        for _ in 0..partitions {
1385            let mut batches = Vec::new();
1386            let mut row_offset: i32 = 0;
1387            while row_offset < n_rows {
1388                let chunk_end = (row_offset + batch_size).min(n_rows);
1389                let chunk_len = (chunk_end - row_offset) as usize;
1390                let mut id_builder = Int32Array::builder(chunk_len);
1391                let mut val_builder = Int32Array::builder(chunk_len);
1392                for i in row_offset..chunk_end {
1393                    id_builder.append_value(i);
1394                    val_builder.append_value(n_rows - 1 - i);
1395                }
1396                batches.push(
1397                    RecordBatch::try_new(
1398                        Arc::clone(&schema),
1399                        vec![
1400                            Arc::new(id_builder.finish()),
1401                            Arc::new(val_builder.finish()),
1402                        ],
1403                    )
1404                    .unwrap(),
1405                );
1406                row_offset = chunk_end;
1407            }
1408            table_partitions.push(batches);
1409        }
1410
1411        fn build_ctx(
1412            session_config: &SessionConfig,
1413            runtime: &Arc<RuntimeEnv>,
1414            schema: &Arc<Schema>,
1415            partitions: &[Vec<RecordBatch>],
1416        ) -> SessionContext {
1417            let session_state = SessionStateBuilder::new()
1418                .with_config(session_config.clone())
1419                .with_runtime_env(Arc::clone(runtime))
1420                .with_default_features()
1421                .build();
1422            let ctx = SessionContext::new_with_state(session_state);
1423            let table = MemTable::try_new(Arc::clone(schema), partitions.to_vec()).unwrap();
1424            ctx.register_table("t", Arc::new(table)).unwrap();
1425            ctx
1426        }
1427
1428        {
1429            let mem_limit: usize = 512 * 1024; // 512 KB pool, 64 KB reserved for merge
1430
1431            let runtime = build_runtime_env(&opts, mem_limit);
1432            assert_eq!(runtime.memory_pool.reserved(), 0);
1433            let ctx = build_ctx(&session_config, &runtime, &schema, &table_partitions);
1434
1435            let df = ctx
1436                .sql("SELECT val FROM t ORDER BY val ASC")
1437                .await
1438                .expect("planning ORDER BY");
1439            let plan = df
1440                .create_physical_plan()
1441                .await
1442                .expect("creating physical plan");
1443            let task_ctx = ctx.task_ctx();
1444
1445            let batches = df_collect(Arc::clone(&plan), task_ctx)
1446                .await
1447                .expect("executing ORDER BY");
1448
1449            let (spill_count, spilled_rows, spilled_bytes) = sum_sort_spill_metrics(&plan);
1450            assert!(
1451                spill_count > 0,
1452                "expected SortExec spill_count > 0 (pool={} KB, reservation=64 KB), got 0",
1453                mem_limit / 1024,
1454            );
1455            assert!(
1456                spilled_rows > 0,
1457                "expected SortExec spilled_rows > 0, got 0 \
1458                 (spill_count={spill_count}, spilled_bytes={spilled_bytes})",
1459            );
1460            assert!(
1461                spilled_bytes > 0,
1462                "expected SortExec spilled_bytes > 0, got 0 \
1463                 (spill_count={spill_count}, spilled_rows={spilled_rows})",
1464            );
1465
1466            let vals: Vec<i32> = batches
1467                .iter()
1468                .flat_map(|b| {
1469                    let col = b.column(0).as_any().downcast_ref::<Int32Array>().unwrap();
1470                    (0..b.num_rows()).map(move |i| col.value(i))
1471                })
1472                .collect();
1473            assert_eq!(vals.len(), n_rows as usize);
1474            for w in vals.windows(2) {
1475                assert!(w[0] <= w[1], "sort order violation: {} > {}", w[0], w[1]);
1476            }
1477
1478            assert!(
1479                std::fs::read_dir(&spill_dir)
1480                    .ok()
1481                    .map(|mut entries| entries.any(|e| {
1482                        e.as_ref()
1483                            .map(|de| de.file_name().to_string_lossy().starts_with("datafusion-"))
1484                            .unwrap_or(false)
1485                    }))
1486                    .unwrap_or(false),
1487                "Expected 'datafusion-*' directory in spill path {:?} \
1488                 (DiskManager should create it on build).",
1489                spill_dir,
1490            );
1491
1492            assert_eq!(runtime.memory_pool.reserved(), 0);
1493        }
1494
1495        let _ = std::fs::remove_dir_all(&spill_dir);
1496    }

1498    /// Walk a physical plan tree, summing spill metrics for every node whose
1499    /// name starts with `"SortExec"`.
1500    fn sum_sort_spill_metrics(plan: &Arc<dyn ExecutionPlan>) -> (usize, usize, usize) {
1501        let mut spill_count = 0usize;
1502        let mut spilled_rows = 0usize;
1503        let mut spilled_bytes = 0usize;
1504
1505        fn walk(plan: &Arc<dyn ExecutionPlan>, sc: &mut usize, sr: &mut usize, sb: &mut usize) {
1506            let name = plan.name();
1507            if name.starts_with("SortExec")
1508                && let Some(m) = plan.metrics()
1509            {
1510                *sc += m.spill_count().unwrap_or(0);
1511                *sr += m.spilled_rows().unwrap_or(0);
1512                *sb += m.spilled_bytes().unwrap_or(0);
1513            }
1514            for child in plan.children() {
1515                walk(child, sc, sr, sb);
1516            }
1517        }
1518
1519        walk(
1520            plan,
1521            &mut spill_count,
1522            &mut spilled_rows,
1523            &mut spilled_bytes,
1524        );
1525        (spill_count, spilled_rows, spilled_bytes)
1526    }
