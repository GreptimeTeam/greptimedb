44/// Query spill mode controlling disk manager behavior.
45#[derive(Debug, Clone, Copy, Serialize, Deserialize, PartialEq, Eq)]
46#[serde(rename_all = "snake_case")]
47pub enum QuerySpillMode {
48    /// Preserve DataFusion default disk manager behavior (OS temp directory).
49    Default,
50    /// Explicitly configure spill path, quota, and compression.
51    Custom,
52    /// Explicitly disable disk spilling; temporary file creation will error.
53    Disabled,
54}

56/// Compression for spilled data files.
57#[derive(Debug, Clone, Copy, Serialize, Deserialize, PartialEq, Eq)]
58#[serde(rename_all = "snake_case")]
59pub enum QuerySpillCompression {
60    /// No compression (default, matches DataFusion default).
61    Uncompressed,
62    /// LZ4 frame compression.
63    Lz4Frame,
64    /// Zstandard compression.
65    Zstd,
66}

68/// Memory pool allocation policy.
69#[derive(Debug, Clone, Copy, Serialize, Deserialize, PartialEq, Eq)]
70#[serde(rename_all = "snake_case")]
71pub enum QueryMemoryPoolPolicy {
72    /// Greedy first-come-first-served (default).
73    Greedy,
74    /// Fair divides memory available after unspillable reservations evenly among
75    /// spillable reservations and may trigger earlier spills.
76    Fair,
77}

79/// Query engine config
80#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
81#[serde(default)]
82pub struct QueryOptions {
83    /// Parallelism of query engine. Default to 0, which implies the number of logical CPUs.
84    pub parallelism: usize,
85    /// Whether to allow query fallback when push down fails.
86    pub allow_query_fallback: bool,
87    /// Memory pool size for query execution. Setting it to 0 disables the limit (unbounded).
88    /// Supports absolute size (e.g., "2GB") or percentage (e.g., "50%").
89    /// When this limit is reached, queries will fail with ResourceExhausted error.
90    pub memory_pool_size: MemoryLimit,
91    /// Whether to expose per-region query load metrics.
92    #[serde(skip)]
93    pub enable_per_region_metrics: bool,
94    /// Experimental: spill-to-disk mode.
95    /// - `default`: preserve DataFusion built-in OS temp directory behavior.
96    /// - `custom`: explicitly configure spill path, max directory size, and compression.
97    /// - `disabled`: explicitly disable disk spilling.
98    pub experimental_spill_mode: QuerySpillMode,
99    /// Experimental: spill directory path. Ignored unless `experimental_spill_mode` is
100    /// `"custom"`. When set, spill files are written into this directory.
101    pub experimental_spill_path: Option<PathBuf>,
102    /// Experimental: maximum total size of the spill directory (data written to spill files).
103    /// Ignored unless `experimental_spill_mode` is `"custom"`. Default: `1GiB`.
104    pub experimental_spill_max_temp_directory_size: ReadableSize,
105    /// Experimental: compression algorithm applied to spilled data.
106    /// Ignored unless `experimental_spill_mode` is `"custom"`. Default: `uncompressed`.
107    pub experimental_spill_compression: QuerySpillCompression,
108    /// Experimental: memory pool allocation policy.
109    /// - `greedy`: Greedy first-come-first-served (default).
110    /// - `fair`: Fair divides memory available after unspillable reservations
111    ///   evenly among spillable reservations and may trigger earlier spills.
112    pub experimental_memory_pool_policy: QueryMemoryPoolPolicy,
113}

138#[derive(Debug, Clone, PartialEq, Eq, Default)]
139pub struct FlowQueryExtensions {
140    /// Maps region id -> lower exclusive sequence bound for incremental reads.
141    pub incremental_after_seqs: Option<HashMap<u64, u64>>,
142    /// Incremental read mode requested by the caller.
143    pub incremental_mode: Option<FlowIncrementalMode>,
144    /// Whether the caller expects per-region watermark metadata in terminal metrics.
145    pub return_region_seq: bool,
146    /// Optional sink table id used to distinguish source scans from sink reads.
147    pub sink_table_id: Option<TableId>,
148}
149
150impl FlowQueryExtensions {
151    /// Parses flow-specific query extensions when any flow key is present.
152    ///
153    /// Returns `Ok(None)` for ordinary queries with no flow-related extensions,
154    /// `Ok(Some(_))` when flow context is present and valid, and `Err(_)` when a
155    /// flow-related extension is present but malformed or incomplete.
156    pub fn parse_flow_extensions(extensions: &HashMap<String, String>) -> Result<Option<Self>> {
157        let has_flow_context = extensions.contains_key(FLOW_INCREMENTAL_AFTER_SEQS)
158            || extensions.contains_key(FLOW_INCREMENTAL_MODE)
159            || extensions.contains_key(FLOW_RETURN_REGION_SEQ)
160            || extensions.contains_key(FLOW_SINK_TABLE_ID);
161
162        if !has_flow_context {
163            return Ok(None);
164        }
165
166        let incremental_mode = extensions
167            .get(FLOW_INCREMENTAL_MODE)
168            .map(|value| match value.as_str() {
169                v if v.eq_ignore_ascii_case(FLOW_INCREMENTAL_MODE_MEMTABLE_ONLY) => {
170                    Ok(FlowIncrementalMode::MemtableOnly)
171                }
172                v if v.eq_ignore_ascii_case(FLOW_INCREMENTAL_MODE_SEQUENCE_RANGE) => {
173                    Ok(FlowIncrementalMode::SequenceRange)
174                }
175                _ => Err(invalid_query_context_extension(format!(
176                    "Invalid value for {}: {}",
177                    FLOW_INCREMENTAL_MODE, value
178                ))),
179            })
180            .transpose()?;
181
182        let incremental_after_seqs = extensions
183            .get(FLOW_INCREMENTAL_AFTER_SEQS)
184            .map(|value| parse_incremental_after_seqs(value.as_str()))
185            .transpose()?;
186
187        let return_region_seq = extensions
188            .get(FLOW_RETURN_REGION_SEQ)
189            .map(|value| parse_bool(FLOW_RETURN_REGION_SEQ, value.as_str()))
190            .transpose()?
191            .unwrap_or(false);
192
193        let sink_table_id = extensions
194            .get(FLOW_SINK_TABLE_ID)
195            .map(|value| {
196                value.parse::<TableId>().map_err(|_| {
197                    invalid_query_context_extension(format!(
198                        "Invalid value for {}: {}",
199                        FLOW_SINK_TABLE_ID, value
200                    ))
201                })
202            })
203            .transpose()?;
204
205        if matches!(
206            incremental_mode,
207            Some(FlowIncrementalMode::MemtableOnly | FlowIncrementalMode::SequenceRange)
208        ) {
209            let mode = if incremental_mode == Some(FlowIncrementalMode::MemtableOnly) {
210                FLOW_INCREMENTAL_MODE_MEMTABLE_ONLY
211            } else {
212                FLOW_INCREMENTAL_MODE_SEQUENCE_RANGE
213            };
214            let after_seqs = incremental_after_seqs.as_ref().ok_or_else(|| {
215                invalid_query_context_extension(format!(
216                    "{} is required when {}={}.",
217                    FLOW_INCREMENTAL_AFTER_SEQS, FLOW_INCREMENTAL_MODE, mode
218                ))
219            })?;
220            if after_seqs.is_empty() {
221                return Err(invalid_query_context_extension(format!(
222                    "{} must not be empty when {}={}.",
223                    FLOW_INCREMENTAL_AFTER_SEQS, FLOW_INCREMENTAL_MODE, mode
224                )));
225            }
226        }
227
228        Ok(Some(Self {
229            incremental_after_seqs,
230            incremental_mode,
231            return_region_seq,
232            sink_table_id,
233        }))
234    }
235
236    pub fn validate_for_scan(&self, source_region_id: RegionId) -> Result<bool> {
237        if self.sink_table_id.is_some() && self.sink_table_id == Some(source_region_id.table_id()) {
238            return Ok(false);
239        }
240
241        if matches!(
242            self.incremental_mode,
243            Some(FlowIncrementalMode::MemtableOnly | FlowIncrementalMode::SequenceRange)
244        ) {
245            let mode = if self.incremental_mode == Some(FlowIncrementalMode::MemtableOnly) {
246                FLOW_INCREMENTAL_MODE_MEMTABLE_ONLY
247            } else {
248                FLOW_INCREMENTAL_MODE_SEQUENCE_RANGE
249            };
250            let after_seqs = self.incremental_after_seqs.as_ref().ok_or_else(|| {
251                invalid_query_context_extension(format!(
252                    "{} is required when {}={}.",
253                    FLOW_INCREMENTAL_AFTER_SEQS, FLOW_INCREMENTAL_MODE, mode
254                ))
255            })?;
256
257            if !after_seqs.contains_key(&source_region_id.as_u64()) {
258                return Err(invalid_query_context_extension(format!(
259                    "Missing region {} in {} when {}={}.",
260                    source_region_id, FLOW_INCREMENTAL_AFTER_SEQS, FLOW_INCREMENTAL_MODE, mode
261                )));
262            }
263        }
264
265        Ok(self.incremental_after_seqs.is_some())
266    }
267
268    pub fn should_collect_region_watermark(&self) -> bool {
269        should_collect_region_watermark(
270            self.return_region_seq,
271            self.incremental_after_seqs.is_some(),
272        )
273    }
274}

276/// Returns whether query-level remote dynamic filter propagation is enabled.
277///
278/// The option defaults to enabled to preserve existing behavior. Callers may set
279/// `query.enable_remote_dynamic_filter_pushdown=false` in query context
280/// extensions to disable FE->DN remote dynamic filter propagation for a single
281/// query.
282pub fn remote_dyn_filter_pushdown_enabled_from_extensions(
283    extensions: &HashMap<String, String>,
284) -> Result<bool> {
285    extensions
286        .get(QUERY_ENABLE_REMOTE_DYNAMIC_FILTER_PUSHDOWN)
287        .map(|value| parse_bool(QUERY_ENABLE_REMOTE_DYNAMIC_FILTER_PUSHDOWN, value.as_str()))
288        .transpose()
289        .map(|value| value.unwrap_or(true))
290}

292/// Returns whether raw Flow query extensions request terminal region watermark collection.
293///
294/// This is only an intent/presence check for transport/scan plumbing; callers that need
295/// validated Flow options must still use [`FlowQueryExtensions::parse_flow_extensions`].
296pub fn should_collect_region_watermark_from_extensions(
297    extensions: &HashMap<String, String>,
298) -> bool {
299    let return_region_seq = extensions
300        .get(FLOW_RETURN_REGION_SEQ)
301        .is_some_and(|value| value.eq_ignore_ascii_case("true"));
302    let has_incremental_after_seqs = extensions.contains_key(FLOW_INCREMENTAL_AFTER_SEQS);
303
304    should_collect_region_watermark(return_region_seq, has_incremental_after_seqs)
305}
306
307fn should_collect_region_watermark(
308    return_region_seq: bool,
309    has_incremental_after_seqs: bool,
310) -> bool {
311    return_region_seq || has_incremental_after_seqs
312}

314fn parse_incremental_after_seqs(value: &str) -> Result<HashMap<u64, u64>> {
315    let raw = serde_json::from_str::<HashMap<String, serde_json::Value>>(value).map_err(|e| {
316        invalid_query_context_extension(format!(
317            "Invalid JSON for {}: {} ({})",
318            FLOW_INCREMENTAL_AFTER_SEQS, value, e
319        ))
320    })?;
321
322    raw.into_iter()
323        .map(|(region_id, raw_seq)| {
324            let region_id = region_id.parse::<u64>().map_err(|_| {
325                invalid_query_context_extension(format!(
326                    "Invalid region id in {}: {}",
327                    FLOW_INCREMENTAL_AFTER_SEQS, region_id
328                ))
329            })?;
330
331            let seq = match raw_seq {
332                serde_json::Value::Number(num) => num.as_u64().ok_or_else(|| {
333                    invalid_query_context_extension(format!(
334                        "Invalid sequence value in {} for region {}: {}",
335                        FLOW_INCREMENTAL_AFTER_SEQS, region_id, num
336                    ))
337                })?,
338                serde_json::Value::String(s) => s.parse::<u64>().map_err(|_| {
339                    invalid_query_context_extension(format!(
340                        "Invalid sequence string in {} for region {}: {}",
341                        FLOW_INCREMENTAL_AFTER_SEQS, region_id, s
342                    ))
343                })?,
344                _ => {
345                    return Err(invalid_query_context_extension(format!(
346                        "Invalid sequence value type in {} for region {}",
347                        FLOW_INCREMENTAL_AFTER_SEQS, region_id
348                    )));
349                }
350            };
351
352            Ok((region_id, seq))
353        })
354        .collect()
355}

357fn parse_bool(option_name: &str, value: &str) -> Result<bool> {
358    match value {
359        v if v.eq_ignore_ascii_case("true") => Ok(true),
360        v if v.eq_ignore_ascii_case("false") => Ok(false),
361        _ => Err(invalid_query_context_extension(format!(
362            "Invalid value for {}: {}",
363            option_name, value
364        ))),
365    }
366}

368/// Parse the scheduled time (in milliseconds since Unix epoch) from extensions.
369///
370/// Returns `Ok(None)` if the extension key is absent, `Ok(Some(millis))` on success,
371/// or `Err` if the value is malformed.
372pub fn parse_scheduled_time_millis(extensions: &HashMap<String, String>) -> Result<Option<i64>> {
373    match extensions.get(FLOW_SCHEDULED_TIME_MILLIS) {
374        Some(val) => val.parse::<i64>().map(Some).map_err(|_| {
375            invalid_query_context_extension(format!(
376                "Invalid value for {}: {}",
377                FLOW_SCHEDULED_TIME_MILLIS, val
378            ))
379        }),
380        None => Ok(None),
381    }
382}
383
384/// Parse the scheduled time from extensions into a [`DateTime<Utc>`].
385///
386/// Returns `Ok(None)` if the extension key is absent, `Ok(Some(datetime))` on success,
387/// or `Err` if the value is malformed.  Millis that produce an out-of-range timestamp are
388/// also rejected.
389pub fn parse_scheduled_time_datetime(
390    extensions: &HashMap<String, String>,
391) -> Result<Option<DateTime<Utc>>> {
392    match parse_scheduled_time_millis(extensions)? {
393        Some(millis) => DateTime::from_timestamp_millis(millis)
394            .map(Some)
395            .ok_or_else(|| {
396                invalid_query_context_extension(format!(
397                    "Out-of-range timestamp for {}: {} ms",
398                    FLOW_SCHEDULED_TIME_MILLIS, millis
399                ))
400            }),
401        None => Ok(None),
402    }
403}

405/// Best-effort helper: extract scheduled time from a [`QueryContextRef`] as a [`DateTime<Utc>`].
406///
407/// Errors are silently swallowed, returning `None`; use [`parse_scheduled_time_datetime`]
408/// at call sites that need to reject malformed scheduled time.
409pub fn scheduled_time_from_ctx(query_ctx: &QueryContextRef) -> Option<DateTime<Utc>> {
410    let extensions = query_ctx.extensions();
411    parse_scheduled_time_datetime(&extensions).ok().flatten()
412}

414/// Carries the scheduled logical "now" through [`ConfigOptions::extensions`] so
415/// that the distributed plan analyzer can apply it during expression
416/// simplification (preventing wall-clock constant-folding of `now()`).

420}
