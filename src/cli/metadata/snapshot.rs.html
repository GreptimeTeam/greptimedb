27/// Subcommand for metadata snapshot operations, including saving snapshots, restoring from snapshots, and viewing snapshot information.
28#[derive(Subcommand)]
29pub enum SnapshotCommand {
30    /// Save a snapshot of the current metadata state to a specified location.
31    Save(SaveCommand),
32    /// Restore metadata from a snapshot.
33    Restore(RestoreCommand),
34    /// Explore metadata from a snapshot.
35    Info(InfoCommand),
36}
37
38impl SnapshotCommand {
39    pub async fn build(&self) -> Result<Box<dyn Tool>, BoxedError> {
40        match self {
41            SnapshotCommand::Save(cmd) => Ok(Box::new(cmd.build().await?)),
42            SnapshotCommand::Restore(cmd) => Ok(Box::new(cmd.build().await?)),
43            SnapshotCommand::Info(cmd) => Ok(Box::new(cmd.build().await?)),
44        }
45    }
46}

48/// Export metadata snapshot tool.
49/// This tool is used to export metadata snapshot from etcd, pg or mysql.
50/// It will dump the metadata snapshot to local file or s3 bucket.
51/// The snapshot file will be in binary format.
52#[derive(Debug, Default, Parser)]
53pub struct SaveCommand {
54    /// The store configuration.
55    #[clap(flatten)]
56    store: StoreConfig,
57    /// The object store configuration.
58    #[clap(flatten)]
59    object_store: ObjectStoreConfig,
60    /// The path of the target snapshot file.
61    #[clap(
62        long,
63        default_value = "metadata_snapshot.metadata.fb",
64        alias = "file_name"
65    )]
66    file_path: String,
67    /// Specifies the root directory used for I/O operations.
68    #[clap(long, default_value = "/", alias = "output_dir")]
69    dir: String,
70}
71
72impl SaveCommand {
73    async fn build(&self) -> Result<MetaSnapshotTool, BoxedError> {
74        let kvbackend = self.store.build().await?;
75        let (object_store, file_path) = build_object_store_and_resolve_file_path(
76            self.object_store.clone(),
77            &self.dir,
78            &self.file_path,
79        )?;
80        let tool = MetaSnapshotTool {
81            inner: MetadataSnapshotManager::new(kvbackend, object_store),
82            file_path,
83        };
84        Ok(tool)
85    }
86}

161        }

165#[async_trait]
166impl Tool for MetaRestoreTool {
167    async fn do_work(&self) -> std::result::Result<(), BoxedError> {
168        let clean = self
169            .inner
170            .check_target_source_clean()
171            .await
172            .map_err(BoxedError::new)?;
173        if clean {
174            common_telemetry::info!(
175                "The target source is clean, we will restore the metadata snapshot."
176            );
177            self.inner
178                .restore(&self.file_path)
179                .await
180                .map_err(BoxedError::new)?;
181            Ok(())
182        } else if !self.force {
183            common_telemetry::warn!(
184                "The target source is not clean, if you want to restore the metadata snapshot forcefully, please use --force option."
185            );
186            Ok(())
187        } else {
188            common_telemetry::info!(
189                "The target source is not clean, We will restore the metadata snapshot with --force."
190            );
191            self.inner
192                .restore(&self.file_path)
193                .await
194                .map_err(BoxedError::new)?;
195            Ok(())
196        }
197    }
198}

269/// Builds the object store and resolves the file path.
270fn build_object_store_and_resolve_file_path(
271    object_store: ObjectStoreConfig,
272    fs_root: &str,
273    file_path: &str,
274) -> Result<(ObjectStore, String), BoxedError> {
275    if let Some(object_store) = object_store.build().map_err(BoxedError::new)? {
276        return Ok((object_store, file_path.to_string()));
277    }
278
279    let absolute_path =
280        resolve_relative_path_with_current_dir(file_path).map_err(BoxedError::new)?;
281    let (root, key) = split_fs_snapshot_path(fs_root, &absolute_path);
282    Ok((new_fs_object_store(&root)?, key))
283}

285/// Separates a local absolute path into an operator root and a relative key.
286fn split_fs_snapshot_path(fs_root: &str, absolute_path: &str) -> (String, String) {
287    let path = Path::new(absolute_path);
288    // OpenDAL keys cannot contain a Windows drive/UNC prefix or a root directory.
289    // Keep the filesystem anchor in the operator root, not in the object key.
290    let anchor: PathBuf = path
291        .components()
292        .take_while(|c| matches!(c, Component::Prefix(_) | Component::RootDir))
293        .collect();
294    let key = path
295        .components()
296        .filter(|c| !matches!(c, Component::Prefix(_) | Component::RootDir))
297        .map(|c| c.as_os_str().to_string_lossy())
298        .collect::<Vec<_>>()
299        .join("/");
300    // Preserve the existing custom-root semantics: cwd-resolved keys are rooted
301    // under --dir. The default root follows the input path's drive on Windows.
302    let root = if fs_root == "/" {
303        anchor.to_string_lossy()
304    } else {
305        fs_root.into()
306    };
307    (root.into_owned(), key)
308}

325    fn create_raftengine_url(path: &std::path::Path) -> String {
326        let mut path = path.to_string_lossy().replace('\\', "/");
327        if !path.starts_with('/') {
328            path = format!("/{}", path);
329        }
330        format!("raftengine://{}", path)
331    }
