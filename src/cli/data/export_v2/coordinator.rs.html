28/// Owned, manifest-independent context shared by all chunk export futures.
29///
30/// `catalog`/`schemas`/`format` are cloned from the manifest up front so the
31/// export futures never borrow the manifest, leaving the coordinator free to
32/// mutate and persist it while chunks are in flight.
33struct ExportContext<'a> {
34    storage: &'a dyn SnapshotStorage,
35    database_client: &'a DatabaseClient,
36    snapshot_uri: &'a str,
37    storage_config: &'a ObjectStoreConfig,
38    catalog: String,
39    schemas: Vec<String>,
40    format: DataFormat,
41    parallelism: usize,
42    experimental_metric_export: bool,
43    resume: bool,
44}
45
46pub struct ExportDataOptions<'a> {
47    pub snapshot_uri: &'a str,
48    pub storage_config: &'a ObjectStoreConfig,
49    pub parallelism: usize,
50    pub chunk_parallelism: usize,
51    pub experimental_metric_export: bool,
52    pub resume: bool,
53}

55pub async fn export_data(
56    storage: &dyn SnapshotStorage,
57    database_client: &DatabaseClient,
58    manifest: &mut Manifest,
59    options: ExportDataOptions<'_>,
60    progress: &dyn ProgressReporter,
61) -> Result<()> {
62    if manifest.chunks.is_empty() {
63        return Ok(());
64    }
65
66    let context = ExportContext {
67        storage,
68        database_client,
69        snapshot_uri: options.snapshot_uri,
70        storage_config: options.storage_config,
71        catalog: manifest.catalog.clone(),
72        schemas: manifest.schemas.clone(),
73        format: manifest.format,
74        parallelism: options.parallelism,
75        experimental_metric_export: options.experimental_metric_export,
76        resume: options.resume,
77    };
78
79    // One progress unit per chunk. Already completed/skipped chunks from a
80    // previous run count up front so the reported total matches the chunk plan
81    // regardless of resume position; each chunk finalized in this run (completed,
82    // skipped, or failed) then increments exactly once.
83    let progress_phase = ProgressPhase::start(
84        progress,
85        "Export data chunks",
86        Some(manifest.chunks.len() as u64),
87    );
88    let already_done = (manifest.completed_count() + manifest.skipped_count()) as u64;
89    if already_done > 0 {
90        progress.inc(already_done);
91    }
92
93    let result = if options.chunk_parallelism <= 1 {
94        export_data_serial(&context, storage, manifest, progress).await
95    } else {
96        export_data_concurrent(
97            storage,
98            manifest,
99            options.chunk_parallelism,
100            progress,
101            |id, range| export_chunk(&context, id, range),
102        )
103        .await
104    };
105
106    progress_phase.finish();
107    result
108}

110/// Exports chunks one at a time, preserving the original serial behavior.
111async fn export_data_serial(
112    context: &ExportContext<'_>,
113    storage: &dyn SnapshotStorage,
114    manifest: &mut Manifest,
115    progress: &dyn ProgressReporter,
116) -> Result<()> {
117    for idx in 0..manifest.chunks.len() {
118        if matches!(
119            manifest.chunks[idx].status,
120            ChunkStatus::Completed | ChunkStatus::Skipped
121        ) {
122            continue;
123        }
124
125        let (chunk_id, time_range) = mark_chunk_in_progress(manifest, idx);
126        manifest.touch();
127        storage.write_manifest(manifest).await?;
128
129        let export_result = export_chunk(context, chunk_id, time_range).await;
130
131        let result = match export_result {
132            Ok(files) => {
133                mark_chunk_completed(manifest, idx, files);
134                Ok(())
135            }
136            Err(err) => {
137                mark_chunk_failed(manifest, idx, err.to_string());
138                Err(err)
139            }
140        };
141
142        manifest.touch();
143        let persistence = storage.write_manifest(manifest).await;
144        if persistence.is_ok() {
145            progress.inc(1);
146        }
147        if let Err(err) = result {
148            if let Err(secondary) = persistence {
149                error!(secondary; "Failed to persist failed export chunk");
150            }
151            return Err(err);
152        }
153        persistence?;
154    }
155
156    Ok(())
157}

159/// Exports up to `chunk_parallelism` chunks concurrently on the client.
160///
161/// The coordinator owns all manifest mutation/persistence: it marks chunks
162/// `InProgress` and persists the manifest before polling their futures, then
163/// applies each chunk result and persists again on completion. The export
164/// futures only run COPY DATABASE and collect files; they never touch the
165/// manifest, so manifest writes stay serialized in this task.
166///
167/// On the first chunk failure we stop scheduling new chunks but let already
168/// in-flight chunks finish and persist their final status, then return the
169/// first error.
170async fn export_data_concurrent<F, Fut>(
171    storage: &dyn SnapshotStorage,
172    manifest: &mut Manifest,
173    chunk_parallelism: usize,
174    progress: &dyn ProgressReporter,
175    export: F,
176) -> Result<()>
177where
178    F: Fn(u32, TimeRange) -> Fut,
179    Fut: std::future::Future<Output = Result<Vec<String>>>,
180{
181    let mut pending = FuturesUnordered::new();
182    let mut next_idx = 0;
183    let mut first_error: Option<Error> = None;
184
185    loop {
186        let mut admitted = Vec::new();
187        // Schedule eligible chunks in order up to the parallelism limit. Once a
188        // failure is seen, stop scheduling but keep draining in-flight chunks.
189        while first_error.is_none() && pending.len() + admitted.len() < chunk_parallelism {
190            let Some(idx) = next_eligible_chunk(manifest, &mut next_idx) else {
191                break;
192            };
193
194            let (chunk_id, time_range) = mark_chunk_in_progress(manifest, idx);
195            admitted.push((idx, chunk_id, time_range));
196        }
197        if !admitted.is_empty() {
198            manifest.touch();
199            match storage.write_manifest(manifest).await {
200                Ok(()) => {
201                    for (idx, chunk_id, time_range) in admitted {
202                        let export = &export;
203                        pending.push(async move { (idx, export(chunk_id, time_range).await) });
204                    }
205                }
206                Err(err) => first_error = Some(err),
207            }
208        }
209
210        let Some((idx, export_result)) = pending.next().await else {
211            break;
212        };
213
214        match export_result {
215            Ok(files) => mark_chunk_completed(manifest, idx, files),
216            Err(err) => {
217                mark_chunk_failed(manifest, idx, err.to_string());
218                if first_error.is_none() {
219                    first_error = Some(err);
220                }
221            }
222        }
223        manifest.touch();
224        match storage.write_manifest(manifest).await {
225            Ok(()) => progress.inc(1),
226            Err(err) if first_error.is_none() => first_error = Some(err),
227            Err(err) => error!(err; "Failed to persist export chunk while draining"),
228        }
229    }
230
231    match first_error {
232        Some(err) => Err(err),
233        None => Ok(()),
234    }
235}

237/// Returns the index of the next chunk eligible for export, scanning forward
238/// from `next_idx` and skipping already Completed/Skipped chunks. Advances
239/// `next_idx` past the returned chunk so each chunk is scheduled at most once.
240fn next_eligible_chunk(manifest: &Manifest, next_idx: &mut usize) -> Option<usize> {
241    while *next_idx < manifest.chunks.len() {
242        let idx = *next_idx;
243        *next_idx += 1;
244        if !matches!(
245            manifest.chunks[idx].status,
246            ChunkStatus::Completed | ChunkStatus::Skipped
247        ) {
248            return Some(idx);
249        }
250    }
251    None
252}

254fn mark_chunk_in_progress(manifest: &mut Manifest, idx: usize) -> (u32, TimeRange) {
255    let chunk = &mut manifest.chunks[idx];
256    chunk.mark_in_progress();
257    (chunk.id, chunk.time_range.clone())
258}
259
260fn mark_chunk_completed(manifest: &mut Manifest, idx: usize, files: Vec<String>) {
261    let chunk = &mut manifest.chunks[idx];
262    if files.is_empty() {
263        chunk.mark_skipped();
264    } else {
265        chunk.mark_completed(files, None);
266    }
267}
268
269fn mark_chunk_failed(manifest: &mut Manifest, idx: usize, error: String) {
270    let chunk = &mut manifest.chunks[idx];
271    chunk.mark_failed(error);
272}

274async fn export_chunk(
275    context: &ExportContext<'_>,
276    chunk_id: u32,
277    time_range: TimeRange,
278) -> Result<Vec<String>> {
279    if context.experimental_metric_export {
280        context
281            .storage
282            .prepare_export_chunk(&context.schemas, chunk_id, context.resume)
283            .await?;
284    }
285    let scheme = StorageScheme::from_uri(context.snapshot_uri)?;
286    let needs_dir = matches!(scheme, StorageScheme::File);
287    let copy_options = CopyOptions {
288        format: context.format,
289        time_range,
290        parallelism: context.parallelism,
291        experimental_metric_export: context.experimental_metric_export,
292    };
293
294    for schema in &context.schemas {
295        let prefix = data_dir_for_schema_chunk(schema, chunk_id);
296        if needs_dir {
297            context.storage.create_dir_all(&prefix).await?;
298        }
299
300        let target = build_copy_target(
301            context.snapshot_uri,
302            context.storage_config,
303            schema,
304            chunk_id,
305        )?;
306        execute_copy_database(
307            context.database_client,
308            &context.catalog,
309            schema,
310            &target,
311            &copy_options,
312        )
313        .await?;
314    }
315
316    let files = list_chunk_files(context.storage, &context.schemas, chunk_id).await?;
317    info!("Collected {} files for chunk {}", files.len(), chunk_id);
318    Ok(files)
319}

321async fn list_chunk_files(
322    storage: &dyn SnapshotStorage,
323    schemas: &[String],
324    chunk_id: u32,
325) -> Result<Vec<String>> {
326    let mut files = Vec::new();
327
328    for schema in schemas {
329        let prefix = data_dir_for_schema_chunk(schema, chunk_id);
330        files.extend(storage.list_files_recursive(&prefix).await?);
331    }
332
333    files.sort();
334    Ok(files)
335}

412    #[tokio::test]
413    async fn test_manifest_failure_stops_admission_and_drains_started_chunks() {
414        use std::sync::atomic::{AtomicUsize, Ordering};
415        for fail_at in [1, 2, 3] {
416            let storage = FailingManifest {
417                fail_at,
418                writes: AtomicUsize::new(0),
419                release: tokio::sync::Notify::new(),
420            };
421            let started = AtomicUsize::new(0);
422            let finished = AtomicUsize::new(0);
423            let second_started = tokio::sync::Notify::new();
424            let mut manifest = pending_manifest(4);
425            let result = tokio::time::timeout(
426                std::time::Duration::from_secs(5),
427                export_data_concurrent(
428                    &storage,
429                    &mut manifest,
430                    2,
431                    &crate::data::progress::NoopProgress,
432                    |id, _| {
433                        let (started, finished, second_started, storage) =
434                            (&started, &finished, &second_started, &storage);
435                        async move {
436                            started.fetch_add(1, Ordering::SeqCst);
437                            match id {
438                                1 => second_started.notified().await,
439                                2 => {
440                                    second_started.notify_one();
441                                    storage.release.notified().await;
442                                }
443                                _ => panic!("new chunk started after manifest persistence failed"),
444                            }
445                            finished.fetch_add(1, Ordering::SeqCst);
446                            Ok(vec![format!("data/public/{id}/a.parquet")])
447                        }
448                    },
449                ),
450            )
451            .await
452            .unwrap();
453            assert!(
454                result
455                    .unwrap_err()
456                    .to_string()
457                    .contains("injected manifest failure")
458            );
459            let expected = if fail_at == 1 { 0 } else { 2 };
460            assert_eq!(started.load(Ordering::SeqCst), expected);
461            assert_eq!(finished.load(Ordering::SeqCst), expected);
462        }
463    }
